بيان ٻڌڻ جون نِيَّتون

عُثمانِ غني جو شان تسليم نه ڪرڻ واري تي انفرادي ڪوشِش

وحي جا انوار پنهنجن اکين سان ڏٺا۔ ٭ هي اهي عظيم هستيون آهن جن جي تربيت (Training) خود خُلقِ عظيم جي پيڪر، تمام نبين جي سَروَر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پاڻ فرمائي۔ ٭ هي اهي عظيم هستيون آهن جن آقا ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي معمولات کي پنهنجي اکين سان ڏٺائون۔ ٭ هي اهي عظيم هستيون آهن جن رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي معجزن کي ڏٺو، ٭ هي اهي عظيم هستيون آهن جن مال، دولت جان سميت هر شيء راھ خدا ۾ قربان ڪري ڇڏي۔ ٭ هي اهي عظيم هستيون آهن جن پنهنجي ٻچڙن جي رت جو نظرانو پيش ڪري اسلام جو شان بلند ڪيو۔ ٭ هي اهي عظيم انسان آهن جن تبليغِ دين جي لاءِ هر طرح جون مشقتون برداشت ڪيون۔ مطلب ته! ٭ هي اهي آهن جن جي قربانين جو ئي صلو آهي جو اڄ اسلام جو جهنڊو هر پاسي لهرائي رهيو آهي۔ تنهنڪري!! انهن جي تعظيم ۽ توقير هر مسلمان تي لازم آهي، انهن جو تذڪرو ادب ۽ محبت سان ڪرڻ هر مسلمان تي ضروري آهي۔ پاڻ رسول اڪرم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جن انهن جي باري ۾ وصيتون ۽ نصيحتون فرمايون۔ اچو! انهيء باري ۾ فرمانِ مصطفيٰ صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّم ٻڌون ٿا:

فرمايائون: اَللهَ اَللهَ فِي اَصْحَابِي منهنجي صحابين جي باري ۾ الله کان ڊڄو! خدا جو خوف ڪريو! لَا تَتَّخِذُوْھُمْ غَرَضًا بَعْدِي انهن کي مون کان پوءِ نشانو نه بڻائجو! فَمَنْ اَحَبَّھُمْ فَبِحُبِّي اَحَبَّھُمْ جنهن انهن کي محبوب رکيو منهنجي محبت جي ڪري محبوب رکيو۔ وَمَنْ اَبْغَضَھُمْ فَبِبُغْضِي اَبْغَضَھُمْ ۽ جنهن انهن سان بغض رکيو اهو مونسان بغض رکي ٿو، ان ڪري انهن سان بغض رکيائين۔ وَمَنْ اٰذَا ھُمْ فَقَدْ اٰذَانِي جنهن انهن کي اِيذاء ڏنو ان مون کي ايذاء ڏنو۔ وَمَنْ اٰذَانِي فَقَدْ اَذَي اللّٰه جنهن مون کي ايذاءُ ڏنو ان بيشڪ الله پاڪ کي ايذاء ڏنو۔ وَمَنْ اٰذَي اللّٰه يوْ شِڪ اَنْ يأْخُذَه جنهن الله پاڪ کي ايذايو قريب آهي ته الله پاڪ ان جي پڪڙ فرمائي۔ (ترمذي،ڪتاب المناقب، باب في من سب اصحاب النبي، ۵/۴۶۳، حديث:۳۸۸۸)

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي ڪم و بيش 107 کان وڌيڪ شُعبن ۾ دِينِ اسلام جو مدني پَيغام عام ڪري رهي آهي۔ انهن شعبن ۾ ’’وَڪالت‘‘ (Advocacy) سان وابسته ماڻهن ۾ نيڪي جي دعوت عام ڪرڻ جي لاءِ ”مَجلِسِ وڪلا“ به قائم آهي۔ جنهن جي ذَرِيعي وڪالت سان

وابسته ماڻهن کي دعوتِ اسلامي جي مَدَني ماحول سان وابسته ڪندي، هن مَدَني مَقصَد جي مُطابِق زِندگي گُذارڻ ۽ فڪرِ آخِرت جو مَدَني ذِهن ڏيڻ ۾ مصروفِ عمل آهي ته ”مون کي پنهنجي ۽ س ڄ ي دُنيا جي ماڻهن جي اِصلاح جي ڪوشش ڪرڻي آهي۔ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ “ ۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ هن مجلس تحت وڪالت جي شعبي سان وابسته عاشقانِ رسول جي گھرن، دفترن، بار ايسوسي ايشن وغيره ۾ وقت بوقت ”مَدَني حلقن“ جو سلسلو ٿيندو رهندو آهي، جڏهن ته هن شعبي جي عاشقانِ رسول ”هفتيوار اِجتماع“ ۽ ”مَدَني مذاڪري“ ۾ به شرڪت جي سعادت حاصل ڪندا آهن۔ ”مجلس مَدرَسة المدينه بالِغان“ جي ڪوشش سان هن شعبي سان وابسته ماڻهن کي ”تعليمِ قرآن“ جي روشني سان مُنَوَّر ڪرڻ جي لاءِ ”مَدرَسة المدينه بالِغان“ جي ترڪيب به ٿيندي آهي، جنهن ۾ فِي سَبِيلِ الله قرآنِ ڪريم جي تعليم ڏني ويندي آهي۔ الله ڪريم ”مجلسِ وڪلاء“ کي مزيد ترقيون نصيب فرمائي۔ اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله عَلٰي مُحَمَّد

پيارا پيارا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي؛ جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو.

(مشکاۃ الصابیح،کتاب الایمان،باب الاعتصام بالکتاب والسنۃ،الفصل الثانی،۱/۵۵،حدیث:۱۷۵)

مطالعو ڪرڻ جا مدني گل

پيارا پيارا اسلامي ڀائرو! اچو! مطالعو ڪرڻ جي باري ۾ ڪجھ مدني گل ٻڌڻ جي سعادت حاصل ڪريون ٿا۔ پهريان 2 فرمان مصطفيٰ صَلَّي اللهُ تَعَالٰي عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحظه ڪريو: (1) فرمايائون: بي شڪ علم سکڻ سان اچي ٿو۔ (ڪنز العمال،۱۰/۱۰۴، حديث: ۲۹۲۵۲) (2) فرمايائون: دنيا ملعون آهي ۽ الله پاڪ جو ذڪر، ان جا ولي يعني دوست ۽ ديني علم پڙھڻ ۽ پڙھائڻ واري کانسواءِ هر شيءِ به ملعون آهي۔ (تِرمِذي، ڪتاب الزهد،۴/۱۴۴،حديث:۲۳۲۹) ٭ مطالعو ايمان جي مضبوطي ۽ پختگي جو سبب آهي۔ (مطالعہ کیا،کیوں اور کیسے؟ص۱۶) ٭ مطالعي سان علم وڌي ٿو۔ (ايضاً ص۱۷) ٭ مطالعو معرفت حاصل ڪرڻ جو ذريعو آهي۔ (ايضاً،ص۱۸) ٭ اهڙن ڪتابن، رسالن ۽ اخبارن کان هميشه پري رهو جيڪي ايمان جي لاءِ زهر قاتل، حياء سوز ۽ اخلاق کان ڪريل هجن۔ (ايضاً،ص۲۹) ٭ پنهنجن اڪابرن جا حالات ڄاڻڻ ۽ انهن جي نيڪ ڪمن کي اپنائڻ جي لاءِ انهن جي احوال ۽ ڪوائف تي مشتمل ڪتابن جو مطالعو به ضروري آهي۔ (ايضاً،ص۳۳) ٭ امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جڏهن توهان ڪو علم حاصل ڪرڻ لڳو يا مطالعو ڪرڻ چاهيو ته بهتر آهي ته توهان جو علم ۽ مطالعو تزڪيهءِ نفس ۽ دل جي اصلاح جو باعث ٿئي۔ (ايضاً،ص۳۲) ٭ ٭ ڪوشش ڪريو ته ڪتاب هر وقت گڏ رهي جو جتي موقعو ملي ڪجھ نه ڪجھ مطالعو ڪيو وڃي ۽ ڪتاب جي صحبت به ميسر رهي۔ (ايضاً،ص۳۶) ٭ مطالعو ڪرڻ کان بعد تمام مطالعي کي شروع کان آخر تائين سرسري نظر سان ڏٺو وڃي ۽ ان جو هڪ خلاصو پنهنجي ذهن ۾ محفوظ ڪيو وڃي۔ (ايضاً،ص 112) ٭ پنهنجو اِحتِساب ڪرڻ به فائديمند آهي ته مون کي هن مطالعي مان ڇا حاصل ٿيو ۽ ڪهڙو مواد ذهن نشين ٿيو ۽ ڪهڙو نه۔ (ايضاً،ص 112) ٭ ڪنهن ڳالھ کي ياد ڪرڻ جي لاءِ اکيون بند ڪري يادداشت تي زور ڏيڻ مفيد آهي۔ (ايضاً، ص 117) ٭ جيڪو مطالعو ڪريو ان کي چڱين نيتن سان پنهنجي گھر وارن ۽ دوستن کي ٻڌايو اهڙي نموني به معلومات جو ذخيرو گھڻي عرصي تائين اسان جي دماغ ۾ محفوظ رهندو۔ (ايضاً، ص112) ٭ جيڪو ڪجھ پڙھون ان جي

دهرائي (Repeat) ڪندا رهو۔ (ايضاً،ص 112) ٭ شيخِ طريقت اميرِ اهلسنّت دَامَتْ بَرَکاتُهمُ الْعَالِيه مدَني مذاڪرن ۾ علمِ دين جا رنگ برنگي مدني گل ارشاد فرمائن ٿا هنن مدني مذاڪرن جي برڪت سان پنهنجي علم تي عمل ڪرڻ، ان کي ٻين تائين پهچائڻ ۽ مزيد مطالعو ڪرڻ جو جذبو پيدا ٿيندو آهي۔ (ايضاً،ص 115)

ايندڙ هفتيواراجتماع جي بيان جون جھلڪيون

پيارا پيارا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! آئنده هفتيوار سنتن ڀري اِجتماع جو موضوع آهي: ’’دل جوئي جون فضيلتون‘‘ جنهن ۾ دل جُوئي جي فضيلت تي حديث سڳوريون ۽ بزرگن جا اقوال پيش ڪيا ويندا۔ اسان جا بزرگ سڳورا دل جوئي جا ڪيترا عادي هئا ان جا به ڪجھ واقعا پيش ڪيا ويندا۔ ڪن ڪن معاملن سان اسان مسلمانن جي دل جوئي ڪري سگھون ٿا اهو به بيان ڪيو ويندو۔ وڌيڪ ڪيئي مدني گل به پيش ڪيا ويندا۔ لهٰذا! آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ اِنفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان چئو: اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([1])

﴿5﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهڙي مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([2])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([3])

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)
