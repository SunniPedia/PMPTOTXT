(3) اُن جو تَعلُّق قرض ۽ ظُلم ٻنهي سان آهي، جيئن ڪنهن جو مال چورايو، ڦُريو، لوٽيو، ڪنهن کان رِشوت ورتي، وياج ورتو يا جوا ۾ مال کٽيو وغيره۔ هن صورت ۾ توبه جو طريقو هي آهي ته اهي حق ادا به ڪري ۽ صاحب ِحق کان معافي به حاصل ڪري۔ جيڪڏهن توبه جا شرط موجود هئا ته توبه ضرور قبول ٿيندي، ڇوته اهو رب تعاليٰ جو وعدو آهي ۽ پنهنجي وعدي جي خلاف ڪرڻ، الله تعاليٰ جي شانِ ڪريمي جي لائق ناهي۔ (صراط الجنان،ج۴،ص۲۳۰)

مٺا مٺا اسلامي ڀائرو! الله تبارڪ وتعاليٰ جون رحمتون بيحد ۽ بي حساب آهن، ان ڪري سان کي گھرجي ته جڏهن به گُناھ ٿين ته ان جي بارگاھ ۾ هڪدم توبه ڪيون۔ جيڪڏهن بَتَقاضاءِ بَشَرِيت وري گُناھ ڪيون ته وري توبه ڪيون، وري خَطا ٿي وڃي ته وري توبه ڪيون، چاهي لک ڀيرا ڀٽڪي وڃون، پر وري اچي ان جي دامنِ ڪرم سان چنبڙي وڃون ۽ هرگز هرگز مايوس نه ٿيون، ۽ يقيناً جيڪي الله عَزَّوَجَلَّ جي بارگاھ ۾ سچي توبه ڪندا آهن الله عَزَّوَجَلَّ انهن کان گھڻو خُوش ٿيندو آهي ۽ انهن تي پنهنجو خاص فَضل وڪرم به ڪندو آهي۔ قرآنِ ڪريم ۾ مختلف هنڌن تي رَبِّ ڪريم عَزَّوَجَلَّ پنهنجي گُناھ گار ٻانهن کي توبه جي ترغيب ڏياري رهيو آهي، جيئن سيپاري 6 سُوْرَة الْمائِده جي آيت نمبر 39 ۾ ارشاد آهي:

فَمَنْ تَابَ مِنْ بَعْدِ ظُلْمِ ه ِ وَأَصْلَحَ فَإِنَّ اللَّ ه َ ي َتُوبُ عَلَ ي ْ ه ِ إِنَّ اللَّ ه َ غَفُورٌ رَحِ ي مٌ

ترجمو ڪنزالايمان: سو جيڪو پنهنجي ظلم کانپوءِ توبه ڪري ۽ پاڻ سُڌرِي وڃي ته الله پنهنجِي مهر (مهرباني) سان رجُوع فرمائِيندو. بيشڪ الله بخشيندڙ مهربان آهي.

دعوتِ اسلامي جي اشاعتي اِداري مڪتبة المدينه جي شايع ٿيل تفسير’’صِراطُ الجِنان‘‘ ۾ هن آيت سڳوري جي تحت لکيل آهي ته توبه نِهايت نَفيس شيء آهي، ڪيترو ئي وڏو گُناھ هجي، جيڪڏهن ان کان توبه ڪئي وڃي ته الله عَزَّوَجَلَّ پنهنجو حق مُعاف فرمائيندو آهي ۽ توبه ڪرڻ واري کي آخرت جي عذاب مان نَجات ڏئي ڇڏيندو آهي۔ (تفسير ِصراط الجنان،ج ۲ ،ص ۴۳۰)

پئے تاجدارِ حَرَم یاالٰہی

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو توبه ڪرڻ نهايت ئي بهترين ۽ پسنديده ڪم آهي، اهڙي طرح هڪ ٻي هنڌ تي الله عَزَّوَجَلَّ قرآنِ ڪريم ۾ توبه جي ترغيب ڏياري ۽ هن جي فضيلت بيان ڪئي آهي، جيئن اِرشاد ٿئي ٿو:

ي َا أَ ي ُّ ه َا الَّذِ ي نَ آمَنُوا تُوبُوا إِلَى اللَّ ه ِ تَوْبَ ة ً نَصُوحًا (پ۲۸،التحريم:۸)

ترجمو ڪنزالايمان: ڇا اي ايمان وارؤ! الله جي طرف اهڙي توبه ڪيو جو اڳتي جي لاءِ نصيحت ٿي وڃي۔

هن آيتِ مُبارَڪه ۾بيان ٿيل لفظ ” نَصُوۡح “ جي معنيٰ الله عَزَّوَجَلَّ جي لاءِ اهڙو خالص ٿيڻ آهي جو ڪنهن قسم جي مِلاوَٽ نه هجي۔ ٻيو خليفو، اَمِيرُالْمُومِنِين فارُوقِ اعظم حضرت سَيدُنا عُمر بن خطاب رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان جڏهن ’ تَوْبَةً نَصُوْحا ‘ جي باري ۾ پڇيو ويو ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ اِرشاد فرمايو: هن مان مُراد هي آهي ته ڪو شخص پنهنجي ڪنهن بُري عمل کان اهڙي توبه ڪري جو وري ڪڏهن به ان گُناھ ۾ نه پوي۔ (تفسيردر منثور،ج۸،ص۲۲۷)

اهڙي طر ح حضرت اِبۡنِ عباس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته

حضرت سَيدُنا مَعاذ بن جبل رَضِىَ اللهُ تَعَالٰی عَـنْهُ عرض ڪيو؛ يارَسُوْلَ الله ” تَوْبَةً نَصُوْحا “ ڇا آهي؟ ته پاڻ ڪريمن صَلَّ ی الله تَعَالٰی عَلَيه وَاٰلِهٖ وَسَلَّم فرمايو ته هن مان مُراد هي آهي ته ٻانهو پنهنجن ڪيل گُناهن تي شرمسار ٿئي، پوءِ الله عَزَّوَجَلَّ کان مُعافي گھري ۽ ان گُناھ جي طرف ڪڏهن به نه موٽي جيئن کير پنهنجي ٿڻ ۾ ٻيهر نه موٽبو آهي۔ ۽ حضرت سَيدُنا اِبنِ مَسعُود رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته’’ تَوْبَةً نَصُوْحا ‘‘ڪرڻ سان تمام گُناھ مُعاف ٿي ويندا آهن"۔ (تفسيردر منثور،ج۸،ص۲۲۷)

(وسائلِ بخشش مُرمّم،ص78)

مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان ته سچّي توبه ڪرڻ جون ڪيتري قدر برڪتون آهن ته توبه ڪرڻ وارن جا سڀئي گُناھ مُعاف ڪيا ويندا آهن۔ تنهنڪري اسان کي به گھرجي ته اسان به الله عَزَّوَجَلَّ جي بارگاھ ۾ ٻاڏائي ڪري سچي توبه ڪيون ۽ جيڪڏهن وري گُناھ ٿي وڃن ته ٻيهر توبه ڪيون ته الله عَزَّوَجَلَّ قرآنِ ڪريم ۾ اهڙن ماڻهن جي تعريف بَيان ڪئي آهي جيئن سيپاري 4 سُوْرَة النِّساۤء آيت نمبر 17 ۾ اِرشاد آهي:

إِنَّمَا التَّوْبَ ة ُ عَلَى اللَّ ه ِ لِلَّذِ ي نَ ي َعْمَلُونَ السُّوءَ بِجَ ه َالَ ة ٍ ثُمَّ ي َتُوبُونَ مِنْ قَرِ ي بٍ فَأُولَئِكَ ي َتُوبُ اللَّ ه ُ عَلَ ي ْ ه ِمْ وَكَانَ اللَّ ه ُ عَلِ ي مًا حَكِ ي مًا

ترجمو ڪنزالايمان: اُها توبھ جنهن جو قُبول ڪرڻ الله پنهنجي فضل جي ذمي لازم ڪيو آهي سا اُنهن جِي آهي جيڪِي نادانِي ۾ ڪا بُرائِي ڪري ويٺا پوءِ ٿورِي دير ۾ توبھ ڪري وٺن، اِهڙن تي الله پنهنجِي رحمت سان رُجوع ڪري ٿو، ۽ الله عِلم و حِڪمت وارو آهي.

مٺا مٺا اسلامي ڀائرو! الله تعاليٰ جي (ٻانهي تي اها) عظيم رحمت آهي جو گُناھ جي بعد توبه ڪرڻ تي مُعاف فرمائي ڇڏيندو آهي ۽ موت جي وَقت تائين توبه قبول ڪندو آهي۔ هتي فرمايو ويو ته جيڪو گُناھ ڪري ٿوري دير ۾ توبه ڪن ته هتي ٿوري دير مان مُراد هڪ ادھ ڪلاڪ يا ٻه (2) چار (4) سال نه آهن بلڪ موت کان پهريان جڏهن به توبه ڪئي اها قريب ئي شمار ٿيندي۔ ها جڏهن موت جو وقت اوچتو اچي وڃي ۽ غيب جو مُعاملو ظاهر ٿي وڃي ته ان وَقت توبه مقبول نه ٿيندي۔ ۽ پڻ اِسلام ۾ توبه جو قانون بنائڻ عين حڪمت ۽ علم تي مشتمل آهي۔ جن دِينَن ۾ توبه ناهي، ان جي مڃڻ وارا گُناھ تي وڌيڪ دلير هوندا آهن ڇوته مايوسي جُرم تي دلير ڪري ڇڏيندي آهي ۽ مُعافي جي اُميد توبه تي اُڀاريندي آهي۔ (صراط الجنان،ج۲،ص۱۶۳)

مٺا مٺا اسلامي ڀائرو! اسان جي پياري آقا، حبيبِ ڪبريا، محمدِ مُصطفيٰ صَلَّ ی الله تَعَالٰی عَلَيه وَاٰلِهٖ وَسَلَّم به توبه جون گھڻيون فضيلتون ۽ هن تي ملڻ وارا اِنعام بيان فرمايا آهن۔ جيئن پاڻ سڳورن صَلَّ ی الله تَعَالٰی عَلَيه وَاٰلِهٖ وَسَلَّم جو فرمانِ عاليشان آهي:'' جڏهن ٻانهو گُناھ ڪندو آهي، وري الله عَزَّوَجَلَّ جي بارگاھ ۾ توبه ڪندو آهي ۽ ان تي قائم رهندو آهي ته الله عَزَّوَجَلَّ ان جو هر نيڪ عمل قَبول ڪندو آهي ۽ ان کان ٿيڻ وارو هر گُناھ بخشي ڇڏيندو آهي ۽ (مُعاف ٿيڻ واري) هر گُناھ جي بدلي جنَّت ۾ ان جو هڪ دَرَجو بُلند ڪندو آهي ۽ الله عَزَّوَجَلَّ ان جي هر نيڪي جي بدلي ۾ ان کي جنَّت ۾ هڪ محل عطا ڪندو آهي ۽ حُورن مان هڪ حُور سان ان جو نِڪاح ڪندو آهي۔'

(بحر الدموع ،ص ۲۱)

کي پسند ڪندو آهي ته ان جو اعمال نامو ان کي خُوش ڪري، ته ان کي گھرجي ته ان ۾ اِستِغفار جو اِضافو ڪري۔'' (مجمع الزوائد ،ڪتاب ، التوبة ، باب الاڪثار من الا ستغفار ،رقم ۱۷۵۷۹، ج۱۰، ص ۳۴۷) هڪ ٻي حديث شريف ۾ آهي؛ اَلتَّائِبُ مِنَ الذَّنْبِ کَمَنْ لَّا ذَنْبَ لَه يعني گُناھ کان توبه ڪرڻ وارو ائين آهي جيئن ان گناھ ڪيو ئي نه۔ (سنن ابن ماجه، ابواب الزھد، باب ذڪر التوبة، الحديث ۴۲۵۰،ج۴، ص۴۹۱)

۽ هڪ حديث شريف ۾ آهي ته جڏهن اِبليس چيو؛ ''اي الله عَزَّوَجَلَّ!مون کي تنهنجي عزّت جو قسم! مان تنهنجي ٻانهن کي ان وقت تائين ڀٽڪائيندو رهندس جيستائين انهن جا رُوح انهن جي جسم ۾ رهندا۔'' ته الله عَزَّوَجَلَّ فرمايو؛ ''مون کي پنهنجي عزّت وجَلال جو قسم! جيستائين اهي مون کان بخشش گھرندا رهندا مان انهن کي بخشيندو رهندس۔''
