شرم ۽ حيا جي پيڪر با پردھ صحابيه جو مديني جو سفر

مسلمانن جي حَليف (مددگار) قبيلي بني خُزاعه جو هڪ شخص پنهنجي اُٺ (Camel) تي سُوار ٿي ڪيڏانهن وڃي رهيو هو ته هو مَڪي پاڪ کان ٿورو پري شَرم ۽ حَيا جي پيڪر هڪ باپرده خاتون کي وِيراني ۾ پيدل سفر ڪندي ڏسي حَيران ٿي ويو۔ ان کي شڪ ٿيو ته اها پاڪيزه خاتون اُنهن مسلمانن مان هوندي جن کي مڪي وارن مديني وڃڻ کان روڪيو آهي۔ ان عورت کان پڇڻ تي جڏهن ان شخص کي يقين ٿي ويو ته ان جي دِل اهو پسند نه ڪيو ته هو نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي ديواني کي اڪيلو سفر ڪرڻ ڏي۔ ته پوءِ ان

پنهنجو سفر روڪي پنهنجو اُٺ ان باپرده عورت کي پيش ڪري مديني شريف

پهچائڻ جو پڪو اِرادو ڪيو ته شَرم ۽ حَيا جي پيڪر ان باپرده عورت به ان کي غيبي اِمداد سمجھي قَبول ڪري ورتائين۔ ائين هڪ اهڙو خاموش سفر شروع ٿيو جنهن تاريخ جي ورقن ۾ اهي سنهري يادون ڇڏيون جنهن تي اڄ به رَشڪ ڪري سگھجي ٿو۔ عشقِ سرڪار ۾ بيقرار اڪيلي ئي مديني جو سفر ڪرڻ واري ان باپرده عورت جي همَّت جي به ڇا ڳالھ ڪجي! اها پاڻ پنهنجي ان سفر جو احوال ڪجھ هيئن بيان ڪن ٿيون ته سفر دوران ان شخص سڄي واٽ ۾ مون سان ڪا به گفتگو نه ڪيائين، بلڪ جڏهن آرام جو وَقت ٿيندو هو ته هو اُٺ کي ويهاري پري هٽي ويندو هو ۽ مان ڪجاوي مان نڪري ڪنهن ڇانو واري وڻ جي هيٺ ويندي هيس، پوءِ هو اُٺ کي مون کان پري ڪنهن ٻي وڻ جي هيٺ ٻڌي ڇڏندو هو ۽ پاڻ به اتي ڪٿي آرام ڪندو هو ۽ جڏهن ٻيهر سفر جو وَقت ٿيندو هو ته اُٺ تي ڪجاوو رکي ان کي مون وٽ ڇڏي وري پري هٽي ويندو هو، منهنجي سُوار ٿيڻ کان بعد خاموشي سان اچي مهار پڪڙي مديني ڏانهن هلڻ لڳندو هو۔ ائين سفر دَوران به مان نِقاب ۾ رهيَس ۽ پردي جي اِهتِمام ۾ مون کي ڪا دُشواري پيش نه آئي۔ الله پاڪ ان شخص کي چڱي جزا عطا فرمائي جو ان نهايت بهترين ڪردار جو مُظاهرو ڪيو۔ جڏهن مان مديني پهتيس ته سڀ کان پهريان اُمُّ المؤمنين حضرت سَيدَتُنا اُمِّ سَلَمَه رَضِىَ اللّٰهُ تَعَالٰى عَنْها جي مبارڪ گھر ۾ حاضِر ٿيس۔ مان ان وَقت به نِقاب ۾ هئس جنهن سبب سيده اُمِّ سَلَمَه رَضِىَ اللّٰهُ تَعَالٰى عَنْها مون کي نه سڃاتو، ته پوءِ مون چهري تان نِقاب کڻي انهن کي پنهنجو تعارُف ڪرايو ۽ جڏهن ٻڌايو ته مون اڪيلي هجرت ڪئي آهي ته اهي حَيران ٿي ويون ۽ حَيران ٿي پڇڻ لڳيون ته ڇا واقعي! تو اڪيلي الله پاڪ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي رِضا جي خاطِر هجرت ڪئي آهي؟ مون اِقرار ڪيو، اڃا اسان ڳالهيون ئي

پياريون پياريون اسلامي ڀينرون! بيان ڪيل حڪايت ۾ جنهن باپرده عورت جو ذِڪرِ خَير ٿيو اها شَرم ۽ حيا جي پيڪر ۽ مشهور صحابِيه حضرت سَيدَتُنا اُمِّ ڪلثوم بنتِ عُقبَه رَضِىَ اللّٰهُ تَعَالٰى عَنْها هيون۔ پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْها مَڪي شريف ۾ مسلمان ٿيون ۽ غُربَت جي ڪري سُواري جو اِنتظام نه ٿي سگھيو ان ڪري پيادل ئي هجرت ڪيائون ، مَدِيني پاڪ پهچي نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان بَيعَت ٿيون ۽ مديني ۾ ان سان حضرت سَيدُنا زَيد بن حارِثه رَضِىَ اللّٰهُ تَعَالٰى عَنْه نڪاح ڪيو۔ (الاستيعاب، ڪتاب ڪني النساء،باب الڪاف: ۳۶۳۷،ام ڪلثوم بنت عقبۃ،۴/ ۵۰۸ملخصا) حضرت سَيدَتُنا اُمِّ ڪلثوم رَضِىَ اللّٰهُ تَعَالٰى عَنْها ايتري قَدَر ڊگھي سفر ۾ جهڙي طرح شَرم ۽ حَيا جي لڄ رکي ۽ پردي جو اِهتِمام ڪيو يقيناً اهو پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْها جو خاصو هو، ان جو اِنعام (Reward) بارگاھ نُبُوّت مان هي مليو ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اُن جي هجرت ۽ پردي جو اِهتمام ڪرڻ تي خوشي جو اِظهار ڪيو۔

ڇا عورَت جو اڪيلو سفر ڪرڻ جائز آهي؟

هتي اهو مَسئَلو به ذهن نشين رهي ته حضرت سَيدَتُنا اُمِّ ڪلثوم رَضِىَ اللّٰهُ تَعَالٰى عَنْها مَڪي شريف کان مديني پاڪ ڏانهن هجرت جو جيڪو سَفَر اڪيلي ۽ بغير مَحرَم جي ڪيو هو، اهو شَريعَت جي مُطابِق هو، جيئن ته حڪيمُ الاُمّت حضرت مفتي احمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: (عَورَت جي اڪيلي سَفَر

ڪرڻ جي) مُمانَعَت (روڪڻ) جي حڪم کان مُهاجِره (هجرت ڪرڻ واري) ۽

ڪافر جي قيد مان نڪرڻ واري عورَت خارِج آهي جو اهي ٻئي عورَتون بغير مَحرَم اڪيليون ئي دارُالسَّلام ڏانهن سَفَر ڪري سگھن ٿيون بلڪ اهو سَفَر انهن تي واجِب آهي۔ (مرآۃ المناجيح، ۴/۹۰) تنهنڪري هن واقعي مان ڪو اهو نه سمجھي ته اڪيلي عورت جو ايتري قَدر پري جو سَفَر ڪرڻ جائز آهي، ڇوته بغير مڙس يا محرم جي عورت جو ٽن ڏينهن جي فاصلي (سفر) واري ڪنهن جڳهه وڃڻ حرام آهي۔ ايتري تائين جو جيڪڏهن عورت وٽ حج جي سفر جو خرچ آهي پر مڙس يا ڪو قابل اطمينان محرم گڏ ناهي ته حج ڪرڻ جي لاءِ به نه ٿي وڃي سگهي، جيڪڏهن وئي ته گنهگار ٿيندي، باوجود ان جي جو فرض حج ادا ٿي ويندو۔ (پردے کے بارے میں سوال جواب ، ص ۱۶۳) جيئن ته بهارِ شَريعَت ۾ آهي ته عورت کي بنا محرم جي ٽن ڏينهن يا وڌيڪ جي راھ وڃڻ (يعني سفر ڪرڻ) ناجائز آهي بلڪ هڪ ڏينهن جي سفر تي وڃڻ به۔ نابالغ ٻار يا مَعتُوه (گھٽ عقل) سان گڏ به سفر نٿي ڪري سگهي، سفر ڪرڻ دوران بالغ محرم يا مڙس جو هئڻ ضروري آهي۔ محرم جي لاءِ ضروري آهي ته سخت فاسق بيباڪ غير محفوظ نه هجي۔ (بهارِ شريعت،(حصه چهارم)۱/ ۷۵۲)

سُبْحٰنَ الله عَزَّ وَجَلَّ! ڪهڙو پاڪيزه دور هو جو مرد هجن يا عورتون سڀئي شرعي پردي جي مڪمل رِعايت ۽ نگاهن جي حفاظت جي مَدَني جذبي سان سَرشار ۽ هڪ ٻي جي عزّت ۽ ناموس جا رکوالا هئا، پر افسوس! جيئن جيئن اسان زمانه رِسالت کان پري ٿينديون پيون وڃون، بي عِلمي ۽ بي شرمي جا نشان ويتر وڌيڪ پخته ٿيندا پيا وڃن۔ ٻين بُرائين سان گڏوگڏ بي پردگي، بي حيائي ۽ بدنگاهي جي تباھ ڪارين اسان جي معاشَري کي تباهي ۽ بربادي جي ڪناري تي کڻي آندو آهي، اَجنبي عورَتن ۽ مَردن ۾ پردو ختم ڪرڻ ۽ انهن

سيپارو 8 سُورَه اَعرَاف جي آيت نمبر33 ۾ ربِّ ڪريم اِرشاد فرمائي ٿو:

پياريون پياريون اسلامي ڀينرون! ٿورو غور ڪريو ته قرآن اسان کي بي حَيائي کان روڪي رهيو آهي، ته جڏهن اسان الله ڪريم ۽ ان جي پياري حبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تي اِيمان آندو آهي ته عقلمندي جي اها تقاضا آهي ته اسان انهن جي فرمانن تي به عمل ڪريون! پر اسان کي ڇا ٿي ويو آهي جو اسان ان احڪامن تي عمل ڪرڻ لاءِ تيار ئي نظر نه پيا اچون؟ پير آهن ته بي حيائي ڏانهن اُٿندا پيا وڃن، هٿ آهن جو بي حيائي ڏانهن وڌندا پيا وڃن، اکيون آهن جو بي حيائي ڏانهن اٿنديون پيون وڃن، سمجھ ۾ نه پيو اچي ته ان بگڙيل مُعاشَري جو رُخ الله ڪريم ۽ ان جي محبوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي فرمانبرداري ڏانهن ڪيئن ڦهريو وڃي۔ هاء! اهڙو دَور اچي چڪو آهي ڄڻ هر ڪو ٻي کان وڌيڪ بي پردَگي ۽ بي حيائي جو مظاهرو ڪرڻ ۾ مشغول آهي۔ ذاتي مُعامَلا هجن يا اِجتمِاعي تقريبون، پاڙو هجي يا بازار هر هنڌ شَرم ۽ حيا جو قَتلِ عام ۽ بي حيائي جي ڌُوم ڌام آهي۔ جنهن کي ڏسو وڌ کان وڌ بي حيائي کي پسند ڪندڙ نظر پيو اچي۔ ٿيندڙ شادين کي ئي ڏسو۔ جيڪي اِبتِدا کان اِنتِها تائين فُضول خَرچي، بدنگاهي، مَخلوط نظام، بي حيائي، ناچ گانن،

بے پردَگی کا خاتِمہ ہو عورَتوں کو دے زیور حیا و شرم کا یاربِّ مُصْطَفٰے

سُبْحٰنَ الله عَزَّ وَجَلَّ! ٻڌو اوهان ته پهريان جي عورتن ۾ اسلامي تعليمات تي عمل ۽ شرعي پردو ڪرڻ جو مَدَني جَذبو ڪيتري قَدَر گھڻو ڀريل هو جو انسان ته انسان درندن جي اڳيان به بي پرده اچڻ کان بچنديون هيون، قربان ان عظيم ماءُ تي جنهن پٽ جي شهادت جي خبر ٻڌي ڪري به روڄ راڙو ڪرڻ، گريبان ڦاڙڻ ۽ شرم ۽ حيا جي چادر لاهڻ جي بدران صَبر ۽ بُردباري جو مظاهرو ڪيو ۽ پردي جي اَهمِّيت کان هڪ گھڙي به غافِل نه ٿيون جڏهن ته حضرت سَيدُنا بابا نِظامُ الدّين رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي والِده سڳوري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ا به سڄي زِندگي نامَحرمن کان پردو ڪيو ايستائين جو دنيا مان ظاهري وصال ڪيو۔ ڇا هاڻ به اسان جون اسلامي ڀينرون انهن جي پاڪيزه سيرت کي نه اپنائيديون؟، ڇا گهٽين ۽ بازارن ۾ بي پرده گھمڻ ڦرڻ جو سِلسِلو اڃا به جاري رهندو؟، ڇا اڃا به نامَحرم رشتيدارن کان بي تَڪلُّفِي ۽ بي پردَگي جي گناھ کان تَوبَه نه ڪئي ويندي؟، ڇا اڃا به نامَحرَمن کان ڳالهيون ٿينديون رهنديون؟، ڇا اڃا به خوشي جي نالي تي شادي وهانء وغيره جي موقعي تي الله پاڪ جون نافرمانيون ٿينديون رهنديون؟ ڇا اڃا به اسان پنهنجي گھر وارن کي بي پردگي جي آفت ۾ مُبتَلا ڏسي خاموشي اِختيار ڪندا رهنداسين۔ ياد رکو! جيڪي ماڻهو قدرت باوُجُود پنهنجي عورَتن ۽ مَحرِمن کي بي پردَگي کان مَنع نٿا ڪن اهي دَيُّوث آهن ۽ دَيُّوث تي جنّت جو داخِلو حرام آهي۔ (مسند امام احمد،۲/۳۵۱ ،حديث:۵۳۷۲ ماخوذاً) تنهنڪري اسان کي گھرجي ته اسان پاڻ به بي پردگي جي آفت کان پري رهون، پنهنجي اکين ۽ ٻين عضوئن جو قفلِ مدينه لڳايون ۽ پنهنجي گھر وارن کي به بي پردگي کان بچڻ جي ترغيب ڏياريندا رهون۔

8 مدني ڪمن مان هڪ مدني ڪم ”مدني انعامات“

پياريون پياريون اسلامي ڀينرون! بي پردَگي جي آفت کان نجات ماڻڻ، اکين ۽ ٻين عضوئن جو قُفلِ مدينه لڳائڻ جي مَدَني سوچ ماڻڻ، شَرم ۽ حيا جو پيڪر بڻجڻ ۽ پنهنجي گھر وارن کي به پردي جو ذِهن ڏيڻ جو طريقو سکڻ جي لاءِ عاشِقانِ رَسول جي مَدَني تحريڪ دعوتِ اسلامي جي مَدَني ماحول سان وابسته رهو ۽ ذيلي حلقي جي 8 مدني ڪمن ۾ وڌي ڪري حصو وٺو.ذيلي حلقي جي 8 مدني ڪمن مان هڪ مدني ڪم ”مدني انعامات تي عمل“به آهي اميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جا عطا ڪيل 63 مدني انعامات نيڪ بڻجڻ ۽ پنهنجو محاسبو ڪرڻ جو بهترين نسخو آهي ۽ پنهنجو محاسبو ڪرڻ جي حوالي سان حضرت سيدنا عمر فاروق رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمايو پنهنجي نفس جو محاسبو ڪريو ان کا پهريان جو توهان جو حساب ورتو وڃي ۽ وزن ڪيو وڃڻ کان پهريان پنهنجي عملن جو خود وزن ڪريو ۽ تمام وڏي پيشي جي لاءِ تيار ٿي وڃو (فيضان احياء العلوم،ص 79) ته پوءِ وقت مقرر ڪري روزانو فڪر مدينه ڪريو (يعني مدني انعامات جي مطابق اڄ ڪيستائين عمل ٿيو) رسالي ۾ ڏنل خانا ڀري هر مدني مهيني جي پهرين تاريخ تي پنهنجي ذميدار اسلامي ڀيڻ کي جمع ڪرايو وڌيڪ مڪتبة المدينه تا شايع ٿيل ڪتاب ”جنت جي طلبگارن جي لاءِ مدني گلدستو“جي ذريعي ٻين اسلامي ڀينرن کي به مدني انعامات تي عمل ڪرڻ جي ترغيب ڏياريو. هر اسلامي ڀيڻ انفرادي ڪوشش ڪرڻ واري ”مدني انعامات“ تي عمل ڪندي هر مهيني مدني انعامات جا گهٽ ۾ گهٽ 26 رسالا تقسيم ڪري اڳين مهيني وٺڻ جي به ڪوشش ڪريو. هدف في ذيلي حلقو گهٽ ۾ گهٽ 12 رسالا آهن اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مدني انعامات تي عمل جون کوڙ بهارون آهن توهان جي ترغيب جي لاءِ هڪ مدني بهار پيش ڪريان ٿي جيئن ته

مدني برقعو پائڻ جي سعادت ملي وئي

باب المدينه ”ڪراچي“جي هڪ اسلامي ڀيڻ دنيا جي رنگينن ۾ گم ٿي ڪري ۽

آخرت جي امتحان کان بي فڪر ٿي ڪري زندگي جا ڏيهاڙا گزاري رهيون هيون هڪ ڏينهن دعوت اسلامي جي مشڪبار مدني ماحول سان وابسته ڪنهن اسلامي ڀيڻ انفرادي ڪوشش ڪندي ان کي عالمي مدني مرڪز فيضاني مدينه جي تهه خاني ۾ ٿيڻ واري اسلامي ڀينرن جي سنتن ڀري اجتماع ۾ شرڪت جي دعوت ڏني.ان جي شفقت ڪرڻ جي سبب ان کي سنتن ڀري اجتماع ۾ شرڪت جي سعادت نصيب ٿي.اتي مدني انعامات جي متعلق بيان جاري هو.انهن تمام غور سان بيان ٻڌڻ شروع ڪري ڏنو.بيان نهايت سوز ڀريو هو انهن تي هڪ دم رقّت طاري ٿي وئي ۽ بيان جي آخر ۾ انهن سچي دل کان نيت ڪئي ته اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ آئينده زندگي مدني انعامات تي عمل ڪندي گزارينديون.پوءِ مدني انعامات تي عمل جي برڪت سان مدني برقعو پائڻ جي سعادت به نصيب ٿي وئي هاڻي انهن پنهنجي زندگي کي هن مدني مقصد جي تحت گزارڻ جو پڪو ارادو ڪري رکيو آهي ته ”مون کي پنهنجي ۽ سڄي دنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻي آهي“ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ پنهنجي اصلاح جي لاءِ مدني انعامات تي عمل ڪرڻو آهي ۽ سڄي دنيا جي ماڻهن جي اصلاح جي ڪو شش جي لاءِ گهر جي محرم مردن کي مدني قافلي ۾ سفر ڪرائڻو آهي. زينت جون سُنّتون ۽ آداب

اچو!مڪتبة المدينه جي ڪتاب ”سُنّتون ۽ آداب“ صَفحي نمبر78 تان زِينَت (سينگار) جون سُنّتون ۽ آداب ٻڌو: ٭ انسان جي وارن جي چوٽي ٺاهي عورَت

پنهنجي وارن ۾ ٻڌي، اهو حرام آهي۔ حديثِ مبارَڪ ۾ ان تي لَعنت آئي بلڪ ان تي به لَعنت آئي آهي جنهن ڪنهن ٻي عورَت جي مٿي ۾ انساني وارن جي چوٽي ٻڌي۔ (درمختار،ڪتاب الحظروالاباحۃ،باب في النظرِ والمس،۹/۶۱۴تا ۶۱۵) ٭ جيڪڏهن اهي وار جنهن جي چوٽي ٺاهي وئي پاڻ ان عورَت جا پنهنجا وار ئي آهن جنهن جي مٿي ۾ لڳائي وئي تڏهن به ناجائز آهي۔ (درمختار، ڪتاب الحظروالاباحۃ،ج۹،ص۶۱۴تا ۶۱۵) ٭ ڪجھ ماڻهو ڇوڪرن جا به ڪن ٽوپائيندا آهن ۽ والي وغيرہ پارائيندا آهن اهو ناجائز آهي ۔يعني ڇوڪرن جو ڪن ٽوپڻ به ناجائز ۽ان ۾ زيور پائڻ به ناجائز۔ (ردالمحتار، ڪتاب الحظر والاباحۃ،فصل في اللبس، ۹/۵۹۸ ملخصاً) ٭ عورتن جي هٿن پيرن تي ميندي لڳائڻ جائز آهي، ننڍن ٻارن جي هٿن پيرن تي ميندي هڻڻ ناجائز آهي، ٻارڙين کي ميندي هڻڻ ۾ حرج ناهي۔ (ردالمحتار،ڪتاب الحظر والاباحۃ،فصل في اللبس،۹/۵۹۹ ملتقطاً) حضرت سَيدُنا حضرت سيّدُنا ابوهريره رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته مديني جي تاجدار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم وٽ هڪ مُخَنَّث (يعني کدڙو) حاضر ڪيو ويو، جنهن پنهنجا هٿ ۽ پير مينديءَ سان رنڱيا هئا۔ ارشاد فرمايائون؛ ان جو ڇا حال آهي؟ (يعني ان ڇو ميندي هنئي آهي؟) ماڻهن عرض ڪيو: هي عورتن جي نقل ڪري ٿو، اسان جي مٺي مدني آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم حڪم فرمايو ته ”هن کي شهر بدر ڪري ڇڏيو، تنهن ڪري ان کي شهر بدر ڪيو ويو، مديني منوره مان ڪڍي ”نَقِيع“ (مديني پاڪ جي ويجھو مقام) ڏانهن موڪليو ويو۔ (ابو داود،ڪتاب الادب،باب في الحڪم في المخنثين،حديث: ۴۹۲۸،۴/ ۳۶۸) جهڙي طرح مردن کي عورتن جي نقل ڪرڻ جائز ناهي، اهڙي طرح عورتون به مردن جي نقل نه ٿيون ڪري سگهن۔ جيئن ته حضرت سيّدُنا ابنِ عباس رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا کان روايت آهي ته رسول اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم لعنت فرمائي، زنانه مردن تي جيڪي عورتن جهڙي صورت بڻائن ۽ مردانه عورتن تي جيڪي مردن جهڙي صورت بڻائن۔ (مسند امام احمد،مسند عبد الله بن

عباس،حديث:۲۲۶۳،۱/ ۵۴۰) ٭ عورتون پنهنجن مڙسن جي لاءِ جائز شين سان، پر گهر جي چؤديواريءَ ۾ سينگار ڪن پر ميڪ اپ ڪري ۽ ٺهي ٺُڪي گهر کان ٻاهر نه نڪرن ڇوته اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: ”عورت پوري جي پوري عورت (يعني لڪائڻ جي شيء) آهي جڏهن ڪا عورت ٻاهر نڪرندي آهي ته شيطان ان کي جهاتي پائي ڏسندو آهي۔ (ترمذي،ڪتاب الرضاع،باب(۱۸)،حديث: ۱۱۷۶،۲/۳۹۲)
