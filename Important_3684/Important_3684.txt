منقول آهي ته حضرت سَيِّدُنا شيخ عبدُالقادر جِيلاني رَضِىَ اللهُ تَعَالٰی عَـنْهُ ڪرسي تي ويهي فرمائي رهيا هئا ته مون حُضُور سيدِعالَم، نُورِمجسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زِيارت ڪئي ته حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن مون کي فرمايو: پُٽ! تون بَيان ڇو نه ٿو ڪرين؟ مون عَرۡض ڪيو: اي منهنجا نانا جان (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم)! آئون هڪ عجمي مرد آہيان، بغداد ۾ فُصَحاء (بهترين ڪلام ڪرڻ وارن) جي سامهون بيان ڪيئن ڪريان؟ سَر ڪ ار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن مون کي فرمايو: پُٽ! پنهنجو وات کول! مون پنهنجو وات کوليو، ته حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن منهنجي وات ۾ ست (7) ڀيرا پنهنجو لُعابِ مُبار ڪ وِڌو ۽ مون کي فرمايو: ماڻهن جي سامهون بيان ڪندو ڪر ۽ انهن کي پنهنجي رَبّ عَزَّوَجَلَّ جي طرف عُمدہ حڪمت ۽ نصيحت سان گُھراءِ! ٻيهر مون نمازِ ظُہۡر اَدا ڪئي ۽ ويهي رهيس، مون وٽ ڪيترائي ماڻهو آيا ۽ مون کي وڏي آواز سان چَوڻ لڳا، ان کان پوءِ مون اَمِيۡرُالمؤمنين حضرت سيدُنا عَلي الۡمُرتَضٰي، شَيرِ خدا كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم جي زِيارت سان مُشرَّف ٿيس، پاڻ فرمايائون: پُٽ! تون بَيان ڇو نه ٿو ڪرين؟ مون عَرۡض ڪيو: اي منهنجا والد! ماڻهو مون سان وڏي آواز سان ڳالهائن ٿا۔ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن فرمايو: پنهنجو وات کول! مون پنهنجو وات کوليو، ته پاڻ منهنجي وات ۾ ڇھ (6) ڀيرا لُعاب وڌو، مون عَرۡض ڪيو ته توهان ست (7) ڀيرا ڇو نه وڌو؟ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن فرمايو: رَسُوۡلُ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ادب جي سبب۔ پوءِ اُهي منهنجي اکين جي سامهون کان

غائب ٿي ويا۔ (بہجۃالاسرار، ص ۵۸ ،از غوثِ پاک کے حالات: ۳۳) انهي واقعي کان پوءِ حُضُور غوثِ اعظم، سَيدُنا شيخ مُحِي الدِّيۡن عبدُالقادر جيلاني رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن ماھِ شَوّالُ الۡمُڪرَّم 521ھ هڪ عظيمُ الشَّان اجتماع ۾

پہريون ڀيرو بيان فرمايو، جنهن ۾ ہيبت ۽ رونق ڀريو پُرڪيف منظر هو، اولياءِ ڪرام ۽ فِرِشتن ان کي ڍڪي رکيو هو، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن ڪتاب ۽ سُنت جي تَصۡريح سان گڏ ماڻهن کي اللہ تبارَ ڪ وَتَعَالٰي جي طرف گھُرايو ته سڀ ماڻهو اِطاعت ۽ فرمانبرداري جي لاءِ جلدي ڪرڻ لڳا۔

(بہجۃالاسرار، ذکروعظہ ،ص ۱۷۴)

لُعاب اپنا چٹایا احمدِ مُختار نے ان کو توپھر کیسے نہ ہوتا بول بالاغوثِ اعظم کا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

غوثِ پاڪ جو پهريون بيان!

م ٺ ا م ٺ ا اسلامي ڀ ائرو! حُضُور غوثِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن نبي ڪ ريم ، رؤفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۽ اَمِيرُالۡمُؤمِنِيۡن حضرت سَيدُنا عَلي الۡمُرتَضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم جي ترغيب ڏيارڻ تي ۽ لُعابِ مُبارڪ کان فَيۡض حاصل ڪرڻ بعد پہرين ئي بَيان سان ماڻهن جا دل اوهان جي طرف مائل ٿي ويا ۽ ماڻهن جا ٽولن جا ٽولا اوهان جي صحبتِ بابرڪت کان فيض حاصل ڪرڻ لڳا۔ سندن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي مَحفل ۾ زِندگي جي تمام شُعۡبن سان تَعَلُّق رکڻ وارا ماڻهو شريڪ هوندا هئا، عُلَماء ۽ فُقَہاء سڀ جمع ہوندا هئا، ”غوث پاڪ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي مجۡلسِ وَعظ جو هي عالَم هو جو هڪ ئي وَقۡت چار چار سو ماڻهون قلم ۽ دواتون (مَس ڪُپڙيون) کڻي ڪري حاضر ہوندا هئا ۽ سندن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جا ملفوظات تحرير ڪندا هئا۔“ (اخبار الاخيار: ۱۲)

مُحَرِّر چار سو مجلس میں حاضرہوکےلکھتےتھے ہوا کرتا تھا جو اِرشادِ والاغوثِ اعظم کا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ مَواعظِ حَسَنه (چڱين نصيحتن) ، صَبۡر و رِضا، زُہۡد و تَقويٰ ۽ خوفِ خُدا جا آئينه دار ہئا، پنهنجي وَعظ ۽ نصيحت جي ذريعي مَحَبَّت ۽ اُخُوَّت جو دَرۡس ڏيندا هئا، ماڻهن کي آٓخِرت جي طرف مُتَوجھ ڪندا هئا، حُبّ جاھ ومال،نِفاق،رِياڪاري،بُغۡض ۽ ڪيني سميت تمام باطني بيمارين کان بچڻ جو درس ڏيندا هئا، جڏهن ته دُنيا جي بي ثَباتي؟، ايمان تي پُخۡتَگي، اعمالِ صالحه (چڱن عملن) جي اَہَميت ۽ ڪامل اَخۡلاق کي اپنائڻ جي ترغيب به ڏيندا هئا، سندن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جا فرامينِ مُقدَّسه دل مان نڪرندا ۽ دل ۾ گھر ڪري ويندا هئا۔ شيخ عَبۡدُالحق محدّثِ دہلوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اِرشاد فرمائن ٿا: حضرت شيخ (عبدُالقادر جيلاني) رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي ڪا محفل اهڙي نه ہوندي هئي، جنهن ۾ غير مُسلم، اوهان جي دَسۡتِ حق پرست تي دولتِ ايمان کان مُشرف نه ٿيندا هجن ۽ نافرمان، ڌاڙيل، گمراھ ۽ بدمَذہب ماڻهون سندن جي ہٿ تي تائِب نه ٿيندا هجن۔ جڏهن پنج سو (500) کان وڌيڪ غير مُسلم ۽ لکين سياھ ڪار سندن جي هٿ تي تائب ٿي چڪا ۽ پنهنجي بداَعمالين کان باز اچي چڪا هئا ته ٻين ماڻهن جي (تعداد) جي باري ۾ ڇا ٿو چئي سگھجي؟ (ہر هڪ توهان جي فرامينِ مُبارڪه کان مُسۡتَفِيد ٿيندو هو) (اخبارالاخيار: ۱۳) اَمْرٌ بِالْمَعرُوف وَ نَہی عَنِ الْمُنْڪر يعني نيڪي جو حڪم ڏيڻ ۽ بُرائي کان منع ڪرڻ جي مُعاملي ۾ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ عملدارن جي پرواھ نه ڪندا هئا ۽ خليفن، وزيرن، سَلاطين ۽ عوام ۽ خواص سڀني کي نيڪي جي دعوت ڏيندا هئا، حضرتِ سَيدُنا امام شمسُ الدّين محمد بن احمد ذہبي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: شيخ عبدُا لقادر جيلاني رَضِىَ اللهُ تَعَالٰی عَـنْهُ بر سرِ منبر اعلانِ حق فرمائيندا هئا ۽ جيڪو ڪنهن ظالم کي ماڻهن تي حاڪم بڻائي، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ان تي اعتراض ڪندا هئا۔ (تاريخ الاسلام للذہبي: ۳۹/۹۹)

پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ پنهنجي عمر مُبارَڪ جا 33 سال درس، تَدريس ۽ فَتويٰ نَويسي ۾ گذاريا، جڏهن ته 40 سال مخلوقِ خُدا ۾ وَعظ ۽ نصيحت جا مدني گُل نڇاوَر ڪيا ۽ 90 سال جي عمر ۾ 561ھ ۾ وِصال فرمايو۔ (اخبار الاخيار: ۹)

سلاطینِ جہاں کیونکر نہ اُن کے رُعب سے کانپیں نہ لایا شیر کو خطرے میں کُتّا غوثِ اعظم کا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

فائديمند ” ڪارتُوس“

م ٺ ا م ٺ ا اسلامي ڀ ائرو! اسان جي م ٿ ي جا تاج، حُضُورِ غوثِ پا ڪ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جيئن ته خُود باعمل هئا، انهيءَ لاءِ سندن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي نيڪي جي دعوت تي مشتمل ڪلما تاثير جو تِير بڻجي ڪري ماڻهن جي سِينن ۾ پيوست ٿي ويا ۽ اُنهن جي دل جي دُنيا بدلجي ويندي۔ ياد رکو! جيڪڏهن ڪارتُوس بہترين بندوق سان هَلايو وڃي ته ان سان شينهن به ماريو وڃي سگھجي ٿو، اسان جي نيڪي جي دعوت جا اَلۡفَاظ ”ڪارتُوس“ وانگر آهن ۽ اسان جو ڪردار بندوق جي مثل آهي، جيڪڏهن نيڪي جي دعوت ڏيڻ ۽ بُرائي کان منع ڪرڻ وارو خُود باعمل ہوندو ته ان جي ڳالھ ۾ به اثر ہوندو، تنهنڪري پنهنجي ڳالھ ۾ تاثير پيدا ڪرڻ جي لاءِ اسان کي پنهنجي عملي حالت به دُرُسۡت ڪرڻ گھرجي۔ حُضُور غوثِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ خُود عمل ڪرڻ جي بغير ڪئي ويندڙ نصيحت جي باري ۾ اِرشاد فرمائن ٿا: خُود عمل ڪرڻ جي بغير، ٻين کي نصيحت ڪرڻ، ڪا اَہَميت نه ٿي رکي۔ (الفتح الرباني، المجلس العشرون ،ص ۷۸) وڌيڪ فرمائن ٿا: خُود عمل نه ڪرڻ، ٻين کي نصيحت ڪرڻ ائين ئي آهي، جيئن بغير دروازي جي گھر ۽ ان ۾ گھر جون ضروري شيون به نه هجن ۽ اهڙو خزانو آهي، جنهن مان ڪجھ خرچ نه ڪيو وڃي ۽ هي ڳالهه اهڙي دعويٰ وانگر آهي، جنهن جو ڪو گواھ نه هجي۔ (الفتح الرباني،المجلس العشرون ،ص ۷۸)

مٺا مٺا اسلامي ڀائرو! جيڪڏهن ڪنهن طبيب وٽ ہر بيماري جي بهتر کان بهتر دَوا مَوۡجُود ہجي ۽ ٻيا ماڻهو ان مان فائدو به حاصل ڪندا ہجن، پر جڏهن خُود ان طبيب کي اهو مَرَض لاحِق ٿئي ته لاپرواهي سان ڪم وٺندو رهي ۽ آخرڪار جان کان هٿ ڌوئي ويهي، ته ان کي بيوقوف ئي چيو ويندو۔ بلڪل اهڙي طرح جيڪي ماڻهون ٻين کي ته گناھن کان بچڻ ۽ نيڪيون

ڪرڻ جي نصيحت ڪندا آهن، پر پاڻ عمل جي ڪوشش نه ٿا ڪن، اهي به سَراسَر نُقصان ۾ آهن۔

نبي مُ ڪ رم، نُورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اِرۡشاد فرمايو:''ڪجھ جنّتي ماڻهو جہنمين جي طرف ويندا، ته ان کان پُڇندا، توهان جہنَّم ۾ ڪنهن سبب جي ڪري داخل ٿيا، حالانڪه اللہ عَزَّوَجَلَّ جي قسم! اسان ته توهان جي ئي تعليم سان جنَّت ۾ داخل ٿيا آهيون۔ ”ته اهي چوندا، اسان جيڪا ڳالھ ڪندا هئاسين، ان تي خُود عمل نه ڪندا هئاسين۔“ (المعجم الکبير،الحديث: ۴۰۵ ،ج ۲۲ ،ص ۱۵۰)

عمل کا ہو جذبہ عطا یا اِلٰہی گناھوں سے مجھ کو بچا یا اِلٰہی

عبادت میں گزرے مِری زِندگانی کرم ہو کرم یا خدا یا اِلٰہی

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

رسولِ ا ڪ رم، شاہ ِبني آدم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اِرۡشاد فرمايو:''ماڻهن کي عِلۡم سيکارڻ ۽ پاڻ کي وِساري ڇڏڻ واري جي مثال ان ڏيئي وانگر آهي جيڪو ماڻهن کي ته روشني ڏيندو آهي، جڏهن ته پاڻ کي ساڙيندو آهي۔“

پر پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ڪشۡف جي ذريعي ان جي مَعصيت (گناھ) ۽ بدڪرداري کي لکيل ڏسي ورتو، ان چور جي دل ۾ خيال آيو: ”شايد هي غوثِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ آهن۔“ سندن کي ان جي دل ۾ پيدا ٿيڻ واري ان خيال جو به علم ٿي ويو، ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمايو: آئون عبدُالقادِر آهيان۔ هي ٻُڌندي ئي اهو چور فوراً اوهان جي مُبارڪ قدمن تي ڪِري پيو ۽ ان جي زبان تي يا سَيِّدِي عَبدَالقَادِرِشَيئًالِلّٰہِ (يعني اي منهنجا آٓقا عبدُالقادر! منهنجي حال تي رحم فرمايو) جاري ٿي ويو۔

کي مَدرَسةُ المدينه (برائي بالغان) جون روحاني فضائون ته ڇا ميسر آيون، منهنجي ته خوش بختي جي سفر جو آغاز ٿي ويو۔ مَدرَسَةُ المدينه (برائي بالغان) جي برڪتن منهنجي دل جي ڪارهه کي خوفِ خدا ۽ عشقِ مُصطفٰي جي ڏيئي سان مُنوَّر ڪري ڇڏيو۔ ان ۾ مون کي قرآنِ ڪريم جي تعليم سان گڏوگڏ سُنتن تي عمل جو جذبو به مليو ۽ ہفتيوار سُنتن ڀري اجتماع ۾ شرڪت جي سعادت به نصيب ٿي۔ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ مدرسةُ المدينه (برائي بالغان) ۾ پڙهڻ جي برڪت سان منهنجي زندگي ۾ مدني بہار اچي وئي، فيشن پرستي ۽ موج مستي کان نجات حاصل ٿي وئي ۽ آئون دعوتِ اسلامي جي مہڪندڙ مہڪندڙ مَدَني ماحول سان وابسته ٿي ويس۔ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ مَدَني قافله ذمه دار جي حيثيت سان مَدَني ڪمن جي ڌومون مچائڻ جي سعادت حاصل آهي۔

م ٺ ا م ٺ ا اسلامي ڀ ائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو.

(مِشْکاۃُ الْمَصابِیح ،ج ۱ ص ۵۵ حدیث ۱۷۵ دارالکتب العلمیۃ بیروت)

سینہ تری سُنَّت کا مدینہ بنے آقا

جنّت میں پڑوسی مجھے تم اپنا بنانا

م ٺ ا م ٺ ا اسلامي ڀ ائرو! اڪثر ڪري اسان کي سفر جي ضرورت پيش ايندي آهي. تنهن ڪري اسان ڪوشش ڪري سفر جون به ڪجهه نه ڪجهه سنّتون ۽ آداب سکي وٺون ته جيئن انهن تي عمل ڪري اسان پنهنجي سفر کي به ثواب حاصل ڪرڻ جو ذريعو بڻائي سگهون.

(1) ٿي سگهي ته خميس واري ڏينهن کان سفر جي شروعات ڪئي وڃي ڇوته خميس تي سفر جي شروعات ڪرڻ سنّت آهي. (اشعۃاللمعات،ج ۵ ،ص ۱۶۱)

(2) جيڪڏهن سهولت هجي ته رات جو سفر ڪيو وڃي ڇوته رات جو سفر جلدي طئي ٿيندو آهي.

(3) جيڪڏهن ڪجهه اسلامي ڀائر گڏجي قافلي جي صورت ۾ سفر ڪن ته ڪنهن هڪ کي امير بڻائي وٺو.

(4) هلڻ وقت عزيزن، دوستن کان غلطيون معاف ڪرايو ۽ جن کان معافي گهري وڃي انهن تي به لازم آهي دل سان معاف ڪري ڇڏين.

(بہارِ شریعت،حصہ ۶ ،ص ۱۹)

(5) سفر جو لباس پائي جيڪڏهن مڪروهه وقت نه هجي ته گهر ۾ چار رڪعتون نفل،” الحمد ۽ قل “ سان پڙهي ٻاهر نڪرو، اهي رڪعتون واپسيءَ تائين گهروارن ۽ مال جي نگهباني ڪنديون.

(6) اسان جڏهن به سفر تي روانا ٿيون ته اسانکي گهرجي ته اسان پنهنجي گهروارن ۽ مال کي الله عَزَّوَجَلَّ جي حوالي ڪري وڃون، الله تبار ڪ وتعاليٰ ئي سڀ کان بهتر حفاظت ڪرڻ وارو آهي. بلڪه ٿي سگهي ته پنهنجي گهروارن لاءِ هي جملا چئي سفر تي روانا ٿيون. اَسْتَوْدِعُکَ اللہَ الَّذِیْ لَایُضِیْعُ وَدَائِعَہٗ۔ ترجمو: مان توهانکي الله عَزَّوَجَلَّ جي حوالي ڪ يان ٿ و جي ڪ و حوالي ڪ يل امانتن کي ضايع ناهي ڪ ندو . (سنن ابن ماجه، کتاب الجهاد، باب تشبيع الغزوة وداداعهم الحديث 2825ج 3،ص 372)

(7) واپار لاءِ سفر ڪرڻ وارن اسلامي ڀائرن کي گهرجي ته هي پنج سورتون پڙهندا ڪن. (1) قُلْ یٰۤاَیُّهَا الْكٰفِرُوْنَۙ(۱) آخر تائين. (2) اِذَا جَآءَ نَصْرُ اللّٰهِ وَ

الْفَتْحُۙ(۱) آخر تائين، (3) قُلْ هُوَ اللّٰهُ اَحَدٌۚ(۱) آخر تائين. (4) قُلْ اَعُوْذُ بِرَبِّ الْفَلَقِۙ(۱) آخر تائين. (5) قُلْ اَعُوْذُ بِرَبِّ النَّاسِۙ(۱) آخر تائين.

(8) جڏهن سيڙهين تي چڙهو يا اونچي جڳھ جي طرف هلو، يا اسان جي بس وغيره ڪنهن اهڙي روڊ وٽان گذري جيڪا اونچائي جي طرف وڃي

ته ” اَللہُ اَکْبَرُ “ چوڻ سُنت آهي ۽ جڏهن سيڙهين تان لهون ته ” سُبْحَانَ اللہِ “ عَزَّوَجَلَّ چوڻ سُنت آهي. مسافر کي گهرجي ته اهو دعا کان غفلت نه ڪري جو اهو جيستائين سفر ۾ آهي ان جي دُعا قبول ٿيندي آهي بلڪه جيستائين گھر نه پهچي وڃي ان وقت تائين دعا مقبول آهي.

(9) منزل تي لهڻ وقت وقتًا فوقتًا اها دُعا پڙهو اِنْ شَآءَ اللّٰہ عَزَّوَجَلَّ هر نقصان کان بچنداسين. دُعا هي آهي: اَعُوْذُ بِکَلِمَاتِ اللّٰہِ التَّآمَّاتِ مِنْ شَرِّمَا خَلَقَ ۔ ترجمو: الله عَزَّوَجَلَّ جي ڪَلِماتِ تامه جي پناھ گھُران ٿو ان جي شر کان جنهن کي اُن پيدا ڪيو.

(کنز العمال ،کتاب السفر ،الفصل الثانی فی آداب السفر،الحدیث ۱۷۵۰۸ ،ج ۶ ،ص ۳۰۱)

(10) سفر جي واپسي تي گھروارن جي لاءِ ڪو تحفو وٺي اچو جو هي سُنت مبارڪه آهي. سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جڏهن سفر کان ڪو واپس اچي ته گھروارن جي لاءِ ڪجھ نه ڪجھ تحفو آڻي، چاهي جھولي ۾ پٿر ئي وجھي اچي.

(کنز العمال ،کتاب السفر ،الفصل الثانی فی آداب السفر ،الحدیث ۱۷۵۰۲ ،ج ۶ ،ص ۳۰۱)

هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي.

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿1﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿2﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿3﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([1])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([2])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([3])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([1])

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([2])

﴿2﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عسا ڪ ر ج 19 ، ص 155 ، حديث 4455)
