موسٰي عَلَيۡهِ السَّلَام کي ان فضيلت سان تعجب ٿيڻ لڳو جيڪي اللہ پاڪ سيدُالانبياء، مڪي مدني مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ۽ اُنهن جي اُمَّت کي عطا فرمائي ته چوڻ

لڳا: ”ڪاش! مان حضرت سيدُنا محمدِ مصطفٰي، احمدِ مُجتبيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اصحاب مان هجان ها۔“ ته اللہ پاڪ انهن جي رضا جي لاءِ 3 آيتون نازل فرمايون، جيئن سيپارو 9، سُورةُ الاَعراف جي آيت نمبر 144 ۽ 145 ۾ ارشاد آهي:

قَالَ یٰمُوْسٰۤى اِنِّی اصْطَفَیْتُكَ عَلَى النَّاسِ بِرِسٰلٰتِیْ وَ بِكَلَامِیْ ﳲ فَخُذْ مَاۤ اٰتَیْتُكَ وَ كُنْ مِّنَ الشّٰكِرِیْنَ(۱۴۴)وَ كَتَبْنَا لَهٗ فِی الْاَلْوَاحِ مِنْ كُلِّ شَیْءٍ مَّوْعِظَةً وَّ تَفْصِیْلًا لِّكُلِّ شَیْءٍۚ-فَخُذْهَا بِقُوَّةٍ وَّ اْمُرْ قَوْمَكَ یَاْخُذُوْا بِاَحْسَنِهَاؕ-سَاُورِیْكُمْ دَارَ الْفٰسِقِیْنَ(۱۴۵) (پ 9 الاعراف 144) ترجموڪنزالايمان: فرمايائِين اي مُوسى مُون توکي ماڻُهن مان چُونڊيو رسالتن ۽ پنهنجي ڪلام سان پوءِ وٺ جيڪِي مُون توکي عطا فرمايو ۽ شُڪر وارن مان ٿِئُ. ۽ اسان اُن جي واسطي تختيُن ۾ لکِي ڇڏِي هر قسم جي نصِيحت ۽ هر شئَ جو تفصِيل ۽ فرمايو اي مُوسى هِن کي مضبُوطِي سان پڪڙ ۽ پنهنجِي قوم کي حُڪم ڏي ته هِن جُون چڱيُون ڳالهيُون اختيار ڪن عنقرِيب مان توهان کي ڏيکارِيندس بي فرمان جو گھر.

سيپارو 9، سُورة الاَعراف جي آيت نمبر 159 ۾ اِرشاد آهي:

وَ مِنْ قَوْمِ مُوْسٰۤى اُمَّةٌ یَّهْدُوْنَ بِالْحَقِّ وَ بِهٖ یَعْدِلُوْنَ(۱۵۹) (پ۹،الاعراف:۱۵۹) ترجموڪنزالايمان: ۽ مُوسى جِي قوم مان هِڪ جماعت آهي جيڪا حق جِي واٽ ٻڌائي ٿِي ۽ اُنهِئَ سان اِنصاف ڪري ٿِي.

اهو ٻڌي حضرت سيدُنا موسٰي عَلَيۡهِ السَّلَام بلڪل راضي ٿي ويا۔

(اللّٰه والوں کی باتيں، ص ۵۱۶ تا ۵۱۹ ملخصا و ملتقطا)

اميرِ اهلسنّت، باني دعوتِ اسلامي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ اُمّتِ مُصطفيٰ ۾ پيدا ٿيڻ جي هن نعمت تي اللہ پاڪ جو شڪر ادا ڪندي بارگاھ اِلٰهي ۾ عرض ڪن ٿا:

شکر ادا ہو کیونکر تیرا پیارے نبی کی اُمّت میں مجھ سے نکمے کو بھی پیدا تُو نے اے رحمٰن کیا

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! ٻڌو اوهان! اُمّتِ مُصطفيٰ کي ربِّ ڪريم ٻه چار يا پنج نه بلڪه بيشمار فضيلتن ۽ برڪتن ۽ ڪيئي خُصوصيتن سان نوازيو آهي، اهو تورات

شريف جنهن کي اللہ پاڪ حضرت سيدنا موسيٰ عَلَيۡهِ السَّلَام تي نازل فرمايو، ان مقدس آسماني ڪتاب جي اندر به هن عظيم اُمّت جي فضيلتن ۽ ڪمال ۽ شاندار خُصوصيات کي بيان ڪيو ويو آهي، ايستائين جو جڏهن موسيٰ عَلَيۡهِ السَّلَام کي عِلم ٿيو ته اهي تمام فضيلتون ۽ ڪمالات اُمّتِ مُصطفيٰ جا آهن، ته پاڻ عَلَيۡهِ السَّلَام اللہ پاڪ جي بارگاھ ۾ هن اُمَّت کي پنهنجي اُمَّت بنائڻ جي اِلتجا (Request) پيش ڪئي، پر جڏهن ان جي اجازت نه ملي ته پوءِ پاڻ عَلَيۡهِ السَّلَام اُمَّتِ محبوب ۾ شامل ٿيڻ جي خواهش جو اِظهار هنن لفظن ۾ ڪيو ته ”ڪاش! مان حضرت سيدُنا محمدِ مصطفٰي، حبيبِ ڪبريا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اصحاب (يعني صحابه) مان هجان ها۔“

هتي ڪجھ مدني گل ذهن نشين ڪريو ته: (1) پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اُمت جو ڪهڙو ئي خاص ولي بلڪه صحابي ۽ اهلِ بيت بلڪه ڪو ملائڪ به ڪنهن به نبي جي برابر قطعاً نٿو ٿي سگھي، جيئن ته بهارِ شريعت ۾ آهي: عقيده (22): انبياءِ ڪرام، تمام مخلوق ايستائين جو رُسُلِ ملائڪه کان افضل آهن۔ ولي چاهي ڪيتري وڏي رتبي وارو هجي، ڪنهن نبي جي برابر نٿو ٿي سگھي۔ جيڪو ڪنهن غيرِ نبي کي ڪنهن نبي کان افضل يا برابر ٻڌائي، ڪافر آهي۔ (حصه 1 ،ص 47) (2) حضور پرنور، سيدالمرسلين، نبي الانبياء ۽ جميع مخلوقاتِ الٰهي کان افضل تر آهن. (فتاوي رضويه جلد۱۵، ص۲۶۸) (3) حضرت موسيٰ عَـلَيْـهِ الـسَّـلاَم جو ان اُمت ۾ شامل ٿيڻ جي خواهش جو اظهار فرمائڻ، ان اُمت جي فضيلت جو پتو ضرور ڏي ٿو پر اهي ان اُمت کان نهايت افضل بلڪ انبياء اُولُو العَزم (انهن 5 انبياء (عَـلَـيْهِمُ السَّلَام) مان آهن جيڪي ٻين نبين (عَـلَـيْهِمُ السَّلَام) کان به افضل آهن۔

اللہ پاڪ جا ڪروڙين ڪروڙ احسان آهي جو ان رسولِ ڪريم، رؤفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صدقي ۾ اسان کي مسلمان بڻايو ۽ اسان کي رحمتِ ڪونين، ناناءِ حسنين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اُمّت ۾ شامل ڪري اسان تي ايترو وڏو احسان ڪيو جو جيڪڏهن اسان ان جي هن احسان تي سڄي زندگي به ان جو شڪر ادا ڪندا رهون تڏهن ان جو حق ادا نٿا ڪري سگھون۔

امير ِاهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ پنهنجي مشهورِ زمانه نعتيه ديوان ”وسائل بخشش“ ۾ لکن ٿا ته

شکر تیرا کہ اُن کی اُمّت میں مجھ کو اے ذُوالجلال رکھا ہے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

اُمّتِ مصطفيٰ جي افضل هئڻ جو سبب

مٺا مٺا اسلامي ڀائرو! جهڙي طرح حضور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) تمام رسولن جا سردار ۽ سڀني کان افضل آهن، بِلا تشبيه حضور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي صدقي ۾ حضور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي اُمت تمام اُمتن کان افضل (آهي)۔ (بهارِ شريعت، ١/٥٤) اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! اسان ڪيترا خوشنصيب آهيون جو اللہ ڪريم جي حبيبِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو دامن ڪرم اسان جي گنهگار هٿن ۾ آيو، يقيناً اسان جا پيارا ۽ مٺا مٺا آقا مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تمام اَنبياءِ ڪرام عَلَيهم الـصَّلٰوة وَالـسَّلَام ۾ سڀني کان افضل ۽ اعليٰ آهن ۽ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صدقي ۾ سندن جي اُمت به سڀني اڳوڻين امتن کان افضل آهي. افضليت جو سبب هرگز هرگز هي ناهي ته هن اُمت ۾ سرمائي دارن جي ڪثرت هوندي يا اهي ماڻهو دنياوي طور تي گھڻا تعليم وارا هوندا، انهن ۾ انجينيئر ۽ ڊاڪٽرن جي گھڻائي هوندي، نه ئي فضيلت جو سبب هي آهي ته هي جنگجو، بهادر ۽ طاقتور هوندا يا اهي انهيءَ ڪري افضل آهن ته نهايت ئي چالاڪ ۽ هوشيارهوندا، بلڪ انهن جي فضيلت جو سبب ته هي آهي جو اهي ڀلائي جو حڪم ڏيڻ ۽ برائي کان روڪڻ جي اهم منصب تي فائز آهن. جيئن سيپارو 4،سوره ٔ آلِ عمران جي آيت نمبر110 ۾ خداءِ رحمٰن جو فرمانِ عاليشان آهي:

كُنْتُمْ خَیْرَ اُمَّةٍ اُخْرِجَتْ لِلنَّاسِ تَاْمُرُوْنَ بِالْمَعْرُوْفِ وَ تَنْهَوْنَ عَنِ الْمُنْكَرِ وَ تُؤْمِنُوْنَ بِاللّٰهِؕ

ترجمو ڪنزالايمان ؛ توهان ڀلا آيو انهن سڀني امتن کان جيڪي ماڻهن ۾ ظاهر ٿيون چڱائي جو حڪم ڪريو ٿا ۽ برائي کان روڪيو ٿا ۽ الله تعاليٰ تي ايمان رکو ٿا۔

تفسيرِ خازن ۾ هن آيتِ سڳوري تحت لکيل آهي: هن اُمَّت کي نيڪي

جو حڪم ڏيڻ ۽ برائي کان منع ڪرڻ جي ڪري ٻين تمام اُمّتن تي فضيلت ڏني وئي آهي ۽ انهيءَ سبب کان اها اُمّت تمام اُمّتن ۾ سڀ کان بهترين اُمّت آهي، ثابت ٿيو! هن اُمّت جي بهترين هئڻ جو سبب (Reason) ان جي افراد جو نيڪي جي دعوت ڏيڻ ۽ برائي کان منع ڪرڻ آهي۔ (تفسيرِ خازن،پ ۴،آل عمران،تحت الآية:۱۱۰ ،۱/۲۸۹)

مٺا مٺا اسلامي ڀائرو! طاعون هڪ اهڙي جان ليوا وبائي بيماري آهي جنهن کي ڊاڪٽر (پليگ) Plague چوندا آهن۔ هن بيماري ۾ ڪنڌ، بغل ۽ ران جي ڪناري ۾ انب جي گھوٽي(ککڙي) برابر ڳوڙهيون نڪرنديون آهن۔ جن ۾ بي پناھ درد ۽ ناقابلِ برداشت سوزش ٿيندي آهي۔ شديد بخار ٿيندو آهي، اکيون ڳاڙهيون ٿي دردناڪ جلن سان شعلي جيان سڙنديون آهن، آخرڪار مريض درد جي شدت ۽ شديد بي چيني ۽ بيقراري ۾ تڙپي تڙپي تمام جلد مري ويندو آهي۔ (عجائب القرآن، ص۲۶۱)

ياد رهي! طاعون جو اهو تباه ڪن مَرَض اڳوڻين اُمّتن جي لاءِ عذاب طور مقرر ڪيو ويو هو، پر اهو اُمَّتِ محبوب جي خُصوصيات مان آهي ته اللہ پاڪ پنهنجي مدني حبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صدقي ۾ هن مرض کي مومنن جي لاءِ رحمت جو سبب بڻايو آهي، جيئن

اُمّتِ محمديه تي الله ڪريم جو خاص فضل ۽ ڪرم

بخاري شريف جي حديثِ پاڪ ۾ آهي ته تمام نبين جي سردار، جنابِ احمدِ مختار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: طاعون هڪ عذاب هو، اللہ پاڪ جنهن تي چاهي ان ڏانهن موڪلي۔ پوءِ اللہ ڪريم مؤمنين جي لاءِ ان کي رحمت بڻائي ڇڏيو۔ ته جيڪو شخص طاعون ڦهلجڻ جي زماني ۾ پنهنجي شهر ۾ صبر سان ثواب جي طلب لاءِ هن اعتقاد سان ترسيل رهي ته ان کي اهو ئي پهچندو جيڪو اللہ ڪريم ان جي لاءِ لکيو آهي ته ان جي لاءِ شهيد جي مثل ثواب آهي ۔ (بخاري ،ڪتاب الطب ، باب اجر الصابر في الطاعون ،۴/۳۰ حديث: ۵۷۳۴)

حضرت علامه غلام رسول رضوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: هن حديث مان واضح ٿئي ٿو ته هن اُمَّتِ مُحَمَّدِيه تي اللہ ڪريم جي وڏي مهرباني آهي ڇوته جيڪا بيماري ٻين اُمّتن جي لاءِ عذاب مقرر ڪئي وئي آهي، اها هن اُمّت جي لاءِ اللہ پاڪ جي رحمت آهي۔ طاعون بني اسرائيل جي لاءِ عذاب ۽ هن اُمّت جي لاءِ رحمت آهي۔ (تفهيم البخاري، ۵/۳۵۵) ٻي هنڌ تي لکيائون: هن اُمّت جي مومنن جي لاءِ طاعُون کي رحمت ڪيو ويو آهي، هن جو رحمت هئڻ هن لحاظ سان آهي ته هي شهيد جي ثواب کي شامل آهي، توڙي جو صورت جي اعتبار سان سخت تڪليف ده آهي۔ (تفهيم ا لبخار ي، ۸/۸۰۰)

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! اللہ ڪريم جي رحمت تي قربان جو ان مديني جي تاجدار، ٻنهي جھانن جي مالڪ و مختار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اُمّت جي لاءِ طاعون جيڪو اڳوڻي اُمّتن جي لاءِ بطورِ عذاب مقرر هو، پر هن اُمّت جي لاءِ ان کي رحمت بڻايو ويو۔ جنهن سان اهو اندازو لڳائڻ ٿورو به مشڪل ناهي ته اللہ پاڪ

پنهنجي پياري رسول، رسولِ مقبول صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ۽ سندن اُمّت سان ڪيتري محبت فرمائي ٿو۔

ٿورو سوچيو! گناهن جي ارتڪاب سبب جيڪڏهن هن اُمّت جي لاءِ به طاعُون کي عذاب بڻايو وڃي ته ڪيتري قدر تڪليف ۽ آزمائش ٿئي ها، لهٰذا اسان کي گھرجي ته اسان اللہ پاڪ جي نافرمانين کان پاڻ به بچون ۽ ٻين کي به بچايون، خوب خوب نيڪيون ڪيون ۽ سُنّتن ڀري زندگي گذاريون۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ عاشِقانِ رَسول جي مَدَني تحريڪ دعوتِ اسلامي جي مَدَني ماحول جي برڪت سان گناهو ں کان نفرت، نيڪين سان محبت ۽ سُنّتن تي عمل جو جذبو نصيب ٿيندو آهي، تنهنڪري هن مدني ماحول سان وابسته رهي 12 مدني ڪمن ۾ عملي طور تي شامل ٿي نيڪين جو ذخيرو گڏ ڪندي راھ آخرت جي لاءِ آخرت جو سامان گڏ ڪريو۔

ذيلي حلقي جي 12 مَدَني ڪمن مان روزانو جو هڪ مَدَني ڪم ”صداءِ مدينه“ به آهي۔ دعوتِ اسلامي جي مَدَني ماحول ۾ مسلمانن کي نمازِ فجر جي لاءِ جاڳائڻ کي ”صَداءِ مدينه“ لڳائڻ چوندا آهن۔ ٭ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ ”صَداءِ مدينه“ جي برڪت سان نمازِ تهجد جي سعادت ملي سگھي ٿي۔ ٭ "صداءِ مدينه" جي برڪت سان نماز جي حفاظت ٿيندي آهي۔ ٭ "صداءِ مدينه" جي برڪت سان مسجد جي پهرين صف ۾ تڪبيرِاُوليٰ سان فجر نماز جي ادائيگي ٿي سگھي ٿي۔ ٭ "صداءِ مدينه" جي برڪت سان "نيڪي جي دعوت" ڏيڻ جو ثواب به حاصل ڪري سگھجي ٿو۔ ٭ "صداءِ مدينه" جي برڪت سان دعوتِ اسلامي جي نيڪ نامي ۽ تشهير ٿيندي۔ ٭ "صداءِ مدينه" لڳائڻ وارو باربار مُسلمانن کي حج ۽ مٺو مدينو ڏسڻ جي دُعا ڏيندو آهي، اللہ پاڪ چاهيو ته اهي دُعائون، ان جي حق ۾ به قبول ٿينديون۔ ٭ "صداءِ مدينه" ۾ پيدل هلڻ جي برڪت سان صحت به سٺي ٿيندي۔ ٭ "صداءِ مدينه" لڳائڻ مسلمانن کي نمازِ فجر جي لاءِ جاڳائڻ آهي، مسلمانن کي نمازِ فجر جي لاءِ جاڳائڻ سُنّتِ مصطفيٰ آهي، مسلمانن کي نماز فجر جي لاءِ جاڳائڻ سنت حيدري ۽ سنت فاروقي آهي، اَمِيرُالمؤمنين حضرت

سَيدُنا عمر فاروقِ اعظم رَضِىَ اللّٰهُ تَعَالٰى عَنْه نمازِ فجر جي لاءِ ماڻهن کي جاڳائيندي مسجد تشريف فرما ٿيندا هئا۔ (طبقات ڪبريٰ،ذڪر استخلاف عمر، ۳/۲۶۳ مفهوماً)

”صداءِ مدينه“ واري مدني ڪم بابت وڌيڪ مفيد ترين معلومات ڄاڻڻ جي جي لاءِ مڪتبة المدينه جي مطبوعه رِسالي ”صداءِ مدينه“ جو مطالعو ڪريو۔

اچو! بطورِ ترغيب ”صداءِ مدينه“ لڳائڻ بابت هڪ مَدَني بهار ٻڌي خوش ٿي وڃو،

صداءِ مدينه ڏيکاري مدينه

مٺا مٺا اسلامي ڀائرو! شبِ قدر اها عظيمُ الشَّان رات آهي جنهن جي اَهمِّيت ۽ فضيلت کي مسلمانن جو ٻچو ٻچو ڄاڻي ٿو، ڇوته هن رات ۾ قُرآنِ ڪريم لاٿو ويو، اها رات هزار (1000) مهينن کان افضل آهي، هن رات جي فضيلت ۾ قرآنِ ڪريم جي ٽيهين (30) سيپاري ۾ هڪ مڪمل (Complete) سُورت به آهي، ان جي علاوه هن رات جا ٻيون به کوڙ فضيلتون ۽ برڪتون ڪتابن ۾ موجود آهن۔ ياد رهي! هن اُمّت کان اڳ ۾ به ڪيئي اُمّتون آيون پر ڪنهن کي به اها عظيمُ الشان نعمت نه ملي، پر ربِّ ڪريم تمام نبين جي سردار، مڪي مديني جي تاجدار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اُمّت کي اهو شرف بخشيو ته ان کي شبِ قدر جهڙي عظيمُ الشان ۽ مبارڪ رات جو تحفو عطا فرمايو، جيئن

شبِ قدر عطا ڪئي وئي

حضرت سَيدُنا عَبْدُ اللہ بن عباس رَضِىَ اللّٰهُ تَعَالٰى عَنْهمَا فرمائن ٿا: سرڪارِ مدينه صَلَّى اللهُ

تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ وٽ بني اسرائيل جي هڪ شخص جو تذڪرو ٿيو، جنهن هڪ هزار مهيني اللہ ڪريم جي راھ ۾ وڙهيو۔ صحابَه ڪرام عَلَيهمُ الرِّضْوَان کي ان بابت گھڻو تعجب ٿيو ۽ تمنا ڪرڻ لڳا: ڪاش! انهن جي لاءِ به ائين ممڪن هجي ها۔ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اللہ ڪريم جي بارگاھ ۾ عرض ڪيو: اي منهنجا ربِّ ڪريم! تو منهنجي اُمّت کي ننڍي عمر عطا ڪئي آهي، هاڻ انهن جا عمل به گھٽ ٿيندا۔ ته اللہ ڪريم پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي شبِ قدر عطا فرمائي ۽ ارشاد فرمايو: اي محمد (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ)! شبِ قدر هزار مهينن کان بهتر آهي جيڪا مون تون کي ۽ تنهنجي اُمّت کي هر سال عطا فرمائي آهي۔ اها رات رمضان ۾ توهان جي لاءِ ۽ قيامت تائين ايندڙ توهان جي اُمّتين جي لاءِ آهي، جيڪا هزار مهينن کان بهتر آهي۔ (الروض الفائق،ص۴۹)

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! ٻڌو اوهان! اسان جا بخشيل آقا، اسان گناهگارن جي گناهن کي بخشائڻ وارا مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي اُمّتين تي ڪيتري قدر مهربان آهن جو جڏهن پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اڳيان بني اسرائيل جي هڪ شخص جو قصو بيان ٿيو ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ غمِ اُمّت ۾ بيقرار ٿي ويا ۽ انهيءَ بيقراري جي عالم ۾ پاڻ سڳورا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ بارگاھ الٰهي ۾ اُمّت جي غم بابت اظهار ڪيو ته اللہ پاڪ رسولِ بيمثال،بي بي آمنه جي لال صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ۽ سندن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي پياري اُمت کي شبِ قدر جهڙي مبارڪ نعمت سان سرفراز فرمايو۔

ياد رهي! غمِ اُمّت جو اهو پهريون واقعو ناهي، پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي پنهنجي اُمت سان ايتري محبت آهي جو ڪيئي موقعن تي اُمت جي گناهگارن کي ياد فرمايو، جيئن دنيا ۾ جَلوه اَفروز ٿيندي ئي حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سَجدو فرمايو ۽ مبارڪ لبن تي اها دُعا جاري جئي: رَبِّ ھَبْ لِي اُمَّتِي يعني پَروَردَگار! منهنجي اُمَّت منهنجي حوالي فرماءِ۔ (فتاويٰ رضويه،۳۰/۷۱۲)

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ

کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([1])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([4])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([5])

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، اللہ عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
