حضرتِ سيدُنا عبدُ اللہ اِبنِ عباس فرمائن ٿا: بني سُليم مان هڪ شخص بارگاھِ رسالتِ مآب ۾ اچي گستاخي ڪئي: ته حضور نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي ارشاد فرمايو: تون آخِرت جي عذاب کان ڊڄ، دوزخ کان خوف رک، خُدا وَحدَه لاَشَرِيڪ جي عبادت ڪر ۽ مان اللہ ڪريم جو ٻانهو ۽ ان جو رسول آهيان، رسول اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي حُسنِ اَخلاق ۽ مؤثر ڪلام کان متاثر

حضرتِ سيدُنا سعد بن عباده رَضِىَ اللهُ تَعَالٰى عَنْهُ پنهنجي ڏاچي ان کي ڏئي ڇڏي۔ پوءِ ارشاد فرمايائون: ڪير آهي جيڪو هن جو مٿو ڍڪي؟ ته اميرالمؤمنين موليٰ مشڪل ڪشا، حضرتِ سيدُنا علي المرتضي کرَّمَ اللّٰه تَعَالٰي وَجْهه الْکرِيم پنهنجو عمامو ان کي ڏئي ڇڏيو۔ حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن ارشاد فرمايو: ڪير آهي جيڪو هن جي لاءِ کاڌي جو اِنتِظام ڪري؟ حضرتِ سيدُنا سلمان فارسي رَضِىَ اللهُ تَعَالٰى عَنْهُ اٿيا ۽ ڪجھ گھرن ڏانهن ويا پر ڪجھ نه مليو۔ پوءِ نُورِ مصطفٰي حضرتِ سيدَتُنا فاطمة الزهره رَضِىَ اللهُ تَعَالٰى عَنْهَا جي گھر تي حاضر ٿي دروازو کڙڪايو۔ پاڻ پڇيائون ڪير آهي؟ عرض ڪيائون! سلمان آهيان۔ فرمايائون: ڪيئن اچڻ ٿيو ؟ پاڻ سڄو واقعو ٻڌايو۔ اهو ٻڌي پاڻ غمغين ٿي ويون ۽ فرمايائون: اي سلمان! خُدا جو قسم! جنهن منهنجي پيءُ کي رسول بڻائي موڪليو! اڄ ٽيون ڏينهن آهي، گھر ۾ سڀئي فاقي سان آهن پر دروازي تي آيا آهيو، خالي ڪيئن موٽايان؟ هي چادر کڻي وڃو ۽ شمعون نالي غيرمسلم وٽ وڃي چئو: فاطمه بنتِ محمد جي چادر رک ۽ ڪجھ جَو قرض ڏيو۔ حضرتِ سيدُنا سلمان فارسي رَضِىَ اللهُ تَعَالٰى عَنْهُ ان چادر کي کڻي ان وٽ ويا ۽ سمورو واقعو بيان ڪري ٻڌايو۔ شمعون ڪجھ دير ان مبارڪ چادر کي ڏسندو رهيو، ان وقت ان تي هڪ ڪيفيت طاري ٿي وئي ۽ چوڻ لڳو: اي سلمان! هي اهي ئي مقدَّس ماڻهو آهن جن جي خبر اللہ پاڪ پنهنجي نبي موسيٰ عَلَيه السَّلَام کي توريت ۾ ڏني آهي، مان سچي دل سان حضرتِ سيدَتُنا فاطمه جي پيءُ مُحمَدٌ رَّسُولُ اللہ تي ايمان آڻيان ٿو۔

هي چئي ڪري ان ڪلمو پڙھيو ۽ مسلمان ٿي ويو ان کان پوءِ ان حضرتِ سيدُنا سلمان رَضِىَ اللهُ تَعَالٰى عَنْهُ کي جَو ڏنا ۽ نهايت ادَب واِحتِرام سان چادر مبارَڪ موٽائي ڏني۔ خاتونِ جنّت شمعون کي دُعاءِ خير ڏني ۽ جَوَ پيهي کاڌو تيار ڪري حضرتِ سيدُناسلمان فارسي رَضِىَ اللهُ تَعَالٰى عَنْهُ کي ڏئي ڇڏيو۔ حضرت سيدُناسلمان فارسي رَضِىَ اللهُ تَعَالٰى عَنْهُ عرض ڪيو: هن مان ڪجھ گھر لاءِ رکي ڇڏيو۔ فرمايائون: بس راھ خدا ۾ ڏيڻ جي نيت سان گھرايو ۽ پچايو آهي، هاڻي هن مان وٺڻ دُرُست نه آهي۔

مٺا مٺا اسلامي ڀائرو! بَيان کي اِختِتام ڏانهن آڻيندي اچو! سيد سڳورن جي تعظيم جي باري ۾ ڪجھ مدني گل ٻڌڻ جي سعادت حاصل ڪيون ٿا۔ پهريان 2 فرمان مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحظه ڪريو۔ (1) فرمايا ئون: جيڪو منهنجي اَهلِ بَيت مان ڪنهن سان به سٺو سُلوڪ ڪندو، مان قِيامت جي ڏينهن ان جو صِلو (بَدلو) ان کي عطا ڪندس۔ (جامع صغير، ص۵۳۳، حديث:۸۸۲۱) (2) فرمايائون: جيڪو شخص اَولادِ عَبدُ المُطَّلِب مان ڪنهن سان دُنيا ۾ نيڪي (ڀلائي) ڪري ان جو صِلو ڏيڻ مون تي لازِم آهي جڏهن اهو قِيامت جي ڏينهن مون سان ملندو۔ (تاريخ بغداد،۱۰/ ۱۰۲،حديث:۵۲۲۱) ٭ سيد سڳورن جي تعظيم فرض آهي ۽ اُنهن جي توهين حرام۔ (کفريه کلمات کے بارے ميں سوال جواب ،ص۲۷۷) ٭ سيد سڳورن جي تعظيم ۽ تڪريم جو اَصل سبب هي آهي ته هي حضرات رسُول ڪائنات صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي جسمِ اَطهر جو ٽڪر آهن۔ (سادات کرام کی عظمت، ص۷) ٭ سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي تعظيم وتوقير مان هي به آهي ته اهي تمام شيون جيڪي حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان نسبت رکن ٿيون انهن جي تعظيم ڪئي وڃي۔ (اَلشِّفا، الباب الثالث في تعظيم امره،فصل ومن اِعظامهٖ...الخ،ص۵۲،الجزء ۲) (سادات کرام کی عظمت، ص۸) ٭ تعظيم جي لاءِ نه يقين دَرڪار آهي ۽ نه ئي ڪا خاص سند جي حاجت تنهنڪري جيڪي ماڻهو سيد سڏرائين ٿا انهن جي تعظيم ڪرڻ گھرجي۔ (سادات کرام کی عظمت ص۱۴) ٭ جيڪو واقعي ۾ سيد نه هجي ۽ ڄاڻي واڻي سيد بڻجندو هجي اهو ملعون (لعنت ڪيل) آهي، نه ان جو فرض قبول ٿئي نه نفل ۔(سادات کرام کی عظمت، ص۱۶) ٭ جيڪڏهن ڪو بد مذهب سيد هئڻ جي دعويٰ ڪري ۽ ان جي بد مذهبي حدِّ ڪفر تائين پهچي چڪي هجي ته هرگز ان جي تعظيم نه ڪئي وڃي۔ (سادات کرام کی عظمت، ص۱۷) ٭ سيدن جي تعظيم حضور اقدس صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي تعظيم آهي۔ (فتاويٰ رضويه،٢٢/ ٤٢٣، ماخوذا) (سادات کرام کی عظمت، ص۸) ٭ استاد به سيد کي مارڻ کان پرهيز ڪري۔ (کفريه کلمات کے بارے میں سوال جواب،ص۲۸۴) ٭ سيد سڳورن کي اهڙي ڪم

تي مُلازم رکي سگھجي ٿو جنهن ۾ ذِلَّت نه هجي البت ذِلَّت وارن ڪمن ۾

انهن کي ملازم رکڻ جائز نه آهي۔ (سادات کرام کی عظمت، ص۱۲) ٭ سيد جي بطورِ سيد يعني هو سيد آهي ان ڪري توهين ڪرڻ ڪفر آهي۔ (کفريه کلمات کے بارے ميں سوال جواب ص۲۷۶)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا

ويندا آهن. ([1])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([2])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([3])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([4])

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، اللہ عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)
