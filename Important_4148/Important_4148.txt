اَلْحَمْدُ لِلّٰہِ رَبِّ الْعٰلَمِیْنَ وَ الصَّلٰوۃُ وَالسَّلَامُ عَلٰی سَیِّدِ الْمُرْسَلِیْنَ ط

فيضانِ صحابه ۽ اهل بيت

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

دُرُود شريف جي فضيلت

نبي ڪريم، روؤفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ عاليشان آهي: جنهن ڏينهن ۽ رات ۾ مون ڏانهن شَوق ۽ مَحَبَّت جي ڪري ٽي ٽي ڀيرا دُرُودِ پاڪ پڙھيو الله پاڪ تي حق آهي ته هو ان جي ان ڏينهن ۽ اُن رات جا گُناھ بخشي ڇڏي۔ (مُعْجَم ڪبير، ۱۸ / ۳۶۲ ،حديث: ۹۲۸)

کعبہ کے بدرُالدُّجیٰ تم پہ کروروں دُرُود طیبہ کے شمسُ الضُّحٰی تم پہ کروروں دُرُود

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿيون. فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِه ٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندَس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندَس ٭ ضرورت پوڻ تي ٻين اسلامي ڀينرن جي لاءِ سِرڪي جاءِ ڪشاده ڪندَس ٭

ڌِڪو وغيره لڳو ته صبر ڪندَس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندَس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ واريءَ جي دِل جوئي جي لاءِ آهسته آواز سان جواب ڏيندَس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندَس ۔ دوران بيان موبائل جي غيرضروري استعمال کان بچندَس،نه بيان ريڪارڊ ڪندَس نه ئي ٻئي ڪنهن قسم جو آواز ڇو ته ان جي اجازت ناهي،جيڪو ڪجهه ٻُدَندَس،ان کي ٻڌي ۽ سمجهي ڪري ان تي عمل ڪرڻ ۽ ان کي بعد ۾ ٻين تائين پهچائي نيڪي جي دعوت عام ڪرڻ جي سعادت حاصل ڪندَس.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! اڄ جي بيان ۾ صحابَه ڪرام ۽ اهل بيت عظام عَلَيۡهِمُ الرِّضْوَان جي مقام ۽ مرتبي ۽ انهن جي شان ۽ عظمت ٻڌڻ جي سعادت حاصل ڪنديون سين۔ پهريان صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان جي شان بابت قرآنِ پاڪ جي آيت ۽ ڪجھ حديث سڳوريون بيان ڪيون وينديون۔ ان کان پوءِ اهلِ بيت ڪرام عَلَيۡهِمُ الرِّضْوَان جي محبت جي اهميت ۽ ان جي متعلق حديث سڳوريون ٻڌنديون سين۔ صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان نبي پاڪ عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام جي آل ۽ اولاد سان ڪهڙي محبت ڪندا هئا، ان جا ڪجھ واقعا به بيان ڪيا ويندا۔ آخر ۾ شهزاده ڪونين، حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه جو تعارف به بيان ڪيو ويندو۔ الله ڪري ته اسان سڄو بيان توجهه سان ٻڌڻ جي سعادت حاصل ڪريون۔ ڪجهه اسلامي ڀينرون دورانِ بيان تسبيح جي ذريعي ذِڪر و درود شريف وغيره پڙھنديون رهنديون آهن، ياد رکو! هي ان جو موقعو نه آهي، اسان جيئن ته علمِ دِين سکڻ جي نيت سان هتي گڏ ٿيون آهيون ته دورانِ بيان سڄو ڌيان بيان ڏانهن ئي هئڻ گھرجي، ائين علمِ دين سکڻ تي مشتمل بيان ٻڌڻ به ذِڪرُ الله ئي ۾ شامل آهي۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

فاروقِ اعظم جي اصحابِ ڪهف سان ملاقات

دعوتِ اسلامي جي اِشاعتي اِداري مڪتبة المدينه جي ٻن جلدن تي مشتمل ڪتاب ”فيضانِ فاروقِ اعظم “ جلد 1 صفحي 304 تي آهي: هڪ ڏينهن الله پاڪ جي محبوب، داناءِ غيوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ رب ڪريم جي بارگاھ ۾ اَصحاب ڪهف سان ملاقات جي آرزو ڪئي ته انهيءَ وقت حضرت سيدُنا جبريل امين عَلَيۡهِ السَّلَام حاضر ٿيا ۽ بارگاھ رسالت ۾ عرض ڪيو،” يارَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ! اوهان انهن کي دنيا ۾ ظاهراً نه ڏسي سگھندو، البت پنهنجي صحابه مان چئن صحابين کي انهن وٽ موڪلي ڇڏيو ته جيئن اهي اوهان جو پيغام اُنهن تائين پهچائن ۽ انهن کي پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تي ايمان آڻڻ جي دعوت ڏين۔ “ هن طرح موڪلجو جو اوهان پنهنجي چادر (Shawl) مبارڪ وڇايو! هڪ پاسي حضرت ابوبڪر صديق رَضِىَ اللهُ تَعَالٰى عَنْه ، ٻي پاسي حضرت عمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰى عَنْه ، ٽئين پاسي حضرت علي المرتضي شير خدا کرَّمَ اللهُ تَعَالٰى وَجْهه الْکرِيم ۽ چوٿين پاسي حضرت ابو ذر غفاري رَضِىَ اللهُ تَعَالٰى عَنْه کي ويهاري ڇڏيو۔ پوءِ ان هوا کي سڏيو جيڪا حضرت سليمان عَلَيۡهِ السَّلَام جي تابع هئي ۽ ان کي حڪم ڪريو ته انهن چئن ئي صحابه ڪرام رَضِىَ اللهُ تَعَالٰى عَنْهم اجمعين کي کڻي ۽ ان غار تائين کڻي وڃي، جتي اصحاب ڪهف آرام فرمائي رهيا آهن۔ “ پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ائين ئي فرمايو، هوا ان چادر جنهن تي چار ئي صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان آرام ۽ سڪون سان ويٺل هئا ان کي کنيو ۽ اصحابِ ڪهف جي غار وٽ چادر کي زمين تي رکي ڇڏيو۔ صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان غار جي ويجھو پهچي ان جي سوراخ تان پٿر کنيو، جيئن ئي روشني اندر پهتي ته اَصحاب ڪهف جي ڪتي هلڪو آواز ڪڍيو ۽ حملو ڪرڻ جي لاءِ ٻاهر نڪتو پر جڏهن ان رسولِ پاڪ، سرورِ ڪائنات صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صحابه کي ڏٺو ته انهن جي قدمن کي چمڻ لڳو ۽ پنهنجو پڇ هِلائڻ لڳو ۽ پوءِ مٿي جي اشاري سان اندر اچڻ جو چيائين۔ چار

ئي صحابَه ڪرام رَضِىَ اللهُ تَعَالٰى عَنْهم اجمعين غار جي اندر ويا ۽ سُتل اصحابِ ڪهف کي هن طرح سلام ڪيائون: ” اَلسَّلَامُ عَلَيکم وَرَحْمَة الله وَبَرَکاتُه ۔ “ الله پاڪ پنهنجي ڪرم سان اصحابِ ڪهف کي بيدار فرمايو ۽ انهن به جواب ۾ سلام ڪيو۔ چئن ئي صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان پنهنجو تعارف ڪرايو ۽ سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو سلام پهچايو۔ انهن سلام جو جواب ڏنو ۽ نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تي ايمان آندائون، دينِ اسلام کي قبول (Accept) ڪيائون ۽ عرض ڪيائون: ’’ اسان جي طرفان پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي بارگاھ ۾ سلام پيش ڪجو۔ “ پوءِ اهي پنهنجي پنهنجي جڳهن تي ٻيهر ليٽي پيا۔ اهي چار ئي صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان ٻيهر چادر تي ويٺا ۽ هوا انهن کي بارگاھ رسالت ۾ پهچائڻ جي لاءِ چادر کي کڻي نڪري پئي۔

هوڏانهن حضرت سيدُنا جبريل امين عَلَيۡهِ السَّلَام سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ وٽ حاضر ٿي ويا ۽ انهن چئن ئي صحابه ڪرام عَلَيۡهِمُ الرِّضْوَان سان جيڪو ڪجھ ٿيو سڀ بيان ڪري ڇڏيائون۔ ۽ جڏهن چار ئي صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان بارگاھ رسالت ۾ حاضر ٿيا ته انهن به پورو واقعو بيان ڪيو ۽ اصحابِ ڪهف جو سلام به پهچايو، اهو ٻڌي نور جا پيڪر، تمام نبين جا سَروَر صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ گھڻو خوش ٿيا، پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ دعا جي لاءِ هٿ کنيا ۽ بارگاھ الٰهي ۾ هن طرح دعا گھريائون: ”اي رب ڪريم! منهنجي، منهنجي رشتيدارن، منهنجن دوستن ۽ مون سان محبت ڪرڻ وارن جي وچ ۾ ڪڏهن به جدائي نه وجھجان ۽ جيڪو مون سان، منهنجي اهلِ بيت سان ۽ منهنجي صحابن سان محبت ڪري ٿو انهن سڀني جي بخشش فرماء۔

(تفسير ثعلبي ،پ ۱۵ ،الڪھف،تحت الآية: ۱۶ ، ۶ / ۱۵۶ ،ملتقطاً و ملخصاًوفيضان فاروق اعظم، ۱ / ۳۰۴ ملخصاً)

پياريون پياريون اسلامي ڀينرون! بيان ڪيل حڪايت ۾ نهايت ئي خوشبودار مدني گل اسان جي ذهنن کي تازگي بخشي رهيا آهن، اچو! اسان به ڪجھ مدني گلن کي پنهنجي دامن ۾ گڏ ڪرڻ جي ڪوشش ڪريون ٿيون.

(1) الله ڪريم نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي مالڪ ۽ مختار بڻايو بلڪ

اهڙي حڪمراني، بادشاهي ۽ شهنشاهي عطا فرمائي آهي جو پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ڪائنات جي هر هر شيء تي الله جي حڪم سان حڪومت ڪن ٿا۔ جيئن جڏهن پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ هوا کي حڪم ارشاد فرمايو ته اها سندن حڪم مڃندي صحابَه ڪرام کي اصحاب ڪهف جي غار(Cave) تائين کڻي وئي، يقيناً اهو سندن هڪ زبردست معجزو آهي ته هوائن تي به رسول اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي حڪمراني جو سڪو قائم آهي۔

خالقِ کُل نے آپ کو مالکِ کُل بنادیا دونوں جہاں ہیں آپ کے قبضہ و اختیار میں

(2) تاجدارِ مدينه، قرارِ قلب و سينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صحابه ڪرام عَلَيۡهِمُ الرِّضْوَان جو شان هي آهي ته انسان ته انسان بي زبان جانور به انهن کي سڃاڻن ٿا ۽ انهن جي تعظيم ۽ عزت ڪن ٿا، جيئن غار ۾ موجود اصحابِ ڪهف جي ڪتي جڏهن پاڻ سڳورن عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام جي صحابين کي ڏٺو ته ان جي مبارڪ قدمن کي چمڻ لڳو، پنهنجو پڇ (Tail) هِلائڻ لڳو ۽ پوءِ مٿي جي اشاري سان اندر اچڻ جو چيائين۔ ڪنهن شاعر ڇا خوب چيو آهي ته

منقول آهي ته حَضرَتِ سَيِّدُنَا اَبُوبَڪرصِدّيق رَضِیَ اللهُ تَعَالٰی عَنْهُ جڏهن اَمِيرُ الۡمُؤمِنِين، خَلِيفَةُ الۡمُسۡلِمِين مُنتَخَب ٿيا ته رسولِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان تَعَلُّق جي سبب پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهُ اہلِ بيتِ اطہار جو تمام گھڻو خَيال رکندا هئا ۽ اَہۡلِ بيتِ اَطہار جي باري ۾ فرمائيندا هئا ته ”نبي ڪريم، رءُوفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا رشتيدار مون کي پنهنجي رشتيدارن کان وڌيڪ عَزِيز آهن۔ “

پياريون پياريون اسلامي ڀينرون! غور ڪريو! اَمِيرُالمؤمنين حَضرتِ سَيدُنا صديقِ اڪبر رَضِىَ اللهُ تَعَالٰى عَنْه کي اهلِ بيت ڪرام سان ڪيتري محبت هئي جو انهن کي پنهنجي گھروارن تي ترجيح ڏيندا هئا۔ اچو صحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان جي اهلِ بيتِ پاڪ سان محبت جا وڌيڪ ڪجھ واقعا ٻڌون ٿيون:

-1 حضرت سيدنا اميرِمعاويه رَضِىَ اللهُ تَعَالٰى عَنْه جي بارگاھ ۾ حضرت سيدنا علي المرتضيٰ رَضِىَ اللهُ تَعَالٰى عَنْه جو تذڪرو ٿيو ته پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه فرمايائون: خدا جو قسم! جڏهن علي المرتضيٰ رَضِىَ اللهُ تَعَالٰى عَنْه گفتگو ڪندا هئا ته سندن آواز ۾ شينهن جيان گجگوڙ (Thunder) هوندي هئي، جڏهن ظاهر ٿيندا هئا ته چنڊ جيان روشن هوندا هئا ۽ جڏهن نوازيندا هئا ته برسات جيان بي انتها عطا ڪندا هئا، ڪجهه حاضرين پڇيو ته اوهان افضل آهيو يا حضرت سيدنا علي المرتضيٰ رَضِىَ اللهُ تَعَالٰى عَنْه ؟ فرمايائون: ’’حضرت سيدنا علي رَضِىَ اللهُ تَعَالٰى عَنْه جا ڪجھ نقوش به آلِ ابوسفيان کان بهتر آهن۔ پوءِ فرمايائون: جيڪو شخص حضرت سيدنا علي المرتضيٰ رَضِىَ اللهُ تَعَالٰى عَنْه جي ساراھ ۾ سندن شان لائق شعر ٻڌاءِ مان ان کي هر شعر جي بدلي هزار دينار انعام ڏيندس۔ “ پوءِ حاضرين شعر ٻڌايا، حضرت سيدنا اميرِ معاويه رَضِىَ اللهُ تَعَالٰى عَنْه فرمائن ٿا: انهن شعرن ۾ جيڪي اوهان ’’اميرالمؤمنين حضرت علي المرتضيٰ رَضِىَ اللهُ تَعَالٰى عَنْه جي شان ۽ عظمت بيان ڪئي پاڻ ان کان به وڌيڪ افضل آهن۔ “ پوءِ حضرت سيدنا عَمرو بن عاص رَضِىَ اللهُ تَعَالٰى عَنْهما حضرت علي المرتضي کَرَّمَ اللّٰه تَعَالٰى وَجْهه الْکرِيم جي شان ۽ عظمت ۾ شعر پڙهيا۔ (الناھية ،فصل تابع في فضائل معاوية، ص ۵۹ م ختصرًا)

-2 منقول آهي ته ڪنهن جنازي ۾ نواسه رسول حضرت سيدنا امام حسين ۽ مشهور صحابي رسول سيدنا ابو هريره رَضِىَ اللهُ تَعَالٰى عَنْه گڏ هئا۔ اتان کان واپسي تي حضرت سيدنا امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه کي ٿڪاوٽ (Weariness) محسوس ٿي ته هڪ جاءِ تي پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه آرام جي لاءِ ڪجھ دير ويهي رهيا۔ حضرت سيدنا ابوهريره رَضِىَ اللهُ تَعَالٰى عَنْه پنهنجي چادر کان حضرت سيدنا امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه جي

پيرن تان گرد و غبار صاف ڪرڻ لڳا ته حضرت سيدنا امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه انهن کي منع فرمايو۔ ان تي حضرت سيدنا ابوهريره رَضِىَ اللهُ تَعَالٰى عَنْه عرض ڪيو: فَوَاللّٰهِ لَوْ يَعْلَمُ النَّاسُ مِنْكَ مَا اَعْلَمُ لَحَمَلُوْكَ عَلَي رِقَابِهِمْ يعني الله پاڪ جو قسم! اوهان جي جيڪا عظمت مون کي خبر آهي جيڪڏهن ماڻهن کي خبر پئجي وڃي ته اهي اوهان کي پنهنجي ڪلهن تي کڻن ۔ (تاريخ ابن عساڪر ، ۱۴ / ۱۷۹ مختصراً)

-3 حضرت امامِ حسن رَضِىَ اللهُ تَعَالٰى عَنْه فرمائن ٿا ته هڪ ڀيري اَمِيرُالمؤمنين حَضرتِ سَيدُنا عمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰى عَنْه مون کي فرمايو: اي منهنجا پٽ! منهنجي تمنا آهي ته اوهان اسان وٽ ايندا ڪريو۔ پوءِ مان هڪ ڏينهن سندن گھر ويس پر پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه حضرت امير معاويه رَضِىَ اللهُ تَعَالٰى عَنْه سان عليحدگي ۾ گفتگو ۾ مصروف هئا ۽ سندن پٽ حضرت عبدالله بن عمر رَضِىَ اللهُ تَعَالٰى عَنْه دروازي تي بيهي انتظار ڪري رهيا هئا۔ ڪجھ دير انتظار کان بعد جڏهن اهي واپس موٽڻ لڳا ته مان به موٽي آيس۔ بعد ۾ امير المؤمنين حضرت سيدُنا عمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰى عَنْه سان منهنجي ملاقات ٿي ته پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه فرمايائون: ” لَمْ اَرَكَ يعني اوهان اسان وٽ ٻيهر آيا ئي نه؟ “ مون عرض ڪيو: ’’اي اميرالمومنين! مان ته آيو هئس پر اوهان حضرت امير معاويه رَضِىَ اللهُ تَعَالٰى عَنْه سان گفتگو ۾ مصروف هئا۔ اوهان جا پٽ عبدالله بن عمر رَضِىَ اللهُ تَعَالٰى عَنْه به ٻاهر بيهي انتظار ڪري رهيا هئا (مون سوچيو جڏهن پٽ کي اندر وڃڻ جي اجازت نه آهي مون کي ڪيئن ٿي سگھي ٿي) لهٰذا مان ان سان گڏ ئي واپس موٽي ويس۔ “ پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه فرمايائون: ” اَنْتَ اَحَقُّ بِالْاِذْنِ مِنْ عَبْدِ الله ابْنِ عُمَرَ اِنَّمَا اَنْبَتَ فِيْ رُؤُوْسِنَاالله ثُمَّ اَنْتُم يعني اوهان منهنجي اولاد کان وڌيڪ هن ڳالھ جا حق دار آهيو ته اندر اچو۔ اسان جي مٿي تي هي جيڪي وار آهن، الله پاڪ کان بعد توهان ئي ته پوکيا آهن۔“ (تاريخ ابن عساڪر ، ۱۴ / ۱۷۶ ، رياض النضره ، ۱ / ۳۴۱)

وَسَلَّمَ سندن ساڄي ڪن ۾ اذان چئي ۽ کاٻي ڪن ۾ تڪبير چئي ۽ پنهنجو لعاب مبارڪ سندن وات ۾ وڌو۔ (ڪنزالعمال، ۸ / ۲۵۲ ، جزء ۱۶ ، حديث: ۴۵۹۹۳ ،ملتقطا) ٭ پاڻ عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه کي دنيا ۾ پنهنجو گل قرار ڏنو۔ (بخاري،ڪتاب فضائل اصحاب النبي،باب مناقب الحسن والحسين، ۲ / ۵۴۷ ، حديث: ۳۷۵۳) ٭ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه پنهنجي وڏي ڀاءُ حضرت امام حسن رَضِىَ اللهُ تَعَالٰى عَنْه وانگر جنتي جوانن جا سردار آهن۔ (سنن الترمذي،ڪتاب المناقب، باب مناقب ابي محمد الحسن بن علي ... الخ، ۵ / ۴۲۶ ، حديث: ۳۷۹۳) ٭ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه (25) حج پيدل ڪيا هئا۔ (تاريخ مدينه دمشق ، ۱۴ / ۱۸۰) ٭ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه علم جي شهر جنابِ رسولِ خدا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ۽ دروازهِ شهرِ علم مصطفيٰ، اميرُ المؤمنين حضرتِ سيدُنا مَوليٰ علي مشڪل ڪشا کرَّمَ اللهُ تَعَالٰى وَجْهه الْکرِيم کان علمِ دين جو قيمتي خزانو ماڻيو۔ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه جي علم سان ڀرپور گُفتگو اهڙي دلڪش هوندي هئي جو ماڻهن جي اها خواهش هوندي هئي ته پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه چپ نه رهن بلڪ عِلم ۽ حِڪمت جا مَدَني گل لُٽائيندا ئي رهن۔ ٭ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه پنهنجي جدِ ڪريم، رؤف ورحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کان ۽ پنهنجي والدِ محترم، والده مُحترمه ۽ اميرُالمؤمنين حضرتِ سيدُنا عمرِ فاروق عَلَيۡهِمُ الرِّضْوَان کان حديثون ٻڌيون ۽ رِوَايت ڪيون۔ ٭ حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه کان رِوَايت ڪرڻ ۾ سندن ڀاءُ امام حسن، سندن شهزادا امام زينُ العابدين، سندن صاحِبزاديون، پوٽو امام باقر عَلَيۡهِمُ الرِّضْوَان سميت ٻيا عالم ۽ مُحَدِّث شامل آهن۔ (الاصابه ، ۲ / ۶۸) حضرت امام حسين رَضِىَ اللهُ تَعَالٰى عَنْه جو مُستقل علمي حلقو مسجدِ نَبوي ۾ لڳل هوندو هو جنهن ۾ پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه ماڻهن کي فِقهي مسئلا ٻڌائيندا هئا۔ (تاريخ دمشق، ۱۴ / ۱۷۹ مفھوماً) الله پاڪ اسان کي سندن نقشِ قدم تي هلڻ ۽ پاڻ رَضِىَ اللهُ تَعَالٰى عَنْه جي شهادت جي ڏينهن جي موقعي تي انهن کي خوب ايصالِ ثواب ڪرڻ جي توفيق عطا فرمائي۔

مجلس رابطه براءِ عالمات

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوت اسلامي سڄي دنيا ۾ 200 کان وڌيڪ ملڪن ۾ ڪم و پيش 104 شعبه جات ۾ نيڪي جي دعوت جون ڌومون مچائڻ ۾ مصروفِ عمل آهي. انهيء شعبن مان هڪ شعبو ”مجلسِ رابطه براءِ عالمات) به آهي. هن شعبي جو بنيادي مقصد سنّي عالمات کي دعوت اسلامي جي ديني خدمتن کان آگاهه ڪرڻ، مدني ڪمن ۾ انهن جي مدد حاصل ڪرڻ، انهن جون دعائون وٺڻ، سنّي جامعات ۽ مدرسن ۾ مدني ڪمن جي ترڪيب بڻائڻ، هفته وارا سنتن ڀري اجتماع ۾ سنّي جامعات ۽ مدرسن جي طالبات جي شرڪت ڪرائڻ آهي. الله ڪريم دعوت اسلامي ۽ ”مجلس رابطه براءِ عالمات “ کي وڌيڪ ترقيون عطا فرمائي ۽ اسان کي دعوت اسلامي جي مدني ماحول ۾ استقامت نصيب فرمائي . اٰمِــيْن بِـجَـاهِ الـنَّـبِـیِّ الْاَمِــيْن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم
