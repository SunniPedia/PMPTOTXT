پيارا پيارا اسلامي ڀائرو! معلوم ٿيو جنهن کي روئڻ نه اچي ان کي روئڻ جي ڪوشش ڪرڻ گھرجي، ڪڏهن ڪڏهن گناهن جي شدّت ۽ دل جي سختي جي ڪري لڙڪ سڪي ويندا آهن۔ ان سختي کي دور ڪرڻ جو هڪ طريقو اهو به آهي ته بک ۽ نفلي روزن جي ڪثرت ڪئي وڃي۔ هن سان دل نرم ٿيندي ۽ خوفِ خدا ۾ روئڻ جي سعادت حاصل ٿي سگھندي۔ زهي قسمت جو رجب جي آمد آمد آهي۔ هن مهيني جي روزن جون وڏيون برڪتون آهن۔ لهٰذا جن کان ٿي سگهي پڪي نيت ڪريو ۽ هن مهيني جا نفلي روزا رکڻ جي سعادت حاصل ڪيو. هن سان نه صرف دل جي سختي دور ٿيندي بلڪ رجب جون برڪتون به نصيب ٿينديون۔ رجب ۽ شعبان جي نفلي روزن جي برڪت سان رمضان جي مهيني جا فرض روزا رکڻ ۾ آساني ٿيندي ۽ ائين فرض روزن جي لاءِ اڳواٽ تياري به ٿي ويندي.

منقول آهي ته ڇهين آسمان جي دروازي تي رجب جي مهيني جا ڏينهن لکيل آهن۔ جيڪڏهن ڪو شخص رجَب ۾ هڪ روزو رکي ۽ ان کي پرهيزگاري سان پورو ڪري ته اهو دروازو ۽ اهو (روزي وارو) ڏينهن هن ٻانهي جي لاءِ الله پاڪ کان مغفِرت طلب ڪندا ۽ عرض ڪندا: ياالله! هن ٻانهي کي بخشي ڇڏ۔ جڏهن ته الله پاڪ جي حبيب، هر عيب کان پاڪ نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ ذِيشان آهي: رجب ۾ هڪ ڏينهن ۽ رات آهي جيڪو ان ڏينهن روزو رکي ۽ رات جو قيام (عبادت) ڪري ته ڄڻ ان سؤ سال جا روزا رکيا ۽ سؤ سال جي شب بيداري ڪئي، اها رجب جي 27 تاريخ آهي. (شُعَبُ الْاِيمَان، ۳/۳۷۴، حديث:۳۸۱۱)

پيارا پيارا اسلامي ڀائرو! رجب المرجب جي روزا رکڻ وارن جا ڀاڳ ئي کلي ويندا آهن. رجب المرجب جا روزا رکڻ وارن جي لاءِ الله پاڪ جنت ۾ خاص محل تيار ڪيو آهي، انهن خوش نصيبن کي الله پاڪ ”رجب“ نامي نهر مان سيراب ڪندو، انهن جي لاءِ جهنم جا در بند ۽ جنت جا در کوليا ويندا آهن، انهن جا روزا گناهن جو ڪفارو بڻجي ويندا ۽ محشر واري ڏينهن جي برداش نه ٿيندڙ گرمي ۽ بُک اُڃ ۾ انهن جي لاءِ کائڻ پيئڻ ۽ آرام ڪرڻ جو بندوبست ڪيو ويندو. نفلي روزن جي ايتري قدر زبردست فضيلتون ۽ برڪتون ٻڌڻ کانپوءِ ته اسان مان هر هڪ کي گهرجي ته فرض روزن سان گڏوگڏ نفلي روزن جو بڪثرت اهتمام ڪندا ڪيون، ماھ رجب المرجب جي اچڻ سان هونئن ئي روزن رکڻ جو ڄڻ موسم شروع ٿي ويندو آهي. پهريان رجب جا روزا ان کانپوءِ ماھ شعبان جا روزا.

اچو شيخ طريقت امير اهلسنت جو نفلي روزن جي ترغيب ۽ فضائلن تي مشتمل هڪ خوبصورت مڪتوب ٻڌون ٿا:

مڪتوب عطار

بِسۡمِ اللهِ الرَّحمٰنِ الرَّحِيمِ ط سگِ مدينه محمّد الياس عطار قادري رضوي عُفِي عَنه جي

طرفان، تمام اسلامي ڀائرن ، اسلامي ڀينرن ، مدارس المدينه ۽ جامعاتُ المدينه جي اساتذه ، طَلَبه معلمات ۽ طالبات جي خدمات ۾ ڪعبهء مُشَـرّفه جي چوڌاري گھمندو،گنبذ خضريٰ کي چمندو رَجَبُ المُرَجب، شَعبانُ المُعَظم ۽ رمضانُ المُبارڪ جي برڪتن سان مالا مال جهومندو سلام،

ا َ لسَّلَامُ عَلَيْکُم وَرَحْمَةُ الله ِوَبَرَکاتُهٗ اَ لۡـحَـمۡدُ لِلهِ رَبِّ العٰلَـمِيۡنَ عَليٰ کُلِ حَال

ہو نہ ہو آج کچھ مرا ذکر حضور ميں ہوا

ورنہ مری طرف خوشی د يکھ کے مسکرائی کيوں

(حدائق بخشش)

اَلۡـحَمۡدُ لله عَزَّوَجَلَّ هڪ ڀيرو وري خوشين جا ڏينهن اچڻ لڳا رجب شريف جي مهيني جي آمد آمد آهي. هن مبارڪ مهيني ۾ عبادت جو ٻج پوکيو ويندو آهي، شَعبانُ المُعَظم ۾ شرمساري جي ڳوڙهن سان پاڻي ڏنو ويندو آهي ۽ رمضان المبارڪ جي مهيني ۾ رحمت جو فصل ڪٽيو ويندو آهي .

ٽن مهينن جا روزا

رجب المرجب جا قدر دانو! تعليم و تعلُّم (يعني سکڻ ۽ سيکارڻ) ۽ ڪسبِ حلال ۾ رڪاوٽ نه هجي، ماءُ پيءُ به بلا وجه منع نه ڪن، ڪنهن جي به حق تلفي نه ٿيندي هجي جلدي جلدي ۽ تمام جلدي مسلسل ٽن مهينن جا يا رمضان المبارڪ جي مڪمل فرض روزن سان گڏ جنهن کان جيترا ٿي سگهن نفل روزن جي لاءِ ڪمر بسته ٿي وڃو، سحري ۽ افطار ۾ گهٽ کائي ڪري پيٽ جو قفل مدينه به لڳايو. ڪاش! هر گهر ۾ ۽ منهنجي سڀني مدارس المدينه ۽ تمام جامعة المدينه ۾ روزن جون بهاريون اچي وڃن. بس پهرين رجب شريف کان ئي روزن جو آغاز فرمائيندا وڃو.

رجب جي ابتدائي ٽن روزن جي فضيلت

رجب شريف جي ابتدائي ٽن روزن جي فضائل جي به ڇا ڳالهه ڪجي!

حضرت سيّدنا عبدالله ابنِ عبّاس رَضِىَ اللهُ تَعَالٰی عَنْهُمَا کان روايت آهي ته بيچين دِلين جي چين، رحمتِ دارَين، تاجدارِ حَرمَين، سروَرِ ڪَونَين، ناناءِ حَسۡنَين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ رحمتِ نشان آهي: ” رجب جي پهرين ڏينهن جو روزو ٽن سالن جو ڪفارو آهي ۽ ٻي ڏينهن جو روزو ٻن سالن جو ۽ ٽين ڏينهن جو هڪ سال جو ڪفارو آهي، پوءِ هر ڏينهن جو روزو هڪ مهيني جو ڪفارو آهي.“ (الجامع الصغير للسيوطي ص 311 حديث 5051 فضائل شهر رجب للخلال ص 7)

ميں گنہگار گناہوں کے سوا کيا لاتا

نيکياں ہوتی ہیں سرکار نِکوکار کے پاس

نفلي روزن جون به ڇا نه خوب بهارون آهن هن ضِمن ۾ ٻه مبارڪ حديثون ملاحظه فرمايو ،

(1) فِـرشتا ڪڏهن دعاءِ مَغفِرت ڪندا آهن؟

حضرت سيّدَتُنا اُمِّ عَمَّارَه رَضِىَ اللهُ تَعَالٰی عَنْها فرمائن ٿيون: حُضُور پاڪ، صاحبِ لولاڪ، سَيَّاحِ اَفلاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مون وٽ تشريف وٺي آيا ته مون پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمت بابرڪت ۾ کاڌو پيش ڪيو ته ارشاد فرمايائون: ”تون به کاءُ“ مون عرض ڪيو: ”مان روزي سان آهيان“ ته رحمتِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: ”جيستائين روزيدار جي سامهون کاڌو کاڌو ويندو آهـي فِرشتا ان روزيدار جي لاءِ دعاءِ مَغفِرت ڪندا رهندا آهن .“ (سنن ترمذي ج 2 ص 205 حديث 785)

(2) روزيدار جون هڏيون ڪڏهن تسبيح ڪنديون آهن؟

حضرت سيّدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ نبيِّ اَڪرم، رحمتِ عالم، نُورِ مُجسم، شاھِ بني آدم، رسولِ مُحۡتَشَم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمتِ مُعَظم ۾ حاضر ٿيا، ان وقت حُضُور پُر نُور، شافعِ يومُ النُّشور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ناشتو ڪري رهيا هئا، فرمايائون: اي بلال! ناشتو ڪريو، عرض ڪيائون: يارسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! مان روزيدار آهيان، ته رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: اسان پنهنجي

روزي کائي رهيا آهيون ۽ بلال جو رزق جنّت ۾ وڌي رهيو آهي. اي بلال! ڇا توکي خبر آهي ته جيستائين روزيدار جي سامهون ڪجھه کاڌو وڃي تيستائين انهي جون هڏيون تسبيح ڪنديون آهن، ان کي فرشتا دعائون ڏيندا آهن .

(شعب الايمان ج 3 ص 297 حديث 3586)

مُفَسِّرِ شهير حڪيم الامت حضرت مفتي احمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا ته هن مان معلوم ٿيو ته جيڪڏهن کاڌو کائڻ دوران ڪو اچي وڃي ته ان کي به کائڻ جي لاءِ سڏڻ سنت آهي، پر دِلي ارادي سان سڏي، ڪوڙي صلاح نه ڪري ۽ اچڻ وارو به ڪوڙ ڳالهائي هيئن نه چوي ته مونکي خواهش ناهي، ته جيئن بک ۽ ڪوڙ گڏجي نه وڃن. بلڪ جيڪڏهن (نه کائڻ چاهي يا) کاڌو گھٽ ڏسي ته چـوي بَارَ ك َ اللهُ (يعـني الله عَزَّوَجَلَّ تو کي برڪت ڏي) هي به معلوم ٿيو ته سروَرِ ڪائنات، شاھِ موجودات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان پنهنجون عبادتون نه لڪائڻ گھرجن، بلڪ ظاهر ڪيون وڃن ته جيئن حضور پُرنُور، شافعِ يومُ النُّشُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان تي گواھ بڻجي وڃن، ۽ اهو ظاهر ڪرڻ رياء ناهي. (حضرت سيدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ جي روزي جو ٻڌي ڪري جيڪو ڪجھ فرمايو ويو ان جي شرح هي آهي) ته يعني اڄ جي روزِي اسان ته پنهنجي هتي ئي کائي وٺون ٿا ۽ حضرت سيّدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ ان جي عِوَض جنّت ۾ کائيندا، اهو عوض (يعني بدلو) بهتر به هوندو ۽ زياده به، حديث بلڪل پنهنجي ظاهري معنيٰ تي آهي، واقعي ان وقت روزيدار جي هر هڏي ۽ جوڙ بلڪ رڳ رڳ تَسۡبِيح (يعني الله عَزَّوَجَلَّ جي پاڪي بيان) ڪندي آهي جنهن جو روزيدار کي پتو ئي ناهي هوندو، پر سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌندا آهن. (مراٰة ج 3 ص 202)

جيڪڏهن مطالعو ڪيو هجي تڏهن به ٻئي رسالا (1) ڪـفـن جـي واپسـي ان سان گڏ رجب جون بهارون ۽ (2) آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو مهينو پڙهي ڇڏيو، ۽ پڻ هر سال شعبان المُعَظّم ۾ فيضان سنّت جو باب فـيـضـانِ رمـضـان ضرور پـڙهي ڇـڏيـنـدا ڪريو . ٿي سگهي ته عيدِ مِعراجُ النبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي نسبت سان

127 يا 27 رسالا يا حسب ِ توفيق فيضانِ رمضان به تقسيم فرمايو ۽ خوب خوب ثواب ڪمايو . تمام اسلامي ڀائرن سان بالعُمُوم ۽ جامعات المدينہ ۽ مدارس المدينہ جي سمورن قاري صاحبان، استادن، ناظمن ۽ طالبن جي خدمت ۾ بالخصوص تڙپندڙ مدني عرض آهي ته براءِ ڪرم! (منهنجي جيئري ۽ مرڻ کان بعد به) زڪوٰة، فطرو، قرباني جون کلون ۽ ٻيا مدني عطيات جمع ڪرڻ ۾ اڳتي وڌي ڪري حصو وٺندا ڪريو. (اسلامي ڀينرون ٻين اسلامي ڀينرن ۽ مَحارِم کي ترغيب ڏيارين) خدا جو قسم! مون کي انهن استادن ۽ شاگردن جي باري ۾ ٻڌي ڪري تمام گھڻي خوشي ٿيندي آهي جيڪي پنهنجي ڳوٺ يا شهر ۾ وڃڻ جي خواهش کي قربان ڪري رمضان المبارڪ جو مهينو جامعات ۾ گذاريندا آهن ۽ پنهنجي مَجلس جي هدايتن جي مطابق چندي جي بستن تي ذميداريون سنڀاليندا آهن. جيڪي استاد ۽ شاگرد بغير ڪنهن عذر جي صرف سُستي يا غفلت جي سبب عدمِ دلچسپي جو مظاهرو ڪندا آهن انهن جي سبب کان منهنجي دل ڏکندي آهي.

خصوصي مدني گُل: جيڪي به اسلامي ڀائر يا اسلامي ڀينرو چندو جمع ڪرڻ چاهن ٿا انهن کي چندي جا ضروري احڪام معلوم هجڻ فرض آهن، تاڪيد سان عرض آهي ته جيڪڏهن پڙهي چڪا آهيو تڏهن به دعوت اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل 107 صفحن تي مشتمل ڪتاب ”چندي جي باري ۾ سوال جواب“ جو ٻيهر مطالعو فرمايو. يا الله عَزَّوَجَلَّ رمضان المبارڪ ۾ چندي ۽ عيد الاضحيٰ ۾ کلن جي لاءِ ڪـوشش ڪري جيڪـي عاشقانِ رسول منهنجي دل خوش ڪندا آهن تون انهن کان هميشه هميشه جي لاءِ راضي ٿي وڃ ۽ انهن جي صدقي مون پاپي ۽ بدڪار، گناهگارن جي سردار کان به هميشه جي لاءِ راضي ٿي وڃ .

هميشه جي لاءِ خوش ٿي وڃ ۽ انهن جي صدقي مون گناهگار ۽ بدڪار، گناهن جي سرداري سان به هميشه جي لاءِ راضي ٿي وڃ، يا الله پاڪ جيڪو اسلامي ڀاءُ ۽ اسلامي ڀيڻ (عذر نه هجڻ جي صورت ۾) هر سال ٽن مهينن جا روزا رکي ۽ هر سال جَمادي الآخرٰي ۾ رسالو ڪفن جي واپسي ۽ رَجَبُ المُرَجَّب ۾ آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو مهينو ۽ شَعبان المُعَظّم ۾ فيضانِ رمضان پڙهي يا ٻڌي وٺڻ جي سعادت حاصل ڪري مون کي ۽ ان کي دنيا ۽ آخرت جون ڀلايون نصيب فرماءِ ۽ اسان کي بي حساب بخشي ڪري جنّت الفردوس ۾ پنهنجي مدني حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پاڙي ۾ گڏ رک. اٰمين بِجاهِ النَّبِيِّ الۡاَمين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

روئڻ جا طبي فائدا

پيارا پيارا اسلامي ڀائرو! اسان خوفِ خدا ۾ روئڻ جي اهميت جي باري ۾ ٻڌي رهيا هئاسين۔ جديد طبي تحقيقات مان ثابت ٿيو آهي ته روئڻ جا به ڪيئي فائدا آهن۔ اچو ڳوڙها ڳاڙڻ جا ڪجھ طبي فائدا ٻڌون ٿا: ٭ ماهرن مطابق اهو پاڻي جيڪو لڙڪن جي صورت ۾ اکين مان نڪرندو آهي، اک مان نڪرندڙ ٻين پاڻين (Waters) کان مختلف هوندو آهي۔ ٭ تحقيق مان معلوم ٿيو آهي ته هر شخص کي هفتي ۾ گھٽ ۾ گھٽ هڪ ڀيرو پندرهن (15)(Fifteen) منٽ تائين روئڻ گھرجي۔ ٭ هفتي ۾ هڪ ڀيرو روئڻ دماغي صلاحيتن تي سٺو اثر وجھندو آهي۔ ٭ ماهرن جو چوڻ آهي ته لڙڪ انساني جسم ۾ موجود ڪوليسٽرول (Cholesterol) کي گھٽ ڪندا آهن۔ ٭ ڳري طبيعت ۾ وهندڙ لڙڪ ذهني دٻاؤ جو خاتمو ڪندا آهن جنهن سان بلڊ پريشر (Blood pressure)، شوگر (Sugar) ۽ دل جو مرض (Heart diseases) لاحق ناهي ٿيندو۔ ٭ لڙڪن کي روڪڻ سان اکين ۾ ڊي هائيڊريشن ٿي ويندي آهي جنهن سان روشني ڪمزور ٿيندي آهي جڏهن ته هفتي ۾ هڪ ڀيرو روئڻ سان روشني بهتر ٿيندي آهي۔ ٭ تحقيق مان اها ڳالھ به سامهون آئي آهي ته لڙڪن جي صورت ۾ جيڪو پاڻي اکين مان نڪرندو آهي ان ۾ ڪيئي طرح جا ڪيميائي جزا هوندا آهن۔ اهڙا لڙڪ انتهائي گھٽ مقدار ۾ به نڪرن ته ان جي

نتيجي ۾ شريانن ۾ رت جا قطرا ڄمڻ جو عمل سست ٿي ويندو آهي ۽ جِلدي مرض (Skin diseases) مان نجات ملندي آهي۔ (مختلف ويب سائٹ سے ماخوذ)

پيارا پيارا اسلامي ڀائرو! ممڪن آهي ته ڳوڙها ڳاڙڻ جا فائدا ٻڌي روئڻ جو ذهن بڻجي رهيو هجي۔ اهو ياد رکو ته شرعاً جيڪو روئڻ پسنديده آهي ۽ جنهن تي ثواب آهي اهو روئڻ الله پاڪ جي لاءِ هجي، آخرت جي فڪر ۾ هجي، خوف خدا جي ڪري هجي۔ جڏهن ته دنيا جي لاءِ روئڻ تي اکين کي فائدو ته ملي سگھي ٿو پر ثواب نه ملندو۔ افسوس! اڄ اسان پنهنجي دنيا چڱي ڪرڻ جي لاءِ ڪيترا بي چين آهيون۔ دنيا سٺي ٿي وڃي، صحت ٺيڪ ٿي وڃي، پريشانيون ختم ٿي وڃن، مصيبتون دور ٿي وڃن، مال ۽ دولت جي ڪثرت ٿي وڃي، قيمتي موبائل ملي وڃي، نئي گاڏي ملي وڃي، مطلب ته! بيشمار دنياوي مقصد آهن جو جن کي ماڻڻ جي لاءِ اسان پنهنجي وس آهر ڪوشش ڪندا آهيون پر افسوس آخرت کي بهتر ڪرڻ جو جذبو اهڙو نظر نٿو اچي جهڙُو هجڻ گهرجي. اي ڪاش! اسان کي به دُنيا جي ناپائيداري جو حقيقي معنٰي ۾ اِحساس ٿي وڃي، اي ڪاش! اسان جي به غفلت جو پردو کڄي وڃي۔ اي ڪاش! اسان کي به امّيدِ رحمت سان گڏوگڏ صحيح معنٰي ۾ خوفِ خدا مُيسر اچي، اي ڪاش! بُري خاتِمي جو ڊپ اسان جي دل ۾ ويهي رهي، اي ڪاش! اسان کي پنهنجي پَروَردگار جي ناراضين جو هر دم خوف رهي، اي ڪاش! سڪرات جي سختين، مَوت جي تڪليفن، پنهنجي غسلِ ميت ۽ تڪفين ۽ تدفين جي ڪيفيتن ۽ مرده حالت ۾ پنهنجي بيوسي جو احساس ٿي وڃي۔ اي ڪاش! قبر جي اونداھين ۽ وَحشتن، منڪر ۽ نڪير جي سُوالن، قَبر جي عذابن جو غم اسان کي هر وقت ستائيندو رهي۔ اي ڪاش! مَحشَر جي گرمين ۽ گھٻراهٺن، پُل صراط جي دَهشتن، بارگاه اِلاهي جي پيشين، ميدانِ قِيامت ۾ ننڍين ننڍين ڳالهين جي پڇاڻي جو ۽ سڀني جي اڳيان عيب کُلڻ جي رُسوائين جو به خوف لاحق ٿي وڃي۔ اي ڪاش! جهنَّم جي خوفناڪ دانهُن، دوزخ جي هولناڪ سزائن، پنهنجي نازن سان پليل بدن جي نَزاڪتن ۽ جنّت

جي عظيم نعمتن سان محرومين جو خوف اسان کي بي چين ڪندو رهي ۽ اي ڪاش! اهو خوف اسان جي لاءِ هدايت ۽ رحمت جو ذَرِيعو بڻجي وڃي۔

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله عَلٰي مُحَمَّد

انبياءِ ڪرام جو خوفِ خدا سبب روئڻ

پيارا پيارا اسلامي ڀائرو! هن ۾ ڪو شڪ ناهي ته انبياء ڪرام عَلَيهمُ السَّلاَم اهي مقدس هستيون آهن جيڪي الله پاڪ جي بارگاھ ۾ تمام مخلوق کان وڌيڪ مرتبو رکن ٿيون۔ هي اهي مقدس هستيون آهن جيڪي يقيناً يقيناً الله پاڪ جي غضب، ان جي عذاب ۽ ان جي ناراضي کان محفوظ آهن۔ ايستائين جو الله پاڪ انهن جي حفاظت جو واعدو ڪري ڇڏيو آهي ته انهن کان گناھ ٿي ئي نٿو سگهي. (بہارِ شریعت ، جلد۱، ح۱، ص 38) بلڪ انهن جو شان ته هي آهي جو جن جي سفارش هي ڪن ته الله پاڪ انهن کي به عذابِ دنيا ۽ آٓخرت کان محفوظ ڪري ڇڏيندو آهي۔ هي مُقدَّس شَخصيتون گناهن کان پاڪ هئڻ ۽ الله پاڪ جي رضا جي منصب تي فائز هئڻ جي باوجود به اشڪباري ۽ آھ ۽ زاري ڪندا آهن۔ اسان جي شفاعت فرمائڻ واري آقا، مديني وارا مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سڄي سڄي رات عبادت ۽ اشڪباري ۾ گذاري ڇڏيندا هئا۔ جڏهن ته انبياء ڪرام عَلَيۡهِمُ السَّلَام ۾ ڪيئي اهڙا آهن جو جن جي اشڪباري ۽ آھ ۽ زاري ڪيئي ڪيئي ڏينهن تائين برقرار رهندي هئي۔ اچو انبياءِ ڪرام جي خوفِ خدا ۾ روئڻ بابت پنج (5) روايتون ٻڌون ٿا:

٭ منقول آهي ته حضرتِ سَيدُنا يحييٰ عَلَيۡهِ السَّلَام جڏهن نَماز جي لاءِ بيهندا هئا ته (خوفِ خدا َ سبب) ايتري قَدر روئندا هئا جو وڻ (Trees) ۽ مِٽي جا ڀِتر به روئڻ لڳندا هئا ايستائين جو سندن عَلَيۡهِ السَّلَام جا والدِ ماجد حضرت سَيدُنا زَڪرِيا عَلَيۡهِ السَّلَام به حضرت يحييٰ عَلَيۡهِ السَّلَام کي ڏسي روئندا هئا ايستائين جو بيهوش ٿي ويندا هئا۔ پاڻ عَلَيۡهِ السَّلَام اهڙي طرح مسلسل لڙڪ ڳاڙيندا رهندا هئا، ايستائين جو لاڳيتو وهندڙ لڙڪن جي ڪري حضرت يحيٰي عَلَيۡهِ السَّلَام

جي ڳلن تي زخم ٿي پيا هئا۔ اهو ڏسي حضرت سَيدُنا يحييٰ عَلَيۡهِ السَّلَام جي امڙ سائڻ سندن جي ڳلن تي اُن واريون پٽيون لڳائينديون هيون۔ هن جي باوُجُود جڏهن پاڻ عَلَيۡهِ السَّلَام ٻيهر نَماز جي لاءِ بيهندا هئا ته وري روئڻ شروع ڪندا هئا، جنهن جي نتيجي ۾ اهي اُن واريون پٽيون آليون ٿي وينديون هيون، جڏهن سندن جي والِده پٽين کي سڪائڻ جي لاءِ نپوڙينديون هيون ۽ پاڻ عَلَيۡهِ السَّلَام پنهنجن لڙڪن کي پنهنجي امڙ جي ٻانهن تي ڪرندو ڏسي بارگاھ اِلاهي ۾ عرض ڪندا هئا: اي الله پاڪ! هي منهنجا لڙڪ آهن، هيءَ منهنجي امڙ آهي ۽ مان تنهنجو ٻانهو آهيان جڏهن ته تون سڀ کان وڌيڪ رَحم فرمائڻ وارو آهين۔ (احياء العلوم ، ڪتاب الخوف والرجاء ،۴/ ۲۲۵ از خوفِ خدا، ص۴۵)

٭ اهڙي طرح حضرت سَيدُنا شعيب عَلَيۡهِ السَّلَام جي باري ۾ اچي ٿو ته پاڻ خوفِ خدا سان ايترو روئيندا هئا جو مسلسل روئڻ جي ڪري سندن اڪثر اکين جي روشني (Sight) ختم ٿي وئي۔ ماڻهن عرض ڪيو: اي الله پاڪ جا پيارا نبي عَلَيۡهِ السَّلَام! اوهان ايترو ڇو روئو ٿا جو اوهان جو نور پڻ ويندو رهيو؟ ارشاد فرمايائون: ٻن ڳالهين جي سبب۔ (۱) ڪٿي منهنجي نظر اهڙي شيء تي نه پوي جنهن کي ڏسڻ کان شريعت منع فرمايو آهي۔ (۲) جيڪي اکيون پنهنجي ربِّ ڪريم جو جلوو ڏسڻ چاهن ٿيون، مان نٿو چاهيان ته اهي ڪنهن ٻي شيء کي به ڏسن، تنهنڪري مون مناسب خيال ڪيو ته نابين (Blind) جيان ٿي وڃان ۽ جڏهن قيامت ۾ منهنجي اک کلي ته يڪدم منهنجي نظر پنهنجي ربِّ ڪريم جو ديدار ڪري۔ ان کان پوءِ پاڻ سٺ سال ظاهري حياتي سان متصف رهيا پر ڪنهن سندن کي اک کولندي نه ڏٺو۔ (خوفِ خدا، ص۴۵)

٭ حضرتِ سيدُنا داؤد عَلَيۡهِ السَّلَام جي باري ۾ اچي ٿو ته هڪ ڀيري چاليھ (40) ڏينهن تائين سجدي جي حالت ۾ روئيندا رهيا ۽ الله پاڪ کان حياء جي سبب آسمان ڏانهن پنهنجو مٿو نه کنيائون۔ ايترو وڌيڪ روئڻ جي ڪري

سندن عَلَيۡهِ السَّلَام جي لڙڪن سان گاھ (Grass) ڦٽي پيو ۽ ان گاھ حضرت داؤد عَلَيۡهِ السَّلَام جي مٿي کي ڍڪي ورتو۔ (حڪايتيں ۽ نصيحتيں، ص۱۳۵ بتغير)

٭ حضرتِ سَيدُنا ابراهيم عَلَيۡهِ السَّلَام جي باري ۾ اچي ٿو ته پاڻ جڏهن نماز جي لاءِ بيهندا هئا ته خوفِ خدا جي سبب ايتري قَدَرگِريو و زاري ڪندا (يعني روئيندا) هئا جو هڪ مِيل جي مفاصلي تان سندن جي سيني ۾ ٿيندڙ گڙگڙاهٽ جو آواز ٻڌڻ ايندو هو۔ (اِحياءُ الْعُلوم، ۴/۲۲۶از نيکی کی دعوت، ص۲۷۳)

٭ حضرتِ سيدُنا عيسٰي رُوحُ الله عَلَيۡهِ السَّلَام جي عادت هئي ته جڏهن ڪو نصيحت آموز واقعو ٻڌندا هئا ته اهڙي طرح روئيندا هئا جهڙي طرح ڪا عورت ٻار فوت ٿيڻ تي روئندي آهي۔ (الل ّٰ ه والوں کی باتيں، ۵/۴۳)

12 مدني ڪمن مان هڪ ”يوم تعطيل اعتڪاف“

پيارا پيارا اسلامي ڀائرو! انبياءِ ڪرام عَلَيۡهِمُ السَّلَام جو خوفِ خدا ۾ ٻڏل رهڻ ۽ خوفِ خدا ۾ ڳوڙها ڳاڙڻ مرحبا۔ اسان کي به پنهنجو پاڻ ۾ خوفِ خدا پيدا ڪرڻ جي ڪوشش ڪرڻ گھرجي۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي جي مدني ماحول جون جتي ۽ بي شمار بهارون آهن اتي هڪ بهار اها به آهي ته هن مدني ماحول جي برڪت سان خوفِ خدا ۾ روئڻ نصيب ٿيندو آهي. اهو ئي مدني ماحول فڪرِ آخرت ۾ بيقراري جي ڪيفيت پيدا ڪندو آهي، اهو ئي مدني ماحول قبر ۽ آخرت جي ياد دل ۾ پيدا ڪندو آهي۔ اهو ئي مدني ماحول غمِ مدينه ۽ عشقِ رسول ۾ اشڪباري جو ذهن بڻائيندو آهي۔ اوهان به هن مدني ماحول سان وابسته ٿي وڃو۔ ذيلي حلقي جي ٻارهن مدني ڪمن ۾ خوب خوب حصو وٺو. خاص ڪري دعوتِ اسلامي تحت ٿيڻ واري هفتيوار سنتن ڀري

اجتماع، هفتيوار مدني مذاڪري ۽ هر مهيني عاشقانِ رسول سان گڏ مدني قافلن ۾ سفر کي پنهنجو معمول بڻايو۔ انهن مدني ڪمن جي برڪت سان عاشقانِ رسول جي صحبت نصيب ٿيندي جيڪا دل ۾ رقت پيدا ڪرڻ سان گڏ فڪرِ آخرت ۾ تڙپڻ جو ذهن بيدار ڪندي۔ ذيلي حلقي جي 12 مَدَني ڪمن مان هفتيوار هڪ مدني ڪم ”يومِ تَعطِيل اِعتِـڪاف“ به آهي۔ موڪل واري ڏينهن شهر جي ڪمزور ۽ قريبي علائقن ۽ ڳوٺن ۾ وڃي اتي جي مسجدن کي آباد ڪرڻ سان گڏوگڏ مقامي عاشقان رسول کي نيڪي جي دعوت ڏئي علم دين سکڻ سيکارڻ جي ترڪيب ڪئي ويندي آهي۔ ٭ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ يوْمِ تَعطِيل اِعتِـڪاف اسلامي ڀائرن کي سُنّتون، آداب ۽ مَدَني دَرس وغيره جو طريقو سيکارڻ جو بهترين ذريعو آهي۔ ٭ يومِ تَعطِيل اِعتِـڪاف جي بَرَڪت سان مسجِدون آباد ٿين ٿيون. ٭ يومِ تَعطِيل اِعتِـڪاف جي بَرَڪت سان مسجد ۾ گذرندڙ هر هر لَمحو عِبادَت ۾ گذارڻ جو موقعو ملندو ٭ يومِ تَعطِيل اِعتِـڪاف جي بَرَڪت سان مسجِد سان مَحَبَّت ڪرڻ ۽ ان ۾ پنهنجو گھڻي کان گھڻو وَقت گُذارڻ جي فضيلت حاصل ٿيندي۔

12 مدني ڪمن مان هفتيوار هن مدني ڪم ”يومِ تعطيل اعتڪاف“ جي تفصيلي معلومات ڄاڻڻ جي لاءِ مڪتبة المدينه جي رسالي ”يومِ تعطيل اعتڪاف“ جو مطالعو ڪريو، تمام اسلامي ڀائر هن رسالي جو لازمي مطالعو ڪن، اهو رسالو مڪتبة المدينه تي دستياب هئڻ سان گڏوگڏ دعوت اسلامي جي ويب سائٽ تان به پڙھي سگھجي ٿو .

هن رسالي جي مطالعي سان اوهان کي خبر پوندي ٭ يومِ تعطيل اعتڪاف ڇا آهي ٭ يومِ تعطيل اعتڪاف جو جدول ٭ مدني ڪمن جي مضبوطي جو ذريعو ٭ ڳوٺ ۽ دعوتِ اسلامي ٭ يومِ تعطيل اعتڪاف بابت ڪجھ سوال جواب ٭ يومِ تعطيل اعتڪاف جا 14 فائدا وغيره

خوفِ خدا ۾ روئڻ جا فائدا

پيارا پيارا اسلامي ڀائرو! اسان خوفِ خدا ۾ روئڻ جي اهميت بابت ٻڌي رهيا هئاسين۔ صاف پاڻي جو اهو وصف آهي ته اهو ڪِن ڪچرو ختم ڪندو آهي۔ جيڪڏهن ڪنهن هنڌ گندگي هجي ۽ پاڻي هاريو ته جتي جتي پاڻي پهچندو اهو هنڌ به صاف (Clean) ٿيندو ويندو۔ خوفِ خدا ۾ نڪرندڙ لڙڪ نڪرندا ته ٻاهر آهن پر انسان جي ”اندر“ کي صاف ڪري ويندا آهن۔ اهي لڙڪ وهندا ظاهر تي آهن پر صفائي انسان جي باطن جي ڪندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ دل جي سختي دور ڪندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ دل جي نرمي جو سبب بڻجندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ قلبي سڪون فراهم ڪندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ گناهن جي ٻوجھ کان نجات ڏيندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ الله پاڪ جي رحمت جو اميدوار ڪندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ جهنم کان آزادي جو سبب ٿيندا آهن۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ جنت جو حقدار بڻائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ قبر جي مرحلن کي آسان بڻائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ عذابِ قبر کان بچائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ قبر جي تنگي ۽ وحشت کي دور ڪري سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ قبر کي گُلِ گلزار بڻائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ سڀاڻي ٿيندڙ ندامت کان بچائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ محشر جي اُس ۽ اُڃ کان بچائي سگھن ٿا۔ ٭ خوفِ خدا سبب نڪرندڙ لڙڪ پل صراط تي آساني جو سبب ٿي سگھن ٿا۔ مطلب ته! ٭ خوفِ خدا سبب نڪرندڙ لڙڪ بظاهر بيڪسي جو اظهار هوندا آهن پر درحقيقت عزتن ۽ عظمتن کي ماڻڻ جو ذريعو بڻجي سگھن ٿا۔

اولياءِ ڪرام جو خوفِ خدا ۾ روئڻ

الله پاڪ جا نيڪ ٻانها خوفِ خدا جي ڪري ڪثرت سان لڙڪ وهائيندا آهن، ڪيئي ولين سڳورن جي باري ۾ منقول آهي ته خوفِ خدا ۾ ڪثرت سان

روئڻ سبب انهن جو نور ختم ٿي ويو پر انهن روئڻ نه ڇڏيو۔ اچو! ترغيب جي لاءِ خوفِ خدا ۾ روئڻ بابت اولياءِ ڪرام جا ڪجھ واقعا ٻڌون ٿا:

1۔ حضرت سَيدُنا ابو بِشر صالح مُرّي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه وڏا نالي وارا محدث هئا۔ پاڻ زبردست مبلغ به هئا۔ بيان جي دوران سندن جي اها ڪيفيت ٿيندي هئي جو رب جي خوف مان ڏڪندا ۽ ڪنبدا هئا ۽ ايتري قدر روئيندا هئا جيئن ڪا عورت پنهنجي سڪيلڌي ٻار جي مري وڃڻ تي روئيندي آهي۔ ڪڏهن ڪڏهن ته گھڻي روئڻ ۽ بدن جي ڏڪڻ سان سندن عضون جا جوڙ پنهنجي جڳھ تان ڪنبڻ لڳندا هئا ۽ سندن جو بيان ٻڌڻ وارن تي اهڙو اثر ٿيندو هو جو ڪي ماڻهو تڙپي تڙپي بي هوش ٿي ويندا هئا ۽ ڪي انتقال ڪري ويندا هئا۔ سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي خوفِ خدا جو اهو عالم هو جو جيڪڏهن ڪنهن قبر کي ڏسي وٺندا هئا ته ٻه ٻه، ٽي ٽي ڏينهن حيران ۽ خاموش رهندا هئا ۽ کائڻ پيئڻ ڇڏي ڏيندا هئا۔ (اوليائے رجال الحديث ص۱۵۱،خوفِ خدا، ص۷۶)

2۔ حضرتِ عمر بن عبدالعزيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو دستور هو هر رات عالمن کي گڏ ڪندا هئا موت، قيامت ۽ آخرت جو تذڪرو ڪندي ايترو روئيندا هئا جو ائين لڳندو هو جو ڄڻ اڳيان جنازو رکيل آهي۔ (مڪاشفةا لقلوب، ص۱۹۶)

3۔ حضرت سَيدُنا عطاء رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي باري ۾ اچي ٿو ته ڪڏهن به ڪنهن پاڻ کي مرڪندي نه ڏٺو، پاڻ جڏهن روئڻ شروع ڪندا هئا ته ٽي ڏينهن ۽ ٽي راتيون مسلسل روئيندا رهندا هئا۔ جڏهن به آسمان تي بادل ظاهر ٿيندا هئا ۽ بجلي کڙڪو ڪندي هئي سندن دل جي ڌڙڪن تيز ٿي ويندي هئي، بدن ڏڪڻ لڳندو هو، بيتاب ٿي ڪڏهن بيهندا ڪڏهن ويهندا هئا، گڏوگڏ روئيندي چوندا هئا: شايد منهنجي لغزشن ۽ گناهن جي ڪري زمين وارن کي ڪنهن مصيبت ۾ مبتلا ڪيو وڃڻ وارو آهي، جڏهن مان مري ويندس ته ماڻهن کي به سڪون حاصل ٿي ويندو۔ (خوفِ خدا، ص۸۵)

سلطانُ الهند حضرت خواجه غريب نواز سيد حسن سنجري اجميري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه زبردست ولي الله هئا۔ پر اُنهن تي خَوفِ خُدا ايتري قدر غالب هو جو هميشه خَشِيتِ اِلاهي سان ڏڪندا ۽ گريه وزاري ڪندا رهندا هئا، مخلوقِ خُدا کي خَوفِ خُدا جي تلقين ڪندي ارشاد فرمائيندا هئا: اي انسانو! جيڪڏهن توهان مٽي هيٺ ستل ماڻهن جو حال ڄاڻي وٺو ته خوف وچان بيهي بيهي پگھرجي وڃو۔ (معین الارواح، ص ۱۸۵ملخصاً)

مجلس لنگرِ رضويه

اي عاشقانِ اوليا! اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوتِ اسلامي جو مدني ماحول جتي خوفِ خدا جو پيڪر بڻائي ٿو اتي اولياءِ ڪرام جي سچي محبت به ڏيندو آهي۔ اوهان به انهيء ماحول سان هردم وابسته رهو۔ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي نيڪي جي دعوت عام ڪرڻ جي لاءِ 107 کان وڌيڪ شعبن ۾ ڪم ڪري رهي آهي۔ انهن مان هڪ مجلس لنگر رضويه به آهي، مختلف موقعن تي هزارين عاشقانِ رسول جي خيرخواهي ڪئي ويندي آهي يعني انهن کي کاڌو کارايو ويندو آهي مثلاً دارالسنه ۾ اچڻ وارن مهمانن، مختلف سنتن ڀريئي اجتماعن، مدني مشورن، مختلف مدني ڪورسز ۾ شرڪت ڪرڻ وارن عاشقانِ رسول جي خيرخواهي ڪئي ويندي آهي، دعوتِ اسلامي جي مدني ماحول ۾ ملڪ ۽ بيرونِ ملڪ هر سال هزارين عاشقانِ رسول سڄي رمضان ۽ آخري عشري جي اِعتڪاف جي سعادت حاصل ڪندا آهن، انهن عاشقانِ رسول جي لاءِ "لنگرِ رضويه" (سحري ۽ افطاري) جو نهايت

وڏو اِنتظام ڪيو ويندو آهي ته جيئن اُهي عاشقانِ رسول دِلجمعي سان علمِ دين جي مدني حلقن ۽ مدني مذاڪرن ۾ ڀرپور توجه سان شرڪت ڪري سگھن۔ دعوتِ اسلامي جي عالمي مدني مرڪز فيضانِ مدينه ڪراچي ۾ ٿيڻ واري مدني مذاڪرن جي اجتماعن ۾ به عاشقانِ رسول جي لاءِ لنگر رضويه جو اهتمام ڪيو ويندو آهي، لنگرِ رَضَويه جي مد ۾ ڪروڙين روپين جا اخراجات ٿيندا آهن، هن نيڪي جي عظيم ڪم ۾ به ڪيئي عاشقانِ رسول پنهنجو حصو شامل ڪندا آهن، هن سال به اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ملڪ ۽ بيرونِ ملڪ سوين هنڌن تي سڄي مهيني ۽ آخري عشري جو اعتڪاف ٿيندو، جنهن ۾ هزارين عاشقانِ رمضان، رمضان جي مبارڪ گهڙين جون برڪتون حاصل ڪرڻ جي سعادت حاصل ڪندا۔ تمام عاشقانِ رسول رضاءِ اِلاهي ۽ ٻين سٺين سٺين نيتن سان "لنگرِ رضويه" جي مد ۾ پنهنجو حصو شامل ڪرڻ جي سعادت حاصل ڪن۔

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله عَلٰي مُحَمَّد

”گھر ۾ اچڻ وڃڻ جون “سنتون ۽ آداب

مٺا مٺا اسلامي ڀائرو! اچو! شيخِ طريقت،اميرِاهلسنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ’’101 مدني گل‘‘ مان ” گھر ۾ اچڻ وڃڻ جون سنتون ۽ آداب“ ٻڌون ٿا: ٭ جڏهن گهران ٻاهر نڪرو ته هي دعا پڙهو: ” بِسْمِ اللهِ تَوَکَّلْتُ عَلَی اللهِ لَاحَوْلَ وَلَا قُوَّۃَ اِلَّا بِاللّٰہِ ‘‘ ترجمو: الله جي نالي سان، مون الله ڪريم تي ڀروسو ڪيو ۔ الله کان سواءِ نه

(گناهن کان بچڻ جي) قوت آهي ۽ نه (نيڪين ڪرڻ جي) طاقت آهي. (ابو داود،۴/۴۲۰،حديث:۵۰۹۵) اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ انهيءَ دعا کي پڙهڻ جي برڪت سان سڌي واٽ تي رهندو مصيبتن کان حفاظت ٿيندي ۽ الله ڪريم جي مدد شاملِ حال رهندي ٭ گهر ۾ داخل ٿيڻ جي دعا؛ اَللّٰھُمَّ اِنِّیْٓ اَسْئَلُکَ خَیْرَ الْمَوْلَجِ وَخَیْرَ الْمَخْرَجِ بِسْمِ اللهِ وَلَجْنَا وَبِسْمِ اللهِ خَرَجْنَا وَعَلَی اللهِ رَبِّنَا تَوَکَّلْنَا ۔ ترجمو: اي الله! آئون تو کان داخل ٿيڻ ۽ نڪرڻ جي ڀلائي گهران ٿو، الله پاڪ جي نالي سان اسان (گهر ۾) داخل ٿياسين ۽ انهي جي نالي سان ٻاهر آياسين ۽ پنهنجي رب ڪريم تي اسان ڀروسو ڪيوسين) (ابوداود،۴/۴۲۰،حديث:۵۰۹۶) دعا پڙهڻ کانپوءِ گهر وارن کي سلام ڪري پوءِ بارگاھِ رسالت ۾ سلام عرض ڪري ان کان پوءِ ”سوره اِخلاص“ شريف پڙهي۔ اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ روزي ۾ برڪت، ۽ گهريلو جهيڙن کان بچاءُ ٿيندو۔ ٭ جيڪڏهن اهڙي گهر (کڻي پنهنجي خالي گهر) ۾ وڃڻ ٿئي جو اُن ۾ ڪو به نه هجي ته هيئن چئو: اَلسَّلَامُ عَلَیْنَا وَعَلٰی عِبَادِ اللهِ الصّٰلِحِیْن (يعني اسان تي ۽ الله عَزَّوَجَلَّ جي نيڪ ٻانهن تي سلام) ملائڪ انهي سلام جو جواب ڏيندا۔ (ردالمحتار ج 9 ص 682) يا هن ريت چوي: اَلسَّلَامُ عَلَیْکَ اَیُّـھَاالنَّبِیُّ (يعني يا نبي توهان تي سلام) ڇوته حضور اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو روح مبارڪ مسلمانن جي گهرن ۾ تشريف فرما هوندو آهي۔ (بهارِشريعت،حصّه۶ا ص۹۶،شرح الشّفاء للقاري،۲/۱۱۸) ٭ جڏهن ڪنهن جي گهر ۾ داخل ٿيڻ چاهيو ته هيئن چئو: اَلسَّل َ امُ عَلَیْکُم ڇا مان اندر اچي سگھان ٿو؟ ٭ جيڪڏهن داخل ٿيڻ جي اجازت نه ملي ته خوشي سان موٽي وڃو، ٿي سگھي ٿو ڪنهن مجبوري جي ڪري گهر جي مالڪ اجازت نه ڏني هجي ٭ جڏهن ڪو توهان جي گهر جو دروازو کڙڪائي ته سنت اها آهي ته پڇو: ڪير آهي؟ ٻاهران واري کي گهرجي ته پنهنجو نالو ٻڌائي؛ مثال طور چوي: ”محمد الياس“ نالو ٻڌائڻ جي بدران انهيءَ موقعي تي ”مدينه!“ ”آئون آهيان!“ ”دروازو کوليو“ وغيره چوڻ سنّت ناهي ٭ جواب ۾ نالو ٻڌائڻ کان پوءِ دروازي کان هٽي بيهو ته جيئن دروازو کلندي ئي گهر ۾ نظر نه پوي۔

پيارا پيارا اسلامي ڀائرو! اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ! آ ئنده هفتيوار سنتن ڀري اِجتماع جو موضوع هوندو ”صدقي جا فضائل“ جنهن ۾ اسان قرآن پاڪ ۽ حديث سڳورين جي روشني ۾ صدقي جا فضائل ٻڌنداسين. ان سان گڏوگڏ بزرگن جا صدقي ڪرڻ جي متعلق واقعا به ٻڌنداسين، ان علاوه وڌيڪ ٻيا کوڙ مدني گل بيان ڪيا ويندا. لهٰذا! آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ اِنفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان چئو: اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ

﴿1﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿2﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿3﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([1])

﴿5﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهڙي مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([2])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([3])

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
