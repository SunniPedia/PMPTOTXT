اَلۡـحَـمۡـدُ لِـلّٰـہِ رَبِّ الۡـعٰـلَـمِیۡنَ وَ الـصَّـلٰـوۃُ وَ الـسَّـلَامُ عَـلٰی سَـیِّـدِ الۡـمُـرۡ سَـلِـیۡنَ

اَمَّـا بَــعۡـدُ فَـاَعُـوۡذُ بِـا لـلّٰـہِ مِـنَ الـشَّـیۡـطٰنِ الـرَّجِیۡمِ ؕ بِـسۡمِ الـلّٰـہِ الـرَّحۡـمٰنِ الـرَّحِـیۡمِؕ

مٺا مٺا اسلامي ڀائرو! اچو شيخِ طريقت، امير اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي نفل روزن جي ترغيب ۽ رجب المرجب جي فضائلن تي مشتمل هڪ خوبصورت مڪتوب ٻڌون ٿا؛

مڪتوبِ عطار

بِسۡمِ اللهِ الرَّحمٰنِ الرَّحِيمِ ط سگِ مدينه محمّد الياس عطار قادري رضوي عُفِیَ عَنه جي طرفان، تمام اسلامي ڀائرن ، اسلامي ڀينرن ، مدارس المدينه ۽ جامعاتُ المدينه جي اساتذه ، طَلَبه معلمات ۽ طالبات جي خدمات ۾ ڪعبهء مُشَـرّفه جي چوڌاري گھمندو،گنبذ خضريٰ کي چمندو رَجَبُ المُرَجب، شَعبانُ المُعَظم ۽ رمضانُ المُبارڪ جي برڪتن سان مالا مال جهومندو سلام،

ا َ لسَّلَامُ عَلَيْکُم وَرَحْمَةُ الله ِوَبَرَکاتُهٗ اَ لۡـحَـمۡدُ لِلهِ رَبِّ العٰلَـمِيۡنَ عَليٰ کُلِ حَال

ہو نہ ہو آج کچھ مرا ذکر حضور ميں ہوا

ورنہ مری طرف خوشی د يکھ کے مسکرائی کيوں (حدائق بخشش)

اَلۡـحَمۡدُ لله عَزَّوَجَلَّ هڪ ڀيرو وري خوشين جا ڏينهن اچڻ لڳا، رجب شريف جي مهيني جي آمد آمد آهي. هن مبارڪ مهيني ۾ عبادت جو ٻج پوکيو ويندو آهي، شَعبانُ المُعَظم ۾ شرمساري جي ڳوڙهن سان پاڻي ڏنو ويندو آهي ۽ رمضان المبارڪ جي مهيني ۾ رحمت جو فصل ڪٽيو ويندو آهي.

ٽن مهينن جا روزا

رجب المرجب جا قدر دانو! تعليم و تعلُّم (يعني سکڻ ۽ سيکارڻ) ۽ ڪسبِ حلال ۾ رڪاوٽ نه هجي، ماءُ پيءُ به بلا وجه منع نه ڪن، ڪنهن جي به حق تلفي نه ٿيندي هجي جلدي جلدي ۽ جلدي مسلسل ٽن مهينن جا يا رمضان المبارڪ جي مڪمل فرض روزن سان گڏ جنهن کان جيترا ٿي سگهن نفل روزن جي لاءِ ڪمر بسته ٿي وڃو، سحري ۽ افطار ۾ گهٽ کائي ڪري پيٽ جو قفل مدينه به لڳايو. ڪاش! هر گهر ۾ ۽ منهنجي جمله مدارس المدينه ۽ تمام جامعة المدينه ۾ روزن جون بهاريون اچي وڃن. بس پهرين رجب شريف کان ئي روزن جو آغاز فرمائيندا وڃو.

رجب جي ابتدائي (پهريان) ٽن روزن جي فضيلت!

رجب شريف جي ابتدائي ٽن روزن جي فضائل جي به ڇا ڳالهه ڪجي! حضرت سيّدنا عبدالله ابنِ عبّاس رَضِىَ اللهُ تَعَالٰی عَنْهُمَا کان روايت آهي ته بيچين دِلين جي چين، رحمتِ دارَين، تاجدارِ حَرمَين، سروَرِ ڪَونَين، ناناءِ حَسۡنَين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ رحمتِ نشان آهي: ” رجب جي پهرين ڏينهن جو روزو ٽن سالن جو ڪفارو آهي ۽ ٻي ڏينهن جو روزو ٻن سالن جو ۽ ٽين ڏينهن جو هڪ سال جو ڪفارو آهي، پوءِ هر ڏينهن جو روزو هڪ مهيني جو ڪفارو آهي.“ (الجامع الصغير للسيوطي ص 311 حديث 5051 فضائل شهر رجب للخلال ص 7)

ميں گنہگار گناہوں کے سوا کيا لاتا

نيکياں ہوتی ہیں سرکار نِکوکار کے پاس

نفلي روزن جون به ڇا نه خوب بهارون آهن هن ضِمن ۾ ٻه مبارڪ حديثون ملاحظه فرمايو ،

(1) فِـرشتا ڪڏهن دعاءِ مَغفِرت ڪندا آهن؟

حضرت سيّدَتُنا اُمِّ عَمَّارَه رَضِىَ اللهُ تَعَالٰی عَنْها فرمائن ٿيون: حُضُور پاڪ، صاحبِ لولاڪ، سَيَّاحِ اَفلاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مون وٽ تشريف وٺي آيا ته مون پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمت بابرڪت ۾ کاڌو پيش ڪيو ته ارشاد فرمايائون: ”تون به کاءُ“ مون عرض ڪيو: ”مان روزي سان آهيان“ ته رحمتِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: ”جيستائين روزيدار جي سامهون کاڌو کاڌو ويندو آهـي

فِرشتا ان روزيدار جي لاءِ دعاءِ مَغفِرت ڪندا رهندا آهن.“

(سنن ترمذي ج 2 ص 205 حديث 785)

(2) روزيدار جون هڏيون ڪڏهن تسبيح ڪنديون آهن؟

حضرت سيّدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ نبيِّ اَڪرم، رحمتِ عالم، نُورِ مُجسم، شاھِ بني آدم، رسولِ مُحۡتَشَم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمتِ مُعَظم ۾ حاضر ٿيا، ان وقت حُضُور پُر نُور، شافعِ يومُ النُّشور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ناشتو ڪري رهيا هئا، فرمايائون: اي بلال! ناشتو ڪريو، عرض ڪيائون: يارسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! مان روزيدار آهيان، ته رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: اسان پنهنجي روزي کائي رهيا آهيون ۽ بلال جو رزق جنّت ۾ وڌي رهيو آهي. اي بلال! ڇا توکي خبر آهي ته جيستائين روزيدار جي سامهون ڪجھه کاڌو وڃي تيستائين انهي جون هڏيون تسبيح ڪنديون

آهن، ان کي فرشتا دعائون ڏيندا آهن . (شعب الايمان ج 3 ص 297 حديث 3586)

مُفَسِّرِ شهير حڪيم الامت حضرت مفتي احمد يار خان عَـلَيْهِ رَحْـمَةُ الْـمَنَّان فرمائن ٿا ته هن مان معلوم ٿيو ته جيڪڏهن کاڌو کائڻ دوران ڪو اچي وڃي ته ان کي به کائڻ جي لاءِ سڏڻ سنت آهي، پر دِلي ارادي سان سڏي، ڪوڙي تواضُع (يعني صلاح) نه ڪري ۽ اچڻ وارو به ڪوڙ ڳالهائي هيئن نه چوي ته مونکي خواهش ناهي، ته جيئن بک ۽ ڪوڙ گڏجي نه وڃن.

بلڪ جيڪڏهن (نه کائڻ چاهي يا) کاڌو گھٽ ڏسي ته چـوي بَارَ ك َ اللهُ (يعـني الله عَزَّوَجَلَّ تو کي برڪت ڏي) هي به معلوم ٿيو ته سروَرِ ڪائنات، شاھِ موجودات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان پنهنجون عبادتون نه لڪائڻ گھرجن، بلڪ ظاهر ڪيون وڃن ته جيئن حضور پُرنُور، شافعِ يومُ النُّشُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان تي گواھ بڻجي وڃن، ۽ اهو ظاهر ڪرڻ رياء ناهي. (حضرت سيدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ جي روزي جو ٻڌي ڪري جيڪو ڪجھ فرمايو ويو ان جي شرح هي آهي) ته يعني اڄ جي روزِي اسان ته پنهنجي هتي ئي کائي وٺون ٿا.

حضرت سيّدنا بلال رَضِىَ اللهُ تَعَالٰی عَنْهُ ان جي عِوَض جنّت ۾ کائيندا، اهو عوض (يعني بدلو) بهتر به هوندو ۽ زياده به، حديث بلڪل پنهنجي ظاهري معنيٰ تي آهي، واقعي ان وقت روزيدار جي هر هڏي ۽ جوڙ بلڪ رڳ رڳ تَسۡبِيح (يعني الله عَزَّوَجَلَّ جي پاڪي بيان) ڪندي آهي جنهن جو روزيدار کي پتو ئي ناهي هوندو، پر سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌندا آهن . (مراٰة ج 3 ص 202)

مطالعو ڪيو هجي تڏهن به ٻئي رسالا (1) ڪـفـن جـي واپسـي ۽ رجب جون بهارون ۽ (2) آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو مهينو پڙهي ڇڏيو، ۽ پڻ هر سال شعبان المُعَظّم ۾ فيضان سنّت جو باب فـيـضـانِ رمـضـان ضرور پـڙهي ڇـڏيـنـدا ڪريو . ٿي سگهي ته عيدِ مِعراجُ النبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي نسبت سان 127 يا 27 رسالا يا حسب ِ توفيق فيضانِ رمضان به تقسيم فرمايو ۽ خوب خوب ثواب ڪمايو . تمام اسلامي ڀائرن سان بالعُمُوم ۽ جامعات المدينہ ۽ مدارس المدينہ جي جمله قاري صاحبان، استادن، ناظمن ۽ طالبن جي خدمت ۾ بالخصوص تڙپندڙ مدني عرض آهي ته براءِ ڪرم! (منهنجي جيئري ۽ مرڻ کان بعد به) زڪوٰة، فطرو، قرباني جون کلون ۽ ٻيا مدني عطيات جمع ڪرڻ ۾ اڳتي وڌي ڪري حصو وٺندا ڪريو. (اسلامي ڀينرون ٻين اسلامي ڀينرن ۽ مَحارِم کي ترغيب ڏيارين)

خدا جو قسم! مون کي انهن استادن ۽ شاگردن جي باري ۾ ٻڌي ڪري تمام گھڻي خوشي ٿيندي آهي جيڪي پنهنجي ڳوٺ يا شهر ۾ وڃڻ جي خواهش کي قربان ڪري رمضان المبارڪ جو مهينو جامعات ۾ گذاريندا آهن ۽ پنهنجي مَجلس جي هدايتن جي مطابق چندي جي بستن تي ذميداريون سنڀاليندا آهن. جيڪي استاد ۽ شاگرد بغير ڪنهن عذر جي صرف سُستي يا غفلت جي سبب عدمِ دلچسپي جو مظاهرو ڪندا آهن انهن جي سبب کان منهنجي دل ڏکندي آهي.

خصوصي مدني گُل: جيڪي اسلامي ڀاءَ يا اسلامي ڀيڻ چندو جمع ڪرڻ چاهن ٿا انهن کي چندي جا ضروري احڪام معلوم هجڻ فرض

آهن، تاڪيد سان عرض آهي ته جيڪڏهن پڙهي چڪا آهيو تڏهن به دعوت اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل 107 صفحن تي مشتمل ڪتاب ”چندي جي باري ۾ سوال جواب“ جو ٻيهر مطالعو فرمايو. يا الله عَزَّوَجَلَّ رمضان المبارڪ ۾ چندي ۽ عيد الاضحيٰ ۾ کلن جي لاءِ ڪـوشش ڪري جيڪـي عاشقانِ رسول منهنجي دل خوش ڪندا آهن تون انهن کان هميشه هميشه جي لاءِ راضي ٿي وڃ ۽ انهن جي صدقي مون پاپي ۽ بدڪار، گناهگارن جي سردار کان به هميشه جي لاءِ راضي ٿي وڃ .

ياالله عَزَّوَجَلَّ! جيڪي اسلامي ڀاءُ يا اسلامي ڀيڻ (عذر نه هجڻ جي صورت ۾) هر سال ٽن مهينن جا روزا رکي ۽ هر سال جَمادي الآخرٰي ۾ رسالو ڪفن جي واپسي ۽ رَجَبُ المُرَجَّب ۾ آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو مهينو ۽ شَعبان المُعَظّم ۾ فيضانِ رمضان پڙهي يا ٻڌي وٺڻ جي سعادت حاصل ڪري مون کي ۽ ان کي دنيا ۽ آخرت جون ڀلايون نصيب فرماءِ ۽ اسان کي بي حساب بخشي ڪري جنّت الفردوس ۾ پنهنجي مدني حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پاڙي ۾ گڏ رک. اٰمين بِجاهِ النَّبِيِّ الۡاَمين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

دعوتِ اسلامي جي طرف کان رَجَبُ المُرَجَّب جي ستاويھين رات، جشنِ معراج النبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سلسلي ۾ ٿيڻ واري اجتماعِ ذِڪر و نعت ۾ تمام اسلامي ڀائر شروع کان آخر تائين شرڪت فرمائيندا
