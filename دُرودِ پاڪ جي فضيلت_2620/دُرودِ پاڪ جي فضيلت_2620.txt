﴿ 2 ﴾ جنهن نوجوان دُنيا جي لذت ۽ ان جي عيش وعِشرت کي ڇڏي ڏنو ۽ پنهنجي جواني ۾ اللہ عَزَّوَجَلَّ جي اِطاعت جي طرف اڳرائي ڪئي، اللہ عَزَّوَجَلَّ ان خوش نصيب کي ٻاهتر (72) صِدّيقِن جي برابر ثواب عطا فرمائيندو۔ ([2])

﴿ 3 ﴾ بيشڪ اللہ عَزَّوَجَلَّ پنهنجي مخلوق ۾ ان خُوبصورت نوجوان کي سڀ کان وڌيڪ پسند فرمائيندو آهي جيڪو پنهنجي جواني ۽ حُسن و جمال کي اللہ عَزَّوَجَلَّ جي عبادت ۾ صَرف ڪري ڇڏيو، اللہ عَزَّوَجَلَّ فرشتن جي سامهون اهڙي ٻانهي تي فَخر ڪندي ارشاد فرمائيندو آهي ته ”هي منهنجو حقيقي ٻانهو آهي۔“ ([3])

الله عَزَّوَجَلَّ جو محبوب ٻانهو

﴿ 4 ﴾ بيشڪ اللہ عَزَّوَجَلَّ ان نوجوان سان مَحَبَّت فرمائيندو آهي جنهن پنهنجي جواني کي اِطاعتِ خُداوَندي ۾ فنا ڪري ڇڏيو هجي۔ ([4])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته جواني جي قَدر ڪرڻ وارن تي اللہ عَزَّوَجَلَّ ڪهڙو خاص فَضل ۽ ڪرم فرمائيندو آهي جو ان کي پنهنجي محبوب ٻانهن ۾ شامل فرمائي وٺندو آهي۔ تنهنڪري نوجوانن جي خِدمت ۾ عرض آهي، ته جيڪڏهن پوڙهائپ ۾ سُڪون ۽ اِطمينان واري زِندگي گُذارڻ جا خواہشمند آهيون ته جواني جي نعمت کي غنيمت ڄاڻيندي ان فاني دُنيا جي پويان ڀڄڻ جي بجائي پنهنجي نَفس کي عبادت ۽ رِياضت جي طرف مائل ڪرڻ جي ڪوشش ڪريون ۔ جڏهن ته هي بي حَد مشڪل آهي، ڇو جو جوانيءَ ۾ اُميدون ۽ خواہشون عُروج تي ہونديون آهن، پر جيڪڏهن اسان ٻين مُعاملن سان گڏوگڏ حُضُور ِاَڪَرَم، نُورِمُجَسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پيروي ڪندي حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي عِبادت ۽ رِياضت سان سينگاريل پاڪيزہ زندگي جي مطابق عمل ڪنداسين، ته اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ اسان جي زندگي ۾ به مدني بہارون اچي وينديون۔

حَضۡرَتِ سَيّدُنَا عَطَاء رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: آئون ۽ مون سان گڏ حَضۡرَتِ سَيّدُنَا اِبنِ عمر ۽ حَضۡرَتِ سَيّدُنَا عُبَيۡد بِن عَمۡروۡ رِضۡوَانُ اللهِ تَعَالٰی عَلَیۡھِمۡ اَجۡمعِیۡن اُمُّ الۡمُؤمِنِيۡن حَضۡرَتِ سَيِّدَتُنا عائشه صدّيقه رَضِیَ اللهُ تَعَالٰی عَنْهَا جي بارگاھِ عاليه ۾ حاضر ٿياسين۔

حَضۡرَتِ سَيِّدُنَا ابنِ عمر رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جن عرض ڪيو: اسان کي رَسُوۡلُ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي متعلق ڪا حيران ڪندڙ ڳالھ ٻڌايو۔ ته پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهَا روئي پيون ۽ ارشاد فرمايو: هڪ رات رَسُول اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مون وٽ تشريف وٺي آيا ۽ فرمائڻ لڳا: مون کي اجازت ڏي ته آئون پنهنجي رَبّ عَزَّوَجَلَّ جي عبادت ڪريان۔ مون عرض ڪيو: مون کي پنهنجي خواہش جي بجائي، توهان جو رَبّ تَعاليٰ جي ويجھو ہئڻ وڌيڪ پسند آهي۔ جيئن ته حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم گھر جي هڪ ڪنڊ ۾ بيهي ڪري ڳوڙها ڳاڙڻ لڳا۔ پوءِ چڱي طرح وُضو ڪري قرآنِ ڪريم پڙھڻ شُروع ڪيو ته ٻيهر لڙڪ وهائڻ لڳا، ايستائين جو حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اکين مُبارَڪ کان نڪرڻ وارا مبارڪ لڙڪ زمين تائين وڃي پہتا۔ ايتري ۾ مُؤذّنِ رَسُول، حَضۡرَتِ سَيّدُنَا بِلَال حَبۡشِي رَضِىَ اللهُ تَعَالٰی عَـنْهُ حاضر ٿيا ته حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي روئيندو ڏسي ڪري عرض ڪيو: يارَسُول اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! منهنجا ماءُ پيءُ، اوهان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي قُربان! ڪنهن شيء اوهان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي روئاريو آهي؟ جڏهن ته توهان جي صَدقي ته اللہ عَزَّوَجَلَّ اوهان کان اڳين جا ۽ پوين جا گُناھ بخش فرمائيندو، ارشاد فرمايو: ڇا آئون شُڪر گذار ٻانهو نه ٿيان؟ ([1])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته! اسان جا بخشيل بخشايل آقا، اسان گھنگارن کي بخشوائڻ وارا مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم معصوم بلڪه سَيدُالمعص ُ ومِين ۽ سَيدُالعَابِدين (عبادت گذارن جا سردار) ہئڻ جي باوُجُود ڪيتري قَدر گِريه ۽ زاري سان اللہُ رَبُّ العٰلمين عَزَّوَجَلَّ جي عِبادت ڪندا هئا، جڏهن ته حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي شان ۽ عظمت ايتري قَدر بُلند ۽ بالا آهي جو اللہ عَزَّوَجَلَّ حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي مالڪ ۽ مُختار بڻايو آهي، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم بِاذنِ الٰهي، پنهنجي اِختيار سان روزِ محشر بخشش سان نااُميد ٿيڻ وارن گنهگارن جي شَفاعت فرمائيندا۔ حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجي شانِ رِفعت نشان بيان فرمائن ٿا: (قيامت جي ڏينهن) سڀ کان پہريان آئون (پنهنجي مزار ِ فائضُ الاَنوار کان) ٻاہر تشريف آڻيندس، جڏهن ماڻهو گروھ جي صُورت ۾ ايندا ته آئون ئي انهن جو رہنُما ہوندس، جڏهن اهي (قيامت جي ہولناڪين جي سبب) خاموش ٿي ويندا، ته آئون ئي ان هن جو خطيب (خطبو پڙهڻ وارو) ہوندس، جڏهن اهي روڪيا ويندا ته آئون ئي انهن جو سِفارشي ہوندس، جڏهن اُهي نااُميد ٿي ويندا ته آئون ئي انهن کي خُوشخبري ٻُڌائڻ وارو ہوندس۔ بُزرگي ۽ (اللہ عَزَّوَجَلَّ جي) تمام خزانن جون چاٻيون، ان ڏينهن منهنجي هٿن ۾ هونديون ۽ آئون اولادِ آدم ۾ اللہ عَزَّوَجَلَّ جي ويجھو سڀ کان وڌيڪ بُزرگي وارو هوندس، هڪ ہزار(1000) خدمت گُذار منهنجي اِرد گرد ہوندا۔ (دارمی،باب ما اعطی النبی من الفضل،۱/۳۹، حدیث:۴۸)

سُبۡحٰن اللّٰه عَزَّوَجَلَّ! قُربان وڃو! اوَّلين ۽ آخرين جي سردار ۽ بِاِذنِ اللہ،

مالڪ ۽ مختار هئڻ جي باوُجُود به حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي شوقِ عبادت جو هي عالَم هو جو ڪثرتِ عبادت جي سبب نازڪ قدمن (پَيرن) تي وَرَم (سوڄ) جا آثار ظاهر ٿي ويندا هئا ۽ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اُمّت جي گُناھگارن جي بخشش جي خاطر آھ و زاري فرمائيندا هئا۔ ان مان بِالخُصُوص ان نوجوانن جي لاءِ نصيحت جا مدني گُل مَوجُود آهن جن جو دل عبادت جي طرف مائل نه ٿو ٿئي ۽ اهي سڄي سڄي رات فُضُول ۽ بيڪار ڪمن ۾ برباد ڪري ڇڏيندا آهن، تنهنڪري اهڙن نوجوانن جي خدمت ۾ عرض آهي ته خدارا! مُحسنِ انسانيت، غمخوارِ اُمَّت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مبارڪ لڙڪن کي ياد ڪريو، دُنيا ۽ آخرت ۾ڪاميابي حاصل ڪرڻ جي لاءِ اَحڪامِ خُداوَندي جي بجا آوري، نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سُنَّتن جي پيروي ۽ اُخروي اِنعامات حاصل ڪرڻ جي حرص ۾ خوب خوب نيڪيون ڪريو۔

ياد رکو! جواني ۾ عبادت جي توفيق نصيب ٿيڻ، ڪنهن نعمتِ عُظميٰ (وڏي نعمت) کان گھٽ ناهي، ڇو جو جواني جي چانئٺ تي قدم رکندي ئي انسان، شيطان جي خطرناڪ چالن، نَفس جي ناجائز خواہشن، بُرن دوستن جي صحبت، دُنيوي مُستقبل بہتر بنائڻ جي فڪرن ۽ فاني دُنيا جي رنگينين ۾ گُم ٿي ڪري دولت ڪمائڻ جي ناجائز طريقن جي سبب گُناهن جي اونڌاھ ۾ ڀٽڪندو ڦرندو آهي ۽ عبادت ۽ رِياضت جي طرف مائل نه ٿي سگھندو۔ ياد رکو! اسان کي تمام ئي مُختصر وقت جي لاءِ دُنيا ۾ موڪليو ويو آهي ۽ ان وَقفي ۾ قبر ۽ حشر جي طويل ترين مُعاملن جي لاءِ تياري به ڪرڻي

آهي، تنهنڪري سمجھدار اهو ئي آهي جيڪو ان مختصر وَقت کي غنيمت ڄاڻيندي قبر ۽ حشر جي تياري ۾ مشغول ٿي وڃي ۽ هڪ گھڙي به پنهنجو قيمتي وَقت فُضُول ڪمن ۾ برباد نه ڪري، ڇو جو معلوم ناهي ته ايندڙ وقت ۾ اهو زِندہ به رهندو يا موت ان کي ڊگھي عرصي جي لاءِ گھِري ننڊ سُمهاري ڇڏيندو۔ تنهنڪري جواني ۽ زندگي کي غنيمت ڄاڻيندي نيڪين ۾ مشغول ٿي وڃو۔

بابُ الۡاسلام (سنڌ) جي مشہور شہر زَم زَم نگر (حيدرآباد) جي مُقيم اسلامي ڀائر (عُمر تقريبًا 28سال) جو بيان آهي: آئون نَائين ڪلاس ۾ پڙھندو هئس ۽ حُصُولِ دُنيا جي جُسۡتجو ۾ مگن هئس۔ منهنجي علائقي جي اسلامي ڀائرن سان منهنجي واقفيت وڌي ته اهي مون کي نيڪي جي دعوت ڏئي ڪري مَسۡجد وٺي آيا۔ جڏهن آئون نَماز پڙھي ڪري مَسۡجد مان نڪرڻ لڳس ته هڪ خَيرخَواھ اسلامي ڀائر (جيڪو مَسۡجد جي دروازي جي ويجھو بيٺل هو) مون کي دَرس ۾ شِرڪت جي گذارش ڪئي۔ آئون دَرسِ فَيضانِ سُنَّت ۾ ويهي رهيس۔ هي منهنجي دعوتِ اسلامي سان رَفاقَت (وابستگي) جي شروعات هئي۔ پوءِ مون اسلامي ڀائرن جي اِنفِرادي ڪوشش مَدرَسَةُ المدينه (بالِغان) ۾ پڙھڻ شُروع ڪري ڇڏيو۔ ہفتيوار سُنَّتن ڀري اجتماع ۾ شِرڪت ڪئي، جتي منهنجي جَذبي کي مديني جا ٻارنهن 12 چنڊ لڳي ويا ۔ ڪجھ ہفتن کان پوءِ اميرِ اہلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جو بيان بَذريعهء ٽيليفون رلي ٿيو۔ جنهن کي اجتماع ۾ شريڪ اسلامي ڀائرن وڏي غور سان ٻُڌو۔ بيان کان پوءِ اميرِ اہلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جن اجتماعي توبھ ۽ بيعت ڪرائي ته آئون به عطّاري بڻجي ويس۔ پوءِ جيئن جيئن وَقت گُذرندو ويو آئون مَدَني ماحول ۾

رَنگجندو ويس ۔ آئون پنهنجي ذيلي حلقي ۾ ٻن سالن تائين مَدَني ڪم ڪندو رہيس۔ مَدَني ماحول جي بَرَڪت سان مون کي فِقۡهي مَسائل سان وڏي دِلچَسپي ٿي۔ عِلمِ دِين سکڻ جي جَذبي جي تحت مون 1999 ء ۾ جامعةُ المدينه (فيضانِ عثمان غني گلستانِ جوہر باب المدينه ڪراچي) ۾ دَرس ِ نَظامي ۾ داخلا ورتي۔ دَرسِ نِظامي سان گڏوگڏ مَدَني ڪم به استِقامت سان ڪرڻ جي سَعادَت حاصل ٿيندي رهي۔ آئون پنهنجي جامِعه ۾مَدَني ڪمن جو خادِم (ذِمه دار) هئس۔ 2005ء ۾ فارِغُ التَّحصِيل هئڻ تي مٺي مٺي مُرشدِ ڪريم اميرِ اہلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جن منهنجي مٿي تي دَستارِ فَضِيلت جي طور تي سبز سبز عمامو شَرِيف سجايو۔ اميرِ اہلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي فيضان جي صَدَقي اڄ به دعوتِ اسلامي جي مدني ماحول ۾ رهندي مُختلف شُعبه جات ۽ مدني ڪمن ۾ نيڪي جي دعوت جي ڌُوم مَچائڻ جي سَعادَت حاصل ٿي رهي آهي۔
