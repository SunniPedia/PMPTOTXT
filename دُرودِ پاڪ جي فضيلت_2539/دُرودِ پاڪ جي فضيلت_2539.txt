دعوتِ اسلامي جي اِشاعتي اِداري مڪتبة المدينہ جي مطبوعه ” عُيونُ الحڪايات “ جلد نمبر 2 جي صفحي نمبر 396 تي حضرتِ سَيدُنا عبدُ الرَّحمٰن بن علي جوزي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن هڪ دِلچَسْپ ۽ سَبَق آموز حڪايت نَقْل ڪئي آهي ته ڪنهن گھر ۾ هڪ عجيب و غريب نانگ رہندو هو جيڪو روزانو سَون جو هڪ آنو (بيضو) ڏيندو هو۔ گھر جو مالڪ مُفت جي دَولت ملڻ تي تمام خُوش هو۔ هن شخص جي گھروارن کي تاڪيد ڪيل هئي ته هي ڳالھ ڪنهن کي نه ٻڌائين۔ ڪيئي مهينن تائين هي سِلْسِلو ائين ئي هلندو رهيو۔ هڪ ڏينهن نانگ پنهنجي ٻِرُ (سُوراخ) مان نِڪتو ۽ سندس ٻَڪري کي ڏنگي ڇڏيو۔ اُن جو زَہْر اهڙو خطرناڪ هو جو ڏسندي ئي ڏسندي ٻَڪري جي مَوْت

واقع ٿي وئي۔ هي ڏسي ڪري سندس گھروارا گھٻرائجي ويا پر اُن شخص هي چئي ڪري اُنهن کي تسلّي ڏني ته اسان کي نانگ کان ملڻ وارو نفعو، ٻڪري جي قيمت کان وڌيڪ آهي تنهنڪري پريشان ٿيڻ جي ضرورت ناهي۔ ڪجھ عرصي کان پوءِ نانگ سندس پَالتو گڏھ کي ڏنگي ورتو جيڪو يڪدم مَري ويو۔ هاڻي ته اهو شخص به گھٻرائجي ويو پر لالچ وچان اُن فورًا خُود تي قابو ڪيو ۽ چوڻ لڳو: آئون ڏسي رهيو آهيان ته هي نانگ اسان کي مُسلسل نُقصان پہچائي رهيو آهي پر جيستائين هي نُقصان جانورن تائين مَحْدُود آهي، آئون صبر ڪندس، اِن کان بعد نه۔ پوءِ ٻن (2) سالن جو عرصو گُذري ويو پر نانگ ڪنهن کي به نه ڏنگيو ، گھروارن به پنهنجي جانورن جي نُقْصان کي وساري ڇڏيو۔ پوءِ هڪ ڏينهن نانگ سندس غُلام کي ڏنگيو ۔ اُن ويچاري مَدَد جي لاءِ پنهنجي مالڪ کي سَڏيو پر اِن کان پهرين جو مالڪ اُن تائين پهچي ها، زَہْر جي سبب غُلام جو جسم ڦاٽي چڪو هو۔ هاڻي اهو شخص پريشان ٿي ڪري چوڻ لڳو: ان نانگ جو زہر ته تمام خطرناڪ آهي، ان جنهن جنهن کي ڏنگيو اهو يڪدم مَوْت جو شڪار ٿي ويو، هاڻي ڪٿي منهنجي گھروارن مان ڪنهن کي نه ڏنگي وٺي ۔ ڪيترائي ڏينهن انهي پريشاني ۾ گُذري ويا ته اِن نانگ جو ڇا ڪيو وڃي؟ دَولت جي حَرص هڪ ڀيري ٻيهر اُن شخص جي اکين تي پَٽي ٻَڌي ڇڏي ۽ هن شخص اهو چئي ڪري پنهنجي گھروارن کي مُطمَئِن ڪري ڇڏيو ته جيڪڏهن اُن نانگ جي سبب اسان کي نُقْصان ٿي رهيو آهي پر سون جا آنا (بيضا) به ته ملي رهيا آهن تنهنڪري اسان کي وڌيڪ پريشان نه ٿيڻ گھرجي ۔ ڪجھ ڏينهن کان بعد نانگ سندس پُٽ کي به ڏَنگي ڇڏيو۔ يڪدم تِرْياق (يعني زہر جو اثَر ختم ڪرڻ واري دَوا) جو اِنتِظام ڪيو ويو پر اُن جو ڪجھ فائدو نه ٿيو ۽ اُن نينگر

جي مَوْت واقع ٿي وئي۔ جَوان پُٽ جي مَوت ماءُ پيءُ تي مصيبت بڻجي آئي، ۽ هاڻي اهي ٻئي غُصّي وچان چوڻ لڳا: هاڻي ان نانگ ۾ ڪا به ڀلائي ناهي، بہتر هي آهي ته اِن ايذائيندڙ کي يڪدم قتل ڪيو وڃي۔ نانگ جڏهن انهن جون ڳالهيون ٻُڌيون ته ڪجھ ڏينهن جي لاءِ غائب ٿي ويو، اِهڙي طرح اُنهن کي سَون جو آنو به نه مِلي سگھيو۔ جڏهن ڪافي عرصو گذري ويو ته سَون جو آنو (بيضو) نه ملڻ جي سبب هن شخص جي لالچي طبيعت ۾ بي چيني ٿيڻ لڳي اهڙي طرح اهي ٻئي زال مڙس نانگ جي ٻِرُ جي ويجھو آيا ۽ دونهين دُکائي ڪري ڄڻ ته انهن نانگ کي صُلح جو پيغام ڏنو۔ حَيرت انگيز طور تي اهو نانگ واپس اچي ويو ۽ اُنهن کي ٻيهر سَون جو آنو (بيضو) ملڻ لڳو۔ مال ۽ دَولت جي حرْص انڌو ڪري ڇڏيو هو ۽ اهي پنهنجي پُٽ ۽ غُلام جي مَوت کي به وساري ويٺا۔ پوءِ هڪ ڏينهن نانگ سندس گھرواريءَ کي سمهندي ڏنگي ورتو، عورت رڙيون ڪيون ته مڙس يڪدم تِرْياق جي ذريعي اُن جو عِلاج ڪرڻ چاهيو پر ٿوري ئي دير ۾ اُن جي گھرواريءَ به تڙپي تڙپي ڪري جان ڏني۔ هاڻي اهو لالچي شخص اڪيلو ٿيو ته اُن نانگ واري ڳالھ پنهنجي ڀائرن ۽ دوستن کي ٻڌائي ڇڏي۔ سڀني اهو مشوَرو ڏنو ته تو وڏي غَلَطي ڪئي، اڃا به وقت آهي سُڌري وڃ ۽ جيترو جلدي ٿي سگھي اُن خطرناڪ نانگ کي ماري ڇڏ۔ بهرحال گھر اچي ڪري اهو شخص نانگ کي مارڻ جي تَڪ ۾ ويهي رهيو۔ هڪدم اُن کي نانگ جي ٻِرُ جي ويجھو هڪ قيمتي موتي نظر آيو جنهن کي ڏسي ڪري اُن جي لالچي طبيعت خُوش ٿي وئي۔ دَولت جي ہَوَس سندس سڀ ڪجھ وِسرائي ڇڏيو هو، اهو چوڻ لڳو: وقت طبيعتن کي بدلائي ڇڏيندو آهي، يقينًا ان نانگ جي طبيعت به بدلجي وئي هوندي جو هي سَون جي آنن (بيضن) جي بجائي هاڻي

حضرت سَيِّدُنا سُفيان ثَوري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا ته حضرتِ سَيدُنا ابوذَر غِفَاري رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن ڪعبي جي ويجھو ٿي ڪري سَڏ ڏنو: اي انسانو! آئون جُندب غِفاري آهيان، هيڏانهن اچو پنهنجي خيرخواھ شَفِيق ڀاءُ جي ويجھو، جڏهن سڀ ماڻهون سندن اِرد گِرد جمع ٿي ويا ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن اُنهن کان پڇيو: هي ٻڌايو ته جيڪڏهن توهان مان ڪنهن جو سَفَر جو اِرادو ہجي ته ڇا اهو پاڻ سان زادِ راھ (سفر لاءِ سامان) نه کڻندو جيڪو سندس ڪم اچي ۽ منزِلِ مَقْصُود تائين پہچائي؟ سڀني عَرض ڪيو: جي ہا، ڇو نه ، ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن (اُنهن کي دُنيا جي حرْص کان بچائڻ ۽ آخرت جي حرْص جي ترغيب ڏيارڻ جي لاءِ نصيحت ڪندي) اِرشاد فرمايو: يقينًا سَفَرِ آخرت اُن سَفر کان تمام طويل (وڏو) آهي جنهن جو توهان (هتي) اِرادو ڪريو ٿا، تنهنڪري (اُن جي لاءِ) اُهي شيون کڻجو جيڪي توهان کي فائدو ڏين۔ ماڻهن پڇيو ته اهي ڪهڙيون شيون آهن؟ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن فرمايو: عظيم مَقصدن جي لاءِ حج ڪريو، قيامت جي ڊگھي ڏينهن کي سامهون رکندي سخت گرمي ۾ روزو رکو، قبر جي وحشت کان بچڻ جي لاءِ رات جي اونڌاهي ۾ نَفلَ پڙھو، (اُن) وڏي ڏينهن ۾بيهڻ جو خيال ڪندي چڱي ڳالھ ڪريو ۽ بُري ڳالھ کان پاسو ڪريو، قيامت جي ڏُکيائي کان بچڻ جي اُمّيد تي پنهنجي مال مان صَدَقو اَدَا ڪريو، دُنيا ۾ ٻه (2) مَجلِسون اِختيار ڪريو هڪ طلبِ حلال جي

لاءِ ۽ ٻي طلبِ آخرت جي لاءِ اِن کان علاوہ ٽين مَجلس جو اِرادو نه ڪريو ڇو جو اها فائدي جي بجائي توهان کي نُقصان پہچائيندي۔ اِهڙي طرح مال جا ٻه حِصّا ڪريو، هڪ حِصّو پنهنجي اہل و عيال تي حلال طريقي سان خرچ ڪريو ۽ ٻيو پنهنجي آخرت جي لاءِ اڳتي وڌائي ڇڏيو (يعني صَدَقو ڪري ڇڏيو) ان کان علاوہ ٽيون حِصّو نه ڪجو جو اهو توهان کي فائدي جي بجائي نُقصان پهچائيندو۔ وري بُلند آواز ۾سَڏيندي اِرشاد فرمايو: اي انسانو! حِرص (کان بچو جو اِن) ۾ توهان جي لاءِ ہلاڪت آهي، توهان حرص کي ڪڏهن پُورا نه ٿا ڪري سگھو۔ (صفۃ الصفوۃ،ابو ذر جُنْدُب بن جُنَادۃ،الجزء:۱، ۱/۳۰۱-۳۰۲، رقم: ۶۴)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! واقعي اسان مان ہر هڪ کي آخرت جو طَوِيل (ڊگھو) سَفَر درپيش آهي ۽ ڪاميابي سان منزلِ مَقْصُود تائين پہچڻ جي لاءِ سفر جي سامان جي صورت ۾ ڪثرت سان نيڪ عملن جي ضرورت آهي، جيڪڏهن قيامت جي ڏينهن نيڪين جو پلڙو وزني هئڻ جي لاءِ هڪ به نيڪي گھٽ ٿي وڃڻ صورت ۾ ماءُ پيءُ، ڀاءُ ڀيڻ ۽ ٻارن ٻچن مان ڪوبه ڪم نه ايندو، تنهن لاءِ جيڪڏهن حريص ٿيڻو آهي ته نيڪين جو حَرِيص بنجڻ گھرجي ته اسان جي پياري آقا، مديني واري مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن به وقتًا فوقتًا آخرت سان مَحَبَّت ڪرڻ ۽ نيڪين جو حَرِيص بنجڻ جي ترغيب اِرشاد فرمائي آهي۔ اچو ته انهي متعلق چار (4) فرامينِ مُصْطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻُڌون ٿا: 1. جيڪو شخص پنهنجي دُنيا سان مَحَبَّت ڪندو آهي اهو پنهنجي آخرت کي نُقْصان پہچائيندو آهي ۽ جيڪو شخص پنهنجي آخرت سان مَحَبَّت

ڪندو آهي اهو پنهنجي دُنيا کي نُقْصان پہچائيندو آهي، بس فنا ٿيڻ واري (دُنيا) تي باقي رہڻ واري (آخرت) کي ترجيح ڏيو۔ (مسند امام احمد،حدیث ابی موسی الاشعری،۷/۱۶۵،حدیث۱۹۷۱۷)

2. حُبُّ الدُّنۡيَا رَأۡسُ كُلِّ خَطِيۡئَةٍ يعني دُنيا جي مَحَبَّت ہر گُناھ جي اَصْل (جڙ) آهي۔ (مو سوعۃ لابن ابی الدنیا،کتاب ذم الدنیا،الجزء الاوّل،۵/۲۲،حدیث۹)

3. يَا عَجَبًا كُلَّ الۡعَجَبِ لِلۡمُصَدِّقِ بِدَارِ الۡخُلُوۡدِ وَهُوَ يَسۡعٰي لِدَارِ الۡغُرُوۡرِ يعني اُن شخص تي وڏو تَعَجب آهي جيڪو هميشه جي گھر (يعني آخرت) جي تَصْدِيق ڪندو آهي جڏهن ته اهو ڌوڪي واري گھر (يعني دُنيا) جي لاءِ ڪوشش ڪندو رهندو آهي۔ (مصنف ابن ابی شیبۃ، کتاب الزھد، باب ما ذکر عن نبینافی الزھد، ۸/۱۳۳، حدیث۶۱)

4. اَحۡرِصۡ عَلٰي مَا يَنۡفَعُكَ وَاسۡتَعِنۡ بِاللّٰهِ وَلَا تَعۡجِزۡ: يعني اُن شيء جي حرص ڪريو جيڪو توهان کي نفعو ڏئي، اللہ عَزَّوَجَلَّ کان مدد گُھرو ۽ عاجز (غافل) نه ٿي وڃو۔ (مسلم،کتاب القدر،باب فی الامر بالقوۃ…الخ، حدیث:۲۶۶۴،ص۱۴۳۲)

شارحِ مُسْلِم حضرت سَيدُنا علّامہ شَرَفُ الدِّين نَوَوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اِن آخري حديثِ پاڪ جي تَحت لکن ٿا: يعني اللہ عَزَّوَجَلَّ جي عبادت ۾ خُوب حرْص ڪريو ۽ اِن تي اِنعام جي لالچ رکو پر اِن عبادت ۾ به پنهنجي ڪوشش تي ڀَروَسو ڪرڻ جي بجائي اللہ تعاليٰ کان مدد گُھرو۔ (شرح نووی،باب الایمان للقدر و الاذعان لہ،الجزء:۱۶، ۸/۲۱۵ملخصاً)

مُفَسِّرِ شَہِير، حَڪِيمُ الاُمَّت مُفتي اَحمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: خيال رهي ته دُنياوي شين ۾ قَناعت ۽ صبْر سٺو آهي پر آخِرَت جي شين ۾ حِرْص ۽ بي صَبْري اعليٰ آهي، دِين جي ڪنهن دَرَجي تي پہچي ڪري قناعت نه ڪريو اڳيان وڌڻ جي ڪوشش ڪريو۔ (م رآۃ المناجیح،۷/۱۱۲)

سُبحٰن اللّٰه عَزَّوَجَلَّ! معلوم ٿيو ته ہر حِرص قَبِيح (بُرو) ناهي بلڪه جيڪو حرْص آخِرَت سان لڳاپيل معاملن تي مَبْنِي ہجي اها قابلِ تعريف آهي، گڏوگڏ هي به معلوم ٿيو ته انسان ڪيتريون ئي نيڪيون ڪري پر نيڪين جي حرص ۾ ڪمي نه اچڻ گھرجي، البت دُنياوي مُعاملات ۾ قناعت ڪرڻ بهترين شيء آهي۔ اچو ته! قناعت جي تعريف ۽ اُن جي فضيلت جي باري ۾ ٻُڌون ٿا۔

شيخُ الحديث حضرتِ علّامہ عبدُالمُصطَفيٰ اَعْظَمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا ته انسان کي جيڪو ڪجھ خُدا (عَزَّوَجَلَّ) جي طرف کان مِلي وڃي ان تي راضي رهندي زندگي گذاريندي حرص ۽ لالچ کي ڇڏي ڏيڻ کي قناعت چوندا آهن۔ (جنتی زیور،ص۱۳۶بتغیر قلیل) جڏهن ته علّامہ مِير سَيد شريف جُرجاني حَنَفي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه قناعت جي تعريف بيان ڪندي اِرشاد فرمائن ٿا: هِيَ السُّكُوۡنُ عِنۡدَ عَدَمِ الۡمَاۡلُوۡفَاتِ يعني روزمرّہ اِسْتِعْمال ٿيڻ وارين شين جي نه ہئڻ تي به راضي رہڻ قناعت آهي۔ (التعريفات للجرجانی،باب القاف،تحت اللفظ:القناعۃ، ص۱۲۶)

مٺا مٺا اسلامي ڀائرو! ياد رکو ته اَصل مالداري هي ناهي ته انسان وٽ گھڻو مال ۽ دولت ہجي بلڪه اَصل مالداري ته هي آهي ته جيڪڏهن مال هجي پر اُن تي قناعت حاصِل ہجي، ڇو جو مال ۽ دولت وارو شخص اهڙو مالدار آهي جو اُن کي ڪيترو ئي مال ڇو نه مِلي وڃي پر مزيد مال جي خواہش باقي رہندي آهي ۽ اِن مال ۾ گھٽتائي به ايندي رهندي آهي جڏهن ته قناعت ڪرڻ وارو شخص اهڙو مالدار آهي جو سندس ڪيترو ئي گھٽ مال ڇو نه ہجي پر اهو مزيد (گھڻي) جي خواهش ناهي ڪندو گڏوگڏ قناعت جي دولت ۾ ڪمي به نه ايندي آهي، بہرحال قناعت بهترين عادت آهي ۽ جنهن کي نصيب ٿي وڃي اهو دُنيا ۽ آخرت ۾ ڪامياب آهي۔ اچو ته قناعت جي رغبت پنهنجي دل ۾ پيدا ڪرڻ جي لاءِ انهي متعلق ڪجھ رِوايتون ٻُڌون:

حضرتِ سَيدُنا موسيٰ ڪليمُ اللہ عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن پنهنجي رَبّ عَزَّوَجَلَّ جي بارگاھ ۾ عَرض ڪيو، اي اللہ عَزَّوَجَلَّ! تنهنجي ٻانهن ۾ سڀ کان وڌيڪ مالدار ڪير آهي؟ اللہ عَزَّوَجَلَّ اِرشاد فرمايو: اهو شخص جيڪو منهنجي ڏنل شيء تي سڀ کان وڌيڪ قناعت ڪرڻ وارو آهي۔ (ابنِ عساکر ،موسیٰ بن عمران بن یصھر بن قامث،۶۱/ ۱۳۹،رقم:۷۷۴۱)

نبين جي سُلطان، رحمتِ عالميان صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ ذيشان آهي: لَيۡسَ الۡغِنٰى عَنۡ كَثۡرَةِ الۡعَرَضِ ، وَلَكِنَّ الۡغِنٰى غِنَى النَّفۡسِ يعني مالداري هي ناهي ته سازوسامان جي ڪثرت ہجي بلڪه اَصل مالداري ته دل جو مالدارہئڻ آهي ۔ (صحیح البخاری،کتاب الرقاق،باب الغنی غنی النفس،۴/ ۲۳۳،حدیث۶۴۴۶)

حضرتِ سَيدُنا عبدُاللہ بن عَمْرو رَضِیَ اللهُ تَعَالٰی عَنْهُمَا فرمائن ٿا،نبي اڪرم، نُورِ مُجَسّم صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم جن اِرشاد فرمايو: قَدۡ اَفۡلَحَ مَنۡ اَسۡلَمَ وَرُزِقَ كَفَافًا وَقَنَّعَهُ اللّٰهُ بِمَا اٰتَاهُ يعني بي شڪ ڪامياب ٿي ويو اهو شخص جنهن اِسلام آندو ۽ اُن کي بَقَدْرِ ڪفايت (ضرورت سارو) رزق ڏنو ويو ۽ اللہ عَزَّوَجَلَّ اُن کي جيڪو ڪجھ ڏنو اُن تي قناعت به عطا فرمائي۔

(مسلم ، کتاب الزکاۃ، باب فی الکفاف والقناعۃ، ص۵۲۴، حدیث: ۱۰۵۴)

رَحمتِ عالَم، نورِ مُجَسَّم صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم جن اِرشاد فرمايو: اَلۡقَنَاعَ ةُ كَنۡزٌ لَا يَفۡنٰي يعني قناعت ڪڏهن نه خَتم ٿيڻ وارو خزانو آهي۔ (الزھد الکبیر،ص۸۸،حدیث: ۱ ۰۴)

حضرتِ سَيدُنا ابُوہُريرہ رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان مَروِي آهي ته تاجدارِ رِسالت، شَہَنْشاہِ نُبوّت صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم جن دُعا ڪئي: اَللّٰهُمَّ اجۡعَلۡ رِزۡقَ اٰلِ مُحَمَّدٍ قُوۡتًا يعني اي اللہ عَزَّوَجَلَّ! محمد (صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم) جي آل کي صرف ضرورت جي مُطابق رِزق عطا فرماءِ۔ (مسلم، کتاب الزکاۃ، باب فی الکفاف والقناعۃ، ص۵۲۴، الحدیث:۱۰۵۴)

اَمِيرِ اَہلس ُ نّت، باني دعوتِ اسلامي حضرتِ علّامہ مولانا ابوبلال محمد الياس عطّار قادِري رَضَوِ ي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه فرمائن ٿا: پنهنجي دؤر جي جيّد عالم حضرتِ سيّدُنا خليل بصري عَـلَيْهِ رَحْـمَةُ الـلّٰـهِ الۡـقَـوِی جي خدمت ۾ ” اهواز“ کان امير (يعني حڪمران) سليمان بن علي جو خصوصي نمائندو حاضر ٿي ڪري عرض گذار ٿيو. شهزادن جي تعليم ۽ تربيـت لاءِ حاڪم توهان کي شاهي درٻار ۾ گهرايو آهي، حضرتِ سيّدُنا خليل بصري عَـلَيْهِ رَحْـمَةُ الـلّٰـهِ الۡـقَـوِی سڪل مانيءَ جو ٽُڪر ڏيکاريندي جواب ارشاد فرمايو، ”مون وٽ جيستائين سڪل مانيءَ جو هڪ ٽڪر موجود آهي مون کي شاهي درٻار جي نوڪريءَ جي ڪا ضرورت ناهي.“

اهڙي طرح مُفتي دعوتِ اسلامي، مُفتي محمد فاروق عطاري المدني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي قناعت جو جذبو به حيرت انگيز ۽ پيروي جي لائق آهي۔ جامعةُ المدينه ہجي يا دارُالاِفتاء، مُفتي دعوتِ اسلامي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن ڪڏهن به اِجاري (پگھار) وڌائڻ جو مُطالَبَو نه ڪيو۔ مرڪزي مجلسِ شوريٰ جي نگران مد ظلہ العالی جو بيان آهي ته ڪجھ ئي وقت پهرين (يعني سندن وفات کان ڪجھ عرصو پهرين) سندن مُشاہَرو (يعني وظيفو) وڌيو هو ته هي منهنجي گھر خود تشريف وٺي آيا۔ انتہائي پريشاني جي عالَم ۾ هئا۔ مون کي فرمائڻ لڳا ته منهنجو اِجارو (پگھار) ڪا في وڌي وئي آهي، مون کي اِن زائد رقم جي حاجت نه آهي، تنهنڪري مون تي مهرباني ڪئي وڃي ۽ منهنجو مُشاہَرو (يعني وظيفو) نه وڌايو وڃي۔ پنهنجي اِنتقال کان ڪجھ عرصي پهرين مُفتي دعوتِ اسلامي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ج ن پنهنجي اسڪوٽر (موٽرسائيڪل) ۽ ليپ ٽاپ (Laptop)، ڪمپيوٽر وغيرہ سڀ وڪڻي ڇڏيو هو ۽ فرمايو ته هاڻي مون کي اِن جي

سُبحٰن اللّٰه عَزَّوَجَلَّ! توهان ڏٺو ته اللہ عَزَّوَجَلَّ جي نيڪ ٻانهن جي سوچ ڪيتري قدر اعلٰي هوندي آهي جو جيڪڏهن سندن جائز طريقي سان به ضرورت کان وڌيڪ مال وغيرہ مِلي رہيو هوندو آهي ته پريشان ٿي ويندا آهن ۽ حتّي الاِمڪان دنيا جي مال کي پنهنجو پاڻ کان پَري رکندا آهن۔ اچو ته اِنهي متعلق شيخِ طريقت، اَمِيرِ اہلسُنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي مَدَني سوچ ۽ دنيا جي مال کان بي رغبتي جي باري ۾ به ٻُڌون ٿا:

شيخِ طريقت، اَمِيرِ اَہلسنّت ،باني دعوتِ اسلامي حضرتِ علّامه مَولانا ابوبلال محمد الياس عطار قادِري رَضَوِي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي ڪُرتي تي سيني جي طرف ٻه کِيسا ہوندا آهن۔ مِسواڪ(ڏندڻ) شريف رکڻ جي لاءِ پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه پنهنجي کاٻي سائيڊ (يعني دل جي طرف) واري کِيسي جي پاسي ۾ هڪ ننڍڙو کِيسو ٺاهرائيندا آهن۔ اِن جو سبب پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جن هي اِرشاد فر مايو ته آئون چاہيان ٿو ته هي سُنّت کي اَدَا ڪرڻ وارو (يعني مِسواڪ) منهنجي دل جي ويجھو رهي۔ اِن جي بَرعَڪس دُنيوِي دَولت کان بي رَغبَتي جو اندازو اِن ڳالھ سان لڳايو ته پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه کي ڏٺو ويو آهي ته جڏهن ڪنهن ضرورت جي وقت کِيسي ۾ رَقَم رکڻي پوندي آهي ته ساڄي پاسي واري کِيسي ۾ رکندا آهن۔ اِن جي حڪمت دَريافت ڪرڻ تي فرمايائون: آئون کاٻي پاسي واري کيسي ۾رَقَم اِن لاءِ نه

فرمان مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ؛ ] 1 [ جتي ڪثرت سان استعمال ڪيو جو انسان کي جيستائين جوتي پاتل هوندي آهي ڄڻ اهو سوار آهي (يعني ٿڪاوٽ گھٽ ٿيندي) (مسلم ص 1161 حديث 2096) ] 2 [ جُتي پائڻ کان پهرين ڇنڊي وٺو ته جيئن ڪو جيت يا پٿري وغيره هجي ته نڪري وڃي ] 3 [ پهرين ساڄو جوتو پايو پوءِ کاٻو ۽ لاهڻ وقت پهرين کاٻو ۽ پوءِ ساڄو۔ نبي پاڪ، صاحب لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان عاليشان آهي: جڏهن توهان مان ڪو جتي پائي ته ساڄي کان ابتدا ڪري ۽ جڏهن لاهي ته کاٻي کان ابتدا ڪرڻ گھرجي ته جيئن ساڄو پير پائڻ ۾ پهرين ۽ لاهڻ وقت آخر ۾ رهي. (بخاري ج 4 ص 65 حديث 5855) ] 4 [ مرد مرداڻو ۽ عورت زنانو جوتو استعمال ڪري ] 5 [ ڪنهن حضرت سيِّدتُنَا عائشه صديقه رَضِیَ اللهُ تَعَالٰی عَنْهَا جن کي عرض ڪيو؛ هڪ عورت (مردن جيان) جوتا پائيندي آهي، انهن فرمايو؛ رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مرداڻي عورت تي لعنت فرمائي آهي۔ (سنن ابي داؤد ج 4 ص 84 حديث 4099) صدرُالشَّريعه، بدرالطَّريقه مفتي محمد امجد علي اعظمي عَلَیهِ رَحْمَةُ اﷲِ القوي بهار شريعت ج 3 صفحي 422 تي فرمائن ٿا؛ يعني عورتن کي مرداڻا جوتا نه پائڻ کپن۔ بلڪ اُهي تمام ڳالهيون جن ۾ مردن ۽ عورتن ۾ امتياز ٿيندو آهي انهن ۾ هر هڪ کي ٻين جي وضع قطع اختيار ڪرڻ (يعني نقل ڪرڻ) جي ممانعت آهي، نه مرد عورت جي وضع اختيار ڪري نه عورت مرد جي۔ ] 6 [ جڏهن ويهو ته جوتا لاهي ڇڏيو جو ان سان پيرن کي آرام ملندو۔

هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جا شايع ٿيل ٻه ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهو۔ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي۔

تری سُنّتوں پہ چل کر میری روح جب نکل کر

چلے تو گلے لگانا مدنی مدینے والے

اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ مان گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن .

(افضل الصلوات علی سید السادات، الصلاۃ السادسۃ والخمسون، ص151ملخصًا)

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سيدُنا انس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. (افضل الصلوات علی سید السادات، الصلاۃ الحادیۃ عشرۃ، ص 65)

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. (القول البدیع، الباب الثانی، ص277،)

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. (افضل الصلوات علی سید السادات، الصلاۃ الثانیۃ والخمسون، ص149)

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِياللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته ائين پڙهندو آهي.

(القول البدیع، الباب الاول، ص125)

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي.

(ا لترغیب والترهیب، ج ۲ ، ص ۳۲۹ ، حدیث ۳۱)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سيدُنا ابنِ عباس رَضِي اللّٰہُ تَعَالٰي عَنہُمَا کان رِوايت آهي ته سرڪارِ مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن.

(مجمع الزوائد،ڪتاب الادعیۃ،باب فی ڪیفیۃ الصلاۃ…الخ،10/254،حدیث:17305)

﴿ 2 ﴾ هر رات عبادت ۾ گذارڻ جو آسان نسخو

غرائب القرآن صفحه 187 تي هڪ روايت نقل ڪئي وئي آهي ته جيڪو شخص رات ۾ ٽي ڀيرا هي دعا پڙهندو ته ان ڄڻ شبِ قدر کي حاصل ڪري ورتو لهٰذا هر رات هي دعا پڙهڻ گهرجي. دعا هي آهي:

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

”يعني خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي.“

(فيضانِ سنّت ج اوّل، ص 1163، 1164)
