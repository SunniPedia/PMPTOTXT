الْـعَـالِـيَـه جي رِسالي "163 مَدَني گل" مان مِسواڪ بابت مدني گل ٻڌون ٿا۔ پهرين ٻه فرمان مصطفيٰ صَلَّى الله تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ملاحظه فرمايو: ٭ ٻه رڪعتون مسواڪ (يعني ڏندڻ) ڪري پڙهڻ بغير مِسواڪ جي 70 رڪعتن کان افضل آهن ـ (الترغيب والترهيب ج 1 ص 102 حديث 18) ٭ مِسواڪ جو استعمال پاڻ تي لازم ڪري ڇڏيو ڇو ته هي وات جي صفائي ۽ ربّ تعالٰي جي رضا جو سبب آهي . (مسند امام احمد بن حنبل ج 2 ص 438 حديث 5869) دعوت اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل ڪتاب ”بهار شريعت“ جلد پهريون صفحي 288 صدرُالشّريعه، بدرُالطريقه، حضرت علامه مولانا مفتي امجد علي اعظمي رَحْمَةُ الله تَعَالٰی عَلَيْه لکن ٿا، مشائخ ڪرام رَحْمَۃُ اللّٰہ ِتَعَالٰی عَلَیْہِمْ اَجْمَعِیْن فرمائن ٿا: جيڪو شخص مسواڪ جو عادي هجي ته مرڻ وقت ان کي ڪلمو نصيب ٿيندو، ۽ جيڪو آفيم کائي ٿو مرڻ وقت ان کي ڪلمو نصيب نه ٿيندو ٭ حضرت سيدنا ابن عباس رَضِىَ الله تَعَالٰی عَنْهُ کان روايت آهي ته مسواڪ ۾ ڏهه خوبيون آهن: (ڪجهه هي آهن) وات کي صاف ڪري ٿو، مسوڙن کي مضبوط ڪري ٿو، نظر تيز ڪري ٿو، بلغم ختم ڪري ٿو، وات جي ڌپ ختم ڪري ٿو، سنت جي موافق آهي، فرشتا خوش ٿي ندا آهن ، ربّ راضي ٿئي ٿو، ٭ سيّدنا امام شافعي رَحْمَةُ الله تَعَالٰی عَلَيْه فرمائن ٿا: چار شيون عقل وڌائينديون آهن: فضول ڳالهين کان پرهيز، مسواڪ جو استعمال، نيڪ ماڻهن جي صحبت، پنهنجي علم تي عمل ڪرڻ (احياءُالعلوم ج 2 ص 27) ٭ مسواڪ پيلو(کبڙ) يا زيتون يا نم وغيره جي ڪَوڙي ڪاٺي جي هجي ٭ مسواڪ جي ٿولھ چيچ جي برابر هجي ٭ ڏندڻ جڏهن استعمال جي قابل نه رهي ته اڇلائي نه ڇڏيو، جو هي سنّت ادا ڪرڻ جو ذريعو آهي، ڪنهن جڳھ تي احتياط سان رکي ڇڏيو يا دفن ڪري ڇڏيو يا

﴿1﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي

پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿2﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿3﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([4])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءِ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

﴿2﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم ۽ ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى الله تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
