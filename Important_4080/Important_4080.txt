تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي رَضاعِي (کير شريڪ) ڀيڻ حضرت شَيمَاء رَضِىَ الله تَعَالٰى عَنها سان هيئن حُسنِ سُلُوڪ فرمايايو:(1) اُنهن جي لاءِ قِيام فرمايائون (يعني بيهي رهيا) (سبل الھدي و الرشاد، ۵/۳۳۳ ملتقطاً و ملخصاً) (2)پنهنجي مُبارَڪ چادر وڇائي ان تي ويهاريائون ۽ (3) هي اِرشاد فرمايائون: گُھر، توکي عطا ڪيو ويندو، سِفارش ڪر، تنهنجي سِفارش قَبول ڪئي ويندي۔ (دلائل النبوۃ للبيهقي،۵/۱۹۹ملتقطاً) هن مثالي ڪرم نوازي دَوران پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مُبارَڪ اکين مان لڙڪ وهي رهيا هئا، (4) هي به اِرشاد فرمايائون: جيڪڏهن چاهين ته عزّت ۽ احترام سان اسان وٽ رهه،(5) واپس موٽڻ لڳيون ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ٽي (3) غلام، هڪ لونڊي ۽ هڪ يا ٻه (2) اُٺ به عطا ڪيائون، (6) جڏهن جِعرَانَه ۾ ٻيهر رَضاعي ڀيڻ سان ملاقات ٿي ته رڍون ٻڪريون پڻ ڏنائون۔ (سبل الھدي و الرشاد،۵/۳۳۳ ملتقطاً و ملخصاً) اسان جي پياري نبي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو پنهنجي رَضاعي ڀيڻ سان حُسنِ سُلُوڪ هر ڀاء کي اهو اِحساس ڏيارڻ جي لاءِ ڪافي آهي ته ڀينرون ڪيتري قَدر وقت پيار ۽ حُسنِ سُلُوڪ جون حقدار آهن۔ حضرت سَيدُنا جابِر بن عبدُالله رَضِىَ اللهُ تَعَالٰى عَنهما صرف پنهنجي نو (9) يا ست (7) ڀينرن جي سار سنڀار لهڻ انهن جي ڦڻي چوٽي ۽ سٺي تَربِيت خاطر بيوه عورت سان نڪاح ڪيائون۔ (مسلم،ڪتاب الرضاع،باب استحباب نڪاح البڪر،ص۵۹۳،۵۹۴،حديث:۳۶۴۱، ۳۶۳۸ ملخصاً)

شيطان جو ڄار:

پياريون پياريون اسلامي ڀينرون! دنيا جي مَحَبَّت شيطان جو اهڙو ڄار آهي جو ان ۾ ڦاسجي انسان نيڪ ڪمن کان پري ٿي ويندو آهي، مثال طور پهريان ته مستحبّات کان پري ٿيندو آهي، پوءِ سُنَّتن کان غافل ٿيندو آهي، ان کان بعد فرض ۽ واجب ڇڏڻ جي عادت بڻجندي آهي ۽ پوءِ آهستي آهستي حرام ڪمن جو عادي بڻجي ويندو آهي۔ ڪوڙ ڳالهائڻ، غيبت ڪرڻ، ٻين جي دل ڏُکائڻ،

ڳائڻ وڄائڻ ٻڌڻ ۽ طرح طرح جي حرام ۽ ناجائز ڪمن ۾ زندگي گذرڻ لڳندي آهي۔ ان جي علاوه هي به ڏٺو ويو آهي ته عام طور دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو پنهنجن کي وساري ويهندو آهي، دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو مخلص دوستن کان محروم ٿي ويندو آهي، دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو غريبن کي حقير ۽ ڪمتر سمجھڻ لڳندو آهي، دُنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو ڪنجوس ٿي ويندو آهي، دُنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو تَڪبُّر جي آفت ۾ مبتلا ٿي ويندو آهي،دنيا ۾ ضرورت کان زِياده مشغول رهڻ واري تي نصيحت اثر ناهي ڪندي، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو حلال ۽ حرام ۾ تميز ناهي ڪري سگھندو، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو حُقُوْقُ الله سان گڏوگڏ حُقُوقُ العِبَاد يعني ٻانهن جي حقن کان به غافل ٿي ويندو آهي، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو چورن، ڌاڙيلن ۽ لٽيرن جو شڪار ٿي ويندو آهي۔ مطلب ته دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو طرح طرح جي آفتن ۾ مبُتلا رهندو آهي۔دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو جيڪڏهن دنيا جي حقيقت کي سمجھي وٺي ته ڪڏهن به ان سان دل نه لڳائي ها۔ قرآنِ ڪريم جي مختلف آيت سڳورين ۾ دُنيا جي حقيقت کي بيان ڪيو ويو آهي جيئن سيپارو 27 سُورَۃُ الْحَدِيد جي آٓيت نمبر 20 ۾ ارشاد آهي:

اِعْلَمُوْۤا اَنَّمَا الْحَیٰوةُ الدُّنْیَا لَعِبٌ وَّ لَهْوٌ وَّ زِیْنَةٌ وَّ تَفَاخُرٌۢ بَیْنَكُمْ وَ تَكَاثُرٌ فِی الْاَمْوَالِ وَ الْاَوْلَادِؕ- (پ ۲۷ ،الحدید: ۲۰)

تَرجَمو ڪنز الايمان: يقين ڄاڻو ته دنيا جي زندگي ته فقط راند روند ئي آهي ۽ سينگار ۽ توهان جو هڪٻئي ۾ فخر ڪرڻ ۽ مال۽ اولاد ۾ هڪٻئي تي زيادتي جي خواهش ڪرڻ .

بيان ڪيل آيت سڳوري تحت تفسير صِراطُ الجِنان ۾ لکيل آهي: هن آيت ۾ دُنيا جي حقيقت بيان ڪئي پئي وڃي ته جيئن مسلمان ان طرف راغب نه ٿين ڇوته دُنيا تمام گھٽ نفعي واري ۽ جلد ختم ٿيڻ واري آهي۔ هن آيت ۾ الله پاڪ دُنيا جي باري ۾ پنج (5) شيون بيان ڪيون آهن (1،2)… دنيا جي زندگي ته صرف راند روند آهي جيڪو ٻارن جو ڪم آهي ۽ صرف ان جي حصول ۾ محنت ۽ مشقت ڪندو رهڻ وقت ضائع ڪرڻ کان سوا ٻيو ڪجھ نه آهي۔(3)… دنيا جي زندگي زِينت ۽ آرائش جو نالو آهي جيڪو عورتن جو شيوو آهي۔ (4،5)…دنيا جي زندگي پاڻ ۾ فَخر ۽ غرور ڪرڻ ۽ مال ۽ اولاد ۾ هڪ ٻئي تي زيادتي چاهڻ جو نالو آهي ۽ جڏهن دنيا جو هي حال آهي ۽ ان ۾ اهڙيون بُرائيون موجود آهن ته ان ۾ دل لڳائڻ ۽ ان جي حصول جي ڪوشش ڪندي رهڻ بدران انهن ڪمن ۾ مشغول هئڻ گھرجي، جن سان اُخروي زندگي سَنورجي سگھي۔ (صراط الجنان، 9/740)

دنيا ۽ مالِ دنيا جي مَحَبَّت ۾ انڌا ٿي رشتا ناطا ٽوڙڻ وارن کي بروزِ قِيامت ڪيتري قدر ذِلّت ۽ رُسوائي کي منهن ڏيڻو پوندو، اچو هن باري ۾ 2 عبرت آموز حڪايتون ٻڌو ۽ عبرت حاصل ڪريو، جيئن

نيري اکين واري بدصورت پوڙهي

حضرت سيدنا فُضَيل بن عِياض رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا، حضرت سيدنا عبدالله بن عباس رَضِىَ اللهُ تَعَالٰی عَنْهُما جن فرمايو؛ قيامت جي ڏينهن نيري اکين واري نهايت بدصورت پوڙهي جنهن جا ڏند اڳيان نڪتل هوندا ماڻهن جي سامهون ظاهر ٿيندي ۽ انهن کان پڇا ڪئي ويندي؛ هن کي سڃاڻو ٿا؟ ماڻهو چوندا؛ اسان ان جي سڃاڻپ کان الله پاڪ جي پناهه چاهيون ٿا، چيو ويندو؛ هي اها ئي دنيا آهي جنهن تي اوهان فخر ڪندا هئا، هن دنيا جي ڪري قطعِ رِحمي ڪندا يعني رشتيداريون ٽوڙيندا هئا، انهيءَ جي ڪري هڪ ٻئي سان حسد ۽ دشمني

ڪندا هئا، پوءِ ان (پوڙهيءَ جھڙي دنيا) کي جهنم ۾ وڌو ويندو ته رڙيون ڪندي؛ اي منهنجا پروردگار! منهنجي پيروي ڪرڻ وارا ۽ منهنجي جماعت ڪٿي آهي؟ الله پاڪ فرمائيندو؛ انهن کي به ان سان گڏ ڪيو. (موسوعۃلابن ابي الدنيا، ۵/ ۷۲ ،رقم:۱۲۳)

مالِ دنيا جون تباھ ڪاريون

حضرت سَيدُنا ابُوهرَيره رَضِىَ اللهُ تَعَالٰی عَنْه کان روايت آهي، نبي پاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن اِرشاد فرمايائون: زمين سون چاندي جي ٿنڀن جھڙا پنهنجي جگر جا ٽڪر ٻاهر ڪڍندي، قاتِل ڏسي چوندو: هن (مال ۽ دَولت) جي ڪري ته مون قَتل ڪيو هو، رشتيداري ٽوڙڻ وارو چوندو: هن (مال و دَولت) سبب مون رشتيداري ختم ڪئي هئي، چور ڏسي چوندو: هن مال جي ڪري ته منهنجو هٿ ڪپيو ويو هو ۔ پوءِ سڀئي ان مال کي ڇڏي ڏيندا ۽ ڪو ان مان ڪجھ نه کڻندو. (مسلم،ڪتاب الزڪاۃ،باب الترغيب في الصدقۃ…الخ،ص ۳۹۲، حديث:۱۰۱۳)

8مَدَني ڪمن مان هڪ مَدَني ڪم ”گهر درس “ به آهي

پياريون پياريون اسلامي ڀينرون! ٻڌو اوهان! دنيا ۽ مالِ دنيا جي مَحَبَّت ڪيتري قدر تباھ ڪندڙ آهي جيڪا انسان کي مائٽي ٽوڙڻ تي مجبور ڪندي آهي، لهٰذا اڃا به وَقت آهي، هوش ۾ اچو، دنيا ۽ مالِ دنيا جي مَحَبَّت کان پيڇو ڇڏايو، ٿوري ٿوري ڳالھ تي رشتيدارن کان رُسڻ جي غيرمناسب عادت کي پنهنجي اندر مان ڪڍي ڇڏيو، انهن جي لاءِ پنهنجي دل نرم رکو، اوهان جو ڪوبه عزيز اوهان کان معافي گھري ته پنهنجي دَولت يا مَنصَب تي ناز ڪرڻ بدران ان کي معافي سان نوازڻ ۾ دير نه ڪريو، اوهان چاهي لک پتي هجو يا ارب پتي پر ناراض رشتيدارن کان پاڻ اڳتي وڌي صُلح ڪري آخرت جي ثواب جا حقدار بڻجي وڃو، انهن جا حق ادا ڪرڻ ۾ غفلت کان ڪم نه وٺو، بِلاوَجه شرعي رشته ختم ڪرڻ کان هميشه جي لاءِ سچّي تَوبه ڪريو ۽ اها سوچ ماڻڻ جي لاءِ عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي جي مَدَني ماحول سان وابسته ٿي ذيلي حلقي جي 8 مَدَني ڪمن جو معمول بڻايو۔ ذيلي حلقي جي 8 مدني ڪمن مان هڪ مدني ڪم ”گهر درس“ به آهي گهر ۾ مدني ماحول بڻائڻ

جي لاءِ روزانو گهٽ ۾ گهٽ هڪ ڀيرو درس ِ فيضان سنت ڏيڻ يا ٻڌڻ جي ترڪيب بڻايو (جنهن ۾ نامحرم نه هجن) امير اهلسنت دامت برڪاتهم العاليہ جي تخريج ڪيل رسالن مان به حسب موقع درس ڏئي سگهجي ٿو (دورانيو 7 منٽ آهي)

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ گهر درس جي برڪت سان علم دين حاصل ڪرڻ جو ثواب ملندو آهي اهڙي طرح گهر درس به ظاهري ۽ باطني اخلاق سنوارڻ ۽ سٺي تربيت جو هڪ بهترين ذريعو آهي تنهنڪري اوهان سڀئي اسلامي ڀينرون گهر درس شروع ڪرڻ جي نيت ڪريو ۽ دعوت اسلامي جي مدني ماحول سان هميشه جي لاءِ وابسته ٿي وڃو. اچو ترغيب خاطر هڪ مدني بهار ٻڌو. جيئن ته

دل کي قرار نصيب ٿي ويو:

باب المدينه (ڪراچي) جي هڪ اسلامي ڀيڻ نمازون قضا ڪرڻ ۽ بي پردگيءَ جهڙن گناهن ۾ گرفتار هئي. افسوس!ان کي گناهه کي گناهه سمجهڻ جو به احساس نه هو ان کي دنياوي سهولتون موجود هجڻ جي باوجو دلي سڪون نصيب نه هو اها عجيب بي چيني ۽ گهٻراهٽ جو شڪار رهندي هئي الحمد لله عَزَّوَجَلَّ ان کي دلي سڪون هن طرح مليو جو ڪجهه اسلامي ڀينرن جي دعوت تي ان کي دعوت اسلامي جي هفته وار سنتن ڀري اجتماع ۾ شرڪت جي سعادت ملي اتي بيان ٻڌو پنهنجي ربّ قدير عَزَّوَجَلَّ جو ذڪر ڪيو ۽ دعا گهري ۽ روئيندي روئيندي پنهنجي گناهن کان توبهه ڪئي ته ان کي ائين محسوس ٿيو ڄڻ ان جي دل تان بار لهي ويو ۽ ان کي قرار نصيب ٿي ويوآهي. الحمدلله عَزَّوَجَلَّ انهيءَ اجتماع ۾ شرڪت جي برڪت سان اهي نه صرف مدني ماحول سان وابسته ٿيون بلڪ امير اهلسنت دامت برڪاتهم العاليہ سان بيعت ٿي ڪري عطّاريه به ٿي ويون. (مون مدني برقعو ڇو پاتو؟ص19)

حضرت سيدنا ابوهريره رَضِىَ اللهُ تَعَالٰی عَنْهُ هڪ ڀيري سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جون حديثون بيان ڪري رهيا هئا، انهيءَ دوران فرمايائون؛ هر قاطعِ رحم (يعني رشتيداري ٽوڙڻ وارو) اسان جي محفل مان هليو وڃي، هڪ نوجوان اُٿي ڪري پنهنجي پڦيءَ وٽ ويو جنهن سان ان جو ڪيئي سالن جو پراڻو جڳهڙو هو، جڏهن ٻئي هڪ ٻئي سان راضي ٿي ويا ته ان نوجوان کي پڦيءَ چيو؛ تون وڃي ان جو سبب پڇا ڪر ته آخر ائين ڇو ٿيو؟ (يعني سيدنا ابوهريره رَضِىَ اللهُ تَعَالٰی عَنْهُ جي اعلان جي حڪمت ڇا هئي؟) نوجوان حاضر ٿي جڏهن پڇا ڪئي ته حضرت سيدنا ابوهريره رَضِىَ اللهُ تَعَالٰی عَنْهُ فرمايائون ته مون حضور انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان هي ٻڌو آهي؛ جنهن قوم ۾ قاطع رحم (يعني رشتيداري ٽوڙڻ وارو) هجي ان (قوم) تي الله ڪريم جي رحمت نازل ناهي ٿيندي۔ (الزواجر،قطع الرحم،۲/۱۵۳)

مجلس”مَدَني اِنعامات “

پياريون پياريون اسلامي ڀينرون! ٻڌو اوهان! پهريان جا مسلمان ڪيتري قَدَر خوفِ خدا رکڻ وارا هئا! خوش نصيب نوجوان الله ڪريم جي ڊپ سبب هڪدم پنهنجي پڦي وٽ پاڻ حاضِر ٿي صُلح جي ترڪيب ڪري ورتي۔ سڀئي مسلمان غور ڪن ته خاندان ۾ ڪنهن ڪنهن سان اڻبڻت آهي، خبر پوڻ کان هاڻي جيڪڏهن شَرعي عذر نه هجي ته فوراً ناراض رشتيدارن کان ”صُلح و صفائي“ جي ترڪيب شُروع ڪريو۔ جيڪڏهن توهان کي جُھڪڻو به پوي ته بيشڪ رِضاءِ الٰهي جي لاءِ جُھڪي پئو۔ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ سَربُلندي ماڻينديون۔ فرمانِ مصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ آهي: مَنْ تَوَاضَعَ لِلّٰه رَفَعَه الله يعني جيڪو الله پاڪ جي لاءِ

عاجِزي ڪندو آهي الله ڪريم ان کي بُلندي عطا ڪندو آهي . (شعب الايمان،باب في حسن الخلق، ۶/۲۷۶، حديث: ۸۱۴۰) رشتيدارن سان حُسنِ سُلوڪ جي مَدَني سوچ بڻائڻ جو هڪ بهترين ذريعو عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي سان وابسته ٿيڻ به آهي۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوتِ اسلامي سڄي دنيا ۾ ڪم و پيش 104 شُعبن ۾ دينِ متين جي خدمت ۾ مصروفِ عمل آهي، انهن مان هڪ شعبو ”مجلس مدني انعامات“ به آهي۔ شيخِ طَرِيقَت،اَميرِاَهلسنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي خواهش مُطابق اسلامي ڀائرن، اسلامي ڀينرن، اسلامي ڀينرن ۽ جامعاتُ المدينه و مَدارِسُ المدينه جي شاگردن ۽ شاگردياڻين کي باعمل بنائڻ جي لاءِ ۽ انهن کي مدني انعامات تي عمل جي ترغيب ڏيارڻ لاءِ ”مجلسِ مدني انعامات“ جو قيام عمل ۾ آندو ويو ، شيخِ طريقت،اميرِ اهلسنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ فرمائن ٿا: ڪاش! ٻين فرض ۽ سُنتن جي بجا آوَري سان گڏوگڏ تمام اسلامي ڀائر ۽ اسلامي ڀينرون انهن مَدَني انعامات کي به پنهنجي زندگي جو دستورُ العَمَل بڻائي وٺن، ڪاش! تمام ذمّه دار اسلامي ڀينرون به پنهنجي پنهنجي حَلقي ۾ انهن (مَدَني انعامات جي رسالن) کي عام ڪن. ڪاش! هر اسلامي ڀيڻ پنهنجي قَبر ۽ آخِرت جي بهتري جي لاءِ انهن مَدَني اِنعامات کي اِخلاص سان اپنائي الله پاڪ جي فضل و ڪرم سان جَنّتُ الفردوس ۾ مَدَني حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّمَ جي پاڙيسري بڻجڻ جو عظيم ترين انعام ماڻي ۔ اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

صله رحمي جا مدني گل

پياريون پياريون اسلامي ڀينرون! بَيان کي ختم ڪندي صِله رِحمي جي باري ۾ ڪجھ مدني گل ٻُڌڻ جي سعادت حاصِل ڪريون ٿييون۔ پهريان ٻه 2 فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحظه ڪريو: (1) هر حُسنِ سُلُوڪ (Politeness) صدقو آهي غني سان هجي يا فقير سان . (مجمع الزوائد ، ڪتاب الزڪوٰۃ ، باب ڪل معروف صدقۃ ، ۳/۳۳۱، حڈيث: ۴۷۵۴) (2) جنهن والدين سان حُسنِ سُلوڪ ڪيو ان کي مبارَڪ هجي ته الله پاڪ ان جي عُمر وڌائي ڇڏي۔ (مستدرڪ،ڪتاب البروالصلۃ،باب من بروالديه…الخ،۵/۲۱۳، حديث: ۷۳۳۹) C صِلَه رِحم واجِب آهي ۽ قَطعِ رِحم حرام ۽ جهنَّم ۾ وٺي وڃڻ وارو ڪم آهي۔ (بهار شريعت، ۳/۵۵۸) C رشتيدارن سان سٺو سُلوڪ ان جو نالو ناهي ته هو سُلوڪ ڪري ته توهان به ڪريو، اها شيء ته حقيقت ۾ مُڪافا ۃ يعني اَدلو بَدلو

آهي ته ان توهان وٽ ڪا شيء موڪلي توهان به ان وٽ ڪا شيء موڪلي، اهو توهان وٽ آيو توهان ان وٽ هليا ويا، حقيقت ۾ صله رحمي هي آهي ته اهو ٽوڙي ته توهان ڳنڊيو، اهو توهان کان جدا ٿيڻ چاهي ٿو، تون ان سان مائٽي جي حقن جي رعايت (يعني لحاظ) ڪر (رَدُّالْمُحتار، ۹/۶۷۸) C صِلَه رِحمي جون مُختلِف صورَتون آهن، انهن کي هدِيو ۽ تحفو ڏيڻ ۽ جيڪڏهن انهن کي ڪنهن ڳالھ ۾ توهان جي اِمداد درڪار هجي ته ان ڪم ۾ اُن جي مدد ڪرڻ،اُنهن کي سلام ڪرڻ، اُنهن جي ملاقات لاءِ وڃڻ، انهن وٽ اٿڻ ويهڻ، انهن سان ڳالھ ٻولھ ڪرڻ، اُنهن سان لُطف و مهرباني سان پيش اچڻ . (ڪتاب الدُرَر الحڪام،۱/۳۲۳) C رشتيدارن سان ناغو ڪري ملندو رهي يعني هڪ ڏينهن ملڻ وڃي ٻي ڏينهن نه وڃي ته اهڙي طرح سان مَحَبَّت ۽ اُلفت وڌندي آهي، بلڪه قَرابَتدارن سان جُمعي جُمعي ملندو رهي يا مهيني ۾ هڪ ڀيرو۔ (ڪتاب الدُرَر الحڪام ، ۱/۳۲۳) C حق ۽ جائز ڳالهين ۾ قبيلي ۽ خاندان وارن کي مُتَّحد ٿيڻ گھرجي يعني جيڪڏهن رشتيدار حق تي هجن ته ٻين سان مقابلو ۽ اِظهارِ حق ۾ سڀئي مُتَّحِد ٿي ڪم ڪن۔ (ڪتاب الدُرَر الحڪام،۱/۳۲۳) C رشتيدار مائٽ حاجت پيش ڪري ته ردّ ڪرڻ (موٽائڻ) گناھ آهي، جڏهن پنهنجو ڪو رشتيدار ڪا حاجت پيش ڪري ته ان جي حاجت روائي ڪري، ان کي رَد ڪرڻ رِشتو ٽوڙڻ آهي۔ (ڪتاب الدُرَر الحڪام،۱/۳۲۳) صِلَه رِحمي جي حوالي سان وڌيڪ معلومات حاصِل ڪرڻ جي لاءِ شيخِ طريقت، اَميرِ اَهلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ”هٿو هٿ پڦي سان صلح ڪري ورتي“ جو مطالَعو ڪريو۔
