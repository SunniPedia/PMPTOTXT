نه زِيادہ طويل آهي نه ئي تمام مختصر، بلڪه متوسط ۽ جامع آهي ﴿ 5 ﴾ تفسير صِراطُ الجنان ۾ حتي الاِمڪان آسان انداز اختيار ڪيو ويو آهي ته جيئن عام اسلامي ڀائر به ان مان فائدو حاصل ڪري سگھن ﴿ 6 ﴾ تفسير صِراطُ الجنان ۾ اعمال جي اِصلا ح جي لاءِ مُعاشرتي بُرائين جو تذڪرو ۽ معاشرتي برائين جا عذابَ بيان ڪيا ويا آهن ﴿ 7 ﴾ تفسير صِراطُ الجنان ۾ جنّت جي اِنعامات جي حصول جي ترغيب ۽ جہنم جي عذابن جي ترہيب موجود آهي ﴿ 8 ﴾ تفسير صِراطُ الجنان ۾ باطني امراض جي متعلق به ڪنهن حد تائين ڪلام ڪيو ويو آهي ﴿ 9 ﴾ تفسير صِراطُ الجنان ۾ والدين، رِشتي دارن، يتيمن، پاڙيسرين وغيرہ سان چڱو سلوڪ ڪرڻ ۽ صِلهء رحمي ڪرڻ جي متعلق به اِصلاحي مواد موجود آهي ﴿ 10 ﴾ تفسير صِراطُ الجنان ۾ عقائدِ اہلسنّت ۽ معمولاتِ اہلسنّت جي دلائل سان گڏ وضاحت ڪئي وئي آهي ﴿ 11 ﴾ تفسير صِراطُ الجنان ۾ موقعي جي مناسبت سان حضورِ پُرنور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۽ صحابه ڪرام عَـلَيْهِمُ الرِّضْوَان ۽ اولياءِ عِظام جي سِيرت ۽ واقعات به ذِڪر ڪيا ويا آهن ﴿ 12 ﴾ تفسير صِراطُ الجنان ۾ آيتن مان حاصل ٿيڻ وارن رنگ برنگي، مختلف ۽ نَرالي خوشبوئن وارا مدني گُل به موجود آهن۔

محبتِ اِلٰہي سان معمور جيڪا آيت اسان ٻُڌي، ان جي متعلق تفسير صِراطُ الجنان ۾ آهي،

اللہ تَعاليٰ جا مقبول ٻانها، سموري مخلوق کان وڌيڪ اللہ تَعاليٰ سان مَحَبَّت ڪندا آهن۔ مَحبتِ اِلٰہي ۾ جيئڻ ۽ مَحبتِ الٰہي ۾ مرڻ، ان جي زِندگي جو مَقصد ہوندو آهي۔ پنهنجي ہر خوشي تي پنهنجي رَبّ عَزَّوَجَلَّ جي رِضا کي ترجيح ڏيڻ، نرم ۽ آرامده بسترن کي ڇڏي ڪري بارگاھِ نياز ۾ سر بَسجود ٿيڻ، يادِ الٰہي ۾ روئڻ، رِضاءِ الٰہي جي حُصُول جي لاءِ تڙپڻ، سردين

جي ڊگھين راتين ۾ قيام ۽ گرمين جي وڏن ڏينهن ۾ روزا، اللہ تَعاليٰ جي لاءِ مَحَبَّت ڪرڻ، ان جي خاطر دُشمني رکڻ، ان جي خاطر ڪنهن کي ڪجھ ڏيڻ ۽ ان جي خاطر ڪنهن کان روڪي وٺڻ، نعمت تي شڪر، مُصيبت ۾ صبر، ہر حال ۾ خُدا تي تَوڪل ڪرڻ، پنهنجي ہر مُعاملي کي اللہ تَعاليٰ جي سپرد ڪري ڇڏڻ، اَحڪام اِلٰہي تي عمل جي لاءِ ہر وَقۡت تيار رہڻ، دل کي غير جي مَحَبَّت کان پاڪ رکڻ، اللہ تَعاليٰ جي محبوبن سان مَحَبَّت ۽ اللہ تَعاليٰ جي دُشمنن سان نفرت ڪرڻ، اللہ تَعاليٰ جي پيارن جو نياز مند رہڻ، اللہ تَعاليٰ جي سڀ کان پياري رسول ۽ محبوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي دل ۽ جان سان محبوب رکڻ، اللہ تَعاليٰ جي ڪلام جي تلاوت، اللہ تَعاليٰ جي مُقرَّب ٻانهن کي پنهنجي دلين جي قريب رکڻ، اِن سان مَحَبَّت رکڻ، مَحَبَّتِ الٰہي ۾ اضافي جي لاءِ ان جي صُحبت اختيار ڪرڻ، اللہ تَعاليٰ جي تعظيم سمجھندي انهن جي تعظيم ڪرڻ، اهي تمام اُمُور ۽ ان کان علاوہ سَوَين ڪم اهڙا آهن، جيڪي مَحَبَّتِ الٰہي جي دليل به آهن ۽ ان جا تقاضا به آهن۔

(تفسيرصِراطُ الجنان،ص۲۶۴)

اميرِ اہلسنّت،باني دعوتِ اسلامي حضرت علامه مولانا ابُوبلال محمد الياس عطار قادِري رضوي، ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه بارگاھ خُداوندي ۾ عرض ڪن ٿا:

(وسائلِ بخشش،ص۸۰)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مڪتبة المدينه جي مطبوعه ڪتاب ” اللہ والوں کی باتیں “ ۾ هڪ اهڙي حديثِ پاڪ لکيل آهي، جنهن کي ٻُڌي ڪري گناہگارن جي اميد وڌي ويندي آهي ۽

نيڪو ڪار ڊڄي ويندا آهن۔ اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ ۔ ان حديثِ پاڪ کي ٻُڌي ڪري توهان کي معلوم ٿيندو ته جيڪو اللہ عَزَّوَجَلَّ سان محبت ڪندو آهي، ان جي رات ڪيئن گذرندي آهي؟ جيئن ته

شَہنشاھِ نُبوّت، مَخزنِ جُودوسخاوت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ جنّت نشان آهي: الله عَزَّوَجَلَّ حضرت سيدنا داوٗد عَلَيهِ السَّلام ڏانهن وحي موڪلي: ”اي داوٗد (عَلَيهِ السَّلام)! گنهگارن کي خوشخبري ڏي ۽ صديقن کي ڊيڄار، انهيءَ تي حضرت سيدنا داوٗد (عَلَيهِ السَّلام) کي تعجب ٿيو، انهن عرض ڪيو: ”ياربّ عَزَّوَجَلَّ آئون گنهگارن کي ڪهڙي خوشخبري ڏيان ۽ صدّيقن کي ڪيئن ڊيڃاريان؟

الله عَزَّوَجَلَّ فرمايو: ”اي داوٗد (عَلَيهِ السَّلام) گنهگارن کي اها خوشخبري ڏي ته ڪو گناھ منهنجي بخشش کان وڏو ناهي ۽ صدّيقن کي ان ڳالهه کان ڊيڄار ته اهي پنهنجي نيڪ عملن تي خوش نه ٿين جو مون جنهن کان به پنهنجي نعمتن جو حساب ورتو ته اهو تباھ ۽ برباد ٿي ويندو، اي داوٗد (عَلَيهِ السَّلام)! جيڪڏهن تون مون سان محبّت ڪرڻ چاهين ٿو ته دنيا جي محبت کي پنهنجي دل مان ڪڍي ڇڏ ڇاڪاڻ ته منهنجي ۽ دنيا جي محبت هڪڙي دل ۾ جمع نٿيون ٿي سگهن، اي داوٗد (عَلَيهِ السَّلام)! جيڪو مون سان محبت ڪندو آهي، اهو رات جو منهنجي بارگاھ ۾ تهجُّد ادا ڪندو آهي جڏهن ته ماڻهو سُتا پيا هوندا آهن، اهو اڪيلائي ۾ مون کي ياد ڪندو آهي جڏهن غافل ماڻهو منهنجي ذڪر کان غفلت ۾ پيا هوندا آهن، اهو منهنجي نعمت تي شڪر ادا ڪندو آهي جڏهن ته وسارڻ وارا مون کان غفلت اختيار ڪندا آهن۔ “

(حلیۃالاولیاء ،عبدالعزیزبن ابی رواد،رقم ۱۱۹۰۶، ج ۸،ص ۲۱۱ ۔الیٰ قولہ الاھلک،بحر الدموع ص۲۱)

مِرے دِل سے دنیا کی چاہت مِٹا کر

کر اُلفت میں اپنی فنا یا اِلٰہی

عبادت میں گزرے مری زندگانی

کرم ہو کرم یا خدا یا الٰہی!

تُو اپنی وِلایت کی خیرات دیدے

مِرے غوث کا واسطہ یا اِلٰہی!

(وسائلِ بخشش ،ص۱۰۵)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اللہ عَزَّوَجَلَّ جي حقيقي مَحَبَّت اُن وَقۡت حاصل ٿي سگھي ٿي، جڏهن اسان کي ان جي حاصل ڪرڻ جا طريقا معلوم ہوندا۔ اللہ عَزَّوَجَلَّ جي مَحَبَّت حاصل ڪرڻ جو سڀ کان اَہَم طريقو هي آهي ته ان جي پياري حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان سچي مَحَبَّت ۽ ہر مُعاملي ۾ توهان جي اِطاعت ڪئي وڃي۔ ڇو جو اللہ عَزَّوَجَلَّ پنهنجي مَحَبَّت جي حُصُول جي لاءِ، نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اطاعت ۽ فرمانبرداري کي شرط قرار ڏنو آهي۔ جيئن ته سيپاري 3 سورهء اٰلِ عمران جي آيت 31 ۾ ارشادِ باري تَعاليٰ آهي:

قُلۡ اِنۡ کُنۡتُمۡ تُحِبُّوۡنَ اللہَ فَاتَّبِعُوۡنِیۡ یُحۡبِبۡکُمُ اللہُ وَ یَغۡفِرۡ لَکُمۡ ذُنُوۡبَکُمۡ ؕ وَ اللہُ غَفُوۡرٌ رَّحِیۡمٌ ﴿۳۱﴾

تَرۡجَمو ڪنز الايمان: اي محبوب تون فرمائي ته اي انسانو! جيڪڏهن توهان اللہ کي دوست رکو ٿا ته منهنجا فرمانبردار ٿي وڃو اللہ توهان کي دوست رکندو ۽ توهان جا گناھ بخشي ڇڏيندو ۽ اللہ بخشڻهار مہربان آهي.

ان آيت مبارڪه مان مَعلوم ٿيو ته اللہ عَزَّوَجَلَّ جي مَحَبَّت جو دعويٰ، اُنهيءَ وَقۡت سچّو ٿي سگھي ٿو، جڏهن اسان سيدِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اطاعت ۽ پيروي ڪنداسين۔ حَڪيمُ الاُمَّت حضرت مُفتي احمد يار خان

نعيمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ان آيت ڪريمه جو خُلاصو بيان ڪندي فرمائن ٿا: اي نبي (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم!) توهان انهن ماڻهن کي فرمائي ڇڏيو، جيڪي توهان جي وسيلي جي بغير اسان جي مَحَبَّت جو دَم ڀرين ٿا يا جيڪي پنهنجو پاڻ کي رَبّ جو پيارو ڄاڻي ڪري توهان کان بي نياز ٿيڻ چاہين ٿا، يا جيڪي توهان جي اطاعت کان سِواءِ ٻين اَسباب سان خُدا تائين پہچڻ چاہين ٿا، انهن سڀني جي لاءِ اعلانِ عام ڪري ڇڏيو ته جيڪڏهن توهان خدا سان مَحَبَّت ڪرڻ چاہيو ٿا ته، نه مون سان مُقابلو ڪريو نه منهنجي برابري جو دَم ڀريو، نه مون کان اڳتي اڳتي هلو، بلڪه غلام بڻجي ڪري منهنجي پويان پويان هلندا اچو، پنهنجا اَقۡوال، اَفۡعال، اَعۡمال، زندگي جي ہر شُعۡبي کي منهنجي مثال بڻايو ۽ مون ۾ فنا ٿي وڃو، پوءِ مُعاملو برعڪس ٿيندو ته رَبّ توهان کي پنهنجو محبوب بڻائي وٺندو ۽ توهان جيڪي چاهيندو، اهو ڪندو، ان سان گڏ هُو توهان جا سمورا گُناھ مُعاف فرمائي ڇڏيندو، ڇو جو اللہ (عَزَّوَجَلَّ) وڏو غَفّار ۽ اَرۡحَمُ الرَّاحمين آهي، توهان پنهنجو پاڻ کي ان جي مغفرت ۽ رحمت جو اہل بڻايو، پوءِ مزو ڏسو۔ (تفسير ِنعيمي،ج۳، ص۳۵۹۔۳۶۰،ملتقطاً) تنهنڪري اسان کي به پنهنجا ڏينهن ۽ رات اللہ عَزَّوَجَلَّ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي احڪامن جي بجاآوري ۾ گذارڻ گھرجن، ڇو جو مَحَبَّت جو تقاضو به اهو ئي آهي ته جنهن سان مَحَبَّت ڪئي وڃي، ان جي اطاعت ۽ فرمانبرداري به ڪئي وڃي۔

حضرت سيدُنا ابُوہُريرہ رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان مَروي آهي ته ٻنهي جہانن جي تاجۡوَر، سلطانِ بَحروبَر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو ته اللہ عَزَّوَجَلَّ ارشاد فرمائي ٿو: ”جنهن، ڪنهن ولي کي اَذِيت ڏني، آئون ان سان اعلانِ جنگ ڪريان ٿو ۽ ٻانهو منهنجو قُرب سڀ کان وڌيڪ فرائض جي ذريعي حاصل ڪندو آهي ۽ نَوافل جي ذريعي مُسلسل قرب حاصل ڪندو رہندو آهي، ايستائين جو آئون ان سان مَحَبَّت ڪرڻ لڳندو آہيان۔ جڏهن آئون ٻانهي کي محبوب بڻائي وٺندو آہيان، ته آئون ان جا ڪن بڻجي ويندو آہيان، جنهن سان اهو ٻڌندو آهي۔ ان جي اَک بڻجي ويندو آهيان ،جنهن سان اهو ڏسندو آهي۔ ان جا ہٿ بڻجي ويندو آهيان، جنهن سان اهو پڪڙيندو آهي ۽ ان جا پير بڻجي ويندو آهيان، جنهن سان اهو هلندو آهي۔ پوءِ اهو مون کان سُوال ڪري، ته آئون ان کي عطا ڪندو آهيان، منهنجي پناھ چاهي، ته آئون ان کي پناھ ڏيندو آہيان۔ (بخاری، کتاب الرقاق، باب التواضع، ۴/۲۴۸،حدیث:۶۵۰۲)

مُفَسّرِ شہير حڪيمُ الاُمَّت حضرت ِ مفتي احمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اِن حديثِ پاڪ جي تحت فرمائن ٿا: ان عبارت جو هي مطلب ناهي ته خدا تَعاليٰ،

وَلي ۾ حُلُول ڪري ويندو آهي، جيئن ڪوئلي ۾ باھ يا گُل ۾ رنگ ۽ بُو، جو خدا تَعاليٰ حُلُول کان پاڪ آهي ۽ هي عقيدو ڪفر آهي بلڪه ان جا ڪجھ مطلب آهن هڪ هي ته وَلي ُ الله جا هي اَعۡضاء، گُناھ جي لائق ناهين رہندا، ہميشه ان کان نيڪ ڪم ئي سرزد ٿيندا آهن، ان تي عبادتون آسان ٿي وينديون آهن، ڄڻ ته سموريون عبادتون ان سان آئون ڪرائي رہيو آهيان يا هي ته پوءِ اهو ٻانهو ان عضوَن کي دُنيا جي لاءِ استعمال ناهي ڪندو، صرف منهنجي لاءِ استعمال ڪندو آهي، ہر شيءِ ۾ مون کي ڏسندو آهي، ہر آواز ۾ منهنجي آواز ٻُڌندو آهي، يا هي ته اهو ٻانهو فَنا فِي الله ٿي ويندو آهي، جنهن سان خُدائي طاقتون ان جي عضوَن ۾ ڪم ڪنديون آهن ۽ اهو اهڙا ڪم ڪري ڇڏيندو آهي جيڪي عقل کان وَراء (سمجھ ۾ نه اچڻ وارا) آهن، حضرت (سَيدُنا) يعقوب عَـلَيْـهِ الـسَّـلَام جن ڪنعان ۾ ويهي مِصۡر کان هليل قميصِ يوسُفي جي خُوشبو سُونگهي ورتي، حضرت (سَيدُنا) سليمان عَـلَيْـهِ الـسَّـلَام جن ٽي (3) مِيل جي فاصلي کان ماڪوڙي جي آواز ٻُڌي ورتي، حضرت آصِف بن بَرخيا جن اَک ڇنڀ کان پہريان يمن کان تختِ بلقيس آڻي ڪري شام ۾ حاضر ڪري ڇڏيو۔ حضرت (سَيدُنا) عمر (رَضِىَ اللهُ تَعَالٰی عَـنْهُ) جن مدينهء مُنوَّرہ کان خُطبو پڙھندي نَہاوند تائين پنهنجي آواز پہچائي۔ حُضُورِ انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن قيامت تائين جا واقعات بچشم مُلاحَظه فرمائي ورتا۔ هي سڀ اِنهيءَ طاقت جا ڪرشما آهن۔ (مرآۃ المناجيح، ۳/۳۰۸)

اللہ عَزَّوَجَلَّ جڏهن پنهنجي ٻانهي سان مَحَبَّت فرمائيندو آهي، ته ان کي دِين جي مَحَبَّت عطا فرمائي ڇڏيندو آهي جيئن ته حديثِ پاڪ ۾ آهي ته اللہ تَعاليٰ دُنيا ان کي به ڏيندو آهي، جيڪو ان کي محبوب ہجي ۽ ان کي به جيڪو محبوب ناهي ۽ دِين صرف ان کي ئي ڏيندو آهي، جيڪو ان جي ويجھو پيارو ہوندو آهي، تنهنڪري جنهن کي اللہ عَزَّوَجَلَّ دِين ڏنو، ان کي مَحبُوب بڻائي ورتائين۔ (''شعب الإیمان''، باب في قبض الید عن الاموال المحرمۃ،الحدیث:۵۵۲۴،ج۴، ص۳۹۵ ۔ ۳۹۶)

اهڙي طرح جڏهن اللہ عَزَّوَجَلَّ ٻانهي سان مَحَبَّت ڪندو آهي ته ان جي ذِڪۡرِ خَير کي پنهنجي مخلوق ۾ عام فرمائي ڇڏيندو آهي، ہر طرف ان جي نيڪ نامي ۽ ڀلائي جا چرچا ہئڻ لڳندا آهن۔ حُضُورِ پاڪ، صاحب ِلولاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عالي شان آهي: اللہ عَزَّوَجَلَّ جڏهن ڪنهن ٻانهي سان مَحَبَّت ڪندو آهي ته حضرت جبريل عَـلَيْـهِ الـسَّـلَام کي گھُرائيندو آهي، پوءِ فرمائيندو آهي ته آئون فُلاڻي سان مَحَبَّت ڪندو آہيان، تون ان سان مَحَبَّت ڪر، جيئن ته حضرت جبريل عَـلَيْـهِ الـسَّـلَام ان سان مَحَبَّت ڪندا آهن، آسمان ۾ اعلان ڪندي چوندا آهن ته اللہ عَزَّوَجَلَّ فُلاڻي سان مَحَبَّت ڪري ٿو، توهان ماڻهون ان سان مَحَبَّت ڪريو، ته ان سان آسمان وارا مَحَبَّت ڪندا آهن، پوءِ ان جي لاءِ زمين ۾ قبوليت رکي ويندي آهي ۽ جڏهن رب تَعاليٰ ڪنهن ٻانهي کان ناراض ٿيندو آهي، ته فرمائيندو آهي ته آئون فُلاڻي کان ناراض آہيان، ته توهان به ان کان ناراض ٿي وڃو، فرمايو ته جبرئيل ان کان ناراض ٿي ويندا آهن، پوءِ آسمان وارن ۾ اعلان ڪندا آهن ته اللہ عَزَّوَجَلَّ فُلاڻي کان ناراض آهي، توهان ماڻهو به ان کان ناراض ٿي وڃو، فرمايو: پوءِ اهي ماڻهون ان سان نفرت ڪندا آهن، پوءِ زمين ۾ ان جي لاءِ نفرت رکي ويندي آهي۔

ٻه فرامين مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم: * عمامي سان گڏ ٻه رڪعتون نماز (پڙهڻ) بغير عمامي جي ستر 70 رڪعتن کان افضل آهي. (الفردوس بماثورالخطاب ج 2 ص 265 حديث 3233) * عمامي سان گڏ نماز ڏهه هزار نيڪين جي برابر آهي. (ايضاً ج 2 ص 406 حديث 3805 فتاويٰ رضويه ج 6 ص 213) * عمامو قبلي ڏانهن منهن ڪري بيهي ٻڌو. (ڪشف الالتباس في استحباب اللباس للشيخ عبدالحق الدهلوي ص 38) * عمامي ۾ سنت هي آهي ته اڍائي گز کان گھٽ نه هجي، نه ڇهه گز کان وڌيڪ، ۽ ان جي بندش گنبد نما هجي. (فتاويٰ رضويه ج22 ص 186) * رومال جيڪڏهن ايترو وڏو هجي جيڪو مٿي کي چڱي طرح ڍڪي ته اهو عمامو ئي آهي ۽ * ننڍو رومال جنهن سان صرف هڪ ٻه وَر اچي سگھن ٻَڌڻ مڪروهه آهي. (فتاويٰ رضويه ج 7 ص 299) * عمامي کي جڏهن نئي سري سان ٻڌڻو هجي ته جيئن ٻڌيل آهي ائين کوليو ۽ هڪ مهل زمين تي نه اُڇلايو . (فتاويٰ عالمگيري ج 5 ص 330) * جيڪڏهن ضرورتاً لاٿو، ۽ ٻيهر ٻَڌڻ جي نيّت هجي ته هڪ هڪ پيچ کولڻ تي

هڪ هڪ گناهه مٽايو ويندو ـ (ملخص از فتاويٰ رضويه ج 6 ص 214) عمامي جو طبي فائدو هي آهي ته: * مٿي اگھاڙي رهندڙ جي وارن تي سردي گرمي ۽ اُس وغيره سڌو سنئون اثر ڪندي آهي ان سان نه صرف وار بلڪ دماغ ۽ چهرو به متاثر ٿيندو آهي ۽ صحت کي نقصان پهچي سگھي ٿو، تنهنڪري سنت جي پيروي جي نيت سان عمامو شريف ٻڌڻ ۾ ٻنهي جهانن ۾ عافيّت آهي.
