اَلْحَمْدُ لِلّٰہِ رَبِّ الْعٰلَمِیْنَ وَ الصَّلٰوۃُ وَالسَّلَامُ عَلٰی سَیِّدِ الْمُرْسَلِیْنَ ط

اَمَّا بَعْدُ فَاَعُوْذُ بِاللّٰہِ مِنَ الشَّیْطٰنِ الرَّجِیْمِ ط بِسْمِ اللہِ الرَّحْمٰنِ الرَّ حِیْم ط

فيضان اَصحابِ صُفه

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا نَبِیَّ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا نُوْرَ اللہ

دُرُود شريف جي فضيلت

حضور نبي رحمت، شفيعِ اُمَّت جن جو فرمان آهي: قِيامت جي ڏينهن الله پاڪ جي عرش جي علاوه ٻي ڪا ڇانوَ نه هوندي، ٽي (3) شَخص الله پاڪ جي عرش جي ڇانوَ ۾ هوندا۔ (1) اُهو شخص جيڪو منهنجي اُمتي جي پريشاني دور ڪري (2) منهنجي سنّت کي زندھ ڪرڻ وارو (3) مون تي ڪثرت سان درود پاڪ پڙهڻ وارو۔ (اَلبُدورُ السّافرة لِلسُّيوطي، ص ۱۳۱حديث: ۳۶۶)

تُم ہو میں تُم پر فِدا تُم پہ کروڑوں دُرُود

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿيون. فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِهٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندَس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندَس ٭ ضرورت پوڻ تي ٻين اسلامي ڀينرن جي لاءِ سِرڪي جاءِ ڪشاده ڪندَس ٭ ڌِڪو وغيره لڳو ته صبر ڪندَس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندَس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ واريءَ جي دِل جوئي جي لاءِ آهسته آواز سان جواب ڏيندَس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندَس ۔ دوران بيان موبائل جي غيرضروري استعمال کان بچندَس،نه بيان ريڪارڊ ڪندَس نه ئي ٻئي ڪنهن قسم جو آواز ڇو ته ان جي اجازت ناهي،جيڪو ڪجهه ٻُدَندَس،ان کي ٻڌي ۽ سمجهي ڪري ان تي عمل ڪرڻ ۽ ان کي بعد ۾ ٻين تائين پهچائي نيڪي جي دعوت عام ڪرڻ جي سعادت حاصل ڪندَس.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! صحابَه ڪرام جو شان ايترو بي مثال آهي جو ڪو به امتي چاهي اهو ڪيتري به وڏي مرتبي تي پهتل هجي، انهن جي مقام ۽ رُتبي کي نٿو پهچي سگھي۔ هي اهي مبارڪ هستيون آهن جن سڀ کان پهريان اسلام قبول ڪيو، هي اهي خوش نصيب آهن جن کي الله پاڪ پنهنجي محبوبِ جي صحبت اختيار ڪرڻ جي لاءِ چونڊيو، هي اهو مقدس گروه آهي جنهن کي حضور سڀ کان پهريان نيڪي جي دعوت ڏني، هي اهي پاڪيزه افراد آهن جن دينِ اسلام کي پکيڙڻ جي لاءِ ڪافرن جي ظلم ۽ ستم کي برداشت ڪيو، هي اهي ماڻهو آهن جن ابتداءِ اسلام جي مشڪل دور ۾ بک اُڃ برداشت ڪري، پيٽ تي پٿر ٻڌي، قريبي رشتيدارن، پنهنجن ۽ پراين جون دشمنيون سهي ڪري به پرچمِ اسلام کي بلند رکيو، يقيناً انهن جي بيحد محنتن ۽ لازوال قربانين جو نتيجو آهي جو اڄ اسان الله پاڪ ۽ ان جي پياري رسول کي ياد ڪندڙ آهيون ۽ هر پاسي دينِ اسلام جون بهارون آهن.

هونءَ ته تمام صحابه ڪرام نبي اڪرم، رسولِ محتشم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا پيارا ۽ هدايت جا روشن تارا آهن، سڀئي اسان جي اکين جو نور ۽

دل جو سُرُور آهن، پر انهن مان ڪجھ صحابه اهڙا به آهن جيڪي پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي خدمت ۾ صرف دين سکڻ جي لاءِ حاضر رهندا هئا۔ انهن کي اصحابِ صُفَه چوندا آهن۔ اڄ اسان انهن ئي جو شان ۽ انهن جي قربانين بابت ٻڌنديون سين۔ انهن اصحابن مان ڪجھ وڏن ۽ مشهور صحابَه ڪرام عَلَیْہِمُ الرِّضْوَان جي سيرتن سان گڏوگڏ ضمناً مدني گل به ٻڌڻ جي سعادت حاصل ڪنديون سين۔ ان سان گڏوگڏ اصحابِ صفّه جي زندگين مان اسان کي ڪهڙو درس ملندو آهي، اهي ڪهڙا سبب آهن جن جي ڪري اصحابِ صفّه پنهنجا گھر ٻار، اهل و عيال ۽ فڪرِ روزگار کان بي نياز ٿي مسجدِ نبوي ۾ ئي رهندا هئا، اهو سڀ به ٻڌڻ جي سعادت حاصل ڪنديون سين۔ اچو سڀ کان پهريان اصحابِ صفّه مان هڪ مشهور صحابي، مقبولِ بارگاھ رسول حضرت سَيدُنا ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ جو واقعو ٻڌون ٿيون جنهن ۾ ٻين اصحابِ صفّه جو ذڪرِ خير به آهي۔

کير جو هڪ پيالو

حضرتِ سيدُنا ابو هريره رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا: بک جي شدت جي ڪري هڪ ڏينهن مان اُن رستي تي ويهي رهيس، جتان ماڻهو ٻاهر ويندا هئا۔ ٻنهي جهانن جا آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مون وٽان گذريا ته مون کي ڏسي مُرڪيا ۽ منهنجو چهرو ڏسي منهنجي حالت سمجهي ويا۔ ارشاد فرمايائون؛ اي ابوهريره! مون عرض ڪيو؛ لَبَّيۡڪ يا رسول الله!(صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) فرمايائون؛ مون سان گڏ اچو، مان پٺيان هلڻ لڳس۔ جڏهن شهنشاھِ بَحۡر و بَر مديني جا تاجوَر صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي مبارڪ گهر ۾ جلوه فرما ٿيا ته اجازت وٺي مان به اندر داخل ٿيس۔ سروَرِ ڪائنات، شاھِ موجودات صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ هڪ پيالي ۾ کير ڏٺو ته فرمايائون؛ ”هي کير ڪٿان آيو آهي؟ گهروارن عرض ڪيو؛ فلاڻي صحابِيءَ يا صحابيه پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جي لاءِ تحفي طور موڪليو آهي۔ فرمايائون؛ ابوهريره! مون عرض ڪيو؛ لَبَّيۡڪ يا رسول الله (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ)! فرمايائون؛ وڃو، اهلِ صُفّه کي سڏي اچو۔ حضرتِ سيّدُنا ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا؛ اهلِ صُفّه عَلَیْہِمُ الرِّضْوَان اسلام جا مهمان آهن، نه انهن کي ٻارن ٻچن جو شوق آهي نه مال ۽ دولت جو ۽ نه اهي ڪنهن شخص جو سهارو وٺندا آهن۔ جڏهن محبوبِ ربّ ذوالجلال صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن وٽ صدقي جو مال ايندو هو ته پاڻ ڪريم اهو مال انهن ڏانهن موڪلي

ڇڏيندا هئا ۽ پاڻ ان مان ڪجهه به نه کڻندا هئا ۽ جڏهن پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن وٽ ڪو هديو ايندو هو ته پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اُنهن ڏانهن موڪليندا هئا، اُن مان پاڻ به اِسۡتِعمال ڪندا هئا ۽ اُنهن کـي به شريڪ فرمائيندا هئا۔ مون کي اها ڳالھ ڏکي لڳي دل ۾ خيال آيو ته اهلِ صُفّه جو هن کير مان ڇا ٿيندو؟ مان هن جو وڌيڪ مُسۡتحِق هئس جو هن کير مان ڪجهه ڍُڪَ پيئان ها ۽ ڪجهه طاقت حاصل ڪيان ها۔ جڏهن اصحابِ صُفّه عَلَیْہِمُ الرِّضْوَان ايندا ته سرڪارِ نامدار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مون کي ئي ارشاد فرمائيندا ته انهن کـي کير پيش ڪر۔ اهڙي صورت ۾ تمام مشڪل آهي جو کير جا ڪجهه ڍڪ مون کي ميّسر ٿين، پر الله ۽ ان جي رسول جي اطاعت ۽ فرمانبرداري کان بغير ڪا واهه نه هئي، مان اصحابِ صُفّه عَلَیْہِمُ الرِّضْوَان کي سڏي آيس۔ اهي آيا انهن تاجدارِ مدينه ءِ منوره، شهنشاھِ مڪه ءِ مڪرمه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کان (اندر اچڻ جي) اجازت طلب ڪئي۔ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اجازت عطا فرمائي ۽ اهي گهر ۾ داخل ٿي ويهي رهيا۔ مٺي مٺي آقا، مديني واري مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو؛ ابوهريره! مون عرض ڪيو؛ لبَّيۡڪ يا رسول الله (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ)! فرمايائون؛”پيالو پڪڙيو ۽ هنن کي کير پياريو“ حضرتِ سيّدُنا ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا؛ مون پيالو پڪڙيو، مان اهو پيالو هڪ شخص کي ڏيندو هئس اهو پيٽ ڀري کير پيئندو ۽ وري پيالو مون کي موٽائي ڏيندو هو، ايستائين جو مان پياريندو پياريندو، سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تائين پهچي ويس ۽ سڀئي ماڻهو سَير ٿي چڪا هئا۔ سرڪارِ مدينه پيالو کڻي پنهنجي هٿ مبارڪ تي رکيو۔ پوءِ مون ڏانهن ڏسي مُرڪيا ۽ ارشاد فرمايائون؛ ”ابو هريره! مون عرض ڪيو؛ لبَّيۡڪ يا رسول الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ! فرمايائون؛ هاڻي مان ۽ تون باقي رهجي ويا آهيون۔ عرض ڪيم؛ يا رسول الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ! اوهان سچ فرمايو، فرمايائون؛ ”ويهو ۽ پيئو۔“ مان ويهي رهيس ۽ کير پيئڻ لڳس۔ پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ فرمايو؛ ”اڃان پيئو“ مون وري پيتو۔ پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مسلسل فرمائيندا رهيا ته؛ ”پيئو“ ايستائين جو مون عرض ڪيـو؛ بس! قسم آَهي اُن ذات جو جنهن پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن کي حق سان مبعوث فرمايو ، هاڻي وڌيڪ گنجائش ناهي، فرمايائون؛ ” مون کي ڏيکاريو“ مون پيالو پيش ڪيو، پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن الله تعاليٰ جي ساراھ ڪئي ۽ بِسۡمِ الله پڙهي

بچيل کير نوش فرمايو (يعني پيتو). (بخاري،ڪتاب الرقاق باب ڪيف ڪان عيش النبي۔۔۔الخ۴ /۲۲۳، حديث:۶۴۵۲)

دينے والا ہے سچا ہمارا نبي!

پياريون پياريون اسلامي ڀينرون! هن واقعي مان معلوم ٿيو ته حضورِ انور، شافعِ محشر مالڪ ۽ مختار آهن، جڏهن چاهين، جيئن چاهين، جيترو چاهين، جتي چاهين تصرف فرمائن، اسان جي آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ وٽ سڀئي اختيار آهن۔ اهو به معلوم ٿيو نبي رحمت، شفيع امت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کان وڏو ڪو سخي ناهي، کير جو صرف هڪ ئي پيالو هو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ خود ئي ان کي استعمال ڪري سگھيا ٿي پر اهو پسند نه ڪيائون بلڪ فيض ۽ سخاوت جا دريا وهائيندڙ آقا، سوالين جا دامن ڀرڻ وارا داتا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اصحابِ صفّه کي گھرائي انهن کي به ۽ جنابِ ابُو هُريره رَضِیَ اللہُ تَعَالٰی عَنْہُ کي به کير پيئاري پنهنجي سخاوت جا دريا وهائي ڇڏيائون۔ ائين ڇو نه ٿئي جو اسان جا آقا، عالم جا داتا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تمام بني آدم ۾ سڀ کان وڌيڪ سخي آهن۔ جيئن ته حضرت انس بن مالڪ رَضِیَ اللہُ تَعَالٰی عَنْہُ کان روايت آهي ته هڪ ڀيري رسولِ خدا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: ڇا توهان کي خبر آهي ته سڀ کان وڏو سخي ڪير آهي؟ صحابَه ڪرام عَلَیْہِمُ الرِّضْوَان عرض ڪيو: الله ۽ ان جو رسول بهتر ڄاڻي ٿو۔ ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ارشاد فرمايائون: الله پاڪ سڀ کان وڌيڪ جوّاد آهي پوءِ تمام بني آدم ۾ سڀ کان وڌيڪ سخي مان آهيان۔ (شعب الايمان،الثامن عشر ،باب في نشر العلم، ۲/۲۸۱، حديث:۱۷۶۷ملتقطا)

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله تَعَالٰي عَلٰي مُحَمَّد

اصحابِ صفه ڪير هئا؟

پياريون پياريون اسلامي ڀينرون! هن واقعي ۾ حضرت ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ جو به ذڪر آهي ۽ ٻين اصحابِ صفه جو به تذڪرو آهي۔ پاڻ حضرت سَيدُنا ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ اصحابِ صفه مان هئا۔ مسجد نبوي شريف عَلٰی صَاحِبِہَا الصَّلٰوۃُ وَالسَّلَام ۾ هڪ چبوترو هو ، جتي(مختلف وقتن ۾) تقريبًا چار يا پنج سؤ فقراءِ مهاجرين ويهندا هئا۔ تنهن ڪري انهن کي اصحابِ صُفّه چوندا آهن۔ انهن وٽ نه گهر هو نه

دُنياوي ساز و سامان ۽ نه ئي ڪو ڪاروبار۔ غُربت جو اِهو حال هو جو اِنهن مان 70 وٽ ته اَوگَهڙ ڍڪڻ جي لاءِ پورو ڪپڙو به نه هوندو هو۔ انهن جي تعداد ۾ ڪمي پيشي ٿيندي رهندي هئي۔ مديني شريف ۾ اچڻ واري جو ڪو ڄاڻ سُڃاڻ وارو نه هوندو هو ته اهو به صُفّه وارن ۾ شامل ٿي ويندو هو۔ (تفسير نعيمي،۳/۱۳۷)

بظاهر ته اهي اَصحابِ صُفّه وڏا ڪمزور (Weak) هئا، بظاهر ته اهي اَصحابِ صُفّه گھڻا تنگ دست هئا، انهن وٽ پائڻ جي لاءِ سٺا ڪپڙا نه هئا، انهن وٽ کائڻ لاءِ غذا نه هئي، انهن وٽ سمهڻ جي لاءِ نرم ۽ گرم بستر نه هئا، انهن وٽ رهڻ جي لاءِ گھر نه هئا، انهن وٽ پنهنجي خدمت جي لاءِ نوڪر چاڪر نه هئا، انهن وٽ ڪم ڪار ڪرڻ جي لاءِ غلام نه هئا، اَلغَرَض! بظاهر اهي گھڻا غريب هئا، پر الله پاڪ جي بارگاھ ۾ مقبول هئا، رسولِ ڪريم، رءوفٌ رّحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا مقرب هئا، قرآنِ پاڪ ۾ خود ربِ ڪائنات انهن جي تعريف ۽ توصيف بيان فرمائي آهي۔ جيئن سيپاري 15،سورة ڪهف،آيت نمبر 28 ۾ ارشاد فرمايو:

وَ اصْبِرْ نَفْسَكَ مَعَ الَّذِیْنَ یَدْعُوْنَ رَبَّهُمْ بِالْغَدٰوةِ وَ الْعَشِیِّ یُرِیْدُوْنَ وَجْهَهٗ وَ لَا تَعْدُ عَیْنٰكَ عَنْهُمْۚ-تُرِیْدُ زِیْنَةَ الْحَیٰوةِ الدُّنْیَاۚ-وَ لَا تُطِعْ مَنْ اَغْفَلْنَا قَلْبَهٗ عَنْ ذِكْرِنَا وَ اتَّبَعَ هَوٰىهُ وَ كَانَ اَمْرُهٗ فُرُطًا(۲۸) (پ۱۵، الڪھف:۲۸)

ترجمو ڪنزالايمان: ۽ پنهنجي پاڻ کي انهن سان مانوس رک جيڪي صبح ۽ شام پنهنجي رب کي پڪاريندا آهن ان جي رضا گهرندا آهن ۽ تنهنجون اکيون انهن کي ڇڏي هيڏي هوڏي نه پون، ڇا تون دنيا جي زندگي جو سامان گهرندين، ۽ ان جو چيو نه ڪر جنهن جي دل اسان پنهنجي ياد کان غافل ڪري ڇڏي ۽ اهو پنهنجي خواهش جي پٺيان هليو ۽ ان جو ڪم حد کا ٽپي ويو.

اَصحابِ صُفّه کي پري نه ڪريو!

تفسيرِ نعيمي ۾ هن آيت سڳوري تحت آهي: (الله پاڪ اهو فرمائي رهيو آهي ته) اي حبيب مڪرم! (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) اوهان پنهنجو پاڻ کي سڄي عمر، سڄي حياتِ طَيبَه هر محفل، هر مجلس، هر گھڙي، ڏينهن رات، صبح شام

انهن ئي ماڻهن کي گڏ رکو جيڪي مخلص (Sincere) آهن، جيڪي مسڪين آهن، جيڪي فقير آهن، جيڪي غريب آهن، جيڪي سادا آهن، جيڪي نهايت خلوص سان پنهنجي رب جي عبادت ڪن ٿا، جيڪي خشوع و خضوع جا پيڪر آهن، جيڪي صبح کان شام تائين، جيڪي فجر کان عشاء تائين، جيڪي پنج ئي وقتن ۾، جيڪي سحري وقت جاڳي، جيڪي رات جو سمهڻ کان پهريان اَلغَرَض! هر وقت ئي، هر ساھ ۾ هر ڪروٽ ۾ ذڪرِ الٰهي ڪندا آهن۔ سڄي دنيا کان منهن موڙي بس پنهنجي موليٰ جي ذات ئي جو ارادو رکن ٿا، ان کي چاهين ٿا، ان جا ئي طلب گار آهن۔ (۽ اي حبيبِ مڪرم!) انهن ڪافرن کان پري ئي رهو جيڪي مختلف وفد بڻجي اوهان وٽ اچن ٿا ۽ اوهان کي پنهنجي ايمان آڻڻ جي لاءِ طرح طرح جا شرط (Conditions) لڳائن ٿا ته يارَسُولَ الله اسان ۽ اسان جو قبيلو اوهان تي تڏهن ايمان آڻيندو جڏهن اوهان پاڻ وٽان، پنهنجي محفل مان انهن غريب اَصحابِ صُفّه کي پري ڪندؤ، يا ڪم از ڪم اسان جي هوندي انهن کي پاڻ وٽ نه اچڻ ڏيو، انهن جي محفل الڳ لڳايو، اسان جي مجلس الڳ سجايو، اسان کي انهن سان گڏ ويهندي شرم اچي ٿو ۽ اسان جي سرداري تي فرق اچي ٿو، بس اها ئي هڪ رڪاوٽ آهي جنهن سبب اسان اوهان جي نبوت ۽ رسالت، صداقت، امانت ۽ ديانت تي اڃا تائين ايمان نه آڻي سگھياسين، (ان تي الله پاڪ ارشاد فرمائي رهيو آهي) اي محبوب! (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) انهن مڪارن جي ڳالهين ۾ نه اچجو ۽ پنهنجي رحمت ۽ عافيت ڀرين نگاهن کي انهن پاڪيزه باطن وارن کان نه هٽائجو ڇوته اوهان جون اهي معرفت واريون نظرون ئي ته انهن حضرات جي ڪُل ڪائنات آهن، اوهان جون محفلون ئي ته انهن جي زندگي آهي، اوهان جي جدائي انهن جي لاءِ تڪليف جو سبب آهي، هن موقعي تي اها آيت سڳوري نازل ٿي ۽ نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي فرمايو ويو ته پنهنجي جان کي انهن ماڻهن سان مانوس رکو جيڪي صبح ۽ شام پنهنجي رب کي پڪارين ٿا ان جي رضا چاهن ٿا يعني جيڪي اِخلاص سان هر وقت الله پاڪ جي فرمانبرداري ۾ مشغول رهنداآهن۔ (تفسير نعيمي، تحت الآية المذڪورة، ۱۵/۵۷۶ بتغير)

پياريون پياريون اسلامي ڀينرون! غور ڪريو! اَصحابِ صُفّه جو مقام الله پاڪ جي بارگاھ ۾ ڪيتري قدر بلند آهي جو انهن جي باري ۾ پنهنجي حبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي هدايت فرمائي ته انهن کي پنهنجي بارگاھ ناز کان پري نه ڪريو۔ تفسير صراط الجنان ۾ آهي: هن آيت سڳوري مان ڪيئي مسئلا معلوم ٿيا،

جيئن: (1) معلوم ٿيو ته چڱن سان گڏ رهڻ چڱو آهي توڙي جو اهي فُقراء هجن ۽ برن سان گڏ رهڻ برو آهي توڙي جو اهي مالدار هجن۔ (2) اهو به معلوم ٿيو ته صبح و شام خصوصيت سان الله پاڪ جو ذڪر ڪرڻ گھڻو افضل آهي۔ (3) نيڪن جون ٻه نشانيون به هن آيت ۾ بيان فرمايون، پهرين هي جو اهي صبح ۽ شام الله پاڪ جو ذڪر ڪن ٿا ۽ ٻي هي جو هر عمل سان الله پاڪ جي رضا ۽ خوشنودي جا طلبگار هوندا آهن۔ (4) هن آيت ۾ دنيا دارن ڏانهن نظر رکڻ ۽ انهن جي پيروي کان منع فرمايو ويو آهي۔ (صراط الجنان، ۵/۵۵۹ ملتقطاً)

ڪجھ مشهور اَصحابِ صُفّه

پياريون پياريون اسلامي ڀينرون! اَصحابِ صُفّه جي شان ۽ عظمت جي ڇا ڳالھ ڪجي! هي اهي خوش نصيب ماڻهو هئا جيڪي مدني حبيب، طبيبن جي طبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صحبت ۾ ئي رهندا هئا، انهن زندگي جون تمام نعمتون گھرٻار، اهل و عيال، مال ۽ دولت، جاھ ۽ ثروت ۽ عيش و عشرت کي دين جي تعليم ۽ اشاعتِ اسلام جي لاءِ قربان ڪري ڇڏيو هو۔ اَصحابِ صُفّه ۾ وڏا وڏا جليل القدر صحابَه ڪرام عَلَیْہِمُ الرِّضْوَان شامل (Included) آهن۔ مشهور اصحابِ صُفّه ۾ حضرت سيدنا ابُوهُرَيره، حضرت سيدنا بلال حَبشي، حضرت سيدنا ابُوذَر غِفاري، حضرت سيدنا عَمَّار بِن ياسِر، حضرت سيدنا سلمان فارسي، حضرت سيدنا حُذيفه بن اليَمان، حضرت سيدنا ابو سعيد خُدري رَضِیَ اللہُ تَعَالٰی عَنْہُم جهڙا جَلِيلُ القَدۡر صحابي شامل آهن۔ (سيرتِ رسولِ عربي، ص۱۱۶ ملخصاً)

اَصحابِ صُفّه جا فاقا

اَصحابِ صُفّه تمام غريب هئا، اڪثر بُک واري زندگي ۾ رهندا هئا، نبي ڪريم، رسولِ عظيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ وٽ جڏهن ڪو تحفو، صدقو يا مالِ غنيمت ايندو هو ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان مان اصحابِ صفه تي خرچ ڪندا هئا۔ ۽ ڪڏهن ڪڏهن ائين به ٿيندو هو ته اهي عظيم ماڻهو ڪيئي ڪيئي ڏينهن تائين بکايل رهندا هئا۔ انهيءِ ڪري اهي تمام ڪمزور، ضعيف، هيڻا ۽ نادار هئا۔ اچو اَصحابِ صُفّه جي غريبي ۽ ڪمزوري جي باري ۾ ڪجھ روايتون ٻڌون ٿيون:

.1 حضرت سيّدُنا فَضالَه بن عُبيد رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا؛ حبيبِ ڪِبۡريا، اِمامُ

الانبياء صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جڏهن ماڻهن کي نماز پڙهائيندا هئا ته ڪجهه صحابه عَلَیْہِمُ الرِّضْوَان نماز ۾ ئي قيام جي حالت ۾ بُک جي سختيءَ جي سبب ڪِري پوندا هئا ۽ اهي اصحابِ صُفه هوندا هئا۔ ايستائين جو اعرابي چوندا هئا ته اهي ديوانا آهن۔ جڏهن سرڪارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ نماز مان فارغ ٿيندا هئا ته انهن ڏانهن توجهه ڪري فرمائيندا هئا، جيڪڏهن توهان کي معلوم ٿي وڃي ته توهان جي لاءِ الله تعاليٰ وٽ ڪيڏو (اجر ۽ ثواب) آهي ته توهان انهيءَ ڳالهه کي پسند ڪريو ته توهان جي بُک ۽ حاجت منديءَ ۾ اضافو ٿئي۔ (فيضانِ سنّت، ص۷۰۲،ترمذي،ڪتاب الزهدباب ماجاء في معيشة ۔۔۔الخ،۴/۱۶۲،حديث:۲۳۷۵)

مصروف رهندا هئا جڏهن ته اسان جو حال هي آهي ته کائڻ پيئڻ جي هر طرح جون سهولتون اسان کي ميسر آهن، علمِ دِين حاصل ڪرڻ جا آسان ذريعا به موجود آهن۔ پر افسوس! صد ڪروڙ افسوس! اسان صرف نفس ۽ شيطان جي ورغلائڻ ۾ اچي ۽ غفلت ۽ سستي جي ڪري هن عظيم سعادت کان محروم آهيون، ايستائين جو فرض ۽ لازم علم ۽ تجويد سان قرآنِ پاڪ سکڻ جي لاءِ به اسان وقت نٿيون ڪڍون۔ ياد رکو! انسان جيستائين ڪنهن ڪم کي ڪرڻ جي لاءِ ذهني طور تي تيار نه هجي، اهو ڪم ڪيترو ئي آسان ڇونه هجي مشڪل لڳندو آهي۔ پر جڏهن ان ڪم کي ڪرڻ جو پڪو ذهن بڻايو وڃي ۽ ان جي لاءِ وس آهر ڪوشش ڪئي وڃي ته مشڪل کان مشڪل ڪم به آسان ٿي ويندو آهي۔ تنهنڪري علم ِدِين جي اهميت (Importance) کي سمجھو ۽ دنيا ۽ آخرت جي بهتري جي لاءِ هن جذبي کي عملي جامو پارائيندي علم دِين سکڻ ۾ مشغول ٿي وڃو۔ ياد رکو! جهڙي طرح کائڻ پيئڻ اسان جي جسم جي غذا (Diet) آهي، اهڙي طرح علم اسان جي دل جي غذا آهي، جيڪڏهن جسم کي کاڌو نه ڏنو وڃي ته هو ڪمزور ٿي ويندو آهي ۽ مسلسل ڪافي ڏينهن تائين بکايل رهڻ جي ڪري موت به ٿي سگھي ٿو، اهڙي طرح علم دل جي غذا آهي جيڪڏهن دل کي علم جي نور سان منور نه رکيو وڃي ته اها به مُرده ٿي ويندي آهي،

دل جي غذا علم ۽ حڪمت آهي

منقول آهي ته هڪ ڀيري حضرت سيدُنا فتح مُوصلي رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ حاضرين کان پڇيائون: جڏهن مريض کي کائڻ پيئڻ ۽ دوا کان روڪيو وڃي ته ڇا اهو مري ناهي ويندو؟ ماڻهن عرض ڪيو: جي ها۔ ته پاڻ فرمايائون: اهو ئي معاملو دل جو آهي جڏهن ان کي ٽن ڏينهن تائين علم ۽ حڪمت کان روڪيو وڃي ته اها به مري ويندي آهي۔ امام غزالي رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ فرمائن ٿا: حضرت سَيدُنا فتح مُوصلي رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ سچ فرمايو ڇوته دل جي غذا علم ۽ حڪمت آهي ۽ هنن ٻنهي سان دل زنده رهندي آهي جيئن جسم جي غذا کائڻ پيئڻ آهي، جنهن علم کي حاصل نه ڪيو ان جي دل بيمار آهي ۽ ان جو موت يقيني آهي پر ان کي هن ڳالھ جو شعور ناهي هوندو ڇوته دنيا ۾ مشغوليت ان جي احساس کي ختم ڪندي آهي ۽ جڏهن موت ان مشغلن کي ختم ڪري ڇڏيندو آهي ته اها گھڻي تڪليف محسوس ڪندي آهي ۽ ان کي بي انتها افسوس ٿيندو آهي ۽ نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي هن فرمانِ عاليشان: اَلنَّاسُ نِيامٌ فَاِذَامَاتُوْا اِنْتَبَھُوْا ۔ يعني ماڻهو

سُتل آهن جڏهن مري ويندا ته بيدار ٿي ويندا۔ جو به اهو ئي مطلب آهي۔ (حليۃ الاولياء،سفيان ثوري،۷/۵۴، حديث:۹۵۷۶) (لباب الاحياء ،ص۲۴) علمِ دِين سکڻ جو ذهن بنائڻ جي لاءِ هن پُرفتن دَور ۾ ”مدني چينل“ تمام وڏو معاون ۽ مددگار آهي، ٭ مدني چينل تي هلندڙ علمِ دِين سان مالا مال مختلف سلسلا عِلمِ دِين سکڻ جو شوق پيدا ڪندا، هفتيوار مدني مذاڪرو پاڻ به پابندي سان ڏسو ۽ پنهنجي گھروارن کي به ڏيکاريو، هن سان به علمِ دين سکڻ جو شوق پيدا ٿيندو، ٭ هر هفتيوار مدني رِسالي جي مطالعي جو معمول بڻايو، ان جي برڪت سان به علمِ دين سکڻ جو شوق پيدا ٿيندو ٭ ماهنامه فيضانِ مدينه جي مطالعه جي برڪت سان به علمِ دين سکڻ جو شوق پيدا ٿيندو۔ ٭ دعوتِ اِسلامي جي ويب سائٽ (www.dawateislami.net) جو وِزٽ ڪريو، سوين موضوعن تي سُنّتن ڀريا بيان (آڊيو/ويڊيو)، مدني مذاڪرا ڏسڻ ٻڌڻ، علمِ دِين سان مالامال ڪتابن ۽ رسالن جي مطالعي جي برڪت سان علمِ دِين سکڻ جو شوق پيدا ٿيندو۔

اَصحابِ صُفّه جو روشن ڪردار

پياريون پياريون اسلامي ڀينرون! اَصحابِ صُفّه عَلَیْہِمُ الرِّضْوَان پنهنجون زندگيون هن انداز سان گذاريون جو اهي اڄ تائين امّت جي لاءِ مشعلِ راھ جي حيثيت رکن ٿا، انهن جو اخلاص، لِلٰهِيّت، تقويٰ، زُهد(دنيا کان بي رغبتي)، عاجزي، شوقِ علم، خدمتِ دين، قناعت(جيڪو ملي ان تي راضي رهڻ)، توڪل(الله تي ڀروسو ، خوفِ خدا، عشقِ رسول سميت تمام سٺيون خوبيون پنهنجو مٽ پاڻ آهن۔ الله پاڪ هن مقدس گروه کي اهو مقام عطا فرمايو آهي جو فقط انهن جو ذڪر ڪرڻ به ان جون رحمتون ماڻڻ جو سبب آهي۔ الله پاڪ هن مقدس گروه جي تذڪرن ۾ به اهڙو اثر رکيو آهي جو فقط انهن جي ذڪر جي برڪت سان دلين جي دنيا بدلجڻ لڳندي آهي، اڄ به هنن پاڪيزه ماڻهن جي قربانين جو تذڪرو ڪيو وڃي ته بدن ڪانڊارجي ويندو آهي، اڄ به هنن پاڪيزه ماڻهن جي بيڪسي ۽ بيوسي جو سوچيو وڃي ته عقل حيران رهجي ويندا آهن ته ايترين مشڪلاتن ۾ به علمِ دين حاصل ڪرڻ جو اهڙو جذبو مرحبا صد مرحبا! اڄ به هنن پاڪيزه ماڻهن جي واقعن کي ٻڌو وڃي ته شايد اکين مان ڳوڙهن جي برسات جاري ٿي وڃي، اڄ به هنن پاڪيزه ماڻهن جي صبر ۽ دينِ اسلام تي ثابت قدمي جي واقعن جو تذڪرو ٿئي ته همت ۽ حوصلي جون نيون منزلون طئي ٿينديون

آهن، اچو انهن مردانِ خدا مان ٻن مردان حق حضرت سلمان فارسي ۽ حضرت ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُما جو ذڪرِ خير ٻڌون ٿيون:

حضرت سلمان فارسي جي مبارڪ سيرت جون جھلڪيون

اَصحابِ صُفّه ۾ هڪ بزرگ هستي حضرت سَيدنا سلمان فارسي رَضِیَ اللہُ تَعَالٰی عَنْہُ جي آهي، پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ جي ڪنيت ابو عبدالله آهي۔ پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ زُهد وقناعت اپنائڻ وارا، تقويٰ ۽ پرهيز گاري جي گُلن کي پنهنجي دامن تي سجائڻ وارا، حضرتِ سيدنا عيسيٰ عَلَیْہِ السَّلَام جا تابعي ۽ رحمتِ عالَم، نورِ مُجَسَّم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صحابي هئڻ جو اعزاز حاصل ڪرڻ وارا مشهور صحابي آهن ۔ (مراٰة المناجيح،۸/ ۵۲۲ملخصاً) حضرت سلمان فارسي رَضِیَ اللہُ تَعَالٰی عَنْہُ جي اِسلام قبول ڪرڻ جو واقعو نهايت ايمان افروز آهي۔

پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ اَصفَهان (ايران) جا رهاڪو هئا۔ پنهنجي ابن ڏاڏن جي دين کان بيزار ٿي دينِ حق جي تلاش ۾ پنهنجي وطن کان نڪري مختلف ديني ۽ مذهبي رهنمائن جي صحبت اختيار ڪئي۔ هڪ رَاهب اوهان کي ٻڌايو ته هاڻي نبي آخِرُ الزَّمان صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي ظهور جو وقت ويجھو آهي جيڪي دينِ ابراهيمي تي هوندا، ان مقام ڏانهن هجرت (Migration) ڪندا، جيڪو ٻه جبلن جي وچ ۾ هوندو، جتي کجور جا وڻ ڪثرت سان موجود هوندا، ان شخص جي ٻنهي ڪلهن جي وچ ۾ مُهرِنَبُوَّت هوندي، هو هديو قبول ڪندا ۽ صدقو نه کائيندا۔ رَاهب جي انتقال کان پوءِ پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ هڪ قافلي سان گڏ سفر ڪري رهيا هئا۔ واٽ ۾ قافلي وارن جي نيت بدلجي وئي ۽ انهن سندن رَضِیَ اللہُ تَعَالٰی عَنْہُ کي هڪ يهودي وٽ وڪڻي ڇڏيو ۔ (طبقاتِ ابنِ سعد، ۴/۵۶) تقريبًا ڏھ ڀيرا وڪاميا ۔ (بخاري،ڪتاب مناقب الانصار باب اسلام سلمان فارسي۲/۶۰۷، حديث:۳۹۴۶) نيٺ وڪامجندي وڪامجندي مدينه مُنَوَّره پهچي ويا۔ پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ هڪ ڀيرو کجيءَ جي وڻ تي چڙھي کجور لاهي رهيا هئا ته خبر ٻڌائون ته نبي آخِرُ الزَّمَان صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ هجرت فرمائي مديني جي ويجھو مقامِ قُبا ۾ تشريف فرما ٿي چڪا آهن، انهيء وقت دل بيقرار ٿي وئي، يڪدم هيٺ لٿا ۽ خبر آڻڻ واري کان ٻيهر اها رُوح پَروَر خبر ٻڌڻ جي خواهش ظاهر ڪئي، يهودي مالڪ پنهنجي غلام جو تَجَسُّس ۽ بيقراري ڏٺي ته ڪاوڙ ۾ اچي ويو ۽ هڪ زوردار ٿڦ هڻي چوڻ لڳو: توکي هنن ڳالهين سان ڪهڙو مطلب؟ وڃ! ٻيهر ڪم تي لڳي وڃ، پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ دل تي پٿر رکي چپ ٿي ويا پر صَبر ۽ قَرار ته لُٽجي چڪو هو، تنهنڪري جيئن

ئي موقعو مليو ته ڪجھ کجورون هڪ ٿالهيءَ ۾ رکي بارگاھ رسالت ۾ حاضر ٿيا ۽ عرض ڪيائون: اهو صدقو آهي، قبول فرمائي وٺو! پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ صحابَه ڪرام کي فرمايائون: توهان کائي وٺو، ۽ پاڻ نه کاڌائون۔ اهڙي طرح هڪ نشاني ته پوري ٿي، هڪ ڀيري ٻيهر کجورن واري ٿالهي کڻي پهتا ۽ عرض گذار ٿيا ته اهو تحفو آهي، قبول فرمائي وٺو۔ رحمتِ عالم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ صحابَه ڪرام کي کائڻ جو اشارو ڪيائون ۽ پاڻ به تناول فرمايائون۔ اهڙي طرح ٻي نشاني به پوري ٿي ان وچ ۾ پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ رحمت عالم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي ٻنهي ڪلهن جي وچ ۾ ”مُهرِ نَبُوت“ کي به ڏسي ورتائون، ان ڪري يڪدم اسلام قبول ڪري ورتو ۽ ان دَر جا غلام بڻجي ويا جن تي بادشاهن جا ڪنڌ جھڪندا آهن۔ (طبقاتِ ابنِ سعد ، ۴/۵۸ ملخصاً) غزوهءِ خندق ۾ خندق کوٽڻ جو مشورو به پاڻ ئي جو هو۔ (طبقاتِ ابنِ سعد، 2/51) پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ کي سرورِ ڪونين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان تمام گهڻي محبت هئي، پنهنجي وقت جو ڪافي حصو دربارِرسالت ۾ گذاري فيضانِ نبوي سان فيضياب ٿيندا هئا، ان جي بدلي ۾ بارگاھِ رسالت مان سَلمَانُ الخَير ۔ (سنن الڪبري للنسائي، ۶/۹، حديث: ۹۸۴۹) ۽ سَلْمَانُ مِنَّا اَهلَ الْبَيت (سلمان اسان جي اَهلِ بيت مان آهن)۔ (مسند بزار، ۱۳/۱۳۹، حديث: ۶۵۳۴) جهڙي خوشخبري ٻڌڻ جي سعادت ماڻي.

ابوهريره رَضِیَ اللہُ تَعَالٰی عَنْہُ جي آستين (Sleeve) ۾ ننڍڙي ٻلي ڏسي فرمايائون: يا اَبَا هرَيرَة (يعني اي ننڍي ٻلي وارا) ، تڏهن کان پاڻ ابوهريره جي ڪنيت سان مشهور ٿي ويا، پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ نبيِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جون ڪيئي حديثون روايت ڪيون آهن جن جو تعداد پنج هزار ٽي سو چوهتر(5374) آهي۔ (عمدة القاري،۱/۱۹۴، تحت الحديث:۹ ، ملخصاً) سندن عِلمِ دين سان شَغَف جو عالَم هي هو جو جيڪا ڳالھ معلوم نه هوندي هئي حُضُورِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کان بنا حجاب جي پڇي وٺندا هئا۔ (الاصابه، ۷/۳۵۴) پنهنجو وڌ کان وڌ وقت نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صحبتِ بابرڪت ۾ گذاريندا هئا۔ (اسد الغابة، ۶/۳۳۸) پاڻ رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا هڪ ڀيري مون بارگاھ رسالت ۾ عرض ڪيو: يارسول الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ! مان اوهان کان فرمان ٻڌندو آهيان پر وسري ويندا آهن؟ ارشاد فرمايائون: ابوهريره! پنهنجي چادر ڦهلايو، مون ڦهلائي ته مالِڪ جنّت، قاسِمِ نعمت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي دستِ رحمت سان چادر ۾ ڪجھ وجھي ڇڏيائون ۽ فرمايائون: هن کي کڻي ۽ سيني سان لڳائي ڇڏ! مون حڪم جي پيروي ڪئي، ان کان بعد مان ڪا به شيء نه وسريس۔ (بخاري،ڪتاب العلم باب حفظ العلم ۱/۶۲، حديث: ۱۱۹ملخصاً)
