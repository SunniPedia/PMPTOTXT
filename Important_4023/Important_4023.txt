نَـوَیْتُ سُنَّتَ الاعْتِکَاف (مون سنّتِ اعتڪاف جي نيّت ڪئي)

مٺا مٺا اسلامي ڀائرو! جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي اعتڪاف جي نيت ڪري وٺو، ڇوته جيستائين مسجد ۾ رهندؤ اعتڪاف جو ثواب ملندو رهندو۔ ياد رکو! مسجد ۾ کائڻ، پيئڻ، سمهڻ يا سحري، افطاري ڪرڻ، ايستائين جو آبِ زم زم يا دَم ڪيل پاڻي پيئڻ جي به شرعاً اجازت ناهي ، البت جيڪڏهن اعتڪاف جي نيت هوندي ته اهي سڀ شيون ضمناً جائز ٿي وينديون۔ اعتڪاف جي نيت به صرف کائڻ، پيئڻ يا سمهڻ جي لاءِ نه هئڻ گھرجي بلڪ ان جو مقصد الله تعاليٰ جي رِضا هجي۔ ”فتاويٰ شامي“ ۾ آهي: جيڪڏهن ڪو مسجد ۾ کائڻ پيئڻ يا سمهڻ چاهي ته اعتڪاف جي نيت ڪري ۽ ڪجھ دير ذِڪرُاللہ ڪري پوءِ جيڪو چاهي (يعني هاڻي چاهي ته کائي پي يا سمهي سگھي ٿو)

دُرُود شريف جي فضيلت

حضرت سيدنا اِبنِ وَهب رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ کان روايت آهي ته حضرت سيدنا ڪعب رَضِیَ اللہُ تَعَالٰی عَنْہُ امّ المومنين حضرت سيدتنا عائشه صديقه رَضِیَ اللہُ تَعَالٰی عَنْہَا جي خدمت ۾ حاضر ٿيا. ماڻهن رسولِ اڪرم صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ جو ذڪر ڪيو ته حضرت سيدنا ڪعب رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمايو: هر ڏينهن ستر هزار ملائڪ لهندا آهن جيڪي رسولِ پاڪ صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ جي قبر شريف کي گهيري وٺندا آهن، پنهنجا پر وڇائيندا آهن ۽ نبيّ ڪريم صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ تي درود شريف پڙهندا رهندا آهن، ايستائين جو

جڏهن شام ڏسندا آهن ته اهي (آسمانن ڏانهن) چڙهي ويندا آهن ۽ انهن وانگر ٻيا (ملائڪ) لهندا آهن، اهي به انهيء طرح ڪندا آهن، ايستائين جو جڏهن زمين کلندي ته حضورِ انور صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ ستر هزار ملائڪن (جي جهرمٽ) ۾ نڪرندا جيڪي حضور صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ کي پهچائيندا.

(مشڪاة، ڪتاب احوال القيامت ۽ بدء الخلق، باب الڪرامات، 2/401، حديث، 5955)

بيان ڪيل حديث پاڪ جي تحت حڪيم الامت حضرت مفتي احمد يار خان نعيمي رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ فرمائن ٿا: خيال رهي ته هميشه سڀئي ملائڪ ئي حضور صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ تي درود موڪليندا آهن پر هي ستر هزار ملائڪ اهي آهن جن کي پوري عمر ۾ هڪ ڀيرو درٻار جي حاضري جي اجازت هوندي آهي. هي حضرات حضوراڪرم صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ جي برڪت حاصل ڪرڻ جي لاءِ حاضري ڏيندا آهن. جيڪو ملائڪ هڪ ڀيرو حاضري ڏئي ويندو آهي ان کي ٻيهر حاضري جو شرف ناهي ملندو سڄي عمر ۾ صرف ڪجهه ڪلاڪ يعني اڌ ڏينهن جي حاضري نصيب ٿيندي آهي. (حديثِ پاڪ جي هن حصي” ايستائين جو جڏهن زمين کلندي ته حضور ستر هزار ملائڪن سان نڪرندا جيڪي حضور کي پهچائيندا.“ جي وضاحت ۾ لکن ٿا:) يعني قيامت جي ان ڏينهن جي ڊيوٽي وارا ملائڪ حضور کي پنهنجي جهرمٽ ۾ وٺي ربّ تعالٰي تائين پهچائيندا گهوٽ وانگر . (مرآة المناجيح، 8/282 کان 283)

امامِ عشق و محبت اعلٰي حضرت مولانا شاه احمد رضا خان رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ حدائق بخشش ۾ انهيء ڳالهه جي طرف اشارو ڪندي ڇا خوب ارشاد فرمائن ٿا:

ستّر ہزار صُبْح ہیں سَتّر ہزار شام

یُوں بندَگیِ زُلف و رُخ آٹھوں پہر کی ہے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿا. فرمانِ مصطفي ٰ صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ: ” نِیَّۃُ الْمُؤمِنِ خَیْرٌ مِّنْ عَمَلِہٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

درِ رسول تي حاضر ٿيڻ وارو بخشجي ويو

دعوت اسلامي جي اشاعتي اداري مڪتبة المدينه جي 413 صفحن تي مشتمل ڪتاب ”عيون الحڪايت“ جلد 2 صفحي نمبر 308 تي آهي:حضرت سيّدنا محمد بن حَرب هلَالي رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہِ جن بيان ڪيو ته ”هڪ ڀيري مان روضهء رسول تي حاضر هئس ته هڪ اعرابي (عرب جو رهاڪو ڳوٺاڻو) آيو ۽ حضور انور، شافعِ مَحشر، محبوب ربّ اڪبر صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ جي بارگاھ بيڪس پناھ ۾ ائين

عرض ڪرڻ لڳو؛ يا رسول الله صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ الله پاڪ اوهان تي جيڪو سچو ڪتاب نازل فرمايو آهي، ان ۾ (سيپارو 5 سُوْرَۃُالنِّسَا ء ج ي) اها آيت به آهي۔

وَ لَوْ اَنَّهُمْ اِذْ ظَّلَمُوْۤا اَنْفُسَهُمْ جَآءُوْكَ فَاسْتَغْفَرُوا اللّٰهَ وَ اسْتَغْفَرَ لَهُمُ الرَّسُوْلُ لَوَجَدُوا اللّٰهَ تَوَّابًا رَّحِیْمًا(۶۴) (پ 5 النساء 64)

ترجموڪنزالايمان ؛ ۽ جيڪڏهن اهي پنهنجي پاڻ تي ظلم ڪن ته اي محبوب تنهنجي بارگاهه ۾ حاضر ٿين پوءِ الله کان معافي گھرن ۽ رسول انهن جي شفاعت فرمائي ته ضرور الله کي گھڻو توبهه قبول ڪندڙ مهربان لهندا۔

اي منهنجا آقا ۽ مولا صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ! مان بخشڻهار ربّ پاڪ کان پنهنجي ڪوتاهين ۽ گناهن جي معافي گهرندي اوهان جي درٻار ۾ حاضر ٿيو آهيان ۽ اوهان کي ربّ جي درٻار ۾ پنهنجو شفيع بڻايان ٿو۔ اهو چئي اهو عاشق رسول روئڻ لڳو ۽ ان جي زبان تي هي شعر هئا؛

ترجمو ؛ (1) اي اها بهترين ذات! جنهن جو مبارڪ وجود ان زمين ۾ دفن ڪيو ويو ته ان جي عمدگي ۽ پاڪيزگيءَ سان ميدان ۽ دڙا (ڀٽ) معطر ٿي ويا۔ (2) منهنجي جان فدا ٿئي ان نوراني قبر تان جنهن ۾ پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن آرام فرمائي رهيا آهن، جنهن ۾ پارسائي سخاوت، عفو ۽ ڪرم جو وڏو قيمتي خزانو آهي ۔

پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو هڪ ٻيو ڪارنامو فتاوي امجديه به آهي. 4 جلدن تي مشتمل هي ڪتاب صدرالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي بهترين فتون جو مجموعو آهي جنهن کي پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اعلٰي حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي وصال کان بعد تحرير فرمايو هو.

قَبر ۽ دفن جا مَدَني گل

مٺا مٺا اسلامي ڀائرو! اچو! قَبر ۽ دَفن جي باري ۾ ڪجھ مَدَني گل ٻڌڻ جي سعادت حاصل ڪريون ٿا۔ v هڪ قَبر ۾ هڪ کان وڌيڪ بِلا ضَرورت دَفن ڪرڻ جائز نه آهي ۽ ضَرورت هجي ته ڪري سگھو ٿا۔ (بهار شريعت جلداوّل ص۸۴۶، عالمگيري ج۱ ص۱۶۶) v جنازو قَبر کان قبلي ڏانهن رکڻ مُستَحب آهي ته جيئن ميت قبلي واري پاسي کان قَبر ۾ لاٿي وڃي۔ قَبر جي پيراندي (يعني پيرن واري جڳھ) رکي مٿي ڏانهن نه آڻيو۔ (بهارِ شريعت ج۱، ص ۸۴۴) v حَسبِ ضَرورت ٻه يا ٽي ۽ بهتر هي آهي ته قَوي (طاقت ور) ۽ نيڪ ماڻهو قَبر ۾ لهن۔ عَورَت جي ميت مَحارِم لاهين اهي نه هجن ته ٻيا رِشتيدار، اهي به نه هجن ته پرهيزگار لاهين۔ (عالمگيري ج ۱ ص۱۶۶) v عَورَت جي ميت کي لاهڻ کان تختا لڳائڻ تائين ڪنهن ڪپڙي کان لڪائي رکو۔ v قَبر ۾ لاهڻ وقت اها دُعا پڙھو: بِسْمِ الله وَبِالله وَعَلٰي مِلَّۃِ رَسُوْلِ الله ۔ (تنوير الابصار ج۳ص۱۶۶) v ميت کي ساڄي پاسي ليٽايو ۽ ان جو چهرو قبلي ڏانهن ڪيو ۽ ڪفن واري پٽي کولي ڇڏيو ڇوته هاڻي ضَرورت ناهي، نه کولي ته به حَرَج نه آهي۔ (عالمگيري ج۱ص۱۶۶،جوهره ص۱۴۰) v ڪفن جي ڳنڍ کولڻ وارو اها دُعا پڙھي: اَللّٰهمَّ لَا تَحْرِمْنَا اَجْرَه وَلَا تَفْتِنَّا بَعْدَه ترجمو: اي الله! اسان کي ان جي اَجر کان محروم نه ڪر ۽ اسان کي ان کان پوءِ واري فتني ۾ نه وجھ۔ (حاشيۃ الطحطاوي علي مراقي الفلاح ص۶۰۹) v قَبر ڪچّي سرن سان بند ڪريو جيڪڏهن زَمين نَرم هجي ته (ڪاٺي جا) تختا لڳائڻ به جائز آهن۔ (بهارِشريعت ج۱ص۸۴۴) v هاڻي مِٽّي ڏني وڃي، مُستَحَب اهو آهي ته سيراندي کان ٻنهي هٿن سان ٽي ڀيرا مٽّي وجھو۔ پهرين ڀيري چئو مِنْھَا

خَلَقْنٰکُمْ ۔ ٻي ڀيري وَفِیْھَا نُعِیْدُ کُمْ ۔ ٽئين ڀيري وَمِنْھَا نُخْرِجُکُمْ تَارَۃً اُخْرٰی چئو۔ هاڻي باقي مِٽّي ڪوڏر وغيره سان وجھو۔ (جوهره ص۱۴۱) v جيتري مِٽّي قَبر مان نڪتي آهي ان کان وڌيڪ وجھڻ مڪروه آهي۔ (فتاويٰ هنديۃ،۱/۱۶۶) v قَبر چَوڪنڊي (يعني چار ڪونڊن واري) نه بڻايو بلڪه ان ۾ ڍال(هيٺاهين) رکو جيئن اُٺ جو ڪوهان، (دفن کان بعد) ان تي پاڻي ڇڙڪڻ بهتر آهي، قَبر هڪ گرانٺ اُونچي هجي يا معمولي وڌيڪ۔ (بهارشريعت،۱/۸۴۶ مُلَخَّصاً) v دفن کان بعد قَبر تي آذان ڏيڻ ثواب جو ڪم ۽ ميت جي لاءِ نهايت نَفعي بخش آهي۔ (فتاويٰ رضويه،۵/۷۰ ۳ماخوذاً) v مُستَحَب هي آهي ته دفن کان بعد قَبر تي سُوره بَقَرَه جو اوّل ۽ اآخِر پڙھو، سيراندي (يعني مٿي واري پاسي) الٓمّٓ کان مُفْلِحُوْن تائين ۽ پيراندي (يعني پيرن ڏانهن) اٰمَنَ الرَّسُوۡلُ کان ختم سورت تائين پڙهو۔ (بهارشريعت،۱/۸۴۶) v شَجَرو يا عَهدنامو قَبر ۾ رکڻ جائز آهي ۽ بهتر اهو آهي ته ميت جي چهري اڳيان قبلي ڏانهن طاق(کُڏو) کوٽي ان ۾ رکو، بلڪ ”دُرِّمختار“ ۾ ڪفن تي عهد نامو لکڻ جائز چيو آهي ۽ فرمايائون ته ان سبب مَغفِرَت جي اُمّيد آهي، v ميت جي سيني ۽ پيشاني تي بِسْمِ الله الرَّحْمٰنِ الرَّحِيۡمِ لکڻ جائز آهي۔ ائين به ٿي سگھي ٿو ته پيشاني تي بِسْمِ اﷲ شريف لکو ۽ سيني تي ڪلمو شريف لَآ اِلٰه اِلَّا الله مُحَمَّدٌ رَّسُوْلُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پر وهنجارڻ کان پوءِ ڪفن پارائڻ کان پهريان ڪلمي واري آڱر سان لکو روشنائي (INK) سان نه لکو۔ (بهارشريعت ،۱/۸۴۸) خ قَبر مان ميت جون هڏيون ٻاهر نڪرن ته انهن هڏين کي دَفن ڪرڻ واجِب آهي۔ (فتاويٰ رضويه،۹/ ۴۰۶ماخوذاً)

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([2])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([3])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([4])

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
