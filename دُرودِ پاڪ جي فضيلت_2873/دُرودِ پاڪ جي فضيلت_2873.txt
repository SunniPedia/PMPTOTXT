2. مٺا مٺا اسلامي ڀائرو! ولي اَقۡرَب يعني ميّت جو سڀ کان ويجھو رشتيدار، جيڪڏهن جنازي نماز نه پڙهي سگھي ته ان کي قبر تي جنازو پڙهڻ جو اختيار آهي، جيئن بهارِ شريعت جلد اول صفحي 838 تي آهي؛ ولي کان علاوه ڪنهن اهڙي جنازي نماز پڙهائي جيڪو ولي تي مُقدم (افضل) نه هجي ۽ ولي ان کي اجازت به نه ڏني هجي، ته جيڪڏهن ولي نماز ۾ شريڪ نه ٿيو ته نماز کي پڙهي سگھي ٿو ۽ جيڪڏهن مڙدي کي دفن ڪيو ويو ته قبر تي به نماز جنازه پڙهي سگھي ٿو ۽ اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجي مبارڪ زماني ۾ تمام مسلمانن جا ولي اقرب هئا: اعليٰ حضرت، اِمام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ارشادفرمائن ٿا ؛ حُضُور سيّدِ عالَم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زماني ۾ تمام مسلمانن جا ولي اَحَق واَقدَم (سڀ کان وڏا مالڪ) پاڻ حُضُور پُرنور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهن۔ (پاڻ) رَسُولُ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اِرشاد فرمائن ٿا: اَنَااَولٰی بِالْمُؤمِنِیْنَ مِنْ اَنْفُسِھِم يعني مان مسلمانن جي جانن جو وڌيڪ مالڪ آهيان. (مسلم،کتاب الفرائض ،باب ترک مالًافلو رثتہ،۸۷۴/۱۶۱۹) ته جيڪو نمازِ جنازہ حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي اِطلاع ڏيڻ جي بنا ماڻهن پڙهيو، پوءِ حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي ٻيهر پڙهائڻ چاهين ته ائين ئي آهي جيئن پهرين جنازي نماز، ولي کان علاوه ڪنهن غير پڙهائي، وليِ اَحَق ٻيهر اعاده ڪرڻ جو اختيار رکي ٿو.

(1) سڄي رَمَضانُ الْمُبارَڪ يا آخري عشري ۾ تربيتي اجتماعي اعتڪاف جي ترڪيب ٿيندي آهي، جنهن ۾ هزارين اسلامي ڀائرن کي فرض عُلُوم سکاريا ويندا آهن ۽ سنتن جي مطابق مدني تربيت ڪئي ويندي آهي .

(2) مدني قافلي جا مُسافر اسلامي ڀائر به مسجدن ۾ قيام ڪندا آهن ائين پنهنجو اڪثر وقت مسجد ۾ گذارڻ علم دين حاصل ڪرڻ جو موقعو ملي ويندو آهي .

(3) بعدِ فجر مدني حلقي جي ترڪيب ٿيندي آهي جنهن ۾ گھٽ ۾ گھٽ ٽي آيتون، ترجمہءِ ڪنزالايمان و تفسيرخزائن العرفان / تفسير نور العرفان يا تفسير صراط الجنان سان گڏ ٻڌايون وينديون آهن.

(4) مختلف نمازن کان پوءِ مدني درس يعني اميرِ اهلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي ڪتابن ۽ رسالن جي ذريعي مدني گل ڏنا ويندا آهن.

(5) دعوتِ اسلامي جي ڪن مدني مرڪزن (فيضانِ مدينه) ۾”مدني تربيت گاهون“ موجود آهن، بنيادي ضرورياتِ دين، نماز جو عملي طريقو، مختلف موضوعن تي سنتون ۽ آداب سکڻ سيکارڻ جو عمل جاري

(6) مدرسة المدينه بالغان جي ذريعي دُرست مخارج سان قرآنِ ڪريم پڙهڻ سيکاريو ويندو آهي ان جي برڪت سان به اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مسجدون آباد رهنديون آهن.

(7) اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ وقت بوقت مختلف ڪورسز (مثلاً 63 ڏينهن جو مدني تربيتي ڪورس، 41 ڏينهن جو مدني انعامات ومدني قافله ڪورس، ڪردار ساز12 ڏينهن جو مدني ڪورس) ۽ مختلف موقعن تي ٿيندڙ مدني مشورن ۽ تربيتي اجتماعن جي ذريعي ڪثير عاشقان رسول مسجدن کي آباد رکڻ جي سعادت حاصل ڪندا آهن.

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته وات ۾ بدبو هجڻ جي صورت ۾ جيستائين بو ختم نه ٿئي مسجد ۾ وڃڻ منع آهي۔شيخِ طريقت، اميرِ اَهلسُنَّت، بانيِ دعوتِ اسلامي حضرت ِ علامہ مولانا ابُو بلال محمد الياس عطارقادري دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ هن باري ۾ مدني گل ارشاد فرمائن ٿا: بُک کان گهٽ کاڌو کائڻ جي عادت بڻايون، يعني اڃا خواهش باقي هجي ته هٿ روڪي ڇڏيون۔ جيڪڏهن خوب پيٽ ڀري کائيندا رهيؤ، ۽ وقت بي وقت سيخ ڪباب، برگر، آلو ڇولا، پزّا، آئس ڪريم، ٿڌيون بوتلون وغيره کائيندا پيئندا رهيؤ انهي سبب پيٽ خراب ٿي ويو ۽ خدا نخواسته وات مان بدبوءِ اچڻ جي بيماري لڳي وئي ته سخت امتحان ۾ پئجي ويندو، ڇو ته وات مان بدبوءِ اچي ته مسجد ۾ داخل ٿيڻ حرام آهي، ايتري تائين جو جنهن وقت وات مان بدبوءِ اچي رهي هجي ان وقت باجماعت نماز پڙهڻ جي لاءِ به مسجد ۾ اچڻ گناهه آهي۔ ڇو ته فڪرِ آخرت جي گهٽتائيءَ سبب ماڻهن جي وڏي اڪثريت ۾ کاڌي جو حرص گهڻو ۽ اڄ ڪلهه هر طرف ”فوڊ ڪلچر“ جو دؤر هلي رهيو آهي۔ هن سبب کان وڏو تعداد آهي جنهن جي وات مان بدبوءِ ايندي آهي۔ منهنجو ڪافي تجربو آهي ته جڏهن ڪو وات ويجهو ڪري ڳالهه ڪندو آهي ته ان جي وات جي بدبوءِ جي سبب ساهه روڪڻو پوندو

حضرت سَيِّدُنا عيسٰي عَلٰی نَبِیِّنَا وَ عَلَیْہِ الصَّلٰوۃُ وَالسَّلَام انهن ماڻهن کي مسجد ۾ وڌيڪ ويهڻ کان منع ڪندا هئا جيڪي مسجد جي آدابن کان اڻ واقف هئا، هڪ ڀيري پاڻ عَلَیْہِ الصَّلٰوۃُ وَالسَّلَام مسجد ۾ ڪن ماڻهن کي ويٺل ڏٺو، جيڪي فضول ڳالهيون ڪري رهيا هئا ته پاڻ پنهنجي چادر کي ويڙهي انهن کي ماريو ۽ اتان کان ڪڍي ڇڏيو ۽ فرمايو؛ توهان الله عَزَّ وَجَلَّ جي گھرن کي دنيا جي بازار ٺاهي رکيو آهي حالانڪ اهي ته آخرت جا بازار آهن.

حضرتِ سيّدُنا سائب بن يزيد رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا، مان مسجد ۾ ليٽل هئس ته مون کي ڪنهن پٿري هنئي۔ مون ڏٺو ته اهي حضرتِ سيّدُنا امير المُؤمنين عمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ هئا۔ انهن مون کي فرمايو: ”وڃ ۽انهن ٻنهي شخصن کي مون وٽ وٺي اچو!“۔ مان انهن ٻنهي کي وٺي آيس۔ حضرتِ سيّدُنا عمر رَضِىَ اللهُ تَعَالٰی عَـنْهُ انهن کان پڇا فرمايو،” توهان ڪهڙي شهر سان تعلق رکو ٿا ؟ عرض ڪيائون، طائف سان. فرمايائون، ”جيڪڏهن توهان مدينہءِ منوره جا رهڻ وارا هجو ها ته مان توهان کي ضرور سزا ڏيان ها (ڇو ته) توهان رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مسجد ۾ پنهنجو آواز بلند ٿا ڪريو! (بخاری،کتاب الصلٰوة،باب رفع الصوت فی المساجد،۱ / ۱۷۸،حدیث۴۷۰)

اچو! مسجد ۾ دنيوي ڳالهين ۽ مسجد ۾ کلڻ جي مذمت تي روايتون ٻڌون ٿا؛

(1) هڪ زمانو اهڙو ايندو جو ماڻهو مسجد ۾ دنيا جون ڳالهيون ڪندا ته ان وقت انهن ماڻهن وٽ توهان نه ويهو، الله عَزَّوَجَلَّ کي انهن ماڻهن جي ڪجھ پرواھ ناهي۔ (شعب الإيمان، باب في الصلٰوت، فصل المشي إلی المساجد، ۳ / ۸۶ ،حدیث: ۲۹۶۲)

(2) مسجد ۾ دنياوي ڳالھ ٻولھ نيڪين کي ائين کائي ڇڏيندي آهي جيئن چوپايا گاھ کي کائيندا آهن. (اتحاف السادة المتقین،کتاب اسرارالصلوة ومھماتھا،الباب الاول،۳ / ۵۰)

(3) مسجد ۾ کِلڻ قبر ۾ اونداهي (آڻيندو) آهي۔“

(الجامع الصغیر، فصل فی المحلی بأل من ھذاالحرف،۱ / ۳۲۲ ،حدیث:۵۲۳۱)

مٺا مٺا اسلامي ڀائرو! انهن سڀني وعيدن کي پيشِ نظر رکندي پاڻ کي هلاڪت کان بچايو ۽ مسجد جو ادب ڪندي انهي ڳالھ جو خيال رکو ته

مسجد ۾ هلڻ مهل پيرن جوڌمڪ(يعني آواز) پيدا نه ٿئي ۽ لٺ (WALKING STICK) ، ڇٽي، هٿ جو پکو، چَپّل، ٿيلهو (BAG) ٿانوَ وغيره ڪا به شيء اهڙي طرح نه رکو جو آواز پيدا ٿئي جيڪڏهن موبائل فون هجي ته مسجد ۾ ان جي گھنٽي به بند رکو، افسوس! اڄڪلھ ان جو احتياط تمام گھٽ ڪيو ويندو آهي ايستائين جو مسجد الحرام ۾ به اهو به عين خانه ڪعبه جي طواف ۾ ماڻهن جي موبائل فون جون گھنٽيون بلڪ مَعَاذَ اللہ عَزَّوَجَلَّ ميوزيڪل ٽيونز (Musical Tunes) وڄنديون رهنديون آهن حالانڪ ميوزيڪل ٽيون ته مسجد جي علاوه به ناجائز ۽ گناھ آهي.(ته مسجد ۾ ته حڪم ويتر وڌيڪ سخت آهي.)

مٺا مٺا اسلامي ڀائرو! سُنَّتن جي خدمت جي لاءِ دعوت اسلامي جي مدني ڪمن ۾ وڌ کان وڌ حصو وٺو۔ ذيلي حلقي جي 12 مدني ڪمن مان هڪ مدني ڪم ”مدني قافلي ۾ سفر ڪرڻ“ به آهي، اَمِيرُالمؤمنين حضرت سَيِّدُنا عُمرفَاروقِ اَعظَم رَضِیَ اللہُ تَعَالٰی عَنْہُ فرمائن ٿا؛ مان ۽ منهنجو هڪ انصاري پاڙي وارو بَنُو اُمَيّه بن زيد (جي پاڙي) ۾ هئا سين، جيڪو مديني شريف جي بلندي تي هو اسان واري واري سان سرڪارِ والا تَبار،شفيعِ روزِ شُمار صَلَّی اللہُ تَعَالٰی عَلَیْہِ وَاٰلِہٖ وَسَلَّمَ جي بارگاھ ۾ حاضر ٿيندا هئاسين، هڪ ڏينهن اهي مديني شريف ويندا ۽ واپس موٽي ان ڏينهن جي وحي جو احوال مون کي ٻڌائيندا ۽ هڪ ڏينهن مان ويندو هئس ۽ موٽي ان ڏينهن جي وحي جي خبر جو حال ان کي ٻڌائيندو هئس۔ (صَحیح بُخاری ج ۱ ص۵۰حدیث۸۹)

مٺا مٺا اسلامي ڀائرو! جيئن ته اسان ٻڌو مدني قافلي ۾ سفر ڪري به مسجدن کي آباد ڪرڻ جي ڪوشش ڪئي ويندي آهي مدني قافلي ۾

سفر ڪرڻ جي سعادت حاصل ڪرڻ وارا عاشقان رسول نه رڳو پاڻ مسجدن کي آباد ڪندا آهن، بلڪ ٻين مسلمانن کي به نيڪي جي دعوت جي ذريعي مسجدون آباد ڪرڻ جي ترغيب ڏيندا آهن، اچو ته اسان به هٿو هٿ مدني قافلي ۾ سفر جي ذريعي الله عَزَّ وَجَلَّ جي گھر ”مسجد“ کي آباد ڪرڻ جي نيت سان مدني قافلي ۾ سفر جي نيت ڪيون ٿا.

اسان کي به هر مهيني گھٽ مان گھٽ ٽن ڏينهن جي مدني قافلي ۾ ضرور سفر ڪرڻ گھرجي جتي اسان کي علم دين حاصل ڪرڻ جو موقعو ملندو اتي نيڪي جي دعوت عام ڪرڻ جو ثواب پڻ ملندو، مدني قافلي ۾ سفر جون گھڻيون برڪتون آهن، ترغيب جي لاءِ هڪ مدني بهار ٻڌون ٿا۔

بابُ المدينه (ڪراچي) جي علائقي ناظم آباد جي مقيم هڪ وڏي عمر جي اسلامي ڀاءُ مدني قافلي مان حاصل ٿيڻ وارين برڪتن جو ڪجھ ائين تذڪرو ڪيو آهي ته مان تقريباً 19 سالن کان ساھ جي مرض ۾ مبتلا هئس، ڪڏهن ته مرض جي شدت جي ڪري مون کي وڌيڪ تڪليف ٿيندي هئي، ڪڏهن آڌي رات جو طبعيت خراب ٿي ويندي هئي ته انهي وقت ڊاڪٽر وٽ وڃڻو پوندو هو، مطلب ته مان بيحد پريشان رهندو هئس، علاج معالجي ۾ ڪا گھٽتائي نه ڪئي، هر ڏينهن اٽڪل 150 روپيا دوائن تي خرچ ٿيندا هئا، جنهن سان وقتي طور تي سڪون ملندو هو پر مون کي مستقل سڪون نه مليو، هر وقت انهيء فڪر۽ پريشاني ۾ رهندو هئس ته ان مرض مان ڪيئن شفا ملندي منهنجي خوش قسمتي جو هڪ ڀيري دعوت اسلامي جي عاشقان رسول سان گڏ مدني قافلي ۾ سفر جي سعادت حاصل ٿي، مدني قافلي جي برڪتن سان جتي منهنجا ڏينهن رات عبادت الهي ۾ گذريا ۽
