تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم به انهيءَ مَقصَد جي لاءِ هن دُنيا ۾ تَشرِيف فرما ٿيا۔ حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان پوءِ اُمَّت کي نيڪي جي دَعوَت ڏيڻ ۽ ان جي تربِيت جو اِهو ڪم بارگاھِ نَبُوَّت جي براھِ راست تَربِيت يافته صَحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان جن سنڀالي ورتو۔ صحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان کان پوءِ به هر دؤر ۾ بُزرگانِ دين اسلامي تعليمات جي نُور سان ماڻهن جي دِلين کي مُنوَّر ڪيو۔ حَضرَتِ سَيِّدُنَا داتا علي هجويري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن به انهيءَ مَقصَد کي پنهنجو شِعار بَڻايو ۽ نيڪي جي دعوت جي هن اَهم فَرِيضي کي نِڀائڻ جي لاءِ مرڪزالاولياء (لاهور) پهچي ويا۔ داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن (مرڪزالاولياء) لاهور ۾ عِلم ۽ حِڪمَت جي اهڙا دَريا وهايا جو اهو شهر جيڪو پهريان ڪُفر ۽ شِرڪ جي اونڌاھ سان ڍڪيل هو، حُضُور سَيِّدُنَا داتا علي هجويري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي ڪوشِشن سان قَلعهء اِسلام بڻجي ويو، داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي حُسنِ اَخلاق، حُسنِ ڪردار ۽ نَرم گُفتار سان ڪيترن ئي دلين ۾ اوهان جي مَحَبَّت راسخ ٿي وئي۔ مرڪزالاولياء (لاهور) ۾ داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي قيام جي مُدَّت تقريبًا ٽيھ (30) سال هئي۔ (اللّٰه کے خاص بندے، ص۴۶۸) هن پُوري عَرصي ۾ داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڏينهن رات دِين جي تَبليغ ۾ مَشغُول رهيا، توهان جي بي داغ سِيرت، دِلڪش گُفتُگو، پُرنُور شَخصِيت ۽ دِلين ۾ لهڻ وارا ارشاداتِ عالِيه ماڻهن کي ڪُفر ۽ ضَلالَت (گمراهي) جي دَلدَل کان ڪڍي ڪري هدايت جي راھ تي گامزَن ڪندا رهيا۔ مرڪزالاولياء (لاهور) ۾ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پنهنجي قِيام گاه جي ويجھي هڪ جڳھ مَسجِد جي تَعمِير جو سنگِ بنياد رکيو ۽ ان مسجد جي تعمير جي وَقت داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن خود مزدورن جيان ڪم ڪيو ۽ وڏي مَحَبَّت ۽ جَذبي سان ان جي تعمير ۾ اڳتي اڳتي رهيا، مرڪزالاولياء (لاهور) شهر ۾ اها پهرين مسجد هئي جيڪا هڪ وَلي الله جي هٿان تعمير ٿي۔ (اللّٰه کےخاص بندے، ص۴۶۹) حَضرَتِ سَيِّدُنَا داتا گَنۡج بَخۡش رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پوري زندگي خُوب مَحَبَّت ۽ لگن سان خدمتِ دين جو ڪم سرانجام ڏنو، ڏُکايل اِنسانِيت کي اَمن ۽

سُڪون جو پَيغام ڏنو ۽ پنهنجي مُرِيدن ۽ مُحِبِّين جي دِيني ۽ دُنياوي حاجتن کي پُورو فرمايو۔ اڄ به داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي مَزارِ فائضُ الاَنوار مان پنهنجي عقيدت مندن جي حاجت رَوائي فرمائيندا آهن، انهن جون پريشانيون حَل فرمائيندا آهن ۽ پنهنجي رُوحاني فيضان سان جنهن کي چاهيندا آهن مالامال ڪندا آهن۔

میں ہوں عصیاں کا مریض اور تم طبیبِ عاصِیاں

ہو عطا مجھ کو گناہوں کی دوا داتا پیا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اَوۡلِياءُالله جي مَزارات تي حاضِري جي بَرڪت سان دُعائون قَبول ٿينديون آهن، مُشڪلِاتن ۽ مُصيبتن کان نَجات ملندي آهي، خاص اِن نَظَريي سان اولياءِ ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي مَزارات تي وڃڻ به اسان جي بُزرگن جو طريقو رهيو آهي۔ جيئن ته

حَضرَتِ سَيِّدُنَا داتا گَنۡج بَخۡش علي هجۡويري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو به هي مَعمُول هو ته

اُهي بُزرگانِ دِين رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي مَزارات تي حاضِري ڏيندا آهن، مَزارات تي حاضِري

جي مُتَعلِّق پنهنجا ڪيترائي واقِعا اُنهن پنهنجي مَشهور ۽ مَعرُوف ڪتاب ”ڪَشۡفُ الۡمَحۡجُوب“ ۾ دَرۡج ڪيا آهن۔ اچو ته اُن مان ڪجھ سماعت ڪريون ٿا: جيئن ته

حَضرَتِ سَيِّدُنَا داتا گَنۡج بَخۡش علي هجْويري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: ”آئون هڪ ڏينهن سفر ڪندي ملڪ شام ۾ مؤذنِ رسول، حضرت سَيِّدُنَا بلال رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي مزار شريف تي حاضر ٿيس، اُتي منهنجي اَک لڳي وئي ۽ مون پاڻ کي مَڪهء مُعظَّمه (زَادَھَا الـلّٰـهُ شَرَفًاوَّتَعۡظِیۡمًا) ۾ ڏٺم۔ ڇا ٿو ڏسان ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم قبيله بني شيبه جي دروازي تي موجود آهن ۽ هڪ وڏي عُمر واري شخص کي ڪنهن ننڍي ٻار وانگر کَڻي ويٺا آهن، مون مَحَبَّت وچان بي قرار

1. ٿي ڪري حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي طرف ڊوڙيس ۽ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مبارڪ قدمن کي بَوسو ڏنم، دل ئي دل ۾ هن ڳالھ تي وڏو حيران به ٿيس ته هي ضعيف شخص ڪير آهي؟ ايتري ۾ الله عَزَّوَجَلَّ جي مَحبوب، داناءِ غُيوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم قوتِ باطني ۽ علمِ غيب جي ذريعي منهنجي حيرت ۽ اِسۡتِعۡجاب (تعجب) جي ڪيفيت ڄاڻِ ورتي ۽ مون سان مخاطب ٿي ڪري فرمايو: ”هي ابوحنيفه آهن ۽ تنهنجا امام آهن۔

(کشف المحجوب ،ص،۲۱۶ ضیاء القرآن ملخصاً)

2. مزيد فرمائن ٿا: هڪ ڀيري مون کي هڪ مُشۡڪل دَرپيش آئي، مون اُن جي حل جي ڪوشِش ڪئي پر ڪامياب نه ٿيس، اِن کان پهريان به مون تي اهڙي ئي مُشۡڪل آئي هئي، ته مون حَضرَتِ شيخ ابو يزيد رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مَزار شَريف تي حاضِري ڏني هئي ۽ منهنجي اها مُشۡڪل آسان ٿي وئي هئي۔ هن ڀيري به مون اِرادو ڪيو ته اُتي حاضِري ڏيان۔ انهيءَ نِيت سان ٽن (3) مهينن تائين اُنهن جي مَزار مُبارَڪ تي چلّه ڪَشي ڪئي، ته جيئن منهنجي مُشۡڪل حل ٿي وڃي۔ (کشف المحجوب، ص۶۵)

3. حضرت اَبُوالعباس قاسم بن مَهدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي باري ۾ حُضُورداتا گنج بخش رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اِرۡشاد فرمائن ٿا ته اڄ تائين انهن جو مَزار ”مَرۡوَ“ (تُرڪمانستان) ۾ مَوۡجُود آهي ۽ تمام مَشهور ۽ مَعۡروف آهن، ماڻهو اُتي مُرادون گھُرڻ ويندا آهن ۽ وڏيون وڏيون مُشڪلاتون حَل ڪرڻ جي لاءِ انهن کان طالبِ اِمداد (مدد گھُرندا) آهن ۽ اُنهن جي اِمداد ڪئي ويندي آهي، اها ڳالھ تمام مُجَرَّب (يعني ڪيترا ئي ڀيرا آزمائي وئي) آهي۔

(کشف المحجوب،ص۱۶۵)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته! سَيِّدُنَا داتا علي هجويري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو به هي عقيدو هو ته نه صرف مَزارات تي وڃڻ باعِثِ برڪت آهي بلڪه اُتي مُشڪلاتون به حَل ٿينديون آهن ۽ هي سڀ صاحِبِ مَزار ئي جو فَيضان هوندو آهي۔ مُمڪن آهي ته ڪنهن کي هي وَسۡوَسو اچي ته اَولياءِ ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جو فَيض ڪيئن مِلي سگھي ٿو؟ ڇو جو اُهي ته وَفات ڪري چڪا آهن۔ ياد رکو! اَولِياءِ ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی ربِّ ڪائنات عَزَّوَجَلَّ جي عِنايتن سان مَزارات ۾ نه صرف حَيات هوندا آهن بلڪه زائِرِين (پنهنجي مَزارات جي زِيارَت ڪرڻ وارن) جي هدايت ۽ مدد به فرمائن ٿا۔

حضرت سَيِّدُنَا اِمام اسماعيل حَقّي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: انبياء، اَولِياء ۽ شُهداء جا جسمَ قَبرن ۾ نه مُتَغَير ٿيندا آهن ۽ نه ئي بوسِيدا ٿيندا آهن، ڇو جو الله عَزَّوَجَلَّ اُنهن جي جِسمن کي اِن خَرابي کان جيڪا گوشت جي ڳَرڻ سَڙڻ کان پيدا ٿيندي آهي ،مَحۡفُوظ رکيو آهي۔ (تفسير روح البيان، ج۳، ص۴۳۹)

شَيخ عبدُالحق مُحدّثِ دِهلوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: الله عَزَّوَجَلَّ جي اَولياء هن دارِ فاني کان دارِ بَقا جي طرف لَڏي ويا آهن ۽ پنهنجي پَرۡوَرۡدۡگار جي آڏو زِندھ آهن اُنهن کي رِزۡق ڏنو ويندو آهي، اُهي خُوش حال آهن ۽ ماڻهن کي ان جو شُعُور ناهي۔ (اشعۃ اللمعات، کتاب الجہاد، باب حکم الاسراء ،۳/ ۴۲۳)

حَضرَتِ علّامه علي قاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: اَوۡلِياءُ الله جون ٻئي حالتون (زِندَگي ۽ مَوت) ۾ اَصۡلاً (ڪو) فَرۡق ناهي،انهيءَ لاءِ چيو ويو آهي ته اهي مرندا ناهن بلڪه هڪ گھر کان ٻي گھر ۾ تشريف وٺي ويندا آهن۔

(مرقاۃ شرح مشکوٰۃ باب الجمعۃ فصل الثالث، ۳/۴۵۹)(ازفتاوی رضویۃ، ج۹، ص۴۳۳)

کون کہتا ہے ولی سَب مَر گئے؟

قَیْد سے چُھوٹے وہ اپنے گھر گئے!

مٺا مٺا اسلامي ڀائرو! انهن جَلِيلُ الۡقَدۡر آئمهء ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي تَصۡريحات (وضاحتن) مان اهو مَعلُوم ٿيو ته اَنبياءِ ڪرام عَـلَـيْهِمُ السَّلَام ، شُهداءِ عُظام ۽ اَولِياءِ ربّ سلام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی سڀ پنهنجي پنهنجي مَزارات ۾ زِندھ هوندا آهن ۽ تَصرُّف به فرمائن ٿا۔ انهيءَ لاءِ صرف عوام ئي نه بلڪه وڏن وڏن عُلَماء ۽ فُضَلاء جو هي مَعمُول رهيو آهي ته اُهي پنهنجي مُشڪلِاتن جي حَل جي لاءِ اَولِياءِ ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي مَزارات تي حاضِري ڏيندا هئا۔ اچو ته اِنهيءَ باري ۾ بُزرگانِ دِين رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جا ٽي (3) فرمان ٻُڌون ٿا۔

جيئن ته

1. مشهور حَنۡبَلي مُحدّث حضرت امام خَلَّال ابوبڪر احمد بن محمد بَغۡدادي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: مون کي جڏهن ڪو مُعامَلو (پريشاني) دَرپيش ايندي آهي، آئون اِمام مُوسيٰ ڪاظِم بن جَعۡفَر صادِق رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جي مَزار تي حاضِر ٿي ڪري سندن جو وسيلو پيش ڪندو آهيان۔ الله عَزَّوَجَلَّ منهنجي مُشۡڪل کي آسان ڪري مون کي منهنجي مُراد عَطا فرمائي ڇڏيندو آهي۔

2. ڪروڙن شافِعِين جا پيشوا حَضرَتِ سيدُنا امام محمد بن اِدريس شافعي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: مون کي جڏهن ڪا حاجت پيش ايندي آهي ته آئون ٻه (2) رَڪعَت نَماز ادا ڪري امامِ اَعظم ابُو حنيفه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مَزارِ پُر اَنۡوار تي وڃي ڪري دُعا گھُرندو آهيان، الله عَزَّوَجَلَّ منهنجي حاجت پوري ڪري ڇڏيندو آهي۔ (الخيرات الحِسان ص ۹۴)

حاضِري جي ڪهڙا ڪهڙا فائدا آهن؟ وغيره وغيره هي سڀ ڄاڻڻ جي لاءِ دعوتِ اسلامي جي اِشاعَتِي اداري مَڪتبةُ الۡمَدِينه جي مطبوعه 48 صفحن تي مُشتَمِل رسالي ” مَزاراتِ اولياء کی حکايتيں “ هديةً حاصِل فرمائي ڪري اِن جو مُطالعوفرمايو، اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ معلومات ۾ڪافي اضافو ٿيندو۔ اچو ته انهيءَ رسالي مان مَزارات تي حاضِري جو طريقو ۽ ان جا مَدَني گُل ٻُڌون ٿا۔

(جيڪڏهن ڪو شَخۡص ولي اللّٰه جي مَزار شريف يا) ڪنهن به مُسَلمان جي قَبۡر جي زِيارَت لاءِ وڃي ته مُستَحَب هي آهي ته پهريان پنهنجي مَڪان تي (غَير مڪروھ وَقۡت ۾) ٻه (2) رَڪعَت نَفۡل پڙھي، هر رَڪعَت ۾ سُوْرَۃُالْفَاتِحَہ کان پوءِ هڪ (1) ڀيرو اٰیَۃُ الْکُرْسِی ۽ ٽي (3) ڀيرا سُوْرَۃُ الْاِخْلَاص پڙھي ۽ اُن نَماز جو ثَواب صاحبِ قَبۡر کي پهچائي، الله عَزَّوَجَلَّ اُن فَوت شُده ٻانهي جي قَبۡر ۾ نُور پيدا ڪندو۽ اِن (ثَواب پهچائڻ واري) شَخۡص کي تمام گھڻو ثَواب عَطا فرمائيندو۔ (فتاوٰي عالمگيري ج۵ص۳۵۰) پوءِ سُٺيون سُٺيون نيّتون ڪرڻ کان پوءِ مَزارات جي طرف روانو ٿئي ۽ (زائر يعني زِيارت ڪرڻ واري کي گھرجي ته اَولِياءِ ڪرام رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي) مَزاراتِ طيبات تي حاضِر هئڻ ۾ پيراندي (يعني پَيرن) جي طرف کان وڃي ۽گهٽ ۾گهٽ چار (4) هٿ جي فاصِلي تي مُواجھ (يعني چِهري جي سامهون) بيهي ۽ مُتَوَسِّط (يعني درمِياني) آواز ۾ (هن طرح) سَلام عَرۡض ڪري: اَلسّلَامُ عَلَیْکَ یَاسَیِّدِیْ وَرَحمَۃُ اللّٰہِ وَبَرَکَاتُہٗ ، پوءِ” دُرُودِ غَوْثيه “ ٽي (3) ڀيرا، اَلۡحَمۡد شريف هڪ (1) ڀيرو، اٰیَۃُ الْکُرْسِی هڪ(1) ڀيرو ، سُوْرَۃُ الْاِخْلَاص ست (7) ڀيرا، پوءِ ” دُرُودِ غَوْثيه “ ست (7) ڀيرا ، ۽ وَقۡت اجازت ڏئي ته سُوْرَۂ یٰسٓ ۽ سورۂ مُلک به پڙھي ڪري الله عَزَّوَجَلَّ کان دُعا ڪري ته الٰهي! هن قِراءَ ت تي مون کي اِيترو ثَواب ڏي جيڪو تنهنجي ڪرم جي قابِل آهي، نه ايترو جيڪو منهنجي عَمَل جي قابِل آهي ۽ ان کي منهنجي طرف کان هن مَقبول ٻانهي کي نَذۡر پهچاءِ۔ پوءِ پنهنجو جيڪو مطلب، جائز (۽)

شرعِي هجي، اُن جي لاءِ دُعا ڪري ۽صاحبِ مَزار جي رُوح کي الله عَزَّوَجَلَّ جي بارگاھ ۾ پنهنجو وسيلو قَرار ڏئي، پوءِ اهڙي طرح سَلام ڪري واپَس اچي۔

مٺا مٺا اسلامي ڀائرو! عام طور تي ڏٺو ويندو آهي ته مَزاراتِ اَوۡلياء تي نِياز به تقسيم ڪيو ويندو آهي، هي به صاحبِ مَزار کي ايصالِ ثواب ڪرڻ جو هڪ طريقو آهي۔ يقينًا الله عَزَّوَجَلَّ جي رِضا حاصِل ڪرڻ جي لاءِ نِياز وغيره تقسيم ڪرڻ جي وڏي فَضِيلت هي، جيئن ته اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فتاويٰ رَضَوِيه جلد 24 صفحي 521 تي لِکن ٿا: کاڌوکارائڻ، لنگر تقسيم به مَنۡدُوب (يعني سٺو عمل) ۽ باعِثِ اجر آهي، حديث ۾ آهي: رَسُولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمائن ٿا: اِنَّ الله عَزَّوَجَلَّ یُبَاہِیْ مَلٰئِکَتَہٗ بِا لَّذِیْنَ یُطْعِمُوْنَ الطَّعَامَ مِنْ عَبِیْدِہٖ۔ يعني الله عَزَّوَجَلَّ پنهنجي اُنهن ٻانهن سان گڏ (هوندو آهي) جيڪي ماڻهن کي کاڌو کارائيندا آهن، فِرِشتن تي مُباهات (يعني فَخۡر) فرمائيندو آهي۔

(الترغيب والترھيب ،ج۲،ص۳۸، الحديث۲۱،ازفتاوي رضويه، ج۲۴،ص۵۲۱)

پر لنگر (نياز) تقسيم ڪندي هن ڳالھ جو خَيال ضَرور رکو ته ڪنهن به طرح لنگر (نياز) جي بي حُرمَتي ۽ بي ادبي نه ٿئي، نه پَيرن ۾ اچي، نه مَزار شريف جو فَرش آلوده ٿئي، ڌِڪ ڌِڪان کان بچڻ جي لاءِ اسلامي ڀائرن کي ويهاري ڪري يا قَطار بڻائي ڪري لنگر (نياز) تقسيم ڪيو وڃي، اچڻ وارن زائرين جي حُقُوق جو خَيال رکيو وڃي جو لنگر (نياز) تقسيم ڪرڻ جي سبب انهن کي حاضِري ڏيڻ ۾ ڪنهن قِسۡم جي تڪليف کي منهن نه ڏيڻو پوي ۽ خاص طور تي مَزار شريف جي تعظيم جو مڪمل اِهتِمام ڪيو وڃي، ائين نه ٿئي جو هڪ طرف لنگر (نياز) تقسيم ڪري اجر ۽ ثَواب جا مُسۡتَحِق بڻجي ۽ ٻي طرف مَزار شريف جي بي اَدَبي جا مُرتَڪب

ٿي وڃي۔ کاڌي جي نِياز سان گڏوگڏ مَڪتبةُ الۡمَدِينه جي مَطبُوعه ڪتابون ۽ رسالا تقسيم ڪري به بي شُمار ثَوابِ جارِيه، صاحِبِ مَزار جي خدمت ۾ پيش ڪري سگھجي ٿو۔ (مزاراتِ اوليا کي حڪايات، ص۱۷)

مٺا مٺا اسلامي ڀائرو! ياد رکو! نه صرف مَزار شريف ۾ لنگر (نياز) تقسيم ڪندي بلڪه هر جڳھ کاڌو کائيندي ۽ کارائيندي اِحتِياط ڪرڻ گھرجي ته ڪٿي کاڌي جا ذرا وغيره ضائع نه ٿين۔ جيڪڏهن ڪٿي ڪو لُقۡمو (گِراھ) ڪِري پوي ۽ تنفيرِ عوام (ماڻهن جي نفرت) جو انديشو به نه هجي ته ماڻهن جي پرواھ نه ڪندي کڻي ڪري کائڻ گھرجي، اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ ان جون برڪتون نصيب ٿينديون ۔

اُمُّ الۡمُؤمِنين سَيِّدتنا عَائِشه صِدِّيقه رَضِیَ اللهُ تَعَالٰی عَنْهَا فرمائن ٿيون،” ٻنهي جهانن جا سلطان، شهنشاههِ ڪون و مڪان، رحمتِ عالميان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم گهر مبارڪ ۾ تشريف کڻي آيا، مانيءَ جو ٽڪر ڪِريل ڏٺائون ته ان کي کڻي ڪري اگهيائون پوءِ کاڌائون ۽ فرمائڻ لڳا، ” عائشه! (رَضِیَ اللهُ تَعَالٰی عَنْهَا) سٺي شيء جو احترام ڪريو، جو اها شيء (يعني ماني) جڏهن ڪنهن قوم کان ڀڄي آهي ته موٽي ناهي آئي“.۔ (ابنِ ماجہ،کتاب الاطعمۃ،باب النھی عن القاء الطعام، ۴/۵۰، حدیث:۳۳۵۳)

مٺا مٺا اسلامي ڀائرو! اڄڪلهه هر ڪو بي برڪتي ۽ تنگدستيءَ جي شڪايت ڪري رهيو آهي. ٿي سگهي ٿو جو مانيءَ جو احترام نه ڪرڻ جــي اها سزا هجي. اڄ شايد ئي ڪو مسلمان اهڙو هجي، جيڪو ماني ضايع نه ڪندو هجي، هر طرف کاڌي جي بي حرمتي جا دل ساڙيندڙ نظارا آهن، شادي جون تقريبات هجن يـا بـزرگانِ ديـن رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی جي نياز جا تبرڪات.

مُسلمانن ۾ اوهان جو فيضان جاري ۽ ساري آهي۔ داتا صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو اَصۡل وطن اَفۡغانِستان جو شهر غزني آهي، پر ماڻهن کي نيڪي جي دعوت ڏيڻ جي لاءِ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پنهنجو وطن ڇڏي ڪري هڪ اَنۡجان شهر ۾ اِقامَت اِخۡتِيار فرمائي ، ان مان اسان کي به هي درس ملي ٿو ته نيڪي جي دعوت ڏيڻ جي لاءِ هر مهيني ڪم اَزۡ ڪم ٽن (3) ڏينهن جي مَدَني قافلي ۾ ضرور سفر ڪرڻ گھرجي ۽ ان راھ ۾ اچڻ وارين مُشڪلات کي خوش دليءَ سان برداشت ڪندي خوب خوب سُنَّتن جي خِدۡمت جي لاءِ تيار رهڻ گھرجي۔ هر شُعۡبي سان مُنۡسلڪ ماڻهن تي اَحۡسن اَنۡداز ۾ اِنۡفرادي ڪوشش ڪرڻ گھرجي ۔

اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ! تبليغِ قرآن وسُنَّت جي عالمگير غير سياسي تحريڪ دعوتِ اسلامي جتي مُختلف شعبه جات ۾ خِدۡمتِ دين جو ڪم سراَنۡجام ڏئي رهي آهي، اُتي رانديگرن جي اِصۡلاح ۽ تَربِيت جي لاءِ به هڪ شعبو بنام ” مجلس اِصلاح براءِ رانديگر “ قائم ڪيو ويو آهي، جنهن جو بُنيادي مَقۡصد راندين سان مُنۡسلڪ ماڻهن ۾ دعوتِ اسلامي جي پيغام کي عام ڪرڻ ۽ انهن کي دعوتِ اسلامي سان وابسته ڪندي ان مَدَني مقصد ” مون کي پنهنجي ۽ سڄي دُنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻي آهي ۔ اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ “ جي مطابق زندگي گذارڻ جو مدني ذِهن ڏيڻ آهي۔ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ! ڪيترن ئي رانديگرن ۽ سندن گھر وارن کي مدني مقصد ” مون کي پنهنجي ۽ سڄي دُنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻي آهي “ جو ذهن ڏيڻ جي ڪوشش جاري آهي ۔

اے دعوتِ اسلامی تیری دھوم مچی ہو

مٺا مٺا اسلامي ڀائرو! سنتن جي خدمت جي لاءِ دعوتِ اسلامي جي تحت ذيلي حلقي جي 12 مَدَني ڪمن ۾ اڳتي وڌي ڪري حصو وٺو۔ ذيلي حلقي جا 12 مَدَني ڪم مُسَلمانن کي راھِ سُنَّت تي هلائڻ ۽ عاشِقانِ رسول ۾ قرآن ۽ سُنّت جو پيغام پهچائڻ ۾ تمام گھڻا مُعاوِن آهن۔ ان 12 مَدَني ڪمن مان هفتيوار هڪ مَدَني ڪم ” هفتيوار سُنّتن ڀري اجتماع ۾ شِرڪت ڪرڻ “ به آهي۔ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ! دعوتِ اسلامي جي تحت ٿيڻ واري هفتيوار سنّتن ڀري اجتماع جو آغاز بعد نمازِ مَغۡرِب سورهء مُلڪ جي تِلاوت سان ٿيندو آهي، فرمانِ مُصۡطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هي: ان ذات جي قسم! جنهن جي قَبضهء قُدرت ۾ منهنجي جان آهي! ڪتابُ الله جي هڪ آيت ٻڌڻ پهاڙ (جَبَلِ صَبِير) جي مثل صَدَقو ڪرڻ جي اجر کان وڌيڪ عظيم آهي۔ (جمع الجوامع، حرف الواو، ۸ / ۸۲، حدیث:۲۴۶۱۵) غور ڪريو! جڏهن هڪ آيت بُڌڻ جو هي فائدو آهي ته مڪمل سُورت کي ٻڌڻ ڪيتري قَدَر اجر ۽ ثَواب جو باعِث ٿيندو۔ تِلاوَت کان پوءِ نَعۡت شريف پڙھي ويندي آهي۔ نَعۡت پڙھڻ ۽ ٻڌڻ جي به ڇا ڳالھ ڪجي؟ سرڪار عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي نَعۡت پڙھڻ، حضرت حَسّان بن ثابت رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي سنّت ۽ نَعۡت ٻڌڻ سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سنّتِ ڪريمه آهي، ان کان پوءِ سُنَّتن ڀري بَيان جي ترڪيب هوندي آهي، جنهن ۾ علمِ دين جا قيمتي موتي چونڊڻ لاءِ ملندا آهن ۔علمِ دين حاصِل ڪرڻ جي فضيلت جي باري ۾ فرمانِ مُصۡطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: سڀ کان افضل صَدَقو هي آهي ته مُسَلمان علم سکي، پوءِ پنهنجي اسلامي ڀائرن کي سيکاري۔ (سنن ابن ماجہ ، کتا ب السنہ ، با ب ثواب معلم الناس با لخیر، رقم ۲۴۳، ج ۱، ص ۱۵۸) ، ذِڪر ۽ دُعا، صَلوٰة وسَلام، نمازِ عشاء، نماز کان پوءِ حلقه لڳندا آهن، جنهن ۾ مختلف موضوعن تي سُنّتون ۽ آداب ٻڌايا ويندا آهن ۽ ڪا هڪ دُعا ياد ڪرائي ويندي آهي، فڪرِ

مدينه جو مدني حلقو هوندو آهي، پوءِ آرام جو وقفو، خوش نصيب عاشقانِ رسول رات اعتڪاف جي سعادت حاصل ڪرڻ کان پوءِ نمازِ تهجد جي برڪتون لُوٽيندا آهن، اذانِ فجر کان پوءِ صداءِ مدينه، نمازِ فجر باجماعت، نماز کان پوءِ مدني حلقي ۾ شرڪت ۽ پوءِ مدني انعام تي عمل ڪندي اِشراق و چاشت جي سعادت حاصل ڪرڻ کان پوءِ صلوٰة و سلام تي اجتماع جو اِختِتام ٿي ويندو آهي۔ اجتماع جي اختتام تي ڪيترائي عاشقانِ رسول سُنّتن جي تربيت جي مدني قافلن ۾ سفر جي سعادت حاصل ڪندا آهن۔

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته؟ هفتيوار سُنَّتن ڀري اجتماع ۾ شِرڪت اسان جي لاءِ ڪيتري قَدَر باعِثِ اجر و ثَواب آهي ۔ تنهنڪري توهان کي مَدَني التجا آهي، سُستي کان پاسو ڪندي ۽ هفتيوار سنّتن ڀري اجتماع ۾ اَوّل تا آخِر شِرڪت کي پنهنجو معمول بڻايو! ۽ بيشمار اجر ۽ ثَواب جا حَق دار بڻجو۔ هفتيوار سنّتن ڀري اجتماع جي برڪت جو اندازو ان مَدَني بهار سان لڳايو!

مَرڪَزُالاَولِياءَ (لاهور) جي علائقي، ملتان روڊ جي هڪ اسلامي ڀاءُ جو ڪجهه اهڙي قسم جو بيان آهي ته مان ڇڙواڳ ۽ تيز طَبِيعَت جو هوندو هئس، پنهنجي مَستي ۾ مَست، دنيا جي محبت ۾ مَگَن، گناهن ۽ غَفلتُن ۾گم هئس ـ ٽِفن وڄائي ٻارن جا گيت ڳائڻ ۽ قَوالن جـون نَقلون ڪرڻ جــي معاملي ۾ سڄي خاندان ۾ مشهور هئس. شادي ۽ ٻين تقريبن ۾ مزاحِيه، ٽَوٽَڪا ۽ فِلمِي غَزلُون ۽ گانا ڳائڻ، بي ڊولي انداز ۾ نچڻ ، قِسم قِسم جا نَخرا ڪري ماڻهن کي کلائڻ منهنجو پسنديده شُغل هوـ اسڪول جو زمانو هو، هڪ عِمامي وارو اسلامي ڀاءُ وڏي ڀاءُ سان گهڻو ملڻ ايندو هو ـ هڪ ڏينهن ڀاءُ منهنجو تعارُف ڪرايو ته ان مونکي تبليغِ قرآن ۽ سنّت جي عالمگير غير سياسي تحريڪ دعوتِ اسلامي جي هفتيوار سُنّتن ڀريي اجتماع جي دعوت پيش ڪئي، مان انهن جي دعوت تي خميس جي ڏينهن سُنّتن ڀريي اِجتماع

۾ پهچي ويس ـ مونکي ڏاڍو سٺو لڳو، ائين مان پابندي سان وڃڻ شروع ڪيو، آهستي آهستي عِمامو شريف به سجائي ڇڏيو، جنهن تي گهر جي ڪِن ڀاتين سختي سان مُخالَفَت ڪئي، ايتري تائين جو اڪثر دفعا مَعَاذَاللّٰه عَزَّوَجَلَّ عِمامو شريف ڇِڪي لاٿو ويندوهو. درس ڏيڻ جي لاءِ منع ڪيو ويندو هو پر سجائڻ جي نيّت هئي انهن سڀني آزمائشن جي باوجود مدني ماحول جي ڪشش ۽ عاشقانِ رسول جو سهڻو سلوڪ ڏينهون ڏينهن مونکي دعوتِ اسلامي جي گهڻو ويجهو ڪندو ويو ـ مڪتبةُ المدينه کان جاري ٿيل سُنّتن ڀريل بيانن جون ڪيسٽون ٻڌڻ سان آٿت ملندي هئي ۽ حوصلو وڌندو ويوـ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ! آهستي آهستي گهر ۾ به مدنِي ماحول ٿي ويو اهي گهر وارا جيڪي سُنّتن ڀريي اجتماع ۽ مدني قافلي ۾ سفر جي اجازت نه ڏيندا هئا، انهن مونکي يَڪمُشت ٻَارهن مَهِينن جي سفر جي اجازت ڏئي ڇڏي گهر ۾ اِسلامي ڀينرن جـو اِجـتماع شـروع ٿي ويو ـ والـد صـاحـب ڏاڙهـي سجائي ورتي. سيّد صاحب يا ماءُ پيءُ اچن ته تعظيم لاءِ اُٿي بيهڻ ثواب آهي۔ حڪيم الامت مفتي احمد يار خان نعيمي عَلَیهِ رَحْمَةُ اﷲِ الغني لکن ٿا: بزرگن جي اچڻ تي اهي ٻئي ڪم يعني تعظيمي قيام ۽ استقبال جائز بلڪ صحابه جي سنت آهن، بلڪ حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي قولي سنّت آهي۔

(مراٰۃالمناجیح ،ج۶، ص۳۷۰، جملہ منقول از سنتیں اور آداب،ص۹۹، بتغیر)

هزارين سنّتون سکڻ جي لاءِ مڪتبة المدينه جا شايع ٿيل ٻه ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهو۔ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي. (101مَدَنی پھول)

عاشقانِ رسول، آئیں سنت کے پھول

دینے لینے چلیں، قافلے میں چلو
