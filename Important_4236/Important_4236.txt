نَوَيۡتُ سُنَّتَ الۡاِعۡتِكَافِ (مون سنّتِ اعتڪاف جي نيّت ڪئي)

مٺا مٺا اسلامي ڀائرو! جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي اعتڪاف جي نيت ڪري وٺو، ڇوته جيستائين مسجد ۾ رهندؤ اعتڪاف جو ثواب ملندو رهندو۔ ياد رکو! مسجد ۾ کائڻ، پيئڻ، سمهڻ يا سحري، افطاري ڪرڻ، ايستائين جو آبِ زم زم يا دَم ڪيل پاڻي پيئڻ جي به شرعاً اجازت ناهي ، البت جيڪڏهن اعتڪاف جي نيت هوندي ته اهي سڀ شيون ضمناً جائز ٿي وينديون۔ اعتڪاف جي نيت به صرف کائڻ، پيئڻ يا سمهڻ جي لاءِ نه هئڻ گھرجي بلڪ ان جو مقصد الله تعاليٰ جي رِضا هجي۔ ”فتاويٰ شامي“ ۾ آهي: جيڪڏهن ڪو مسجد ۾ کائڻ پيئڻ يا سمهڻ چاهي ته اعتڪاف جي نيت ڪري ۽ ڪجھ دير ذِڪرُاللہ ڪري پوءِ جيڪو چاهي (يعني هاڻي چاهي ته کائي پي يا سمهي سگھي ٿو)

دُرُود شريف جي فضيلت

حضور ا ڪ رم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عظمت نشان آهي: جِبريل (عَلَيه السَّلام) مون کي عرض ڪيو ته رَبّ ڪريم فرمائي ٿو: اي محمد! (صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) ڇا توهان هن ڳالھ تي راضي ناهيو ته توهان جو اُمَّتي توهان تي هڪ ڀيرو دُرُود موڪلي، مان ان تي ڏھ رَحمتون نازل ڪيان ۽ اوهان جي اُمَّت مان جيڪو هڪ سلام موڪلي، مان ان تي ڏھ سلام موڪليان۔ (مش ڪ اۃ، ڪ تاب الصلاۃ ،باب الصلاۃ علي النبي وفضلها،۱ / ۱۸۹،حديث: ۹۲۸ملتقطا) حڪيمُ الاُمَّت حضرت مفتي احمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: رَبّ (ڪريم) جي سلام موڪلڻ مان مُراد يا ته مَلائڪن جي ذريعي ان کي سلام موڪلڻ آهي يا آفتن ۽ مُصيبتن کان سلامت رکڻ۔ (مرآۃ المناجيح، ۲/۱۰۲)

دُنیاوآخِرت میں جب میں رہوں سلامت پیارے پڑھوں نہ کیوں کر تم پر سلام ہر دم

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مختصر وضاحت: اي اسان جا مهربان آقا! اسان گناهه ڪرڻ ۾ ڪا ڪمي نه ڪئي پر پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ عطا فرمائڻ ۾ ڪا ڪمي نه فرمائي. اوهان تي ڪروڙ ڀيرا درود نازل ٿين.

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿا. فرمانِ مصطفي ٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِه ٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم ال ڪ بير ،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس ٭ ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس ٭ ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله! تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي جي لاءِ بلند آواز سان جواب ڏيندس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ اڄ جي بيان ۾ اسان ”آقا ڪريم صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي اُمّت سان محبت“ جي باري ۾ ٻڌنداسين ته بروزِ قيامت پاڻ ڪريم

صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ فڪرِ اُمّت ۾ ڪيتري قدر غمگين هوندا، گنهگار اُمّتين جي آخرت سنواريندا، پاڻ سڳورن صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي صدقي ۾ اُمّتِ مُصطَفٰي تي دنيا ۾ جيڪي عنايتون ٿيون ۽ جيڪي آخرت ۾ ٿينديون انهن جون ڪجھ جھلڪيون به بيان ٿينديون، رسول اڪرم صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ دنيا ۾ پنهنجي اُمّت کي ڪيترو ياد فرمايائون ۽ اُمّت جي ياد ۾ پاڻ صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ڪيتري قدر گريه و زاري ڪندا رهيا اهو به ٻڌنداسين، بيان جي آخر ۾ اِنْ شَآءَ الله عَزَّ وَجَلَّ مرحبا يا مصطفٰي جون ڌومون مچائيندي، سرڪار جي ولادت جي جشن جون خوشيون ٻڌائيندو پيارو پيارو مڪتوبِ عطار به ٻڌڻ جي سعادت حاصل ڪنداسين” الله ڪري جو اسان سڄو بيان توجه سان ٻڌڻ جي سعادت حاصل ڪري سگھون۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

ڏ سو هو آيو اسان جو حامي ۽ مدگار

حضرت سيدُنا عبدُالله بن عُمَر رَضِىَ اللّٰهُ عَنْهما کان روايت آهي ته اُمت تي مهربان، محبوبِ رحمٰن صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن ارشاد فرمايو: قِيامت جي ڏينهن حضرت سَيدُنا آدم عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام عرش جي ويجھو وڏي ميدان ۾ تشريف فرما هوندا، سندس کي ٻه ساوا ڪپڙا پاتل هوندا، پنهنجي اولاد مان هر ان شخص کي ڏسي رهيا هوندا جيڪو جنَّت ۾ وڃي رهيو هوندو ۽ پنهنجي اَولاد مان ان کي به ڏسي رهيا هوندا جيڪو دَوزَخ ۾ وڃي رهيو هوندو۔ انهيء دوران آدم عَلَيۡهِ السَّلَام منهنجي هڪ اُمَّتِي کي دَوزَخ ۾ ويندي ڏسندا۔ سَيدُنا آدم عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام سڏيندا، يا احمد(صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ)! يا احمد (صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ)! مان چوندس: لَبَّيڪ اي ابُوالبَشَر! حضرت سَيدُنا آدم عَلَيۡهِ السَّلَام چوندا: اوهان جو هي اُمَّتِي دَوزَخ ۾ وڃي رهيو آهي۔ اهو ٻڌي مان وڏي چُستي سان تيز تيز (قدمن سان) ملائڪن جي پويان ويندس ۽ چوندس: اي منهنجي رَبّ جا ملائڪو! ترسو۔ اهي عرض ڪندا: اسان مُقَرَّر ڪيل ملائڪ آهيون، جنهن ڪم جو اسان کي الله پاڪ حڪم ڏنو آهي ان جي نافرماني ناهيون ڪندا، اسان اهو ئي ڪندا آهيون جنهن جو اسان کي حڪم

مليو آهي۔ جڏهن مان غمگين ٿي ويندس ۽ پنهنجي ڏاڙھي مُبارڪ کي کاٻي هٿ سان پڪڙي عرش ڏانهن هٿ جو اشارو ڪندي عرض ڪندس: اي منهنجا پَروَردگار! ڇا تو مون سان واعدو ناهي ڪيو ته مون کي منهنجي اُمَّت جي باري ۾ رُسوا نه ڪندين۔ عرش کان سڏ ٿيندو: اي ملائڪو! محمد(صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي اِطَاعت ڪريو ۽ ان کي موٽائي ڇڏيو۔ پوءِ مان پنهنجي جھولي مان سفيد ڪاغذ ڪڍندس ۽ اُن کي مِيزان جي ساڄي پَلڙي ۾ وجھي چوندس، ’’ بِسمِ ﷲِِِ ‘‘ته اهو نيڪين وارو پلڙو برائين واري پَلڙي کان وزني ٿي ويندو۔ آواز ايندو: خوش بَخت آهي، سعادت وارو ٿي ويو آهي ۽ ان جو مِيزان وزني ٿي ويو آهي۔ ان کي جَنَّت ۾ وٺي وڃو۔ اهو ٻانهو چوندو: اي منهنجي پاڪ پَروَردگار جا ملائڪو! ترسو، مان هن ٻانهي سان ڳالھ ته ڪيان جيڪو پنهنجي رَبّ جي بارگاھ ۾ وڏي عزت رکي ٿو۔ پوءِ هو چوندو: منهنجا ماءُ پيءُ اوهان تي فِدا ٿين، اوهان جو مبارڪ چهرو ڪيترو حسين آهي ۽ اوهان جي صورت ڪيتري خُوبصُورت آهي، اوهان منهنجي ڏوهن کي مُعاف فرمايو ۽ منهنجي لڙڪن تي رَحم ڪيو، (اوهان ڪير آهيو؟)۔ ته مان ان کي چوندس: مان تنهنجو نبي محمد (صَلَّى اللهُ عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) آهيان ۽ هي تنهنجو اهو دُرُود آهي جيڪو تُون مون تي موڪليندو هئين، ان توکي پورو نفعو پهچايو جيتري توکي ان جي ضرورت هئي۔ (موسوعۃ ابن ابي الدنيا في حسن الظن باللّٰه، ۱ / ۹۱،حديث:۷۹)

مٺا مٺا اسلامي ڀائرو! اسان رسول انور، شافعِ محشر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پنهنجي امت سان مَحَبَّت جا واقعات ٻڌي رهيا آهيون، اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ اسان کي فخر آهي ته اسان به رحمت عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا اُمّتي آهيون، هاڻي اسان پنهنجي باري ۾ غور ڪريون ته ڇا اسان کي به پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان اهڙي ئي محبت آهي جهڙي هڪ اُمّتي کي پنهنجي آقا سان هئڻ گھرجي، مثال طور آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته نمازن جو پابند هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو امتي ته پنهنجي ضرورت جا مسئلا ۽ ٻيا فرض ۽ واجب علوم ڄاڻڻ وارو هجڻ گهرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته تقويٰ ۽ پرهيزگاري جو پيڪر هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو امتي ته پنهنجون شرعي ذميداريون پوريون ڪرڻ وارو هجڻ گهرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته تلاوتِ قرآن جو حريص هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته ناجائز ۽ حرام ڪمن کان بچڻ وارو بلڪ پنهنجي اولاد وغيره کي انهن ڪمن کان بچائڻ وارو، انهن جي صحيح اسلامي تربيت ڪرڻ وارو هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته خوفِ خدا وارو هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته پنهنجي ۽ سڄي دنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻ وارو هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته بهترين خوبين سان آراسته هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته بي حيائي جي ڪمن کان بچڻ وارو هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته باطني برائين کان پري هئڻ

گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته مسلمانن جو خيرخواه هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي رحمتِ ڪونين، ناناءِ حَسَنَين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اطاعت ۽ فرمانبرداري ڪرڻ وارو هئڻ گھرجي، آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو اُمّتي ته سُنّتن جي پيروي ڪرڻ وارو هئڻ گھرجي۔ الله ڪريم اسان سڀني کي آقا ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سنتن تي عمل ڪرڻ جي توفيق نصيب فرمائي.

جشنِ وِلادت جي باري ۾ مڪتوبِِ عطَّار

مٺا مٺا اسلامي ڀائرو! صَفَرُالْمُظَفَّر جو مُبَارَڪ مهينو پنهنجي اِختِتَام ڏانهن وڃي رهيو آهي۔ پوءِ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ رَبِيعُ الاَوَّل جو نُور بار مهينو پنهنجي آب و تاب سان جَلوَه گَر ٿيندو۔ شَيخِ طريقت، اَميرِاَهلسُنَّت، باني دعوتِ اسلامي حضرت علامه مولانا ابُو بلال محمد الياس عطّار قادِرِي رَضَوِي ضِيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جَشنِ مِيلَاد ملهائڻ واري عاشقانِ مِيلاد کي پنهنجي مڪتوب (Letter) ۾ ڪجھ اَهمّ مَدَني گل عطا ڪيا آهن۔ اچو! اسان به اهو ”م ڪ توبِ عطّار“ تَوَجُّه سان ٻڌڻ جي سَعادت حاصل ڪريون ٿا:

مڪتوبِ عطّار

بِسْم اللّٰه الرَّحْمٰنِِ الرَّحِيم ط سگِ مدينه محمدالياس عطّارقادِرِي رَضَوِي عُفِي عَنْه جي طرفان تمام عاشقان رسول اسلامي ڀائرن / اسلامي ڀينرن جي خدمت ۾ جشن ولادت جي خوشي ۾ لهرائيندي ساون سـاون جهـنڊن ۽ جڳمڳ ڪندڙ بلبن ۽ ننڍڙين ننڍڙين بتين کي چمندو جهومنـدو ماکي کان به مٺو مڪي مدني سلام،

اَلسَّلامُ عَلَيکمْ وَرَحْمَۃُ اﷲِ وَبَرَکاتُه اَلْحَمْدُ لِلّٰه رَبِّ الْعٰلَمِينَ عَلٰي کلِ حَال

تم بھی کرکے اُن کا چرچا اپنے دِل چمکاؤ

اُونچے میں اُونچا نبی کا جھنڈا گھر گھر میں لہراؤ

ربیعُ النُّور اُمّیدوں کی دُنْیا ساتھ لے آیا دُعاؤں کی قَبولیَّت کو ہاتھوں ہاتھ لے آیا

مرد لاءِ ڏاڙهي ڪوڙائڻ يا هڪ مٺ کان گهٽائڻ ٻئي حرام آهن، اسلامي ڀيڻ جو بي پردگي ڪرڻ حرام آهي. براءِ ڪرم ربيع النور شريف جي برڪت سان اسلامي ڀائر هميشه جي لاءِ هڪ مٺ ڏاڙهي ۽ اسلامي ڀينرون مستقل شرعي پڙدو ۽ زهي قسمت جو مدني برقعو پائڻ جي نيت ڪن .

جُھک گیا کعبہ سبھی بُت مُنہ کے بَل اَوندھے گِرے دَبدبہ آمد کا تھا ، اَھْلاً وَّسَھْلاً مرحبا

سنتن ۽ نيڪين تي استقامت حاصل ڪرڻ جو عظيم مدني نسخو هي آهي ته تمام عاشقان رسول اسلامي ڀائر ۽ اسلامي ڀينرون روزانو فڪر مدينه ڪندي مدني انعامات جو رسالو ڀري هر مهيني جمع ڪرائڻ جي نيت ڪن، جيڪڏهن نيت هجي ته هٿ کڻي چئو: اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ (ياد رکو جيڪڏهن دل ۾ نيت نه هئڻ جي باوجود ڄاڻي واڻي ڏيکاءُ جي لاءِ هٿ کڻڻ ته مان به نيت ڪيان ٿو گناھ آهي۔)

باب ماجاء في ثواب من عزي مصابا، رقم ۱۶۰۱ ،۲/ ۲۶۸) ٭ تعزيت جي معنيٰ آهي: مصيبت زده کي صبر جي تلقين ڪرڻ۔ ٭ تعزيت مسنون (يعني سنّت) آهي۔ (بهار ِشريعت ،۱/۸۵۲) ٭ دفن کان پهرين به تعزيت جائز آهي، پر افضل هي آهي ته دفن کان بعد ٿئي اهو ان وقت آهي جو ميت جا وارث (عزيز) ميت تي روڄ راڙو نه ڪندا هجن، نه ته انهن جي تسلي جي لاءِ دفن کان پهريان ئي ڪري۔ (الجوهرۃ النيرۃ، ڪ تاب الصلاۃ، باب الجنائز، ص۱۴۱) ٭ مستحب هي آهي ته ميت جي سڀني مائٽن سان تعزيت ڪري، ننڍي وڏي مرد و عورت سڀني سان پر عورت سان ان جا محارم ئي تعزيت ڪن۔ تعزيت ۾ اهو چوي، الله پاڪ ميت جي بخشش فرمائي ۽ ان کي پنهنجي رحمت ۾ جڳھ عطا فرمائي۔ (بهار شريعت،۱ / ۸۵۲) ٭ ميت جي عزيزن جو گھر ۾ ويهڻ ته ماڻهو ان جي تعزيت لاءِ اچن هن ۾ حرج ناهي ۽ گھر جي دروازي تي يا شارعِ عام (ما ڻ هن ج ي گذرگاھ) تي دري وڇائي ويهڻ بُري ڳالھ آهي۔ (الدرالمختار، ڪ تاب الصلاۃ، مطلب في ڪ راھۃ الضيافۃ من اھل الميت،۳ /) ٭ ۽ دفن کان بعد ميت جي گھراچڻ ۽ تعزيت ڪري پنهنجي پنهنجي گھر وڃڻ جيڪڏهن اتفاقاً هجي ته حرج ناهي ۽ ان جي رسم ڪرڻ نه گھرجي ۽ ميت جي گھر تعزيت جي لاءِ ماڻهن جو رش ڪرڻ دفن کان پهريان هجي يا بعد انهيءَ وقت هجي يا ڪنهن ٻي وقت خلاف اَوليٰ آهي ۽ ڪن ته تو گناھ به ناهي۔ (بهار شريعت،۱ / ۸۵۳) ٭ جيڪو هڪ ڀيرو تعزيت ڪري ان کي ٻيهر تعزيت جي لاءِ وڃڻ مڪروه آهي۔ (الدر المختار، ڪ تاب الصلاۃ، باب صلاۃ الجنازۃ، ج۳، ص۱۷۷)

(1) جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

(2) سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

(3) رَحمت جا ستّر دروازا

(5) قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَي ۡ هِ وَاٰلِهٖ وَسَلَّمَ
