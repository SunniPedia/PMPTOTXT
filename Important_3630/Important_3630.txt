صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ٻڌو اوهان ته رَبُّ العٰلَمِين عَزَّ وَجَلَّ کي پنهنجي مَدَني حبيب صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ ۽ سندن نالي مبارڪ جو ادب ڪيتري قَدَر عزيز آهي جو ان پنهنجي پاڪيزه ڪلام ۾ پنهنجن ٻانهن کي رسول اڪرم صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ ۽ سندن مبارڪ نالي جي ادب ڪرڻ جو حڪم اِرشاد فرمايو۔ تنهنڪري اسان کي گھرجي ته اسان به هن فرمانِ خُداوَندي ۽ مُفَسِّرِنِ سڳورن رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيهمْ اَجْمَعِين جي اِرشادن تي عمل ڪندي مدني مصطفيٰ صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ کي پُڪارڻ وقت ادب ۽ تَعظِيم جو لحاظ ڪندي هميشه صِفاتي نالن مثال طور يارَسُولَ الله ، ياحَبِيبَ الله وغيره اِستِعمال ڪرڻ جي عادت بڻايون۔ ائين ته پاڻ صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جا سمورا نالا ئي يقيناً لائقِ ادب ۽ تَعظِيم ۽ خُدا عَزَّ وَجَلَّ جا مَحبوب آهن پر انهن مان نالي ”اَحْمَد“ ۽ ناليِ ” مُحَمَّد“ جو شان سڀ کان جُدا (Unique) آهي ڇوته هي ٻه اهڙا مُبَارَڪ نالا آهن جيڪي الله عَزَّ وَجَلَّ کي تمام گھڻا مَحبوب آهن، جيئن ته

الله عَزَّ وَجَلَّ جا مَحبوب نالا

صَدرُ الشَّرِيعه، بَدرُ الطَّريقه حضرت علّامه مَولانا مُفتي محمد امجد علي اَعظَمِي رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيه فرمائن ٿا: حُضُورِ اَڪرم صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جا نالا مبارڪ ”محمد“ ۽ ”احمد“ آهن ۽ ظاهر اهوئي آهي ته اهي ٻئي نالا خود الله تعاليٰ پنهنجي محبوب صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جي لاءِ منتخب فرمايا، جيڪڏهن اهي ٻئي نالا خدا (عَزَّوَجَلّ) وٽ وڌيڪ پيارا نه هجن ها ته پنهنجي محبوب صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جي لاءِ پسند نه فرمائي ها ۔ (بهار شريعت ،حِصّه۱۶، ۳/ ۶۰۱) وري انهن ٻنهي مان ” مُحَمَّد “

جي نالي کي جيڪا اَهمِيت ۽ فضيلت حاصل آهي اهو هر عقلمند شخص آساني سان سمجھي سگھي ٿو۔ پر اَفسوس ته جيئن جيئن اسان زَمانَه نَبَوِي کان پري ٿيندا پيا وڃون ائين اسان جي معاشري ۾ جَهالَت جي اونداهي وڌندي پئي وڃي۔ جنهن جو هڪ چٽو مِثال هي آهي ته بعض نادان پنهنجي اَولاد جي لاءِ اهڙن نالن جي ڳولها ۾ رهن ٿا جيڪو گھر، خاندان، پاڙي ۾ پري پري تائين ڪنهن جو به نه هجي، جيڪو به ٻڌي واھ واھ ڪري، واھ! ڪيترو زَبَردَست نالو رکيو آهي، اَڙي هي نالو ته اسان پهرين دفعي ٻڌو آهي وغيره وغيره۔اهڙي طرح بعض نادان بي معنيٰ نالا يا مَعَاذَ الله عَزَّوَجَلَّ غير مُسلِمن جي نالن تي پنهنجي اَولاد جا نالا رکندا آهن، اَلْاَمَان وَالْحَفِيظ ۔ يقيناً هي سڀ عِلمِ دِين کان دُوري ۽ نالي رکڻ جي اَهمِيت کان ناواقِفي جو نتيجو آهي۔ ياد رکو! نالي جو تعلّق صرف دنيوي زندگي تائين ناهي بلڪ جڏهن ميدان محشر قائم ٿيندو ته انسان کي انهيءَ نالي سان سڏيو ويندو جنهن نالي سان ان کي دنيا ۾ سڏيو ويندو هو، جيئن ته

حضرت سيِّدُنا ابو درداء رَضِىَ اللهُ تَعَالٰى عَـنْهُ کان روايت آهي، حضورِ پاڪ، صاحب لولاڪ، سيّاح افلاڪ صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جن فرمايو؛ قيامت جي ڏينهن توهان پنهنجي ۽ پنهنجي ابن ڏاڏن جي نالن سان پڪاريا ويندؤ، تنهنڪري توهان پنهنجا سٺا نالا رکو۔ ۔ (ابو داود،ڪتاب الادب،باب في تغيير الاسماء،۴/۳۷۴، حديث: ۴۹۴۸)

مشهور مُفَسِّرِ قُرآن،حڪيمُ الاُمَّت مُفتي اَحمد يار خان نعيمي رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيه تفسيرِ نعيمي ۾ فرمائن ٿا ته لفظِ ”مُحَمَّد“ جي معنيٰ آهي؛ هر طرح تعريف ڪيل، هروقت، هرزماني، هر زبان ۾ حمد ۽ ثناء ڪيل۔ حقيقت هي آهي ته جيئن حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ سموري مخلوق ۾ افضل، سڀني رسولن جا سردار آهن اهڙي طرح حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جو اسم شريف به سڀني نبين جي نالن جو سردار آهي۔ هن پاڪ نالي جا بيشمار فضائل آهن جن مان ڪجھ هي آهن؛

(1) سڀني جا نالا ان جا ماءُ پيءُ رکندا آهن لقب قوم ڏيندي آهي خطاب حڪومت کان ملندو آهي، پر حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جن جو نالو، لقب ۽ خطاب سڀ الله عَزَّوَجَلّ جي طرفان آهن جو عبدالمطلب (رَضِىَ اللهُ تَعَالٰى عَـنْهُ) ملائڪ جي بشارت سان اهو نالو رکيو۔ (2) ٻين جا نالا پيدائش جي ستين ڏينهن رکيا ويندا آهن پر حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جن جو پاڪ نالو جهان جي پيدائش کان پهريان عرش اعظم تي لکيو ويو هو ۽ حضرت عيسيٰ عَـلَيْـهِ السَّلَام حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جي ولادت باسعادت کان تقريبًا 600 سال پهريان پنهنجي قوم کي فرمايو؛ اِسۡمُهٗ اَحۡمَد انهن جو نالو پاڪ احمد آهي، گذريل قومون سندن پاڪ نالي جي برڪت سان دعا گھرنديون هيون۔ (3) حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جو نالو ”محمد“ وڏو جامع آهي، جنهن ۾ حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جا بيشمار فضائل بيان ٿي ويا آهن، آدم جي معنيٰ آهي مٽي مان پيدا ٿيڻ وارا، ابراهيم جي معنيٰ آهي ”مهربان پيءُ، نُوح جي معنيٰ آهي ”خوف خدا وچان گريه و زاري ۽ نوحو ڪرڻ وارا“ انهن سڀني نالن ۾ هڪ

هڪ وصف جي طرف اشارو آهي پر ”محمد“ جي معنيٰ آهي هر طرح سان هر وصف ۾ بيحد تعريف ڪيل، انهيءَ ۾ حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جي لاتعداد خوبين ۽ ڪمالن ڏانهن اشارو ٿي ويو۔ (5) لفظ ”محمد“ ۾ غيبي خبر به آهي جو هميشه دنيا ۽ آخرت ۾ ان جي هر جڳھ هر طرح حمد و ثناء پئي ٿيندي رهندي۔ انهيءَ خبر جي صداقت اسان پنهنجي اکين سان ڏسي رهيا آهيون جو اڄ به حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جي برابر ڪنهن جي به تعريف نٿي ٿئي، بلڪ جيڪي حضور انور صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جن سان وابسته ٿي ويا انهن جي به تعريف ٿي وئي، فرش تي ان جون ڌومون ته عرش تي ان جي واکاڻ. صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اسان نالي ”مُحَمَّد“ جي بَرَڪتن جي باري ۾ ٻڌي رهيا آهيون ۽ هي به ٻڌو ته اسان جا بُزرگ سڳورا رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيهمْ اَجْمَعِين هن نالي پاڪ جو تمام گھڻو ادب ڪندا هئا، ٿيڻ ته ايئن گھرجي ته اسان به بُزرگن جي طريقي تي عمل ڪندي هن پاڪ نالي جي عِزَّت ۽ حُرمَت جو خيال ڪريون ۽ دل ۽ جان سان ان جو خوب خوب ادب ڪريون پر اَفسوس صَد ڪروڙ اَفسوس ته هاڻي هن پاڪ نالي جو احترام به ختم ڪيو پيو وڃي، مَثال طور بعض واهيات ۽ بي باڪ نادان زبان جا ايتري قَدَر تيز هوندا آهن جو هو عام ماڻهن سان گڏ ”مُحَمَّد“ نالي واري مسلمانن کي به مُعاف نه ڪندا آهن ۽ اُنهن کي به مَعَاذَ الله عَزَّ وَجَلَّ برين گارين ۽ بُرن لقبن سان سڏيندا آهن، ائين ئي ڪجهه نادان شَرعي اَحڪام کان ناواقِفي جي ڪري هر شخص جي نالي کان اڳ ۾ ”مُحَمَّد“ لکندا يا پڙھندا آهن جڏهن ته ڪڏهن معنيٰ جي اِعتِبار سان ڪجهه نالن کان پهريان ”مُحَمَّد“ لکڻ پڙھڻ درست ناهي هوندو ته پوءِ نالي ”مُحَمَّد“ جي بَرَڪت ته ڇا ملندي رهندو بي اَدَبي جو وَبال هٿي تي هاڻي هڪ طرف ته انهن جھڙن نادانن جو طَريقڪار آهي ته ٻي پاسي انهن باادب هستين جو لائقِ تقليد ڪردار اسان جي سامهون آهي جيڪي نامُوسِ مُصطَفٰي سان گڏوگڏ نالي ”مُحَمَّد“ جي عِزَّت ۽ حُرمَت جا سچا پاسبان ۽ نالي ”مُحَمَّد“ جي ادب جي مُعَامَلي ۾ زَبَردَست مَدَني ذِهن رکڻ وارا هئا۔ اچو! نالي ”مُحَمَّد“ جو ادب ڪرڻ بابت هڪ سبق حاصل ڪرڻ وارو واقعو ٻڌون ٿا،

مفتي اعظم اصلاح ڪئي

سراجُ العارفين حضرت علامه مولانا غلامِ آسي رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيه فرمائن ٿا؛ مون پنهنجي نالي جي شروع ۾ برڪت جي لاءِ لفظ ”محمد“ شامل ڪري ڇڏيو هو۔ ان تي شهزاده ءِ اعليٰ حضرت مفتي اعظم مولانا مصطفيٰ رضا خان رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيه تنبيھ فرمائي جو اتي اسمِ رسالت (يعني محمد) نه هجڻ گھرجي۔ مون فورًا عرض ڪيو ته حضور پوءِ ”محمد عبدُالحيّ“ جو ڇا حڪم آهي؟ ان جي جواب ۾ حضرت فرمايو؛ ڪٿي عبدالحي ۽ ڪٿي غلامِ آسي؟ علامه (غلامِ آسي رحمة الله) فرمائن ٿا؛ ”اهو جواب ٻڌي مان حيرت ۾ پئجي ويس ۽ حضرت جي تَفَقُّه في الدّين (دين جي سمجهه) جي عظمت دل ۾ چڱي طرح ويهي وئي۔“

حضور مفتئ اعظم رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيه پنهنجي ارشاد سان اها رهنمائي فرمائي آهي جو جنهن نالي جي شروع ۾ لفظ ”محمد“ آندو وڃي، جيڪڏهن انهي نالي جو اطلاق لفظ ”محمد“ تي درست هجي ته اتي لفظ ”محمد“ شروع ۾ آڻڻ درست ٿيندو (جيئن محمد صادق) ۽ جيڪڏهن انهي نالي جو اطلاق لفظ ”محمد“ تي درست نه هجي ته اتي لفظ ”محمد“ شروع ۾ آڻڻ درست نه ٿيندو (جيئن محمد غلام حسين) ۔ حضور رحمت عالم صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ عبدُالحيّ آهن تنهن ڪري محمد عبدُالحيّ چوڻ درست آهي پر غلامِ آسي نه آهن ان ڪري ”محمد غلامِ آسي (يعني ”آسي“ جو غلام)“ چوڻ نامناسب آهي۔ (جهانِ مفتي اعظم ،ص ۴۵۱،بتغير قليل)

ان جي عِلاوه به ڪجهه اهڙا نالا آهن جو جن کان اڳ ۾ نالي ” مُحَمَّد “ جو اِطلاق دُرُست ناهي هوندو مثال طور عَبدُالمُصطَفٰي، غُلامِ غَوث، غُلامِ جِيلاني، عُبَيدِ رَضا وغيره پر ڪڏهن ڪڏهن هنن نالن کان به اڳ ۾ ” مُحَمَّد “ لکيو ويندو آهي البت اها ڳالھ به ذِهن نشين رکو ته نالي ” مُحَمَّد “ جو اِطلاق ڪهڙي نالي تي دُرُست آهي ۽ ڪهڙي نالي تي نه ان جو تَعَين ڪرڻ جي هر ڪنهن کي اِجازت ناهي! تنهنڪري ان جي لاءِ ڪنهن مُستَنَد سُنّي عالِمِ دِين يا

مٺا مٺا اسلامي ڀائرو! اوهان ٻڌو ته اسان جا بُزرگانِ دِين رَحْمَةُ اللّٰهِ تَعَالٰى عَلَيهمْ اَجْمَعِين ڪيتري قَدَر غَيرَت مَند هئا جو جڏهن اهي حضرات نالي ”مُحَمَّد“ جي بي اَدَبي ٿيندي ڏسندا هئا ته بي قَرار ٿي ويندا هئا ۽ ان جي ايماني غَيرت کي جوش اچي ويندو هو تنهنڪري اهي حضرات هٿو هٿ بي اَدَبي ڪرڻ واري جي اِصلاح (Reformation) ڪندا هئا۔ اي ڪاش ته اسان کي به انهن بُزرگن جو صَدَقو نصيب ٿئي ۽ اسان به انهن جي تعليمات تي عمل ڪندي نالي ”مُحَمَّد“ جو ادب ڪرڻ وارا ۽ ان پاڪيزه مُبَارَڪ نالي جي بي اَدَبي ٿيندي ڏسي هٿو هٿ بي اَدَبي ڪرڻ واري جي اِصلاح ڪرڻ وارا بڻجي وڃون۔ اَلْحَمْدُ لِلّٰه عَزَّ وَجَلَّ اڄ جي هن نازڪ دور ۾ دعوتِ اسلامي مسلمانن جي اندر بُزرگن جي ان مَدَني سوچ کي پيدا ڪرڻ لاءِ،اُنهن کي نالي ”مُحَمَّد“ جو ادب ڪرڻ ۽ نالي ”مُحَمَّد“ جي بي اَدَبي ڪرڻ وارن جي اِصلاح ڪرڻ جو ذِهن ڏيڻ جي لاءِ هر دم ڪوشش ڪري پئي تنهنڪري اوهان به مَدَني ماحول سان وابسته رهو ۽ ادب ۽ تَعْظِيم جو پيڪر بڻجڻ جي لاءِ ذيلي حلقي جي 12 مَدَني ڪمن ۾ وڌ کان وڌ حِصّو وٺو۔

ذيلي حلقي جي 12 مدني ڪمن مان هفتيوار هڪ مدني ڪم ”يومِ تَعطِيل اِعتِـڪاف“ به آهي۔ موڪل واري ڏينهن شهر جي ڪمزور ۽ ويجهي علائقن ڳوٺن ۾ وڃي اتي جي مسجدن کي آباد ڪرڻ سان گڏوگڏ مقامي عاشقان رسول کي نيڪي جي دعوت ڏئي علم دين سکڻ سيکارڻ جي ترڪيب ڪئي ويندي آهي ٭ اَلْحَمْدُ لِلّٰه عَزَّ وَجَلَّ يوْمِ تَعْطِيل اِعْتِـڪاف اسلامي ڀائرن کي سُنّتون،۽ آداب ۽ مَدَني دَرس وغيره جو طريقو سيکارڻ جو بهترين ذريعو آهي ٭ يومِ تَعطِيل اِعتِـڪاف جي بَرَڪت سان مسجِدون آباد ٿين ٿيون ٭ يومِ تَعطِيل اِعْتِـڪاف جي بَرَڪت سان مسجد ۾ گذرندڙ هر هر لَمحو عِبادَت ۾ شُمار ٿيندو ٭ يومِ تَعطِيل اِعتِـڪاف جي بَرَڪت سان مسجِد سان مَحَبَّت ڪرڻ ۽ ان ۾ پنهنجو گھڻي کان گھڻو وَقت گُذارڻ جي فضيلت حاصل ٿيندي۔ اَلْحَمْدُ لِلّٰه عَزَّ وَجَلَّ پنهنجو گھڻو وَقت مسجد ۾ گذارڻ جي به ڇا خوب فضيلت آهي

حضرت سَيدُنا ابُو سَعِيد خُدرِي رَضِىَ اللهُ تَعَالٰي عَـنْه ڪان روايت آهي ته تاجدارِ رسالت، شهنشاھ نُـبُوّت صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ جن فرمايو: جڏهن توهان ڪنهن شخص ڪي ڏسو ته هو مسجد ۾ ڪثرت سان اچ وڃ رکڻ وارو آهي ته ان جي اِيمان جي گواهي ڏيو ڇوته الله عَزَّ وَجَلَّ ارشاد فرمائي ٿو:

اِنَّمَا یَعْمُرُ مَسٰجِدَ اللّٰهِ مَنْ اٰمَنَ بِاللّٰهِ وَ الْیَوْمِ الْاٰخِرِ وَ اَقَامَ الصَّلٰوةَ وَ اٰتَى الزَّكٰوةَ (پ ۱۰ ، التوبه:۱۸)

بابُ المدينه (ڪراچي) جي هڪ اسلامي ڀاءُ جي اڳوڻي زِندگي گُناهن ۾ گذري رهي هئي، لغڙن اڏائڻ جو به شوقين هو، ويڊيو گيمز ۽ بلورراند کيڏڻ وغيره ان جي عادت ۾ شامل هو۔ هر هڪ جي مُعامَلي ۾ ٽنگ اَڙائڻ، ماڻهن سان جھيڙو ڪرڻ، ڳالھ ڳالھ تي ڦڏو ڪرڻ جهڙين بُرائين ۾ گِرِفتار هو۔ خُوش قسمتي سان هڪ اسلامي ڀاءُ جي اِنفِرادي ڪوشِش تي هو رَمَضانُ المُبَارَڪ جي آخري عَشَري ۾ پنهنجي عَلائقي جي مسجد ۾ اعتڪاف ڪيو، جتي ان کي سٺا سٺا خواب نظر آيا ۽ گھڻو سُڪون مليو۔ ان کان پوءِ ان وڌيڪ 2 سال اعتڪاف جي سَعَادَت حاصِل ڪئي۔ هڪ ڀيري ان جي مسجِد جي مُؤَذِّن صاحِب اِنفِرادي ڪوشِش ڪري اُن کي تبليغِ قرآن ۽ سنّت جي عالمگير غير سياسي تَحريڪ دعوتِ اسلامي جي عالَمي مَدَني مرڪز فَيضانِ مدينه(باب المدينه ڪراچي) ۾ هفتيوار سُنّتن ڀري اِجتِماع ۾ وٺي آيو۔ هڪ مُبَلِّغِ دعوتِ اسلامي سُنّتن ڀريو بَيان ڪري رهيا هئا جنهن کي سفيد لِباس ۽ ناسي چادرپاتل هئي، چهري تي هڪ مٺ ڏاڙھي ۽ مٿي تي سائي رنگ جي عِمامي شريف جو تاج سجايل هو۔ اهڙو بارونق چهرو ان زِندگي ۾ پهرين ڀيري ڏٺي هو۔ مُبَلِّغ جي چهري جي ڪشِش ۽ نُورانيت ان جي دل کٽي ورتي ۽ هو دعوتِ اسلامي جي مَدَني ماحول سان وابسته ٿي ويو۔ اَلْحَمْدُ لِلّٰه عَزَّ وَجَلَّ اُن هڪ مُٺ ڏاڙھي به رکي ورتي آهي۔

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم ۽ ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
