مٺا مٺا اسلامي ڀائرو! غور ڪريو! اسان وٽ کاڌو گھٽ هجي ۽ کائڻ وارا ماڻهو گھڻا ٿي وڃن ته کاڌي جي مقدار وڌائڻ کان علاوه ٻيو ڪو چارو ناهي هوندو، تقريبًا چار ڪلو اٽو ۽ ٻڪري جو ٻچو چاهي ڪيترو به صحت مند هجي مشڪل سان پنجاھ سٺ ماڻهن جي پيٽ ڀري کائڻ جو سامان بڻجي سگھي ٿو۔ پر نبي رحمت، شفيعِ امت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي لعابِ دهن جي برڪت سان ايترو کاڌو نه صرف صحابه ڪرام عَلَيهمُ الرِّضْوَان جي مڪمل جماعت کي پورو ٿي ويو بلڪ جيترو پچايو هو اوترو ئي باقي بچيل رهيو۔ ۽ پوءِ ٻڪري جي بچيل هڏين تي ڪلام پڙھيائون ته اهو گوشت کل سميت پهريان جھڙي ٻڪري ٿي ڪن ڇنڊيندي بيهي رهي، اها حڪايت سرڪارِ مدينه، قرارِ قلب و سينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي ڪيئي ڪمالن جي جامع آهي۔

مشهور مفسرِ قرآن، حڪيم الامت حضرت مفتي احمد يار خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هن حديثِ پاڪ جي تحت جيڪو ڪجھ ارشاد فرمايو آهي اچو! ان مان ڪجھ مدني گل ٻڌون ٿا: ٭ جن خوشنصيبن کاڌو کاڌو انهن جي تعداد چوڏهن سؤ هئي۔ تنهن مان هڪ هزار ته خندق کوٽڻ وارا هئا ۽ چار سو اهي حضرات هئا جيڪي بعد ۾بچيل رهيا، جيڪي مدينه منوره جي گھرن، بازارن وغيره ۾هئا، مدينه منوره جا ٻار عورتون به ان دعوت ۾شامل ٿيا ۔ مطلب ته کائڻ وارن جا ميلا لڳي ويا هئا۔ خوش نصيب هئا اهي ماڻهو جيڪي ان برڪت واري کاڌي سان مشرف ٿيا۔ ٭ حضور اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ انهن سڀني جي دعوت ڪري ڇڏي، ان ڏينهن لنگر حضور جو هو، گھر حضرت جابر جو هو، تنهنڪري اهو اعلان ۽ دعوت بلڪل درست آهي۔ جيڪا شيء استعمال سان گھٽجي نه اها مالڪ جي بغير اجازت استعمال ڪري سگھجي ٿي، جيئن ڪنهن جي چراغ جي روشني ۾ مطالعو ڪرڻ، ڪنهن جي ديوار مان ڇانوَ وٺڻ۔ اڄ اهو کاڌو انهن کائڻ وارن جي استعمال سان گھٽ نه ٿيندو تنهنڪري حضرت جابر جي بغير اجازت حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سڀني کي دعوت ڏئي ڇڏي۔ ٭ حضرت جابر (رَضِىَ اللهُ تَعَالٰی عَـنْهُ) سمورن ماڻهن کي دعوت ڏيڻ ۽ انهن ۾ اعلان ڪرڻ سان حيران ٿيا، پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ انه ن جي حيراني ملاحظه فرمائي ۽ تسلي ڏيڻ جي لاءِ اهو فرمايائون:گھٻرايو نه الله فضل ڪندو جيڪو آڻيندو اهو کارائيندو

، توهان ايترو ڪجو جو منهنجي اچڻ کان پهريان ديڳڙي چُلھ تان نه لاهجو ۽ اٽو پچائڻ نه شروع ڪجو پوءِ قدرتِ خدا جو نظارو ڏسجو۔ ٭ هن واقعي ۾ حضورِ انور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي لعاب شريف جا کوڙ معجزا آهن: ٻوٽين ۾ ڪثرت ۽ برڪت،ٻوڙ جي پاڻيءَ ۾ برڪت ، ٻوڙ جي لوڻ مرچ مصالحي گيھ ۾ برڪت ۽ ڪثرت، اٽي ۾ برڪت ۽ ڪثرت، جنهن ڪاٺي سان اهي شيون پچايون ويون ان ۾ برڪت، ماني پچائڻ واري جي هٿ ۾ قوت ۽ طاقت نه ته ايتري وڏي جماعت جي دعوت جي لاءِ ڪيئي مڻ گوشت ڪاٺيون اٽو گھرجي ها، پچائڻ وارا به گھڻا کپن ۽ کوڙ تندور کپن ها جيئن ته اڄڪلھ شادين جي دعوتن ۾ ڏٺو ويندو آهي۔ ٭ (حضرت) موسيٰ عَلَيه السَّلَام جي عصا سان پاڻي جا ٻارهن چشما پٿر (Stone) مان ڦٽي نڪتا، هتي حضور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي لعاب سان ديڳڙي مان ٻوٽين،ٻوڙ جا چشما ڦٽي نڪتا ۔

معجزي جي تعريف

حڪيم الاُمّت مفتي احمد يار خان رَحْمَةُ اللهِ عَلَيۡه معجزي جي تعريف بيان ڪندي فرمائن ٿا: اهو ڪم جنهن جي مقابلي سان بلڪ ان جي سمجھ کان مخلوق عاجز هجي ان کي ”معجزو “ چوندا آهن۔ شريعت جي اصطلاح ۾ معجزو هر اهو عجيب ۽ غريب خلافِ عادت ڪم آهي جيڪو نبوت جي دعوي ڪرڻ واري جي هٿان ظاهر ٿئي۔ نبوت جي دعوي کان پهريان جيڪو (خلافِ عادت ڪم) نبي جي هٿان ظاهر ٿئي ان کي ”اِرهاص “ چوندا آهن۔ اولياءُ الله جي هٿان جيڪا عجيب ڳالھ ظاهر ٿئي ان کي ”ڪرامت “ چوندا آهن۔ عام مؤمنن جي هٿان جيڪڏهن، ڪڏهن ڪا عجيب ڳالھ ظاهر ٿئي ان کي ”مَعُونَت “ چوندا

آهن۔ ۽ ڪافرن جي هٿان جيڪي عجوبه (عجيب و غريب ڪم) ظاهر ٿين ان کي ” اِستِدرَاج “ چوندا آهن۔ مفتي صاحب رَحْمَةُ اللهِ عَلَيۡه مزيد فرمائن ٿا: سمورن نبين جا معجزا قصا (parables) بڻجي ويا، اسان جي حضور جا کوڙ معجزا قيامت تائين ڏسڻ ۾ ايندا، ذڪرِ ڪثير، محبوبيتِ قرآنِ مجيد، پٿرن، جانورن تي حضور جو نالو لکيل ملڻ وغيره اهي زنده جاويد معجزا آهن۔ حضور جا اولياءالله، انهن جون ڪرامتون حضور جا زنده معجزا آهن۔ (مرآة المناجيح، ۸/۱۶۲ملخصاً)

جامعُ الُمعجِزات

مٺا مٺا اسلامي ڀائرو! مُعجِزَو نبي جي نَبُوَّت جو دليل هوندو آهي، تنهنڪري الله پاڪ هر نَبي کي ان دَور جي ماحول ۽ ان جي اُمَّت جي مِزاجِ عَقل ۽ فَهم (سوچڻ سمجھڻ جي صلاحيت) جي مُناسِب مُعجِزَات سان نوازيو هو۔ مثال طور حَضرتِ مُوسيٰ عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي دورِ نَبُوَّت ۾ جيئن ته جادُو (Magic) جي ذريعي ڪارناما پنهنجي ترقّي جي اعليٰ تَرين مَنزل تي پهتل هئا، ان ڪري الله پاڪ اوهان کي ”يدِبَيۡضا“ ۽ ”عَصا“ جا مُعجِزَا عطا ڪيا، جن سان اوهان جادُوگرن جي سَاحِرانه (جادوئي) ڪارنامن تي هن طرح غَلَبو حاصل ڪيائون جو تمام جادُوگر سَجدي ۾ ڪري پيا ۽ حَضرتِ مُوسيٰ عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام تي اِيمان آندائون۔ اهڙي طرح حَضرتِ عيسيٰ عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي زماني ۾ عِلمِ طِب اِنتهائي تَرقّي تي پهتل هو ۽ ان دَور جي طَبِيبن يعني ڊاڪٽرن وڏين وڏين بيمارين جو عِلاج ڪري پنهنجي فنّي مَهارَت سان سڀني اِنسانن کي مَسحُور (پنهنجي قابو ۾) ڪري رکيو هو، ان ڪري الله پاڪ حَضرتِ عيسيٰ عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي پيدائشي اَنڌن ۽ ڪوڙھين کي شِفا ڏيڻ ۽ مُردن کي زِنده ڪرڻ جو مُعجزو عطا فرمايو، جنهن کي ڏسي پاڻ عَلَيۡهِ السَّلَام جي دَور جي طبيبن (يعني ڊاڪٽرن) جا هوش اُڏامي ويا ۽ اهي حَيران ٿي ويا، آخرڪار انهن هنن مُعجِزن کي اِنساني ڪمالن کان بلند مڃي سندن نَبُوَّت جو اِقرار ڪيائون۔ اَلغرض هر نَبي کي ان دَور جي ماحول مُطابِق ۽ ان جي قَوم جي مِزاج ۽ انهن جي طبيعت جي مُناسِب ڪنهن کي هڪ، ڪنهن کي ٻه، ڪنهن کي ان کان وڌيڪ مُعجِزَات عطا ٿيا، پر نبي آخرُالزّمان، سَرورِ ڪون و مَڪان صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جيئن ته تمام نبين جا به نَبي آهن ۽ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي سِيرتِ مُقدَّسه تمام اَنبِياء عَلَيهمُ الصَّلٰوة السَّلَام جي مُقدَّس زِندگين جو خُلاصو (Summary) آهي

۽ رسول ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي تَعلِيم تَمام اَنبياءِ ڪرام عَلَيهمُ الصَّلٰوة السَّلَام جي تَعليمات جو عطر (يعني نِچوڙ) آهي ۽ پاڻ سڳورا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ دُنيا ۾ هڪ عالمگير ۽ اَبَدي دِين کڻي تَشرِيف فرما ٿيا ۽ عالمِ ڪائنات ۾ اَوَّلين ۽ آخرين جون تمام قومون ۽ مِلّتون پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مُقدَّس دَعوت جي مُخاطَب هيون، ان ڪري الله پاڪ سندن مُقدَّس ذات کي اڳوڻن نبين جي سمورن مُعجزن جو مَجموعو (Collection) بڻائي ڇڏيو ۽ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي قِسم قِسم جي اهڙن بيشُمار مُعجزن سان سرفراز فرمايو جيڪو هر طَبقي، هر ٽولي، هر قوم ۽ تَمام اَهلِ مَذاهب جي مِزاجِ عَقل ۽ سمجهه جي لاءِ ضَروري هو۔ يعني الله پاڪ تمام نبين جي سردار، محبوبِ غفار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي اهڙا اهڙا معجزا عطا ڪيا جيڪي هر قوم بلڪ اسلام جي علاوه ٻين مذهبن سان تعلق رکڻ وارن جي مزاج ۽ سوچ جي به مطابق هئا. (سيرتِ مُصْطَفٰے، ص ۷۱۲ تا ۷۱۴،ملخـصًا و ملتقطاً) ان کان علاوه بي شُمار اهڙن مُعجزن سان به الله پاڪ پنهنجي آخِري پَيغمبر، شَفيعِ مَحشر صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کي مُمتاز فرمايو، جيڪي پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جون خاصيتون چيو وينديون آهن۔ يعني هي پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا اهي ڪمال ۽ مُعجزا آهن، جيڪي ڪنهن ٻي نَبي ۽ رَسُول کي عَطا نه ڪيا ويا۔ (سيرتِ مُصْطَفٰے، ص ۸۲۰ ملخـصاً) تنهنڪري اسان جي آقا، مڪي مَدَني مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي ذات اُها ذات آهي جنهن ۾ سڀني معجزن کي جمع ڪيو ويو آهي۔

(مسلم،ڪتاب الفضائل،باب في معجزات النبي ، ص ۹۶۳،حديث ۵۹۴۶)

2۔ مشهور صحابي رسول حضرت ابوهريره رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو بيان آهي ته مان حضورِ اقدس صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي خدمتِ اقدس ۾ ڪجھ کجيون (Dates) کڻي حاضر ٿيس ۽ عرض ڪيو ته يارَسُولَ الله! (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) هنن کجين۾ برڪت جي دعا ڪريو۔ رسول ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ انهن کجورن کي گڏ ڪري برڪت جي دعا ڪئي ۽ ارشاد فرمايائون ته توهان هنن کي پنهنجي توشه دان (ٿهيلي) ۾ رکي وٺو ۽ توهان جڏهن چاهيو هٿ وجھي ان مان ڪڍندا رهو پر ڪڏهن توشه دان ڇنڊي بلڪل خالي نه ڪجو۔ پوءِ حضرت ابوهريره رَضِىَ اللهُ تَعَالٰی عَـنْهُ انهن کجين مان پاڻ به کائيندا هئا، ماڻهن کي به کارائيندا هئا ۽ مڻن جي حساب کان الله پاڪ جي راھ ۾ به ڏيندا هئا۔ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ هميشه ان ٿيلهي کي پنهنجي چيلھ سان ٻڌي رکندا هئا ايستائين جو حضرت عثمان رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي شهادت جي ڏينهن ۾ اها ٿيلهي انهن جي چيلھ تان ڪٽجي ڪٿي ڪري پئي۔ (ترمذي، ڪتاب المناقب، باب مناقب لابي هريره ،۵/۴۵۴، حديث: ۳۸۶۵)

3۔ حضرت بي بي ام سُلَيم رَضِىَ اللّٰهُ عَنْها وٽ هڪ ٻڪري هئي۔ انهن ان جي کير مان گيھ ٺاهي هڪ مشڪيزي ۾ گڏ ڪيو، جڏهن مشڪيزو ڀرجي ويو ته ٻانهيءَ جي هٿان اهو مشڪيزو رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي بارگاھ ۾ موڪلي ڇڏيو ته جيئن پاڻ سڳورا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان جي ذريعي ٻوڙ بڻائن۔ رسول اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايائون: انهن جو مشڪيزو خالي ڪري موٽائي ڏيو۔ خالي مشڪيزو گھر پهتو ۽ جڏهن بي بي ام سُلَيم رَضِىَ اللّٰهُ عَنْها ڏٺائين ته گھڻو حيران ٿيون ته اهو مشڪيزو جيئن ڀريل هو تئين ئي ڀريل آهي ۽ ان مان گيھ به وهي رهيو آهي، ٻانهيء کان پڇيائين

ته ڇا اهو مشڪيزو سرڪار عالي وقار صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي بارگاھ ۾ نه کڻي وئي هئينءَ؟ ان چيو: مون ائين ئي ڪيو جيئن اوهان چيو هو، ڀلي اوهان پاڻ وڃي تصديق (Verify) ڪري وٺو۔ حضرت ام سليم رَضِىَ اللّٰهُ عَنْها بارگاھ رسالت مآب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ۾ حاضر ٿيون ۽ عرض ڪيو ته: مون اوهان جي خدمت ۾ گيھ جو مشڪيزو موڪليو هو، ان ذات جو قسم جنهن اوهان کي هدايت ۽ دينِ حق سان گڏ موڪليو آهي، اهو مشڪيزو پهريان وانگر ڀريل آهي ۽ ان مان انهيءَ طرح گيھ وهي رهيو آهي۔ ته رحمت عالم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: ڇا تون هن ڳالھ سان تعجب ڪرين ٿي؟ الله پاڪ توکي کارايو آهي جهڙي طرح تو ان جي نبي کي کارايو، کائو ۽ کارايو۔ اُم سُلَيم رَضِىَ اللّٰهُ عَنْها چون ٿيون ته: مان گھر موٽيس ۽ مون ان گيھ کي مختلف پيالن ۾ وجھي ڇڏيو ۽ ڪجھ گيھ ان مشڪيزي ۾ رهڻ ڏنو جنهن سان اسان هڪ يا ٻن مهينن تائين ٻوڙ ٺاهيو۔ (مجمع الزوائد و منبع الفوائد، ڪتاب علامات النبوه ،باب معجزاته في الطعام و برڪته فيه،۸/۵۴۳، حديث:۱۴۱۲۶)

مدني چينل ڏسڻ جون برڪتون ۽ ترغيب

مٺا مٺا اسلامي ڀائرو! عشقِ رسول جو ڄام پيئڻ جي لاءِ مدني چينل ڏسندا رهو. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مدني چينل ڏسڻ جون تمام گهڻيون برڪتون آهن

٭ مثال طور مدني چينل ڏسڻ جي برڪت سان ديني معلومات ۾ اضافو ٿيندو آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان نماز روزو وضو غسل ۽ ٻيا فقهي مسئلا سکڻ جي سعادت نصيب ٿيندي آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان خوب خوب سنتون سکڻ جي لاءِ ملنديون آهن۔ ٭ مدني چينل ڏسڻ جي برڪت سان بزرگانِ دين رَحْمَةُ اللهِ تَعَالٰی عَلَيْهم اجمعين جي خوفِ خدا ۽ عشقِ مصطفٰي سان ڀريل واقعا ٻڌڻ جي توفيق ملندي آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان ديدارِ اميرِاهلسنت نصيب ٿيندو آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان دعوت اسلامي جي اپ ڊيٽس حاصل ٿيندي آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان عمل جو جذبو نصيب ٿيندو آهي۔ ٭ مدني چينل ڏسڻ جي برڪت سان تلاوت، نعت ۽ منقبتون ٻڌڻ جون سعادتون ملنديون آهن، تنهنڪري جڏهن به ڪنهن سان ملاقات ٿئي يا فون تي ڳالهه ٻولھ يا ڪاٿي تقريب ۾ وڃو، ڪاٿي سنتن ڀريو بيان ڪرڻ جي ترڪيب هجي هر هڪ کي مدني چينل ڏسڻ جي دعوت ڏيو ڇو ته مدني چينل ڏسڻ جي دعوت ڏيڻ به نيڪي جي دعوت ڏيڻ جو ذريعو آهي. الله ڪري ته اسان پاڻ به مدني چينل ڏسون ۽ گهروارن کي به مدني چينل ڏسڻ جو ذهن ڏيون.

ويهڻ جون ڪجهه سُنتون ۽ آداب

مٺا مٺا اسلامي ڀائرو! اچو ويهڻ جون ڪجھ سنتون۽ آداب ملاحظه فرمايو ٭ سُرين زمين تي رکو ۽ ٻئي گوڏا اُڀا ڪري ٻنهي ٻانهن کي گوڏن جي چوڌاري ورايو ۽ هڪ هٿ سان ٻئي کي پڪڙيو، ائين ويهڻ سنّت آهي، (پر ان

دؤران گوڏن تي ڪا چادر وغيره وجهڻ بهتر آهي) (مرآۃالمناجيح،۶/۳۷۸) ٭ پلٿي هڻي ويهڻ به نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان ثابت آهي۔ ٭ جتي ڪجهه اُس ۽ ڪجهه ڇانوَ هجي اتي نه ويهو۔ الله عَزَّوَجَلَّ جي محبوب، داناءِ غُيُوب، مُنَزَّهٌ عَنِ الۡعُيُوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”جڏهن توهان مان ڪو ڇانوَ ۾ هجي ۽ ان تان ڇانوَ ختم ٿي وڃي ۽ هو ڪجهه اُس ۽ ڪجهه ڇانوَ ۾ رهجي وڃي ته ان کي گهرجي ته ان جاءِ تان اٿي وڃي۔ (ابو داود،ڪتاب الادب،باب في الجلوس بين الظل و الشمس، حديث ۴۸۲۱، ج۴،ص۳۳۸) ٭ قبلي ڏانهن منهن ڪري ويهو۔ (رسائل عطاريه حصو 2 ص 229) اعليٰ حضرت امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه لکن ٿا: پير ۽ استاد جي جاءِ تي انهن جي غير موجودگي ۾ به نه ويهي۔ (فتاويٰ رضويه، ج 24 ص 369، 424) ٭ جڏهن به اجتماع يا مجلس ۾ اچو ته ماڻهن کي اورانگهي اڳتي نه وڌو جتي جڳهه ملي اتي ئي ويهي رهو۔ ٭ جڏهن ويهو ته جوتا لاهي ويهو توهان جي پيرن کي آرام ملندو ۔ (جامع صغير ،حديث:۵۵۴،ص۴۰) ٭ مجلس مان فارغ ٿي ٽي ڀيرا هيءَ دعا پڙهو ته گناهه معاف ٿي ويندا ۽ جيڪو اسلامي ڀاءُ ڀلائي ۽ ذِڪر واري مجلس ۾ پڙهي ته ان جي لاءِ خير تي مهر لڳائي ويندي، اها دعا هي آهي؛ سُبْحَانَک اللّٰھُمَّ وَبِحَمْدِک لَا اِلٰه اِلَّا اَنْتَ اَسْتَغْفِرُک وَاَتُوْبُ اِلَيک ترجمو: ” تنهنجي ذات پاڪ آهي ۽ اي الله! سڀ خوبيون تنهنجي لاءِ ئي آهن، تو کان سواءِ ڪو معبود ناهي، توکان بخشش گهران ٿو ۽ تو ڏانهن توبهه ڪريان ٿو۔ (سنن ابي داؤد،ڪتاب الادب، باب في ڪفارة المجلس، الحديث: ۴۸۵۷، ۴/۳۴۷) ٭ جڏهن ڪو باعمل عالم يا متقي شخص يا سيّد صاحب يا ماءُ پيءُ اچن ته تعظيم لاءِ اُٿي بيهڻ ثواب آهي۔ ٭ مفتي احمد يار خان نعيمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه لکن ٿا؛ بزرگن جي اچڻ تي اهي ٻئي ڪم يعني تعظيمي قيام ۽ استقبال(خوش آمديد چوڻ) جائز بلڪ صحابه جي سنت آهن ۔ (مراٰةالمناجيح ،ج۶، ص ۳۷۰، جمله منقول از سنتيں ۽ آداب،ص۹۹، بتغير)

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي

هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([4])

﴿5﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاري ڇڏيو، تنهن تي صَحابَه ڪرام

رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
