وات ڦاڙي ان تي حملو ڪرڻ وارا آهن۔ اهو خوفناڪ منظر ڏسي اهو خوف ۽ دَهشَت سبب رڙيون ڪرڻ لڳو، جنهن سبب اَمِيرُالمؤمنين رَضِىَ اللهُ تَعَالٰی عَـنْهُ بيدار ٿي ويا ۽ مُلاحظه فرمايو ته عجمي شخص کليل تلوار هٿ ۾ کڻي گهڻو ڏڪي رهيو آهي۔ پاڻ ان جي رڙين ۽ دَهشَت جو سبب معلوم ڪيو ته ان سَچ سَچ سمورو واقِعو بيان ڪري ڇڏيو پوءِ ان بلندآواز سان ڪلمو پڙھي اِسلام قبول ڪري ورتو ۽ اَمِيرُالمؤمنين رَضِىَ اللهُ تَعَالٰی عَـنْهُ ان سان نهايت ئي شَفقَت ڀريو ورتاؤ ڪري ان جي قُصُور کي مُعاف ڪري ڇڏيو۔ (اِزالة الخِفاء،فصل رابع، ۴/۱۰۹)

سگهي، اهڙو شخص عاجزيءَ تي به قادر ناهي هوندو، جيڪا تقويٰ ۽ پرهيزگاري جي جڙ آهي، ڪينو (يعني دل ۾ لڪيل دُشمني) به نٿو ڇڏي سگهي، پنهنجي عزت بچائڻ جي لاءِ ڪوڙ ڳالهائيندو آهي، ان ڪوڙي عزت جي ڪري غصو نٿو ڇڏي سگهي، حسد کان نٿو بچي سگهي، ڪنهن جي خير خواهي نٿو ڪري سگهي، ٻين جي نصيحت قبول ڪرڻ کان محروم رهي ٿو، ماڻهن جي غيبت ۾ مبتلا ٿي ويندو آهي مطلب ته متڪبر ماڻهو پنهنجو ڀرم رکڻ جي لاءِ هر برائي ڪرڻ تي مجبور ۽ هر سٺي ڪم ڪرڻ کان لاچار ٿي ويندو آهي۔“ (احياء العلوم، ج۳ ،ص ۴۲۳) وڌيڪ فرمائن ٿا: تڪبُّر جو اظهار ڪڏهن ته انسان جي عادتن ۽ طور طريقن سان ٿيندو آهي جيئن ناراض ٿيڻ، بيزار ٿيڻ، گهورڻ مٿي کي هڪ طرف جهڪائڻ، ٽنگ تي ٽنگ رکي ويهڻ، ٽيڪ لڳائي کائڻ، آڪڙجي هلڻ وغيره ۽ ڪڏهن ڳالهائڻ سان مثلاً هي چوڻ: ”(تون منهنجي سامهون ڪجھ به نه آهين) تنهنجي اها هِمّت جو مون کي جواب ٿو ڏئين“وغيره وغيره ۔ مطلب ته مختلف حالن، ڳالهين ۽ ڪمن جي ذريعي تڪبّر جو اظهار ٿي سگهي ٿو۔ وري ڪِن مغرورن ۾ اظهار جا سڀ انداز ڏٺا ويندا آهن ۽ ڪِن مُتَڪَبِّرن ۾ ڪجهه۔ پر ياد رهي ته اهي سڀ ڳالهيون اُن وقت تڪبر جي دائري ۾ اينديون، جڏهن ته دل ۾ تڪبر موجود هجي، فقط انهن شين کي تڪبر نٿو چئي سگهجي۔ (ماخوذ از احياء العلوم ،ج۳، ص۴۳۴)

’’جيڪو پنهنجي مسلمان ڀاءُ جي لاءِ تواضُع اختيار ڪندو آهي الله عَزَّوَجَلَّ ان کي بُلندي عطا ڪندو آهي ۽ جيڪو مسلمان ڀاءُ تي بلندي چاهيندو آهي الله عَزَّوَجَلَّ ان کي پستي ۾ وجھندو آهي۔ (معجم الاوسط، من اسمه محمد، ۵/۳۹۰، الحديث: ۷۷۱۱)

موزا لاهي ڪلهي تي رکي ورتا!

الله عَزَّوَجَلَّ اميرالمومنين فاروقِ اَعظَم رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي عاجزي ۽ انڪساري جهڙي صفت جي ڪري نهايت بُلند مقام ۽ مرتبو عطا فرمايو، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ الله عَزَّوَجَلَّ جي طرفان ملڻ واري هن عزّت ۽ مرتبي کان علاوه ماڻهن طرفان ملندڙ عزّت ۽ شهرت جا بلڪل خواهش مند نه هئا، جيئن ته هڪ ڀيري پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ

ملڪ شام ڏانهن تشريف کڻي ويا ته حضرت سَيدُنا ابُوعبيده رَضِىَ اللهُ تَعَالٰی عَـنْهُ به پاڻ سان گڏ هئا ايستائين جو پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ هڪ اهڙي هنڌ تي پهتا، جتي گوڏن تائين پاڻي هو، پاڻ پنهنجي اُٺڙي تي سوار هئا، اُٺڙي تان لٿا ۽ پنهنجا جورابا لاهي پنهنجي ڪلهي تي رکي ورتا، پوءِ اُٺڙي جي لغام پڪڙي پاڻي ۾ داخل ٿيا ته حضرت ابُوعبيده رَضِىَ اللهُ تَعَالٰی عَـنْهُ عرض ڪيو: اي اميرُالمؤمنين! رَضِىَ اللهُ تَعَالٰی عَـنْهُ ، اوهان جيڪو اهو ڪم ڪري رهيا آهيو، مون کي اهو پسند ناهي ته هتي جا ماڻهو اوهان کي نظر کڻي نهارين۔ حضرت عُمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ ارشاد فرمايو: ’’افسوس! اي اَبُوعُبيده! جيڪڏهن اها ڳالھ اوهان کان علاوه ڪو ٻيو ڪري ها ته مان ان کي اُمَّتِ محمدي جي لاءِ عبرت بڻايان ها، اسان هڪ غريب قوم هئاسين، پوءِ الله تعاليٰ اسان کي اسلام جي ذريعي عزّت عطا فرمائي، جڏهن به اسان الله تعاليٰ جي عطا ڪيل عزّت کان عِلاوه (ڪنهن کان) عزّت حاصِل ڪرڻ چاهينداسين ته الله تعاليٰ اسان کي رُسوا ڪري ڇڏيندو۔

(الزواجر عن اقتراف الڪبائر، الباب الاول في الڪبائر الباطنة۔۔۔ الخ، ۱/۱۶۱)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اميرُالمُؤمنين حضرت سَيدُنا فاروقِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي عاجزي ۽ اِنڪساري ته ڏسو جو وقت جو حاڪم ٿي به ڄڻ ان ڳالھ کي ناپسند ڪري رهيا آهن ته ماڻهو مون کي منهنجي منصب جي ڪري عزّت (Protocol) ڏين، مون کي ته بس الله عَزَّوَجَلَّ جي طرفان ملڻ واري عزّت ئي ڪافي آهي ۽ ٻيو اسان آهيون جو هر مُعاملي ۾ پنهنجي عزّت ۽ شُهرت ۽ پنهنجي

واه واه چاهيون ٿا، ڪڏهن ڪڏهن اها خواهش به جوش ڏيندي آهي ته جيڪڏهن اميرن ۽ صاحبِ مَنصب حضرات ۾ به شهرت ملي وڃي ته داد ۽ تحسين سان گڏوگڏ گھڻا انعامات به حاصِل ٿي ويندا! الله عَزَّوَجَلَّ سيپاري 4سوره آلِ عمران آيت نمبر188 ۾ ارشاد فرمائي ٿو:

لَا تَحۡسَبَنَّ الَّذِیۡنَ یَفۡرَحُوۡنَ بِمَاۤ اَتَوۡا وَّ یُحِبُّوۡنَ اَنۡ یُّحۡمَدُوۡا بِمَا لَمۡ یَفۡعَلُوۡا فَلَا تَحۡسَبَنَّہُمۡ بِمَفَازَۃٍ مِّنَ الۡعَذَابِ ۚ وَ لَہُمۡ عَذَابٌ اَلِیۡمٌ ﴿۱۸۸﴾

ترجموڪنز الايمان: هرگِز اُنهن کي نه ڀانءِ جيڪِي پنهنجي ڪَئي تي خوش ٿِين ٿا ۽ گُهرن ٿا ته بِنا ڪم جي اُنهن جِي تعرِيف ٿئي اِهڙن کي عذاب کان هرگِز پري نه ڄاڻندا ۽ اُنهن لاءِ دردناڪ عذاب آهي.

مڪتبة المدينه جي شايع ٿيل تفسير صِراطُ الجنان جلد 2 صفحي 117 تي آهي: هن آيت ۾ خُود پسندي ڪرڻ وارن جي لاءِ وعيد آهي ۽ انهن جي لاءِ جيڪي حُبِّ جاه يعني عزّت، واکاڻ، شهرت حاصل ڪرڻ جي تمنّا ۾ مبُتلا آهن۔ جڏهن ڪنهن شخص جي دل ۾ اها آرزُو پيدا ٿيڻ لڳي ته ماڻهو ان جا شيدائي ٿين، هر زبان انهن جي واکاڻ ڪري، سڀ منهنجي ڪمال جو اعتراف ڪن، مون کي هر جڳھ عزّت کان نوازيو وڃي، عالِم نه آهيان پوءِ به علّامه صاحب چيو وڃي، ملڪ ۽ قوم جي ڪا خدمت نه ڪئي پوءِ به مِعمارِ قوم چيو وڃي، نجات ڏيندڙ سمجھيو وڃي، محسنِ قوم قرار ڏنو وڃي، منهنجو تعارف بهترين لقبن سان ٿئي، مُلاقات پُرتپاڪ انداز ۾ ڪئي وڃي، سلام جُھڪي ڪيو وڃي ته ان کي گھرجي ته پنهنجي دل تي غور ڪري وٺي ته ڪٿي اها حُبِّ جاه جي شڪار ته نه ٿي چڪي آهي، جيڪڏهن ائين هجي ته ان آيت مان سبق حاصل ڪندي تمام جلدي ان کان ڇوٽڪاري جي ڪوشش ڪري۔ ياد رکو! خُود پسندي ۽ حُبِّ جاه جي مرض ۾ مبُتلا شخص اُخروي انعامات کان محروميءَ جو شڪار ٿيندو آهي ۽ دل ۾ مُنافَقت جي زيادتي، قلبي نُورانيت کان محرومي، دين جي خرابي ۾ مبُتلا ٿي ويندي آهي ۽ بُرائي کان منع ڪرڻ

۽ نيڪي جي دعوت ڏيڻ کان محرومي، ذِلّت ۽ رُسوائي جو مقدر ٿيڻ، اُخروي لَذت کان محرومي، قلبي سُڪون جي بربادي ۽ دولتِ اِخلاص کان محرومي جهڙن نُقصانن کي منهن ڏيڻو پئجي سگھي ٿو، ان ڪري ان کي گھرجي ته دنيا جي بي ثباتي، تعريف پسندي جي مذمت، مَنصَب ۽ مرتبي جي حوالي سان اُخروي مُعاملن ۽ بُزرگانِ دِين جي احوال ۽ اَقوال جو تمام گھڻو مُطالعو ڪري ته جيئن هنن مذموم مرضن کان نجات جي ڪا صورت ٿئي۔

اچو!خُودپسندي ۽ حُبِّ جاه کان بچڻ جي لاءِ ٽي (3) فرامينِ مُصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌون ٿا:

1. ’’جيڪو شهرت طلب ڪندو (قيامت جي ڏينهن) ان جي عيبن جي تشهير ٿيندي ۽ جيڪو شخص ماڻهن کي ڏيکارڻ جي لاءِ عمل ڪندو الله تعاليٰ ان کي ان جو بدلو ڏيندو۔ (بخاري، ڪتاب الرقاق، باب الرياء والسمعة، ۴/۲۴۷، الحديث: ۶۴۹۹)

2. ’’ٻه بکايل بگھڙ ٻڪرين جي ڌَڻ ۾ ڇڏيا وڃن ته اهي ايترو نُقصان نٿا ڪن جيترو مال ۽ مرتبي جو حرص ڪرڻ وارو پنهنجي دِين جي لاءِ نُقصان دِه آهي۔ (ترمذي،۴/۱۶۶، الحديث: ۲۳۸۳)

3. ’’جيڪڏهن خُود پسندي انساني شڪل ۾ هجي ها ته اها سڀ کان بد صُورت انسان هجي ها۔ (الفردوس بماثور الخطاب، باب اللام، ۲/۱۹۳، الحديث: ۵۰۶۴)

4. صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

حضرت سيدُنا ابُو رافع رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي، فرمائن ٿا ته هڪ ڀيري امير المؤمنين حضرت سيدُناعمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ منهنجي ويجھو کان گذريا ته مان قرآنِ پاڪ جي خُوبصورت لهجي ۾ تلاوت ڪري رهيو هئس، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ٻڌي ارشاد فرمايو: یَا اَبَا رَافِعْ! لَاَنْتَ خَیْرٌ مِّنْ عُمَرَ تُؤَدِّیْ حَقَّ اللہِ وَحَقَّ مَوَالِیْكَ يعني اي ابُو رافع! يقيناً توهان عمر کان بهتر آهيو ته توهان الله عَزَّ وَجَلَّ جا حق ۽ پنهنجي آقا جا حق چڱي طرح اَدا ڪيو ٿا۔‘‘

عاجزي پيداڪرڻ جو هڪ طريقو اهو به آهي ته پنهنجو سامان پاڻ کڻو ۽ جيڪڏهن ڪو عذر نه هجي ته گھر جو ڪم ڪرڻ جي به عادت بڻايو، فرمانِ مصطفي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: ”جنهن پنهنجو سامان پاڻ کڻي ورتو اهو تڪبُّر کان آزاد ٿي ويو۔“ (شعب الايمان، باب في حسن الخلق، فصل في التواضع، الحديث: ۸۲۰۱،ج۶،ص۲۹۲)

پيارا آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پاڻ به پنهنجي گھر جي ڪم ڪار ۾ مشغول رهندا هئا'' (بخاري،ج۱،ص۲۴۱) انهيءَ سنت تي عمل ڪندي صدرُالشَّريعه،بدرُالطَّريقه مفتي

محمد امجد علي اعظمي رَحْمَة اللّٰه ِتَعَالٰي عَلَيه گھر جي ڪم ڪار کان عار (شرم) محسوس نه ڪندا هئا، گھر ۾ ڀاڄيون ڇهليندا، ڪٽيندا ۽ ٻيا ڪم به ڪندا هئا۔

(ماهنامه اشرفيه،صدر الشريعه نمبر،ص۵۴)

الله عَزَّ وَجَلَّ اسان کي به گھر جو ڪم پاڻ ڪرڻ جي توفيق عطا فرمائي۔

(4)هنن عملن کي اختيار ڪيو!

اهڙا عمل اختيار ڪيو، جيڪي تڪبُّر کي ختم ڪرڻ وارا آهن، غريب ۽ نيڪ مسلمانن جي صُحبت ۾ ويهو، اِنْ شَآءَ الله عَزَّوَجَلَّ ان سان به تڪبّر ختم ٿيندو، عاجزي پيدا ٿيندي۔

ٻڪري جي کَل تي ويهڻ جي برڪت

دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل 1250 صفحن تي مشتمل ڪتاب ”بهارِ شريعت“ جلد اول حصو 2 صفحي 403 تي آهي: ”ٻڪري ۽ گهٽي جي (دباغت ٿيل يعني مخصوص طريقي سان خشڪ ڪيل) کل تي ويهڻ ۽ پائڻ سان طبيعت ۾ نرمي ۽ انڪساري پيدا ٿيندي آهي۔ “ (بهارشريعت،جلد اوّل، ص۴۰۳، بتغير) اَلۡحَمۡدُلله عَزَّوَجَلَّ شيخِ طريقت امير اهلسنّت دَامَتۡ بَرَكَاتُهُمُ الۡعَالِيَه پنهنجي ويهڻ جي جڳهه تي عام طور تي ٻڪري جي (خشڪ ٿيل) کَل وڇائڻ جو اهتمام فرمائيندا آهن بلڪ بيان جي دوران به اڪثر پاڻ دَامَتۡ بَرَكَاتُهُمُ الۡعَالِيَه جي ويهڻ جي جڳهه تي ٻڪري جي کَل وڇايل هوندي آهي۔ اوهان به ڪوشش ڪري پنهنجي ويهڻ جي جڳهه تي ٻڪري يا گهٽي جي (خشڪ ٿيل) کَل وڇائي ويهو ۽ انهن جي برڪتن جو اکين ڏٺو نظارو ڪريو۔

(5)صدقو ڏيو!

صدقو ۽ خيرات ڪرڻ به اهڙو عمل آهي جيڪو عاجزي پيدا ڪندو ۽ تڪبُّر کي ختم ڪندو آهي۔ فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: '' مسلمان جو صدقو عمر ۾ اضافي جو سبب ۽ بُري موت کان بچائي ٿو ۽ الله تعاليٰ ان جي سبب تڪبُّر ۽ فخر کي دور فرمائي ڇڏي ٿو۔“ (مجمع الزوائد ،ڪتاب الزڪاة، باب فضل الصدقة، رقم ۴۶۰۹،ج ۳، ص ۲۸۴) ان

ڪري پنهنجا صدقات، زڪوٰة، خيرات دعوتِ اسلامي کي ڏئي مدني ڪمن ۾ مُعاوِن بڻجو۔

(6) حق ڳالهه تسليم ڪري وٺو

مٺا مٺا اسلامي ڀائرو! عاجزي پيدا ڪرڻ وارن ڪمن مان هڪ اهو به آهي ته جيڪڏهن ڪير به اسان جي ڪنهن غلطي جي خبر ڏئي ته جلدي پنهنجي غلطي مڃڻ گهرجي، چاهي هو عمر، تجربي ۽ رُتبي ۾ اوهان کان گهٽ ئي ڇو نه هجي۔ ۽ جڏهن ڪنهن اسلامي ڀاءُ سان اختلافِ راءِ ٿئي، پوءِ اوهان تي ظاهر ٿئي ته اهو حق تي آهي ته ضد ڪرڻ جي بدران مڃي وٺو، پوءِ اُن جي سامهون اِن ڳالهه جو اعتراف ڪندي حق جي بيان تي اُن جي تعريف به ڪريو ته ”اوهان درست فرمائي رهيا آهيو الله تعاليٰ اوهان کي جزاءِ خير عطا فرمائي۔“ اهڙي موقعي تي پنهنجي غلطيءَ مڃڻ اهو نفس تي تمام گھڻو دشوار آهي، پر لاڳيتو ائين ڪرڻ سان حق جو اعتراف ڪرڻ اوهان جي عادت بڻجي ويندي ۽ ان جي برڪت سان تڪبُّر کان به جان ڇٽي ويندي۔ اِنْ شَآءَ الله عَزَّ وَجَلَّ

غلطي جو اعتراف!

جليلُ القَدر محدِّث امام دارِ قُطني رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جڏهن ننڍي عمر جا شاگرد هئا ته هڪ ڏينهن حضرت سيدنا امام انباري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي درسگاھ ۾ حاضر ٿيا، حديث لکرائڻ مهل امام انباري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن هڪ راوي جي نالي ۾ غلطي ڪئي، حضرت سيدنا دار قطني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ادب جي سبب امام انباري رَحْمَة اللّٰه ِتَعَالٰي عَلَيه کي ته ٽوڪي نه سگهيا پر جيڪو اُنهن جو آواز شاگردن تائين پهچائيندو هو ان کي ان غلطي کان آگاھ ڪري ڇڏيو۔ جڏهن ٻئي جمعي تي امام دارِ قُطني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه وري مجلسِ درس ۾ ويا ته امام انباري رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جو حق پسندي جو جوش ۽ بي نفسي جو عالم ڏسو جو انهن ڀري مجلس جي سامهون اهو اعلان

فرمايو ته اُنهي ڏينهن فُلان نالي ۾ مون کان غلطي ٿي هئي ته هِن نوجوان شاگرد مون کي آگاھ ڪيو هو۔ (تاريخِ بغداد،ج۳،ص۴۰۰)

(7) نمايان حيثيت جا طلبگار نه بڻجو

مٺا مٺا اسلامي ڀائرو! عاجزي اپنائڻ جي لاءِ هن ڳالھ جو به خيال رکو ته جڏهن اسان پنهنجي ساٿين سان گڏ هجون يا ڪنهن محفل ۾، ڪڏهن به ان خواهش کي دل ۾ جڳهه نه ڏيون ته مون کي نمايان حيثيت ڏني وڃي، مٿانهين جڳهه تي ويهاريو وڃي، منهنجو آڌر ڀاءُ ڪيو وڃي۔ ها! ڪنهن از خود اوهان کي نمايان جڳهه تي ويهڻ جي درخواست ڪئي ته قبول ڪرڻ ۾ حرج ناهي۔ جيئن

اميرالمؤمنين حضرت سيدنا موليٰ عليُّ المرتضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم ڪِٿي تشريف فرما ٿيا ميزبان سندن لاءِ مسند (يعني تخت) کڻي آيو، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ان تي رونق افروز ٿيا ۽ فرمايائون: ڪو گڏھ ئي عزت جي ڳالهه قبول نه ڪندو۔

(فتاوٰي رضويه ،ج۲۳،ص۷۲۰)

(8) غريبن جي دعوت به قبول ڪريو

مٺا مٺا اسلامي ڀائرو! صرف اميرن سان تعلق وڌائڻ ۽ انهن وٽ دعوتن تي وڃڻ جا عادي نه بڻجو بلڪ پنهنجي ڄاڻ سُڃاڻ وارن ۾ غريبن کي به شامل ڪريو ۽ جڏهن اُهي اوهان کي دعوت ڏين ته قبول ڪريو۔

غريبن تي خصوصي شفقت

مُحَدِّثِ اعظم پاڪستان حضرت علامه مولانا محمد سردار احمد چشتي قادري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه عام طور تي مالدارن جي دعوت قبول نه ڪندا هئا پر اُن جي اُبتڙ جيڪڏهن ڪو غريب مسلمان دعوت جي پيشڪش ڪندو هو ته جيستائين ممڪن هوندو پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه قبول فرمائي وٺندا هئا ۽ انهن جي معمولي ۽ سادي کاڌي تي به اُنهن جي تعريف فرمائيندا هئا ته جيئن انهن کي ڪو دلي ڏک نه رسي۔ هڪ ڀيري هڪ غريب ماڻهو جي دعوت تي پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ان جي

گهر تشريف وٺي ويا، اُتي وڃي خبر پئي ته ان جو ڪکن ڪانن جو ٺهيل گهر ۽ بدبودار علائقي ۾ هو، پر پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن اُن جي دلجوئي جي لاءِ ان وٽ ماني کاڌي ۽ پنهنجي ڪنهن عمل سان ان غريب کي محسوس ٿيڻ نه ڏنو جو پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڌپ محسوس ڪري رهيا آهن، باوجود ان جي جو عام حالتن ۾ معمولي ڌپ به سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي لاءِ ناپسنديده هوندي هئي۔

(حيات محدّث اعظم،ص۱۸۴)

(9) ڪپڙن ۾ سادگي اختيار ڪريو

مٺا مٺا اسلامي ڀائرو! ڪپڙن ۾ سادگي اختيار ڪرڻ سان طبعيت ۾ عاجزي پيدا ٿيندي آهي، جڏهن ته عُمده لباس پائڻ سان تڪبُّر ۽ رِياڪاري پيدا ٿيڻ جو انديشو آهي۔ دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل 312 صفحن تي مشتمل ڪتاب، ”بهارِ شريعت“ حصو 16 صفحي 52 تي صدرُالشَّريعه، بدرُالطَّريقه حضرت علامه مولانا مفتي محمد امجد علي اعظمي لکن ٿا: ايترو لباس جنهن سان اوگھڙ ڍڪجي ۽ گرمي سردي جي تڪليف کان بچي فرض آهي ۽ ان کان وڌيڪ جنهن سان مقصد زينت هجي ۽ اهو جو جڏهن ته الله (عَزَّوَجَلَّ) ڏنو آهي ته ان جي نعمت جو اظهار ڪيو وڃي، اهو مستحب آهي۔ خاص موقعي تي مثلاً جُمعي يا عيد جي ڏينهن بهترين ڪپڙا پائڻ مُباح (جائز) آهن۔ ان قِسم جا ڪپڙا روز نه پائي ڇاڪاڻ ته ٿي سگهي ٿو ته آڪڙ اچڻ لڳي ۽ غريبن کي جِن وٽ اهڙا ڪپڙا ناهن حقارت ڀري نگاھ سان ڏسي، تنهنڪري ان کان بچڻ ئي گهرجي ۽ تڪبر جي طور تي جيڪو لباس هجي اهو منع ٿيل آهي، تڪبر آهي يا نه ان جي سُڃاڻپ ائين ڪري جو انهن ڪپڙن کي پائڻ کان پهرين پنهنجي جيڪا حالت هئي جيڪڏهن پائڻ کان پوءِ به اُها ئي حالت آهي ته خبر پئي ته انهن ڪپڙن سان تڪبر پيدا نه ٿيو، جيڪڏهن اها حالت هاڻي باقي نه رهي ته تڪبر اچي ويو، تنهنڪري اهڙي ڪپڙي کان بچي جو تڪبر ڏاڍي خراب صفت آهي۔

(ردالمحتار، ڪتاب الحظر و الإباحة، فصل في اللبس، ج۹، ص۵۷۹)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

(10)عاجزي جي فضيلتن جو مُطالعو ڪيو:

مٺا مٺا اسلامي ڀائرو! عاجزي اپنائڻ جي لاءِ مڪتبة المدينه جي هنن ڪتابن ۽ رسالن ’’تڪبُّر‘‘صفحو 70 کان 96، احياء العلوم جلد 3 صفحو 999 کان 1010، سيرتِ رسولِ عربي صفحو 340 کان 346، فيضانِ عائشه صديقه 560 کان 570، سيرتِ مصطفيٰ "606 تا 609 جو مطالعو ڪيو، اِنْ شَآءَ الله عَزَّوَجَلَّ ان جي برڪت سان عاجزي ۽ انڪساري اختيار ڪرڻ جو جذبو پيدا ٿيندو۔

بيان جو خلاصو

مٺا مٺا اسلامي ڀائرو! اَلْحَمْدُ لِلّٰه عَزَّوَجَلَّ اڄ جي بيان ۾ اسان اَمِيرُالمؤمنين حضرت سيدُنا عمر فاروقِ اعظم رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي سادگي ۽ عاجزي جا واقعا ٻڌا۔ تنهنڪري اسان کي به گھرجي ته اسان به غرور ۽ تڪبُّر کان بچي عاجزي ۽ اِنڪساري کي پنهنجي ڪردار ۽ عادت جو حصّو بڻايون ۽ بيان ۾ ذِڪر ڪيل انهن تمام طريقن تي عمل جي ڪوشش ڪيون، جن تي عمل ڪرڻ سان عاجزي ۽ انڪساري ڪرڻ جو جذبو پيدا ٿئي ٿو ۽ تڪبُّر سان نفرت ٿئي ٿي۔

اچو!شيخِ طريقت،اميرِاَهلسنّت دَامَتۡ بَرَكَاتُهُمُ الۡعَالِيَه جي رسالي ’’ 101 مَدَني گل ‘‘ مان ننهن ڪٽڻ جون سنّتون ۽ آداب ٻڌون ٿا۔ * جمعي جي ڏينهن ننهن ڪٽڻ مستحب آهن۔ ها جيڪڏهن وڌي ويا هجن ته جمعي جو انتظار نه ڪريو۔ (دُرِّمُختار،۹/۶۶۸) صدرُ الشَّريعه، بدرُ الطَّريقه مولانا امجد علي اعظمي فرمائن ٿا: منقول آهي: ”جيڪو جمعي جي ڏينهن ننهن ڪٽي الله تعاليٰ اُن کي ٻئي جمعي تائين مصيبتن کان محفوظ رکندو ۽ ٽي ڏينهن وڌيڪ يعني ڏهه ڏينهن تائين۔ هڪ روايت ۾ هي به آهي ته جيڪو جمعي جي ڏينهن ننهن ڪٽي ته رحمت نازل ٿيندي ۽ گناهه ڇڻندا۔ (دُرِّمُختار،رَدُّالْمُحتَار، ۹/۶۶۸، بهارِشريعت حصّه۱۶، ۲۲۵،۲۲۶) * هٿن جا ننهن ڪٽڻ جي منقول طريقي جو خلاصو پيش خدمت آهي: پهرين ساڄي هٿ جي شهادت واري آڱر کان شروع ڪري ترتيبوار چيچ سميت ننهن ڪٽيا وڃن پر آڱوٺو ڇڏيو وڃي، هاڻي کٻي هٿ جي چيچ کان شروع ڪري ترتيبوار آڱوٺي سميت ننهن ڪٽيو، پوءِ هاڻي آخر ۾ ساڄي هٿ جي آڱوٺي جو ننهن ڪٽيو وڃي۔ (دُرِّمُختار،۹/۶۷۰، اِحْياءُ الْعُلُوم، ۱/۱۹۳) * پيرن جي ننهن ڪٽڻ جي ڪا ترتيب منقول ناهي۔ بهتر اهو آهي ته ساڄي پير جي چيچ کان شروع ڪري ترتيبوار آڱوٺي سميت ننهن ڪٽيو پوءِ کاٻي پير جي آڱوٺي کان شروع ڪري چيچ سميت ننهن ڪٽيو۔ (دُرِّمُختار،۹/۶۷۰، اِحْياءُ الْعُلُوم، ۱/۱۹۳) * جنابت جي حالت (يعني غسل فرض ٿيڻ جي صورت) ۾ ننهن ڪٽڻ مڪروهه آهي ۔ (عالَمگيري،۵ /۳۵۸) * ڏند سان ننهن ڪٽڻ مڪروهه آهي ۽ ان سان برص (يعني ڪوڙهه) جي بيماري ٿيڻ جو انديشو آهي۔ (اَيضاً) * ننهن ڪٽڻ کان پوءِ انهن کي دفن ڪري ڇڏيو ۽ جيڪڏهن اڇلائي

ڇڏيو ته به حرج ناهي ۔ (اَيضاً) | ڪٽيل ننهن بيتُ الخلا يا غسل خاني ۾ اڇلائڻ مڪروهه آهن، جو ان سان بيماري پيدا ٿئي ٿي ۔ (اَيضاً) | اربع جي ڏينهن نَنهن نه ڪٽڻ گھرجن، ڇوته ڪوڙهه جي بيماري ٿيڻ جو انديشو آهي پر جيڪڏهن اوڻيتاليهه ڏينهن کان نه ڪٽيا هئا، اڄ اربع چاليهون ڏينهن آهي جيڪڏهن اڄ نه ڪٽيا ته چاليهه ڏينهن کان وڌيڪ ٿي ويندا ته اُن تي واجب ٿيندو ته اڄ جي ئي ڏينهن ڪٽي ان ڪري جو چاليهه ڏينهن کان وڌيڪ ننهن رکڻ ناجائز ۽ مڪروهه تحريمي آهي۔ (تفصيلي معلومات جي لاءِ فتاويٰ رضويه جلد22 صَفحَه 574کان685 جو مطالعو ڪيو) | ڊگها ننهن شيطان جي نشست گاهه آهن يعني انهن تي شيطان ويهندو آهي۔ ۔ (اِتحافُ السّادَة للزّبيدي،۲/۶۵۳)
