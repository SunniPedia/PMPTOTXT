هڪ حاضري رهجي وئي!

مٺا مٺا اسلامي ڀائرو! ماھـهِ شَوّالُ المُڪَرَّم پنهنجي رَحمتن ۽ بَرڪتن جي بهار لٽائيندو اختتام جي طرف وڌي رهيو آهي ۽ ان کان پوءِ ماھـِه ذُوالقَعدةُ الحرام جي آمد آمد آهي۔ هي اهو مُبارَڪ مهينو آهي، جنھن جي ٻين رات دُنياءِ سُنيت جا عظيم پيشوا، سُنين جا رَہبَر ۽ رَہنما، آسمانِ رَضويت جا چمڪندڙ ستارا، خليفہء اَعليٰ حضرت، مُصنّفِ بہارِشريعت، صَدرُالشَّريعہ، بدرُ الطّريقہ حضرتِ علامہ مفتي اَمجد علي اعظمي عَلَیۡهِ رَحْمَةُ الـلّٰـهِ الۡغَنِی جن هن فاني دنيا مان لا ڏاڻو فرمايو۔ يعني 2 ذُوالقَعدةُ الحرام تي سندن عرس مبارڪ آهي، اچو ته! انهيءَ نسبت سان صَدرُالشَّريعہ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مبارڪ زندگي جي مختلف گوشن جا پيارو ذڪر ٻُڌون ٿا۔ سڀ کان پهريان نهايت ايمان اَفروز حڪايت ٻُڌون ٿا، جيئن ته

حضرت صَدرُالشَّريعہ، مُفتي امجد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اجمير شريف ۾ هڪ ڀيري شديد بُخار ۾ مُبتَلا ٿي ويا، ايستائين جو غَشي طاري ٿي وئي۔ ٻنپہرن کان پهريان غَشي طاري ٿي ۽ عَصر تائين رهي۔ حافِظِ ملّت مولانا عبدُالعزيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه خدمت جي لاءِ حاضِر ٿيا، صدرُ الشَّريعہ، بَدرُ الطّريقہ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي جڏهن ہوش آيو تو سڀ کان پهريان هي پڇيائون: ڇا وقت ٿيو آهي؟ ظُہر جو وَقت آهي يا نه؟ حافِظِ ملّت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن عرض ڪيو ته هي ٽائم ٿيو آهي، هاڻي ظُہر جو وَقت ناهي۔ هي ٻڌي ڪري ايتري تڪليف پهتي جو اکين مان لڙڪ جاري ٿي ويا ۔حافِظ ملّت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پڇيو: ڇا توهان کي ڪٿي سُور آهي، ڪٿي تڪليف آهي؟ فرمايائون: (تمام وڏي) تڪليف آهي ته۔۔۔۔۔۔ ظُہر

جي نَماز قَضاء ٿي وئي۔'' حافِظ ملّت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن عرض ڪيو: توهان بيہوش هئا، بيہوشي جي حالت ۾ نَماز قَضا ٿيڻ تي قيامت ۾ پڇا ڳاڇا ناهي۔ فرمايائون: توهان پڇا ڳاڇا جي ڳالھ ڪري رهيا آهيو، مقرر وقت تي دَربارِ الٰهي عَزَّوَجَلَّ جي هڪ حاضري کان ته محروم رہيس۔

(تذڪرهء صدر الشريعه ،ص:۳۰)

باجماعت نماز جو جَذبو

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته صَدرُ الشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه نماز جا ڪيتري قَدَر پابند ۽ نماز سان ڪيڏي نه مَحَبَّت فرمائڻ وارا هئا جو شريعت جي طرف کان رُخصت جي باوُجُود به بيهوشي جي حالت ۾ نَماز ڇڏجڻ تي ايتري قَدر ڏک ۽ صدمو ٿيو جو مبارڪ اکين مان لڙڪ نڪري آيا. اللہ عَزَّوَجَلَّ انهن جي صدقي اسان کي به نمازن سان مَحَبَّت ۽ عِبادت جي لَذّت نصيب فرمائي۔ حضرتِ صدرُالشريعه، بَدرُ الطريقه مفتي محمد اَمجد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ان تي (به) تمام سختي سان پابند هئا جو مسجد ۾ حاضِر ٿي ڪري باجماعت نماز پڙهن۔ بلڪه جيڪڏهن ڪنهن وَجہ سان مُؤذِّن صاحِب مُقرَّروَقت تي نه پہچندا هئا ته خُود اَذان ڏيندا هئا۔ قديم (پراڻي) مڪانِ عاليشان کان مسجد بلڪل قريب هئي اتي ڪا به ڏکيائي نه هئي پر جڏهن نئين مڪانِ عاليشان قادِري منزل ۾ رِہائش پَذير ٿيا ته آس پاس ۾ ٻه مسجدون هيون۔ هڪ بازار جي مسجد ٻي وڏي ڀاءُ جي گهر جي ڀرسان جيڪا ’’نوّاڪي مسجد‘‘ جي نالي سان مشہور آهي۔ هي ٻئي مسجدون فاصلي تي هيون۔ ان وَقت بينائي (نظر) به ڪمزور ٿي چڪي هئي، بازار

واري مسجد ڪجهه ويجھي هئي پر رستي ۾ بي جوڙ ناليون هيون۔ تنهن لاءِ ’’نوّاڪي مسجد‘‘ نماز پڙهڻ ايندا هئا. هڪ ڀيري ايئن ٿيو جو صُبح جي نَمازجي لاءِ وڃي رهيا هئا، رستي ۾ هڪ کُوھ هو، اڃا ڪجھ اونداهي هئي ۽ رستو به خراب هو، بي خَيالي ۾ کُوھ تي چڙهي ويا، عنقريب کُوھ ۾ پير هلي وڃي ها۔ ايتري ۾ هڪ عورت اچي وئي ۽ زور سان رڙيون ڪرڻ لڳي! ”اڙي مولوي صاحِب کُوھ آهي رُڪجي وڃو! نه ته کوھ ۾ ڪِري پوندؤ!“ هي ٻڌي ڪري پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه قَدم روڪي ورتو ۽ پوءِ کُوھ تان لهي ڪري مسجد ويا. ان جي باوجود مسجد جي حاضري نه ڇڏي. (تذڪرهء صدرالشريعه،ص:۳۱)

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي نمازِ باجماعت اَدا ڪرڻ جو ڪيترو جَذبو حاصل هو جو وڏي عمر ۾ جڏهن ته بينائي (نظر) به ڪافي ڪمزور ٿي چڪي هئي، ايتري قَدَر مَشَقَّت جي باوُجود نمازِ باجماعت اَدا فرمائڻ جي لاءِ مسجد ۾ حاضر ٿيندا هئا. پر افسوس! ڪيترائي مُسلمان فقط سُستي، غَفلَت، معمولي پريشاني ۽ بيماري جي حالت ۾ نمازِ باجماعت ڇڏي ڏيندا آهن، تنهنڪري اسان به گھرجي ته پنج وقت نمازِ باجماعت اَدا ڪرڻ جي پابندي ڪريون جو مسجد ۾ نماز لاءِ وڃڻ واري جا گناھ معاف ٿيندا آهن، جيڪڏهن جماعت سان نماز پڙهي ته ان جي مَغفِرت ٿي ويندي آهي ۽ جيڪڏهن جماعت نڪري وڃي پوءِ به ان فضيلت جو حَقدار ٿيندو. جيئن ته

حضرتِ سَيّدُنا سَعيد بن مُسَيب رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته جڏهن هڪ اَنصاري صحابي رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي موت جو وَقت قريب آيو ته فرمائڻ لڳا ته: آئون توهان کي فقط حُصُولِ ثواب جي نيت سان هڪ حديث ٻڌايان

ٿو، مون رَسُولُ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي فرمائيندي ٻڌو ته ”توهان مان جيڪو ڪامل وُضُو ڪري ۽ مسجد جي طرف هلي، ته ساڄو قدم کڻڻ تي اللہ عَزَّوَجَلَّ ان جي لاءِ هڪ نيڪي لکندو آهي ۽ کاٻو قَدم رکڻ تي اللہ عَزَّوَجَلَّ ان جو هڪ گُناھ مٽائي ڇڏيندو آهي، هاڻي تنهنجي مرضي توهان مسجد جي ويجهو رہو يا پري. پوءِ جڏهن اهو مسجد ۾ ايندو آهي ۽ باجماعت نَماز اَدا ڪندو آهي ته ان جا گُناھ مُعاف ڪيا ويندا آهن، پوءِ جيڪڏهن اهو مسجد ۾ حاضِر ٿئي ۽ ان جون ڪجھ رَڪعتون نڪري چڪيون هجن ۽ ڪجھ باقي رهجي ويو هجن، اُن کي گهرجي ته جيڪي رَڪعتون اهو حاصل ڪري سگھي، اِمام سان گڏ پڙهي وٺي ۽ بقايا رڪعتون مڪمل ڪري وٺي تڏهن به ان جي مَغفِرت ڪئي ويندي آهي ۽ جيڪڏهن اهو مسجد ۾ (جماعت سان نماز پڙهڻ جي نيت سان) حاضِر ٿيو، پر جماعت ٿي چڪي هجي ته ان کي گهرجي ته پنهنجي نماز اَدا ڪري وٺي جو ان جي لاءِ به اها ئي بشارت آهي.“

(ابوداود، ڪتاب الصلٰوة ،باب ماجاء في الھدي في المشي الي الصلٰوة، رقم ۵۶۳ ، ج ۱ ،ص ۲۳۳)

ياد رهي! ته ڄاڻي واڻي نمازِ باجماعت جو وَقت گذاري ڇڏڻ واري کي هي فضيلت ته پري جي ڳالھ بلڪه جماعت ڇڏڻ جو گُناھ ملندو.

حضرتِ صدرُالشريعه، بَدرُ الطريقه مفتي محمد اَمجد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ۱۳۰۰ ھ مُطابق 1882ء ۾ مَشرقي يو پي (ہند) جي ڳوٺ مدينة العُلَماء گھوسي ۾ پيدا ٿيا. توهان جا والدِ ماجد حڪيم جمالُ الدين عَـلَيْهِ رَحْ ـ مَةُ اللهِ الۡمُبِین ۽ ڏاڏا خُدا بخش رَحْمَةُ اللهِ تَعَالٰی عَلَيْه طِب جي فن جا ماهر هئا. اِبتدائي تعليم توهان پنهنجي ڏاڏي حضرت مولانا خُدا بخش رَحْمَةُ اللهِ تَعَالٰی عَلَيْه

کان گھر ۾ حاصل ڪئي، پوءِ پنهنجي ڳوٺ ۾ ئي مدرَسہ ناصِرُالعُلُوم ۾ وڃي ڪري گوپال گنج جي مولوي الٰہي بخش صاحب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان ڪجھ تعليم حاصل ڪئي. پوءِ جونپور پہچي ڪري پنهنجي سؤٽ ڀاءُ ۽ اُستاذ مولانا محمد صديق رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان ڪجھ اَسباق پڙهيا، پوءِ حضرت علّامہ ہِدايتُ اللہ خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان علمِ دين جون برڪتون حاصل ڪيون ۽ اتان کان دَرسِ نِظامي مڪمل ڪئي. پوءِ دورہء حديث جي تڪميل پيلي ڀيت ۾ اُستاذُ المُحدِّثين حضرت مولانا وَصي احمد مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان ڪئي. حضرت مُحدِّثِ سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي هوشيار شاگرد جي اَعليٰ صَلاحيتن جو اِعتِراف هنن لفظن ۾ ڪيو: ”مون کان جيڪڏهن ڪنهن پڙهيو ته امجد علي پڙهيو۔“

(تذڪرهء صدر الشريعه،ص۵)

حُليو مُبارڪ:

پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه نِہايت ئي خُوبصورت گندمي رنگ، مَضْبُوط ۽ طاقتور جسم جا مالڪ هئا۔ چہرو مبارڪ پروقار ۽ بارونق، رَنگت گَندمي، ڪُشادا پيشاني سان گڏ وڏيون وڏيون چمڪدار اکيون، ڀرون گھاٽيون ۽ ڏاڙهي مبارڪ به ڪافي گھاٽي هئي. جسم مبارڪ ڀريل هو، ٿُلهو متارو ۽ سهڻي بيهڪ وارو (Smart) هو ۽ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو قَد مُبارڪ درميانو هو. (سوانحِ صدر الشريعه، ص۱۰،ملخصاً)

حافظي جي قوت جي مضبوطي:

سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو حافظو تمام مَضبوط هو. (طالب علمي جي دوران) حافِظي جي قُوت، شوق ۽ محنت ۽ ذِہانت جي ڪري سڀ طالبِ علمن

کان بہتر سمجھيا ويندا هئا۔ هڪ ڀيري ڪتاب ڏسڻ سان يا ٻڌڻ سان سالن تائين ايئن ياد رهندو هو، جيئن هاڻي هاڻي ياد ڪيو يا ٻڌو هجي. 3 ڀيرا ڪنهن عِبارَت کي پڙهي وٺندا هئا ته ياد ٿي ويندي هئي.

(تذڪرهءصدرالشريعه،ص۷)

تدريس جو آغاز:

صُوبي بِہار (ہند پٽنا) ۾ مدرسہ اَہلسنَّت هڪ اعلٰي درس گاھ هئي. ان مدرَسي جي مُتَولي قاضي عبدُالوحيد مرحوم جي دَرخواست تي حضرت مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن مدرسه اَھلسنَّت (پٽنا) جي صَدرِ مُدَرِّس جي لاءِ صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو انتخاب فرمايو۔ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اُستاد محترم جي دُعائن جي ڇانَو ۾’’پٽنا‘‘ پهتا ۽ پھرين ئي سبق ۾ علم جا اهڙا دريا وهايا جو عُلَماء وطَلَبه واھ واھ ڪرڻ لڳا۔ (خليفه اَعليٰ حضرت ،حضرت مولانا) قاضي عبدُالوحيد (عظيم آبادي) رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جيڪي پاڻ به وڏا عالِم هئا، (اُنھن) صَدرُ الشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي علمي وَجاھت ۽ انتِظامي صَلاحيت کان مُتَاثر ٿي ڪري مدرَسي جا تعليمي اُمُور سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي سِپرد ڪري ڇڏيا. (تذڪرهء صدرالشريعه،ص۷،ملتقطاً)

اعليٰ حضرت جي زيارت:

ڪجھ عرصي کان بعد قاضي عبدُالوحيد رَحْمَةُ اللهِ تَعَالٰی عَلَيْه بانيِ مدرسه اَھلسنَّت (پٽنا) شديد بيمار ٿي ويا ۔ قاضي صاحب هڪ نِھايت دين دار ۽ دين پروَر شخص هئا، پاڻ پوري زندگي خِدمتِ دين ئي کي پنهنجو مقصد بڻايو. سندن پرهيزگاري ۽ مدني سوچ جي ئي ڪَشِش هئي جو منهنجا آقا اَعليٰ حضرت، اِمامِ اَھلسُنَّت، مُجَدّدِ دين ومِلَّت مولانا شاھ

امام اَحمد رضا خان عَـلَيْهِ رَحْمَةُ الـرَّحْمٰن ۽ حضرت قبله مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جهڙا مصروف بُزُرگانِ دين قاضي صاحِب جي عيادت جي لاءِ روهيل کنڊ کان پٽنا تشريف وٺي آيا. ان موقعي تي حضرت صَدرُالشريعه، بَدرُالطّريقه مفتي محمداَمجد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پھرين دفعي منهنجي آقا اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي زيارت ڪئي۔ اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي شخصيت ۾ اهڙي ڪَشِش هئي جو بي اِختيار صَدرُالشريعه، بَدرُ الطريقه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي دل سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي طرف مائل ٿي وئي ۽ توهان اُستادِ محترم حضرت سيدُنا مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مشوري سان سلسله عاليه قادِريه ۾ اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان بَيعَت ٿي ويا. منهنجا آقا اَعليٰ حضرت ۽ مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيۡھُمَا جي مَوجُودَگي ۾ ئي قاضي صاحب جن وَفات ڪئي۔ اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن نمازِ جنازه پڙهائي ۽ مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن قَبر ۾ لاٿو.

(تذڪرهء صدرالشريعه،ص۹،ملتقطاً)

علمِ طِب ۾ مَھارت:

قاضي صاحب مَرحُوم جي اِنتقال کان پوءِ مدرسو اهڙن ماڻهن جي هٿن ۾ پهتو، جن جو علمِ دين سان بلڪل به تَعلُّق نه هو۔ اندازي سان ايئن مَعلوم ٿو ته خِدمتِ دين جيڪو اصلي مقصد آهي، هاڻي هتي ممڪن ناهي، تنهنڪري پهرين رَمَضانُ المبارڪ 1326ھ ۾ وطن واپس اچي ويا ۽ مدرسي ۾ اِستِعيفا موڪلي ڇڏي۔ ڇو جو توهان جو خانداني پيشو حڪمت هو، تنهنڪري والدصاحب جي مشوري سان ان فن کي حاصل ڪرڻ جي لاءِ حڪيم عبدُالوَلي صاحب وٽ لکنؤ هليا ويا. ٻن سالن ۾ مڪمل فن حاصل ڪرڻ بعد وطن اچي ويا ۽ مَطب

(يعني ڪلينڪ) شُروع ڪري ڇڏي۔ خانداني پيشو ۽ خُداداد قابليت جي بنياد تي دواخانو نھايت ڪاميابي سان هلڻ لڳو. (سوانحِ صدرالشريعه،ص۳۶) (مَعاش جي سِلسلي کان مُطمئن ٿي ڪري) 1329ھ ۾ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي اُستاد حضرت مُحدِّثِ سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ۽ مُرشدِ بَرحق اَعليٰ حضرت امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي زيارت جي لاءِ سفر جو ارادو ڪيو۔ حضرت مُحدِّثِ سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي خدمت ۾ جڏهن پهتا ۽ انهن کي هي معلوم ٿيو ته انهن جو لائق فائق ۽ محنتي شاگرد تدريس (پڙهائڻ) ڇڏي ڪري دواخاني ۾ مشغول ٿي ويو آهي ته بيحَد غمگين ٿيا۔ (ملخص از سيرتِ صدر الشريعه،ص۳۷) (مُحدِّثِ سُورتي سان ملڻ کان بعد) جيئن ته صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو اِرادو بريلي شريف حاضِر ٿيڻ جو به هو، بريلي شريف وڃڻ وَقت مُحدِّث سُورتي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن هڪ خَط اِن مضمون جو اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي خِدمت ۾ تحرير فرمائي ڇڏيو هو ته ’’جهڙي طرح ممڪن ٿئي توهان اِن (يعني حضرت صَدرُ الشريعه، بَدرُالطّريقه مفتي محمد اَمجد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه) کي خِدمتِ دين ۽ علمِ دين جي طرف مُتوجِّه ڪريو۔“ (تذڪرهء صدر الشريعه،ص۱۲) صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه خُود فرمائن ٿا: آئون جڏهن اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي بارگاھ ۾ حاضِر ٿيس ته پڇيائون: مولانا ڇا ڪندا آهيو؟ مون عرض ڪيو: طبيب آهيان۔ اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن فرمايو: ’’مَطَب به سٺو ڪم آهي، اَلْعِلْمُ عِلْمَانِ عِلْمُ الْاَدْیَانِ وَعِلْمُ الْاَبْدَان (يعني علم ٻه آهن: علمِ دين ۽ علمِ طِب) ، ته مَطَب ڪرڻ ۾ هي خَرابي آهي ته صُبح صُبح قارُورھ (يعني پيشاب) ڏسڻو پوندو آهي‘‘ اِن ارشاد کان بعد مون کي قارُورھ (پيشاب) ڏسڻ کان اِنتھائي نفرت ٿي وئي ۽ اهو اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو ڪشف هو ڇو جو آئون مرض

سڃاڻڻ ۾ قارُورھ (يعني پيشاب) ئي کان مدد وٺندو هئس ۽ هي (اعليٰ حضرت جي) ڪرامت هئي جو قارُورھ بيني يعني مريضن جي پيشاب ڏسڻ کان نفرت ٿي وئي. (تذڪرهء صدر الشريعه،ص۱۳،ملتقطاً) (ان مُلاقات ۾) اعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه نھايت شفقت ۽ ڪرم سان پيش آيا ۽ اُنھن صَدرُ الشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي پاڻ وٽ رهڻ جو حڪم فرمايو ۽ دل لڳائڻ جي لاءِ ڪجھ تحريري ڪم وغيره سِپرد فرمائي ڇڏيو، صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو تقريباً ٻه مهينا بريلي شريف ۾ قيام رھيو۔ رمضان جي آمد جي پيشِ نظر صدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن گھر وڃڻ جي اجازت طلب ڪئي ته منهنجي آقا اعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن ارشاد فرمايو:”وڃو! پر جڏهن به گُهرايان ته فوراً اچي وڃجو.“ (ملخص از تذڪرهء صدر الشريعه،ص۱۲) گھر وڃڻ جي ڪجھ مهينن کان بعد بريلي شريف کان خَط پھتو ته توهان فوراً هليا اچو. بهرحال صَدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ٻيهر بريلي شريف حاضِر ٿي ويا ۽ اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن بريلي شريف ۾ توهان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مُستقل قيام جو انتِظام فرمائي ڇڏيو۔ اهڙي طرح صدرُالشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن 18سال منهنجا آقاءِ نعمت اعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي صحبت بابرڪت ۾ گذاريا۔ (تذڪرهء صدر الشريعه، ص۱۴، ملتقطاً) بريلي شريف ۾ ٻه مستقل ڪم هئا، هڪ مدرَسو (منظرُالاسلام) ۾ تدريس (پڙهائڻ) ، ٻيو پريس جو ڪم، يعني ڪاپين ۽ پرُوفن جي تصحيح، ڪتابن جي روانگي، خطن جا جواب، آمد ۽ خرچ جا حساب، هي سڀ ڪم پاڻ انجام ڏيندا هئا. انهن ڪمن کان علاوه اعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا بعض مُسَوَّدا (اهو ڪم جيڪو سَرسَري طور تي) شروع کان صاف لکڻ، فتوون جي نقل ۽ انهن جي خدمت ۾ رهي ڪري فتوٰي لکڻ هي ڪم به مستقل طور تي

انجام ڏيندا هئا۔ پوءِ شھر ۽ بيرونِ شھرجي اڪثر تبليغِ دين جي جلسن ۾ به شرڪت فرمائيندا هئا. (تذڪرهء صدر الشريعه،ص۱۴، ملتقطاً)

مون کي سون سان تورين ها.“ (تذڪرهء صدرالشريعه، ص۴۵) ۽ يقينًا هي سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو پاڪ ۽ ھند جي مُسلمانن تي تمام وڏو احسان آهي ته اُنھن وڏن عربي ڪتابن ۾ پکڙيل فقهي مَسئلن کيِ تحريري گلدستي جي صورت ۾ هڪ مَقام تي جمع ڪري ڇڏيو. ان ۾ بيشُمار مَسئلا اهڙا آهن جنهن جو سکڻ ھر اسلامي ڀاءُ ۽ اسلامي ڀيڻ تي فرضِ عَين آهي. ان جي تصنيف جا اَسباب ذِڪر ڪندي صَدرُ الشريعه رَحْمَةُ اللهِ تَعَالٰی عَلَيْه لکن ٿا:’’اُردو زبان ۾ هن وقت تائين ڪا اهڙو ڪتاب تصنيف ناهي ٿيو، جيڪي صحيح مَسئلن تي مُشتمل ھجي ۽ ضَروريات جي لاءِ ڪافي ۽ مڪمل ھجي.‘‘ (تذڪرهء صدر الشريعه، ص۴۵، ملتقطاً) انهيءَ لاءِ پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هي عظيم الشَّان ڪتاب تحرير فرمايو. جيئن ته خُود اِرشاد فرمائن ٿا:’’ان ڪتاب ۾ وس آهر اها ڪوشش ھوندي ته عِبارَت تمام آسان ھجي جو سمجھڻ ۾ مشڪل نه ٿئي ۽ گھٽ علم وارا ۽ عورَتون ۽ ٻار به ان مان فائدو حاصل ڪري سگھن“۔ (تذڪرهء صدر الشريعه،ص۴۶) اِن تصنيف جي هڪ خوبي اها به آهي ته اَعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن بھارِشريعت جي ٻين، ٽين ۽ چوٿين حصي جو مُطالَعَو فرمائي ڪري ان تي تقريظ به لِکي آهي. هن ڪتاب ۾ جٿي ڪٿي فرض عُلُوم سان گڏوگڏ، قُرآني آيتن جا نُور، مبارڪ حديثن جي بهار ۽ ھر حصي ۾ پکڙيل بي شُمار خُوشبودار شَرعي مَسئلا پنهنجي خوشبو مهڪائي رهيا آهن. يقينًا ان ڪتاب کي پڙهڻ سان نه صِرف دل ۽ دِماغ روشن ٿيندا آهن بلڪه ڪيترن ئي مُعامَلن ۾ شَرعي رَھنمائي به نصيب ٿيندي آهي. بھارِ شريعت جي ان عظيم عِلمي ذَخيرن کي وڌ کان وڌ فائديمند بنائڻ جي لاءِ دعوتِ اسلامي جي مجلس اَلمَدينة الْعِلمية جي مدني عُلماء

ترجمه ڪنزالايمان جو تحفو:

صَدرُالشريعه، بَدرُ الطّريقه مفتي اَمجد علي اعظمي عَلَیۡهِ رَحْمَةُ الـلّٰـهِ الۡغَنِی جن بريلي شريف ۾ قيام جي دوران صحيح ۽ غلطين کان پاڪ اَحاديثِ نَبويّه ۽ اَئمّه جي اقوالن جي مُطابق هڪ ترجمي جي ضَرورت محسوس ڪندي قرآنِ پاڪ جي ترجمي جي لاءِ اَعليٰ حضرت عَـلَيْهِ رَحْـمَةُ ربّ العزّت جي بارگاھِ عظمت ۾ دَرخواست پيش ڪئي ته اِرشاد فرمايو: ’’هي ته تمام ضَروري آهي پر ڇپجڻ جي ڇا صُورت ھوندي؟ ان جي طباعَت جو ڪير اِھتِمام ڪندو؟ باوُضو ڪاپين کي لکڻ، با وُضو ڪاپين ۽ اکرن کي صحيح ڪرڻ ۽ تصحيح به اهڙي ھجي جو اِعراب نُقطا يا علامتن جي به غلطي نه رهي، پوءِ هي سڀ شيون ٿي وڃڻ کان بعد سڀ کان وڏي مُشڪل ته هي آهي ته پريس ۾ ھر وَقت باوُضُو رهي، بِغير وُضو نه پٿر کي هٿ لڳائي ۽ نه ڪَٽي، پٿر ڪٽڻ ۾ به اِحتياط ڪئي وڃي ۽ ڇپجڻ ۾ جيڪي جوڙيون نڪرنديون آهن ان کي به وڏي احتياط سان رکيو وڃي.

پاڻ عرض ڪيائون:” اِنْ شَآءَ الله عَزَّوَجَلَّ جيڪي ڳالهيون ضروري آهن، ان کي پُوري ڪرڻ جي ڪوشش ڪئي ويندي، بِالفرض مڃي وٺجي ته اسان کان ايئن نه ٿي سگھيو ته جڏهن هڪ شيءِ موجود هجي ته ٿي سگھي ٿو ته آئنده ڪو شخص ان جي طباعت (ڇپائي) ڪرڻ جو انتِظام ڪري ۽ مخلوقِ خُدا کي فائدو پھچائڻ جي ڪوشش ڪري ۽ جيڪڏهن هن وَقت هي ڪم نه ٿي سگھيو تو آئنده ان جي نه ٿيڻ جو وڏو افسوس ٿيندو۔‘‘ سندن جي عرض ڪرڻ کان پوءِ ترجمي جو ڪم شُروع ڪيو ويو، بِحَمدِاللّٰھ عَزَّوَجَلَّ سندن رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي نيڪ ڪوششن جي خواھش موجب ڪاميابي حاصل ٿي ۽ اڄ مسلمانن جو وڏو تعداد مُجدِّد اعظم، اِمامِ اَھلسُنَّت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي لکيل قرآنِ پاڪ جي صحيح ترجمي ’’ترجمو ڪنزالايمان‘‘ کان فائدو حاصل ڪري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه (يعني صدرالشريعه) جا احسان مند آهن ۽ اِنْ شَآءَ الله عَزَّوَجَلَّ هي سلسلو قيامت تائين جاري رهندو.

(تذڪرهء صدرالشريعه ،ص:۱۷)

مٺا مٺا اسلامي ڀائرو! اسان کي به نيڪي جي ڪمن ۾ حصو وٺڻ، نيڪيون ڪرڻ گُناھن کان بچڻ، ٻين کي بچائڻ ۽ نيڪي جي دعوت عام ڪرڻ جي لاءِ ذيلي حلقي جي 12 مدني ڪمن ۾ اڳتي وڌي ڪري حصو وٺڻ گھرجي. ذيلي حلقي جي 12 مدني ڪمن مان هڪ مدني ڪم ھفتيوار”مدني مُذاڪرو“ به آهي. ماڻهن ۾ عِلم و حڪمت جا مدني گُل لٽائڻ، انهن جي تربيت ڪرڻ ۽ علمِ دين سيکارڻ بُزرگانِ دين رَحِمَهُمُ الـلّٰـهُ الْـمُبین جو معمول رھيو آهي۔ بُزرگانِ دين جو هڪ تعداد آهي جيڪو علمِ دين جي روشني ۽ ماڻهن کي ھدايت جو سامان فراھم ڪرڻ لاءِ وقتًا فوقتًا مختلف طريقن سان علمِ دين سيکارڻ جي ترڪيب ڪندا آهن. عُلماءِ دين ۽ اَولياءِ ڪاملين جي صُحبت جو فيض، انهن جي بيان جي چاشني ۽ انهن جي ڪردار جي دِلڪشي ھزارين ماڻهن جي لاءِ راھـِه ھدايت جو ذريعو بڻجندي آهي. انهيءَ سبب حضرت سَيّدُنا حڪيم لقمان رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن پنهنجي صاحبزادي کي وَصيت ڪندي ارشاد فرمايو: اي منهنجا پُٽ! جڏهن تون ڪنهن اهڙي گروھ کي ڏسين جو اهي اللھ عَزَّوَجَلَّ جو ذِڪر ڪندا هجن ته انهن جي صُحبت اِختيار ڪر، ان لاءِ جو جيڪڏهن تون عالِم به هوندين ته تنهنجو علم وڌيڪ فائدو ڏيندو ۽ جيڪڏهن تون جاھل هوندين ته اهي توکي علمِ دين سيکاري ڇڏيندا ۽ جڏهن اللھ عَزَّوَجَلَّ جي رحمتن جو وسڪارو انهن تي وسندو ته توکي به انهن جي صُحبت ۾ ھئڻ جي سبب ان مان حصو ملندو ۽ اهڙي مجلس ۾ نه ويهجان جتي اللھ عَزَّوَجَلَّ جو ذِڪر نه هجي، ان لاءِ جو جيڪڏهن تون عالِم به هوندين ته تنهنجو علم فائدو نه ڏيندو ۽ جيڪڏهن تون جاھل هوندين ته انهن جي

صُحبت تنهنجي جھالت ۽ گمراهي کي وڌيڪ وڌائي ڇڏيندي. جيڪڏهن انهن تي غَضبِ الٰهي ٿيو ته انهن سان گڏ هئڻ جي سبب تون به ڦاسي ويندي. (روح البيان:۱/۲۵۷) هڪ ٻئي مقام تي پنهنجي پُٽ کي وَصِيت ڪندي اِرشاد فرمايو: عُلما جي مجلس کي لازِم پڪڙ ۽ ڏاهن ماڻهن جي گفتگو غور سان ٻڌندو ڪر! ڇو جو اللھ عَزَّوَجَلَّ مُرده دل کي حڪمت جي نُور سان ايئن ئي زِنده فرمائي ڇڏيندو آهي جيئن ڪنهن زمين کي مِينهن جي وسڪاري سان زِنده فرمائيندو آهي۔ (السراج المنير:۳/۱۸۵)

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته! رَئيسُ الحڪماء حضرت سَيّدُنا حڪيم لقمان رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن ڪهڙا علم ۽ حڪمت ڀريا مدني گُل عطا فرمايا ته عُلماء جي صُحبت اختيار ڪرڻ گھرجي، انهيءَ ۾ فائدو آهي. عالِم ھجي تڏهن به عُلماء جي مجلس سان جڙيل رهي ۽ جاھل ھجي تڏهن ته ان تي لازِم آهي ته اهو عُلماء جي صحبت اختيار ڪري. علم ۽ حڪمت ڀري ڳالهين سان جتي فڪر ۽ نظر جي روشني حاصل ٿيندي آهي، اتي علمِ دين سکڻ جو ثواب به ھٿ ايندو آهي. اَلْحَمْدُلِلّٰھ عَزَّوَجَلَّ شيخ طريقت، اميرِ اَھلسُنَّت، بانيءِ دعوتِ اسلامي حضرت علامه مولانا ابُو بلال محمد الياس عطار قادري رضوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه به اَسلاف جي طريقي تي عمل ڪندي آسان اَنداز ۾ قرآن ۽ حديث جي روشني ۾ نه صِرف شَرعي سوالن جا جواب ارشاد فرمائيندا آهن بلڪه گھڻو ڪري قيمتي طِبّي معلومات کان آگاهي عطا فرمائيندا آهن، مشڪلاتن جا حل ارشاد فرمائيندا آهن، مختلف علمي مدني گُلن جي ذريعي انهن کي وڌيڪ مُنفرِد بڻائيندا آهن ۽ ڏسڻ، ٻڌڻ وارن جي دِلين ۾ خوفِ خدا ۽ عشقِ مصطفٰي جي شمع به (روشن) فرمائيندا آهن. شيخ طريقت،

اَميرِ اَھلسنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي ھر ڪوشش جو مقصد راھ راست تا ڀٽڪيل ماڻهن کي راھ تي آڻڻ ۽ انهن کي سنتن جو پابند بنائڻ آهي. اَلۡحَمۡدُلِلّٰھ عَلٰي اِحۡسَانِھ ٖ بي شمار اسلامي ڀائر ۽ اسلامي ڀينرون توهان دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي ان مدني مُذاڪرن جي ذريعي گُناھن کان تائب ٿي ڪري اسلامي تعليمات کي اپنائڻ وارا ٿي چُڪا آهن. اچو ته! توهان جي ذَوق اَفزائي جي لاءِ هڪ مدني بھار ٻڌايان ٿو:

فيشن پرست ڪيئن سڌريو!

لَيّه شھر (صوبو پنجاب، پاڪستان) جي مُقيم هڪ نوجوان جو بيان آهي ته آئون سنتن کان پَري فيشن جي نَشي ۾ چُور هئس، نوان نوان عجيب و غريب انداز وارا ڪپڙا پائڻ، فُضُول ڪمن ۾ پنهنجا قيمتي لمحا ضايع ڪرڻ منهنجو معمول هو. ذِڪرِ الٰھي کان بلڪل غافل ٿي چڪو هئس، نيڪين ڀري زندگي گذارڻ جو ذِھن ڪجھ هن طرح بڻيو جو هڪ ڀيرو’’مدني مُذاڪرو‘‘ٻڌڻ جي سَعادت حصي ۾ ڇا آئي، منهنجي زندگي جو رُخ ئي بدلجي ويو. اميرِاَھلسنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي آسان اَنداز ۾ بيان ٿيڻ واري بي شمار مَعلُومات جو’’اَنمول خزانو‘‘ لُٽڻ جو سُنھري موقعو مليو، خوفِ خدا ۽ عشقِ مُصطفٰي جي روشني سان منهنجي ڪاري دل روشن ٿي وئي، گذريل زندگي تي شرمندگي ٿيڻ لڳي، مون پنهنجي بقايا زندگي کي غنيمت ڄاڻيندي فيشن جي نحوست کان جان ڇڏائي ۽ سنتن تي عمل ڪرڻ ۽ نمازن جي پابندي جو پڪو ارادو ڪري ورتو، جلد ئي مٿي تي سائو سائو عمامو شريف سجائي ورتو، ڏاڙهي شريف سان چھرو پر نُور ڪري ورتو ۽ نيڪين تي

اِستقامت حاصل ڪرڻ جي لاءِ دعوتِ اسلامي جي مُشڪبار مدني ماحول سان وابسته ٿي ويس. نيڪي جي دعوت جون ڌومون مَچائڻ جي جذبي جي تحت ھر مھيني 3 ڏينهن جي مدني قافلي ۾ سفر منهنجو معمول بڻجي ويو. اَلْحَمْدُ لِلّٰھ عَزَّوَجَلَّ تحرير لکڻ وقت مبلّغِ دعوت اسلامي جي حيثيت سان سنتن جي خدمت ڪري رهيو آهيان.

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

14،15،16آگسٽ مدني قافلا:

مٺا مٺا اسلامي ڀائرو! سڀاڻي 14 آگسٽ آهي، ان ڏينهن کي اللہ عَزَّوَجَلَّ ۽ رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي فضل ۽ ڪرم سان اسان کي نعمتِ عظيم ”يعني وطنِ عزيز ملڪ پاڪستان" نصيب ٿيو، بدقسمتي سان اڄ اِن نعمت جو شڪر ڪرڻ جي بجاءِ نه ڄاڻ ڪهڙن ڪهڙن گناہن جا بازار گرم ڪري ربِّ غفار جي ناراضي جو سامان ڪيو ويندو آهي، اللہ عَزَّوَجَلَّ سلامت رکي، اميرِ اہلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه کي، جن قدم قدم تي هن اُمّت جي خيرخواهي ۽ رہنمائي فرمائي آهي، توهان ان موقعي تي به عاشقانِ رسول کي گناہن کان بچڻ ۽ اللہ عَزَّوَجَلَّ جو شڪر بجا آڻيندي نيڪين ۾ گذارڻ جو مدني ذہن عطا فرمايو آهي.

مٺا مٺا اسلامي ڀائرو! اميرِ اہلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي ترغيب تي 14 آگسٽ تي ملڻ واري نعمت جي شڪراني ۽ نيڪي جي دعوت جون ڌومون مچائڻ جي لاءِ سڀاڻي اِنْ شَآءَ الله عَزَّوَجَلَّ! ڪيترائي عاشقانِ رسول 3 ڏينهن جي مدني قافلي ۾ سفر جي سعادت حاصل ڪندا، توهان به ہمّت ڪريو ۽ پاڻ کي مدني قافلي ۾ سفر جي پيش ڪريو.

جشنِ آزادي ۾ شڪراني جا نفل پڙهو ۽ شڪراني جي طور تي اللہ جي رضا جي لاءِ روزو رکو.

شيخ طريقت، اميرِ اَھلسُنَّت، باني دعوتِ اسلامي حضرت علامه مولانا ابُوبلال محمد الياس عطار قادري رضوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه فرمائن ٿا: جيڪي به اسلام سان محبت ڪندا آهن، پاڪستان سان محبت ڪندا آهن، اُنهن جي خدمت ۾ عرض آهي ته توهان دعوتِ اسلامي وارن سان گڏ مدني قافلي ۽ مدني چينل تي وقت گذارين، سحري ڪن ۽ روزو رکن.

مزيد فرمائن ٿا جنهن جو خلاصو هي آهي: ڪجھ ماڻهو جشنِ آزادي ملهائي ڪري پنهنجي لاءِ بربادي جو سامان ڪندا آهن: مثلاً ہوائي فائرنگ، ميوزيڪل پروگرام ڪيا ويندا آهن، قانون جي خلاف ورزي ٿيندي آهي، وڌيڪ شراب پيتو ويندو آهي، بيهوده ڊانس ڪيا ويندا آهن.

ويهڻ جون ڪجهه سنتون ۽ آداب

مٺامٺا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو۔ (مِشڪاة المَصابيح ، ج۱ ص۵۵ حديث ۱۷۵ دارالکتب العلمية بيروت)

سینہ تری سنّت کا مدینہ بنے آقا جنت میں پڑوسی مجھے تم اپنا بنانا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

اچو ته ويهڻ جون سنتون ۽ آداب سماعت ڪريون!

* … ويهڻ جي جڳھ زمين تي رکو ۽ گوڏن کي اُڀو ڪري ٻنهي هٿن سان گھيري وٺو ۽ هڪ هٿ سان ٻئي کي پڪڙي وٺو، اهڙي طرح ويهڻ سُنت آهي (پر ان دوران گوڏن تي ڪا چادر وغيرہ ويڙهي وٺڻ بہتر آهي) (مراٰة المناجيح، ج۶، ص۳۷۸ ملخصاً) *… چارزانو (يعني پلٿي هڻي) ويهڻ به نبيِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان ثابت آهي۔ *… جتي ڪجھ اُس ۽ ڪجھ ڇانَو هجي اتي نه ويهو۔ اللہ عَزَّوَجَلَّ جي مَحبوب، داناءِ غُيوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جڏهن توهان مان ڪو ڇانو ۾ هجي ۽ ان تان ڇانو رخصت ٿي وڃي ۽ اهو ڪجھ اُس ڪجھ ڇانَو ۾ رهجي وڃي ته ان کي گھرجي ته اتان کان اُٿي وڃي۔ (سنن ابي داؤد، ڪتاب الادب، باب في الجلوس بين الظل و الشمس، الحديث۴۸۲۱،ج۴،ص۳۳۸) *… قبلي رُخ ٿي ڪري ويهو۔ (رسائل عطاريه، حصہ۲،ص۲۲۹) *… اعليٰ حضرت، امامِ اہلسنت، مَولانا شاہ امام احمد رضا خان عَـلَيْهِ رَحْمَةُ الـرَّحْمٰن لکن ٿا: پير ۽ استاد جي نشست (مَسند) تي انهن جي غير موجودگي ۾ به نه ويهي۔ (فتاوٰي رضويہ، ج۲۴، ص۳۶۹/۴۲۴) *… جڏهن ڪنهن اِجتِماع يا مجلِس ۾ اچڻ ٿئي ته ماڻهن کي اورانگھي ڪري اڳيان نه وڃو جتي جڳھ ملي اتي ويهي رهو۔ *… جڏهن ويهو ته جُوتا لاهي وٺو توهان جي پَيرن کي آرام ملندو۔ (الجامع الصغير، الحديث۵۵۴، ص۴۰) *… مجلس کان فارغ ٿي ڪري هي دعا ٽي ڀيرا پڙهي وٺو ته گناہ معاف ٿي ويندا۔ ۽ جيڪو اسلامي ڀاءُ مجلسِ خير ۽ مجلسِ ذڪر ۾ پڙهي ته ان جي لاءِ ان خير تي مُہر لڳائي ويندي۔ اُها دُعا هي آهي:

سُبْحَانَکَ اللّٰھُمَّ وَبِحَمْدِکَ لَاۤ اِلٰہَ اِلَّاۤ اَنْتَ اَسْتَغْفِرُ کَ وَاَتُوْبُ اِلَیْکَ ترجمو: تنهنجي ذات پاڪ آهي ۽ يا اللہ! تنهنجي لاءِ ئي تمام خوبيون آهن، تو کان سوا ڪو معبود ناهي، توکان بخشش گھُران ٿو ۽ تنهنجي طرف توبه ڪريان ٿو۔

(سنن ابي داؤد، ڪتاب الادب، باب في ڪفار ۃ المجلس، الحديث ۴۸۵۷، ج۴، ص۳۴۷)

*… ڪنهن عالِم با عمل يا متقي شَخص يا سيّد صاحب يا والدين جي اچڻ تي تعظيمًا اُٿي بيهڻ ثواب آهي۔ حڪيم الامّت مفتي احمد يار خان نعيمي عَلَیۡهِ رَحْمَةُ الـلّٰـهِ الۡغَنِی لکن ٿا: بزرگن جي اچڻ تي هي ٻه ڪم يعني تعظيمي قيام (اٿي بيهڻ) ۽ استقبال جائز بلڪه سنتِ صحابه آهي۔

(مراٰة المناجيح، ج۶، ص۳۷۰، جملہ منقول از سنت یں اور آداب ، ص۹۹، بتغير)

هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ”بهار شريعت“ حصو 16 ۽ 120 صفحن جو ڪتاب ”سنتين اور آداب“ هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي.

عاشقانِ رسول، آئیں سنت کے پھول

دینے لینے چلیں، قافلے میں چلو

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ درود شريف

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ مان گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زيارت ڪندو ۽ قبر ۾ داخل ٿيندي وقت اهو به ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن ، ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سيدُنا انس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪو شَخص اهو دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو اهو دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([1])

﴿ 4 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سيدُنا ابنِ عباس رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: انهي درود شريف کي پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تي نيڪيون لکندا آهن. ([2])

﴿5﴾ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَیۡهِ رَحْمَةُ الـلّٰـهِ الۡھَادِی ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿ 6 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي ۽ صِدّيقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي وچ ۾ ويهاريو، انهي ڪري صَحاب َ ه ڪرام رَضِیَ اللهُ تَعَالٰی عَنْهم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: اهو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته ائين پڙهندو آهي. ([1])

﴿ 7 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَن ْ زِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو فرمان معظم آهي: جيڪو شخص هيئن درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

﴿2﴾ هر رات عبادت ۾ گذارڻ جو آسان نسخو

غرائب القرآن صفحه 187 تي هڪ روايت نقل ڪئي وئي آهي ته جيڪو شخص رات ۾ ٽي ڀيرا هي دعا پڙهندو ته ان ڄڻ شبِ قدر کي حاصل ڪري ورتو لهٰذا هر رات هي دعا پڙهڻ گهرجي. دعا هي آهي:

لَاۤ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

”يعني خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي.“

(فيضانِ سنّت ج اوّل، ص 1163، 1164)
