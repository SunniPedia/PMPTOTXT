مٺا مٺا اسلامي ڀائرو! ياد رکو ته ڪو به مُلڪ خواه مَعَاشِي طور تي ڪيترو ئي تَرَقّي يافته ڇو نه هجي، پر عام طور تي اُن ۾ ماڻهن جو هڪ اهڙو طبقو به هوندو آهي جيڪو مُختلف سببن جي ڪري غُربت ۽ اَفْلَاس جو شِڪار هوندو آهي ۔ اهڙن ماڻهن جي ڪفَالَت جي ذِمّيداري، الله عَزَّوَجَلَّ صاحبِ حَيثِيت ماڻهن جي حوالي ڪئي آهي۔ جيئن الله عَزَّوَجَلَّ مالدار مسلمانن تي زڪوٰة فرض ڪئي، ته جيئن اُهي پنهنجي زڪوٰة جي ذريعي مُعَاشري جي ڪمزور ۽ نادار طبقي جي مَدَد ڪن ۽ دولت ڪجھ ماڻهن جي هٿن ۾ قيد ٿيڻ بدران مُعاشري جي ستايل غريب ماڻهن تائين به پهچندي رهي ته جيئن مُعَاشري ۾ مَعَاشِي توازُن جي فضَا قَائم ٿئي۔

ياد رهي ته جيڪڏهن الله عَزَّوَجَلَّ چاهي ها ته سڀني کي دولت مند ڪري ڇڏي ها ۽ ڪو به شخص غريب نه هجي ها، پر اُن پنهنجي مَشِيت سان ڪنهن کي اَمِير بڻايو ته ڪنهن کي غريب ته جيئن اَمِير کي ان جي دولت ۽ غريب کي ان جي غُربت جي سبب آزمائي۔ جيئن ته سيپاري 8 سورةُ الاَنعام جي آيت نمبر 165 ۾ ارشادِ خُداوندي عَزَّ وَجَلَّ آهي؛

وَ ه ُوَ الَّذِ ي جَعَلَكُمْ خَلَائِفَ الْأَرْضِ وَرَفَعَ بَعْضَكُمْ فَوْقَ بَعْضٍ دَرَجَاتٍ لِ ي َبْلُوَكُمْ فِ ي مَا آتَاكُمْ (پ۸، الانعام:۱۶۵)

تَر ْ جموڪنز الايمان: ۽ اُهو ئِي آهي جِنهن زمِين ۾ توهان کي نائِب ڪ يو ۽ توهان مان ه ڪ ڙن کي ٻِين تي ڪ يئِي درجا بُلندِي ڏنائِين جيئن توهان کي آزمائِي اُن شيءِ ۾ جي ڪ ا توهان کي عطا ڪ يائِين ۔

صدْرُ الْافَاضِل حضرت علّامه مولانا سَيد مُفتي محمد نعيمُ الدّين مُراد آبادي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه هن آيت تحت لکن ٿا: يعني آزمائش ۾ وجھي ته توهان نعمت، عزت ۽ مال حاصل ڪري شُڪر گُذار ڪيئن رهو ٿا ۽ پاڻ ۾ هڪ ٻئي سان ڪهڙي قسم جو سُلوڪ ڪيو ٿا۔ ([1])

مٺا مٺا اسلامي ڀائرو! بيان ڪيل آيت سڳوري ۽ ان جي تفسير مان معلوم ٿيو ته دُنيا دارُالامتحان (يعني آزمائش جو گھر) آهي، هتي ڪنهن کي نعمتن ۽ نوازشن جي ذريعي آزمايو پيو وڃي، ته ڪنهن کي انهن شين کان محروم ڪري آزمائش ۾ وڌو پيو وڃي ته نعمتون ملڻ تي ڪير ڪهڙي انداز ۾ شُڪر يا ناشُڪري ڪري ٿو ۽ نعمتون نه ملڻ تي ڪير ڪهڙي طرح صبر يا بي صبري ڪري ٿو، تنهنڪري اسان کي کپي ته هرحال ۾ صبر ۽ شُڪر سان ڪم وٺندي شرعي اَحڪامن تي عمل سان گڏوگڏ خُوش دِلي سان گهڻو ڪري صَدَقو ۽ خَيرات ڪيون ۽ پنهنجي آخرت جي لاءِ اَجر ۽ ثَوَاب جو ذَخِيرو گڏ ڪرڻ ۾ مصروف رهون۔ قرآنِ ڪريم ۾ مختلف هنڌن تي صَدَقو ۽ خَيرات ڪرڻ جي ترغيب ۽ تاڪيد بيان ڪئي وئي آهي، جيئن ته سيپاري 27 سورة الحديد جي آيت نمبر 10 ۾ ارشادِ خداوندي آهي:

وَمَا لَكُمْ أَلَّا تُنْفِقُوا فِ ي سَبِ ي لِ اللَّ ه ِ وَلِلَّ ه ِ مِ ي رَاثُ السَّمَاوَاتِ وَالْأَرْضِ (پ۲۷،الحديد:۱۰)

تَر ْ جموڪنز الايمان: ۽ توهان کي ڇا آهي جو الله جي راھ ۾ خرچ نٿا ڪ يو حالان ڪ آسمانن ۽ زمين سڀ جو وارث الله ئي آهي.

”سڀ جو وارث الله عَزَّ وَجَلَّ ئي آهي“ ان جي وضاحت ڪندي صدرُ الافَاضِل حضرت علّامه مولانا سَيد مفتي محمد نعيمُ الدّين مُراد آبادي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن

ٿا: توهان برباد ٿي ويندؤ ۽ مال ان جي مِلڪ ۾ رهجي ويندو ۽ توهان کي خرچ ڪرڻ جو ثواب به نه ملندو ۽ جيڪڏهن توهان خُدا جي راھ ۾ خرچ ڪيو ته ثواب به ملندو۔ ([1]) اهڙي طرح هڪ ٻي هنڌ تي صَدَقي جي ترغيب ڏيندي فرمايو ويو:

لَنْ تَنَالُوا الْبِرَّ حَتَّى تُنْفِقُوا مِمَّا تُحِبُّونَ وَمَا تُنْفِقُوا مِنْ شَ ي ْءٍ فَإِنَّ اللَّ ه َ بِ ه ِ عَلِ ي مٌ

(پ۴،اٰلِ عمران:۹۲)

تَر ْ جموڪنز الايمان: اوهان ايستائين چ ڱ ائِئَ کي هرگِز نه پھچندؤ جيس تائ ين پنهنجِي پيارِي شيءِ خرچ نه ڪ ندؤ ۽ اوهان جو ڪ ُجھ ه خرچ ڪ ندؤ سو الله کي معلُوم آھي .

صدْرُ الْافَاضِل حضرت علّامه مولانا سَيد مفتي محمد نعيمُ الدين مُراد آبادي عَلَيه رَحمَۃُ اللّٰه ِالْهادِي هن آيت جي تفسير ۾ فرمائن ٿا ته حضرت ابنِ عُمَر رَضِي اللّٰه تَعَالٰي عَنْهمَا فرمايو ته هتي خَرچ ڪرڻ عام آهي، تمام صَدَقات جيڪي يعني واجِب هجن يا نفلي سڀئي هن ۾شامل آهن۔ حسن جو قَول آهي ته جيڪو مال مُسلمانن کي پسند هجي ۽ ان کي رضاءِ الٰهي جي لاءِ خَرچ ڪري، اهو هن آيت ۾ داخل آهي، چاهي هڪ کجور ئي هجي۔

(حضرتِ سَيدُنا) عمر بن عبدُ العزيز (رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه) شَڪر (کنڊ) جون ٻوريون خريدي صَدَقو ڪندا هئا، انهن کي چيو ويو: ان جي قيمت ڇو نٿا صَدَقو ڪيو؟ فرمايائون: شَڪر مون کي پسند ۽ وڻندڙ آهي، آئون اهو چاهيان ٿو ته خدا جي راھ ۾ پياري شيءَ خَرچ ڪيان۔ ([2])

میں سب دولت رہِ حق میں لُٹا دُوں

شہا ایسا مجھے جذبہ عطا ہو

(وسائلِ بخشش مُرمّم،ص 316)

مٺا مٺا اسلامي ڀائرو! ڏٺو توهان ته پاڻ الله تعاليٰ اسان کي پنهنجو مَحْبُوب ترين مال پنهنجي راھ ۾ خَرچ ڪرڻ جي ترغيب اِرشاد فرمائي رهيو آهي، تنهنڪري اسان کي گھرجي ته ڪنجُوسي ڪرڻ جي بدران، دل کولي صَدَقو ۽ خَيرات ڪيون ۔ ظاهر آهي ته اسان وٽ جيڪو ڪجھ به آهي، اهو الله عَزَّوَجَلَّ جو ئي ڏنل آهي، تنهنڪري ان جي ڏنل مال مان ان جي رِضا جي لاءِ صَدَقو ۽ و خَيرات نه ڪرڻ، الله عَزَّوَجَلَّ جي طرفان ملڻ وارين نعمتن کان مَحرومي جو سبب به بڻجي سگھي ٿو۔

حضرت اَسْماء بِنتِ ابوبڪر رَضِي اللّٰه تَعَالٰي عَنْهمَا کان روايت آهي، فرمائن ٿيون ته مُحَمَّد رسولُ الله صَلَّي اللّٰه تَعَالٰي عَلَيه واٰلِهٖ وَسَلَّمَ مون کي فرمايو: هٿ نه روڪيو، نه ته توهان کان به روڪيو ويندو۔ ([1])

مٺا مٺا اسلامي ڀائرو! مال يقيناً الله عَزَّوَجَلَّ جي هڪ نهايت ئي عظيمُ الشان نعمت آهي، تنهنڪري جنهن کي الله تعاليٰ مال ۽ دولت جي نعمت کان سرفراز فرمايو آهي، ان کي کپي ته اهو ان نعمت جي قدر ڪندي ان کي بِلا ضرورت گڏ ڪري نه رکي ۽ نه ئي فضول ڪمن ۾ ضائع ڪري، بلڪ پنهنجي ماءُ پيءُ، گھرواري، ٻارن ٻچن، ڀيڻن ڀائرن، رشتيدارن، غريبن، مِسڪينن ۽ محتاجن تي جائز طريقي سان خرچ ڪري صدقي ۽ خَيرات جا فائدا ۽ ثَمَرات حاصل ڪرڻ جي ڀرپور ڪوشش ڪري، ڇو ته جيڪو شخص مال کي اِخلاص سان نيڪي جي ڪمن ۾ خرچ ڪندو آهي، اُن جي دُنيا ۽ آخرت سنورجي ويندي آهي، اهڙو ٻانهو الله عَزَّ وَجَلَّ ۽ ان جي مدني حبيب صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ جو محبوب ۽ آخرت جي ثواب جو حقدار ٿيندو آهي۔ قرآن شريف کان علاوه

احاديثِ مُبارڪه ۾ به ڪيئي هنڌن تي صَدَقي ۽ خَيرات جون ترغيبون ۽ فضيلتون بيان ڪيون ويون آهن۔ اچو! ترغيب جي لاءِ هن باري ۾ 5 مبارڪ حديثون ٻڌون ٿا:

مٺا مٺا اسلامي ڀائرو! اوهان ڏٺو ته صَدَقو، مصيبت ۽ بَلا کي روڪيندو آهي، صَدَقو عمر کي وڌائيندو آهي، صَدَقو بُري موت کي پري ڪندو آهي، صَدَقو تڪبُّر کي پري ڪندو آهي، صَدَقو فخر کي پري ڪندو آهي، صَدَقو قبر جي گرمي ختم ڪندو آهي، صَدَقو قِيامت جي ڏينهن ڇانو ٿيندو، صَدَقو دوزخ جي عذاب کان بچائيندو۔

دے نَزع و قبر و حشر میں ہر جا اَمان اور

دوزخ کی آگ سے بچا یا ربِّ مُصطفٰے

(وسائلِ بخشش مُرمّم ،ص132)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اسان جي اسلاف ۽ بُزرگ سڳورن ۾ صَدَقي ۽ خَيرات جو جذبو ايتري قدر گھڻو هو جو ڪڏهن ڪڏهن ته اُهي مُبارڪ هستيون پنهنجي ضرورتن کي پنهنجي مسلمان ڀاءُ جي ضرورت تي قُربان ڪري ڇڏينديون هيون۔ اچو ترغيب جي لاءِ هڪ عظيمُ الشان واقعو ٻڌون ٿا؛

اُمُّ المؤمنين، حضرتِ سَيدَتُنا عائشه صِدّيقه، طَيبَه، طاهره رَضِي اللّٰه تَعَالٰي عَنْهَا کان روايت آهي ته هڪ مسڪين اوهان کان سوال ڪيو جڏهن ته پاڻ رَضِي الله تَعَالٰي عَنْهَا روزي سان هيون، ۽ گھر ۾ سِواءِ هڪ ماني جي ٻيو ڪجھ نه هو۔ پاڻ پنهنجي ٻانهيءَ کي فرمايائون: ان کي اها ماني ڏئي ڇڏيو، ته ٻانهيءَ چيو: اوهان جي اِفطاري جي لاءِ هن کان علاوه ٻيو ڪجھ ناهي، حضرت سيدَ تُنا عائشه صِدّيقه رَضِي الله تَعَالٰي عَنْهَا فرمايو: هن کي اها ماني ڏئي ڇڏ، ٻانهيءَ چوي ٿي ته مون اها ماني ان کي ڏئي ڇڏي، اڃا شام نه ٿي جو اهلِ بيت يا ڪو ٻيو شَخص جيڪو

هديو ڏيندو هو پاڻ رَضِي الله تَعَالٰي عَنْها کي هديي طور هڪ ٻڪري موڪلي، کڻي اچڻ وارو ان گوشت کي ڪپڙي ۾ ڍڪي کڻي آيو۔ پاڻ رَضِي الله تَعَالٰي عَنْهَا مون کي سڏي فرمايائون: وٺ ان مان ڪجھ کاءِ اهو توهان جي ان ماني کان بهتر آهي۔ ([1])

مٺا مٺا اسلامي ڀائرو! اهو هو الله وارن جو طرزِ عمل جو جيڪو ڪجھ به هوندو اهو صَدَقو ڪري ڇڏيندا هئا اهو ئي سبب آهي جو الله عَزَّ وَجَلَّ انهن جي صَدَقي جي ڪري انهن کي بهتر کان بهتر بدلو عطا فرمائي ٿو۔جيئن ته

شيخ ِطريقت، امير اهلسنت، باني دعوتِ اسلامي حضرت علّامه مولانا ابو بلال محمد الياس عطّار قادري رضوي ضيائي دَامَتْ بَـرَکاتُهمُ الْعَالِيه پنهنجي تصنيف ”فيضانِ سُنّت“ جي صفحي 513 تي هڪ واقعو نقل ڪن ٿا: پنهنجي دؤر جا ابدال حضرت سيّدُنا ابو جعفر بن خطاب عَلَیۡهِ رَحْمَةُ اللهِ الۡتَوَّاب فرمائن ٿا، منهنجي دروازي تي هڪ سائل (يعني سواليءَ) صدا هنئي مون گهرواريءَ کان پُڇيو ته ڇا تو وٽ ڪجهه آهي؟ جواب مليو چار(4) بيضا (يعني آنا) آهن مون چيو؛ سوالي کي ڏئي ڇڏيو، انهن ڏيئي ڇڏيا. سائل بيضا وٺي هليو ويو، اڃا ٿوري دير ئي گذري جو مون وٽ هڪ دوست بيضن سان ڀريل هڪ ٽوڪري موڪلي. مون گهران پُڇيو، هن ۾ ڪل ڪيترا بيضا آهن؟ انهن چيو؛ 30. مون چيو توهان ته فقير کي 4 بيضا ڏنا هئا، هي 30 ڪهڙي حساب سان آيا! چوڻ لڳيون ٽيهه بيضا صحيح آهن ۽ باقي 10 ڀڳل آهن، حضرت سيّدُنا شيخ علامه يافعي يمني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا، ڪجهه ماڻهو هن حڪايت جي متعلق بيان ڪن ٿا ته سائل کي جيڪي بيضا ڏنا ويا هئا انهن مان ٽي صحيح ۽ هڪ ڀَڳل هيو. ربّ تعاليٰ هر هڪ جي بدلي ۾ ڏهه ڏهه

اَمِيرُالمُؤمنين حضرت سيدُنا عَلِيّ الْمُرتضٰي، شيرِخُدا کرَّمَ اللّٰه تَعَالٰي وَجْهه الْکرِيم فرمائن ٿا: جيڪڏهن توهان وٽ دُنيا جي دولت اچي ته ان مان ڪجھ خَرچ ڪيو ڇو ته خَرچ ڪرڻ سان اها ختم ناهي ٿيندي ۽ جيڪڏهن دُنيا جي دولت توهان کان منهن ڦيرڻ لڳي تڏهن به ان مان ڪجھ خَرچ ڪيو ڇو ته ان کي باقي ناهي رهڻو۔ ([1])

حضرت سيدُنا حسن بصري رَحْمَۃُ اللّٰهِ تَعَالٰي عَلَيه کان پڇيو ويو ته سخاوت ڇا آهي؟ فرمايائون: الله عَزَّوَجَلَّ جي لاءِ پنهنجو مال خُوب خَرچ ڪرڻ۔ پوءِ پڇيو: حَزم (اِحْتياط) ڇا آهي؟ فرمايائون: الله عَزَّوَجَلَّ جي لاءِ مال روڪي رکڻ۔ پوءِ پڇيو ويو: اِسْراف ڇا آهي؟ فرمايائون: اِقْتِدار جي چاهت ۾ مال خَرچ ڪرڻ ۔ ([2])

(وسائلِ بخشش مُرمّم،ص82)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! ڪجھ ماڻهو محتاجي جو بهانو بڻائي صَدَقي ۽ خَيرات جي معاملي ۾ سُستي ۽ بُخل ڪندا آهن ۽ حال پڇڻ تي شڪوا شڪايات جا انبار لڳائي ڇڏيندا آهن۔ ياد رکو! صَدَقي ۽ خَيرات ڪرڻ سان نه ته مال ۾ ڪا گھٽتائي ٿيندي آهي ۽ نه ئي راھ خدا ۾ مال ڏيڻ جي لاءِ امير ۽ اعليٰ هئڻ شرط آهي جيئن ته سرڪارِ ابدِ قرار، شفيعِ روزِ شمار صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ جو فرمانِ دلنشين آهي: مَا نَقَصَ مَالٌ مِنْ صَدَقَۃِِ يعني صَدَقو ڏيڻ سان مال گھٽ ناهي ٿيندو۔ ([2]) هڪ ٻي هنڌ تي ارشاد فرمايائون: فَاتَّقُوا النَّارَ وَلَوْ بِشِقِّ تَمْرَۃٍ يعني باھ کان بچو، جيتوڻيڪ کجور جي هڪ ٽُڪڙي جي ذريعي. ([3])

هن حديث شريف ۾ انهن ماڻهن جي لاءِ دلاسو آهي، جيڪي مَعاشي پريشانين ۾ مبُتلا هئڻ جي باوجود صَدَقو ۽ خَيرات ته ڪندا آهن پر معمولي شيون صَدَقو ڪرڻ جي ڪري دل ننڍي ڪندا آهن حالانڪ صَدَقو چاهي گھٽ

هجي يا وڌيڪ، جيڪڏهن حلال مال مان اِخلاص سان ڏنو وڃي ته يقيناً ثواب جي لحاظ کان نهايت ئي بهترين آهي جيئن ته اعليٰ حضرت اِمام احمد رضا خان رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جا والدِ گرامي حضرت مولانا مُفتي نَقي علي خان رَحمَۃُ اللّٰه ِتَعَالٰي عَلَيه ارشاد فرمائن ٿا: صَدَقو ڏيڻ ۾ ٿوري شيء کي به حقير نه سمجھو، جيتوڻيڪ وڌيڪ جي استطاعت به هجي، هٿ پهچندو آهي (يعني انسان صَدَقو ڏيڻ جو اِرادو ڪندو آهي) پر شيطان روڪيندو آهي، نفس رڪاوٽ وجھندو آهي، هڪ شيطان ڇا ستّر (70) شيطان صَدَقي کان پري رکندا آهن۔

حديث شريف ۾ ارشاد آهي: انسان پنهنجو صَدَقو ستّر (70) شيطانن جي ڄاڙي چيري ڪري ڪڍندو آهي۔ ([1]) ته اهڙي حالت ۾ ٿورو ئي ڏي ۽ ان کي حقير سمجھي بالڪل لَا تَعَلُّق نه ٿئي، ڇو ته ٿوري شيء کان به محتاج جي ضرورت ڪنهن حد تائين پوري ٿيندي ۽ بُخل جي جَڙ دل تي ڄمڻ ۾ ڪجھ ته گهٽتائي ايندي۔ ([2])

مٺا مٺا اسلامي ڀائرو! يقيناً هيءَ دُنيا فاني آهي ۽ ان سان دل لڳائڻ تمام وڏي ناداني آهي، ڇو ته اسان مان هر ڪنهن کي هڪ نه هڪ ڏينهن مَوت اچڻو آهي، تنهنڪري عقلمندي جو تقاضو هي آهي ته انسان الله عَزَّوَجَلَّ ۽ اُن جي مدني حبيب صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ جي ٻڌايل ڪمن ۾ لڳي وڃي، عبادتِ الٰهي جي ڪثرت ڪري، سنّتن ڀري زندگي گُذاري، ضرورت مطابق حلال روزي ڪمائي، مال سان هرگز دل نه لڳائي، حرام مال حرام ڪمائڻ کان بالڪل بچي ۽ هميشه صرف ۽ صرف حلال ئي ڪمائي ۽ هن حلال مال کي به پنهنجي جائز حاجتن ۽ ضرورتن ۾ خرچ ڪندو رهي ۽ انهيءَ حلال مال مان راھِ خدا ۾

خُوب خُوب صَدَقو ۽ خَيرات ڪري غريبن، مسڪينن، حاجت مندن، پنهنجي قريبي رشتيدارن ۽ ديني شاگردن جو سهارو بڻجي، انهن جي ڏُک درد کي پنهنجو ڏُک دَرد سمجھي ۽ اُنهن جون دعائون وٺڻ جي ڪوشش ڪري۔ پر افسوس! اڄڪلھ صَدَقي ۽ خَيرات جي ذريعي غريبن جي مَدَد ڪرڻ ۽ نيڪي جي ڪمن ۾ مال خرچ ڪرڻ جي بدران فضول ۽ غير ضروري ڪمن جي لاءِ ته پنهنجي خزاني جا مُنهن ته کوليا ويندا آهن، پر هن ڳالھ جي طرف ڪو ڌيان نه ڏنو ويندو آهي ته اسان جي چوڌاري مُعاشري جا ڌڪاريل ڪيئي اهڙا ماڻهو به موجود آهن، جنهن وٽ نه ته مٿو ڍڪڻ جي لاءِ مُناسب جڳھ آهي، نه تَن تي پائڻ جي لاءِ سٺو لباس آهي ۽ نه ئي پيٽ جي باھ وسائڻ جي لاءِ ضرورت مطابق ماني. تنهنڪري اسان کي گھرجي ته پنهنجي جائز خُوشين سان گڏوگڏ وس آهر اهڙن سفيد پوش ماڻهن جي ضرورتن جو به خيال رکون ۽ انهن سان همدردي ڀريو سُلوڪ ڪيون۔

نُور جا پيڪر، سڀني نبين جا سرور صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ فرمائن ٿا: هڪ گرھ ماني، هڪ مُٺ کجور يا ان کان علاوه ڪا ٻي شيء جنهن سان مسڪين کي فائده پهچي۔ ان جي ڪري الله عَزَّوَجَلَّ ٽن (3) ماڻهن کي جنّت ۾ داخل فرمائيندو آهي۔ (1) صاحبِ خانه جنهن حڪم ڏنو (2) گھرواري جيڪا ان کي تيار ڪندي آهي (3) خادم جيڪو مسڪين کي ڏئي ايندو آهي۔ پوءِ رحمتِ ڪونين، ناناءِ حسنين صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ فرمايو: حمد آهي الله عَزَّوَجَلَّ جي لاءِ، جنهن اسان جي خادمن کي به نه ڇڏيو (يعني انهن کي به خير جي ڪم ۾ پنهنجي قُدرت جي مُطابق حصّو ملائڻ ۽ صَدَقو غريب تائين پهچائڻ جي ڪري جنّت ۾ داخل فرمائي ڇڏيندو آهي) ۔ ([1])

مٺا مٺا اسلامي ڀائرو! ڪيئي ڀيرا ائين به ٿيندو آهي جو عام غريبن جي ته اسان خُوب مَدَد ڪندا آهيون ۽ نيڪ ڪمن ۾ به مالي طور تي وڌ کان وڌ حصو وٺندا آهيون، پر چراغ تلي اندهيرا جي مطابق پنهنجن ئي ضرورت مند رشتيدارن کي وساري ڇڏيندا آهيون،تنهنڪري انهن کي به ياد رکيو وڃي ۽ اُنهن جي خُود داري کي تڪليف پهچائڻ بدران نهايت اَحسن انداز ۾ انهن جي مَدَد ڪئي وڃي بلڪ پهريان انهن کي ترجيح ڏني وڃي۔

مٺا مٺا اسلامي ڀائرو! دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي مختلف مدني ڪمن ۾ عملي طور شامل ٿيڻ گھرجي، اَلْحَمْدُلِلّٰه عَزَّ وَجَلَّ! اڄ جي هن پُرفِتَن دَور ۾ تبليغِ قرآن و سُنّت جي عالمگير غيرسياسي تحريڪ، دعوتِ اسلامي اسان کي سُنّتن ڀريو پاڪيزه ماحول مُهيا ڪري رهي آهي۔ هن مدني ماحول سان وابسته ٿي خبر ناهي ڪيترن ئي گُناهگارن کي الله عَزَّ وَجَلَّ جي فضل ۽ ڪرم سان تَوبھ جي تَوفيق ملي ۽ هاڻي اُهي صلوٰة و سُنّت جا پابند ٿي هڪ باعمل ۽ باڪردار مُسَلمان جي حَيثِيت سان زندگي گذاري رهيا آهن ۔ اچو! دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي پنهنجي پاڻ کي مدني ڪمن ۾ مشغول رکي نيڪين جو ذخيرو گڏ ڪرڻ واري هڪ عاشقِ رسول جي مدني بهار ٻڌون ٿا:

لائنزايريا، بابُ المدينه (ڪراچي) جي هڪ نوجوان اسلامي ڀاءُ جو ڪجھ هن طرح بيان آهي ته پهريان مان به هڪ عام ماڊرن ۽ بي نمازي ڇوڪرو هئس، زندگي جا ڏينهن ۽ راتيون غفلتن ۽ گناهن ۾ گذري رهيا هئا۔ رَمَضانُ المبارَڪ 1423ھ ۾ هڪ اسلامي ڀاءُ مون تي انفرادي ڪوشش ڪندي اسان جي

ئي علائقي جي فيضانِ رضا مسجِد (لائنز ايريا) ۾ ٿيڻ واري سنّتن ڀري اِجتِماعي اِعتِڪاف ۾ ويهڻ جي ترغيب ڏياري، مون هاڪار ڪئي ۽ گھر وارن کان اجازت وٺي رَمَضانُ المبارَڪ جي آخِري عشري ۾ مُعتَڪف ٿي ويس۔ اِعتڪاف ۾ عاشِقانِ رسول جي صحبتن جي برڪتن کان خوب مالا مال ٿيس ۽ اِعتِڪاف ۾ سڄي زندگي جي لاءِ پنج وقتو نمازي بڻجڻ جو پختو عزم ڪيو، ٻين گناهن سان گڏوگڏ ڏاڙھي ڪٽائڻ کان به توبه ڪئي۔ هٿوهٿ عمامو شريف به سجائي ورتو ۽ سنّت جي مطابِق مَدَني لباس جي به نيت ڪئي۔ عيد جي ٻي ڏينهن عاشِقانِ رسول سان گڏ 3 ڏينهن جي مَدَني قافِلي ۾ سنّتن ڀريو سفر ڪيو ۽ هن مبارڪ سفر جي بَرَڪت سان مان تبليغِ قرآن ۽ سنّت جي عالمگير غير سياسي تحريڪ دعوتِ اسلامي جو ئي ٿي ويس، الله عَزَّ وَجَلَّ ڪري جو مرڻ گھڙيءَ تائين دعوتِ اسلامي جو مَدَني ماحول مون کان جدا نه ٿئي۔ هاڻي مان فيشن ايبل ماڊرن ڇوڪرو نه رهيس، اعتِڪاف ۽ هٿو هٿ مَدَني قافِلي جي سفر جي دوران عاشقانِ رسول جي قُرب اَلْحَمْدُلِلّٰه عَزَّوَجَلَّ مون کي ڇا مان ڇا بڻائي ڇڏيو. مون تي الله عَزَّوَجَلَّ جو وڏو ڪرم آهي جو مان هي بيان ڏيڻ وقت پنهنجي علائقي ۾ مَدَني انعامات جي ذمّيدار جي حيثيت سان سنّتن جي خدمت ڪري رهيو آهيان۔

مٺا مٺا اسلامي ڀائرو! جڏهن به صَدَقات ۽ خَيرات جو ذڪر ٿيندو آهي ، ان جي فضيلت بيان ڪئي ويندي آهي ته عموماً اسان جي ذهن ۾ فوراً اهو تَاثُّر قائم ٿي ويندو آهي ته اِن مان مراد اُهي صَدَقات آهن جنهن جو تعلق مال سان هوندو آهي ۽ اهي فضيلتون به انهيءَ صَدَقي سان خاص آهن، حالانڪ نبي ڪريم، رؤفٌ رَّحيم صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ اهڙي صَدَقن جو به ذِڪر ڪيو آهي جو جن ۾ مال ۽ دولت خرچ ڪرڻ بغير به صَدَقي جا فضائل حاصل ڪري سگھجن ٿا، اهڙي طرح ڪجھ صَدَقات اهڙا به آهن جن جو شمار وڌيڪ فضيلت وارن صَدَقات ۾ ٿيندو آهي۔ انهن افضل صَدَقن ۾ مالي صَدَقا به شامل آهن ۽ اهي به جن ۾ مال خرچ نٿو ڪرڻو پوي، اچو پهريان ان صَدَقن جي باري ۾ ٻڌون ٿا ،جن ۾ مال خرچ نٿو ڪرڻو پوي:

پنهنجي ڀاءُ جي لاءِ مسڪرائڻ صَدَقو آهي۔نيڪي جي دعوت ڏيڻ ۽ بُرائي کان روڪڻ صَدَقو آهي۔ راھ ڀٽڪڻ واري جي رهنمائي ڪرڻ صَدَقو آهي۔ ڪمزور نگاھ واري (يا نابين) جي مَدَد ڪرڻ صَدَقو آهي۔ رستي تان پٿر، ڪنڊو يا هڏي هٽائڻ صَدَقو آهي۔ پنهنجي ڏول سان پنهنجي ڀاء جي ڏول ۾ پاڻي وجھڻ صَدَقو آهي ([1]) قرض ڏيڻ صَدَقو آهي۔ ([2]) شر ڦهلائڻ کان پاسو ڪرڻ صَدَقو

آهي۔ ([1]) ٻن ماڻهن جي وچ ۾ انصاف ڪرڻ، جانور تي سوار ٿيڻ يا سامان کڻڻ ۾ ڪنهن جي مَدَد ڪرڻ، چڱي ڳالھ ڪرڻ، نماز جي ادائگي جي لاءِ هلڻ ۽ راھ مان تڪليف ڏيندڙ شيء کي پري ڪرڻ صَدَقو آهي۔ ([2]) وڻ لڳائڻ يا فصل پوکڻ جنهن مان ڪو ماڻهو يا جانور کائي اهو به صَدَقو آهي۔ ([3])

اچو! هاڻي اُنهن صَدَقات جي باري ۾ ٻڌون ٿا، جنهن کي رسولُ الله صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ ني افضل صَدَقو ارشاد فرمايو آهي، يعني جيڪڏهن ڪو مسلمان ان صَدَقات کي اَدا ڪري وٺندو، ته اِنْ شَآءَ الله عَزَّ وَجَلَّ اُن کي افضل صَدَقي جو ثواب حاصل ٿي ويندو اُن مان ڪجھ پيشِ خدمت آهن:

تنگدست جو وس آهر صَدَقو ڏيڻ ([4]) تندرستي جي حالت ۾ صَدَقو ڪرڻ، مال جي حرص جي باوجود صَدَقو ڪرڻ ۽ پنهنجي محتاجي جو خوف هئڻ باوجود صَدَقو ڪرڻ افضل ترين صَدَقات مان آهي ([5]) اهڙي طرح ڪنهن مسلمان جو علم ِ دين سکڻ، پوءِ پنهنجي اسلامي ڀاءُ کي سکائڻ۔ ([6]) بکايل کي پيٽ ڀري کاڌو کارائڻ ([7]) رمضان ۾ صَدَقو ڪرڻ۔ ([8]) ڪينو رکڻ واري رشتييدار کي صَدَقو ڏيڻ۔ ([9]) رُٺل مسلمانن ۾ صلح ڪرائڻ ([10]) جائز سفارش ڪري ڪنهن قيدي کي رهائي ڏيارڻ، ڪنهن جو رت ڪِرڻ کان بچائڻ، پنهنجي مسلمان ڀاءُ

سان احسان ڪرڻ۔ ([1]) ۽ سٺي گفتگو ڪرڻ . ([2])

اهي سڀئي ڪم مختلف احاديثِ مُبارڪه جي روشني ۾ افضل صَدَقي ۾ شُمار ٿيندا آهن، جن مان ڪجھ مالي ۽ ڪجھ غير مالي صَدَقا آهن، جن ۾ مال خرچ ڪرڻ بِنا ئي صَدَقي جو ثواب حاصل ٿي ويندو آهي۔

مٺا مٺا اسلامي ڀائرو! اَلْحَمْدُلِلّٰه عَزَّوَجَلَّ تبليغِ قرآن وسنّت جي عالمگير غير سياسي تحريڪ دعوتِ اسلامي اهلِ حق جي عالمگير غير سياسي، مسجد ڀريو تحريڪ آهي جيڪا تبليغِ دين جي تقريباً 102 شعبن ۾ نيڪي جون ڌومون مچائڻ ۾ سرگرمِ عمل آهي۔ هن نازُڪ دَور ۾ جڏهن باطل قوّتون چئني پاسي کان سڄي طاقت سان مسلمانن جي ايمان کي برباد ڪرڻ ۽ اُنهن کي گمراهي جي کڏ ۾ ڪيرائڻ جي ڪوششن ۾ مشغول آهن، اهڙي ۾ دعوتِ اسلامي اُمّيد جو ڪِرڻو بڻجي چمڪي ۽ هن مدني تحريڪ اُمّت جي ٻڏندڙ ڪِشتي کي سهارو ڏيڻ، سڄي دُنيا ۾ تعليمِ قرآن ۽ سُنّت کي عام ڪرڻ، نُورِ قرآن سان سينن کي سرشار ۽ اُمَّتِ مُصطَفٰي کي خوابِ غفلت مان بيدار ڪرڻ جي جذبي جي تحت مختصر عرصي ۾ گھڻن ئي ملڪن ۽ شهرن ۾ مدني منّن ۽ منّين جي لاءِ سوين مدرسا ”مدرسۃ المدينه“ جي نالي سان، جڏهن ته اسلامي ڀائرن ۽ اسلامي ڀينرن جي تعليم ۽ تربِيت جي لاءِ جامعات بنام ”جامعۃ المدينه“ جو بُنياد رکيو، هن ۾ بِلامبالغه هزارين شاگرد ۽ شاگردياڻيون قُرآن ۽ سنت جي مُفت تعليم حاصل ڪندا آهن، جتي انهن کي زيورِ قرآن ۽ علمِ دين نيز شرعي ۽ اخلاقي تربِيت سان آراسته ڪرڻ جو خصوصيت سان اهتمام ڪيو

ويندو آهي، ته جيئن دَورِ طالبِ علمي کان فراغت بعد قوم جا اُهي حقيقي مِعمار مُعاشري جي بگاڙ کي سڌارڻ ۾ پنهنجو ڪردار اَدا ڪندي، شيخِ طريقت،اميرِ اهلسنت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي عطا ڪيل هن مدني مقصد جي تحت زندگي گُذارڻ وارا بڻجي وڃن ته ” مون کي پنهنجي ۽ سڄي دنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻي آهي ۔ اِنْ شَآءَ الله عَزَّ وَجَلَّ “

ياد رکو! دعوتِ اسلامي جي تحت هلڻ وارا اهي مدارِسُ المدينه و جامعاتُ المدينه دِيني فضليت جي لحاظ سان وڏي اهميت جا حامل آهن، اهو ئي سبب آهي جو انهن مدارِسُ المدينه ۽ جامعاتُ المدينه کي قائم رکڻ جي لاءِ اِنهن جي مختلف اَخراجات جي مَد ۾ سالانو ڪروڙين روپيا خرچ ڪيا ويندا آهن، جنهن جي لاءِ ڪڏهن مدني چينل جي ذريعي به ٽيلي ٿون (يعني مدني عطيات جمع ڪرڻ جي مهم) جي به ترڪيب ٿيندي آهي، تنهنڪري اوهان سان به عرض آهي ته علمِ دين جي طالبن جي دُعائن مان حِصّو پائڻ، دعوتِ اسلامي جي ترقّي ۽ بقا ۽ مدارِس المدينه و جامعات المدينه جي نظام کي مضبوط کان مضبوط تَر بڻائڻ جي لاءِزڪوٰ ۃ ، فطرات، صَدَقات ۽ خَيرات، نفلي عطيات ۽ عُشر وغيره جي ذريعي نه صرف پاڻ تعاوُن ڪيو بلڪ پنهنجي عزيزن، پاڙيسرين ۽ دوست اَحباب تي انفرادي ڪوشش ڪري اُنهن کي به هن جي ترغيب ڏياريو، اسان جي زبان چَورڻ ۽ اسان جي اِنفرادي ڪوشش سان جيڪڏهن ڪنهن جو مدني ذهن بڻجي ويو ۽ اُن پنهنجي زڪو ۃ ، فطرات، صَدَقات ۽ خَيرات،نفلي عطيات يا عُشر وغيره دعوتِ اسلامي کي مدني ڪمن جي لاءِ ڏني ته اها اسان جي لاءِ به ثوابِ جاريه جو ذريعو بڻجي ويندي۔

مٺا مٺا اسلامي ڀائرو! اَلْحَمْدُلِلّٰه عَزَّ وَجَلَّ هزارين عاشقانِ رسول، مختلف انداز سان "تبليغِ قرآن وسُنّت جي عالمگيرغير سِياسي مسجد ڀريو تحريڪ دعوتِ اسلامي جي مدني ڪمن جي لاءِ مالي تعاون جي سعادت حاصل ڪندا رهندا

آهن، ٿي سگھي ٿو ته اسان مان ڪنهن جي ذهن ۾ سوال اچي ته ڀلا مان ڪهڙي نموني پنهنجو حصو دعوتِ اسلامي جي مدني ڪمن جي لاءِ شامل ڪري سگھان ٿو؟

اچو هڪ نهايت ئي آسان طريقو اوهان جي خدمت ۾ عرض ڪيان ٿو ته جنهن جي ذريعي غريب کان غريب به دعوتِ اسلامي جي مدني عطيات ۾ پنهنجو حصو شامل ڪرڻ ۾ ڪامياب ٿي سگھي ٿو، اهو ڇا آهي، ”مدني عطيات بڪس (Box) “ جي ذريعي مالي تعاون۔

دعوتِ اسلامي جي شعبي ” مجلس مدني عطيات بڪس “ جي طرفان هڪ بڪس جي ترڪيب ڪئي وئي آهي، اهي مدني عطيات بڪس دُڪانن، ڪارخانن، مارڪيٽن، شاپنگ مالز، ميڊيڪل اسٽورز ۽ دفترن وغيره ۾ رکڻ سان گڏوگڏ گھرن ۾ به رکيو ويندو آهي، ته جيئن اسان پنهنجي آساني کي ڏسندي روزانو ڪجھ نه ڪجھ رقم ان بڪس ۾ وجھندا رهون جيڪي دُڪاندار آهن اهي سٺين سٺين نيتن سان پنهنجي گراهڪن (Customers) تي اِنفرادي ڪوشش ڪري انهن کي به راھ خدا ۾ مال خرچ ڪرڻ جي ترغيب ۽ فضائل ٻڌائي انهن کي به پنهنجو حصو شامل ڪرڻ جي ترڪيب ڪيو ته مدينه مدينه۔

مشورةً عرض آهي ته اسان روزانو هڪ رقم خاص ڪيون مثال طور 5 روپيا ئي سهي، پوءِ ان جي مطابق اسان روزانو پنهنجو حصو وجھندا وڃون ۽ مجلس مدني عطيات بڪس جي طئي شده طريقي جي مطابق اهي مدني عطيات جمع به ڪرائي ڇڏيون۔ جيڪي دُڪانن وغيره تي بڪس رکيا ويندا آهن، هن کي "مدني عطيات بڪس" ۽ جيڪي گھرن ۾ بڪس رکيا ويندا آهن، اُن کي"گھريلو صدقه بڪس" چئبو آهي۔

حديثِ پاڪ ۾ آهي: اِنَّ الدَّالَّ عَلَي الْخَيرِ کفَاعِلِهٖ يعني بيشڪ نيڪي جي راھ ڏيکارڻ وارو ، نيڪي ڪرڻ واري جيان آهي۔ ([1])

وَ لَا تَمۡـشِ فِـی الۡاَرۡضِ مَـرَحًا ۚ اِنَّکَ لَـنۡ تَخۡرِقَ الۡاَرۡضَ وَ لَنۡ تَبۡلُغَ الۡجِبَالَ طُوۡلًا 0

ترجموڪنزالايمان: ۽ زمين ۾ آ ڪ ڙ ڪ ري نه هل بيش ڪ تون زمين کي هرگز چيري نه سگهندين ۽ هرگز پهاڙن جي اوچائي کي نه پهچندين .

* فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: هڪ شخص ٻه چادرون پائي تڪَبُّر سان هلي رهيو هو ۽ گهمنڊ ۾ هو اهو زمين ۾ داخل ڪيو ويو ۽ اهو قيامت تائين زمين ۾ داخل ڪيو ويندو. ([1]) * رسولِ اڪرم،نورِ مُجَسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هلندا هئا ته ڪجهه اڳيان طرف جهڪي هلندا هئا ڄڻ ته پاڻ مٿانهون کان هيٺ پيا لهن ـ ([2]) * جيڪڏهن ڪا رڪاوٽ نه هجي ته رستي جي ڪناري ڪناري وچٿري چال هلو، نه ايترو تيز جو ماڻهن جون نگاهون اوهان ڏانهن ٿين ته تڪڙو تڪڙو ڪيڏانهن وڃي رهيو آهي! نه ايترو آهستي جو ڏسڻ واري کي اوهان بيمار لڳو * رستي تي هلندي پريشان نظري(يعني بلاضرورت هيڏانهن هوڏانهن ڏسڻ) سنت ناهي، هيٺ نگاهه ڪري باوقار طريقي سان هلو * هلڻ يا ڏاڪڻ تان لهڻ مهل هي احتياط ڪيو ته جوتن جو آواز پيدا نه ٿئي * رستي تي ٻه عورتون بيٺيون هجن يا وڃي رهيون هجن ته انهن جي وچ مان نه گذرو جو حديث پاڪ ۾ هن جي مُمانعت آيل آهي. ([3]) * ڪجھ ماڻهن

جي عادت هوندي آهي ته رستي تي هلندي جيڪا به شيء سامهون آئي ان کي لتون هڻندي گذرندا آهن، هي بلڪل غير مُهَذَّب طريقو آهي ائين ڪرڻ سان ڪنهن جو پير زخمي ٿيڻ جو انديشو آهي ۽ اخبار يا لکائي وارن دٻن، پيڪٽن، ۽ مِنرل واٽر جي خالي بوتلن وغيره کي لت هڻڻ بي ادبي به آهي .
