پيارا پيارا اسلامي ڀائرو! يقيناً عقل مند اهو ئي آهي جيڪو موت کان پهريان موت جي تياري ڪري، الله ۽ رسول جي حڪم جي مطابق زندگي گذاري، ربّ جا احڪام سمجھي، اُنھن کي مڃي ۽ انهن تي عمل ڪري، سعادت مند اهو ئي آهي جيڪو سفرِ آخرت جو راهي بڻجڻ کان پهريان ئي زادِ راھ تيار ڪري، ياد رکو دنيوي زندگي بيحد مختصر آهي۔ ٭ اسان جو هر پساھ اسان کي موت جي ويجھو ڪري رهيو آهي، ٭ اسان جو هر پساھ اسان کي دُنيوي زندگي جي خاتمي ڏانهن کڻي وڃي رهيو آهي، ٭ اسان جو هر پساھ اسان کي قَبر جي کڏ جي ويجھو ڪري رهيو آهي۔ ٭ اسان جو هر پساھ ڄڻ ته عالمِ دُنيا کان عالَمِ برزخ ڏانهن وڏي تيزي سان وڌندڙ هڪ قدم آهي، ٭ اسان جو هر پساھ اسان کي موت جي ملائڪ سان ملائڻ جي طرف کڻي وڃي رهيو آهي، ٭ اسان جو هر پساھ دنيا جي سفر کي مختصر ڪندو پيو وڃي، ٭ اسان جو هر پساھ اسان کي راھ آخرت سان ملائڻ جو وسيلو بڻجي رهيو آهي۔ جيئن ئي اهو ساھ اسان جو بند ٿيو ، ته اسان جي عمل وارو سلسلو به رڪجي ويندو، پوءِ حسرت ۽ افسوس جي سِوا ڪجھ نه ورندو، ايتري به مُهلت نه ڏني ويندي جو هڪ ڀيرو ” سُبْحَانَ الله “ چئي پنهنجي نيڪين ۾ ئي اضافو ڪري سگھون۔ انهيءَ ڪري دنيا ۾ مليل هن زندگي کي غنيمت سمجھندي نيڪيون ڪمائڻ ۽ آخرت کي بهتر بڻائڻ ۾ لڳي وڃڻ گھرجي۔

پيارا پيارا اسلامي ڀائرو! هن ۾ ڪو شڪ ناهي ته هر شيء جو ڪو نه ڪو مقصد هوندو آهي، اسان جي پائڻ لاءِ ڪپڙا هجن، لکڻ جي لاءِ قلم هجي، رهڻ جي لاءِ گھر هجي، هٿ ۾ ٻڌل گھڙي هجي، تيز رفتار بائيڪ (موٽرسائيڪل) هجي يا هوا ۾ اُڏامندڙ جهاز هجن، سڀني جو ڪجھ نه ڪجھ مقصد ضرور آهي ۽ هر هڪ شيء پنهنجي مقصد جي تڪميل جو سبب بڻجي رهي آهي، ٿورو سوچيو! جڏهن ڪائنات جي هر شيء پنهنجي اندر ڪو نه ڪو مقصد

رکي ٿي ته ڇا انسان بي مقصد پيدا ڪيو ويو آهي؟ ڇا هن اشرفُ المَخلُوقات جي پيدا ڪرڻ جو ڪو مقصد ناهي؟ ڇا حضرتِ انسان جي پيدائش بِلاوجه ۽ بِلا مقصد آهي؟ نه! نه هرگز نه! انسان کي هن دنيا ۾ بيڪار ناهي پيدا ڪيو ويو جيئن سيپاري 18 سورة المُؤمِنُون آيت نمبر115 ۾ ارشاد آهي:

اَفَحَسِبْتُمْ اَنَّمَا خَلَقْنٰكُمْ عَبَثًا وَّ اَنَّكُمْ اِلَیْنَا لَا تُرْجَعُوْنَ(۱۱۵) (پ۱۸،المؤمنون:۱۱۵)

ترجمو ڪنزالايمان: پوءِ ڇا توهان اهو سمجهيو آهي ته اسان توهان کي بيڪار پيدا ڪيو آهي ۽ توهان کي اسان جي طرف موٽڻو نه آهي.

اي عاشقانِ رسول! غور ڪريو! قرآنِ پاڪ جي هن آيت سڳوري مان معلوم ٿي رهيو آهي ته ڪو خاص مقصد آهي جنهن جي لاءِ انسان کي پيدا ڪيو ويو آهي، هن مقصد جي رهنمائي ڪندي سيپاري 27، سُورة الذّٰرِيٰت آيت نمبر 56 ۾ اِرشاد آهي:

وَ مَا خَلَقْتُ الْجِنَّ وَ الْاِنْسَ اِلَّا لِیَعْبُدُوْنِ(۵۶) (پ۲۷، الذّرِيٰت:۵۶)

ترجموڪنزُالايمان: ۽ مان جن ۽ انسان فقط انهيءَ لاءِ پيدا ڪيا جو منهنجي بندگي ڪن.

هن آيت مان معلوم ٿيو ته انسانن ۽ جِنّن کي بيڪار پيدا ناهي ڪيو ويو بلڪ انهن جي پيدائش جو اصل مقصد اهو آهي ته هو الله پاڪ جي عبادت ڪن۔

پيارا پيارا اسلامي ڀائرو! هن ۾ ڪو شڪ ناهي ته دنيا آخرت جي کيتي آهي، دنيا ۾ ڪيو ويندڙ هر عمل آخرت جي لاءِ وڏي اهميت رکي ٿو، آخرت کي بهتر ڪرڻ جي لاءِ نهايت ضروري آهي ته عملن کي ٺيڪ ڪيو وڃي۔ الله پاڪ جي احڪامن کي سمجھيو وڃي ۽ ان جي مطابق عمل ڪيو وڃي۔ پنهنجي اصل مقصد عبادتِ الٰهي کي پيشِ نظر رکي حڪمِ اِلٰهي جي مطابق عبادتِ اِلٰهي ادا ڪنداسين ۽ ربّ ڪريم جو فضل شاملِ حال رهيو ته آخرت

کي بهتر بڻائڻ گھڻو آسان ٿي ويندو۔

دنيا جو امتحان ۽ آخرت جو امتحان

مَدرسن، جامعات ۽ اسڪولن، ڪاليجن جا جڏهن امتحان ٿيندا آهن ته ڏٺو ويندو آهي ته شاگرد ان جي ڪيئن تياري ڪري رهيا هوندا آهن، محنتي شاگردن کي پيپر جي علاوه ٻي ڪنهن شيء جو هوش ناهي رهندو، نه کائڻ ياد رهندو آهي نه پيئڻ جو هوش رهندو آهي، صرف هڪ ئي ڌُن هوندي آهي ته امتحان جي تياري ڪرڻي آهي ۽ هر اهو سوال جنهن جو اندازو هجي ته اهو پيپر ۾ اچي سگھي ٿو، ان جي خاص طور تي تياري ڪئي ويندي آهي، اهڙي وقت ۾ والدين جو به شاگردن کي ڀرپور ساٿ حاصل هوندو آهي ڇوته انهن کي خبر آهي ته پٽ ڪامياب ٿيندو ته ڪماؤ پٽ ٿيندو۔ اولاد دُنيوي امتحان ۾ ڪامياب ٿيندو ته زندگي بهتر ٿيندي، ٿورو سوچيو! جڏهن دنيا جي خاطر اسان ۽ اسان جو اولاد ايترو مگن ٿي ويندو آهي، ته پوءِ آخرت جي امتحان جي تياري جي لاءِ ته دُنيا جي امتحان کان وڌيڪ ڪوشش ڪرڻ گھرجي۔

پيارا پيارا اسلامي ڀائرو! اچو! ٿوري دير جي لاءِ اسان غور سان پنهنجو جائزو وٺون ٿا ته ڇا ڪڏهن آخرت جي امتحان جي تياري لاءِ به ڪڏهن فڪر ٿيو آهي؟ ڇا ڪڏهن آخرت جي اِمتحان جي ڪاميابي جي لاءِ به اسان بي چين ٿيا آهيون۔ اعليٰ حضرت امام اهلسنّت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا شعر ڪيتري قدر آخرت ڏانهن مائل ڪرڻ وارا آهن، ٻڌو!

(1) دنيا جو سفر ڪنڊن يعني غفلتن، مال ۽ اولاد جي آزمائش وغيره سان ڀريل آهي ۽ ان تي وڌيڪ مصيبت اها آهي ته پير به زخمي آهن، قدم قدم تي رڪاوٽون آهن، خدا خير ڪري خبر ناهي اهو سفر ڪيئن طئي ٿيندو۔ (2) اسان ته ڄاڻي واڻي اهو ئي ڪم ڪريون ٿا جن کان الله پاڪ ۽ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم منع فرمايو آهي ته اسان جو ڇا حال ٿيندو ۽ اسان ڪيئن نجات ماڻينداسين ؟ (3) يعني اسان ته پنهنجي خيال مطابق وڏي احتياط سان لڪي لڪي گناھ ڪيا آهن جڏهن ته اسان کي اها خبر به هئي ته الله پاڪ اسان جي هر لڪيل، ظاهر کان آگاھ آهي پر ماڻهن جو ڊپ هو خدا جو ڊپ نه هو۔ (4) اڙي او مجرم! تون ڪهڙو نادان آهين ڏسين نٿو ته تنهنجي مٿي تي هر وقت موت جي تلوار لٽڪي رهي آهي ۽ توکي ڪا پرواھ ئي ناهي، هاڻي توکي خبر پئجي وئي هوندي ته اهڙي مجرم سان ڇا ٿيندو۔ (5) اسان جي سوچ به ڪيتري ٻاراڻي آهي جو ڪم ته اهڙا ڪيون ٿا جو قيد خاني ۾ وڌا وڃون پر اميدون جنت جي باغن ۾ وڃڻ جون رکيون ويٺا آهيون، هنن حالتن ۾ اسان سان ڇا ٿيندو، خدا اسان سان اسان جي عملن جي مطابق معاملو ڪرڻ بدران عفو و ڪرم وارو معاملو فرمائي۔ (6) ڇا تو کي معلوم نه آهي ته تنهنجي ۽ جنت جي وچ ۾ باھ جو دريا آهي جنهن کي پار ڪري ئي جنت ۾ داخلا ٿي سگھي ٿو۔ (7) ربّ ڪريم جي بارگاھ ته اها بارگاھ آهي جو جتي ڪوڙ بهانو، مڪر ۽ فريب هلي ئي نٿو سگھي، اتي پنهنجي ڏوهن جي اقرار ڪرڻ کانسواءِ ڪا ٻي راھ نه هوندي۔ الله ڪريم اسان جي حال تي رحم فرمائي۔ آمين

پيارا پيارا اسلامي ڀائرو! مسلمان هئڻ ناطي جڏهن اسان کي يقين آهي ته

اسان کي هڪ ڏينهن مرڻو آهي، اونداهي قبر ۾ لهڻو آهي، خبر ناهي ڪيترا سال عالمِ برزخ (برزخ جي لفظي معنيٰ اوٽ ۽ پردي جي آهي ۽ مرڻ جي بعد کان وٺي قِيامت ۾ اُٿڻ تائين جو وقفو "برزخ" چورائي ٿو) ۾ رهڻو آهي، پوءِ قِيامت قائم ٿيڻي آهي، پوءِ ميدانِ محشر ۽ حساب ڪتاب جو معاملو دَرپيش ٿيڻو آهي ، ته اسان کي هنن سمورن مرحلن جي تياري ڪرڻ گھرجي، هڪ مسلمان جي آخرت جي تياري جو انداز ڪيئن هئڻ گھرجي ۽ هڪ ٻانهو ڪهڙي طرح هميشه پنهنجي آخرت کي ياد رکي سگھي ٿو؟ اچو هن جا ٻه طريقا ٻڌون ٿا۔

پيارا پيارا اسلامي ڀائرو! آخرت جي تياري ڪرڻ جي لاءِ موت کي ياد ڪرڻ نهايت ضروري آهي، ڇوته موت آخرت جي پهرين ڏاڪڻ آهي ۽ موت آخرت جو پهريون مرحلو آهي۔ موت کي ڪثرت سان ياد ڪرڻ ڄڻ ته آخرت جي سمورن مرحلن کي ياد ڪرڻ آهي جو جڏهن ٻانهو موت کي ياد ڪندو ته ربّ کي راضي ڪرڻ وارن ڪمن ڏانهن وڌندو، اهو ڄاڻڻ جي ڪوشش ڪندو ته ڪهڙي ڪم ۾ منهنجي ربّ جي رِضا آهي ۽ ڪهڙو ڪم ان کي ناراض ڪرڻ وارو آهي، پوءِ اهو ربّ جي احڪامن جي مطابق ان جي عبادت ۾ مشغول ٿي ويندو۔ الله وارا سڄي زندگي فڪرِ آخرت ۾ گذاري ڇڏيندا هئا، ان ڪري جو اهي ڪثرت سان موت کي ياد ڪندا هئا ۽ انهن کي موت زندگي کان وڌيڪ عزيز هوندو هو۔

جيئن منقول آهي ته حضرت سيدُنا امام محمد غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هڪ ڀيري مڪه معظمه ۾ تشريف فرما هئا۔ امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ظاهري شان ۽ شوڪت سان بي نياز هئا، ان ڪري سندن کي نهايت ساده ۽ معمولي قسم جو لباس پاتل هئو۔ حضرت سيدُنا عبدالرحمن طُوسي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه عرض ڪيائين: اوهان وٽ هن جي علاوه ٻيو ڪو ڪپڙو نه آهي۔ اوهان وقت جا اِمام ۽ قوم جا پيشوا آهيو، هزارين ماڻهو اوهان جا مُريد آهن؟ امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جواب ڏنائون: اهڙي شخص جو لباس ڇو ڏسو ٿا جيڪو هن دنيا ۾ هڪ مسافر جيان مقيم هجي ۽ ج يڪو هن ڪائنات جي رنگينن کي فاني ۽ وقتي تصور ڪندو آهي۔ جڏهن ڪائنات جي آقا، مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن دنيا ۾ مسافر جيان رهيا ۽ ڪجھ مال دولت گڏ نه ڪيائون ته منهنجي ڇا حيثيت ۽ حقيقت آهي۔ ‘‘ (احياء العلوم، ۱/۱۹)

جنازي جون سُنّتون ۽ آداب

پيارا پيارا اسلامي ڀائرو! اچو! شَيخِ طريقت،اَميرِاَهلسُنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ”مڙدي جي صدما“ مان جنازي جون سُنّتون ۽ آداب ٻڌون ٿا: پهريان 2 فرمانِ مصطَفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحَظَه ڪريو: (1) جڏهن ڪو جنّتي شخص فوت ٿي ويندو آهي ته الله پاڪ حياء ڪندو آهي ته انهن ماڻهن کي عذاب ڏئي جيڪي ان جو جنازو کڻي هليا ۽ جيڪي ان جي پٺيان هليا ۽ جن ان جي جنازي نماز ادا ڪئي۔ (فِردَوس الْاخبار،۱/۲۸۲،حديث ۱۱۰۸) (2) مؤمن ٻانهي کي مرڻ کان پوءِ سڀ کان پهرين جزا اها ڏني ويندي جو ان جي جنازي ۾ شريڪ ٿيڻ وارن جي بخشش ڪئي ويندي آهي۔ (مُسندُ البزار، ۱۱/ ۸۶، حديث: ۴۷۹۶) ٭ جنازي نماز ۾ رضاءِ الٰهي، فرضِ ڪفايه جي ادائيگي، ميت ۽ ان جي عزيزن جي دلجوئي وغيره سٺين سٺين نيتن سان شرڪت ڪرڻ گھرجي. ٭ جنازي سان گڏ ويندي پنهنجي انجام جي باري ۾ سوچيندا رهو ته جيئن اڄ هن کي کڻي پيا هلن، اهڙي طرح هڪ ڏينهن مون کي به کڻي هلندا، جهڙي طرح ان کي مٽي جي مڻن هيٺ دفن ڪيو وڃڻ وارو آهي، اهڙي طرح مون کي به دفنايو ويندو، اهڙي طرح غور و فڪر ڪرڻ

عبادت ۽ ثواب جو ڪم آهي. ٭ جنازي کي ڪلهو ڏيڻ ثواب جو ڪم آهي۔ حديثِ پاڪ ۾ آهي؛ جيڪو جنازي کي چاليھ قدم کڻي هلي، ان جا چاليھ ڪبيره گناھ معاف ڪيا ويندا ۽ هڪ ٻي حديث پاڪ ۾ آهي؛ جيڪو جنازي جي چئني پاون کي ڪلهو ڏيندو، الله عَزَّوَجَلَّ ان جي مستقل بخشش فرمائي ڇڏيندو۔ (جوهره، ص۱۳۹، دُرِّمُختار،۳/۱۵۸،۱۵۹،بهارِ شريعت،۱/۸۲۳) ٭ سنّت هي آهي ته هڪ ٻئي پٺيان چئني پاون کي ڪلهو ڏئي ۽ هر ڀيري ڏھ ڏھ قدم هلي۔ پوري سنّت هي آهي ته پهريان ساڄي پاسي جي سيرانديءَ کي ڪلهو ڏي پوءِ ساڄي پيراندي کي، بعد ۾ کاٻي سيراندي کي ۽ پوءِ کاٻي پيراندي کي ۽ ڏھ ڏھ قدم هلي ته ڪل چاليھ قدم ٿيا۔ (عالمگيري،۱/۱۶۲،بهارِشريعت،۱/۸۲۲) ٭ جنازي کي ڪلهو ڏيڻ وقت ڄاڻي واڻي ايذاء ڏيڻ واري انداز ۾ ماڻهن کي ڌڪا ڏيڻ جيئن ڪجهه ماڻهو ڪنهن شخصيت جي جنازي ۾ ڪندا آهن اهو ناجائز ۽ حرام ۽ جھنم ۾ وٺي وڃڻ وارو ڪم آهي. ٭ مڙس پنهنجي گھرواريءَ جي جنازي کي ڪلهو به ڏئي سگهي ٿو، قبر ۾ به لاهي سگهي ٿو ۽ منهن به ڏسي سگهي ٿو، صرف غسل ڏيڻ ۽ بنا رُڪاوٽ جي جسم کي هٿ لڳائڻ منع آهي ۔ (بهارِشريعت،۱/۸۱۲،۸۱۳) ٭ جنازي سان گڏ بلند آواز سان ڪلمو شريف يا ڪلمو شهادت يا حمد ۽ نعت وغيره پڙهڻ جائز آهي۔ (فتاويٰ رضويه،۹/۱۳۹تا۱۵۸)

طرح طرح جون هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ”بهار شريعت“ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں ۽ آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي.

مجھ کو جذبہ دے سفر کرتا رہوں پَروَردگار سُنّتوں کی تَرْبِیَت کے قافِلے میں بار بار

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

ايندڙهفتيواراجتماع جي بيان جون جھلڪيون

پيارا پيارا اسلامي ڀائرو! اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ! آئنده هفتيوار سنتن ڀري اِجتماع جي بيان جو موضوع هوندو ” زندگي جو مقصد۔ ‘‘ اسان جي زندگي جو مقصد ڇا آهي، اسان هن مقصد کي ماڻڻ جي لاءِ ڪهڙا عمل ڪري رهيا آهيون ۽ ڪهڙا ٻيا عمل ڪري سگھون ٿا، اسان جي بزرگن جو زندگي گذارڻ جو ڪهڙو انداز هوندو هو جڏهن ته اسان جي زندگي ڪئين گذري رهي آهي ۽ اسان کي زندگي ڪهڙي طرح گذارڻ گھرجي۔ ان جي علاوه ڪيئي سبق آموز واقعا ۽ هنن واقعن مان حاصل ٿيڻ وارا مدني گل به آئنده هفتيوار سُنّتن ڀري اجتماع ۾ بيان ڪيا ويندا۔ لهٰذا! آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ اِنفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان چئو: اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

(1) جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

(2) سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

(3) رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

(4) ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([1])

(5) قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهڙي مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪارمدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([2])

(6) دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([3])

(1) هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

(2) ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)
