اي عاشقانِ علماء و اوليا! اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي سڄي دُنيا ۾ خدمتِ دين جي ڪم و بيش 105 شُعبن ۾ سُنّتن جون ڌُومون مچائي رهي آهي، جن مان هڪ شعبو ” مجلس رابطه بِالعلماء “ به آهي۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ شَيخِ طريقت، اميرِاهلسُنَّت، باني دعوتِ اسلامي حضرت علامه مولانا ابو بلال محمد الياس عطار قادِرِي رَضَوِي ضِيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي علماءِ اهل سنت سان مَحَبَّت جي نتيجي ۾ عاشقان رسول جي مدني تحريڪ دعوت اسلامي هڪ شُعبو بنام ’’ مجلس رابطه بِالعلماء ‘‘ به قائم ڪيو آهي۔ ته جيئن ان جي ذريعي عاشقانِ رسول عالم سڳورن ۽ مَشائخِ عِظّام (۽ مسجدن جي امامن، خطيبن، مُدرسن) کي عاشقانِ رسول جي مدني تحريڪ دعوت اسلامي جي دِيني خِدمتن کان آگاه ڪيو وڃي، انهن سان رابطو ڪري انهن کي دعوت اسلامي جي مدني ماحول سان وابسته ڪيو وڃي ۽ ان سان دعوتِ اسلامي جي مَدَني ڪمن ۾ مُعاوَنت حاصل ڪئي وڃي۔ ۽ انهن جون دُعائون ورتيون وڃن ۽ عاشقانِ رسول مَدرسن ۽ جامعات ۾ دعوتِ اسلامي جي مَدَني ڪمن جي ترڪيب ڪئي وڃي۔ انهن جي طالبن ۽ استادن کي دعوت اسلامي جي هفته وار اجتماع ۽ مدني مذاڪري ۾ شرڪت ڪرائي وڃي.

سلام ڪرڻ جون سنتون ۽ آداب

اچو! شيخِ طريقت، اميرِاهلسُنَّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ”101 مدني گل“ مان سلام جي باري ۾ ڪجھ سنتون ۽ آداب ٻڌون ٿا: ٭ مسلمان سان ملاقات ڪرڻ وقت ان کي سلام ڪرڻ سنّت آهي۔ ٭ مڪتبة المدينه جي شايع ٿيل ڪتاب بهار شريعت جلد 3 صفحي 459 تي لکيل مسئلي جو خلاصو آهي: ”سلام ڪرڻ وقت دل ۾ اها نيّت هجي ته جنهن کي سلام ڪري رهيو آهيان ان جو مال ۽ عزت، آبرو سڀ ڪجھ منهنجي حفاظت ۾ آهن ۽ مان انهن مان ڪنهن شيء ۾ دخل اندازي ڪرڻ حرام سمجھان ٿو“ (بهار شريعت ،۳/۴۵۹،حصه ۱۶ ملخصا) ٭ ڏينهن ۾ ڪيترائي ڀيرا ملاقات ٿئي، هڪ ڪمري کان ٻئي ڪمري ۾ بار بار اچ وڃ ٿئي، اتي موجود مسلمانن کي سلام ڪرڻ ثواب جو ڪم آهي ٭ سلام ۾ اڳرائي ڪرڻ سنّت آهي۔ ٭ سلام ۾ اڳرائي ڪرڻ وارو الله پاڪ جو مُقرّب آهي. ٭ سلام ۾ اڳرائي ڪرڻ وارو تَڪَبُّر کان به بَري آهي جيئن منهنجي مڪي مدني آقا، مٺي مٺي مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو فرمان آهي: سلام ۾ اڳرائي ڪرڻ وارو تَڪَبُّر کان بَري آهي۔ (شعب الايمان ، ڪ تاب، ۶ / ۴۳۳،حديث: ۸۷۸۶) ٭ سلام (۾ اڳرائي) ڪرڻ واري تي 90 رحمتون ۽ جواب ڏيڻ واري تي 10 رحمتون نازل ٿينديون آهن. (ڪ يميائے سعادت،۱ / ۳۹۴) ٭ اَلسَّلَامُ عَلَیْکُمْ چوڻ سان 10 نيڪيون ملن ٿيون، گڏوگڏ وَرَحْمَۃُ الله به چوندا ته 20 نيڪيون ٿي وينديون ۽ وَبَرَکَاتُہٗ شامل ڪندؤ ته 30 نيڪيون ٿي وينديون۔ ڪي ماڻهو سلام سان گڏ جنّت المقام ۽ دوزخ الحرام جا الفاظ به وڌائيندا آهن اِهو غلط طريقو آهي، بلڪ ڪي ته مَعاذالله عَزَّوَجَلَّ ايتري تائين بڪندا آهن، اوهان جا ٻار اسان جا غلام. ٭ سلام جو جواب فوراً ۽ ايتري آواز سان ڏيڻ واجِب آهي جو سلام ڪرڻ وارو ٻڌي وٺي۔

ايندڙ هفتيواراجتماع جي بيان جون جھلڪيون

پيارا پيارا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! آئنده هفتيوار اِجتماع جي بيان جو موضوع هوندو”شيطان جي انسان سان دشمني“ يقينًا هر شخص پنهنجي دشمن جي شر کان بچڻ جي ڪوشش ڪندو آهي ۽ دشمن جيترو وڌيڪ مظبوط هجي، ان کان محفوظ رهڻ جي ايتري ئي وڌيڪ احتياط ڪئي ويندي آهي، انسان جو سڀ کان وڌيڪ ۽ خطرناڪ دشمن شيطان آهي. اولياء ڪرام کان وٺي عام انسانن تائين هرهڪ سان دشمني رکندو ۽ انهن کي ڀٽڪائڻ جي لاءِ طرح طرح جي حيلن ۽ بهانن سان ڪم وٺندو آهي، الله پاڪ قرآنِ ڪريم ۾ ڪيترن ئي مقامن تي شيطان جي دشمني جي سڃاڻپ ڪرائي آهي، آئنده هفتي شيطان جي انسان سان دشمني جي متعلق آيتون، مبارڪ حديثون، بزرگانِ دين جا واقعا ۽ شيطان جي وار کي ناڪام بڻائڻ جا طريقا به ٻڌنداسين. تنهنڪري آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ انفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان سان چئو: اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

(1) جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

(2) سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

(3) رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

(4) ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([4])

(5) قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهڙي مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

(6) دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

(1) هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

(2) ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عسا ڪ ر ج 19، ص 155، حديث 4455)
