هڪ زمانو اهڙو ايندو

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا نَبِیَّ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا نُوْرَ اللہ

نَوَيۡتُ سُنَّتَ الۡاِعۡتِكَافِ (مون سنّتِ اعتڪاف جي نيّت ڪئي)

مٺا مٺا اسلامي ڀائرو! جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي اعتڪاف جي نيت ڪري وٺو، ڇوته جيستائين مسجد ۾ رهندؤ اعتڪاف جو ثواب ملندو رهندو۔ ياد رکو! مسجد ۾ کائڻ، پيئڻ، سمهڻ يا سحري، افطاري ڪرڻ، ايستائين جو آبِ زم زم يا دَم ڪيل پاڻي پيئڻ جي به شرعاً اجازت ناهي ، البت جيڪڏهن اعتڪاف جي نيت هوندي ته اهي سڀ شيون ضمناً جائز ٿي وينديون۔ اعتڪاف جي نيت به صرف کائڻ، پيئڻ يا سمهڻ جي لاءِ نه هئڻ گھرجي بلڪ ان جو مقصد الله تعاليٰ جي رِضا هجي۔ ”فتاويٰ شامي“ ۾ آهي: جيڪڏهن ڪو مسجد ۾ کائڻ پيئڻ يا سمهڻ چاهي ته اعتڪاف جي نيت ڪري ۽ ڪجھ دير ذِڪرُاللہ ڪري پوءِ جيڪو چاهي (يعني هاڻي چاهي ته کائي پي يا سمهي سگھي ٿو)

دُرُود شريف جي فضيلت

اُمُّ المؤمنين حضرت سَيدَتُنا عائشه صِدِّيقَه رَضِىَ اللّٰهُ تَعَالٰى عَنْها کان روايت آهي ته نبي رَحمت، شفيعِ اُمّت، محبوبِ ربُّ العزّت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ شفاعت نشان آهي: ” مَنْ صَلّٰي عَلَي يوْمَ الْجُمُعَۃِ کانَتْ شَفَاعَۃٌ لَّه عِنْدِي يوْمَ الْقِيامَۃ (يعني) جيڪو شخص جُمُعي جي ڏينهن مون تي دُرُود شريف پڙھندو ته بروزِ قيامت ان جي شفاعت منهنجي ذِمّه ڪرم تي هوندي۔“ (ڪنز العمال،ڪتاب الاذڪار،الباب السادس في الصلاۃ عليه،الجزء الاول،۱/۲۵۵ ،حديث: ۲۲۳۶)

شافِعِ روزِ جَزا تم پہ کروروں دُرُود دافِعِ جُملہ بَلا تم پہ کروروں دُرُود

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿا. فرمانِ مصطفي ٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِه  “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ .

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس ٭ ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس ٭ ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي جي لاءِ بلند آواز سان جواب ڏيندس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اڄوڪو دؤر قِيامت جي نشانين ۽ ان کان پهريان ظاهر ٿيڻ وارن بيشمار فتنن سان ڀريل آهي۔ هر ايندڙ ڏينهن نون نون فتنن سان گڏ ظاهر ٿي رهيو آهي۔ ڪيتري اَفسوس جي ڳالھ آهي جو جنهن اُمّت جي حقيقي خَير خواھ حُضور نبي ڪريم، محبوبِ ربِّ عظيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

پنهنجي ظاهري حياتي ۾ فتنن جي دَور جي هولناڪين جي باري ۾ فڪر مند رهندا هئا، آزمائشن، پريشانين ۽ فتنن جي هن هولناڪ زماني ۾ منهنجي اُمّت ڪٿي حق واري واٽ تان ڀٽڪي نه وڃي، اڄ انهيءَ اُمّت جا ماڻهو انهن فتنن ۾ پئجي پنهنجن ئي هٿن سان دِين ۽ ديانت، حق ۽ اَمانت ۽ باهمي مَحَبَّت و شَرافت جو سر عام قَتلِ عام ڪري رهيا آهن۔ تنهنڪري حالات هن ڳالھ جي تقاضا ڪري رهيا آهن ته هر طبقي سان تَعَلُّق رکندڙ مسلمانن تائين نبي غيب دان، مڪي مدني سُلطان صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا اهي اِرشاداتِ عالي شان پهچايا وڃن، جن ۾ پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ الله ڪريم جي عطا ڪيل عِلمِ غَيب جي ذريعي 14 سو سال پهريان ئي موجوده ۽ آئنده دور جي فِتنن جي پيشن گوئي ڪندي اسان کي انهن جي ڀڙڪندڙ باھ مان اٿڻ وارن شُعلن کان پنهنجو دامن بچائڻ جي ترغيب ڏياري آهي۔ اچو! هن باري ۾ ڪجھ اَحاديثِ مبارَڪه ۽ ان مان حاصِل ٿيڻ واري نصيحت جا مَدَني گل گڏ ڪريون ٿا، جيئن

گھر وارن جي هٿان هلاڪت

غَيب ڄاڻڻ وارا آقا، مڪي مدني مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ عبرت نشان آهي: ماڻهن تي هڪ زمانو اهڙو ايندو جو ان شخص جي علاوه ڪنهن جو دِين محفوظ نه رهندو، جيڪي پنهنجي دِين کي کڻي (يعني ان جي حفاظت جي خاطر) هڪ جبل کان ٻئي جبل ۽ هڪ غار کان ٻئي غار ڏانهن ڀڄي۔ ان وقت معيشت (روزي وغيره) جو حصول الله پاڪ کي ناراض ڪرڻ بغير نه ٿيندو۔ جڏهن اها صُورتِ حال هوندي ته انسان پنهنجي ٻارن ٻچن هٿان هلاڪت ۾ پئجي ويندو، جيڪڏهن ٻار ٻچا نه هوندا ته والدين جي هٿان ان جي هلاڪت ٿيندي ۽ جيڪڏهن والدين به نه هجن ته ان جي هلاڪت رشتيدارن يا پاڙيسرين جي هٿان ٿيندي۔“ صحابَه ڪرام عَلَيهمُ الرِّضْوَان عرض ڪيو: يارَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! اهو ڪئين ٿيندو؟ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”اهي ان کي غُربت تي عار (شرم) ڏياريندا، ان وقت اهو پنهنجو پاڻ کي هلاڪت جي هنڌن ڏانهن وٺي ويندو.“ (الزهد الڪبير للبيهقي،الجزء الثاني،فصل في ترڪ الدنيا … الخ، ص۱۸۳، حديث: ۴۳۹)

مٺا مٺا اسلامي ڀائرو! بَيان ڪيل حديثِ پاڪ ۾ جهڙي طرح مَردن جي لاءِ عبرت ۽ نصيحت جا مَدَني گل آهن، اتي انهن عَورَتن جي لاءِ به هن ۾ عبرت آهي، جيڪي پنهنجي مڙسن کي انهن جي آمدني تي طرح طرح جا طعنا ڏيندي ڪجھ هن طرح جا تيز تکا جملا چونديون آهن: ”اڙي ڏس! فُلاڻي وٽ ته ڪيئي ڪيئي بنگلا، فيڪٽريون ۽ جائيدادون (Properties) آهن پر اوهان مون کي ڪرائي جي ننڍڙي گھر ۾ ويهاريو آهي، منهنجو ته هتي ساھ ٿو گُھٽجي! مون کي ڪشادو گھر کپي، فُلاڻي کي ته ڏسو هو پنهنجي فيملي سان گڏ عاليشان گاڏين ۾ گُھمي ڦري ٿو پر اوهان مون کي بسن، ٽيڪسين ۽ رڪشن جا ڌڪا کارايو ٿا، فُلاڻي ته لوڊشيڊنگ کان بچڻ جي لاءِ جنريٽر ورتو آهي اوهان گھٽ مان گھٽ يو پي ايس (U.P.S) يا چارجنگ فين ئي وٺي ڏيو، فُلاڻي ته هن عِيد تي پنهنجي ٻارن جي اَمڙ کي ايترن هزارن جو لباس يا سون (Gold) جو سيٽ ٺاهرائي ڏنو آهي پر اوهان مون کي عِيد تي سون جي منڊي يا بريس ليٽ يا واليون ئي وٺي ڏيو ها، فُلاڻي ته پنهنجن ٻارن جي امڙ کي فلاڻي شاپنگ مال مان شاپنگ ڪرائي آهي تنهنڪري مون کي به ڪنهن سٺي شاپنگ مال مان شاپنگ ڪرڻي آهي۔ فُلاڻي کي ڏسو ته ڪيترو خوشحال ٿي ويو آهي، اوهان به ته ڪجھ ڪريو، فُلاڻي جي پگھار لکن ۾ آهي پر اوهان ايترو عرصو ڪم ڪرڻ کان بعد به اتي جا اتي بيٺا آهيو۔“ وغيره جڏهن ته ٻارن جون فرمائشون ان جي علاوه آهن۔ ته روز روز جڏهن اهو شخص فرما ئشن ۽ طعنن کي ٻڌندو آهي ته ذِهني اَذِيت ۽ مايوسي ان کي هر پاسي کان وڪوڙي وٺندي آهي، ان کي ڪجھ سمجھ ناهي ايندو ته آخر هو ڪري ته ڇا ڪري، جيئن ته ان کي ٻارن ٻچن جون فرمائشون به پُوريون ڪرڻون هونديون آهن ان وٽ وسائل به گھٽ هوندا آهن، ان ڪري هو انهن جون اجايون فرمائشون پوري ڪرڻ جي لاءِ حرام و حلال جي پروا ڪرڻ بغير ناجائز ذريعا اختيار ڪري پنهنجي قَبر ۽ آخرت کي داءُ تي لڳائي ڇڏيندو آهي، جيئن ته

حلال و حرام جي مُعامَلي ۾ بي پرواهي

حضرت سَيدُنا ابُوهرَيره رَضِىَ اللّٰهُ تَعَالٰى عَنْه کان روايت آهي ته نبي پاڪ، صاحبِ

لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: يَاْتِى عَلَى النَّاسِ زَمَانٌ،لاَ يُبَالِى الْمَرْءُ مَا اَخَذَ مِنْهُ اَمِنَ الْحَلاَلِ اَمْ مِنَ الْحَرَام يعني ماڻهن تي هڪ زمانو اهڙو ايندو جو ماڻهو کي هن ڳالھ جي ڪا پروا نه هوندي ته ان (مال) ڪٿان حاصل ڪيو؟ حرام مان يا حلال مان۔

(بخاری، کتاب البیوع، باب من لم یبالی من حیث … الخ، ۲/۷، حدیث: ۲۰۵۹)

حڪيم الاُمَّت مفتي احمد يار خان رَحْمَة اللّٰهِ تَعَالٰى عَلَيه هن حديث جي شرح ۾ فرمائن ٿا: ”يعني آخري زماني ۾ ماڻهو دِين کان بي پروا ٿي ويندا، پيٽ جي فڪر ۾ هر طرح ڦاسي ويندا، آمدني وڌائڻ، مال گڏ ڪرڻ جي فڪر ڪندا، هر حرام وحلال وٺڻ تي دلير ٿي ويندا جيئن اڄڪلھ عام آهي۔“ (مراٰۃالمناجیح،۴/۲۲۹)

مٺا مٺا اسلامي ڀائرو! ياد رکو! جن اهل وعيال جي لاءِ اسان ڏينهن رات ڪمايون ٿا، اهي سڀاڻي قيامت جي ڏينهن اسان جي رُسوائي جو سبب بڻجي سگھن ٿا، توڙي جو ماءُ پيءُ گھرواريءَ ٻارن ۽ قَرابتدارن جي حُقُوق جي ادائيگي يقيناً اسان جي ئي ذميداري آهي، پر جيڪڏهن اسان انهن جون خواهشون پُوريون ڪرڻ ۽ انهن جي طعنن کان بچڻ جي لاءِ حرام ۽ حلال جي پروا ڪرڻ بغير مال و دولت گڏ ڪندا رهياسين، مشڪل وقت ۾ انهن کي صبرو شڪر، توڪل وقناعت جو ذهن نه ڏنو ۽ حلال وحرام جي تميز نه سيکاري ته ٿي سگھي ٿو ته سڀاڻي قيامت جي ڏينهن اهي ئي اسان جي خلاف بارگاھ الٰهي ۾ مُقدمو ڪن ۽ اسان جي پڪڙ ٿي وڃي، جيئن ته

بارگاهه الٰهي ۾ دعويٰ

حضرتِ سَيدُنا فقيه ابوللَّيث سَمر قندي رَحْمَة اللّٰهِ تَعَالٰى عَلَيه نَقل ڪن ٿا: مروي آهي ته مرد سان تعلق رکندڙن ۾ پهرين ان جي زال ۽ اولاد آهي. اهي سڀ (يعني زال ۽ ٻار قيامت ۾) الله پاڪ جي سامهون بيهي ڪري عرض ڪندا: ”اي اسان جا ڪريم رب! اسان کي هن شخص کان اسان جو حق وٺي ڏي، ڇاڪاڻ ته هن ڪڏهن اسان کي ديني معاملن جي تعليم نه ڏني ۽ هي اسان کي حرام کارائيندو هو، جنهن جي اسان کي خبر نه هئي.“ پوءِ انهيءَ شخص کي حرام

ڪمائڻ تي ايتري قدر ماريو ويندو جو هن جو گوشت ڇڻي ويندو. پوءِ ان کي ميزان (ساهمي) وٽ آندو ويندو. ملائڪ جبل جي برابر ان جون نيڪيون آڻيندا ته انهيءَ جي خاندان مان هڪ شخص اڳيان وڌي ڪري چوندو: ”منهنجون نيڪيون گهٽ آهن.“ ته هو ان جي نيڪين مان کڻي وٺندو. پوءِ ٻيو اچي ڪري چوندو: ”تو مون کي وياج کارايو هو.“ ۽ ان جي نيڪين مان کڻي وٺندو. اهڙي طرح ان جا گهر وارا انهيءَ جون سڀ نيڪيون کڻي ويندا ۽ هو پنهنجن گهروارن ڏانهن حسرت ۽ مايوسيءَ سان ڏسي چوندو: ”هاڻي منهنجي ڳچي تي اُهي گناهه ۽ ظلم رهجي ويا، جيڪي مون توهان جي لاءِ ڪيا هئا.“ ملائڪ چوندا: ”هي اُهو بدنصيب شخص آهي، جنهن جون نيڪيون هن جا گهر وارا کڻي ويا ۽ هي انهن جي ڪري دوزخ ۾ هليو ويو.“

(الروض الفائق،الباب الثامن في عقوبة قاتل … الخ، ص۴۰۱)

گر تُو ناراض ہوا میری ہلاکت ہوگی

ہائے! میں نارِ جہنّم میں جلوں گا یاربّ!

دَرْدِ سَر ہو یا بُخار آئے تڑپ جاتا ہوں

میں جہنّم کی سزا کیسے سہوں گا یاربّ!

مٺا مٺا اسلامي ڀائرو! غور ڪريو! ان شخص جي بد نصيبي جو ڪهڙو عالَم هوندو جنهن جون سموريون نيڪيون ان جا گھروارا کڻي ويندا ته هو پاڻ ڪنگال ٿي ويندو. تنهنڪري غنيمت سمجھندي غفلت جي ننڊ مان بيدار ٿيڻ گھرجي، الله پاڪ ۽ اُن جي پياري حبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي فرمانبرداري ڪندي پنهنجي اولاد جي دِيني تربيت ڪريو، دُنيوي تعليم جي فِڪر ڪرڻ سان گڏوگڏ پنهنجي اولاد کي قرآنِ ڪريم پڙھڻ به سيکاريو وڃي، وضو ۽ غُسل، پاڪي ۽ ناپاڪي وغيره، اسلامي آداب به سيکاريا وڃن، انهن جا اَخلاق به سنوارڻ جي ڪوشش ڪئي وڃي، الله پاڪ ۽ ان جي پياري مَحبوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي محبت سيکاري وڃي۔ آخرت جو فڪر ڏياريو وڃي، موت جي تياري ڪرائي

وڃي، مختلف گناهن جي باري ۾ ٻڌايو وڃي ۽ انهن گناهن کان بچايو به وڃي۔

ڪٿي ائين نه ٿئي جو اهو وَقت (Time) اچي جنهن جي نشاندهي الله پاڪ جي مَحبوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ فرمائي هئي ته جنهن ۾ سُنّت تي عمل ڪرڻ، دِين تي قائم رهڻ ۽ ان راھ ۾ ايندڙ تڪليفن تي صبر ڪرڻ گھڻو مشڪل هوندو۔ اچو! هن باري ۾ 2 فرمانِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ٻڌو ۽ پنهنجو پاڻ کي سُنّت جو پابند بڻائڻ جي ڪوشش ڪريو:

سُنّتون اپنائڻ وارو ٽانڊي جهلڻ واري وانگر هوندو

(1) اِرشاد فرمايائون: اَلْمُتَمَسِّكُ بِسُنَّتِي عِنْدَ اِخْتِلَافِ اُمَّتِي كَالْقَابِضِ عَلَى الْجَمْر يعني امت جي (فتني ۽) فساد جي وقت منهنجي سنّت تي عمل ڪرڻ وارو باهه جوٽانڊو جهلڻ واري وانگر هوندو۔ (نوادر الاصول، الاصل الثالث عشر،الجزء الاول، ص۶۸، حديث:۸۷)

(2) اِرشاد فرمايائون: يَاْتِي عَلَى النَّاسِ زَمَانٌ الصَّابِرُ فِيْهِمْ عَلٰى دِينِهٖ كَالقَابِضِ عَلَى الجَمْرِ يعني ماڻهن تي اهڙو زمانو ايندو جو انهن ۾ پنهنجي دين تي صبر ڪرڻ وارو ٽانڊو جهلڻ واري جيان هوندو۔ (ترمذي، ڪتاب الفتن،باب ما جاء في النھي عن سب الرياح،۴/ ۱۱۵،حديث: ۲۲۶۷)

حڪيمُ الاُمّت، حضرت مفتي احمد يار خان رَحْمَة اللّٰهِ تَعَالٰى عَلَيه بيان ڪيل حديث پاڪ جي تحت فرمائن ٿا؛ ”اهو زمانو قيامت جي ويجھو هوندو جنهن جي شروعات اڄ ٿي چڪي آهي۔ اڄڪلھ ديندار بڻجي رهڻ مشڪل آهي۔ اڄ ڏاڙهي رکڻ، نماز جي پابندي ڪرڻ مشڪل ٿي ويو آهي۔ وياج کان بچڻ ته تقريبًا ناممڪن ئي آهي۔“ وڌيڪ فرمائن ٿا؛ ”جيئن هٿ ۾ ٽانڊو رکڻ تمام وڏي صبر واري جو ڪم آهي اهڙي طرح ان وقت مخلص، ڪامل مسلمان بڻجڻ سخت مشڪل ٿي پوندو۔“ (مرآۃالمناجيح، ۷/۱۷۲ملخصا)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دِين تي هلڻ وارن جي لاءِ آزمائش

مٺا مٺا اسلامي ڀائرو! مفتي صاحِب حديث پاڪ جي هي شرح پنهنجي دور مطابق ڪئي هئي ۽ هاڻي ته ان کان ڪيئي ڀيرا وڌيڪ بيباڪي ۽ بي راھ روي جو دور هلي رهيو آهي، اها ڳالھ ڪنهن کان لڪيل ناهي ته في زمانه جيڪي اسلامي ڀينرون بُرقعو پائن ٿيون، شَرعي پردو ڪن ٿيون ۽ شريعت و سُنّت مطابق زندگي گذارڻ چاهين ٿيون، انهن جي لاءِ سخت آزمائش آهي، جيڪي اسلامي ڀائر شريعت مطابق جائز لباس پائن ٿا، پنهنجي مٿي تي عمامي شريف جو تاج سجائن ٿا، سُنّتن جا پيڪر بڻجي شريعت وسُنّت جي پابندي ڪرڻ چاهن ٿا، انهن جي لاءِ به سخت آزمائش آهي. اهي اسلامي ڀائر ديانت داري سان پنهنجي فرائض جي منصب کي ادا ڪندي وياج ۽ رِشوت جي نحوست کان بچندي رِزقِ حلال ڪمائڻ چاهين ٿا انهن جي لاءِ به سخت آزمائش آهي۔ اهي عاشقانِ رسول جيڪي شريعت جي دائري ۾ رهندي، ناجائز رسمن کان بچندي شادي جي تقريب ڪرڻ چاهين ٿا، انهن جي لاءِ سخت آزمائش آهي۔ هڪ پاسي ته مُعاشَري (Society) جا فرد انهن ويچارن کي طرح طرح سان ستائن ٿا، انهن جي دل ڏُکائن ٿا ۽ اُنهن جي حوصَله شِڪني ڪن ٿا ۽ ٻي پاسي نيڪ عملن جي بجا آوري ۾ نَفس ۽ شَيطان آڏو ايندا آهن۔

اَميرِاَهلسُنّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه پنهنجي هڪ ڪلام ۾ هنن حالتن جي ڪجھ هن طرح عَڪاسي ڪئي آهي:

تِری سُنّت پہ چلتا ہوں تو طعنے لوگ دیتے ہیں شہنشاہِ مدینہ! نَفْس بھی مجھ کو ستاتا ہے

پاڻ اِنصاف ڪريو!

مٺا مٺا اسلامي ڀائرو! انسان کان غلطيون ٿينديون رهن ٿيون ليڪن افسوس جي ڳالھ هي آهي ته جيڪڏهن ڪنهن مذهبي ۽ دين تي عمل ڪرڻ واري

شخص کان ڪا غلطي ٿي وڃي ته ان کي مشهور ۽ شرمسار ڪيو ويندو آهي، اهڙي طرح اسلام ۽ شريعت و سنّت تي هلڻ وارن کي بدنام ڪيو ويندو آهي، جنهن سبب ماڻهن جي دلين ۾ انهن لاءِ نفرت پيدا ٿيندي آهي، نتيجي ۾ عوام انهن کان مُتَنَفِّر ٿيندو آهي، انهن کي حقارت جي نگاھ سان ڏسندو آهي ۽ پنهنجي دنيا ۽ آخرت برباد ڪندو آهي۔ اهڙي مشڪل حالتن ۾ دين تي قائم رهڻ ۽ نماز ۽ سنت جي پابند اسلامي ڀائرن ۽ اسلامي ڀينرن کي وڌيڪ محتاط رهڻ گھرجي۔

ياد رکو: معمولي بي احتياطي به ڪڏهن ڪڏهن وڏي نقصان جو سبب بڻبي آهي جيڪڏهن واقعي اوهان مدني ڪم ڪرڻ چاهيو ٿا ته جيستائين شريعت حڪم نه ڏئي هرگز ڪنهن سُنّي کي پنهنجو مخالف نه بڻايو۔ اوهان جي هلڻ ڦرڻ کي ماڻهو غور سان ڏسندا هوندا تنهنڪري ڪو اهڙو ڪم نه ڪيو جو دعوتِ اسلامي تي آڱر کڄي۔“ بهرحال احتياط جي باوجود مشڪلاتون هجن، ماڻهو طعنا ڏين يا گھر وارا سنتن جي مطابق زندگي گذارڻ ۾ رڪاوٽون وجهن ته گھٻرائڻ نه گھرجي جو جنهن ڪم ۾ وڌيڪ مشڪلاتون هونديون آهن ته ان جو ثواب به زياده هوندو آهي۔ اچو انهيءَ سلسلي ۾ 2 فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ملاحظه فرمايو ۔

(1) فرمايائون؛ افضل ترين عبادت اها آهي جنهن ۾ زحمت زياده هجي۔ (ڪشف الخـفا ء،حرف الھمزة مع الفاء، ۱/ ۱۴۱)

(2) فرمايائون: جيڪو امت جي فساد جي وقت منهنجي سنّت کي مضبوطي سان وٺي ان کي سؤ شهيدن جو ثواب ملندو۔ (الزھد الڪبير للبيھقي،ص۱۱۸، حديث:۲۰۷)

حضرت مفتي احمد يار خان رَحْمَة اللّٰهِ تَعَالٰى عَلَيه انهيءَ حديث پاڪ جي شرح ۾ فرمائن ٿا؛ ”جڏهن ته شهيد هڪ دفعو تلوار جو زخم کائي پار ٿي ويندو آهي پر هي الله پاڪ جو ٻانهو سڄي عمر ماڻهن جا طعنا ۽ زبانن جا گھاءُ سهندو رهندو

آهي۔ الله پاڪ ۽ رسول (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) جي خاطر سڀ ڪجھ برداشت ڪندو آهي۔ ان جو جهاد، جهادِ اڪبر(نفس جي خلاف جهاد) آهي جيئن هن زماني ۾ ڏاڙهي رکڻ، وياج کان بچڻ وغيره۔“ (مرآۃالمناجيح،۱/۱۷۳)

ظاهر ۾ دوست ،باطن ۾ دشمن

مٺا مٺا اسلامي ڀائرو! ايندڙ زماني ۾ ظاهر ٿيڻ وارن فتنن جي باري ۾ هڪ حديث شريف ۾ آهي ته نبي پاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو؛ ” آخري زماني ۾ ڪجهه اهڙا ماڻهو به ٿيندا جيڪي ظاهر ۾ دوست هوندا ۽ باطن ۾ (اندروني طور) دشمن“ عرض ڪيو ويو؛ ”يارسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! اهو ڇاجي ڪري هوندو؟“ ارشاد فرمايائون؛ ” (پنهنجي دنيا کي بهتر بڻائڻ جي لاءِ) هڪ ٻئي جي طرف رغبت (لالچ) ۽ هڪ ٻئي کان ڊڄڻ جي ڪري هوندو۔“

حضرت سَيدُنا اِبراهيم تَيمِي رَحْمَة اللّٰهِ تَعَالٰى عَلَيه فرمائن ٿا: مان مَوت ۽ (مرڻ کان بعد هڏين جي) بوسيدگي کي ياد ڪرڻ جي لاءِ ڪثرت سان قَبرِستان ۾ ايندو ويندو هئس، هڪ رات مان قَبرِستان ۾ هئس ته مون تي ننڊ غالِب اچي وئي ۽ مان سمهي پيس، مون خواب ۾ هڪ کليل قَبر ڏٺي ۽ هڪ چوڻ واري کي اهو چوندي ٻڌو: اها زنجير پڪڙيو ۽ ان جي وات ۾ داخِل ڪري ان جي شَرَمگاھ مان ڪڍو۔ ته اهو مُردو چوڻ لڳو: اي ربِّ ڪريم! ڇا مون قرآن ناهي پڙھيو؟ ڇا مون تنهنجي حُرمَت واري گھر جو حج (Hajj) ناهي ڪيو؟ پوءِ هو اهڙي طرح هڪ کان بعد ٻي نيڪي ڳڻائڻ لڳو ته مون ٻڌو: تون ماڻهن جي اڳيان اهي عمل ڪندو هئين پر جڏهن تون تنهائي ۾ هوندو هئين ته نافرمانين جي ذريعي مون سان مُقابَلو ڪندو هئين ۽ تو منهنجو ڪجھ خيال نه ڪيو۔

مجلس ذرائع آمد و رفت

مٺا مٺا اسلامي ڀائرو! ٻڌو اوهان! الله پاڪ جي عِلاوه ڪنهن ٻي جي لاءِ عمل ڪرڻ ڪيتري قدر تباھ ڪن آهي جو ان جي نحوست جي ڪري وڏيون وڏيون نيڪيون به ضائع ٿي وينديون آهن۔ تنهنڪري اسان کي گھرجي ته اسان پنهنجي ظاهر

۽ باطِن کي هڪ رکون، پنهنجي دل ۾ حقيقي خوفِ خدا پيدا ڪريون، رِيا ڪاري جي تباھ ڪارين کان پنهنجو پاڻ کي بچايون، هر عمل صِرف ۽ صِرف رِضاءِ اِلٰهي جي لاءِ ئي ڪيون ۽ اها سوچ ماڻڻ جي لاءِ عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي جي مَدَني ماحول سان وابسته ٿي وڃون۔

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي تحت ڪم و پيش 104 شعبا قائم آهن، انهن مان هڪ ”مجلسِ ذرائع آمد و رفت“ به آهي، هن شعبي جو ڪم ائيرلائنز، ائير پورٽس، ريلويز، پبلڪ ٽرانسپورٽ، پرائيوٽ ٽرانسپورٽ، گُڊز اِسٽيشنز ۽ انهن سان لاڳاپيل مُختلِف اِسٽينڊز (مثال طور بس، ٽرڪ، ٽيڪسي، موٽر سائيڪل ۽ سائيڪل اسٽينڊ، تمام قسم جي ورڪشاپس، اسپئير پارٽس دوڪان وغيره) ۽ مارڪيٽن ۽ شورومن (مثال طور ڪار، رڪشه، موٽر سائيڪل، سائيڪل) سان وابسته اسلامي ڀائرن ۾ عاشقانِ رسول جي مَدَني تحريڪ دعوتِ اسلامي جي مَدَني پيغام کي عام ڪرڻ ۽ اُنهن کي دعوتِ اسلامي جي مَدَني ماحول سان وابسته ڪندي هن مَدَني مَقصَد ”مون کي پنهنجي ۽ سڄي دنيا جي ماڻهن جي اِصلاح جي ڪوشش ڪرڻي آهي، اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ “ جي مطابق زندگي گذارڻ جو مَدَني ذِهن ڏيڻ آهي.

سان ڪجھ ڪم ناهي ۔ (شعب الايمان، باب في الصلوات، فصل المشي الي المساجد،۳/۸۶، حديث:۲۹۶۲)

حڪيمُ الاُمَّت حضرت مفتي احمد يار خان رَحْمَة اللّٰهِ تَعَالٰى عَلَيه هن حديث پاڪ جي شرح ۾ فرمائن ٿا؛ ”يعني الله پاڪ انهن تي ڪرم نه ڪندو، نه ته ربّ ڪريم کي ڪنهن ٻانهي جي ڪا ضرورت ناهي، هو ضرورتن کان پاڪ آهي۔ (مِرآةالمناجيح،۱/۴۵۷)

مسجد ۾ دنياوي ڳالهيون ڪرڻ جو وبال

ياد رکو: دنيا جي ڳالهين جي لاءِ مسجد ۾ وڃي ويهڻ حرام آهي۔ مسجد ۾ دنيا جو ڪلام نيڪين کي ائين کائيندو آهي جيئن باھ ڪاٺي کي۔ اهو مُباح (جائز) ڳالهين جو حڪم آهي پوءِ جيڪڏهن ڳالهيون جيڪڏهن بُريون آهن ته ان جو وبال ڪيترو هوندو، ٻئي سَخت حرام دَر حرام، مُوجِبِ عذابِ شديد (سخت عذاب جو سبب) آهي۔ (فتاويٰ رضويه، ۸/۱۱۲)

اهو به ياد رکو ته مسجد ۾ شور ۽ شَر ڪرڻ حرام آهي ۽ دُنيوي ڳالھ جي لاءِ مسجد ۾ ويهڻ حرام، ۽ نماز جي لاءِ وڃي دُنيوي تَذڪرو مسجد ۾ مَڪرُوه ۽ وُضو ۾ بي ضرورت دُنيوي گفتگو نه ڪرڻ گھرجي، ۽ غِيبَت ڪرڻ وارن ۽ تُهمَت لڳائڻ وارن مُنافِقن مُفسِدن (خرابي پيدا ڪرڻ وارن) کي ڪڍائڻ تي قادِر هجي ته ڪڍائي ڇڏجي، جڏهن ته فِتنو نه ٿئي نه ته پاڻ انهن وٽان اٿي وڃي۔ (فتاويٰ رضويه، ۸/۱۱۲)

فتنن کان ڪيئن بچو وڃي؟

مٺامٺا اسلامي ڀائرو! بيشڪ اڄوڪو دور فتنن سان ڀريل آهي، هر گزرندڙ

ڏينهن سان گڏ ئي هڪ نئون فتنو ڏسڻ اچي ٿو، هاڻي سوال اهو پيدا ٿئي ٿو ته اسان انهن فتنن کان پنهنجي پاڻ کي ڪيئن بچائي سگهون ٿا؟ پوءِ اچو! موجوده دور جي فتنن کان بچڻ جي لاءِ 7 طريقا پيشِ خدمت آهن، ٻڌو ۽ عمل جي ڪوشش ڪيو.

(1) فتنن کان حفاظت ۽ ايمان جي سلامتي جي لاءِ چڱي صحبت اختيار ڪيو. هلندڙ زماني ۾ عاشقانِ رسول جي مدني تحريڪ دعوت اسلامي چڱي صحبت ماڻڻ جو هڪ بهترين ذريعو آهي، اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوت اسلامي مسلمانن کي فتنن کان بچائيندي، انهن کي ايمان جي سلامتي جو ذهن ڏيندي ۽ انهن کي حقيقي معنٰي ۾ سنت جو آئينه دار بڻائيندي آهي. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ هن مدني ماحول کان وابسته ٿيڻ جي برڪت سان هاڻي تائين بيشمار ماڻهن جي اصلاح ٿي آهي، نه صرف انهن جو بلڪ انهن جي نسلن جو ايمان محفوظ ٿي ويو. (2) پنهنجي پاڻ کي فتنن کان بچائڻ جي لاءِ ضرورت جي علاوه گهر کان ٻاهر نڪرڻ کان بچو جو انهيء ۾ عافيت آهي. ڪڏهن ڪڏهن انسان پنهني نفس تي اعتماد ڪندي بغير ضرورت گهر کان ٻاهر نڪري پوندو آهي پر ناواقفي ۾ ڪنهن نه ڪنهن فتني جو شڪار ٿي گمراهي جي گهري کڏ ۾ ڪري پوندو آهي، تنهنڪري هن معاملي ۾ پنهنجي نفس تي بلڪل به اعتماد نه ڪرڻ گهرجي ڇو ته نفس تي اعتماد ڪرڻ وارو وڏي ڪوڙي تي اعتماد ڪندو آهي. (غيبت جون تباهه ڪاريون، 254) (3) هن نئين دور ۾ سوشل ميڊيا فتنن کي پکيڙڻ جو هڪ خطرناڪ هٿيار بڻجندو پيو وڃي، اسلام جون دشمن قوتون پنهنجي مقصدن جي خاطر ان جو غلط استعمال ڪري ظاهر طور تي اسلام جي شعار جو مذاق اڏائي رهيون آهن، سوشل ميڊيا جي ذريعي اسلام جي خلاف سازشون ڪيون پيون وڃن، انهيء جي ذريعي مسلمانن کي بي حيائي ۽ فحاشي جي گند ۾ ڌڪيو پيو وڃي، تنهنڪري سوشل ميڊيا جي تباهه ڪارين کان پنهنجي پاڻ کي ۽ پنهنجي نسلن کي بچائي رکو، بالفرض ان کان بغير گزارو نه هجي ته پوءِ دارالافتا اهلسنت کان ان باري ۾ شرعي رهنمائي ضرور

حاصل ڪيو. (4) دعوت اسلامي جي تحت مختلف مسجدن ۾ روزانو عشاء جي نماز کان پوءِ لڳڻ واري مدرسة المدينه بالغان ۾پڙهڻ پڙهائڻ به فتنن کان بچڻ ۾ گهڻو معاون ۽ مددگار آهي، اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مدرسة المدينه بالغان جي برڪت سان عاشقانِ رسول جي صحبت ميسر ايندي، مسجدن ۾ ويهڻ جو ثواب ملندو ۽ ڪيئي شيطاني فتنن کان اسان جي حفاظت رهندي. (5) فتنن کان بچڻ بچائڻ جي لاءِ مدني چينل ڏسڻ ڏيکارڻ جو ذهن بڻايو. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مدني چينل اهو واحد چينل آهي جنهن تي نشر ٿيڻ وارو سلسلو عين شريعت جي مطابق هوندو آهي، مدني چينل عشقِ رسول جا جام پياريندو آهي، مدني چينل تي درست اسلامي تعليم کي اجاگر ڪيو ويندو آهي، مدني چينل مقدس هستين جو ادب سيکاري ٿو، تنهنڪري پاڻ به مدني چينل ڏسو ۽ گهروارن کي به ڏسڻ جي ترغيب ڏياريو، (6) مدني قافلن ۾ سفر ڪرڻ به فتنن کان بچڻ جو بهترين ذريعو آهي. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ مدني قافلن ۾ عاشقانِ رسول جي صحبت ملندي آهي، سکڻ سيکارڻ جي حلقن ۾ شرڪت جي سعادت نصيب ٿيندي آهي، فڪرِ آخرت جو جذبو ملندو آهي ۽ موت جي تياري جو ذهن بڻجندو آهي، تنهنڪري دعوت اسلامي جي تحت سفر ڪرڻ وارن مدني قافلن ۾ سفر کي پنهنجو معمول بڻايو اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ فتنن کان حفاظت رهندي. (7) فتنن کان حفاظت جي لاءِ ذيلي حلقي جي 12 مدني ڪمن کي پنهنجو لازم عمل بڻايو. اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ ان جون برڪتون پنهنجي کليل اکين سان ڏسڻ نصيب ٿينديون. الله ڪريم هر طرح جي فتني کان اسان جي حفاظت فرمائي، اسان کي ۽ اسان جي آئنده نسلن کي به انهن فتنن کان محفوظ فرمائي.

اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

مٺامٺا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّم َ جو فرمان جنّت نشان آهي:

جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو. (مشکاۃ الصابیح،کتاب الایمان،باب الاعتصام بالکتاب والسنۃ،۱/۹۷،حدیث:۱۷۵)

سُنَّتیں عام کریں دِین کا ہم کام کریں نیک ہوجائیں مُسَلمان مدینے والے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

سُرمو لڳائڻ جون سنتون ۽ آداب

اچو! شيخ طريقت،اميرِ اَهلسُنّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي رسالي 101 مدني گل صفحي نمبر27 تان سُرمو لڳائڻ جون سنتون ۽ آداب ٻڌون ٿا: فرمانِ مُصطَفٰي صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ آهي ؛ ”سڀني سرمن مان بهتر سرمو ”اِثمِد“ آهي جو اهو نظر کي روشن ڪري ۽ پلڪون وڌائي ٿو“۔ (ابن ماجه،۴/ ۱۱۵ ،حديث:۳۴۹۷)

٭ پٿر جو سرمو استعمال ڪرڻ ۾ حرج ناهي ۽ ڪارو سرمو يا ڪجل زينت جي قصد (يعني زينت جي ارادي) سان مرد کي لڳائڻ مڪروهه آهي ۽ زينت جو ارادو نه هجي ته ڪراهت ناهي۔ (فتاويٰ هنديه،۵ /۳۵۹) ٭ سرمو سمهڻ مهل استعمال ڪرڻ سنّت آهي۔ (مرآۃالمناجيح ،۶/۱۸۰) ٭ سرمي استعمال ڪرڻ جي ٽن منقول طريقن جو خلاصو پيش خدمت آهي: (1) ڪڏهن ٻنهي اکين ۾ ٽي ٽي سرائيون (2) ڪڏهن ساڄي اک ۾ ٽي ۽ کاٻي ۾ ٻه، (3) ته ڪڏهن ٻنهي اکين ۾ ٻه ٻه ۽ وري آخر ۾ هڪ سرائي کي سرمي واري ڪري ان کي واري واري سان ٻنهي اکين ۾ لڳايو۔ (شعب الايمان ج 5 ص 218، 219) اهڙي طرح ڪرڻ سان اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ٽنهي طريقن تي عمل ٿيندو رهندو۔ ٭ تڪريم جا جيترا به ڪم هوندا هئا اسان جا پيارا آقا صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ ساڄي پاسي کان شروع ڪندا هئا، تنهنڪري پهريان سرمو ساڄي اک ۾ پوءِ کاٻي اک ۾ پايو۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

طرح طرح جون هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب“ سنتیں ۽ آداب ” هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي.

’’قافلے‘‘ کو جو ہر وقت تیّار ہے مرحبا اس سے عطارؔ کو پیار ہے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([1])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿5﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام

رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

﴿2﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)
