يزيد پليد جي فتنه انگيزين جي باري ۾ سرڪارِ نامدار،غَيبن تي خبردار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اِنهي پيشنگوين جي سبب مشہور صحابي رسول حضرت سَيدُنا اَبُو ہُرَيرہ رَضِىَ اللهُ تَعَالٰی عَـنْهُ پنهنجي وصال کان پهريان هي دُعا گھُرندا هئا: اَعُوْذُ بِاللہ مِنْ رَأْسِ السِّتِّیْنَ وَاِمَارَۃِ الصِّبْیَان ”يعني اي اللہ عَزَّوَجَلَّ! آئون توکان پناھ ٿو گُھران، 60 ہجري جي آغاز ۽ ڇوڪرن جي حڪومت کان۔“ بس اوهان جي هي دُعا قبول ٿي ۽ 59 ہجري ۾ (بمقام مدينهء طَيِّبه ۾) سندن جو اِنتقال ٿيو (۽ ٺيڪ 60 ہجري ۾ يزيد پليد مَسنَدِ سَلطَنَت تي نمودار ٿيو۔) ([2])

صَدرُ الافَاضِل حضرت علّامہ مولانا مفتي سَيد محمد نعيمُ الدّين مُراد آبادي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه يزيد پليد جي حضرت سيدنا امام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ سان دُشمني جو سبب بيان ڪندي ۽ شہادتِ امام حُسَين کان پوءِ يزيد جي طرف کان مڪّي ۽ مديني شريف جي مسلمانن تي ٿيڻ واري ظُلم ۽ زِيادَتي جو تَذڪرو ڪندي ارشاد فرمائن ٿا ته حضرت امام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو وُجُودِ مُبارَڪ يزيد جي آزادين جي لاءِ هڪ زبردست مُحتَسب (يعني حساب وٺڻ وارو) هو، اهو ڄاڻيندو هو ته امام حسين (رَضِىَ اللهُ تَعَالٰی عَـنْهُ) جي زمانهء مُبارَڪ ۾ ان کي کُلي ڪري کيڏڻ جو موقع نه ملندو ۽ ان جي ڪنهن به اُبتي حرڪت ۽ گمراہي تي حضرت امام رَضِىَ اللهُ تَعَالٰی عَـنْهُ صبر نه فرمائيندا، ان کي نظر ايندو هو ته امام رَضِىَ اللهُ تَعَالٰی عَـنْهُ جهڙو دِين دار جو اثر ہروقت ان جي مٿي تي گھمي رہيو آهي، انهي سبب جي ڪري اهو اڃا به وڌيڪ حضرت امام جي جان جو دشمن هو ۽ انهي ڪري حضرت امام رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي شہادت ان جي لاءِ باعثِ مَسَرَّت ہوئي۔ حضرت امام رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو انتقال فرمائڻ هو، يزيد کُلي ڪري (يعني بالڪل آزاد ٿي ويو) مزيد ڪيترن ئي نمونن ۽ قِسمن جي گناھن جو بازار گرم ٿي ويو، حرام ڪاري، ڀاءُ ڀيڻ جو نڪاح، سُود، شراب، اعلانيه رائج ٿيا، نمازن جي پابندي ختم ٿي وئي، بغاوت ۽ سرڪشي اِنتہا کي پهچي، خَباثَت کي ايتري قدر مضبوط ڪيو جو مُسلِم بن عُقبه کي ٻارنهن ہزار (12000) يا ويھ ہزار (20,000) جر وڏي لشڪر سميت مدينهء طِيبَه تي چڙھائي جي لاءِ موڪليو۔ هي 63 ہجري جو واقعو آهي۔ ان نامراد

لشڪر مدينهء طِيبَه (زَادَھَا الـلّٰـهُ شَرَفًاوَّتَعۡظِیۡمًا) ۾ اهو طُوفان برپا ڪيو جو اللہ جي پناھ! ، قتل، غارت ۽ طرح طرح جا ظلمَ، رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَصَحْبِہٖ وَ بَارَک وَسَلَّم جي ساٿين تي ڪيا۔ اتي جي رهڻ وارن جا گھر لُٽي ورتا، ست سو (700) صحابه عَـلَيْهِمُ الرِّضْوَان کي شہيد ڪيو ۽ ٻين عام شهرين کي مِلائي ڪري ڏھ ہزار (10,000) کان وڌيڪ شہيد ڪيا ، نوجوانن کي قيد ڪري ورتو، اهڙيون اهڙيون بد تميزيون ڪيون، جن جو ذڪر ڪرڻ ناگوار آهي۔ مسجدِ نَبَوِي شريف (عَـلٰي صاحِبِھَاالـصَّلٰوة ُ وَالـسَّلَام) جي ٿنڀَ ۾ گھوڙا ٻَڌيا، ٽن ڏينهن تائين مسجد شريف ۾ ماڻهو نماز کان مُشَرَّف نه ٿي سگھيا۔ صرف حضرت سَعِيد بن مُسَيب رَضِىَ اللهُ تَعَالٰی عَـنْهُ مجنون بڻجي ڪري اُتي حاضر رهيا۔ حضرت سَيدُنا عبدُ اللہ بن حَنظَلَه رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا جن فرمايو ته يزيدين جون بُريون حرڪتون ان حد تائين پہچي ويون جو اسان کي خوف ٿيڻ لڳو ته ڪٿي اِن بدڪارين جي سبب آسمان کان پٿر نه وسن۔ (الصواعق المحرقہ ،الباب الحادی عشر،الخاتمہ فی بیان اعتقاد اہل السنۃ، ص۲۲۱ملخصاً) پوءِ هي شرير لشڪر، مَڪهء مُڪرَّمَه جي طرف روانو ٿيو، رستي ۾ امِيرِ لشڪر مري ويو ۽ ٻيو شخص ان جي قائم مقام ڪيو ويو۔ مَڪهء مُعَظَّمَه (زَادَھَا الـلّٰـهُ شَرَفًاوَّتَعۡظِیۡمًا) پہچي ڪري اُن بي دِينن مِنجَنِيق (يعني مِنجَنِيق پٿر اڇلائڻ جو اوزار(گليل) ہوندو آهي جنهن سان پٿر اڇلائي ڪري ماريو ويندو آهي ان جو نشانو وڏو زبردست ۽ پَري تائين مار ڪندو آهي) ان سان پٿرن جي بارش ڪئي۔ ان پٿرن جي وَسڪاري سان حَرَم شريف جو صحنِ مُبارَڪ پٿرن سان ڀرجي ويو۽ مسجدِ الحرام جا ٿنڀا ٽٽي پيا ۽ ڪعبهء مُقَدَّسه جي غِلاف شريف ۽ ڇت کي اُن بي دِينن ساڙي ڇڏيو۔ انهي ڇت ۾ ان دُنبي جا

سڱ به تَبَرُّڪ جي طور تي محفوظ هئا، جيڪي حضرت سَيدُنا اسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي فِديي ۾ قُرباني ڪيا ويا هئا، اهي (سڱ) به سڙي ويا، ڪعبهء مُقَدَّسه ڪيترن ڏينهن تائين بي لباس (بي غلاف) رہيو ۽ اُتي جا رهواسي يزيدي لشڪر جي طرف کان پہچڻ واري سخت مُصيبت ۾مبُتلا رهيا۔ (خلاصہ سوانح کربلا،ص ۱۷۷ تا۱۷۹ ،ملخصاً)

مٺا مٺا اسلامي ڀائرو! يزيد پليد جيستائين زِندھ رہيو ظُلم ۽ سِتَم ٿيندا رهيا۔ ان جي پوري زندگي بي رحمي جي افسوس ن اڪ داستان آهي، يزيد جا ہٿ مڪّي ۽ مديني ۽ شُہداءِ ڪربلا جي مظلومن جي رت سان رنگيل آهن، اِقتِدار جي ہَوَس (لالچ) ان کي چَريو ۽ اَفرادي قوّت ان کي مَغرُور بڻائي ڇڏيو هو، هئڻ ته ائين گھرجي ها ته اهو راڪبِ دوشِ مُصطَفٰي،جِگر گوشهء مُرتَضٰي، دِلبَندِ فاطمه، سُلطانِ ڪربلا، سَيدُ الشُّہَدا، اِمامِ عالي مقام، اِمامِ عرشِ مقام،امامِ ہُمّام، اِمامِ تِشنه ڪام جي فضيلتن ۽ ڪمالن جي مُتَعَلِّق فرامينِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي پيشِ نظر رکي ڪري اُن جي قَدر و مَنزِلَت جو اِعتِراف ڪري ها، اُنهن جي ۽ اُنهن جي ساٿين جي خدمت ڪري جنّت حاصل ڪري وٺي ها، پر آھ! ان ته شيطانِ لَعِين ۽ نَفسِ اَمّارہ جي غُلامي جو طَوق پنهنجي گلي ۾ وجھي رکيو ۽ مُسلسل بي انصافي جو مُظاہَرو ڪندو رہيو، آخرڪار پنهنجي ناپاڪ منصوبن تي عَمَل ڪندي ان ظالِم ۽ جابِر ۽ فاسق وفاجِر گُلشنِ اہلِ بَيت کي جنهن بي دَردي سان اُجاڙيو ۽

پياري آقا، مڪي مدني مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي نازُڪ ۽ سهڻن گُلن کي جنهن بي دَردي سان چيڀاٽيو ان جي تصوُّر سان ئي دل لرزي پوي ٿو ۽ اکيون آليون ٿيون وڃن۔

ان گُستاخن، بي اَدَبن ۽ غَدّارن ان ڳالھ جي ڪا پرواھ نه ڪئي ته نبي اَڪرَم،نُورِ مُجسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن حضرت سَيدُنا اِمام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ سان پنهنجي مَحَبَّت ۽ اُن جي عظمت ۽ فضيلت کي ڪيتري قدر تاڪيد سان گڏ بيان فرمايو: هُمَا رَيْحَانَتا ی َ مِنَ الدُّنْيا ، يعني حَسَن ۽ حُسَين (رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا) دُنيا ۾ منهنجا ٻه(2) گُل آهن۔ ([1]) مزيد ارشاد فرمايو: اِنَّ الْحَسَنَ وَالْحُسَيْنَ سَيِّدا شبابِ اَهْلِ الْجَنَّة ، يعني حَسَن ۽ حُسَين (رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا) جنّتي جوانن جا سردار آهن۔ ([2])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

سُلطانِ ڪربلا، حضرت سَيدُنا اِمامِ عالي مقام، اِمام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ جيئن ته تمام گھڻا رَحم دل هئا، تنهنڪري پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ يزيد پليد جي ظالم ۽ جابر ساٿين کي ميدانِ جنگ ۾ دعوتِ فڪر ڏيندا رهيا، گڏوگڏ پنهنجي اَہَمّيت ٻُڌائي ڪري اُنهن کي جنگ و جِدال ۽ ظُلم و سِتَم کان باز اچڻ جي مُسلسل نصيحت فرمائيندا رهيا، جيئن ته ميدانِ ڪربلا ۾ حُجّت پوري ڪرڻ جي لاءِ حضرت سَيدُنا امام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ

جن پنهنجي گھوڙي تي سُوار ٿي ڪري يزيدي لشڪر جو رُخ ڪيو ۽ پوءِ پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جن ايتري بلند آواز ۾ سڏ ڏنو، جنهن کي سڀني ماڻهن ٻڌو، جيئن ته ارشاد فرمايو: اي انسانو! منهنجي ڳالھ ٻڌو ۽ جلد بازي جو مُظاہَرو نه ڪريو، ايستائين جو آئون اوهان کي ان شيء جي مُتَعَلّق نصيحت نه ڪريان، جيڪا مون تي لازِم ٿي چڪي آهي ۽ پنهنجي اچڻ جو عُذر بيان نه ڪري وٺان۔ بس جيڪڏهن توهان منهنجو عُذر قبول ڪري وٺو، منهنجي ڳالھ جي تصديق ڪريو ۽ منهنجي باري ۾ اِنصاف سان ڪم وٺو ته توهان ان معاملي ۾ بامُراد ٿي ويندا ۽ توهان کان منهنجي مُتَعَلّق ڪو مُواخَذو (يعني سوال) به نه ٿيندو۔ ہا! جيڪڏهن توهان منهنجو عُذر قبول نه ڪندؤ ته ٻڌو! پوءِ هي آياتِ مُبارَڪه تلاوت فرمائي:

فَاَجۡمِعُوۡۤا اَمۡرَکُمۡ وَ شُرَکَآءَکُمۡ ثُمَّ لَا یَکُنۡ اَمۡرُکُمۡ عَلَیۡکُمۡ غُمَّۃً ثُمَّ اقۡضُوۡۤا اِلَیَّ وَ لَا تُنۡظِرُوۡنِ ﴿۷۱﴾

ترجمو ڪنز الايمان؛ پوءِ توهان سمورا پنهنجي ڪوڙن معبودن سميت پنهنجو ڪم پڪو ڪيو ته پوءِ توهان جي ڪم ۾ ڪو چگهه توهان تي نه رهي پوءِ جيڪو ٿي سگهيوَ منهنجي واسطي فيصلو ڪيو ۽ مون کي مهلت نه ڏيو

ڪو مال ضائع ڪيو آهي جو ان جي بدلي مال چاهيو ٿا يا توهان کي پنهنجي زخمين جو قِصاص درڪار آهي (آخر ڪهڙي شيء جو بدلو چاهيو ٿا) ؟ اهي بدبخت خاموش رهيا، پاڻ فرماياؤن: اي شَبث بن رِبۡعي، اي حجّار بن اَبۡجر، اي قَيس بن اَشۡعث، اي زيد بن حارث! ڇا توهان مونکي هي خط موڪلي نه گهرايو هو؟ انهن صاف انڪار ڪري ڇڏيو ۽ چيو اسان ته ايئن نه ڪيو هو. پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمايو: جي ها خدا عزوجل جو قسم توهان ئي ته ائين ڪيو هو. پوءِ فرماياؤن: اي انسانو! جيڪڏهن توهان منهنجي بيعت ڪرڻ پسند نه ٿا ڪيو ته مون کي ڇڏي ڏيو ته مان ڪنهن محفوظ جڳهه ڏانهن هليو وڃان. بدنصيب قَيس بن اَشۡعث چيو، توهان ابنِ زياد بدنهاد جو حڪم مڃي وٺو (ته توهان کي ڇوٽڪارو ملي سگهي ٿو) پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمايو الله عزوجل جو قسم مان هرگز اُن جي بيعت نه ڪندس. الله جا ٻانهؤ مان پنهنجي ۽ توهان جي ربّ جي پناهه گهران ٿو هن کان ته توهان مونکي سنگسار ڪيو. مان توهان جي ۽ پنهنجي ربّ جي پناهه گهران ٿو هر ان مُتَڪَبّر کان جيڪو حساب جي ڏينهن تي يقين نه ٿو رکي. (الڪامل في التاريخ، سنة احدي و ستين ذڪر مقتل الحسين،ج 3،ص 418)

مٺا مٺا اسلامي ڀائرو! ياد رکو! جيڪڏهن يزيدِ پليد پاڻ ابِن زياد نِهاد جي فوج ۾ شامل نه رهيو مگر شهيدانِ و اسيرانِ ڪربلا تي جيتري قدر ظلم۽ جفا ڪيو ان ۾ ابن زياد بدنهاد کي نه صرف يزيد پليد جي ڀرپور رِضا مندي ۽ هن جي مڪمل تائيد و حمايت حاصل رهي بلڪه هن فاسِق و فاجر شخص جي حڪم تي ظالمن جي سردار ابن زياد بد نهاد ۽ هن جي لشڪرن اهل بيت اطهار جي سرِ عام گستاخيون ڪيون ۽ سخت اُس ۾ ميدان ڪربلا جي سر زمين کي جان نثارن جي مبارڪ خون ۾ رنگين ڪيو، لهٰذا يزيد جهڙي پليد بدبخت کي بي قصور قرار ڏيڻ، اهل بيت اطهار رضوان الله تعالي عليهم اجمعين سان غداري ، محبان اهل بيت جي دل آزاري، الله عزوجل ۽ هن جي رسول صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم جي ناراضگي ۽ پڻ الله عزوجل جي قهر ۽ غضب ۽ جنّهم جي حقداري جو سبب آهي. هن معاملي ۾ خُلفاءِ راشيدن جو طرزِ عمل اسان جي لاءِ مشعلِ راهه آهي، اهي حضرت اهل بيت اطهار بالخصوص امام عالي مقام حضرت سيدنا امام حسين

اعلي حضرت، امام اهلسنت مولانا شاهه احمد رضا خان عَـلَيْهِ رَحْمَةُ الـرَّحْمٰنْ جي بارگاهه ۾ جڏهن يزيد جي باري ۾ سوال ڪيو ويو ته يزيد کي پليد چوڻ شرعًا جائز آهي يا نه، پڻ ان جي نالي سان گڏ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه چوڻ درست آهي يا نه؟ ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ارشاد فرماياؤن؛ يزيد بيشڪ پليد هو. ان کي پليد چوڻ ۽ لکڻ جائز آهي، ۽ ان کي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه نه چئو مگر ناصبي (يعني پنهنجي سيني ۾ حضرت علي ۽ حسن ۽ حسين رَضِىَ اللهُ تَعَالٰی عَـنْهُم سان بغض ۽ ڪينو رکڻ وارو هو جو) اهل بيتِ رسالت جو دشمن آهي . وَ العِیاذُ بِا للّٰہِ تَعالیٰ (فتاويٰ رضويه، ج 14، ص 603، ملخصًا)

حضرت علامه ابن حجر هيتمي مڪي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا؛ حضرت سيدنا صالح بن احمد بن حنبل رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي والد حضرت سيدنا امام احمد بن حنبل رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن کي عرض ڪيو؛ (بابا جان!) هڪ قوم اسان تي هي ڳالهه منسوب ڪري ٿي ته اسان يزيد جا حمايتي آهيون. پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن فرمايو؛ يَا بُنَيَّ وَهَل يَتَوَلّٰى يَزِيۡدَ اَحَدٌ يُؤمِنُ بِاللّٰهِ اي منهنجا پٽ! ڇا الله تعالي تي ايمان رکڻ وارو به يزيد پليد جو حمايتي ٿي سگهي ٿو؟

ٿيندا رهيا. يزيدن جي بي مروتي ۽ بي غيرتي جو عالم هي هو ته کير پياڪ ننڊڙو علي اصغر رَضِىَ اللهُ تَعَالٰی عَـنْهُ اُڄ جي سختي سان تڙپندو رهيو، آخرڪار تير جو شڪار ٿي ڪري شهادت سان سرفراز ٿيا. گمراهي جي علمبردارن چمن زهرا جي سرسبز باغ کي ڏينهن ڏٺي جو تباهه و برباد ڪري رکي ڇڏيو. مگر قربان وڄو ميدان جنگ جي بهادر شهسوار، اهلبيت اطهار رَضِىَ اللهُ تَعَالٰی عَـنْهُ م اَجْمَعِیْن جي روشن تارن ۽ گلشن فاطمه جي جگر پارن جن جي همت ۽ صبر و استقلال تان جو ايتري قدر دل ڏکائيندڙ مصائب و آلام ۾ مسلسل ٽي ڏينهن بک ۽ اڄ برداش ڪري به قبر و استقلال سان گڏ باطل جي خلاف وڙهندا رهيا ۽ صبر جو دامن مضبوطيءَ سان پڪڙي رکيو، ڇو ته هي هڪ آزمائش ٿي ۽ الله وارن جي هميشه کان اها شان رهي آهي ته آزمائشن ۾ صبر و برداشت سان ڪم وٺندا آهن. جئين ته

مٺا مٺا اسلامي ڀائرو! بيشڪ اسيرانِ ۽ شهيدانِ ڪربلا جو هي صبر قابلِ تحسين ۽ پيروي ڪرڻ جهڙو آهي. تنهن ڪري اسان کي به گهرجي ته جيڪڏهن اسان تي ڪائي اچانڪ مصيبت اچي، مثلًا قريبي رشتيدار وفات ڪري وڃي، يا ٻار بيمار ٿي وڃي، يا ڦُر ٿي وڃي، يا ڪوئي حادثو پيش اچي، پوليس گرفتار ڪري وٺي، يا مقدمو قائم ٿي وڃي، رقم يا گاڏي وغيره گم يا چوري ٿي وڃي، سردي ۽ گرمي يا لوڊ شيڊنگ ۽ مهانگائي شدت اختيار ڪري وڃي، نوڪري کان ڪڍيو وڃي الغرض مصيبتن جو هجوم ٿي وڃي ان وقت ڪربلا جي مظلومن جي مصائب و آلام کي ياد ڪري صبر و استقلال جو مظاهرو ڪندي صبر صبر ۽ صرف صبر کان ڪم وٺجي ته مصيبتن تي صبر ڪرڻ وارن کي الله تعالي عَزَّوَجَلَّ پنهنجي فضل و ڪرم سان بروز قيامت بيشمار انعام و اڪرام سان سرفراز فرمائيندو.

حسن اخلاق جي پيڪر، نبين جي تاجور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو؛ الله عَزَّوَجَلَّ فرمائي ٿو؛ جڏهن مان پنهنجي ڪنهن ٻانهي جي بدن يا ان جي اولاد يا ان جي مال جي طرف ڪائي مصيبت موڪليان، پوءِ اهو صبر و جميل سان ان جو استقبال ڪري، ته قيامت

جي ڏينهن مون کي ان کان حيا ايندو ته مان ان جي لاءِ ميزان قائم ڪيان يا ان جو اعمال نامو کوليان. (ڪنز العمال، الجزء 2،3، ص 115، حديث 6558)

دعوت اسلامي جي زيرِ اهتمام جامعة المدينه جي سڀ کان پهرين شاخ 1995ع ۾ باب المدينه ڪراچي ۾ نيو ڪراچي جي علائقي گودهرا ڪالوني ۾ کولي وئي ۽ هن وقت پاڪستان جي ڪيترن ئي شهرن کان علاوه الحمد الله عَزَّوَجَلَّ دنيا جي مختلف ملڪن مثلًٰا هند، جنوبي آفريقا، انگلينڊ، نيپال ۽ بنگلاديش وغيره ۾ به جامعة المدينه للبنين ۽ جامعة المدينه للبنات قائم آهن. جنهن ۾ هزارين اسلامي ڀائر ۽ اسلامي ڀينرون علمِ دين حاصل ڪري نه صرف پاڻ فائدو حاصل ڪري رهيا آهن بلڪه ٻين کي به علم جي نور سان منور ڪرڻ ۾ مصروف آهن. جامعة المدينه جي خصوصيت هي به آهي ته هنن جامعات المدينه ۾ طلبه کي نه صرف ديني تعليم سان آراسته ڪيو ويندو آهي بلڪه انهن جي اخلاقي ۽ روحاني تربيت جي طرف ڀرپور توجه ڏني ويندي آهي. شيخ طريقت امير اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ هنن طلباء سان پنهنجي محبت جو اظهار ڪندي ارشاد فرمائن ٿا؛ مان دعوتِ اسلامي جي جامعة المدينه جي طلبه ڪرام سان گهڻي محبت ڪريان ٿو ۽ انهن جي صدقي سان پنهنجي لاءِ دعاءِ مغفرت ڪندو آهيان. الحمد الله عَزَّوَجَلَّ جامعة المدينه جا طلبه پنج وقت نماز کان علاوه ٻيا نوافل به پڙهندا آهن ۽ گهڻا طلبه ملي ڪري صلوٰة التوبه تهجد اشراق ۽ چاشت جي نمازن جو اهتمام به ڪندا آهن. تنهن ڪري اسان کي به گهرجي ته پنهنجي ٻارن کي جامعة المدينه ۾ داخل ڪرائي انهن جي دنيا ۽ آخرت بهتر بڻايون .

سنئون اثر ڪندي آهي ان سان نه صرف وار بلڪ دماغ ۽ چهرو به متاثر ٿيندو آهي ۽ صحت کي نقصان پهچي سگھي ٿو، تنهن ڪري سنت جي پيروي جي نيت سان عمامو شريف ٻڌڻ ۾ ٻنهي جهانن ۾ عافيّت آهي * طِبّي تحقيق مطابق مٿي جي سور لاءِ عمامو شريف پائڻ تمام فائديمند آهي * عمامو شريف پائڻ سان دماغ کي قوت ملندي ۽ حافظو مضبوط ٿيندو آهي * عمامو شريف ٻڌڻ سان دائمي نزلو ناهي ٿيندو يا ٿيندو ته ان جا اثرات گھٽ هوندا آهن * عمامي شريف جو شِملو هيٺين دڙ جي فالج وغيره کان بچائي ٿو ڇو ته شِملو حرام مغز کي موسمي اثرن مثلاً سردي گرمي وغيره کان تحفظ فراهم ڪندو آهي * شِملو ”سَرسام“ جي مرض جو خطرو گھٽ ڪندو آهي. دماغ جي وَرَم (يعني سوڄ) جي مرض کي سر سام چوندا آهن .

هزارين سنّتون سکڻ جي لاءِ مڪتبة المدينه جا شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنّتن جي تربيت جو هڪ بهترين ذريعو دعوتِ اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنّتن ڀريو سفر آهي.
