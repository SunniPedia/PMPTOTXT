صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مُصطَفٰي ڪريم جي پنهنجي ڏوهٽي تي شفقت

حضرت سيدنا ابو هريره رَضِىَ اللهُ عَنْه فرمائن ٿا: مون پنهنجي اکين سان ڏٺو ۽ ڪنن سان ٻڌو ته هڪ ڏينهن رسولِ ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ حضرت سيدنا امام حسين رَضِىَ اللهُ عَنْه کي هٿ مان پڪڙيو هو، امام حسين رَضِىَ اللهُ عَنْه جا پير نبي ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي مبارڪ پيرن تي هئا، حضور انور صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ فرمائي رهيا هئا: ننڍڙا شهزادا! مٿي چڙهندا اچو، امام حسين رَضِىَ اللهُ عَنْه جيڪي اڃا گھڻا ننڍا هئا، سرڪار صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جو هٿ مبارڪ پڪڙي مٿي چڙھڻ لڳا، ايستائين جو پاڻ سڳورن صَلَّى اللّٰهُ عَلَيۡهِ

واٰلِهٖ وَسَلَّمَ جي سيني اقدس تي پنهنجا قدم رکي ڇڏيا۔ پاڻ سڳورن صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جن فرمايائون: وات کوليو! امام حسين رَضِىَ اللهُ عَنْه وات کوليو ته پاڻ ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ انهن جي منهن کي چُمي ورتو، پوءِ بارگاھ الٰهي ۾ دعا فرمائي: اَللّٰهُمَّ اَحِبَّه فَاِنِّيْ اُحِبُّهٗ الٰهي! مان هن کي محبوب رکان ٿو تون به هن کي محبوب رک۔ (الادب المفرد،باب الانبساط إلى الناس،ص ۷۲،حديث:۲۴۹،واللفظ له، الاستيعاب ، ۱/۴۴۶)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

ٽيون مدني گل اهو حاصل ٿيو ته نبي ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جيئن ته تمام جهان وارن جي لاءِ رحمت بڻائي موڪليا ويا آهن، تنهنڪري جھڙي طرح پاڻ ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي رحمتن ۽ شفقتن جي درياء وڏن وڏن کي فيض ياب ڪيو بلڪ ڪري

سرڪار صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي پنهنجي ڏوهٽيءَ تي شفقت

جنتي گھر دارُ الفَرح ڪنهن جي لاءِ آهي؟

حضرت سَيدَتُنَا عائشه صِدّيقه رَضِیَ اللهُ تَعَالٰی عَنْهَا روايت ڪن ٿيون ته رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”بيشڪ جنّت ۾ هڪ گهر آهي جنهن کي ”دار الفرح“ چيو ويندو آهي ان ۾ اهي ئي ماڻهو داخل ٿيندا جيڪي ٻارن کي خوش ڪندا آهن۔ (جامع صغير ، ص۱۴۰، حديث۲۳۲۱)

پي ا ريون پياريون اسلامي ڀينرون! ٻار الله پاڪ جي نعمت هوندا آهن۔ انهن سان پيار ۽ محبت ڀريو ورتاءُ (Behave) ئي ڪرڻ گھرجي۔ ٻار پنهنجا هجن يا ٻين جا، ٻار ٻار ئي هوندا آهن۔ انهن سان هميشه مهرباني سان پيش اچڻ گھرجي۔ ڪي اسلامي ڀينرون اهڙيون خشڪ مزاج هونديون آهن جيڪي پنهنجي ٻارن تي به شفقت نٿيون ڪن۔

حضرتِ سَيدُنا ابوهريره رَضِىَ اللهُ عَنْه کان روايت آهي ته رسولُ الله صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ حضرتِ سَيدُنا حسن بن علي رَضِىَ اللهُ عَنْهُمَا کي چميائون ته هڪ شخص عرض ڪيو: منهنجا ڏھ ٻار آهن مون (اڄ تائين) انهن مان ڪنهن کي ناهي چميو، ته پياري آقا صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ انهن کي ڏٺو پوءِ ارشاد فرمايائون: جيڪو رحم ناهي ڪندو اهو رحم ناهي ڪيو ويندو۔ (بخاري،ڪتاب الادب،باب رحمة الولد۔۔۔، ۴/۱۰۰، حديث:۵۹۹۷)

پي ا ريون پياريون اسلامي ڀينرون! اسان جا آقا، مديني وارا مصطفيٰ صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ ٻارن سان حسنِ سلوڪ ڪرڻ ۾ بي مثل ۽ بي مثال هئا، سندن حسنِ سلوڪ جا ڪجھ مثال اسان ٻڌا۔ اسان کي به رسول اڪرم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي مبارڪ طريقي تي هلڻ گھرجي۔ يقيناً ٻار الله پاڪ جي تمام وڏي نعمت آهن۔ ٻارن سان ئي گھر جي اَڱڻ ۾ بهار ايندي آهي۔ ننڍڙي ٻارن جي هڪ هڪ ادا والدين جي قرار ۽ سُڪون جو سبب بڻجندي آهي۔ انهن جي خوشين سان گھر ۾ خوشيون ٿينديون آهن ۽ انهن جي پريشان ٿيڻ سان سڄو گھر پريشان ٿي ويندو آهي۔ ٻار ماءُ پيءُ جي جگر جو ٽڪر هوندا آهن۔ ٻار والدين جي اکين جي ٿڌاڻ هوندا آهن۔ ٻار والدين جي دلين جو چين هوندا آهن۔ ٻار والدين جي دل جو قرار هوندا آهن۔ اڄڪلھ ڪجهه والدين ٻارن جي ظاهري زيب و زينت، سٺي غذا، عمده ڪپڙا ۽ ٻين ضرورتن جو ته پورو پورو خيال رکندا آهن۔ پر ٻارن جي بهترين اخلاقي ۽ روحاني تربيت ڪرڻ ۾ ڪوتاهي ۽ سستي ڪري ويندا آهن۔ ٻارن سان پيار محبت ڪرڻ انهن جو حق آهي پر محبت جي غلبي ۾ انهن چنڊ جھڙن ٻارن جي تربيت کان پاسو ڪرڻ تمام وڏي ناداني آهي۔ اهڙا نورِ نظر ۽ چنڊ جهڙا ٻار جڏهن وڏا ٿيندا آهن ته گھر جي اَڱڻ ۾ تاريڪي جا پاڇا ڇانئجي ويندا آهن۔ ان جو سبب اهو ئي هوندو آهي ته انهن جي ننڍپڻ ۾ ئي جيڪڏهن مناسب تربيت ڪئي وڃي ها ته انهن جو ڪردار سڄي خاندان جي لاءِ قابلِ رشڪ هجي ها۔ معاشري ۾ انهن جي تعريف ڪئي وڃي ها۔

ٻارن جي سٺي تربيت ڪريو!

ياد رکو! ٻارن جو ننڍپڻ سادي ڪاغذ جيان هوندو آهي، ان ۾ جيڪو به لکيو وڃي هميشه جي لاءِ نقش ٿي ويندو آهي۔ تنهنڪري ننڍپڻ ئي ۾ انهن جي تربيت

جي بهترين ڪوشش ڪئي وڃي۔ ان تربيت ۾ به شفقت و نرمي ۽ پيار و مَحبّت جو پهلو غالب رهڻ گھرجي ته جيئن اهي وڏا ٿي مُعاشَري جو بهترين فرد بڻجي سگھن، ننڍي ڄمار هئڻ ۽ شعور ۽ سمجھ گھٽ هئڻ جي ڪري ٻارن جي طبيعت نهايت حُسّاس هوندي آهي ۽ انهن جي دل شيشي کان وڌيڪ نازڪ هوندي آهي۔ جيڪڏهن ٻارن کي ٿوري ٿوري ڳالھ تي ڇنڊ پٽجي بلڪ انهن کي ماريو به وڃي ته انهن جي طبيعت ۾ چِڙچڙو پن، پيدا ٿي سگھي ٿو، بلڪه بار بار تشدد ڪرڻ انهن کي بدتميزي ۽ بدمزاجي تي اڀاري سگھي ٿو، ائين ان قيمتي هيري جي غلط تراش خراش ان کي بي مُلھ ۽ بيڪار بڻائي ڇڏيندي آهي جڏهن ته ٻارن جي دلجوئي ڪرڻ ۽ انهن سان مِلي ڪري رهڻ، چڱن ڪمن تي انهن جي مناسب حوصله افزائي ڪرڻ، انهن سان پيار و محبت وارو ورتاؤ ڪرڻ، انهن جي شڪوا شڪايتن تي مناسب ڪاروائي ڪرڻ، انهن کي راند روند جا سٺا ۽ محفوظ موقعا ڏيڻ سٺيون ۽ سبق آموز آکاڻيون ٻڌائڻ مثال طور قرآني واقعن جي لاءِ عجائب القرآن ڪتاب، صحابه رَضِیَ اللہُ عَنْہُم اَجْمعین جي حالات جي لاءِ ”ڪراماتِ صحابه“ ٻارن جي ڪهاڻين جي لاءِ ”فرعون جو خواب، پٽ هجي ته اهڙو، نور وارو چهرو، ڪوڙو چور وغيره.

هي سمورا طريقا ٻارن ۾ خود اعتمادي، برداشت ۽ حُسنِ اَخلاق جهڙيون خوب صورت وصفون پيدا ڪندا آهن۔ اهڙي طرح ڪڏهن ٻارن کان غلطي ٿي به وڃي ته ڪاوڙ ۾ لال پيلي ٿيڻ جي بدران نرمي سان انهن جي اصلاح ڪريو ۽ آئنده اهڙن ڪمن کان بچڻ جو ذهن ڏيو۔ ياد رکو! ٻارن سان وڏو بڻجي رهڻ ڪمال نه آهي بلڪ ٻارن سان ٻار بڻجي رهڻ ۽ انهن سان ملي ڪري رهڻ ئي ڪمال آهي۔

ٻارن جي اسلامي تربيت جي لاءِ ڪجھ اهم مدني گل

(1) ٽن سالن واري عمر جي ٻار جي سامهون، سيکارڻ جي نيت سان ” الله، الله“ چئو ته جيئن ان جي وات مان پهريون لفظ”الله“ نڪري ۽ جڏهن ڪجھ لفظ ڳالهائڻ لڳي ته لفظ”الله“ ڳالهائڻ جو عادي ٿي وڃي.

(2) ٽي کان چار سال جي عمر ۾ پورو ڪلمو لَاۤاِلٰہَ اِلَّااﷲُمُحَمَّدٌ رَّسُول اللہ (صَلَّی اللّٰہُ عَلَیْہِ واٰلِہٖ وَسَلَّمَ) سيکاريو.

(3) چئن سالن جي عمر کان ”مدني قاعدي“ جي ترڪيب ڪيو. ڪجھ بزرگن جو معمول رهيو آهي ته اهي ٻارن جي ”رسمِ بسم الله“ (يعني

بسم الله شريف پڙهائڻ جي تقريب) 4 سال، 4 ماھ ۽ 4 ڏينهَن جي عمر ۾ ڪندا هئا.

(4) چئن سالن جي عمر جي ٻارن کي کائڻ پيئڻ جا آداب (مثال طور هٿ ڌوئي کائڻ، ويهي کائڻ، مٿو ڍڪي کائڻ وغِيره)، هلڻ ڦرڻ جا، ڳالهائڻ جا، ڪپڙا صحيح طرح پائڻ جا، حياء جا آداب سيکارڻ شروع ڪري ڇڏيو. اها عمر ابتدائي آداب سيکارڻ جي لحاظ کان تمام اهم آهي.

(5) سالن جي عمر کان ٻارن کي پياري آقا مديني واري مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۽ ٻين نبين سڳورن جا معجزا، صحابه ڪرام ۽ اهلبيتِ اطهار(رَضِیَ اللهُ تَعَالٰی عَنْهُم) جا واقعا، اولياءِ ڪرام(رَحْمَةُ اللهِ تَعَالٰی عَلَيْهم) جو ڪرامتون ٻڌائڻ ۽ نعتن ۽ منقبتن جا ڪجھ شعر ياد ڪرائڻ شروع ڪيو، ٿي سگهي ته گهر ۾ وقت به وقت نياز جي ترڪيب بڻايو هن جو فائدو اهو حاصل ٿيندو جو ننڍپڻ کان ئي ٻارن جي دلين ۾ بزرگن جي محبت پيدا ٿي ويندي ۽ ننڍپڻ جي محبت ممڪن آهي مرڻ مهل تائين باقي رهي.

(6) کان 7 سالن جي عمر تائين محبت ۽ پنهنجائپ سان علماءِ ڪرام، قاري صاحب، امام صاحب، استاد صاحب ۽ وڏن جو ادب ڪرڻ سيکاريو. ان جو هي فائدو ٿيندو جو اولاد والدين جو به ادب ڪندو ۽ شادي کانپوءِ ڇوڪري پنهنجي مڙس جو به احترام ڪندي.

(7) اسلامي تاريخ جي مطابق جنهن ڏينهن اهي 7 سالن جا هجن، جيتري قدر ممڪن ٿي سگهي ته ان ڏينهن گهر ۾(پردي ۽ شرعي احڪام جي پابندي سان) ڪا تقريب رکي ڪري، زندھ هجي ته پيءُ نه ته ڪو سرپرست ان کي نماز جو حڪم ڏيئي، ان سان ان جي دماغ ۾ نماز جي اهميت اجاگر ٿيندي بلڪ سموري خاندان جي ٻارن جي دل ۽ دماغ ۾ اجاگر ٿي ويندي.

(8) کان 9 سال جي عمر شرعي مسئلن ۽ ديني تربيت جي لاءِ سڀ کان اهم ترين وقت آهي. نماز جو حڪم ڏيڻ کان پوءِ هاڻي نرمي سان نماز جي طرف آڻيو، ٿي سگهي ته نماز جي حاضري واري ڪاپي جي ترڪيب بڻايو، آلارم وغيره جي ذريعي پاڻهي اٿڻ جو عادي بڻايو، نماز جا وقت ڏسڻ سيکاريو ۽ نماز واري نقشي جي ڪاپي ڏياريو، گڏو گڏ نماز، سورة فاتحه ۽ ٻين سورتن ۽ اذڪارِ نماز(جيڪو ڪجھ نماز ۾ تلاوت کان

علاوه پڙهيو ويندو آهي) به سيکاريو.

(9) انهي عمر ۾ آهسته آهسته ٻار جي عقل مطابق اسلامي عقائد ٻڌايا وڃن. وضو ۽ نماز جو عملي طريقو سيکاريو وڃي، غسل ۽ تيمم ڪرڻ، روزي رکڻ ۽ ڪپڙا پاڪ ڪرڻ جو طريقو ٻڌايو وڃي(مثال طور ٻڌايو وڃي ته جيڪڏهن ڪپڙن تي رت لڳي وڃي ته هيئن پاڪ ٿيندو وغيره).

(10) هاڻي ان جي اخلاقن کي سڌارڻ جي لاءِ هلاڪت ۾ وجهڻ وارين شين مثلاً(حرص، لالچ، حبِ دنيا، حبِ جاه، عجب، تڪبر، خيانت، ڪوڙ، ظلم، فحش، غيبت، حسد، ڪينو وغِيره) جون خرابيون بيان ڪيو. اهڙي طرح نجات ڏيارڻ وارين ڳالهين مثلاً (توڪل، قناعت، زهد، اخلاص، تواضع، امانت، صدق، عدل، حياء، دل جي سلامتي ۽ زبان وغيره) جون خوبيون بيان ڪيو.

(11) هاڻي پهرين درجي جا ڪتاب پڙهائڻ شروع ڪري ڏيو ته جيئن بالغ ٿيڻ کان پهريان اهي ٻار فرض علوم ڄاڻڻ وارا ٿين.

(12) پهرين درجي کان پوءِ پٽ هجي ته سورة مائده ۽ ڌيءُ هجي ته سورة نور جو ترجمو ڪنز الايمان(تفسير سان) پڙهايو يا ان کان مڪمل ڪرايو ۽ ٿي سگهي ته ان جي مڪمل ٿيڻ تي گهر ۾(پردي ۽ شرعي احڪام جي پابندي سان) تقريب رکو ته جيئن انهن سورتن جي تعليم جو رواج پئجي وڃي.

(13) جڏهن پٽ بالغ ٿيڻ جي ويجهو ٿئي ته پيءُ ۽ ڌيءُ بالغ ٿيڻ جي ويجهو هجي ته ماءُ خاص مسئلا پاڻ سيکارين، هي ڪم هر گز ٻئي جي حوالي نه ڪن.

(14) ٻارن جي بالغ ٿيڻ کانپوءِ به قرآن پاڪ پڙهندي رهڻ ۽ فرض علوم جو مطالعو ڪندي رهڻ جي طرف توجھ ڏياريندا رهو.

(15) ٻارن جي تربيت جي حوالي سان وڌيڪ معلومات حاصل ڪرڻ جي لاءِ دعوتِ اسلامي جي اداري مڪتبة المدينه جي ڪتاب فيضانِ مدني مذاڪره قسط24” بچو ں ک ی ترب ی ت کب اور ک ی س ے ک ی جائ ے ؟ “ ۽ فيضانِ مدني مذاڪره قسط 38 ”چھ ٹی ا ں ک ی س ے گزار یں “ ۽ تربيتِ اولاد جو مطالعو ڪيو.

صَلُّو ْا عَلَی الْحَبِیْب! صَلَّی اللہُ عَلٰی مُحَمَّد

سرڪارِ مدينه صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جا ٻارن سان شفقت ڀريا انداز

پي ا ريون پياريون اسلامي ڀينرون! هميشه ٻارن سان الفت ڀريو ورتاءُ ڪيو، غور ڪريو! ننڍپڻ ۾ چنڊ سان کيڏڻ واري مصطفيٰ، بُرّاق جهڙي اعليٰ ترين سواري تي سوار ٿيڻ واري آقا عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام پاڻ پنهنجي ڏوهٽن يعني حسنينِ ڪريمين رَضِىَ اللهُ عَنْهمَا جي لاءِ سواري بڻيا۔

رسول اڪرم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جو ٻارن سان شفقت جو عالم اهو هو جو نماز دوران سجدي جي حالت ۾ پٺ مبارڪ تي ٻارن جي موجودگي محسوس فرمائي سجدو ڊگھو ڪري ڇڏيندا هئا۔ (مسنداحمد، ۵/۴۲۶، حديث:۱۶۰۳۳ ملخصاًو مفهوماً)

ڪڏهن ڪڏهن نماز دوران ٻار جي روئڻ جو آواز ايندو هو ته پاڻ عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام نماز ۾ تخفيف فرمائيندا هئا۔ (بخاري،ڪتاب الاذان، باب من اخف الصلاة۔۔۔،۱/۲۵۳،حديث:۷۰۷ مفهوماً)

ٻارن سان شفقت ۽ مهرباني جو اهڙو ورتاؤ ڏسي حضرت انس رَضِىَ اللهُ عَنْه چون ٿا: نبي پاڪ عَلَيۡهِ الصَّلٰوةُ وَالسَّلَام ماڻهن ۾ سڀ کان وڌيڪ ٻارن تي شفقت فرمائڻ وارا هئا۔ (مسند ابو يعلٰي، ۳/۴۲۵، حديث:۴۱۸۱)

پي ا ريون پياريون اسلامي ڀينرون! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّمَ جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو. اچو شيخِ طريقت ،اميرِ اهلسُنَّت دَامَتْ بَرَکاتُهمُ الْعَالِيه جي رسالي ’’ 163 مدني گل ‘‘ مان مٿي جي وارن وغيره جون سنتون ۽ آداب ٻڌَئون ٿيون ٭ عورت جو مٿو ڪوڙائڻ حرام آهي ٭ (خلاصو فتاوي رضويه ج 22ص 994) عورت کي مٿي جا وارڪٽرائڻ جيئن هن زماني ۾ نصراني عورتن ڪٽرائڻ شروع ڪيا آهن ناجائز ۽ گناهه آهن ۽ ان تي لعنت آئي آهي ٭ مڙس ائين ڪرڻ جو چيو تڏهن به اهو ئي حڪم آهي ته عورت ائين ڪرڻ ۾ گنهگار ٿيندي ڇو ته شريعت جي نافرماني ڪرڻ ۾ ڪنهن (يعني ماءُ پيءُ يا مڙس وغيره) جو چوڻ نه ورتو ويندو ٭ (بهار شريعت ج 3ص 588) ننڍڙين ٻارڙين جا وار به مرداڻي انداز سان نه ڪٽايو، ننڍپڻ کان ئي انهن کي وڏا وار رکڻ جو ذهن ڏيو ٭ ڪجهه ماڻهو ساڄي يا کاٻي پاسي سينڌ ڪڍندا آهن اهو سنت جي خلاف آهي ٭ سنّت هي آهي ته وچ مان سينڌ ڪڍي وڃي ٭ اڄڪلهه قينچي يا مشين جي ذريعي وارن کي مخصوص نموني تي ڪٿي وڏا ته ڪٿي ننڍا ڪري ڇڏبا آهن اهڙا وار رکڻ سنت ناهي ٭ فرمان مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم: جنهن جا وار هجن اهو انهن جي تعظيم ڪري ٭ (ابودائود ج 4 ص103 حديث 4163) يعني انهن کي ڌوئي، تيل هڻي ۽ ڪنگي ڪري ٭ مٿي ۾ مهندي لڳائي سمهڻ نه گھرجي، هڪ حڪيم جي بقول اهڙي طرح مهندي لڳائي سمهڻ سان مٿي وغيره جي گرمي اکين ۾

هفته وار سنتن ڀريو اجتماع

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ هزارين مقامن تي اسلامي ڀائرن ۽ اسلامي ڀينرن جا هفته وار سنتن ڀريا اجتماع ٿيندا آهن، جن ۾ هر هفتي لکين اسلامي ڀائر ۽ اسلامي ڀينرون شرڪت ڪري علمِ دين حاصل ڪندا آهن، صرف پاڪستان ۾ اسلامي ڀائرن جا 529 جڏهن ته اسلامي ڀينرن جا 6501 هفته وار اجتماع ٿيندا آهن ۔ فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جيڪو الله پاڪ جي رضا خاطر صدقو ڪري ته اهو(صدقو) ان جي ۽ باهه جي وچ ۾ پردو بڻجي ويندو آهي ۔ (شُعَبُ الايمان، باب في الزڪاة،التحريض علي صدقة التطوع،۳/۲۱۴، حديث:۳۳۵۳)

ملڪ ۽ بيرون ملڪ دعوت اسلامي جي تحت هلڻ وارا مدارس المدينه، جامعاة المدينه ۽ ٻين شعبن جي سمورن خرچن جي لاءِ پنهنجي مدني عطيات(چندي) سان تعاون فرمايو ۽ ثوابِ جاريه جون حقدار بڻجي وڃو.

نيت ڪريو ته پنهنجا تمام واجب ۽ نفلي صدقا(زڪوٰة، صدقو، خيرات وغيره) عاشقان رسول جي مدني تحريڪ دعوت اسلامي کي ڏينديون سين اِنْ شَــآءَالـلّٰـه

سڀئي اسلامي ڀينرون هي پمفلٽ”ماهِ رمضان ۾ گناهه ڪرڻ واري جي قبر جو ڀيانڪ منظر“ مڪتبة المدينه کان وڌيڪ تعداد ۾ نه ته گهٽ مان گهٽ 12 يا حسبِ توفيق خريد ڪري تقسيم فرمايو، ۽ پنهنجي گهر ۾ ظاهري جڳهه تي لڳايو. هن جي برڪت سان گناهن کان بچڻ جو ذهن بڻجندو. اِنْ شَــآءَالـلّٰـه

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ ماههِ رمضان المبارڪ ۾ شيخِ طريقت اميراهلسنت بعدِ نماز عصر ۽ تراويح کان پوءِ مدني مذاڪري جي لاءِ وقت عطا فرمائيندا آهن جيڪو مدني چينل تي براءِ راست ڏيکاريو ويندو آهي سڀني اسلامي ڀينرن کي التجاء آهي ته لازمي هنن ٻنهي مدني مذاڪرن ۾ پاڻ به شرڪت ڪن بلڪ گهر جي ٻين ڀاتين کي به اهتمام سان هنن مدني مذاڪرن کي ڏيکارڻ جي ترڪيب بڻائن. اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ علم ۽ حڪمت جا بيشمار موتي ملندا.
