اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

دُرُود شريف جي فضيلت

اسان بيڪسن جي مددگار، بروزِ محشر اسان جي شفاعت فرمائڻ واري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عالي شان آهي: منهنجو جيڪو امتي اخلاص سان مون تي هڪ ڀيرو درودِ پاڪ پڙهندو الله پاڪ ان تي ڏهه رحمتون نازل فرمائيندو، ان جا ڏهه درجا بلند فرمائيندو، ان جي لاءِ ڏهه نيڪيون لکندو ۽ ان جا ڏهه گناهه مٽائي ڇڏيندو ۔ (سنن کبرٰی،کتاب عمل الیوم …الخ،باب ثواب الصلاۃ…الخ،۶/۲۱،حدیث:۹۸۹۲)

کعبے کےبدر الدُّجیٰ تم پہ کروڑوں دُرُود طیبہ کے شمسُ الضُّحٰی تم پہ کروڑوں دُرُود

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿيون. فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِهٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير، سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندَس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندَس ٭ ضرورت پوڻ تي ٻين اسلامي ڀينرن جي لاءِ سِرڪي جاءِ ڪشاده ڪندَس ٭ ڌِڪو وغيره لڳو ته صبر ڪندَس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندَس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ واريءَ جي دِل جوئي جي لاءِ آهسته آواز سان جواب ڏيندَس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندَس ۔ دوران بيان موبائل جي غيرضروري استعمال کان بچندَس،نه بيان ريڪارڊ ڪندَس نه ئي ٻئي ڪنهن قسم جو آواز ڇو ته ان جي اجازت ناهي،جيڪو ڪجهه ٻُدَندَس،ان کي ٻڌي ۽ سمجهي ڪري ان تي عمل ڪرڻ ۽ ان کي بعد ۾ ٻين تائين پهچائي نيڪي جي دعوت عام ڪرڻ جي سعادت حاصل ڪندَس.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! جُمَادي الاُخريٰ اسلامي سال جو ڇهون مهينو آهي، هن جي 22 تاريخ تي عاشقِ اڪبر، سالارِ صحابه، پيڪرِ صدق و وفا، يارِ غار ۽ يارِ مزار، پهرين خليفهءِ راشد اَمِيرُالمؤمنين حَضرتِ سَيدُنا ابوبڪر صديق رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو يومِ عُرس آهي. انهيء مناسبت سان اڄ اسان سندن سيرت و ڪردار بابت ٻڌنديون سين. هن ۾ ڪجھ شڪ ناهي ته حضرتِ سَيدُنا صِدِّيقِ اَڪبَر رَضِىَ اللهُ تَعَالٰی عَـنْهُ هر لحاظ کان اَشرَف ۽ اَعليٰ آهن، اِمامت هجي يا خِلافت، ڪرامت هجي يا شَرَافت، صَدَاقت هجي يا شُجَاعت، خوفِ خدا هجي يا عشقِ مصطفيٰ، اَلغَرَض! هر اِعتِبار سان صديقِ اڪبر جو هڪ اهم مقام (Important) آهي، اڄ جي بيان ۾ اسان سندن ذڪرِ خير، تعارف، فضائل ۽ خصوصيات ۽ سندن زندگي جا واقعا ٻڌنديون سين، ڪاش! سڄو بيان سٺين سٺين نيتن سان ٻڌڻ نصيب ٿي وڃي ۔ اچو! سڀ کان پهريان حضرت سَيد نا صديقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو هڪ واقعو ٻڌون ٿيون:

رسول الله جي فرمان سبب روئڻ

حضرتِ سَيدُنا زَيد بن ارقم رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته اسان هڪ ڀيري حضرتِ سَيدُنا ابُوبَڪر صِدِّيق رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي بارگاھ ۾ ويٺا هئاسين ته پاڻي ۽ ماکي

آندو ويو ۽ جيئن ئي سندن جي ويجھو ڪيو ويو ته صديقِ اڪبر جن زارو قطار روئڻ شروع ڪري ڏنو ۽ روئندا رهيا ايستائين جو تمام صحابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ به روئڻ لڳا، صَحَابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ روئي ڪري ماٺ ٿي ويا پر صديقِ اڪبر جن روئندا رهيا، صَحَابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ صديقِ اڪبر جن کي ڏسي وري روئڻ لڳا، نيٺ حضرتِ سَيدُنا ابُوبَڪر صِدِّيق رَضِىَ اللهُ تَعَالٰی عَـنْهُ پنهنجا نيڻ اُگھيا ته صَحَابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ جن عرض ڪيو اي رسولُ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا خليفہ! اوهان کي ڪهڙي شيء روئاريو، فرمايائون؛ هڪ ڀيري مان الله جي محبوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي بارگاھ ۾ ويٺو هئس ته اوچتو مون ڏٺو ته رسولِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پاڻ تان ڪا شيء پري ڪري رهيا آهن، جڏهن ته مون کي ان وقت ڪا شيء نظر نه اچي رهي هئي، مون عرض ڪيو؛ يارَسُولَ اللہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! اوهان ڪهڙي شيء کي هٽائي رهيا آهيو، فرمايائون؛ دنيا منهنجو ارادو ڪيو هو مون ان کي چيو ته پري ٿي وڃ، ته ان مون کي چيو؛ اوهان ته پاڻ کي مون کان بچائي ورتو، پر اوهان کان بعد وارا مون کان نه بچي سگھندا ـ “ (شعب الايمان ، باب في الزھد وقصر الامل، 7/343، حديث:10518)

پياريون پياريون اسلامي ڀينرون! بيان ڪيل واقعي مان معلوم ٿيو ته دنيا جي محبت چڱي شيء نه آهي، تڏهن ته اسان جي پياري پياري آقا، مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن دنيا کي ٺڪرائي ڇڏيائون، ياد رکو! ٭ دنيا جي محبت ۾ گرفتار ٿي انسان آخرت کي وساري ڇڏيندو آهي، ٭ دنيا جي محبت ۾ مبتلا ٿي انسان گناهن جي دلدل ۾ دٻجندو ويندو آهي، ٭ دنيا جي محبت ۾ پئجي انسان حلال ۽ حرام ۾ تميز ڪرڻ ڇڏي ڏيندو آهي، ٭ دنيا جي محبت جو شڪار ٿيڻ وارو انسان نيڪين کان پاسو ڪندو آهي، ٭ دنيا جي محبت سان ناطو جوڙڻ وارو انسان پنهنجي موت ۽ قبر کي وساري ڇڏيندو آهي، ٭ دنيا جي محبت ۾ گم رهڻ وارو انسان پنهنجي آخرت کي خراب ڪري لاهيندو آهي، ٭ دنيا جي محبت کي دل ۾ جڳھ ڏيڻ وارو انسان پنهنجي آخرت (Hereafter) کي وساري گھاٽي جو سودو ڪندو آهي، اَلغَرَض! ٭ دنيا جي محبت جو شڪار ٿيڻ وارو انسان تمام وڏي نقصان جو شڪار ٿي ويندو آهي۔ انهيء ڪري ٭ الله وارا دنيا جي محبت کان پري رهندا هئا ايستائين جو معمولي شيون ڏسي روئي پوندا هئا جيئن ته ٭ اوهان ٻڌو ته سَيدُنا صِدِّيق اَڪبَر رَضِىَ اللهُ تَعَالٰی عَـنْهُ ماکي مليل پاڻي کي ڏسي ايترا رُنا جو صَحَابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ سندن جي روئڻ واري اها حالت ڏسي نه

سگھيا ۽ اهي به سڏڪا ڀري روئڻ لڳا، صديقِ اڪبر صرف انهي خوف جي ڪري روئي رهيا هئا ته ڪٿي ايئن نه ٿئي جو آئون به دنيا جي محبت ۽ پُر فريب رونقن ۾ غافل ٿي پنهنجي پالڻهار جي ڪا نافرماني ڪري ويهان ۽ پوءِ منهنجو ڪريم ربّ مون کان ناراض ٿي وڃي، جڏهن ته صديقِ اڪبر جي اها شخصيت آهي جو جن کي پياري آقا مٺي مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم دنيا ئي ۾ جنّت جي بشارت عطا فرمائي هئي، پر ان جي باوجود سندن تي خوفِ خدا جو ڪيتري قدر غلبو هو، هڪ پاسي سَيدُنا صِدِّيق اَڪبَر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو خوفِ خدا آهي ۽ ٻي پاسي اسان آهيون جو ڪجھ خبر ناهي ايمان تي خاتمو ٿيندو يا نه؟ مرڻ وقت ڪلمو نصيب ٿيندو يا نه؟ قبر ۾ منڪر نڪير جي سوالن جا جوابات به ڏئي سگھنديون سين يا نه؟ قبر ۾ مصطفٰي ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي سڃاڻپ به نصيب ٿيندي يا نه؟ قبر جنّت جو باغ بڻجندي يا مَعَاذَالله! جهنّم جي کڏ! پر پوءِ به ڪو خوف ناهي، اعمال نامو نيڪين کان خالي هئڻ جي باوجود نيڪين جي جُستجُو نٿيون ڪيون، بلڪ ڏينهن رات بس هڪ ئي ڌُن ۾ مگن آهيون، دولت گڏ ڪري وٺون، گھر ۽ بنگلا هجن، بينڪ بيلنس ۾ اضافو ٿي وڃي، خوب جائيداد ۽ مال هجي، سٺا سٺا کاڌا کائي وٺون، مهانگين گاڏين ۾ گهمي وٺون، سڄي دنيا جي سير ڪري وٺون، ٻارن جو مستقبل (Future) ٺاهي وٺون، ٻارن جون شاديون ڪري وٺون۔ مطلب ته! اسان دنيا کي ئي سب ڪجھ سمجھي ورتو آهي۔ اي ڪاش! اسان خوابِ غفلت مان بيدار ٿي وڃون، دنيا جي محبت کان پيڇو ڇڏائي اسان کي به خوفِ خدا نصيب ٿئي!

افسوس! في زمانه اسان ٻين مُعاملن سان گڏوگڏ پاڙيسرين جي حقن جي ادائيگي جي مُعاملي ۾ به پستي جي اونهي کڏ ۾ ڪرنديون پيون وڃون، هڪ ئي گهٽي، پاڙي ۾ سالن جا سال گُذري وڃڻ جي باوُجُود پنهنجي پاڙيواريءَ جي سڃاڻپ، ان جي موجودگي ۽ پاڙي جي حق کان غَفلت وڌندي پئي وڃي اسان کي بلڪل خبر ناهي هوندي ته اسان جي پاڙي ۾ ڪير رهي ٿي، ان جو نالو ڇا آهي، اسان ته پنهنجو پاڻ ۾ هن طرح گم آهيون جو پاڙيواري بيمار ٿي وڃي

(روزانو سنتن ڀريو بيان يا مدني مذاڪرو ٻڌڻ وارن سان اميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ تمام گهڻا خوش ٿيندا آهن)

دل خوفِ خدا سان ڏڪي ويو

باب المدينه (ڪراچي) جي هڪ اسلامي ڀيڻ فلمون ڊراما شوق سان ڏسندي ۽ نمازون قضا ڪري ڇڏيندي هئي. هڪ ڏينهن ان کي دعوت اسلامي جي مدني ماحول سان وابسته هڪ اسلامي ڀيڻ اميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي ٿيڻ واري”مدني مذاڪري“ جي ڪيسٽ ٻڌڻ جي لاءِ ڏني. ان مدني مذاڪرو ٻڌڻ شروع ڪري ڏنو. هڪ سوال جي جواب ۾ اميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ قبر جي اڪيلائين ۽ پريشانين جو اهڙو نقشو ڇڪيو جو ان اسلامي ڀيڻ جو دل خوفِ خدا سان ڏڪي ويو ۽ قبر جي وحشت ناڪ تصور ان جي جسم کي ڪانڊاري ڇڏيو. ان گهٻرائجي انهيءَ وقت سمورن گناهن کان توبهه ڪئي ۽ نمازن جي پابند بڻجي وئي. انهيءَ مدني مذاڪري جي برڪت سان ان کي فلمن ڊرامن جي نحوست کان به نجات ملي وئي. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ هاڻي نه صرف اها پاڻ دعوت اسلامي جي اسلامي ڀينرن جي هفته وار سنتن ڀري اجتماع ۾ شريڪ ٿينديون آهن بلڪ ٻين اسلامي ڀينرن کي به ان جي دعوت ڏينديون آهن. (فلمي اداڪاره جي توبهه ص20)

”جيڪڏهن توهان کي به دعوت اسلامي جي مدني ماحول جي ذريعي ڪا مدني بهار يا برڪت ملي هجي ته آخر ۾ مدني بهار مڪتب تي جمع ڪرايو.“

پياريون پياريون اسلامي ڀينرون! حضرت سَيدُنا صِدِّيقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي سيرتِ مُبارڪه مَحَبَّتِ رسول جي حقيقي آئينه دار هئي، جڏهن ته سندن ڪردار به عشقِ رسول جو بهترين مِعيار هو، اچو! حضرت سَيدُنا صِدِّيقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي عشقِ مُصطَفٰي تي مشتمل ايمان اَفروز واقعو ٻڌون ٿيون ته جيئن اسان جي مَحَبَّتِ رسول ۾ وڌيڪ اضافو ٿئي۔ جيئن

اسلام جي شروعات ۾ جڏهن صحابَه ڪرام جو تعداد (38) ٿيو ته حضرت سَيدُنا ابُوبڪر صدّيق رَضِىَ اللهُ تَعَالٰی عَـنْهُ سرڪارِ عالم، نُورِ مجسّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان اسلام جو کلي ڪري اِعلان ڪرڻ جي اِجازت طلب ڪيائين ۽ اِصۡرار ڪندا رهيا، ايستائين جو حُضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اظهارِ اسلام جي اِجازت عطا فرمائي۔ پوءِ حضرت سَيدُنا ابُوبڪر صدّيق رَضِىَ اللهُ تَعَالٰی عَـنْهُ ماڻهن کي اسلام جو خُطبو ڏيڻ جي لاءِ بيٺا ته پيارا آقا، مديني وارا مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم به اتي تشريف فرما هئا۔ اهڙي طرح پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي پهرين ڀيري نيڪي جي دعوت ڏيڻ جو شرف حاصل ٿيو۔ مڪي جي مُشرڪن جڏهن مُسلمانن کي ظاهر ظهور دعوتِ اسلام ڏيندي ڏٺو ته انهن کي ڪاوڙ آئي ۽ اهي حضرت سَيدُنا ابُو بڪر صديق رَضِىَ اللهُ تَعَالٰی عَـنْهُ ۽ ٻين مُسلمانن تي ڪاھ ڪري آيا ۽ انهن کي مارڻ شُروع ڪري ڇڏيو، صدّيقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي پڻ نهايت بُري طرح سان ماريو، ايستائين جو عُتبه بن ربيعه نالي خبيث، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي ويجھو آيو ۽ پنهنجي ناپاڪ جُتي پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي مُبارَڪ چهري تي هڻڻ لڳو ۽ سندن جي پيٽ تي چڙهي هلڻ لڳو ايستائين جو صديقِ اڪبر بي هوش ٿي ويا، زخمن جي ڪري سندن جي چهري جي سڃاڻپ مشڪل ٿي وئي۔ جڏهن سندن جي قبيلي بَنُو تَيم جي ماڻهن کي خبر پئي ته اهي ڊوڙندا آيا ۽ سندن کي مُشرڪن کان ڇڏائي گھر وٺي آيا، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي تشويشناڪ حالت ڏسي ائين لڳي رهيو هو ته زندھ بچي نه سگھندا، سندن والد ابُو قَحافه ۽ بَنُو تَيم جا ماڻهو نهايت پريشان هئا ۽ مسلسل پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان گفتگو ڪرڻ جي

ڪوشش ڪري رهيا هئا، آخرڪار ڏينهن جي آخري حصي ۾ اوهان کي هوش اچي ويو۔ پر هوش ۾ ايندي ئي زبان تان پهريون جملو اهو نڪتو ته رَسُوۡلُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ڪهڙي حال ۾ آهن؟ سندن جي اها ڳالھ ٻڌي قبيلي جا ڪيئي ماڻهو ناراض ٿي هليا ويا، سندن والده جڏهن ڪجھ کائڻ پيئڻ جي لاءِ چوندي هئي ته پاڻ صرف هڪ ئي جملو فرمائيندا هئا: رَسُوۡلُ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ڪهڙي حال ۾ آهن ؟ مون کي صرف انهن جي خبر ڏيو، جڏهن سندن کي خبر ملي ته رحۡمتِ عالم، نُورِ مُجَسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم خيريت سان آهن ۽ دارِ اَرقَم ۾ تشريف فرما آهن ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمايائون: خدا جو قسم! مان ان وقت تائين ڪجھ نه کائيندس ۽ نه پيئندس جيستائين پنهنجي محبوب آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي بذاتِ خود ڏسي نه وٺان۔ جڏهن سڀئي ماڻهو هليا ويا ته سندن والده ۽ اُمِّ جميل بنۡتِ خطّاب صديقِ اڪبر کي سهارو ڏئي شهنشاھِ مدينه، قرارِ قلب و سينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي بارگاھ ۾ وٺي ويون، جڏهن پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجي عاشقِ زار کي ڏٺو ته آبديده ٿي ويا ۽ اڳتي وڌي انهن کي هٿن ۾ ورتو ۽ انهن کي چمڻ لڳا، اهو منظر ڏسي تمام مسلمان به فرطِ جذبات ۾ صديقِ اڪبر ڏانهن وڌيا، زخمن سان ڀريل صديقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي ڏسي حضور عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام تي گھڻي رِقّت طاري ٿي، ان تي صديقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ عرض ڪيو: يارَسُوۡلَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم منهنجا ماءُ پيءُ اوهان تي قُربان، مان ٺيڪ آهيان بس چهرو ٿورو زخمي ٿي ويو آهي۔ (تاريخ مدينة دمشق،۳۰/۴۹، ملخصاً)

یہ اِک جان کیا ہے اگر ہوں کروڑوں

ترے نام پر سب کو وارا کروں میں

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون!! ٻڌو اوهان ته عاشق ِاڪبر سَيدنا صديقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان ڪيتري ڪامل درجي جو عشق هو، جسم زخمن جي ڪري چُور چُور آهي، جسم ۾ تڪليف آهي ۽ چهرو زخمي آهي،

پر پوءِ به حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي ياد فرمائي رهيا آهن ۽ اهو ئي نه بلڪ ڇا فرمايائون ته تيستائين سڪون نه ملندو جيستائين حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مُبارڪ چهري جي زيارت نه ڪري وٺان۔

اي محبتِ رسول جي دعوٰي ڪرڻ واريون اسلامي ڀينرون! سيدنا صديقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ ته دشمنن جي اڳيان به نيڪي جي دعوت ڏين ۽ اسان پنهنجي اسلامي ڀينرن کي گناهن ۾ ڏسڻ جي باوجود به نيڪي جي دعوت نه ڏيون، سيدنا صديق اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ نيڪي جي دعوت جي خاطر تڪليفون برداشت ڪن ۽ اسان جي لاءِ اسلامي ڀينرن محبت جي گلن جي ورکا ڪن اسان پوءِ به نيڪي جي دعوت جون ڌومون نه مچايون، سيدنا صديق اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ ته نيڪي جي دعوت جي خاطر تڪليفون برداشت ڪن ۽ اسان کي طرح طرح جون سهولتون ميسر هئڻ جي باوجود به نيڪي جي دعوت نه ڏيون ته سوچيو اسان محبتِ رسول ۾ سچيون آهيون يا اها محبت صرف زبان جي حد تائين آهي، ياد رکو! مَحَبَّت جون ڪجھ نشانيون هونديون آهن، جن کي هڪ عاشقِ صادق جي عشق ۽ مَحَبَّت جو دليل سمجھيو ويندو آهي، انهن نشانين کي ڏسي ٻين کي يقين ٿي ويندو آهي ته واقعي ان کي فُلاڻي سان بيحد مَحَبَّت آهي، مثال هر مُعاملي ۾ محبوب جي اِطاعت ڪرڻ، هر وَقت محبوب جي ذِڪر سان پنهنجي زبان کي تَر رکڻ، محبوب جي پسند کي اپنائڻ، محبوب جي ناپسنديده شين کان پري رهڻ۔ پوءِ جڏهن اها ئي مَحَبَّت پنهنجي اِنتها کي پهچي ويندي آهي ته عشق جو رُوپ اختيار ڪندي آهي۔ اسان جي بُزرگن کوڙ هنڌن تي عاشقن جي مختلف نشانين کي بيان فرمايو آهي۔ اچو! محبتِ رسول جون نشانيون ٻڌون ٿيون: جيئن

محبتِ رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جون نشانيون:

اِمَام قَسطلاني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه محبتِ رسول جون نشانيون بيان ڪن ٿا، جيڪو انهن کي اپنائي، اهو حقيقي عاشقِ رسول بڻجي ويندو۔ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ

(۱) اِقتدا (پيروي ڪرڻ) عشقِ رسول جي سڀ کان وڏي نشاني هي آهي جو پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اِقتدا ۽ پيروي، محبوبِ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ

وَسَلَّم جي سُنّت کي اپنائڻ، سندن جي رستي تي هلڻ، پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سيرت مان رهنمائي حاصل ڪرڻ ۽ سندن جي شريعت جي حُدود تي قائم رهڻ۔

(المواهب اللدنيه ،۲/۴۹۱)

(۲) شريعت تي راضي رهڻ: حضور نبي اڪرم، شهنشاھِ اُمم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت جي نشانين مان هڪ ڳالھ اها آهي ته هن عشق ۽ مَحَبَّت جي دعويٰ ڪرڻ وارو پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي شريعت تي راضي هجي۔

(المواهب اللدنيه ،۲/۴۹۳)

(۳) حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي دين جي مدد ڪرڻ: حضور نبي اڪرم، سراپا جُود و ڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت جي نشانين مان هڪ ڳالھ اها ته ”پنهنجي قول ۽ فعل جي ذريعي سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي دين جي مدد ڪري، پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي شريعت جو دِفاع ڪري ۽ سخاوت ۾ پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سيرتِ طيبه کي اپنائي، بُردباري ،صبر، تَواضُع (عاجزي) وغيره ۾پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مبارڪ سِيرت جي پيروي ڪئي وڃي ۔ (المواهب اللدنيه ،۲/۴۹۴)

(۴) مصيبتن تي صبرڪرڻ: نبي اڪرم، رسولِ محتشم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت جي نشانين مان هڪ ڳالھ اها ته ”مَصيبتن تي صبر ڪري، ڇوته هن مَحَبَّت سان مُحبّ کي اهڙي لذت حاصل ٿيندي آهي جو اهو مصيبت کي وساري ڇڏيندو آهي۔ (المواهب اللدنيه ،۲/۴۹۵)
