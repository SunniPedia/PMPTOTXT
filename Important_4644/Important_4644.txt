جنتي گھر دارُ الفَرح ڪنهن جي لاءِ آهي؟

حضرت سَيدَتُنَا عائشه صِدّيقه رَضِیَ اللهُ تَعَالٰی عَنْهَا روايت ڪن ٿيون ته رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”بيشڪ جنّت ۾ هڪ گهر آهي جنهن کي ”دار الفرح“ چيو ويندو آهي ان ۾ اهي ئي ماڻهو داخل ٿيندا جيڪي ٻارن کي خوش ڪندا آهن۔ (جامع صغير ، ص ۱۴۰ ، حديث ۲۳۲۱)

پيارا پيارا اسلامي ڀائرو! ٻار الله پاڪ جي نعمت هوندا آهن۔ انهن سان پيار ۽ محبت ڀريو ورتاءُ (Behave) ئي ڪرڻ گھرجي۔ ٻار پنهنجا هجن يا ٻين جا، ٻار ٻار ئي هوندا آهن۔ انهن سان هميشه مهرباني سان پيش اچڻ گھرجي۔ ڪي ماڻهو اهڙا خشڪ مزاج هوندا آهن جيڪي پنهنجي ٻارن تي به شفقت نٿا ڪن۔

حضرتِ سَيدُنا ابوهريره رَضِىَ اللهُ عَنْه کان روايت آهي ته رسولُ الله صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ حضرتِ سَيدُنا حسن بن علي رَضِىَ اللهُ عَنْهُمَا کي چميائون ته هڪ شخص عرض ڪيو: منهنجا ڏھ ٻار آهن مون (اڄ تائين) انهن مان ڪنهن کي ناهي چميو، ته پياري آقا صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ انهن کي ڏٺو پوءِ ارشاد فرمايائون: جيڪو رحم ناهي ڪندو اهو رحم ناهي ڪيو ويندو۔ (بخاري ، ڪ تاب الادب ، باب رحمة الولد ، ۴/ ۱۰۰ ، حديث: ۵۹۹۷)

۽ ڪجھ ماڻهو اهڙا به هوندا آهن جيڪي پنهنجن ٻارن لاءِ ته وڏا شفيق ۽ مهربان هوندا آهن پر ٻين جي ٻارن تي شفقت ۽ مهرباني ڪرڻ ڄڻ انهن جي ڊڪشنري ۾ شامل ئي ناهي، اهڙن ماڻهن کي ٻين جي ٻارن سان وڏي سخت الرجي ٿيندي آهي ۽ پنهنجي ٻارن جي علاوه ٻين جي ٻارن سان انهن جو رويو انتهائي سخت هوندو آهي۔ هن طرح جا ماڻهو پنهنجي ٻارن جي وڏي کان وڏي شرارت تي به سُڪون سان ويٺل رهندا آهن پر ٻين جي ٻارن جي معمولي شرارت تي به ڪاوڙ ۾ لال پيلا ٿي ويندا آهن۔ انهن جا پنهنجا ٻار طوفانِ بدتميزي به مچائن ته اهي پُرسڪون رهندا آهن جڏهن ته غيرن جي ٻارن جي هلڪي آواز تي به اهي وڙهڻ لڳندا آهن۔ اهو انتهائي غلط رويو (Conduct) آهي۔

ياد رکو! ٻار ته آخر ٻار هوندا آهن ۽ شرارتون ڪرڻ انهن جي فطرت ۾ شامل هوندو آهي پر ٻين جو ٻار ڪا شرارت يا غلطي ڪري ويهي ته ڀري محفل ۾ ان ٻار ۽ ان جي والدين کي دڙڪا ڏيڻ ۽ مارڪٽ ۽ لڙائي لاءِ تيار ٿيڻ سراسر ناداني ۽ بيوقوفي آهي۔ ۽ يقيناً اهو سڀ علمِ دين کان دوري ۽ تعليماتِ مُصطَفٰي کان لاعلمي جو نتيجو آهي۔ ياد رهي ته ٻارن تي شفقت به انهن جو حق آهي ۽ انهن جي تربيت به، تنهنڪري هن معاملي ۾ ميانه روي لازم آهي جو ٻار نه ئي ڊپ جي ڪري بلڪل گهٻرائجي وڃن ۽ نه ئي اسان جي بي جا نرميءَ سان بگڙي وڃن.

پاڻ سڳورن صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي ته اها عادتِ مبارَڪه هئي ته پاڻ صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي شفقت ۽ مَحَبَّت صرف پنهنجن ٻارن، ٻارڙين، ڏوهٽن ۽ ڏوهٽين سان خاص نه هئي بلڪ ٻين مسلمانن جا ٻار به شفقتِ مصطفيٰ جون خوب خوب برڪتون ماڻيندا هئا۔ اچو! ترغيب جي لاءِ ٻارن تي شفقتِ مصطفيٰ جون ڪجھ ايمان افروز جھلڪيون ملاحظه ڪريو، جيئن

نبي اڪرم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي پياري صَحابي حضرت سيدنا جابِر بِن عبد الله رَضِىَ اللهُ عَنْه جو بيان آهي ته سرڪارِ مدينه صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ کي تحفي ۾ هڪ ٿانوءَ ۾ حلوو پيش ڪيو ويو ته پاڻ سڳورن صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ اسان سڀني کي ٿورو ٿورو حلوو ڏنو، جڏهن منهنجي واري آٓئي ۽ مون کي هڪ ڀيري ڏيڻ کان پوءِ فرمايائون: ڇا توهان کي اڃا ڏيان؟ مون عرض ڪيو: جي ها، ته رسول اڪرم

صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ منهنجي ننڍي ڄمار جي ڪري مون کي مزيد (يعني وڌيڪ) ڏنائون، ان کان پوءِ جيڪي ماڻهو باقي رهي ويا هئا انهن کي انهن جو حصو ڏنو ويو۔ (شُعَبُ الْاِيمان ، ۵/ ۹۹ ، حديث: ۵۹۳۵ مُلَخَّصاً)

سُبْحٰنَ الله! سُبْحٰنَ الله! اسان جي پياري آقا صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ ننڍڙي صحابي حضرتِ سيدنا جابر بن عبدالله کي ٻين صحابه کان وڌيڪ حلوو ڏئي ڪهڙي طرح انهن کي خوش ڪيائون جو انهن ان واقعي کي ٻڌايو جيڪو حديث جي ڪتابن ۾ محفوظ ٿي ويو۔ نبي عَلَيه السَّلَام جو ٻارن تي شفقت جو هڪ ٻيو واقعو ٻڌو۔

نُغَيرجو ڪهڙو حال آهي؟

حضرتِ سَيدُنا اَبو طلحه رَضِىَ اللهُ عَنْه جي هڪ پٽ جو نالو ’’اَبُو عُمَير‘‘ هو، ان وقت هو کير پياڪ ٻار هئا هڪ ننڍي جھرڪي انهن سان گھڻي مانوس هئي، انهن سان کيڏندي هئي۔ نبي اڪرم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جڏهن انهن جي گھر تشريف کڻي ويندا هئا ته ارشاد فرمائيندا هئا: ’’ يا اَبَا عُمَير! مَا فَعَلَ النُّغَيرُ ‘‘ يعني اي اَبُو عُمَير، نُغَير (پکيءَ) ڇا ڪيو؟ (بخاري ، ڪ تاب الادب ، باب الانبساط الي الناس ، ۴/ ۱۳۴ ، حديث: ۶۱۲۹)

واه واه! منهنجا ماءُ پيءُ پياري آقا صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ تي قربان! ڪيترو پيارو انداز هو سندن جو۔ پيارا نبي عَلَيه السَّلَام جڏهن سفر تان واپس موٽندا هئا ته سندن انداز ڪهڙو هوندو هو، اچو ٻڌو!

صحابه ڪرام عَلَيهمُ الرِّضْوَان کي فرمائيندا هئا ته انهن کي پاڻ سان گڏ سوارين تي ويهاريو۔ ڪڏهن ڪڏهن اهي ٻار هن ڳالھ تي فخر ڪندا هئا ۽ هڪ ٻئي کي چوندا هئا ته رَسُولُ الله صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ مون کي پنهنجي اڳيان سوار ڪيو ۽ توکي پنهنجي پويان سوار ڪيو ۽ ڪي ٻار هينئن فخر جو اظهار ڪندا هئا ته سرڪارِ مدينه صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ انهن جي باري ۾ صحابه ڪرام عَلَيهمُ الرِّضْوَان کي فرمايائون ته انهن کي پاڻ سان سوارين تي ويهاري ڇڏيو۔ (احياء العلوم ، ۲/ ۷۱۲) ياد رکو ته هي پاڻ ڪريم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جو شان هو موجوده دور ۾ اسان مان اهڙي طرح رستي مان ٻين جي ٻارن کي ڪو به پنهنجي سواريءَ تي نٿو ويهاري سگهي. ڇو ته هن سان تمام گهڻا مسئلا پيدا ٿيڻ جو انديشو آهي.

پيارا پيارا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! آئنده هفتي اسان ”قرآنِ پاڪ جي اَهمِّيت“ جي باري ۾ ٻڌنداسين۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ رمضان المبارڪ جو رحمتن ڀريو مهينو اسان جي درميان جلوه گر آهي۔ هن مُبارڪ مهيني ۾ ڪيئي عاشقانِ رسول نماز، روزو، تراويح وغيره ۾ مشغول ٿي ويندا آهن۔ اهڙي طرح هڪ تعداد آهي جيڪي هن مهيني ۾ تلاوت قرآنِ پاڪ جي سعادت ماڻيندي آهي۔ رمضان کي قرآنِ پاڪ سان هڪ خاص نسبت حاصل آهي، انهيء مناسبت سان آئنده هفتي جي بيان ۾ اسان قرآنِ پاڪ جي اهميت، ان جون فضيلتون ۽ فائدا ۽ بزرگانِ دِين جي قرآنِ ڪريم سان محبت جي واقعن سان گڏوگڏ ٻيا کوڙ مدني گل ٻڌڻ جي سَعادَت حاصل ڪنداسين. لهٰذا! آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ اِنفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان چئو: اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ .

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

) 1 (جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي

هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

) 2 (سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

) 3 (رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

) 4 (ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([4])

) 5 (قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام

رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

) 6 (دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

) 1 (هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

) 2 (ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عسا ڪ ر ج 19 ، ص 155 ، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
