اَلْحَمْدُ لِلّٰہِ رَبِّ الْعٰلَمِیْنَ وَ الصَّلٰوۃُ وَالسَّلَامُ علٰی سَیِّدِ الْمُرْسَلِیْنَ ط

اَمَّا بَعْدُ فَاَعُوْذُ بِاللّٰہِ مِنَ الشَّیْطٰنِ الرَّجِیْمِ ط بِسْمِ اللہِ الرَّحْمٰنِ الرَّ حِیْم ط

اولياءِ ڪِرام جون برڪتون

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا نَبِیَّ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا نُوْرَ اللہ

نَوَيۡتُ سُنَّتَ الۡاِعۡتِكَافِ (مون سنّتِ اعتڪاف جي نيّت ڪئي)

مٺا مٺا اسلامي ڀائرو! جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي اعتڪاف جي نيت ڪري وٺو، ڇوته جيستائين مسجد ۾ رهندؤ اعتڪاف جو ثواب ملندو رهندو۔ ياد رکو! مسجد ۾ کائڻ، پيئڻ، سمهڻ يا سحري، افطاري ڪرڻ، ايستائين جو آٓبِ زم زم يا دَم ڪيل پاڻي پيئڻ جي به شرعاً اجازت ناهي ، البت جيڪڏهن اعتڪاف جي نيت هوندي ته اهي سڀ شيون ضمناً جائز ٿي وينديون۔ اعتڪاف جي نيت به صرف کائڻ، پيئڻ يا سمهڻ جي لاءِ نه هئڻ گھرجي بلڪ ان جو مقصد الله تعاليٰ جي رِضا هجي۔ ”فتاوي شامي“ ۾ آهي: جيڪڏهن ڪو مسجد ۾ کائڻ پيئڻ يا سمهڻ چاهي ته اعتڪاف جي نيت ڪري، ڪجھ دير ذِڪرُاللہ ڪري پوءِ جو چاهي ڪري (يعني هاڻي چاهي ته کائي پي يا سمهي سگھي ٿو) ۔

دُرُود شريف جي فضيلت

سَرڪارِ مدينه، راحتِ قلب و سِينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فَرمانِ رَحمت نِشان آهي: جيڪو مون تي دُرُود پاڪ پڙھندو آهي،ان جو دُرُود مون تائين پهچي ويندو آهي، مان ان جي لاءِ اِستِغفار ڪندو آهيان ۽ ان جي عِلاوه ان جي لاءِ ڏھ (10) نيڪيون لکيون وينديون آهن۔ (معجم اوسط ،من اسمہ احمد،۱/ ۴۴۶،رقم: ۱۶۴۲)

ذِکْر و دُرُود ہر گھڑی وِردِ زَباں رہے میری فُضُول گوئی کی عادَت نکال دو

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿا. فرمانِ مصطفي ٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِہٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس ٭ ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس ٭ ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي جي لاءِ بلند آواز سان جواب ڏيندس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺامٺا اسلامي ڀائرو! اَولِياءِ ڪرام رَحْمَةُ اللهِ تَعَالٰى عَلَيْهم اَجْمَعِيۡن اهي مُقَدَّس تَرِين هستِيون آهن جِن تي جبل سَلامَتي موڪليندا آهن۔ اَولِياءِ ڪرام کان وَحشِي جانوَر مانُوس هوندا آهن۔اَولِياءِ ڪرام کان جانوَر بَرَڪت حاصِل ڪندا آهن، اَولِياءِ ڪرام کان وڻ به قُرب ماڻيندا آهن۔ شَيطان اَولِياءِ ڪرام جي ويجھو ناهي ويندو۔ اَولِياءِ ڪرام دُنيا جي خَزانن ڏانهن نگاهون کڻي به ناهن نهاريندا۔ جِن جبلن (Mountains) تي اَولِياءِ ڪرام جا قَدَم پئجي ويندا آهن اهي جبل ٻين جبلن تي فَخر ڪندا آهن۔ اَولِياءِ ڪرام جڏهن دنيا کان تشريف کڻي ويندا آهن ته دُنيا وارا انهن جي جُدائي تي غمگين ٿيندا آهن۔ اَولِياءِ ڪرام جنّت جي

مَحَلّات ۾ رهندا۽ جنّت جي باغن ۾ سير ڪندا، اَولِياءِ ڪرام جنّت جي مٿاهين واري جاءِ ۾ رهندا ۽ جنّت جي عُمدَه سُوارين تي سُوار ٿيندا، اَولِياءِ ڪرام جنّت ۾ الله پاڪ جي ڪلام کي ٻڌندا ۽ ان جي دِيدار سان مُشَرَّف به ٿيندا۔ (حڪايتيں اور نصيحتيں،ص۱۳۸ملتقطاًوملخصاً)

اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ اڄ اسان اَولِياءِ ڪرام رَحْمَةُ اللهِ تَعَالٰى عَلَيْهم اَجْمَعِيۡن جي بَرَڪتن تي مُشتَمِل اِيمان اَفروز واقِعا ۽ حِڪايتون ٻڌڻ جي سَعادَت حاصِل ڪنداسين۔ اچو! سڀ کان پهريان مَشهور وَلِيه حضرت سَيدَتُنا رابِعه بَصرِيه رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا جي بَرَڪتن تي مُشتَمِل هڪ حِڪايت ٻڌ ڻ ک ان پهريان اوهان رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا جو تعارُف ٻڌون ٿا۔

حضرت رابعه بصريه رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا جو تعارف

”ماهنامه فَيضانِ مدينه“ صَفحي نمبر 39 تي آهي ته حضرت سَيدَتُنا رابِعه بَصرِيه رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا عابِده، زاهده، نيڪ ۽ پارسا ۽ خَوفِ خُدا واري عورت هئي، ڏينهن جو رَوزو رکنديون هيون ۽ تِلاوتِ قُرآنِ پاڪ ڪنديون هيون۔ سندن جي اڳيان ڪو جَهنَّم جو تذڪرو ڪندو هو ته خَوفِ خُدا جي ڪري بي هوش ٿي وينديون۔ (جنتي زيور،ص۵۴۱) پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا پيرين اگھاڙي، پَيدل بَيتُ الله شَرِيف جو حَج ڪيو، ڪعبي شريف پهچندي ئي(خوف ۽ خَشِيت سَبَب) بي هوش ٿي ڪِري پيون۔ (الروض الفائق،المجلس الثامن في ذڪر حجاج بيت الله …الخ،ص۶۰) پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا خُوب عِبادت ۽ رِياضَت ڪنديون هيون، ننڊ جو غَلَبو ٿيندو هو ته گھر ۾ هلڻ شُروع ڪنديون هيون ۽ پنهنجو پاڻ کي چونديون هيون: رابِعه! اها به ڪا ننڊ آهي، ان جو ڪهڙو مَزو؟ ان کي ڇڏ ۽ قَبر ۾ مَزي سان ڊگھي مُدَّت جي لاءِ ستل رهجان، اڄ ته توکي گھڻي ننڊ ناهي آئي پر ايندڙ رات ۾ خُوب ننڊ ايندئي،همَّت ڪر ۽ پنهنجي ربّ (عَزَّوَجَلَّ) کي راضي ڪري وٺ۔ پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا پنچاھ سال (Fifty years) ائين گذاري ڇڏيا جو نه ڪڏهن بِستر تي سُتيون ۽ نه ئي ڪڏهن وهاڻي تي مٿو رکيائون، ايستائين جو پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا اِنتِقال ڪري ويون۔ (حڪايات الصالحين،ص ۴۰ ملخصاً)(ماهنامه فيضان مدينه،اگست ۲۰۱۷،ص۳۹) الله ڪريم اِن جي طُفَيل اسان کي به عِبادت جو ذوق نصيب فرمائي۔

اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

چور وَلي بڻجي ويو

منقول آهي ته هڪ چور رات جي وقت حضرتِ سيدَتُنا رابعه بصريه رَحْمَةُ اللهِ تَعَالٰى عَلَيْها جي گھر ۾ داخل ٿيو، ان هر پاسي سڄي گھر جي تلاشي ورتي پر هڪ ڪونئري جي علاوه ڪا شيء نه ڏٺي، جڏهن ان نڪرڻ جو ارادو ڪيو ته پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْها فرمايائون: ''جيڪڏهن تون چالاڪ ۽ هوشيار چور آهين ته بنا ڪنهن شيء کڻي وڃڻ جي نه ويندين۔''ان چيائين:''مون کي ته ڪا شيء نه ملي۔'' پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْها فرمايائين: ''اي غريب شخص! ان ڪونئري مان وضو ڪري ڪمري ۾ داخل ٿي ۽ ٻه رڪعت نماز ادا ڪر، هتان کان ڪجھ نه ڪجھ کڻي ويندين۔'' ان سندن رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا جي چوڻ مطابق وضو ڪيو ۽ جڏهن نماز جي لاءِ بيٺو ته حضرت سيدَتُنا رابعه رَحْمَةُ اللهِ تَعَالٰى عَلَيْها ائين دُعا ڪئي: ''اي منهنجا ربّ عَزَّوَجَلَّ! هي شخص مون وٽ آيو پر ان کي ڪجھ نه مليو، هاڻي مون ان کي تنهنجي بارگاھ ۾ بيهاري ڇڏيو آهي، ان کي پنهنجي فضل ۽ ڪرم سان محروم نه ڪجان۔'' جڏهن هو نماز کان فارغ ٿيو ته ان کي عبادت جي لذت نصيب ٿي۔ پوءِ، هو رات جي آخري حصي تائين نماز ۾ مشغول رهيو۔ جڏهن سحري جو وقت ٿيو ته پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْها ان وٽ ويون ته ان کي سجدي جي حالت ۾ پنهنجي نفس کي ڌڙڪا ڏيندي ۽ هي چوندي ڏٺو: جڏهن منهنجو رب عَزَّوَجَلَّ مون کان پڇندو: ''ڇا تو کي حيا نه آيو ته تون منهنجي نافرماني ڪندو رهين؟ ۽ منهنجي مخلوق کان گناھ لڪائيندو رهين ۽ هاڻي گناهن جي ڀَري کڻي منهنجي بارگاھ ۾ پيش آهين، جڏهن هو مون تي غضب فرمائيندو ۽ پنهنجي رحمت ڀري بارگاھ مان پري ڪندو ته ان وقت مان ڪهڙو جواب ڏيندس؟ پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْها پڇيائين:''اي ادا! رات ڪيئن گذري ؟'' چيائين: ''خيريت سان گذري، عاجزي ۽ انڪساري سان مان پنهنجي رَبّ عَزَّوَجَلَّ جي بارگاھ ۾ بيٺل رهيس ته ان منهنجي خراب عادتن کي درست ڪري ڇڏيو،منهنجو عذر قبول ڪري ورتو ۽ منهنجي گناهن کي بخشي ڇڏيو ۽ مون کي منهنجي مطلوب ۽ مقصود تائين پهچائي ڇڏيو۔ پو اهو شخص چهري تي حَيراني ۽ پريشاني جا آثار کڻي هليو ويو۔

حضرت سيدَتُنا رابِعه بَصرِيه رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا پنهنجا هٿ آسمان ڏانهن بلند ڪيا ۽ عَرض گذار ٿيون: اي منهنجا رَبّ! هي شخص تنهنجي بارگاھ ۾ هڪ گھڙي بيٺو ته تو ان کي قبول ڪري ورتو ۽ مان ڪڏهن کان تنهنجي بارگاھ ۾ بيٺي آهيان، ته ڇا تو مون کي به قَبول فرمائي ورتو آهي؟ اوچتو پاڻ رَحْمَةُ اللهِ تَعَالٰى عَلَيْهَا دِل جي ڪنن سان اهو آواز ٻڌو: اي رابِعه! اسان ان کي تنهنجي ئي ڪري قَبول ڪيو ۽ تنهنجي ئي ڪري پنهنجو قُرب عطا ڪيو آهي۔ (الروض الفائق،ص ۱۵۹ ملخصاً)

مَحَبَّت میں اپنی گُما یااِلٰہی نہ پاؤں میں اپنا پَتا یااِلٰہی

رہُوں مَسْت و بے خُود میں تیری وِلا میں پِلا جام ایسا پِلا یااِلٰہی

مِرے دِل سے دُنیا کی چاہت مِٹا کر کر اُلْفَت میں اپنی فَنا یااِلٰہی

(وسائلِ بخشش مرمم،ص۱۰۵)

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! ٻڌو اوهان ته اَولِياءِ ڪرام رَحْمَةُ اللهِ تَعَالٰى عَلَيْهم اَجْمَعِيۡن جي قُرب جون ڪهڙيون ڪهڙيون بَرَڪتون ظاهر ٿينديون آهن جو انهن جي دَر تي ايندڙ ڪڏهن به هٿين خالي موٽايو ناهي ويندو، بلڪ پنهنجي خالي دَامَن کي موتين جي مُرادن سان ڀري ۽ خُوب خُوب بَرَڪتون ماڻي موٽندو آهي، ايستائين جو جيڪڏهن ڪو چور به انهن جي بارگاھ ۾ اچي وڃي ۽ انهن جي ناياب صُحبَت جي نعمت ماڻڻ ۾ ڪامياب ٿي وڃي ته پوءِ اهو چور، چور نٿو رهي بلڪ ربّ ڪريم ان گناهگار ۽ بَدڪردار کي به هنن مَقبول ٻانهن جي نِگاھ وِلايت، دِل جي گهرائين مان نڪرندڙ پُرتاثير دُعائن ۽ پاڪيزه صُحبَت جي بَدولَت ان جي زَنگ آلُود باطِن کي چَمڪائي ان کي گُناهن تي نَدامَت، نَماز، تَقويٰ ۽ پرهيزگاري، سچّي تَوبه ۽ نيڪين جي تَوفِيق، فِڪرِ آخِرت جهڙي شاندار نعمتن سان سَرفَراز ڪري بَخشش ۽ مَغفِرت جي سَند عطا فرمائيدو آهي، تنهنڪري اسان کي به کپي ته اسان به اَدَب جو دامَن پڪڙي انهن کان دُنياوِي مطلبن ۽ مقصدن بدران

اَولِياءِ ڪرام الله ڪريم جي رَحمت آهن،جيڪي عيش ۽ راحت جي زندگي گذارڻ بدران هميشه سادگي کي اپنائيندا آهن، اَولِياءِ ڪرام شَرِيعَت جي پابندي ڪندا آهن۽ باطِن کي صاف ڪندا، اَولِياءِ ڪرام الله پاڪ جي ٻانهن کي الله پاڪ سان ملائيندا ۽ اُنهن کي به الله ڪريم جو پِيارو بڻائيندا آهن، اَولِياءِ ڪرام جي صُحبَت ۾ ويهڻ سان دُنيا ۽ آخِرت سَنوَرجي ويندي آهي ۽ کوڙ ڀلائيون نصيب ٿينديون آهن، اَولِياءِ سڳورا، الله تعاليٰ جي علاوه ٻي ڪنهن کان نٿا ڊڄن، اَولِياءِ ڪرام مُتَّقِي ۽ پَرهيزگار هئڻ جي باوُجود جَهنَّم (Hell) جي خَوف مان ڏڪندا رهندا آهن، اَولِياءِ ڪرام ڏينهن رات عِبادتِ الٰهي ۾ مشغول رهندا آهن، اَولِياءِ ڪرام مخلوق جي مدد ڪندا آهن، اَولِياءِ ڪرام کي الله پاڪ جي عطا سان عِلمِ غَيب حاصِل ٿيندو آهي، اَولِياءِ ڪرام دِلين جي اَحوال کان باخَبَر هوندا آهن ايستائين جو اَولِياءِ ڪرام دُنيا جي اندر ئي جنّتي مَحل جو حقدار بَڻائي ڇڏيندا آهن۔ اچو! مَشهور مَجذُوب ۽ زَبَردَست وَلِي حضرت سَيدُنا بهلول دانا رَحْمَةُ اللهِ تَعَالٰى عَلَيْه جي هڪ اِيمان اَفروز ڪرامت جي باري ۾ ٻڌو ۽ پنهنجي دل ۾ مَحَبَّتِ اَولِيا اُجاگر ڪريو، جيئن ته

ترجمو ڪنز الايمان: ۽ تُون اُنهن کي جاڳندو سمجھندين حالانڪ اُهي سُتل آهِن. ۽ اسان سڄي کٻي سندن پاسا مٽائِيندا آهيُون ۽ سندن ڪُتو پنهنجُون ڪَرائيون پکيڙِيندڙ آهي غار جِي چائٺ تي .

هن آيت ۾ اَصحابِ ڪهف (رَضِي الله تَعَالٰي عَنْهم) جيڪي اَولِيا آهن۔انهن جون ٽي (3) ڪرامتون بَيان ٿيون: (1) جاڳڻ وانگر هاڻي تائين سمهڻ، (2) ربّ جي طرفان پاسا مٽائڻ، زمين جو انهن جي جسمن کي نه کائڻ ۽ بغير غذا جي باقي رهڻ، (3) انهن جي ڪتّي جو هاڻي تائين ليٽ يو رهڻ هي به انهن جي ڪرامت آهي نه ئي ڪتي جي۔ (علم القرآن،ص ۱۸۸بتغير)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺامٺا اسلامي ڀائرو! اَولِياءِ ڪرام رَحْمَةُ اللهِ تَعَالٰى عَلَيْهم اَجْمَعِيۡن جي بَرَڪتن جي به ڇا ڳالھ ڪجي! هي اهي عظيم هستيون آهن جو جِن جو ڪردار هڪ صاف شفاف آئيني وانگر هوندو آهي، جيڪي مَخلوقِ خُدا جي هدايت جي لاءِ هر وَقت تيار رهندا آهن، جن جي دِل ۾ اُمَّت جي اصلاح جو جَذبو تمام گھڻو موجود هوندو آهي، جن جا بيان، اِرشاد ۽ تَحريرن جي بَرَڪتن سان گناهگار ۽ بَدڪار ماڻهو قُرآن ۽ سُنّت جي راھ تي هلندا جڏهن ته غير مُسلِم اسلام جي دَولت کان سَرفَراز ٿيڻ جي سَعادت ماڻيندا آهن۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ اڄ جي هن پُرفِتَن زماني ۾ پندرهين صَدِي جي عظيم عِلمي ۽ رُوحاني شَخصِيت، شَيخِ طَرِيقَت، اَميرِ اَهلسُنّت، باني دعوتِ اسلامي حضرت علّامه مَولانا ابُوبِلال محمد الياس عطّار قادِرِي رَضَوِي ضِيائي دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه اها عظيم هستِي آهي، جِن ۾ بَيان ڪيل اَولِياءِ ڪامِلِين

رَحْمَةُاللهِ تَعَالٰى عَلَيْهم اَجْمَعِيۡن جون نِشانيون آسانيءَ سان ڏسي سگھجن ٿيون ۔ اچو! اسان به ان عظيم ولي ڪامِل جي بَرَڪتن جو مُختَصَر تَذڪرو ٻڌون ٿا، جيئن ته

فَيضانِ اَميرِ اَهلسُنّت جون بَرَڪتون

شَيخِ طَرِيقَت، اَميرِاَهلسُنّت دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه فرض ۽ واجِبات جي ادائيگي سان گڏوگڏ سُنّتن ۽ مُسْتَحَبّن تي عمل پيرا ٿي نيڪي جي دَعوَت جون اهڙيون ڌُومون مچايون جو لکين مسلمانن خاص ڪري نَوجوانن کي گناهن کان تَوبه جي تَوفِيق مِلي ۽ هو توبه ڪري قُرآن ۽ سُنّت جي راھ تي هلڻ لڳا۔ جيڪي بي نمازي هئا نمازي بلڪ مسجدن جا اِمام بَڻجي ويا، بَدنِگاهي ڪرڻ وارا نِگاهون هيٺ رکڻ جي سُنّت تي عمل ڪرڻ جي سَعادت ماڻڻ لڳا، چمڪيلا پر ڪشش لِباس پائي ڳچيءَ ۾ رئو لڙڪائي تفريح گاهن جي زينت بڻجڻ واريون عورتون بي پردگي کان اهڙي توبهه ڪيائون جو مدني برقعو انهن جي لباس جو اهم حصو بڻجي ويو۔ ماءُ پيءُ سان گستاخانه اَنداز اختيار ڪرڻ وارا با ادب ٿي ويا، جن جي حَرَڪتن سبب ڪڏهن سڄو پاڙو بيزار هو، اهي سڄي عَلائقي جي اکين جا تارا بڻجي ويا، چوري ۽ ڌاڙي جا عادِي ٻين جي عِزت ۽ آبرُو جي حِفاظت ڪرڻ لڳا، ڪنهن غريب کي ڏسي تَڪبُّر سبب منهن ڦٽائيندڙ عاجِزِي جا پيڪر بڻجي ويا، هر وَقت حَسَد جي باھ ۾ سڙندڙ ٻين کي تَرَقِّي جون دُعائون ڏيڻ لڳا، ڳائڻ وڄائڻ جا شَوقِين، سُنّتن ڀريا بَيان ۽ مَدَني مُذاڪرا ٻڌڻ جا عادِي ٿي ويا، فُحْش ڪلامي ڪرڻ وارا نَعتِ مُصطَفٰي پڙھڻ ۽ عشق مصطفٰي ۾ جُھومڻ جهومائڻ لڳا، يورپي ملڪن جي رنگينين کي ڏسڻ جا خواھش مند ڪعبي شريف ۽ گنبذِ خضرٰي جي زيارت لاءِ بيقرار رھڻ لڳا، مال جي مَحَبَّت ۾ گِرِفتار رهڻ وارا فڪرِ آخِرت جي مَدَني سوچ رکندڙ بڻجي ويا، شراب پيئڻ وارا عشقِ مُصطَفٰي جا جام پيئڻ لڳا، فُضُولِيات ۾ وَقت بَرباد ڪرڻ وارا پنهنجو وَقت عِبادت ۾ گُذارڻ لڳا، فُحش رَسالا ۽ ڊائجسٽ جا شَوقِين اميرِاَهلسُنّت دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه ۽ مڪتبةُ المدينه جا ڪتاب ۽ رسالا ۽ ماهنامه فَيضانِ مدينه جو مُطالَعو ڪرڻ لڳا، تَفرِيح (Amusement) جي خاطِر سَفَر جا عادِي شخص عاشِقانِ رَسول سان گڏ راھ خُدا ۾ سَفَر ڪرڻ وارا

بڻجي ويا، ”کاؤ پِيو ۽ جان بڻايو“ جي نعري کي پنهنجي زِندگي جو مَقصَد سمجھڻ وارن هن مَدَني مَقصَد کي اپنائي ورتو ته ” مون کي پنهنجي ۽ سڄي دنيا جي ماڻهن جي اصلاح جي ڪوشش ڪرڻي آھي. اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ “ اميرِ اَهلسُنّت دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه جو اهو فَيضان صِرْف مُسلمانن تائين مَحدود ناهي رهيو بلڪ ڪثير غير مُسلمن کي به نُورِ اسلام نصيب ٿيو۔ (تذڪره امير اهلسنت(قسط1)،ص۸تا۱۰بتغير)

امير اهلسنت جي هن فيضان کي ڪنهن عاشق رسول ڪيتري جامع ۽ پياري انداز ۾ اشعار جي صورت ۾ لکيو آهي،جيئن ته لکن ٿا:

مَسْلک کا تُو اِمام ہے اِلیاس قادِرِی تَدْبِیر تیری تام ہے اِلیاس قادِرِی

سُنّت کی خوشبؤوں سے زَمانہ مہک اُٹھا فَیْضان تیرا عام ہے اِلیاس قادِرِی

سَر پہ عِمامہ ماتھے پہ سجدوں کا نُور ہے جو بھی تیرا غُلام ہے اِلیاس قادِرِی

ہے دَعْوَتِ اِسلامی کی دُنیا میں دُھوم دھام مَقبول تیرا کام ہے اِلیاس قادِرِی

(محبوبِ عطار کی 122حکایات،ص۲۰۳)

12مدني ڪمن مان هڪ مدني ڪم ’’هفتيوار مدني مذاڪرو‘‘

مٺا مٺا اسلامي ڀائرو! ٻڌو اوهان ته! شَيخِ طَرِيقَت، اَميرِ اَهلسُنّت دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه ڪيتري قَدَر عظيم وَلِي آهن جو جِن جي بَرَڪت ۽ دِيني خِدمتن سان مُعاشَري ۾ تمام ٿوري ئي وَقت ۾ حَيرت اَنگيز مَدَني اِنقِلاب بَرپا ٿي ويو۔ اچو! اوهان به فَيضانِ اَميرِ اَهلسُنّت دَامَتْ بَـرَ ک اتُهمُ الْعَالِيه سان فَيضياب ٿيڻ جي لاءِ فَيضانِ اَولِيا سان مالا مال عاشِقانِ رَسُول جي مَدَني تَحريڪ دَعوتِ اِسلامي جي مدني ماحول سان وابسته ٿي 12 مَدَني ڪمن ۾ حصّو وٺو۔ 12 مَدَني ڪمن مان هفتيوار هڪ مَدَني ڪم ”مَدَني مذاڪرو“ به آهي۔

٭ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ ”مَدَني مذاڪرو“ ڏسڻ، ٻڌڻ جي بَرَڪت سان شَرعي مُعامَلَات ۾ اِحتياط ڪرڻ جو ذِهن نصيب ٿيندو آهي ٭ مَدَني مذاڪري جي بَرَڪت سان عاشقانِ رسول جي صُحبت مُيسَّر ٿيندي آهي، ٭ مَدَني مذاڪري جي بَرَڪت سان عمل جو جَذبو وڌندو ٭ گناهن کان نفرت پيدا ٿيندي آهي ٭ مَدَني مذاڪري جي بَرَڪت

سان دعوتِ اسلامي جي مَدَني ماحول ۾ اِستقامت نصيب ٿيندي آهي۔ ٭ عاشقانِ رسول جي مدني تحريڪ " دعوتِ اسلامي " جي اپ ڊيٽس (Updates) ملندي آهي۔ ٭ مَدَني مذاڪرو علمِ دِين جي ترقي جو سبب آهي ٭ مَدَني مذاڪرو، اميرِ اهلسنت جي زندگي جي سوين بلڪ هزارين تجربات سان مدني تربيت حاصل ڪرڻ جو بهترين ذريعو آهي۔ ٭ مَدَني مذاڪري ۾ دِيني معلومات ملڻ سان گڏوگڏ اَخلاقي تربيت به ٿيندي آهي۔ ٭ مدني مذاڪري جي بَرَڪت سان اميرِ اَهلسُنَّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه سان ڪيل مُختلف سُوالات بابت دِلچسپ جوابات جي صُورت ۾ عِلْمِ دِين حاصل ٿيندو آهي ۽ عِلمِ دِين سکڻ سيکارڻ جي ته ڇا ڳالھ آهي جو رَحمتِ عالَم، نُورِ مُجَسَّم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ اِرشاد فرمائن ٿا: جيڪو الله پاڪ جي فرضن جي باري ۾ هڪ يا ٻه يا ٽي يا چار يا پنج ڪلِمات سکي ۽ ان کي چڱي طرح ياد ڪري ۽ پوءِ ماڻهن کي سيکاري ته هو جنّت ۾ داخِل ٿيندو۔ (الترغيب والترهيب، ڪتاب العم،۱ /۵۴، حديث:۲۰)

مٺا مٺا اسلامي ڀائرو! اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ ڪيئي اسلامي ڀائر مدني مذاڪري جي برڪت سان پنهنجي گُناهن ڀري زِندگي کان توبه ڪري چڪا آهن۔ ان ڪري اچو! هٿو هٿ نيت ڪريون ٿا ته اسان به هر هفتي ”مَدَني مذاڪري“ ۾ اَوَّ ل تا آخر شرڪت کي يقيني بڻائينداسين ۽ ٻين اسلامي ڀائرن کي به مدني مذاڪري ۾ شرڪت جي دعوت ڏيندا رهنداسين، اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ان جون خوب خُوب برڪتون حاصل ٿينديون۔ هفتيوار مدني مذاڪري جي علاوه مختلف موقعن تي به مدني مذاڪرن جو سلسلو ٿيندو آهي، مثال طور مُحَرَّمُ الْحَرَام جا 10 مدني مذاڪرا، ماهه ربيعُ الاول (ٻارهين) جا 12 مدني مذاڪرا، ماهه ربيعُ الاخر (يارهين) جا 11 مدني مذاڪرا، رمضان ۾ روزانو 2 مدني مذاڪرا، ذُوالحِجَّة الحَرام جا 10مدني مذاڪرا وغيره۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ هنن مدني مذاڪرن ۾ به اوّل تا آخر شرڪت ڪري دُنيا ۽ آخرت جون بيشمار برڪتون ۽ ڀلائيون حاصل ڪري سگھجن ٿيون۔ اچو تَرغِيب جي لاءِ مدني مذاڪرو ڏسڻ جي هڪ مدني بهار ٻڌو ۽ خوش ٿيو، جيئن ته

مدني مذاڪري سڌاري ڇڏيو

واه ڪينٽ (پنجاب، پاڪستان) جوهڪ اسلامي ڀاءُ دعوت اسلامي جي مدني ماحول کان اڳ کوڙ نوجوانن جيان ڪافي اَخلاقي بُرائين ۾ مبُتلا هو۔ فلمون ڊراما ڏسڻ، ،راند روند ۾ وَقت برباد ڪرڻ ان جو محبوب وندر هو۔ گھر ۾ مدني چينل هلڻ جي برڪت سان ان کي مدني مُذاڪرو ڏسڻ جي سَعادت نصيب ٿي، اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ اڳوڻن سڀني گُناهن کان توبه ڪري فرض ۽ واجب جو عامِل بڻجي ويو، چهري تي هڪ مٺ ڏاڙھي شريف سجائي ورتي ۽ مدني حُليو به اَپنائي ورتو آهي، الله عَزَّوَجَلَّ جي ڪرم سان انهن جي والِدَين خوشي سان ’’وقفِ مدينه‘‘(دعوتِ اسلامي جي ڪمن لاءِ پيش) ڪري ڇڏيو آهي۔

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([1])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([2])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([3])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([1])

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([2])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
