۽ اَحاديثِ مُبارَڪه ۾ به مختلف هنڌن تي اوهان رَضِىَ اللهُ تَعَالٰی عَـنْهُ جا فضائل ۽ مَناقِب به بيان ڪيا ويا آهن، اچو! هن باري ۾ 2 فرمان مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌون ٿا:

1. مَنْ سَرَّهُ اَنْ يَّنْظُرَ اِلٰى شَهِيْدٍ يَمْشِى عَلٰى وَجْهِ الْاَرْضِ فَلْيَنْظُرْ اِلٰى طَلْحَةَ بْنِ عُبَـيدِالله يعني جيڪو زمين تي هلندڙ ڦرندڙ (يعني زندهه) شهيد جي زيارت سان خُوش ٿيڻ چاهي ٿو ته اهو طلحه بن عُبَيدُ الله (رَضِىَ اللهُ تَعَالٰی عَـنْهُ) کي ڏسي وٺي۔

2. طَلْحَةُ وَالزُّبَيْرُ جَارَاىَ فِى الْجَنَّۃِ يعني طلحه ۽ زُبَير (رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا) جنّت ۾ منهجا پاڙيسري ٿيندا۔ (ترمذی، کتاب المناقب،باب مناقب ابی محمد طلحۃ بن عبید اللہ،حدیث:۳۷۶۲، ۵/۴۱۳)

بزرگانِ دِين رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی اَجۡمعِیۡن ڪيئن هئا؟ ته جيئن توهان کي پنهنجي خرابي ۽ صحابه ڪرام عَـلَيْهِمُ الرِّضْوَان جي فضيلت معلوم ٿي وڃي۔ ڪن صحابه سڳورن عَـلَيْهِمُ الرِّضْوَان وٽ مال هو، پر انهن جو مقصد سوال کان بچڻ ۽ راھ خدا ۾ خرچ ڪرڻ هو، ان ڪري انهن حلال ڪمايو، پاڪيزه کاڌو، ميانه رَوِي سان خرچ ڪيو ۽ (صدقي ۽ سخاوت جي ذريعي پنهنجي مال کي) پنهنجي آخرت جي لاءِ اڳتي موڪليو، جيڪو ڪجھ انهن تي مالي حقَ لازم هئا، ان ۾ انهن ڪوتاهي نه ڪئي ۽ نه ئي بخل ڪيو، بلڪ انهن اڪثر مال الله عَزَّوَجَلَّ جي رِضا جي لاءِ خرچ ڪري ڇڏيو، ايستائين جو انهن مان ڪن ته سڄو مال ئي راھ خدا ۾ خرچ ڪري ڇڏيو، ۽ تنگي جي وقت ۾ اڪثر الله عَزَّوَجَلَّ جي رضا کي پنهنجي ذات تي ترجيح ڏني، توهان ٻڌايو! ڇا توهان به اهڙا آهيو؟ الله جو قسم! توهان انهن سان ذري برابر به مُشابَهت نٿا رکو۔ اڪابر صحابه ڪرام عَـلَيْهِمُ الرِّضْوَان مسڪين رهڻ پسند ڪندا هئا، محتاجي جي ڊپ کان بي خوف رهندا هئا ۽ رزق جي معاملي ۾ الله عَزَّوَجَلَّ تي يقين رکندا هئا۔ الله عَزَّوَجَلَّ جيڪو ڪجھ انهن جي لاءِ مُقَدَّر فرمايو آهي، ان تي خُوش رهندا هئا، مصيبت ۽ آفت واري حالت ۾ رَبّ عَزَّوَجَلَّ جي رضا تي راضي، خُوشحالي ۾ شڪر گذار، تنگي ۾ صابر، راحت ۾ ان جي ساراھ ڪرڻ وارا هئا۔ الله عَزَّوَجَلَّ جي لاءِعاجزي ڪرڻ وارا ۽ غرور وتڪبُّر ۽ مال جي گھڻائي تي فخر کان پري رهڻ وارا هئا، انهن دنيا کي ٺوڪر هنئي ۽ ان جي سختين تي صبر ڪيو ۽ ان جو ڪوڙو ڍڪ پيتو ۽ ان جي نعمتن ۽ آسائشن کان بي پرواھ رهيا، ڇا توهان به اهڙا آهيو؟ جڏهن دُنيا انهن ڏانهن مُتَوَجّھ ٿيندي هئي ته اهي غمگين ٿي ويندا هئا ۽ (عاجزي ڪندي) فرمائيندا هئا ته اهو ڪهڙي گناھ جي سزا آهي، جيڪا انهن کي هڪدم ملي آهي، ۽ جڏهن فقر کي پاڻ ڏانهن ايندي ڏسندا هئا ته فرمائيندا هئا، صالحين عَلَيهم رَحْمَۃُ اللّٰه ِالْمُبِين جي نشاني جو اچڻ مبارڪ هجي ۔جڏهن انهن کي مالي آسُودَگي حاصل ٿيندي ته غمگين ٿيندا ۽ ڊپ ڪندا ۽

اچو سخاوت جو وصف حاصل ڪرڻ جي لاءِ ترغيب طور ڪي تدبيرون ٻڌون ٿا: (1) وقتاً فوقتاً بارگاھ الٰهي ۾ هن وصف جي لاءِ دعا ڪيو ڇو ته دُعا جي برڪت سان وڏا وڏاڪم بڻجي ويندا آهن ۽ حديثِ شريف ۾ فرمايو ويو آهي: اَلدُّعَآءُ سِلَاحُ الْمُؤمِنِ يعني دُعا مومن جو هٿيار آهي۔ ([2]) (2)سخاوت جي فضيلت بابت آيتن، حديثن، روايتن ۽ واقعن کي هر هر پڙهو يا ٻڌو ۽ ان تي عمل ڪرڻ جي ڪوشش ڪيو ۔(3) بُخْل جي مَذَمّت بابت آيتن، حديثن، روايتن ۽ واقعن ۾غور و فڪر ڪيو ۽ ان کان بچڻ جي ڪوشش ڪندا رهو (4)سخاوت جي صلي ۾ ملڻ واري عظيمُ الشَّان ثواب ۽ رِضاءِ الٰهي ۽ بعض صُورتن (مَثَلاً زڪوٰۃ ۽ صدقاتِ واجبه جي ادائيگي) ۾ بُخل جي ڪري ملڻ واري دردناڪ عذاب ۽ الله عَزَّوَجَلَّ جي ناراضگي کي اکين آڏو رکو. (5) پنهنجي مَوت کي ڪثرت سان ياد ڪيو ته منهنجي مرڻ کان پوءِ

اهو مال منهنجي ڪنهن ڪم نه ايندو ۽ سڀ هتي ئي رهجي ويندو، ان ڪري سخاوت ڪرڻ ۾ ئي عافيت آهي۔ (6) نَفس ۽ شيطان جڏهن به بُخل ڪرڻ جو مشوره ڏين ته هميشه ان جو الٽ ڪيو، سخاوت جي ڪري مال ۾ گھٽتائي بدران الله عَزَّوَجَلَّ جي ذات تي ڀروسو ڪيو ۽ نفس تي جبر ڪندي نيڪ ڪمن ۾ خرچ ڪرڻ ۽ غريبن جي مدد ڪرڻ کان هرگز گرگز نه ڪيٻايو (7) اهڙن ماڻهن جي صحبت اِختِيار ڪيو جن جي برڪت سان خرچ ڪرڻ جو ذهن بڻجي ۽ اهڙن کان پري ڀڄو جيڪي ڪنجوسي ڏانهن رهنمائي ڪن (8) سخاوت جو وصف حاصل ڪرڻ جي لاءِ مڪتبة المدينه طرفان شايع ڪيل حُجَّۃُ الْاِسْلَام حضرت سَيدُنا امام محمد بن محمد غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا ڪتاب مثال طور اِحياءُ العُلوم (مُتَرْجَم،جلد3) ، مُکاشَفَۃُ الْقُلُوب ، مِنْهاجُ الْعَابِدِين ۽ شيخِ طريقت، اميرِ اهلسنّت حضرت علامه مولانا ابو بلال محمد الياس عطار قادري رضوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي رِسالي "مديني جي مڇي" ۽ المدينةُالعلميه جا ڪتاب ۽ رسالا مثال طور ” ضيائے صَدَقات “ و” صدقے کا انعام “ وغيره جو مُطالعو ڪيو (9) روزانو فڪرِ مدينه ڪندي مدني انعامات جو رسالو به پُر ڪيو، اَ لْحَمْدُلِلّٰہ عَزَّوَجَلَّ مدني انعامات جي ذريعي شيخِ طريقت،امِيرِ اهلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه پنهنجي مُريدن ۽ محبت ڪندڙن کي جتي ٻين سٺين سٺين صفتن ۽ عادتن اختيار ڪرڻ جي ترغيب ڏياري آهي اتي سخاوت جھڙي عمده صفت اپنائڻ جي لاءِ انهن کي عالم سڳورن جي خير خواهي ڪرڻ جو به ذهن ڏنو آهي جيئن ته مدني انعام نمبر 62 آهي:” ڇا توهان هن مهيني ڪنهن سني عالم (يا امامِ مسجد، مؤذن، خادم) کي 112 روپيا يا گهٽ ۾ گهٽ 12 روپيا تحفةً پيش ڪيا؟ (نابالغ پنهنجي ذاتي رقم نه ٿو ڏئي سگهي) ۔“ اِنْ شَآءَالله عَزَّوَجَلَّ

مٺا مٺا اسلامي ڀائرو! سخاوت جي علمبردار حضرت سَيدُنا طلحه بن عُبَيدُ الله رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي فيض مان فيضياب ٿيڻ ۽ صحابه ڪرام عَـلَيْهِمُ الرِّضْوَان جي سيرت تي عمل ڪرڻ جي لاءِ دعوتِ اسلامي جي مدني ماحول سان وابسته رهو ۽ ذيلي حلقي جي 12 مدني ڪمن ۾ وڌ کان وڌ حصو وٺو، ذيلي حلقي جي 12 مدني ڪمن مان هفتي وار هڪ مدني ڪم ”علاقائي دَورو براءِ نيڪي جي دعوت“به آهي۔ نيڪي جي دعوت ڏيڻ ته اهڙو اهم فريضو آهي جو سڀني نبين سڳورن عَلَيهمُ الصَّلٰوۃُ وَالسَّلَام بلڪ خُود سيدُالْانبياء، محبوب خدا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي به هن مَقصَد جي لاءِ دُنيا ۾ موڪليو ويو، انهن مُقدَّس هستين کوڙ مشڪلاتون ۽ تڪليفون برداشت ڪرڻ باوجود به نيڪي جو حڪم ڏيڻ ۽

برائي کان منع ڪرڻ جي هن عظيم فريضي کي نه ڇڏيو ۔علّامه زُرقاني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه شرح مواهب ۾ فرمائن ٿا ته خاص ڪري حج جي زماني ۾ حضور نبي ڪريم،رؤفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مُبارَڪ عادت هئي ته اهلِ عرب جا پري پري کان آيل قبيلا جيڪي مڪي ۾ گڏ ٿيندا هئا، ته پاڻ سڳورا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سڀني قبيلن ۾ دورو فرمائي ماڻهن کي اسلام جي دعوت ڏيندا هئا، اهڙي طرح عرب ۾ وقت بوقت گھڻا ئي ميلا لڳندا هئا جنهن ۾ پري پري کان قبيلا گڏ ٿيندا هئا، انهن ميلن ۾ به پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تبليغِ اسلام جي لاءِ تشريف وٺي ويندا هئا ۔ (شرح المواھب، ذکر عرض المصطفٰی...الخ، ۲ / ۷۳ ملخصاً)

اَلْحَمْدُلِلّٰه عَزَّوَجَلَّ دعوتِ اسلامي جو مدني ماحول اسان کي به اَنبياءِ ڪرام عَلَيهمُ الصَّلٰوۃُ وَالسَّلَام ۽ سَيدُ الاَنبيا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي هن پياري پياري سُنّت تي عمل ڪرڻ جي لاءِ هر هفتي ’’علاقائي دَورو براءِ نيڪي جي دعوت“ ۾ شرڪت ڪرڻ جي ترغيب ڏياري ٿو۔ اَلْحَمْدُ لِلّٰه عَزَّوَجَلَّ علاقائي دَوره براءِ نيڪي جي دعوت جي برڪت سان مسجد ۾ نمازين جو تعداد وڌي ويندو آهي، ڪڏهن ڪڏهن ته ائين به ٿيو ته علاقائي دَورو براءِ نيڪي جي دعوت جي ذريعي مُعاشري جا بگڙيل ماڻهو دعوت ِاسلامي جي مدني ماحول سان وابسته ٿي صلوٰة و سُنَّت تي عمل ڪرڻ وارا بڻجي ويا، تنهن ڪري اسان کي به وقت ڪڍي هن عظيم مدني ڪم ۾ وڌ کان وڌ حصو وٺڻ گھرجي۔
