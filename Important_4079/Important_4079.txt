دعوتِ اسلامي جي اِشاعتي اِداري مڪتبة المدينه تان جاري ڪيل اپريل 2017 جي ”ماهنامه فيضانِ مدينه“ جي صَفحي نمبر 46 تي آهي: نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي رَضاعِي (کير شريڪ) ڀيڻ حضرت شَيمَاء رَضِىَ الله تَعَالٰى عَنها سان هيئن حُسنِ سُلُوڪ فرمايايو:(1) اُنهن جي لاءِ قِيام فرمايائون (يعني بيهي رهيا) (سبل الھدي و الرشاد، ۵/۳۳۳ ملتقطاً و ملخصاً) (2)پنهنجي مُبارَڪ چادر وڇائي ان تي ويهاريائون ۽ (3) هي اِرشاد فرمايائون: گُھر، توکي عطا ڪيو ويندو، سِفارش ڪر، تنهنجي سِفارش قَبول ڪئي ويندي ۔ (دلائل النبوۃ للبيهقي،۵/۱۹۹ملتقطاً) هن مثالي ڪرم نوازي دَوران پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مُبارَڪ اکين مان لڙڪ وهي رهيا هئا، (4) هي به اِرشاد فرمايائون: جيڪڏهن چاهين ته عزّت ۽ احترام سان اسان وٽ رهه،(5) واپس موٽڻ لڳيون ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ٽي (3) غلام، هڪ لونڊي ۽ هڪ يا ٻه (2) اُٺ به عطا ڪيائون، (6) جڏهن جِعرَانَه ۾ ٻيهر رَضاعي ڀيڻ سان ملاقات ٿي ته رڍون ٻڪريون پڻ ڏنائون۔ (سبل

الھدي و الرشاد،۵/۳۳۳ ملتقطاً و ملخصاً) اسان جي پياري نبي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو پنهنجي رَضاعي ڀيڻ سان حُسنِ سُلُوڪ هر ڀاء کي اهو اِحساس ڏيارڻ جي لاءِ ڪافي آهي ته ڀينرون ڪيتري قَدر وقت پيار ۽ حُسنِ سُلُوڪ جون حقدار آهن۔ حضرت سَيدُنا جابِر بن عبدُالله رَضِىَ اللهُ تَعَالٰى عَنهما صرف پنهنجي نو (9) يا ست (7) ڀينرن جي سار سنڀار لهڻ انهن جي ڦڻي چوٽي ۽ سٺي تَربِيت خاطر بيوه عورت سان نڪاح ڪيائون۔

مٺا مٺا اسلامي ڀائرو! بَيان کي ختم ڪندي صِله رِحمي جي باري ۾ ڪجھ مدني گل ٻُڌڻ جي سعادت حاصِل ڪريون ٿا۔ پهريان ٻه 2 فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحظه ڪريو: (1) هر حُسنِ سُلُوڪ (Politeness) صدقو آهي غني سان هجي يا فقير سان۔ (مجمع الزوائد ، ڪتاب الزڪوٰۃ ، باب ڪل معروف صدقۃ ، ۳/۳۳۱، حڈيث: ۴۷۵۴) (2) جنهن والدين سان حُسنِ سُلوڪ ڪيو ان کي مبارَڪ هجي ته الله پاڪ ان جي عُمر وڌائي ڇڏي۔ (مستدرڪ،ڪتاب البروالصلۃ،باب من بروالديه…الخ،۵/۲۱۳، حديث: ۷۳۳۹) ٭ صِلَه رِحم واجِب آهي ۽ قَطعِ رِحم حرام ۽ جهنَّم ۾ وٺي وڃڻ وارو ڪم آهي۔ (بهار شريعت، ۳/۵۵۸) ٭ رشتيدارن سان سٺو سُلوڪ ان جو نالو ناهي ته هو سُلوڪ ڪري ته توهان به ڪريو، اها شيء ته حقيقت ۾ مُڪافا ة يعني اَدلو بَدلو آهي ته ان توهان وٽ ڪا شيء موڪلي توهان به ان وٽ ڪا شيء موڪلي، اهو توهان وٽ آيو توهان ان وٽ هليا ويا، حقيقت ۾ صله رحمي هي آهي ته اهو ٽوڙي ته توهان ڳنڊيو، اهو توهان کان جدا ٿيڻ چاهي ٿو، تون ان سان مائٽي جي حقن جي رعايت (يعني لحاظ) ڪر. (رَدُّالْمُحتار، ۹/۶۷۸) ٭ صِلَه رِحمي

جون مُختلِف صورَتون آهن، انهن کي هدِيو ۽ تحفو ڏيڻ ۽ جيڪڏهن انهن کي ڪنهن ڳالھ ۾ توهان جي اِمداد درڪار هجي ته ان ڪم ۾ اُن جي مدد ڪرڻ،

اُنهن کي سلام ڪرڻ، اُنهن جي ملاقات لاءِ وڃڻ، انهن وٽ اٿڻ ويهڻ، انهن سان ڳالھ ٻولھ ڪرڻ، اُنهن سان لُطف و مهرباني سان پيش اچڻ. (ڪتاب الدُرَر الحڪام،۱/۳۲۳) ٭ رشتيدارن سان ناغو ڪري ملندو رهي يعني هڪ ڏينهن ملڻ وڃي ٻي ڏينهن نه وڃي ته اهڙي طرح سان مَحَبَّت ۽ اُلفت وڌندي آهي، بلڪه قَرابَتدارن سان جُمعي جُمعي ملندو رهي يا مهيني ۾ هڪ ڀيرو۔ (ڪتاب الدُرَر الحڪام ، ۱/۳۲۳) ٭ حق ۽ جائز ڳالهين ۾ قبيلي ۽ خاندان وارن کي مُتَّحد ٿيڻ گھرجي يعني جيڪڏهن رشتيدار حق تي هجن ته ٻين سان مقابلو ۽ اِظهارِ حق ۾ سڀئي مُتَّحِد ٿي ڪم ڪن۔ (ڪتاب الدُرَر الحڪام،۱/۳۲۳) ٭ رشتيدار مائٽ حاجت پيش ڪري ته ردّ ڪرڻ (موٽائڻ) گناھ آهي، جڏهن پنهنجو ڪو رشتيدار ڪا حاجت پيش ڪري ته ان جي حاجت روائي ڪري، ان کي رَد ڪرڻ رِشتو ٽوڙڻ آهي۔ (ڪتاب الدُرَر الحڪام،۱/۳۲۳) صِلَه رِحمي جي حوالي سان وڌيڪ معلومات حاصِل ڪرڻ جي لاءِ شيخِ طريقت، اَميرِ اَهلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ”هٿو هٿ پڦي سان صلح ڪري ورتي“ جو مطالَعو ڪريو۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

طرح طرح جون هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنتیں ۽ آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي.

ايندڙ هفته وار اجتماع جي بيان جون جهلڪيون

مٺا مٺا اسلامي ڀائرو! ايندڙ هفته وار اجتماع ۾ جيڪو بيان ٿيندو، اهو به نهايت اهم موضوع آهي، قرباني جا ڏينهن ويجهو آهن، انهن جي اهميت، فضيلت، تعريف، فضائل، اسان ايندڙ بيان ۾ ٻڌنداسين، بدقسمتي سان ڪيئي

بدنصيب انسان جانورن تي ظلم ڪندا آهن، ان ظلم جون مثالون بيان ڪيون

وينديون ۽ جيڪي خوش نصيب جانورن تي رحم ڪندا آهن، ان جا ڪهڙا ڪهڙا فائدا آهن اهي به ٻڌايا ويندا، ڪنهن ڪتي کي پاڻي پياريو ته ان جي بخشش ٿي وئي، اهو واقعو به آئنده هفتي بيان ڪيو ويندو ۽ امير اهلسنت جي مرشد ضياءالدين مدني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو عرس مبارڪ به ويجهو آهي، ته سندن سيرت جون ڪجهه جهلڪيون به اسان ٻڌنداسين، ايندڙ جمعي رات به توهان جي حاضري ٿيندي، نه صرف پنهنجي حاضري بلڪ ٻين کي به ترغيب ڏياري گڏ وٺي اچجو.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ 6 درود شريف ۽ 2 دعائون

(1) جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

(2) سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([1])

(3) رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

(4) ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

(5) قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو

هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

(6) دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

(1) هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

(2) ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)
