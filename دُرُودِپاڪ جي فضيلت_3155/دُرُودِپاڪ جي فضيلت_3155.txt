صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

حضرتِ سَيدُنا فُضَاله بن عُبَيد رَضِى الله تَعَالٰي عَنْه کان روايت آهي ته رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم (مسجد ۾) تشريف فرما ها ته هڪ ماڻهو آيو، ان نماز پڙھي ۽ پوءِ ائين دُعا گھري: اَللّٰهمَّ اغْفِرْ لِي وَارْحَمْنِي يعني اي الله عَزَّ وَجَلَّ! مون کي بخشي ڇڏ ۽ مون تي رَحم فرما ءِ ۔ رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: عَجِلْتَ

اَيُّهَاالْمُصَلِّيۡ يعني اي نمازي تو جلدي ڪئي،(پوءِ دُعا جو طريقو بيان ڪندي ارشاد فرمايائون) جڏهن تون نماز پڙهي ڪري ويهين ته پهريان الله تعاليٰ جي اهڙي حمد ڪر جيڪا ان جي لائق آهي، پوءِ مون تي دُرُودِ پاڪ پڙھ ۽ ان کان پوءِ دُعا گھر۔ راوي جو بيان آهي ته ان کان پوءِ هڪ ٻئي شخص نماز پڙھي، پوءِ الله عَزَّوَجَلَّ جي حمد بيان ڪئي ۽ نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي دُرُودِپاڪ پڙھيو ته نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشادفرمايو: اَيُّهَا الْمُصَلِّي اُدْعُ تُجَبْ ، اي نمازي! تون دُعا گھر، قبول ڪئي ويندي ۔ ([1])

مٺا مٺا اسلامي ڀائرو! هن حديث شريف مان معلوم ٿيو ته جيڪڏهن دُعا گھرڻ وارو قَبُوليت جو طالب آهي ته ان کي گھرجي ته دُعا جي شروع ۾ الله عَزَّوَجَلَّ جي حمد ۽ ساراھ به ڪري ۽ نبي ڪريم، رَؤفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي دُرُودِپاڪ به پڙھي بلڪ آخر ۾ به حمد ۽ دُرود پڙھڻ گھرجي جيئن ته دعوتِ اسلامي جي اِشاعتي اِداري مڪتبة المدينه جي شايع ٿيل 318 صفحن تي مشتمل ڪتاب ”فضائلِ دُعا“ صفحي 68 تي اعليٰ حضرت جا والد، حضرتِ علّامه مولانا نقي علي خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه دُعا جا آداب بيان ڪندي اِرشاد فرمائن ٿا: دُعا جي لاءِ اوّل ۽ آخر حمدِ اِلٰهي ادا ڪري ڇوته الله عَزَّوَجَلَّ کان وڌيڪ ڪو پنهنجي حمد کي دوست رکڻ وارو ناهي، ٿوري حمد تي گھڻو راضي ٿئي ٿو ۽ بي شمار عطا فرمائي ٿو،

وڌيڪ فرمائن ٿا ته اَوَّل ۽ آخر نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۽ سندن آل ۽ اَصحاب تي دُرُود موڪلي ڇوته دُرُود الله تعاليٰ جي بارگاھ ۾ مَقبول آهي ۽ الله تعاليٰ جي ڪرم کان بعيد آهي ته دُعا جي اَوَّل ۽ آخر کي قبول فرمائي ۽ دَرميان کي رَد ڪري ڇڏي۔ (فضائلِ دُعا،ص۶۸ملخصاً)

حضرت سَيدُنا عُمَر بن خطاب رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: اِنَّ الدُّعَاءَ مَوْقُوْفٌ بَينَ السَّمَاءِ وَالْاَرْضِ لَا يصْعَدُ مِنْه شَيئٌ حَتّٰي تُصَلِّيَ عَلٰي نَبِيِّکَ صَلَّي اللّٰه عَلَيه وَسَلَّمَ ، يعني دُعا زمين ۽ آسمان جي وچ ۾ روڪي ويندي آهي، جيستائين تون پنهنجي نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي دُرُود نه موڪلين، بلند ناهي ٿيندي۔ ([1]) حضرتِ علي المرتضٰي رَضِى الله تَعَالٰي عَنْه کان روايت آهي: اَلدُّعَاءُ مَحْجُوْبٌ عَنِ اللّٰهِ حَتّٰي يُصَلّٰي عَلٰي مُحَمَّد ٍ وَّاَهلِ بَيتِهٖ يعني دُعا الله تعاليٰ کان حِجاب ۾ آهي، جيستائين محمد صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۽ سندن اَهلِ بيت تي دُرُود نه موڪليو وڃي۔ ([2])

اعليٰ حضرت، امام احمد رضا خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: اي عزيز! دُعا طَائِر (يعني پکي) آهي ۽ دُرُود شَهپَر (يعني ان جو سڀ کان وڏو پر آهي) ، پکي بنا پرن جي ڇا اڏامي سگھي ٿو!

پکيءَ جي ٻانهُن جو سڀ کان وڏو پر، جنهن جي بغير ڪو پکي اڏامي ناهي سگھندو ان کي شَهپر چيو ويندو آهي ۔ يعني دُعا هڪ پکي ۽ دُرُودِ پاڪ ان جي شَهپر جيان آهي تنهنڪري اهڙو پکي جنهن جو شَهپر ئي نه هجي ته اهو ڇا اڏامي سگھندو! ائين ئي اها دُعا جيڪا دُرُودِ پاڪ کان خالي هجي ڪيئن مَقبُول ٿي سگھي ٿي؟ ([3])

مٺا مٺا اسلامي ڀائرو! اسان جي پياري آقا، مڪي مدني مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مُبارڪ زندگي جي ڏينهنَ مبارڪ جو مطالعو ڪيو وڃي ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي معمولاتِ زندگي ۾ مختلف دُعائون، سندن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو معمول رهيون، حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو هي مبارڪ عمل اُمّت کي سيکارڻ جي لاءِ به هو، مثال طور گھر ۾ داخل ٿيڻ جي دُعا، گھران نڪرڻ وقت جي دُعا، سمهڻ کان پهريان جي دُعا، بيدار ٿيڻ کان پوءِ جي دُعا، ماني کائڻ کان پهريان جي دُعا، ماني کائڻ کان پوءِ جي دُعا، لباس پائڻ جي دُعا، تيل لڳائڻ جي دُعا وغيره۔ پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم نه صرف وقتاً فوقتاً دُعا جي اهميت کي بيان فرمايو بلڪه ڏينهن رات دُعا ۾ مصروف رهڻ جي ترغيب به ڏياري، اچو دُعا جي اهميت ۽ فضيلت جي باري ۾ ڪجھ احاديثِ مُبارڪه ٻڌون ٿا:

1. پياري آقا ،مڪي مدني مُصْطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: الدُّعَاءُمُخُّ العِبَادَةِ يعني دُعا عبادت جو مَغَز آهي۔ ([1])

1. حضرت ابو هريره رَضِى الله تَعَالٰي عَنْه فرمائن ٿا ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: الله عَزَّ وَجَلَّ جي ويجھو ڪا شيء دُعا کان وڌيڪ بزرگ ناهي۔ ([1])

2. حضرت عبدُالله ب ن عمر رَضِى الله تَعَالٰي عَنْه فرمائن ٿا ته حضور ِاقدس صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: جنهن شخص جي لاءِ دُعا جو دروازو کوليو ويو ته ان جي لاءِ رحمت جو دروازو کوليو ويو الله عَزَّوَجَلَّ کان ڪيل سوالن مان پسنديده سوال عافيت جو آهي۔ جيڪي مصيبتون نازل ٿي چڪيون ۽ جيڪي نازل نه ٿيون، ان سڀ ۾ دُعا سان نفعو ٿيندو آهي، ته اي الله عَزَّوَجَلَّ جا ٻانهؤ! دُعا ڪرڻ کي (پاڻ تي) لازم ڪيو۔ ([2])

3. حضرت جابر بن عبدُ الله رَضِى الله تَعَالٰي عَنْه فرمائن ٿا ته حضورِ انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ڇا مان توهان کي اها شيء نه ٻڌايان جيڪا توهان کي توهان جي دُشمن کان نجات ڏي ۽ توهان جي رِزق کي ڪشاده ڪري ، رات ڏينهن الله عَزَّوَجَلَّ کان دُعا گھرندا رهو ڇوته دُعا مومن جو هٿيار آهي۔ ([3])

مٺا مٺا اسلامي ڀائرو! اوهان ٻڌو ته احاديثِ مُبارڪه ۾ دُعا گھرڻ جا ڪيترا فضائل بيان فرمايا ويا آهن جو دُعا مصيبتن ۽ تڪليفن کان ڇوٽڪاري جي لاءِ به فائدي مند آهي ۽ رِزق ۾ برڪت ۽ ڪشادَگي جو سبب به آهي، دُشمنن کان نجات جو ذريعو به آهي ۽ مومن جو هٿيار به ڇوته مومن ٻانهو دُعا جي ذريعي وڏي کان وڏي مصيبت ۽ مشڪل کان مشڪل حالتن جو مقابلو ڪري سگھي ٿو، تنهنڪري هر مسلمان کي گھرجي ته خُوشي ۽ غمي هر حالت

۾ الله عَزَّ وَجَلَّ کي ياد ڪري ۽ دُنيا و آخرت جي ڀلائين کي حاصل ڪرڻ ۽ طرح طرح جي مشڪلن کان نجات جي لاءِ بارگاھ الٰهي ۾ ٻاڏائي ڪري دُعا گھري ۽ دُعا ۾ غفلت نه ڪري ۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! ياد رهي ته دُعا افضل ترين عبادت آهي، جنهن جي ذريعي هڪ مسلمان پنهنجي رَبّ عَزَّ وَجَلَّ سان ڄڻ ته هم ڪلام ٿيندو آهي، حديثِ پاڪ ۾ آهي: اَفْضَلُ الْعِبَادَةِ الدُّعَاءُ يعني دُعا افضل ترين عبادت آهي۔ ([1]) ان ڪري جهڙي طرح تمام عبادتن جا ڪجھ نه ڪجھ شرائط ۽ آداب هوندا آهن، جن تي انهن عبادتن جي قبوليت ۽ دُرُستگي جو دار و مدار هوندو آهي، اهڙي طرح دُعا جا به ڪجھ آداب آهن، دُعائن جي قبوليت ۾ انهن آداب کي هڪ خاص اهميت حاصل آهي ، جيڪڏهن دُعا ۾ انهن آداب جو خيال رکيو وڃي ته قبوليتِ دُعا جا اِمڪان وڌي ويندا آهن، تنهنڪري جيڪڏهن اسان چاهيون ٿا ته اسان جون دُعائون بارگاھ اِلٰهي ۾ قبول ٿين ته اسان کي کپي ته دُعا جي آداب جو خاص خيال رکندا ڪيون۔

اَلحمدُلِلهِ عَزَّوَجَلَّ دُعائن جي قبوليت جي آدابِ جي معلومات حاصل ڪرڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل 326 صفحن تي مشتمل ڪتاب ”فضائلِ دُعا“ جو مطالعو ڪرڻ بي حد مُفيد آهي، اهو ڪتاب درحقيقت اعليٰ حضرت رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جي والد محترم حضرت علّامه مولانا نقي علي خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه

جي تصنيف ” اَحْسَنُ الْوِعَاءِ لِاٰدَابِ الدُّعَاءِ “ جي تسهيل آهي، هن ڪتاب ۾ دُعا جا فضائل، دُعا جا آداب، قبوليتِ دُعا جا سبب، قبوليتِ دُعا جا وقت، قبوليتِ دُعا وارا هنڌ، دُعائن جي قبول ٿيڻ ۾ اسمِ اعظم جي اهميت، قبوليتِ دُعا ۾ رُڪاوٽ بڻجڻ وارا سبب، دعا گھرڻ جا فائدا ۽ دُعا بابت اهم سوالن جوابن کي بيان ڪيو ويو آهي۔ شايد اهو چوڻ غلط نه ٿيندو ته اهو ڪتاب هر گھر جي ضرورت آهي، تنهنڪري اڄ ئي هن ڪتاب کي مڪتبة المدينه جي بستي تان هديةً طلب فرمايو، پاڻ به ان جو مُطالعو ڪيو ۽ ٻين کي به ان جي ترغيب ڏياريو، ۽ دعوتِ اسلامي جي ويب سائٽ www.dawateislami.net تان هن ڪتاب کي رِيڊ (يعني پڙھي) به سگھجي ٿو، ڊاؤن لوڊ (Download) ۽ پرنٽ آؤٽ (Print Out) به ڪري سگھجي ٿو۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

اچو دُعا جا ڪجھ آداب ۽ ان ضمن ۾ قبوليتِ دُعا جا واقعا ٻڌون ٿا۔

پنهجي رَبّ کي محبوب نالن سان پڪاريو

مٺا مٺا اسلامي ڀائرو! دُعا جي آداب مان اهو به آهي ته دُعا ڪرڻ وارو شروع ۾ الله عَزَّ وَجَلَّ کي ان جي محبوب نالن سان سڏي۔ رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمائن ٿا: الله تعاليٰ پنهنجي اسمِ پاڪ” ا َرْحَمُ الرَّاحِمِين “ تي هڪ ملائڪ مقرر فرمايو آهي ته جيڪو شخص ان کي ٽي (3) ڀيرا چوي ٿو، ملائڪ نِدا ڪندو آهي(يعني آواز ڏيندو آهي): گُھر اَرْحَمُ الرَّاحِمِين تو ڏانهن متوجهه ٿيو۔ ([1])

اهڙي طرح پنج (5) ڀيرا ” يا رَبَّنَا “ چوڻ به دُعا جي قبوليت ۾ گھڻو اَثر

رکي ٿو۔ قرآنِ ڪريم ۾ ان لفظِ مبارڪ کي پنج (5) ڀيرا ذڪر ڪري ان کان پوءِ ارشاد فرمايو:

فَاسۡتَجَابَ لَہُمۡ رَبُّہُمۡ (پ۴، اٰل عمران: ۱۹۵)

ترجمو ڪنز الايمان: پوءِ اُنهن جي پالڻهار سندن دُعا قُبول ڪئِي

امام جعفر صادق رَضِى الله تَعَالٰي عَنْه کان منقول آهي:جيڪو شخص بي وسي وقت پنج (5) ڀيرا ” يا رَبَّنَا “ چوي ته جنهن جو خوف رکي ٿو الله تعاليٰ ان کي ان شيء کان امان بخشيندو ۽ جيڪا شيء چاهي ٿو،عطا فرمائيندو۔ ([1])

دُعا کان پهريان نيڪ عمل

مٺا مٺا اسلامي ڀائرو! دُعا جي آداب مان هڪ ادب اهو به آهي ته دُعا کان پهريان ڪو نيڪ عمل ڪيو وڃي جيئن حضرت سَيدُنا امام جَزري رَحْمَة اللّٰه ِتَعَالٰي عَلَيه تحرير فرمائن ٿا: اٰ دَابُ الدُّعَاءِ مِنْھَا تَقْدِيمُ عَمَلٍ صَالِحٍ وَذِکرُهُ عِنْدَالشِّدَّةِ يعني هر مشڪل ۾ الله عَزَّ وَجَلَّ کي ياد ڪرڻ ۽ دُعا کان پهريان نيڪ عمل ڪرڻ دُعا جي آداب مان آهي۔ ([2]) جڏهن ڪو ٻانهو اِخلاص سان ڪيل نيڪ عمل کي الله عَزَّ وَجَلَّ جي بارگاھ ۾ وسيلو بڻائي دُعا گھرندو آهي ته الله عَزَّ وَجَلَّ ان جي صدقي ٻانهي جي دُعا قبول ڪندو آهي۔اچو هن باري ۾ هڪ ايمان افروز واقعو ٻڌون ٿا:

دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل 656 صفحن تي مشتمل ڪتاب ” فيضانِ رياضُ الصالحين “ جي صفحي 139 تي آهي:

حضرتِ سَيدُنا عبدُ الله بن عمر رَضِى الله تَعَالٰي عَنْه فرمائن ٿا ته رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: گذريل زماني ۾ ٽي (3) شخص ڪيڏانهن وڃي رهيا هئا ته برسات ٿيڻ لڳي ته اهي پناھ وٺڻ جي لاءِ هڪ غار ۾ داخل ٿيا، اوچتو جبل تان هڪ چٽان (ڇپ) ڪري جنهن غار جو منهن بند ڪري ڇڏيو۔ اهو ڏسي انهن چيو: هن مصيبت مان نجات جو هڪ ئي طريقو آهي ته اسان پنهنجن پنهنجن نيڪ عملن جو وسيلو الله عَزَّوَجَلَّ جي بارگاھ ۾ پيش ڪري دُعا ڪيون۔ پوءِ، انهن مان هڪ چيو: ياالله عَزَّوَجَلَّ! منهنجا ماءُ پيءُ نهايت پوڙها ٿي ويا هئا، مان انهن کان اڳ ۾ نه ته پنهنجي اهل و عيال کي پيئڻ جي لاءِ کير ڏيندو هئس نه ئي پنهنجي نوڪرن کي، هڪ ڏينهن مان وڻن جي تلاش ۾ گھڻو پري هليو ويس، جڏهن واپس موٽيس ته منهنجا والدين سمهي رهيا هئا، مان کير کڻي انهن وٽ آيس ته انهن کي ستل ڏٺم، مون نه انهن کي جاڳائڻ مناسب سمجھيو ۽ نه ئي انهن کان پهريان اهل وعيال مان ڪنهن کي ڏيڻ پسند ڪيو، بلڪ مان کير جو پيالو کڻي پنهنجي والدين وٽ بيٺل رهيس، جڏهن صبح ٿيو ته مون انهن کي کير پيش ڪيو۔ ياالله عَزَّوَجَلَّ! جيڪڏهن مون اهو عمل صرف تنهنجي رضا جي لاءِ ڪيو هو ته اسان کي اِن مصيبت مان نجات عطا فرما! ان جي دُعا سان ڇِپ (پٿرجي) ڪجھ سِرڪي پئي، پر اڃا به ايتري جڳھ نه ٺهي جو اُهي نڪري سگھن، پوءِ ٻئي چيو: ياالله عَزَّوَجَلَّ! مون کي منهنجي چاچي جي ڌيءُ ماڻهن ۾ سڀ کان وڌيڪ محبوب هئي، مون ان سان بُرائي جو اِرادو ڪيو ته ان انڪار ڪري ڇڏيو، پوءِ اها ڏڪار(تنگدستي) ۾ مبتلا ٿي ته مون وٽ آئي مون ان شرط تي ان کي 100 دينار ڏنا ته هوءَ منهنجي خواهش پوري ڪري، هوءَ مجبور هئي تيار ٿي وئي، جڏهن مان ان سان اڪيلائي ۾ ويس ۽ ان تي قابو ٿيس ته ان چيو: الله عَزَّ وَجَلَّ کان ڊڄ ۽ ناحق مُهر کي نه ٽوڙ (يعني ان بُري ڪم

کان پاسو ڪر) اهو ٻڌي مون ان کي ڇڏي ڏنو ۽ مون بُرائي کان پاسو ڪيو، حالانڪه مون کي ان سان گهڻي محبت هئي، پوءِ مون ان کان اهي دِينار به واپس نه ورتا، ياالله عَزَّوَجَلَّ! جيڪڏهن مون اهو عمل صرف تنهنجي رضا جي لاءِ ڪيو هو ته اسان کي اِن مصيبت مان نجات عطا فرما! ڇِپ ڪجھ وڌيڪ سرڪي پر اڃا به اهي ٻاهر نه نڪري سگھيا ها۔ ٽئي چيو: ياالله عَزَّوَجَلَّ! مون ڪجھ مزدورن کان ڪم ڪرايو ۽ سڀني جي مزدوري ڏئي ڇڏي، پر ان مان هڪ مزدور اُجرت ڇڏي هليو ويو، مون اها رقم واپار ۾ لڳائي ڇڏي ته ان جي رقم وڌندي رهي ڪجھ عرصي کان پوءِ اهو آيو۽ چيائين: اي الله عَزَّوَجَلَّ جا ٻانها! منهنجي اُجرت مون کي ڏئي ڇڏ، مون چيو: اِهي اُٺ، ڳئون، ٻڪري ۽ غلام جيڪي ڪجھ توهان ڏسي رهيا آهيو، اهي سڀ توهان جا آهن۔ ان چيو: اي الله عَزَّوَجَلَّ جا ٻانها! مون سان مذاق نه ڪر۔ مون چيو: مان مذاق نه پيو ڪيان، اِهي سڀ ڪجھ توهان جا ئي آهن۔ پوءِ، اهو سڀ مال کڻي هليو ويو ۽ ڪجھ به نه ڇڏيائين۔ ياالله عَزَّ وَجَلَّ! جيڪڏهن منهنجو اهو عمل تنهنجي رضا جي لاءِ هو ته اسان کي ان مصيبت مان نجات عطا فرما! پوءِ ڇپ وڌيڪ سرڪي ۽ اهي ٽئي ٻاهر نڪري پنهنجي منزل ڏانهن روانا ٿيا۔ ([1])

گھرندي؟ چيائين، عرصو گذري ويو پر مُراد پوري ئي نه پئي ٿئي۔ فرمايائون: مان مزارِ پاڪ تي حاضِري ڏئي ٿوري دير ۾ واپس موٽان ٿو جيڪڏهن اکيون روشن ٿيون ته ٺيڪ، نه ته قتل ڪرائي ڇڏيندس۔ اهو چئي فقير تي پهرو لڳرائي بادشاھ حاضِري جي لاءِ اندر ويو۔ هوڏانهن فقير تي گِريو طاري هو ۽ روئي روئي فرياد ڪري رهيو هو: يا خواجه رَحْمَة اللّٰه ِتَعَالٰي عَلَيه! پهريان ته صرف اکين جو مسئلو هو هاڻي ته جان جو خطرو آهي، جيڪڏهن اوهان ڪرم نه فرمايو ته مارجي ويندس۔ جڏهن بادشاھ حاضري ڏئي واپس موٽيو ته ان جون اکيون روشن ٿي چڪيون هيون۔ بادشاھ مسڪرائي فرمايو ته توهان ايستائين بي دِلي ۽ بي توجُّهي سان دعا گھري رهيا هيو ۽ هاڻي جان جي خوف سبب توهان دل جي تڙپ سان سوال ڪيو ته توهان جي مُراد پوري ٿي وئي۔

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته الله عَزَّوَجَلَّ جي ولين جي مزارات تي خُلوصِ دل سان انهن جي وسيلي سان ڪيل دُعا به قبول ٿيندي آهي، تنهنڪري اسان کي به گھرجي ته پنهنجي حاجتن جي لاءِ وقتاً فوقتاً اولياء ڪرام رَحْمَة اللّٰه تَعَالٰي عَلَيهمْ اَجْمَعِين جي مزارات تي حاضري ڏيندا رهون ۽ انهن جي وسيلي سان دُعائون گھرون۔ والدِ اعليٰ حضرت، حضرتِ علّامه مولانا نقي علي خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه دُعا جا آداب بيان ڪندي فرمائن ٿا ته الله تعاليٰ جي نالن ۽ صِفتن ۽ ان جي ڪتابن خصوصاً قرآن ۽ ملائڪ ۽ انبياء ڪرام عَلَيهمُ الصَّلٰوة وَالسَّلَام بالخصوص حضور عَلَيه الصَّلٰوة وَالسَّلَام ۽ الله عَزَّ وَجَلَّ جي ولين خصوصاً حضور غوثِ اعظم رَضِى الله تَعَالٰي عَنْه کي پنهنجي حاجتن جي پورو ٿيڻ جي لاءِ وسيلو بڻائي جو محبوبانِ خُدا جي وسيلي سان

دُعا قبول ٿيندي آهي۔ (فضائلِ دُعا،ص۷۱بتغیر قلیل) قرآنِ پاڪ ۾ الله عَزَّوَجَلَّ پنهنجي بارگاھ ۾ وسيلو اختيار ڪرڻ جو حڪم ڏيندي ارشاد فرمائي ٿو؛

وَ ابۡتَغُوۡۤا اِلَیۡہِ الۡوَسِیۡلَۃَ: ” ترجمو ڪنز الايمان: ۽ ان جي طرف وسيلو ڳوليو “ ([1]) تنهنڪري دُعائن جي قبوليت جي لاءِ الله عَزَّوَجَلَّ جي پيارن کي وسيلو بنائڻ گھرجي ۽ ٿي سگھي ته نيڪ ٻانهن کان پنهنجي لاءِ دُعا به ڪرائڻ گھرجي، ڇوته الله عَزَّوَجَلَّ پنهنجي نيڪ ٻانهن جي دُعاءِ خير ۽ دُعاءِ نقصان ضرور قبول فرمائيندو آهي۔جيئن ته

حضرت الحاج مولانا عبدُ المصطفي اعظمي رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: الله تعاليٰ جي محبوب ٻانهن عالمن، ولين، ۽ سمورن نيڪن جي ڪنهن جي خلاف دُعائون گھرڻ نهايت ئي خطرناڪ ۽ هلاڪت ڪندڙ بلائون آهن۔ انهن بزرگن جي دُعاءِ ضَرَر ۽ ڦٽڪار اها تلوار آهي جنهن جي ڪا ڍال ناهي ۽ اهو تباهي ۽ بربادي جو اهو زهر آلود تِير آهي جنهن جو نشانو ڪڏهن خطا ناهي ٿيندو، تنهنڪري هر مسلمان تي لازم آهي ته سڄي زندگي هر هر قدم تي اهو ڌيان رکي ته ڪڏهن به الله تعالي ٰ جي نيڪ ٻانهن جي شان ۾ ذرّي برابر به بي ادبي نه ٿئي ۽ بزرگانِ دِين مان ڪنهن جي به دُعاءِ ضَرَر(بد دعا) نه وٺو بلڪ هميشه ان ڪوشش ۾ لڳل رهي ته خُدا عَزَّ وَجَلَّ جي نيڪ ٻانهن جون دُعائون ملنديون رهن ڇوته نيڪ ٻانهن جون نقصان جي لاءِ دُعائون بربادي جو خوفناڪ سگنل (Signal اِشارو) ۽ انهن جون نيڪ دُعائون آبادي جو مٺو ميوو آهي۔ ([2])

اچو ان باري ۾ هڪ ايمان افروز حڪايت ٻڌو ۽ پنهنجي دل ۾ اولياءِ ڪرام رَحِمَهمُ اللّٰه ُ السَّلَام جي مزيد عظمت پيدا ڪيو:

اکيون بي نور ٿي ويون

حضرت سَيدُنا عثمان بن عطاء رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا ته حضرت سيدنا ابومسلم خولاني رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جڏهن مسجد کان پنهنجي گھر ڏانهن ويندا هئا ته دروازي تي پهچي الله اَڪبَر جي صدا بلند ڪندا ها، جواب ۾ سندن زوجه ء محترمه رَحْمَة اللّٰه ِتَعَالٰي عَلَيها به الله اَڪبَر چوندي هئي۔ پوءِ جڏهن پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه اڱڻ ۾ ويندا ها ته الله اَڪبَر چوندا ها، جواب ۾ سندن گھرواريءَ به الله اَڪبَر چوندي هئي۔ جڏهن ڪمري ۾ داخل ٿيندا ها ته وري الله اَڪبَر چوندا ها ۽ جواب ۾ سندن زوجه محترمه به الله اَڪبَر چوندي هئي، اهو پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جي هر ڏينهن جو معمول هو هڪ رات جڏهن پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه گھر موٽي آيا ۽ پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه دروازي وٽ پهتا ۽ معمول مطابق الله اَڪبَر چيو پر جواب نه مليو، پوءِ جڏهن اڱڻ ۾ پهچي الله اَڪبَر چيو تڏهن به جواب نه مليو، جڏهن ڪمري ۾ پهتا ۽ الله اَڪبَر چيائون تڏهن به سندن زوجه جواب ۾ الله اَڪبَر نه چيو، پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جي زوجه محترمه سندن کي ماني ڏني ۽ چُپ چاپ زمين تي ويھي رهي، ائين پئي لڳو جيئن هوءَ اوهان سان ناراض آهي، سندن گھر ۾ روشني جي لاءِ ڏيئو تائين نه هو، پر پاڻ پوءِ به صبر ۽ شڪر ڪندڙ هئا۔ جڏهن سندن پنهنجي زوجه کي ناراض ڏٺائون ته ان کان معلوم ڪيائون: اي الله عَزَّ وَجَلَّ جي ٻانهي! تون ڇو پريشان آهين؟ اهو ٻڌي هوءَ چوڻ لڳي: توهان جو امير المؤمنين حضرت سَيدُنا امير ِمُعاويه رَضِى الله تَعَالٰي عَنْه وٽ وڏو مرتبو آهي، اهي توهان جي نهايت تعظيم ڪندا آهن، جيڪڏهن اوهان اُنهن کان هڪ خادم گھرو ته اُهي ضرور اوهان کي عنايت فرمائيندا، اسان وٽ هڪ به خادم ناهي جيڪو اسان جي خدمت ڪري سگھي، خادم اچي ته اسان کي آساني ٿيندي، اِهو ٻڌي اوهان دُعا جي لاءِ هٿ کنيا ۽ اهڙي طرح بارگاھِ خُداوندي عَزَّوَجَلَّ ۾ عرض گذار

ٿيا: اي منهنجا پروردگار عَزَّوَجَلَّ! ان کي انڌو ڪري ڇڏ جنهن منهنجي گھر وارن جو ذهن خراب ڪيو آهي ۽ اسان ۾ فتنو وجھڻ جي ڪوشش ڪئي آهي، پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جي دُعا جو اثر فوراً ظاهر ٿيو ۽ پاڙيسرين مان هڪ عورت جون اکيون اوچتو بي نور ٿي ويون جيڪا پنهنجي گھر ۾ هئي ۽ اصل ٿيو هينئن جو انهيء عورت اچي سندن جي زوجه محترمه کي چيو هو ته جيڪڏهن تون پنهنجي مڙس کي چوين ته اهي اميرالمؤمنين حضرت سَيدُنا امير معاويه رَضِى الله تَعَالٰي عَنْه کان غلام حاصل ڪري سگھن ٿا، ۽ جيڪڏهن توهان کي غلام ملي ويو ته اهڙي طرح توهان جي زندگي پُرسُڪون ٿي ويندي۔ ڪجھ نظر نه اچڻ جي ڪري ان گھر وارن کي چيو: توهان ڏيئا ڇو وِسايا آهن؟ گھر وارن چيو: ڏيئا ته ٻري رهيا آهن، شايد! تنهنجون اکيون بي ڪار ٿي ويون آهن، هاڻي اها عورت نهايت پريشان ٿي ۽ جڏهن ان کي خبر پئي ته هي حضرت سَيدُنا ابو مُسلم خولاني رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جي دُعا جو اثر آهي ته هوءَ پنهنجي حرڪت تي نهايت شرمسار ٿي ۽ پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه کان معافي گھري ۽ زار وقطار روئڻ لڳي، ۽ عرض ڪرڻ لڳي: مون کي الله عَزَّ وَجَلَّ جي رضا جي خاطر معاف فرمائي ڇڏيو ۽ الله عَزَّ وَجَلَّ جي بارگاھ ۾ دُعا فرمايو ته منهنجي نظر موٽي اچي، پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه کي ان تي ترس اچڻ لڳو ۽ پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه الله عَزَّوَجَلَّ جي بارگاھ ۾ دُعا جي لاءِ هٿ کنيا ۽ ان جي نظر جي لاءِ دُعا ڪئي پاڻ رَحْمَة اللّٰه ِتَعَالٰي عَلَيه اڃا دُعا کان فارغ به نه ٿيا جو ان عورت جون اکيون روشن ٿي ويون ۽ اها بلڪل ٺيڪ ٿي وئي ۔ ([1])

آهي۔ ([1]) ٿورو سوچيو ته جتي ڇما ڇم رحمتون وسي رهيون هجن اتي دُعائون ڇو قبول نه ٿينديون۔ اعليٰ حضرت رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: جماعت ۾ بَرَڪت آهي ۽ مسلمانن جي ميڙ ۾ دُعا گھرڻ قبوليت جي گھڻو ويجھو آهي عُلَماء فرمائن ٿا: جتي چاليھ (40) نيڪ مُسلمان گڏ ٿيندا آهن انهن مان هڪ ولي الله ضَرور هوندو آهي۔ ([2]) تنهنڪري هفتيوار سُنّتن ڀري اجتماع ۾ پابندي سان شرڪت کي پنهنجو معمول بڻائي وٺو، سُنّتن ڀرين اجتماعن ۾ گھريل دُعائن جا ثمرات ڪهڙي طرح ظاهر ٿيندا آهن۔ ٻڌو ۽ خوش ٿيو،

بابُ المدينه ڪراچي جي هڪ اسلامي ڀاءُ جي بيان جو خُلاصو آهي، جواني ۽ چڱي صحّت مون کي مغرور بڻايو هو، نِت نوان فينسي لباس سبرائڻ، رات دير تائين آواره گردي ۾ وقت وڃائڻ، جُوا ۾ پيسا برباد ڪرڻ وغيره منهنجو معمول هو۔ والِدَين سمجھائي سمجھائي ٿڪجي پيا ايستائين جو ڪڏهن ته منهنجي اصلاح جي لاءِ دُعا ڪندي امڙ سائيڻ جا ڳوڙها نڪري ايندا هئا۔ دعوتِ اسلامي سان وابسته هڪ اسلامي ڀاءُ مون کي هفتيوار سنّتن ڀري اجتماع جي دعوت پيش ڪندا ها پر آئون ٽاريندو رهيس، آخرڪار اسان ٻئي اجتماع گاھ ۾ پهچي وياسين، خُدا جو قسم! مون زندَگي ۾ ڪڏهن اهڙي رُوحانيت نه ڏٺي نه ٻڌي هئي، جڏهن رِقّت انگيز دُعا شُروع ٿي ته اجتِماع ۾ شريڪ ٿيندڙن جي سُڏڪڻ جو آواز بُلند ٿيڻ لڳو ايستائين جو مون جھڙو پٿر دل ماڻهو به سڏڪا ڀري روئڻ لڳو، مون پنهنجي گُناهن کان توبه ڪئي ۽ دعوتِ اسلامي جي مَدَني ماحول جو ئي بڻجي ويس ۔

شعر

مٺا مٺا اسلامي ڀائرو! اوهان ڏٺو ته نيڪ ماڻهن جي دُعائن ۾ شريڪ ٿي دُعا گھرڻ جون ڪهڙيون ڪهڙيون برڪتون آهن، واقعي اسان کي خبر ناهي ته ڪهڙي نيڪ ٻانهي جي دُعا بارگاھِ الٰهي ۾ مقبول ٿي وڃي ۽ ان جي صدقي دُعا ۾ شريڪ سڀني ماڻهن جا ٻيڙا پار ٿي وڃن۔ اچو هن باري ۾ هڪ واقعو ٻڌون ٿا، جيئن

حضرتِ علّامه جلالُ الدّين سُيوطِي شافِعي رَحْمَة اللّٰه ِتَعَالٰي عَلَيه نَقل ڪندي فرمائن ٿا: حضرتِ سيدُنا يزيد بن هارون رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: مون حضرتِ سيدُنا ابواسحق محمد بن يزيد واسِطي رَحْمَة اللّٰه ِتَعَالٰي عَلَيه کي خواب ۾ ڏٺو ته پڇا ڪئي: الله عَزَّوَجَلَّ اوهان سان ڪهڙو مُعامَلو ڪيو؟ ته اُنهن فرمايو: منهنجي مغفِرت ڪئي وئي۔ مون پڇا ڪئي: مغفِرت جو سبب ڇا ٿيو؟ فرمايائون: هڪ ڀيري حضرتِ سيدُنا ابو عَمرو بَصري رَحْمَة اللّٰه ِتَعَالٰي عَلَيه جُمُعي جي ڏينهن اسان وٽ تشريف فرما ٿيا ۽ دُعا ڪئي ته اسان آمين چئي، بس ان ڪري اسان جي مغفِرت ٿي وئي۔ ([1])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! دُعا جي آداب مان هي به آهي ته جڏهن اسان پنهنجي لاءِ دُعا گھرون ته سڀني اهلِ اسلام کي ان دُعا ۾ شريڪ ڪيون۔ اعليٰ حضرت امام احمد رضا خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا ته جيڪڏهن دُعا گھرڻ وارو پاڻ عطا لائق ناهي ته ڪنهن ٻانهي جو طُفَيلي ٿي مراد تائين پهچي ويندو۔

حضرت سَيدُنا ابو الشيخ اصبهاني رَحْمَة اللّٰه ِتَعَالٰي عَلَيه حضرت سَيدُنا ثابت بُناني رَحْمَة اللّٰه ِتَعَالٰي عَلَيه کان روايت ڪئي: اسان سان ذڪر ڪيو ويو جيڪو شخص مسلمان مَردن ۽ عورتن جي لاءِ دُعاءِ خير ڪندو آهي قيامت ۾ جڏهن ان جي مجلسن تان گذرندو ته هڪ چوڻ وارو چوندو: هي اهو آهي جيڪو اوهان جي لاءِ دُنيا ۾ دُعاءِ خير ڪندو هو پوءِ اهي ان جي شفاعت ڪندا ۽ رب جي بارگاھ ۾ عرض ڪري ان کي جنت ۾ وٺي ويندا۔

قرآن ِڪريم ۾ مسلمانن جي لاءِ دُعا ڪرڻ جي باري ۾ الله عَزَّ وَجَلَّ جو ارشاد آهي:

وَ اسۡتَغۡفِرۡ لِذَنۡۢبِکَ وَ لِلۡمُؤۡمِنِیۡنَ وَ الۡمُؤۡمِنٰتِ ؕ (پ۲۶، محمد: ۱۹)

ترجمو ڪنز الايمان: ۽ اي محبوب پنهنجي خاص ۽ عام مسلمان مردن ۽ عورتن جي گناهن جي معافي گهرو.

حديث ۾ آهي:نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هڪ شخص کي اَللّٰھُمَّ اغْفِرْلِيْ (يعني اي الله! منهنجي مغفرت فرما) چوندي ٻڌو، فرمايائون: جيڪڏهن سڀني مسلمانن کي دُعا ۾ شامل ڪرين ها ته تنهنجي دُعا مقبول ٿئي ها۔ ([1])

حضرتِ سَيدُنا عُباده بن صامِت رَضِى الله تَعَالٰي عَنْه فرمائن ٿا ته مون رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي هي فرمائيندي ٻڌو: مَنِ اسْتَغْفَرَ لِلْمُوْمِنِيْنَ وَالْمُوْمِنَاتِ كَتَبَ اللهُ لَهُ بِكُلِّ مُوْمِنٍ وَ مُوْمِنَةٍ حَسَنَةً يعني جيڪو شخص سڀني مسلمان مردن ۽ عورتن جي لاءِ استغفار ڪري، الله تعاليٰ ان جي لاءِ هر مسلمان مرد ۽ مسلمان عورت جي بدلي هڪ نيڪي لکندو۔ (مجمع الزوائد، کتاب التوبۃ، باب الاستغفار للمؤمنین والمؤمنات، ۱۰/۳۵۲، حدیث: ۱۷۵۹۸

تنهنڪري اسان کي گھرجي ته پنهنجي دُعائن ۾ پنهنجي مسلمان ڀائرن کي به ياد رکندا ڪيون، ۽ والدين بلڪ پنهنجن استادن جي لاءِ به ضرور دُعا ڪيون، ياد رهي ته دِيني استاد روحاني پيءُ جي حيثيت رکي ٿو۔ان جي لاءِ دُعا ڪرڻ انسان جي حق ۾ نعمتن جو سبب آهي، حديثِ پاڪ ۾ آهي: اِذَاتَرَكَ الْعَبْدُ الدُّعَاءَ لِلْوَالِدَيْنِ فَاِنَّهُ يَنْقَطِعُ عَنْهُ الرِّزْقُ يعني جڏهن ٻانهو والدين جي لاءِ دُعا ڪرڻ ڇڏي ڏيندو آهي ته ان وٽان رِزق روڪيو ويندو آهي۔ ([1]) تنهنڪري اُستادن ۽ ماءُ پيءُ جي لاءِ به لازمي دُعا ڪرڻ گھرجي۔ جڏهن به دُعا گھرون دُعا ۾ نهايت عاجزي ۽ گِريه وزاري ڪيون، روئيندي ۽ ٻاڏائيندي الله عَزَّوَجَلَّ جي بارگاھ ۾ پنهنجي بي وسي ۽ لاچاري جو اظهار ڪندي پنهنجي حاجتن جي لاءِ التجائون ڪيون۔

حضرت علّامه، مولانا نقي علي خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه هن دُعا جي آدب جي تحت فرمائن ٿا: جيتري قدر هيڏانهن عاجزي گھڻي هوندي، اتان کان لطف وڪرم وڌيڪ ٿيندو۔ وڌيڪ فرمائن ٿا ته دُعا ۾ لڙڪ وهائڻ جي ڪوشش ڪري جيتوڻيڪ هڪ ئي ڦڙو هجي ڇوته قبوليت جي نشاني آهي، روئڻ نه اچي ته روئڻ جھڙو چهرو ٺاهي ڇوته نيڪن جهڙي صورت به نيڪ آهي۔ اُمُّ المؤمنين حضرتِ سَيدَتُنا عائشه صِدّيقه رَضِياللّٰه تَعَالٰي عَنْها فرمائن ٿيون ته رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: الله تعاليٰ دُعا ۾ ٻاڏائڻ وارن کي پسند ڪندو آهي۔ ([2])

رونے والی آنکھیں مانگو ، رونا سب کا کام نہیں

ذِکرِ محبت عام ہے لیکن، سوز محبت عام نہیں

مٺا مٺا اسلامي ڀائرو! دُعا جي قبوليت جي لاءِ الله عَزَّوَجَلَّ جي رحمت تي نظر رکندي دُعا گھرڻ گھرجي، ۽ دُعا جي قَبولِيت ۾ جلدي ڪرڻ کان بچڻ گھرجي، ڪجھ نادان مَعَاذالله عَزَّوَجَلَّ اهڙيون ڳالهيون ڪندا آهن ته اسان ايتري عَرصي کان دُعائون گھري رهيا آهيون، بُزُرگن کان به دُعائون ڪرايون، ڪو پير فقير ناهي ڇڏيو، هي وظيفا پڙهندا آهيون اهي ورد به پڙهيا آهن، فُلاڻي فُلاڻي مَزار تي به وياسين پر الله عَزَّ وَجَلَّ اسان جي حاجت پوري ڪري ئي نٿو، حالانڪ ڪڏهن ڪڏهن قَبولِيتِ دُعا جي تاخِير ۾ گهڻيون مصلحتون به هونديون آهن جيڪي اسان جي سمجھ ۾ نٿيون اچن، تنهنڪري دُعا ۾ جلدي نه ڪرڻ گھرجي، دُعا جي قَبولِيت ۾ جلدي ڪرڻ ۽ قبوليت ۾ تاخير تي اُڪتائڻ وارن کي سمجھائيندي اعليٰ حضرت امام احمد رضا خان رَحْمَة اللّٰه ِتَعَالٰي عَلَيه پنهنجي مخصُوص انداز ۾ نصيحت جا جيڪي مدني گل عطا فرمايا، انهن جو خُلاصو ٻڌو ۽ قبوليتِ دُعا ۾ جلدي ڪرڻ کان پاڻ کي بچايو:

دُنْياوي آفيسرن کان ڪم ڪڍڻ جي خواهشمندن کي ڏٺو ويندو آهي ته ٽن ٽن سالن تائين اميد ۽ انتظار ۾ گذاريندا آهن، صبح ۽ شام سندن دروازن تي ڀَڄندا آهن، ۽ اهي آفيسر کين لِفٽ ناهن ڪرائيندا، جواب ناهن ڏيندا، دڙڪا ڏيندا آهن خار کائيندا ۽ چهري ۾ گُهنج وجهندا آهن، انهن آفيسرن وٽ ڌڪا کائڻ ۾ کين سال گذرن پر اميد اڃا به پهرين ڏينهن جيان مضبوط آهي، دنياوي افسرن وٽ ڌڪا کائڻ وارا نه اميد ٽوڙين نه انهن جو پيڇو ڇڏين پر افسوس! الله عَزَّ وَجَلَّ جي دروازي تي پهريان ته اچي ئي ڪير ٿو! ۽ اچي به ته بي دليو ٿي، جلدي ڪرڻ، سڀاڻي ٿيڻ وارو ڪم اڄ ٿي وڃي۔ هڪ هفتو ڪجھ پڙهندي ته ڇا گذريو، هاڻي شڪايت ٿيڻ لڳي، سائين! پڙهيو ته هو پر ڪجھ اثر نه ٿيو! اهي بيوقوف پنهنجي لاءِ قبوليّت جو دروازو پاڻ بند ڪري ڇڏيندا آهن۔

مُحَمَّدٌرَّسُوۡلُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمائن ٿا: يُسْتَجابُ لِاَحَدِکمْ مَالَمْ يعَجِّلْ يقُوْلُ دَعَوْتُ فَلَمْ يسْتَجِبْ لِي توهان جي دعا قبول ٿيندي آهي جيستائين تڪڙ نه ڪريو هي نه چئو ته مون دعا گھري هئي قبول نه ٿي۔ ([1]) وڌيڪ فرمائن ٿا: ڪجھ ته ان دير سبب اهڙا بي قابو ٿي ويندا آهن جو وظيفن ۽ دعائن جي اثر کان بي اعتقاد، بلڪه نَعُوْذُ بِالله عَزَّ وَجَلَّ! الله عَزَّوَجَلَّ جي ڪرم جي واعدي کان بي اعتماد ٿي ويندا آهن. اهڙن کي چيو وڃي ته اي بي حيا! بي شرمو! ٿورو پنهنجي گريبان ۾ جهاتي پائي ڏسو۔ جيڪڏهن ڪو توهان جو هڪ جيڏو دوست توهان کي هزار ڀيرا ڪجھ پنهنجا ڪم چوي ۽ توهان ان جو هڪڙو ڪم به نه ڪريو ته پنهنجو ڪم ان کي چوندي، پهريان ته پاڻ لڄي ٿيندؤ جو اسان ته سندس چوڻ ورتو ئي ناهي هاڻي ڪهڙي منهن سان ان کي ڪم لاءِ چئون؟ ۽ جيڪڏهن ضروري هيو ته چئي به ڏنؤ ۽ ان دوست نه ڪيو ته ڪهڙيءَ ريت به ڏوراپو ڪونه ڏيندؤ۔ ظاهر آهي پاڻ ئي سمجهو ٿا جو مون (ان جو ڪم) ڪٿي ڪيو هو؟ جو هُو ڪري!

هاڻي ڏسو ته توهان الله عَزَّوَجَلَّ جي ڪيترن حڪمن تي عمل ڪريون ٿا؟ ان جا حڪم نه مڃڻ ۽ پنهنجي گذارش جي هر صورت ۾ قبوليّت چاهڻ ڪيڏي نه بي حيائي آهي!
