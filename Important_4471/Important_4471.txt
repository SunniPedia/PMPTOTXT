٭ حضرت سَيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي ڪنيت ابُوحامِد، لَقَب حُجَّة الاِسلَام (اسلام جو دليل) ۽ مبارڪ نالو محمد بن محمد غزالي شافعي آهي، پاڻ ۴۵۰ھ ۾ خراسان جي ضلعي طُوس جي علائقي طابِران ۾ پيدا ٿيا۔ (اتحاف السادة المتقين ،مقدمة الڪتاب ،۱/۹) ٭ حضرت امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو شُمار انهن مُقدَّس هستين ۾ ٿئي ٿو، جن پنهنجي سڄي زِندگي علم حاصل ڪرڻ ۽ تبليغِ دين جي لاءِ وَقف ڪري ڇڏي هئي۔ ٭ حضرت سَيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اِحياءِ دِين جي لاءِ اهڙا اهڙا ڪارناما سر اَنجام ڏنا جو پنهنجي وَقت جا مُجَدِّد ٿي مشهور ٿيا۔ ٭ حضرت سَيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ابتدا ۾ وڏي شان ۽ شوڪت سان زندگي گذاريندا هئا پر بعد ۾ زُهد واري انداز کي اختيار فرمايائون۔ حضرت سَيدُنا ابُومنصور سعيد بن محمد رَحْمَةُ اللهِ تَعَالٰی عَلَيْه بيان فرمائن ٿا: جڏهن پهرين ڀيري حضرت سيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه عالمانه شان وشوڪت سان بغداد ۾ داخل ٿيا ته اسان سندن لِباس ۽ سواري جي قيمت لڳائي ته اها پنج سو (500) دِينار (يعني 500 سؤ سون جا سِڪا) ٿي، پوءِ جڏهن پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه زُهد ۽ تَقويٰ اِختيار ڪري بغداد ڇڏيائون ۽ مختلف مَقامن جو سفر ڪندا رهيا ۽ ٻيهر جڏهن بغداد ۾ داخل ٿيا ته اسان سندن لباس جي قيمت لڳائي ته اها پندرهن (15) قيراط (يعني ڪجهه معمولي سِڪا) ٿي۔“ (المنتظم في تاريخ الملوڪ والامم، ۹/۱۲۶، الجزء: ۱۷)

امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ۽ احياءُ العلوم

٭ ” امام غزالي جو ڪتاب احياء العلوم “ اَخلاقيات جي مَوضُوع تي بي مثال ڪتاب آهي۔ ٭ ” احياء العلوم “ پڙھي آخرت جي تياري جو ذهن ٺهندو آهي، ٭ ”احياء العلوم“ آخرت جي تياري تي اُڀارڻ وارو ڪتاب آهي، ٭ احياءُ العلوم نفس جي چُنگل مان ڇوٽڪارو ڏيارڻ وارو عظيمُ الشّان ڪتاب آهي، ٭ ” احياء العلوم “ بار بار پڙھڻ لائق ڪتاب آهي۔ ٭ ” احياء العلوم “ ۾ ظاهري عُلُوم سان گڏوگڏ باطِني عُلُوم جو به بيان ڪيو ويو آهي۔ ٭ ” اِحياءُ العلوم “ دُنيا سان گڏ آخرت کي به بهتر بنائڻ جو ذريعو آهي۔ ٭ ”احياء العلوم“ شيطان جي شرارتن کي سڃاڻڻ جو مؤثر ذريعو آهي، ٭ ” احياء العلوم “ جو مُطالَعو ۽ ان ۾ بيان ٿيڻ

وارين ڳالهين تي عمل دلين جي صفائي جو سبب آهي۔ ٭ ” احياء العلوم “ انسان کي ’’ڪامل انسان‘‘ بنائڻ وارو ڪتاب آهي۔ اميرِ اهلسنَّت، باني دعوتِ اسلامي حضرت علامه مولانا ابُوبلال محمد الياس عطار قادري رضوي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ هن ڪتاب جي اَهميت بيان ڪندي فرمائن ٿا: باطن جي اِصلاح ۾ حُجَّةُ الاِسلام حضرت سَيدُنا امام محمد بن محمد بن محمد غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو مون تي وڏو اِحسان آهي۔ سَيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا ڪتاب مِنهاجُ العابِدِين ۽ اِحياءُ العُلُوم وغيره پڙھندي ڪيئي ڀيرا اهو محسوس ٿيندو آهي، ڄڻ مون کي ئي ڪن پڪڙي سمجھائي رهيا آهن ته ”وڏو نيڪ سڏائين ٿو ٿورو پنهنجو اندر ته ڏس! ت و ۾ ته هي به خرابي آهي ۽ تنهنجي اندر ته هوءَ به بُرائي آهي، ۽ جڏهن به پڙهندو آهيان ائين لڳندو آهي ته رُوح کي نئي نئي غِذا ملي رهي آهي، سَيدُنا امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا ڪتاب هڪ ٻه ڀيرا پڙهي رکڻ وارا نه آهن، زِندگي جي آخري پساھ تائين پڙهڻ لائق آهن۔“

اهو به ياد رهي ته هلاڪت ۾ وجھڻ وارن عملن جي ضروري معلومات ۽ انهن کان بچڻ جي طريقن بابت ڄاڻ فرض آهي، اهڙي طرح نجات ڏيارڻ وارن عملن جي ضروري معلومات ۽ انهن کي حاصل ڪرڻ جا طريقا ڄاڻڻ به فرض آهن ۽ احياء العلوم ان علم کي حاصل ڪرڻ جو بهترين ذريعو آهي۔

نجات ڏيارڻ وارن عملن جي معلومات جي لاءِ مڪتبة المدينه جو ڪتاب " نجات دِلانے والے اعمال کی معلومات مع 41 حڪايات " به بهترين مددگار آهي۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

8 مَدَني ڪمن مان هڪ مَدَني ڪم ’’مدني انعامات‘‘

8 مدني ڪمن مان هڪ مدني ڪم ”مدني انعامات “ تي عمل به آهي اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ امير اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جا عطا ڪيل 63 مدني انعامات

نيڪ بڻجڻ ۽ پنهنجي احتساب جو بهترين نسخو آهي. تنهنڪري وقت مقرر ڪري روزانو فڪرِ مدينه ڪيو. (يعني مدني انعامات جي مطابق اڄ ڪيستائين عمل ٿيو) رسالي ۾ ڏنل خانا ڀري ڪري هر مدني مهيني جي پهرين تاريخ تي پنهنجي ذميداراسلامي ڀيڻ کي جمع ڪرايو وڌيڪ مڪتبة المدينه جي شايع ڪيل ڪتاب ”جنت جي طلبگارن جي لاءِ مدني گلدستو“ جي ذريعي ٻين اسلامي ڀينرن کي به “مدني انعامات“ تي عمل ڪرڻ جي ترغيب ڏياريو. هر اسلامي ڀيڻ انفرادي ڪوشش ڪرڻ واري ”مدني انعامات“ تي عمل ڪندي هر مهيني مدني انعامات جا گهٽ ۾ گهٽ 26 رسالا تقسيم ڪري ايندڙ مهيني وٺڻ جي به ڪوشش ڪريو. حدف في ذيلي حلقو گهٽ ۾ گهٽ 12 رسالا آهن.

اچو ترغيب جي لاءِ مدني انعامات تي عمل جي هڪ مدني بهار ملاحظه ڪيو جيئن ته

مدني انعامات تي عمل جي برڪت سان چل مدينه جي سعادت ملي وئي

باب المدينه (ڪراچي) جي هڪ اسلامي ڀيڻ جنهن جي عملي زندگي ڪجهه درست نه هئي خاص ڪري نمازن جي پابندي سان محروم هئي وڌيڪ هي جو فيشن ڪرڻ گانا باجا ٻڌڻ جي نحوست ۾ گرفتار هئي، ان جي پڦاٽ (جيڪو دعوت اسلامي جي مشڪبار مدني ماحول سان وابسته هو) انفرادي ڪوشش ڪندي ان جي ڀاءُ کي دعوت اسلامي جي هفته وار سنتن ڀري اجتماع ۾ شرڪت جي نه صرف دعوت ڏني بلڪ پاڻ سان گڏ وٺي وڃڻ شروع ڪري ڏنو. ان جو ڀاءُ سنتن جي اجتماع کان واپسي اچي اجتماع جي ڪيفيت ٻڌائيندو هو جنهن ۾ سيدي اعلٰي حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو ذڪر خير ٻڌڻ جي لاءِ ملندو هو جنهن جي سبب ان اسلامي ڀيڻ کي دعوت اسلامي جي مدني ماحول سان پنهنجائپ محسوس ٿيڻ لڳي . اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ انهي پنهنجائپ جي سوچ ان کي پهريون ڀيرو 1985 جي سالانه سنتن ڀري اجتماع جي خصوصي نشست تي اڀاريو. تنهنڪري اها به اسلامي ڀينرن جي اجتماع ۾ شريڪ ٿي وئي جتي ان پردي ۾

رهي ڪري سنتن ڀري اجتماع ۾ ٿيڻ وارو بيان ٻڌو ۽ روئي ڪري دعا گهري. اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ انهي اجتماع ۾ شرڪت جي برڪت سان ان کي گناهن کان توبهه ڪرڻ جي سعادت نصيب ٿي، فڪر آخرت ملي. جنهن تي استقامت ماڻڻ جي لاءِ انهن مدني انعامات تي عمل ڪرڻ شروع ڪري ڏنو. مدني انعامات جي برڪت سان اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ ان کي چل مدينه جي سعادت به حاصل ٿي . (اسلامي ڀينرن جي نماز، ص 290)

ٻيو طريقو نيڪ ماڻهن جي صحبت اختيار ڪريو!

پياريون پياريون اسلامي ڀينرون! اسان پنهنجي آخرت جي تياري ڪرڻ جا طريقا ٻڌي رهيون هيوسين، آخرت جي تياري ڪرڻ جو هڪ طريقو اهو به آهي ته اسان الله وارن جا تذڪرا ٻڌون ۽ انهن جي سيرت ۽ ڪردار جو مطالعو ڪري ان کي پنهنجي زندگي ۾ نافذ ڪرڻ جي به ڪوشش ڪريون ۔ جهڙي طرح ڪو شخص پنهنجي دُنيا کي بهتر کان بهترين ڪرڻ جي لاءِ فڪر مند رهي ٿو، الله پاڪ جا اهي نيڪ ٻانها ان کان وڌيڪ پنهنجي آخرت جي بهتري جي لاءِ مصروفِ عمل هوندا آهن، اهي ماڻهو پنهنجون زندگيون هن طرز سان گذاريندا آهن جو بعد وارن جي لاءِ صرف انهن جو ذڪر ڪرڻ به الله پاڪ جون رحمتون ماڻڻ جو سبب بڻجي ويندو آهي ۔ ان ڪري جڏهن ڪو الله وارن سان وابسته ٿي وڃي ته اهو به انهن جي رنگ ۾ رنگجي آخرت کي بهتر بنائڻ ڏانهن مائل ٿي ويندو آهي ۔ انهن جي سيرت ۽ ڪردار جو مطالعو ڪرڻ وارو بڻجي ته جيڪڏهن الله پاڪ جي رحمت شامل ٿي وڃي ته ان ۾ به اهو ئي انداز پيدا ٿيڻ شروع ٿي ويندو جيڪو آخرت جي تياري ۾ فائديمند ٿي سگھي ۔ ان کي به اها مدني سوچ ملي ويندي، جنهن سان آخرت بهتر ٿي سگھي ٿي ۔ ڇوته الله پاڪ جا اهي نيڪ ٻانها پنهنجي آخرت کي بهتر بڻائڻ جو ڪو به موقعو هٿ مان وڃڻ ناهن ڏيندا، هر اهو ڪم جنهن سان انهن جي

آخرت سنورجي سگھي ٿي، اهي نيڪ انسان يڪدم ڪندا آهن، تنهنڪري اسان کي به ڪوشش ڪرڻ گهرجي ته اسان جو اٿڻ ويهڻ، ميل جول اهڙين اسلامي ڀينرن سان هجي جيڪي هن فاني دنيا جي رنگينين کي وساري فڪرِ آخرت ڏانهن مائل هجن، خوفِ خدا ۽ عشقِ مصطفٰي جي لازوال دولت سان مالا مال هجن، محبتِ صحابه ۽ اهلِ بيت جي دولت سان سرشار هجن، مٿي کان پيرن تائين سنتن جون پابند هجن، اطاعتِ الٰهي ۽ اتباعِ رسول جو مقدس جذبو رکنديون هجن، جن جي صحبت جي برڪت سان عمل جو جذبو وڌندو هجي، جن جي صحبت جي برڪت سان گناهن کان نفرت ۽ فڪرِ آخرت نصيب ٿيندي هجي.

صحبت ضرور رنگ آڻيندي آهي!

حضرت سَيدُنا مُصْلحُ الدِّين سعدي شيرازي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه سٺي صحبت جي اهميت کي نهايت خوب صورت انداز ۾ بيان فرمايو آهي جيئن پاڻ ”گلستانِ سعدي“ ۾ فرمائن ٿا، جنهن جو ترجمو ڪجھ هن طرح آهي: يعني هڪ ڏينهن خوشبو واري مٽي حَمَّام ۾ مون کي هڪ دوست جي هٿن مان ملي، مون ان مٽي کي چيو ته تون مُشڪ آهين يا عنبر! تنهنجي دلڪش خوشبو مون کي مَست ۽ بيخود ڪيو آهي (هي ٻڌي مٽي چيو) مان ته حقير (يعني ادنيٰ) مٽي هئس، پر هڪ عرصي تائين مان گُلن جي صحبت ۾ رهيس پوءِ هم نشين جي جمال مون ۾ اثر ڪيو (ته مان خوشبودار ٿي ويس) نه ته مان ته اها ئي خاڪ ۽ مٽي آهيان جيڪا پهريان هئس۔ (گلستان سَعدي ص۴)
