پاڻ رَحْمَۃُ الله تَعَالٰي عَلَيه تمام وڏا عابد ۽ زاهد، الله عَزَّ وَجَلَّ جي معرفت ۽ خوف رکڻ وارا هئا، پنهنجي علم ذريعي هميشه رضاءِ الٰهي عَزَّوَجَلَّ تلاش ڪندا هئا۔

امامِ اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه اڳ ۾ تِجارت ڪندا هئا، هڪ ڏينهن پاڻ رَحْمَۃُ الله تَعَالٰي عَلَيه جي مُلاقات امام شَعْبي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه سان ٿي، انهن امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کان پُڇيو: اوهان ڪهڙو ڪم ڪندا آهيو؟ امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جواب ڏنو ته مان بازار ۾ ڪاروبار ڪندو آهيان۔ امام شَعبي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه اِرْشاد فرمايو: بازار ۾ ڇو مصروف آهيو؟ عالمن وٽ اچ وڃ ڪيو۔ امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا مون عرض ڪيو: عالمن وٽ گھٽ ويندو آهيان، پاڻ رَحْمَۃُ الله تَعَالٰي عَلَيه فرمايو: اوهان عِلْمِ دِين کان غافل نه ٿيو! بلڪ اوهان جي لاءِ ته لازمي آهي ته علم ۽ عُلَمائن جي مجلس ۾ شريڪ رهو، مان اوهان ۾ عِلْمِ دِين جي سُڌ ٻُڌ ۽ ان جي دانشمندي جو نشان ڏسان پيو۔ امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا امام شَعبي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جي اها ڳالھ منهنجي دل ۾ ويهي رهي ۽ مون بازار ۾ ويهڻ ڇڏيو ۽ عِلْمِ دِين ڏانهن مُتوجّه ٿيس، الله عَزَّوَجَلَّ مون کي ان جو گھڻو نَفْعو عطا فرمايو ۔ (المناقب للموفق،الجزء الاول ، ص۵۹) منقول آهي ته عُمر جي آخري حصّي ۾ حَضْرتِ سَيدُنا امامِ اعظم ابُوحنيفه رَحْمَۃُ الله تَعَالٰي عَلَيه کان ڪنهن سوال

ڪيو ته اوهان هن بُلند مرتبي تي ڪيئن پهتا؟ ته پاڻ رَحْمَۃُ الله تَعَالٰي عَلَيه اِرْشاد فرمايائون: ’’ مون پنهنجي علم ذريعي ٻين کي فائدو پهچائڻ ۾ ڪڏهن به ڪنجوسي کان ڪم نه ورتو ۽ جيڪو مون کي نه ايندو هو ان بابت ٻين کان فائده حاصل ڪرڻ ۾ ڪڏهن به شرم محسوس نه ڪيو۔“ (الدرالمختار، المقدمۃ، ج۱، ص۱۲۷)

مٺا مٺا اسلامي ڀائرو! اسان جا امام اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه نهايت زبردست ذهن، عُمده فڪر ۽ شاندار ذهانت جا مالڪ هئا، سرڪارِ اعليٰ حضرت، امام اهلسنّت، مولانا شاھ اِمام احمد رضا خان رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فتاويٰ رضويه شريف ۾ نقل فرمائن ٿا ته حَضْرتِ سَيدُنا امام شافعي رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمايو: سڄي جهان مان ڪنهن جو به عقل

(امام اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه) جي عقل برابر ناهي۔ حضرت بَڪر بن حُبَيش رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه چيو: جيڪڏهن (امام) ابُو حنيفه جو عقل ۽ سندن زماني وارن جا عقل گڏ ڪيا وڃن ته انهن سڀني عقلن جي مجموعي تي سندن عقل غالب اچي ويندو۔ (الخيرات الحسان،الفصل العشرون ،ص۶۲،از فتاويٰ رضويه۱/۱۴۸) جهڙي طرح ذهانت وفطانت ۾ پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جو ڪو ثاني نه هو اهڙي طرح پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه تقويٰ وطهارت، عِبادت و رِياضت ۽ خوف وخشيت ۾ به بي مثال هئا، پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جي زُهد و تَقويٰ، عِبادت و رِياضت جي گواهي ڪيئي جَلِيلُ الْقدر بُزرگانِ دين پڻ ڏني آهي، اچو! هن باري ۾ ڪجھ عالم سڳورن جا اقوال ٻڌون ٿا:

حضرتِ عبدُالله بن مُبارَڪ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: هڪ ڀيرو مان ڪوفي ويس ۽ اتي جي ماڻهن کان پڇيو: هتي سڀ کان وڏو زاهد ۽ عابد ڪير آهي؟ ماڻهن چيو: امامِ اعظم ابُوحنيفه (رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه) حضرت ابنِ مُبارَڪ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه وڌيڪ فرمائن ٿا: مون امام اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کان وڌيڪ پرهيزگار ڪنهن کي نه ڏٺو، توهان اهڙي شَخْص جي ساراھ ڪرڻ جي ڇا قدرت رکو ٿا، جن جي سامهون ڪافي مال پيش ڪيو ويو پر اُنهن ان مال جي ڪا پرواھ نه ڪئي، ان سبب انهن کي دُرن سان ماريو ويو، پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه ا ٓ سائش و تڪليف ٻئي حالتون الله عَزَّ وَجَلَّ جي عِبادت ۾ گُذاريون ۽ ان شيء کي ڪڏهن قبول نه ڪيو جنهن جي لاءِ ماڻهو

2. امام بخاري رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جا اُستادِ محترم، حضرتِ سَيدُنا مڪي بن ابراهيم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: مون اهلِ ڪوفه مان ڪيئي عالمن جي صحبت اختيار ڪئي پر انهن مان ڪنهن کي امام اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کان وڌيڪ مُتَّـقي ۽ پرهيزگار نه ڏٺو۔ (الخيرات الحسان، ص۵۸)

3. حضرتِ يزيد بن هارون رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: مون هڪ هزار اُستادن وٽان علم حاصل ڪيو پر امام صاحب کي تقويٰ ۽ زبان جي حفاظت ۾ سڀني کان وڌيڪ ڏٺو۔ (الخيرات الحسان، ص۵۸)

4. حضرت وڪيع رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه پنهنجي پاڻ تي لازم ڪيو هو ته جيڪڏهن سچي ڳالھ تي به قسم کنيو ته هڪ دِرهم صدقو ڪندس، هڪ ڀيرو قسم کنيو ته هڪ دِرهم صدقو ڪيو، ان کان پوءِ اهو لازم ڪيو ته هاڻي جيڪر قسم کنيو ته هڪ دينار صدقو ڪندس ته جڏهن به قسم کڻندا ها ته هڪ دِينار صدقو ڪندا ها۔ (الخيرات الحسان، ص۵۸)

مٺا مٺا اسلامي ڀائرو! ياد رکو! تقويٰ ۽ پرهيز گاري آخرت جي سَفر جي ڪاميابي جي لاءِ نهايت اَهميت رکي ٿي، هن نعمتِ عُظمٰي جي بغير اُخروي ڇوٽڪارو حاصل ڪرڻ وڏو مشڪل آهي، ڇوته عِبادات کي ادا ڪرڻ ۽ گُناهن کان باز رهڻ جو عظيم ذريعو الله عَزَّ وَجَلَّ جو خوف ۽ تقويٰ اِختيار ڪرڻ آهي۔

تفسيرِ ڪبير ۾ تقوي جي تعريف اها لکيل آهي ته توهان پنهنجي باطن کي الله عَزَّ وَجَلَّ جي لاءِ اهڙو ئي سنواريو جيئن توهان پنهنجي ظاهر کي مَخْلُوق جي لاءِ سنواريو ٿا۔ اهو به چيو ويو آهي ته مُتَّـقي اهو آهي جيڪو مُصطَفٰي ڪريم صَلَّي الله تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّم جي راھ تي هلي، دُنيا کي پٺ جي پويان رکي، پنهنجي نَفْس کي اِخلاص ۽ وفا جو عادي بڻائي ۽ حرام ۽ نافرماني وارن ڪمن کان اِجتناب ڪري۔ (تفسيرِ ڪبير،تحت الآيۃ: ذلڪ الڪتاب لاريب فيه،ج۱،ص۲۶۸) حضرت سُفيان ثوري رَضِي الله تَعَالٰي عَنْه فرمائن ٿا ته پرهيز گارن کي مُتَّـقي ان ڪري چيو ويندو آهي جو اهي اهڙين شين کان به بچندا آهن جنهن کان عام طور تي بچڻ دُشوار هوندو آهي۔ (تفسير در منثور ،ج۱،ص۶۱)

مَعْلُوم ٿيو ته الله عَزَّ وَجَلَّ وٽ عزت ۽ فضيلت جو دارومَدار مال و دولت ۽ دُنيوي عزت و مَنصب هرگز ناهي بلڪ الله عَزَّوَجَلَّ جا پسنديده ماڻهو تقويٰ و پرهيز گاري اختيار ڪرڻ وارا آهن۔ مُتَّـقي ماڻهن جو شان بيان ڪندي الله عَزَّوَجَلَّ قرآنِ پاڪ ۾ اِرْشاد فرمائي ٿو: ﴿ اِنۡ اَوۡلِیَآؤُہٗۤ اِلَّا الۡمُتَّقُوۡنَ (پ۹، الانفال:۳۴) تَر ْ جمو ڪنزالايمان؛ ان جا اولياء ته پرهيز گار ئي آهن ﴾ يعني تقويٰ اِختيار ڪرڻ وارا ئي الله عَزَّوَجَلَّ جا دوست آهن۔ يقيناً اسان مان هر هڪ جي اها خواهش هوندي ته ان کي الله عَزَّوَجَلَّ جي رضا حاصل ٿئي ته قرآنِ پاڪ ۾ ان جي ولايت حاصل ڪرڻ جو طريقو تقويٰ اِختيار ڪرڻ ٻڌايو ويو آهي۔

(2) تقويٰ کي حاصل ڪرڻ جو هڪ طريقو اهو به آهي ته اِنْسان غافل ۽ بُرن ماڻهن جي صحبت ۾ اٿڻ ويهڻ ڇڏي ڏئي ۽ مُتَّـقي، پرهيز گار ماڻهن جي صحبت اختيار ڪري، ته جيئن انهن جي صحبت جي تاثير سان ان جي دل ۾ به تقويٰ ۽ پرهيزگاري پيدا ٿئي۔

(3) تقويٰ حاصل ڪرڻ جي لاءِ بُزرگانِ دين جي سيرت مان تقويٰ ۽ پرهيزگاري جا واقعا پڙھو يا ٻڌو ته جيئن انهن مُبارَڪ هستين جي واقعن مان سبق حاصل ڪري اسان به هن پاڪيزه صِفَت کي پنهنجي ذات تي نافذ ڪرڻ ۾ ڪامياب ٿي سگھون۔ دعوتِ اسلامي جي اِشاعتي اِداري مڪتبة المدينه جي شايع ٿيل اِحياء العلوم جلد2، صفحي 367 تان " مُتّقيوں کی حکايتیں "پڙھڻ سان به اِنْ شَآءَ الله عَزَّ وَجَلَّ تقويٰ اِختيار ڪرڻ جو ذهن بڻجندو۔

حضرتِ سَيدُنا امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جي باري ۾ اچي ٿو ته پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه عِبادت دوران خوفِ خدا کان روئيندا رهندا هئا، ايستائين جو رات جو ايتري قدر آه و زاري ڪندا هئا جو پاڙيسرين کي به پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه تي رحم ايندو هو، حضرت سيدُنا حَفْص بن عبدُالرَّحمٰن رَحْمَۃُ الله تَعَالٰي عَلَيه امامِ اعظم جو رات مهل ڪثرت سان عبادت

جو تذڪره ڪجھ ائين فرمائن ٿا جو پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه ٽيهن (30) سالن تائين سڄي رات هڪ رڪعت ۾ قرآنِ ڪريم جي تلاوت ڪندا هئا، چيو وڃي ٿو ته جنهن جڳھ سندن وفات ٿي ان هنڌ تي پاڻ رَضِي الله تَعَالٰي عَنْه ست هزار (7000) ڀيرا قرآنِ ڪريم ختم ڪيو هو۔ حضرت سيدُنا اسد بن عَمرْو رَحْمَۃُ الله تَعَالٰي عَلَيه فرمائن ٿا ته امامِ اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه چاليھ (40) سالن تائين عشاء جي وضو سان فجر نماز ادا ڪئي۔

حضرت سيدُنا زائِدَه رَحْمَۃُ الله تَعَالٰي عَلَيه فرمائن ٿا ته مون امامِ اعظم ابوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه سان گڏ عشاء نماز ادا ڪئي۔ نماز جي بعد ماڻهو هليا ويا ۽ مان مسجد ۾ ئي ترسي پيس۔ منهنجو اِرادو هو جو پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کان هڪ مسئلو معلوم ڪندس پر پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کي مسجد ۾ منهنجي موجودگي جو علم نه ٿيو ۽ پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه قرآنِ ڪريم جي تلاوت شُروع ڪري ڇڏي، جڏهن پاڻ هن آيت سڳوري تي پهتا،

فَمَنَّ اللّٰہُ عَلَیۡنَا وَ وَقٰىنَا عَذَابَ السَّمُوۡمِ ﴿۲۷﴾ (پ۲۷، الطور: ۲۷)

تَر ْ جموڪنز الايمان: پوءِ الله اسان تي احسان ڪيو ۽ اسان کي ساڙيندڙ عذاب کان بچائي ورتائين۔

ته طُلوعِ فجر تائين ان کي دُهرائيندا رهيا۔ (تاريخ بغداد،۱۳/ص۳۵۲، ۳۵۵، بتغير)

اهڙي طرح هڪ رات پاڻ رَحْمَۃُ الله تَعَالٰي عَلَيه مسجد ۾ ڪنهن قرآن جي قاري کي اها آيتِ مُبارَڪه تلاوت ڪندي ٻڌو، ﴿ اِذَا زُلۡزِلَتِ الۡاَرۡضُ زِلۡزَالَہَا ۙ﴿۱﴾

(پ۳۰ ، الزلزال:۱) تَر ْ جموڪنز الايمان: جڏهن زمين ڌڏائي وڃي جيئن ان جو ڌوڏڻ مقرر ٿيل آهي ﴾ ته خوف جي شِدَّت کان فجر تائين پنهنجي ڏاڙھي مُبارَڪ هٿ ۾ پڪڙي اها ڳالھ هر هر چوندا رهيا ته ” اسان کي ذرّي برابر گُناھ جي به سزا ڏني ويندي۔“ (حڪايتيں ۽ نصيحتيں ،ص۳۳۵)

عطا ہو خوفِ خدا خدارا، دو اُلفتِ مُصْطَفٰی خدارا کروں عمل سُنَّتوں پہ ہر دم، امامِ اعظم ابوحنيفه

(وسائلِ بخشش مُرمّم، ص:۵۷۳)

مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان اسان جا امامِ اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه خوفِ خدا جي سبب روئي روئي سڄي رات هڪ ئي آيت سڳوري جي تلاوت ڪندا رهيا ۽ فڪرِ آخرت ۾ اهڙا مگن رهيا جو رات گُذرڻ جو به احساس نه ٿيو۔ جڏهن ته اسان جو مُعاملو هي آهي جو ڏينهن ته الله عَزَّوَجَلَّ جي نافرماني ۽ ناراضگي وارن ڪمن ۾ گُذاريندا آهيون، راتيون به گُناهن ۾ گذرينديون آهن، دوستن سان گڏ فلمون ڊراما ڏسي پنهنجيون اکيون حرام سان ڀريندا آهيون، شادي وغيره جي محفلن ۾ سڄي رات شور شرابي سان ماڻهن جون ننڊون خراب ڪندا آهيون، انٽرنيٽ ۽ موبائل جي نائٽ پيڪجز تي فُضول ۽ بي حيائي و بي شرمي وارين ڳالهين ۾ ۾ سڄي رات برباد ڪندا آهيون، اسان مان هر هڪ پنهنجي دل کان پڇي ته ڇا اسان به ڪڏهن پنهنجي بُزرگن وانگر سڄي رات عبادت ۾ گذاري آهي؟ ڇا اسان جي دل ۾ به تلاوتِ قرآن جو شوق پيدا ٿيو؟ ڇا اسان به ڪڏهن رات جي اونداهي ۾ قبر

جي تاريڪي جو تَصَوُّر ڪيو آهي؟ قبر وحشر جي هولناڪين کي ياد ڪري خوف خدا مان لڙڪ جو هڪ قطرو به وهايو آهي؟ جيڪڏهن جواب ’’نه‘‘ ۾ اچي ته اڄ کان اهي سڀئي ڪم ڪرڻ جي هٿو هٿ نِيت ڪري وٺو ۽ دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي وڃو، اِنْ شَآءَالله عَزَّوَجَلَّ ان جي برڪت سان فرائض جي پابندي سان گڏوگڏ نفلي عبادات ڪرڻ جو به ذِهن بڻندو۔ ۽ ذيلي حلقي جي 12 مدني ڪم ڪرڻ جي به ڀرپُور ڪوشش ڪيو، جيڪڏهن اسان مدني ڪمن جا عامل بڻجي وياسين ته ٻيون برڪتون ماڻڻ سان گڏوگڏ روزانو تلاوتِ قرآن شريف ترجمي ۽ تفسير بيان ڪرڻ يا ٻڌڻ جي به سعادت ملي ويندي.

ا َلْحَمْدُلِلّٰه عَزَّوَجَلَّ ذيلي حلقي جا 12 مَدَني ڪمن مان روزانو جو هڪ مَدَني ڪم فجر کان پوءِ ” مَدَني حلقه“ به آ هي۔ جنهن ۾ روزانو، ٽي (3) قرآني آيتن جي تلاوت مع ترجَمه ڪنْزالايمان ۽ تفسيرخزائنُ العرفان يا تفسير نُورُالعرفان يا تفسيرصِراطُ الجنان، درسِ فيضانِ سُنَّت (4صفحا) ۽ منظوم شَجْره قادِريه، رَضَويه، ضيائيه، عطاريه پڙھيو ويندو آهي۔ انهن سڀني ڪمن جا دُنيا ۽ آخرت ۾ بي شُمار فائدا آهن، منقول آهي ته ٻنهي عالَم جي مالڪ ومختار صَلَّي الله تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّم اِرْشاد فرمايو: جس شَخْص قرآنِ پاڪ سکيو ۽ سکاريو ۽ جيڪو ڪجھ قرآن شريف آهي ان تي عمل ڪيو، قرآن شريف ان جي شفاعت ڪندو۔ (تاريخ دمشق لابن عساڪر ج۴۱ ص۳) ان کانپوءِ فيضانِ سُنَّت جو درس ڏنو ويندو آهي جنهن

ذريعي علمِ دين جو فيضان عام ٿيندو آهي پوءِ شَجْره قادريه، رضويه، ضيائيه، عطاريه پڙھيو ويندو آهي، جنهن ۾ الله عَزَّوَجَلَّ جي نيڪ ۽ بَرگُزيده ٻانهن جو ذڪر ٿيندو آهي، اهڙي طرح نيڪن جي تذڪري جون برڪتون ماڻڻ جي سعادت ملندي آهي، حضرتِ سُفيان بن عُيينَه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جو فرمان آهي: ” عِنْدَذِکرِالصَّالِحِينَ تَنْزِلُ الرَّحْمَۃُ “ يعني نيڪ ماڻهن جي تذڪري مهل رَحمت نازل ٿيندي آهي۔ (حِليۃُ الاولياء، رقم ۱۰۷۵۰، ج ۷ ، ص۳۳۵) آخر ۾ اِشراق و چاشت جي نفلن تي مدني حلقي جو اِختتام ٿيندو آهي۔ مَدَني حلقي ۾ جيڪو شجرو شريف پڙھيو ويندو آهي، ان جي برڪتن جي به ڇا ڳالھ آهي، ان کي روزانو پڙھڻ سان مشڪلون آسان ٿينديون آهن، ذِهني سُڪون نصيب ٿيندو آهي ۽ بيماريون به ختم ٿينديون آهن، اچو! ترغيب جي لاءِ هڪ مدني بهار ٻڌون ٿا:

هڪ اسلامي ڀاءُ جي تحرير جو خُلاصو آهي ته تقريباً پوئين 4 سالن کان مان مٿي جي سور جي مَرَض ۾ مبُتلا هئس جيڪو ڪنهن صُورت ختم ٿيڻ جو نالو نه پئي ورتو، ڪافي علاج مُعالجو ڪرايو پر ان باوُجُود فرق نه ٿيو۔ انهيءَ دَرد مان ڇوٽڪاري جي صُورت ائين بڻي جو هڪ ڏينهن خوش قسمتي سان شيخِ طريقت، اميرِ اهلسنّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي عطا ڪيل ورد ۽ وظيفن تي مُشتمل قادريه رضويه عطاريه سِلسلي جو شجرو شريف مون کي مليو ته برڪت حاصل ڪرڻ جي نِيت سان ان ۾ مَوْجُود شجرو شريف پڙھڻ لڳس، انهيءَ دوران مون محسوس ڪيو ته ان جي برڪت سان منهنجي مٿي جي سور ۾

ڪمي واقع ٿي آهي جيڪا منهنجي لاءِ حيران ڪُن هئڻ سان گڏوگڏ وڏي خوش آئند ڳالھ به هئي، مون ٻيهر ان کي پڙھڻ شُروع ڪيو ته وڌيڪ دَرد ۾ اِفاقو ٿيو، اهو ڏسي نه صرف منهنجو ايمان تازه ٿي ويو بلڪ منهنجي بيماري جو علاج به مون کي ملي ويو۔ اَلْحَمْدُلِلّٰه عَزَّوَجَلَّ هاڻي روزانو شجرو شريف پڙھڻ جي ترڪيب آهي، جنهن جي برڪت سان منهنجي مٿي جو سور بلڪل ختم ٿي ويو آهي۔ پر جنهن ڏينهن پڙهڻ رهجي ويو ته دَرد ٻيھر موٽي ايندو آهي، ان ڪري مان روزانو شجرو شريف پڙھندو آهيان۔ سڄي زندگي هن کي ڪڏهن به ترڪ نه ڪرڻ جي نِيت ڪئي آهي۔

جيڪي عاشقانِ رسول اميرِاهلسنّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جا مُريد ياطالب نه آهن، جيڪڏهن اهي به شجره قادريه، رضويه عطاريه جي برڪتن مان فيض ياب ٿيڻ چاهين ٿا ته اَمِيرِاهلسُنَّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي ذريعي سلسله ءِ عاليه قادريه رضويه عطاريه ۾ مُريد يا طالب بڻجي وڃن، اُنهن کي به اَمِيرِاهلسُنَّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي صدقي شجري شريف جون خوب خوب برڪتون نصيب ٿينديون۔

اگر دَرْدِ سَر ہو کہيں کينسر ہو

دِلائے گا تم کو شِفا مَدَنی ماحول

شِفائيں مليں گی بَلائيں ٹليں گی

يقيناً ہے برکت بھرا مَدَنی ماحول

(وسائلِ بخشش مُرمّم ص: 648)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

حضرتِ مُصعب رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: هڪ ڀيري خليفي منصور امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کي ڏھ هزار درهم ڏيڻ چاهيا ٿي، امامِ اعظم

رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه سوچيو ته مون جيڪڏهن ان مال کي واپس ڪيان ٿو ته خليفي کي بُرو لڳندو ۽ مان ان کي پاڻ وٽ رکندس ته اهو مون کي ناپسند آهي۔ آخرڪار مون سان مشوره ڪيو، مون چيو ته اهو مال خليفي جي نگاھ ۾ تمام گھڻو آهي، جڏهن اهو ڏيڻ جي لاءِ اوهان کي سڏيو وڃي ته پاڻ فرمائي ڇڏجو: ”مون کي اميرُ المُوْمنين کان اهڙي اميد نه ھئي۔“ پوءِ جڏهن امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه کي ڏھ هزار درهم ڏيڻ جي لاءِ سڏيو ويو، ته پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه اهو ئي جملو اِرْشاد فرمايو، جڏهن منصور کي اها خبر ملي ته امامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه ائين چيو آهي ته (خليفي ايتري وڏي رقم کي حقير سمجھي ٺڪرائڻ جي ڪري امام اعظم کي دنياوي مال جو متوالو سمجھيو ۽) ان امام اعظم کي اهي پيسا نه ڏنا۔ (الخيرات الحسان،الفصل الخامس والعشرون، ص:۸۳)

مٺا مٺا اسلامي ڀائرو! ان واقعي مان معلوم ٿيو ته امام اعظم ابُوحنيفه رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه مالدارن طرفان مال حاصل ڪرڻ جا خواهشمند ۽ حريص بلڪل نه هئا ان ڪري ته خليفي جي طرفان تحفي طور ملڻ وارا درهم واپس موٽائي ڏنا۔

شيخِ طريقت،اَمِيرِاهلسنَّت حضرت علامه مولانا ابُوبلال محمد الياس عطار قادري رضوي ضيائي دَامَتْ بَـرَکاتُهمُ الْعَالِيه مالدارن کان پري رهڻ جي نصيحت ڪندي ارشاد فرمائن ٿا: اَربابِ اِقْتِدار ۽ سرمايه دار ماڻهن کان پري رهڻ ۾ ئي عافيت آهي۔ انهن جون دعوتون کائڻ ۽ انهن جا تحفا قبول ڪرڻ ۾ آخِرت جي لاءِ شدِيد خطرا آهن، ڇوته انهن جون دعوتون

کائڻ ۽ تحفي قبول ڪرڻ واري جو انهن جي خُوشامد ڪرڻ ۽ خوامخواه ها ۾ ها ملائڻ کان بچڻ نهايت ئي مشڪل هوندو آهي۔ حديث شريف ۾ ارشاد آهي؛ ''جيڪو ڪنهن غَني (يعني مالدار) جي ان جي مالداري جي سبب تواضُع ڪري ان جو اڌ دين ويندو رهيو۔ (ڪشف الخفاء ج۲ص۲۱۵ حديث ۲۴۴۲،ملتقطاً) اعليٰحضرت، امام اهلسنَّت مولانا احمد رضا خان رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه هن حديثِ پاڪ جي تَحت فرمائن ٿا: ” دنياوي مال جي لاءِ تواضُع الله عَزَّ وَجَلَّ جي خاطِر عاجزي ڪرڻ ناهي (ان ڪري) اها حرام ٿي۔“ (ذَيلُ المُدَّعالِاحسنِ الوِعاء ،ص۶۶،ملخصاً)

کيوں پِھريں شوق میں ہم مال کے مارے مارے

ہم تو سرڪار کے ٹکڑوں پہ پَلا کرتے ہيں

§ پاڻ رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه ڪوفي کان مديني جو وڏو سفر اِختيار ڪري مديني کان اچڻ واري گراهڪ کي 600 دِرهم به واپس ڪيا ۽ ڪپڙو به اُن وٽ رهڻ ڏنو۔

§ الله عَزَّوَجَلَّ اسان کي به پنهنجي نيڪ ٻانهن خاص ڪري اِمامِ اعظم رَحْمَۃُ اللّٰه ِتَعَالٰي عَلَيه جي نقشِ قدم تي هلندي زندگي گذارڻ جي توفيق عطا فرمائي ۔ ا ٰمِين بِجَاه النَّبِي الْاَمِين صَلَّي الله تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي؛ جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو. ([1])

سينہ تری سُنَّت کا مدينہ بنے آقا

جنّت میں پڑوسی مجھے تم اپنا بنانا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

اچو! شيخِ طريقت،اميرِاهلسنَّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي رسالي ’’163 مَدني گل‘‘ مان مسواڪ جون سنّتون ۽ آداب ٻڌون ٿا۔

پهريان ٻه فرمانِ مصطَفٰي صَلَّي اللّٰه تَعَالٰي عَلَيه وَاٰلِهٖ وَسَلَّمَ مُلاحَظه ڪيو: * ٻه رڪعتون مسواڪ (يعني ڏندڻ) ڪري پڙهڻ بغير مِسواڪ جي 70 رڪعتن کان افضل آهن ۔ ([2]) * مِسواڪ جو استعمال پاڻ تي لازم ڪري ڇڏيو ڇو ته هي وات جي صفائي ۽ ربّ تعاليٰ جي رضا جو سبب آهي ۔ ([3]) * حضرت سيدنا اِبن عباس رَضِىَ اللهُ تَعَالٰی عَنْهُ کان روايت آهي ته مسواڪ ۾ ڏهه خوبيون آهن: وات کي صاف ڪري ٿو، مسوڙن کي

مضبوط ڪري ٿو، نظر تيز ڪري ٿو، بلغم ختم ڪري ٿو، وات جي ڌپ ختم ڪري ٿو، سنت جي موافق آهي، ملائڪ خوش ٿين ٿا، ربّ عَزَّوَجَلَّ راضي ٿئي ٿو، نيڪي وڌائيندو ۽ معدي کي درست ڪندو آهي (جمع الجوامع ۵/۲۴۹،حديث۱۴۸۶۷) * حضرت سيّدنا امام شافعي عَلَيْهِ رَحمَةُ اللهِ الۡـقوي فرمائن ٿا: چار شيون عقل وڌائينديون آهن: فضول ڳالهين کان پرهيز، مسواڪ جو استعمال، نيڪ ماڻهن جي صحبت، پنهنجي علم تي عمل ڪرڻ ۔ ([1]) * مسواڪ پيلو يا زيتون يا نم وغيره جي ڪَوڙي ڪاٺي هجي مسواڪ جي ٿولھ چيچ جي برابر هجي ۔ * مسواڪ جڏهن استعمال جي قابل نه رهي ته اڇلايو نه، جو هي سنّت ادا ڪرڻ جو ذريعو آهي، ڪنهن جڳھ تي احتياط سان رکي ڇڏيو يا دفن ڪري ڇڏيو يا سمنڊ ۾ وجھي ڇڏيو ۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد
