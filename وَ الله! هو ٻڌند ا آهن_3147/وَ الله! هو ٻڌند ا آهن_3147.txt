حضرت سيدنا ابُوهرَيره رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته هڪ ڀيري نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اسان وٽ تشريف فرما ٿيا، پاڻ ڪريمن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان گڏ حسنينِ ڪريمين رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا به هئا، انهن مان هڪ ساڄي ڪلهي تي ۽ ٻيا کاٻي ڪلهي تي سوار هئا، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻنهي کي واري واري سان چُمي رهيا هئا۔ هڪ شخص عرض ڪيو: يارسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم!

اوهان انهن سان مَحَبَّت ڪيو ٿا؟ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: ها! جنهن انهن سان مَحَبَّت ڪئي، ان مون سان مَحَبَّت ڪئي ۽ جنهن انهن سان بُغض رکيو، ان مون سان بُغض رکيو۔

(مستدرڪ،ڪتاب معرفة الصحابة، رڪوب الحسن و الحسين ۔۔الخ،۴/۱۵۶،حديث: ۴۸۳۰)

حضرت سيدنا ابُو هرَيره رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي محبتِ رسول جو اهو معيار بارگاھِ رسالت مان عطا ٿيو هو، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ سڄي عمر ان کي اپنائي رکيو، جڏهن مروان بن حڪم حضرت سَيدنا ابُو هرَيره رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي بارگاھ ۾ وفات جي وقت حاضر ٿيو ۽ عرض ڪيائين: ”جڏهن کان اوهان جي صُحبت اختيار ڪئي آهي، مون اوهان جي ذات ۾ حضراتِ حسنينِ ڪريمين رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا جي مَحَبَّت تمام گھڻي ڏٺي آهي۔“ اهو ٻُڌي حضرت سيدنا ابُوهرَيره رَضِىَ اللهُ تَعَالٰی عَـنْهُ بيقرار ٿي بيهي رهيا ۽ ارشاد فرمايائون: هڪ ڀيري اسان نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان گڏ ڪيڏانهن وڃڻ جي لاءِ نڪتاسين، اڃا ڪجھ ئي رستو طئي ڪيو هو ته پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن حسنينِ ڪريمين رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا جي روئڻ جو آواز ٻڌو ۽ اهي ٻئي ان وَقت پنهنجي والده وٽ هئا، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جلدي جلدي هلندي، انهن وٽ تشريف کڻي ويا۔ مون پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي سيده فاطمة الزَّهرا رَضِىَ اللهُ تَعَالٰی عَـنْها کي اهو فرمائيندي ٻڌو: منهنجي پٽن کي ڇا ٿيو؟ عرض ڪيائون: اُڃ (يعني اُڃ جي ڪري ٻئي روئي رهيا آهن۔) نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پاڻي ڀرڻ جي لاءِ مَشڪيزي ڏانهن وڌيا، پر ان ۾ پاڻي مَوجُود نه هو ڇوته اُنهن ڏينهن ۾ پاڻي جي ايتري سخت قِلَّت هئي جو ماڻهو به پاڻي جي ڳولا ۾ رهندا هئا۔ نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ماڻهن کي پُڪاري فرمايائون: ڇا توهان مان ڪنهن وٽ پاڻي آهي؟

هر هڪ ڪجاوي (سوارين تي ويهڻ جي لاءِ اُٺ جي چيلھ تي ٺاهيل ڏولي) تي لٽڪيل مَشڪن ۾ پاڻي ڏٺو پر انهن مان هڪ ڦڙو به نه مليو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن خاتونِ جنّت، سيده فاطمة الزّهرا رَضِىَ اللهُ تَعَالٰی عَـنْها کي فرمايائون: هڪ ننڍڙو مون کي ڏيو۔ خاتونِ جنّت سيده فاطمه رَضِىَ اللهُ تَعَالٰی عَـنْها هڪ ٻار کي پردي مان ڏئي ڇڏيو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اُنهن کي کڻي پنهنجي سيني سان لڳايو پر اهو وڌيڪ اُڃ جي ڪري مُسلسل روئي رهيو هو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اُن جي وات ۾ پنهنجي زبان مُبارڪ وڌي ته اهي ان کي چُوسڻ لڳا ايستائين جو سيراب ٿي ويا، (حضرت سيدنا ابُوهريره رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته) مون ٻيهر انهن جي روئڻ جو آواز نه ٻڌو، جڏهن ته ٻيا (شهزاده) اهڙي طرح مُسلسل روئي رهيا هئا، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: ”ٻئي کي به مون کي ڏيو۔“ خاتونِ جنّت سيده فاطمه رَضِىَ اللهُ تَعَالٰی عَـنْها ٻئي کي به حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي حوالي ڪري ڇڏيو، حضور نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم انهن سان به ساڳيو ئي مُعاملو فرمايو، (يعني ان جي وات ۾به پنهنجي مُبارڪ زبان وڌي ته اهي به سيراب ٿي خاموش ٿي ويا) ان کان پوءِ ٻئي شهزادا اهڙا ماٺ ٿيا جو مون ٻنهن جو ٻيهر روئڻ جو آواز نه ٻڌو۔ حضرت سيدنا ابُو هُرَيره رَضِىَ اللهُ تَعَالٰی عَـنْهُ مزيد فرمائن ٿا: مان انهن سان ڇو نه مَحَبَّت ڪيان جڏهن ته مون نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي انهن سان ائين ڪندي ڏٺو آهي۔ (معجم ڪبير ،بقية اخبارحسن بن علي بن أ بي طالب، ۳/۵۰،حديث:۲۶۵۶)

مٺا مٺا اسلامي ڀائرو! اوهان ڏٺو ته صحابه ڪرام رَضِىَ اللهُ تَعَالٰی عَـنْهُم جي شان ۾ ٿوري گُستاخي ڪرڻ واري جو اَنجام ڪيترو خطرناڪ ۽ عبرتناڪ ٿيو، واقعي اها حقيقت آهي ته الله وارن جي شان ۾ گستاخي ڪرڻ يا انهن کي ڪنهن به طرح سان تڪليف پهچائڻ، الله عَزَّ وَجَلَّ جي عذاب کي دعوت ڏيڻ واري ڳالھ آهي۔ انهن بُزرگن جي شان ۾ گُستاخي ۽ بي ادبي ڪرڻ وارا دُنيا ۾ ته ذليل ۽ رُسوا ٿيندو ئي آهي آخرت ۾ به ذِلّت اُن جو مُقدَّر ٿيندي۔ يزيد پليد به اهو بد نصيب شخص آهي جنهن نه رڳو صحابه ڪرام رَضِىَ اللهُ تَعَالٰی عَـنْهُم جي توهين جو مُرتڪب ٿيو بلڪ ان جي پيشاني تي اهلِ بَيتِ ڪرام عَلَيهمُ الرِّضْوَان کي بي گُناھ شهيد ڪرڻ جو ڪارو داغ (به) آهي ۽ جنهن جي هر زماني ۾ اسلامي دُنيا مذمّت ڪندي رهي آهي ۽ قيامت تائين ان جو نالو حَقارت (ذِلَّت) سان ورتو ويندو۔ اهو بَد باطن، سِياه دل (شخص) 25 هجري ۾ دَمِشق ۾ پيدا ٿيو۔ نهايت ٿلهو، بدنُما، بداَخلاق، بد مِزاج، فاسِق، فاجِر، شرابي، بدڪار، ظالِم، بي اَدب، گُستاخ هو۔ ان جون شرارتون ۽ بريون حرڪتون اهڙيون هيون جو جن سان بدمُعاشن کي به شرم ٿيندو۔

(خلاصه ازسوانحِ ڪربلا،ص۱۱۱،ملتقطاً)

صَدْرُ الْافَاضِل حضرت علّامه مولانا مُفتي سَيد محمد نعيمُ الدّين مُراد آبادي رَحْمَة اللّٰه ِتَعَالٰي عَلَيه ارشاد فرمائن ٿا ته شهزاده ڪونين، حضرت امام حُسَين رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو وُجُودِ مُبارَڪ يزيد جي آزادين جي لاءِ هڪ زبردست مُحتَسِب (يعني حساب وٺڻ وارو) هو، ان کي خبر هئي ته پاڻ (رَضِىَ اللهُ تَعَالٰی عَـنْهُ) جي مُبارَڪ وقت ۾ اُن کي کُلي بُرا فعل ڪرڻ جو موقعو نه ملندو ۽ ان جي ڪنهن به غلط حرڪت ۽ گمراهي تي حضرت امام حسين رَضِىَ اللهُ تَعَالٰی عَـنْهُ صبر نه فرمائيندا، ان کي نظر ايندو هو ته امام (حسين) رَضِىَ اللهُ تَعَالٰی عَـنْهُ جهڙي دِيندار جو ڪوڙو هر وقت ان جي مٿي تي وسي رهيو هو، انهيء ڪري اهو ويتر وڌيڪ حضرت امام (حسين) رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي جان جو دُشمن هو ۽ ان ڪري حضرت امام (حسين) رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي شهادت ان جي لاءِ خوشي جو باعث ٿي۔

حضرت امام (حسين) رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي شهادت ٿي، يزيد بلڪل آزاد ٿي ويو ۽ مختلف قسمن جا گُناھ عام ٿي ويا۔ حرام ڪاري، ڀاءُ ڀيڻ جو نڪاح، وياج، شراب، اعلانيه رائج ٿي ويو، نمازن جي پابندي کڄي وئي، بَغاوت ۽ سرڪشي اِنتها تائين پهتي، خَباثت ايتري ته وڌي وئي جو مُسلِم بن عُقبه کي ٻارهن هزار (12000) يا ويھ هزار (20,000) جو لشڪر ڏئي مدينه طَيبَه تي چڙھائي جي لاءِ موڪليو۔ ان نامُراد لشڪر مدينه طَيبَه (زَادَها اللّٰه شَرَفًاوَّتَعْظِيماً) ۾ اهو طُوفان برپا ڪيو جو الله جي پناھ! قتل وغارت ۽ طرح طرح جا ظلم، رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پاڙي وارن تي ڪيا، اتي رهڻ وارن جي گھرن ۾ لُٽ مار ڪئي، ست سو (700) صحابه عَلَيهمُ الرِّضْوَان کي شهيد ڪيو ۽ ٻين عام رهواسين کي مِلائي ڏھ هزار (10,000) کان وڌيڪ کي شهيد ڪيو، ڇوڪرن کي قيدي ڪري ورتو، اهڙيون اهڙيون بد تميزيون ڪيون،

جن جو ذڪر ڪرڻ ناگوار آهي۔ مسجدِ نَبَوِي شريف (عَلٰي صَاحِبِھَاالصَّلٰوة وَالسَّلَام) جي سُتونن ۾ گھوڙا ٻڌيا، ٽن ڏينهن تائين مسجد شريف ۾ ماڻهو نماز نه پڙهي سگھيا۔ صرف حضرت سَعِيد بن مُسَيب رَضِىَ اللهُ تَعَالٰی عَـنْهُ دِيوانو بڻجي اتي حاضر رهيا۔ حضرت سَيدُنا عبدُالله بن حَنظَلَه رَضِىَ اللهُ تَعَالٰی عَـنْهُم ا جن فرمايو ته يزيدين جون بُريون حرڪتون ايتري حد تائين پهتيون جو اسان کي انديشو ٿيڻ لڳو ته انهن جي بدڪارين جي ڪري ڪٿي اسان تي آسمان تان پٿر نه وسن۔ (الصواعق المحرقه ،الباب الحادي عشر،الخاتمه في بيان اعتقاد اهل السنة، ص۲۲۱ملخصاً) پوءِ اهو شرير لشڪر، مَڪه مُڪرَّمَه ڏانهن روانو ٿيو، راھ ۾ لشڪر جو امير مري ويو ۽ ٻيو شخص اُن جو قائم مقام ڪيو ويو۔ مَڪه مُعَظَّمَه (زَادَها اللّٰه شَرَفًاوَّتَعْظِيماً) پهچي انهن بي دِينن مِنْجَنِيق (اهو پتّھر اڇلائڻ جو آله هوندو آهي جنهن سان پٿر اڇلايو ويندو آهي ان جي زَد وڏي زبردست ۽ پري تائين مار هوندي آهي) سان پٿرن جي بارش ڪئي۔ ان سنگ باري سان حَرَم شريف جو مُبارَڪ صحن پٿرن سان ڀرجي ويو ۽ مسجد ِحرام جا سُتون ٽُٽي پيا ۽ ڪعبه مُقَدَّسه جي غِلاف شريف ۽ ڇت کي انهن بي دِينن ساڙي ڇڏيو۔ ان ڇت ۾ ان گھٽي (دُنبي) جا سڱ به تَبَرُّڪ طور تي محفوظ هئا، جيڪي حضرت سَيدُنا اسمٰعيل عَلٰي نَبِينَا وَعَلَيه الصَّلٰوة وَالسَّلَام جي فِديي ۾ قُربان ڪيو ويو هو، اهي (سڱ) به سڙي ويا، ڪعبو شريف ڪئي ڏينهن تائين بي لباس رهيو ۽ اتي جا رهواسي (يزيدي لشڪر جي طرفان پهچڻ واري) سخت مُصيبت ۾ مبُتلا رهيا۔

(خلاصه سوانح ڪربلا،ص ۱۷۷ تا۱۷۹ ،ملخصاً)

مٺا مٺا اسلامي ڀائرو! اوهان ٻڌو ته يزيد پليد پنهنجي دَورِ حڪومت ۾ بدڪرداري جو ثُبُوت ڏيندي ڪئي برائين کي عام ڪيو مثال طور، مَحارِم (اهي رشتيدار جن سان نڪاح حرام آهي ان) سان نڪاح ۽ وياج وغيره

کي ان بي دِين عَلانِيه عام ڪيو، مدينه طيبه ومڪه مڪرمه جي بي حُرمتي ڪرائي، صحابَه ڪرام ۽ اهلِ بيتِ اَطهار کي ظُلم وسِتم کان پوءِ شهيد ڪرايو، گانا باجا ٻڌڻ، خُوب شراب پيئڻ، نماز نه پڙھڻ، الغرض اهو بد بخت هر اهو ڪم ڪندو هو، جنهن کان شريعت سڳوري منع ڪيو هو۔ يقيناً انسان عزّت وشهرت، مال ودولت ۽ حڪومت ملڻ سان سرڪش ۽ بي باڪ بڻجي ويندو آهي ۽ پوءِ آهسته آهسته دين کان پري ۽ دُنيا کي گھڻو ويجھو ٿي ويندو آهي، پنهنجي تخت ۽ تاج جي لاج رکڻ جي لاءِ الله عَزَّ وَجَلَّ جي نافرمانين ۽ ان جي پيارن جي گُستاخين کان به ناهي مڙندو۔ يزيد پليد به اِقتِدار و حڪومت جي مَحَبَّت ۾ مبُتلا ٿي اهڙو سَرڪش ٿيو جو مَعَاذَ الله عَزَّوَجَلَّ صحابه ڪرام عَلَيهمُ الرِّضْوَان جي شان ۾ گستاخيون ڪيون، انهن کي تڪليفون ڏنيون، انهن جي دل ڏکائي حالانڪ صحابه ڪرام عَلَيهمُ الرِّضْوَان ته اهي مقدّس هستيون آهن، جن جي شان ۽ عظمت جي باري ۾ خود سُلطانِ دو جهان، رَحمتِ عالميان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: اَكْرِمُوا اَصْحَابِي فَاِنَّهُمْ خِيَارُكُمْ يعني منهنجي صحابه (عَلَيهمُ الرِّضْوَان) جي عزّت ڪيو ڇو ته اهي توهان مان وڌيڪ نيڪ آهن۔ (مشڪاة المصابيح، ڪتاب المناقب، باب مناقبالصحابة، ۲/۴۱۳،حديث: ۶۰۱۲) مزيد فرمايائون: ” خَيْرُ اُمَّتِي اَلْقَرْنُ الَّذِينَ يَلُوْنِي ثُمَّ الَّذِينَ يَلُوْنَهُمْ ثُمَّ الَّذِينَ يَلُوْنَهُمْ يعني منهنجي اُمّت مان سڀ کان بهتر منهنجي زماني وارا آهن (يعني صحابه ڪرام عَلَيهمُ الرِّضْوَان) پوءِ ان کان بعد وارا (يعني تابعين) پوءِ بعد وارا آهن (تبعِ تابعين رَحْمَة الله تَعاليٰ عَلَيهمْ اَجْمَعِين ۔) “ (مسلم،ڪتاب فضائل الصحابه باب فضائل الصحابه...الخ ،ص،حديث: ۱۳۷۲)

نمایاں ہے اسلام کے گلستاں میں

ہر اک گل پہ رنگِ بہارِ صحابہ

یہ مہریں ہیں فرمان ِخَتمُ الرُّسُل کی

ہے دِینِ خدا شاہکارِ صحابہ

انہیں میں ہیں بدر واُحد کے مجاہد

لقب جن کا ہے جاں نثارِ صحابہ

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! يزيد پليد صحابه ڪرام عَلَيهمُ الرِّضْوَان جو مُبارڪ زمانو ڏٺو پوءِ به انهن جو قُرب حاصل نه ڪري سگھيو ۽ نه ئي انهن جي تعظيم ڪري پنهنجي لاءِ آخرت جي نجات جو سامان ڪري سگھيو بلڪ ان رَسُولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پاڪ گھراڻي تي تمام گھڻو ظُلم و سِتم ڪيو ۽ انهن تي مصيبتن جو سبب بڻيو۔ يزيد پليد اهو سڀ تاج و حڪومت ۽ دُنياوي مال ۽ دولت جي مَحَبَّت ۾ ڪيو، ڇوته ان ظالمِ بَد اَنجام کي امامِ عالي مقام، سَيدُنا امام حسين رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي ذاتِ گرامي مان پنهنجي اِقتِدار بابت خطرو محسوس ٿيندو هو حالانڪ حضرت سَيدُنا امام حسين رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي ناپائيدار دنيا جي ڪهڙي ضرورت! پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ته ڪلھ به اُمَّتِ مُسلمه جي دِلين جا تاجدار هئا، اڄ به آهن ۽ باقي دُنيا تائين رهندا۔ پر ان يزيد بدبخت مال ودولت جي نَشي ۾ بدمست ٿي پنهنجي آخرت سان گڏوگڏ دُنيا به تباھ وبرباد ڪري ڇڏي۔ واقعي دُنيا جي مَحَبَّت هر فِتني فساد جو سبب آهي، اها سڄي تباهي دُنيا جي مَحَبَّت ئي مچائي آهي، دُنيا جي مَحَبَّت انسان کي ظالم بڻائيندي آهي، دُنيا جي مَحَبَّت انسان کي بي حِس و بيباڪ بڻائيندي آهي، دُنيا جي مَحَبَّت انسان کي پٿر دِل بڻائيندي آهي، دُنيا جي مَحَبَّت عملن کي ضائع ڪندي آهي، دُنيا جي مَحَبَّت دِين کي نُقصان پهچائڻ جو سبب آهي، دُنيا جي مَحَبَّت گمراهي جو سبب آهي، دُنيا جي مَحَبَّت انسان کي نيڪين کان پري ڪندي آهي، دُنيا جي مَحَبَّت ئي گُناهن تي دلير ڪندي آهي۔ اچو دُنيا جي مذمّت تي 3 فرامينِ مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌون ٿا:

1. دُنيا جي مَحَبَّت سمورن گُناهن جي جڙ آهي۔ (ڪتاب ذم الدنيا مع موسوعة الامام ابن ابي الدنيا ج۵ ص۲۲ حديث۹)

1. ڇھ (6) شيون عمل کي ضائع ڪنديون آهن: (۱) مخلوق جي عيبن جي ڳولا ۾ مصروف رهڻ (۲) دل جي سختي (۳) دُنيا جي مَحَبَّت (۴) حَياء جي گھٽتائي (۵) ڊگھي اُميد ۽ (۶) حد کان وڌيڪ ظُلم۔'' (ڪنزالعمال، ۸/۳۶،جزء ۱۶،الحديث:۴۴۰۱۶)

2. 'ٻن بکايل بگھڙن کي جيڪڏهن ٻڪرين جي ڌڻ ۾ ڇڏيو وڃي ته ايترو نُقصان نٿا پهچائن جيترو مال ۽ دولت جو حِرص (لالچ) ۽ دُنيا جي مَحَبَّت انسان جي دِين کي نُقصان پهچائيندا آهن۔ (جامع الترمذي ،ڪتاب الزھد،باب حديث ماذئبان جائعان،الحديث:۲۳۷۶،ص۱۶۶)
