صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مان به نيت ڪيان ٿو * الله جي رضا حاصل ڪرڻ ۽ ثواب حاصل ڪرڻ جي لاءِ بيان ڪندس * ڏسي بيان ڪندس * سيپاري 14 سورة النحل آيت نمبر 125: اُدۡعُ اِلٰی سَبِیۡلِ رَبِّکَ بِا لۡحِکۡمَۃِ وَ الۡمَوۡ عِظَۃِ الۡحَسَنَۃِ ” ترجمو ڪنزالايمان: پنهنجي ربّ جي واٽ جي طرف سڏ پڪي سمجهه ۽ چڱي نصيحت سا ن“ ۽ بخاري شريف (حديث: 4361) جي هن فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيهِ وَاٰلِهٖ وَسَلَّم: بَلِّغُوۡا عَنِّيۡ وَلَوۡ آيَة [1] يعني ” منهنجي طرفان پهچايو جيتوڻيڪ هڪ ئي آيت هجي“ ۾ ڏنل احڪام جي پيروي ڪندس * نيڪي جو حڪم ڏيندس ۽ بُرائي کان منع ڪندس * شعر پڙهڻ، عربي، انگريزي ۽ ڏکيا لفظ ڳالهائڻ دؤران دل جي اخلاص تي توجُّھ رکندس يعني پنهنجي علم جو رُعب ويهارڻ مقصود ٿيو ته ڳالهائڻ کان بچندس * مدني قافلي، مدني انعامات ۽ علاقائي دوره براءِ نيڪي جي دعوت وغيره جي رغبت ڏياريندس * ٽهڪ ڏيڻ ۽ ڏيارڻ کان بچندس * نظر جي حفاظت بڻائڻ جي خاطر حتي الامڪان نگاهون جهڪائي رکندس۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

حضرتِ سيدنا ابي بن ڪعب رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته حضورِ انور، شافع محشر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو. مون وٽ حضرتِ

جبريل عَـلَيْـهِ الـسَّـلاَم شبِ برات ۾حاضر ٿيا ۽ مون کي چيائون اٿو نماز ادا فرمايو. مون پڇيو: اي جبرائيل! هي ڪهڙي رات آهي؟ عرض ڪيو: اي محمد صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! هي اها رات آهي جنهن ۾ آسمان ۽ رحمت جا 300 دروازا کوليا ويندا آهن، الله عَزَّوَجَلَّ سان گڏ شريڪ ڪرڻ وارا، پاڻ ۾ بغض ۽ ڪينو رکڻ وارا، شرابين ۽ بدڪارن کان علاوه سڀني جي مغفرت ڪئي ويندي آهي. انهن ماڻهن جي ان وقت تائين مغفرت نه ٿيندي جيسيتائين اهي سچي توبهه نه ڪن. البته شراب جي عادي جي لاءِ رحمت جي دروازن مان هڪ کوليو ويندو آهي ايسيتائين جو اهو توبهه ڪري، ته جڏهن اهو توبهه ڪري ڇڏيندو آهي ته ان جي مغفرت ڪئي ويندي آهي. اهڙي طرح ڪيني رکڻ واري جي لاءِ به رحمت جي دروازن مان هڪ دروازو کوليو ويندو آهي ايستائين جو اهو پنهنجي ساٿي (يعني جنهن سان ڪينو رکندو آهي) سان ڪلام نه ڪري ڇڏي، جڏهن اهو ان سان ڪلام ڪري ڇڏيندو آهي ته ان جي به مغفرت ڪئي ويندي آهي، حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: اي جبرائيل! جيڪڏهن اهو پنهنجي ساٿي سان ڪلام نه ڪري، ايستائين جو شبِ براءت گذري وڃي ته.........؟ حضرت سيدنا جبرائيل عَـلَيْـهِ الـسَّـلاَم عرض ڪيو: جيڪڏهن اهو انهيءِ حالت تي برقرار رهيو ايستائين (نزع جو عالم طاري ٿيڻ جي سبب) ان جي سيني ۾ ساهه اٽڪڻ لڳي ته ان جي لاءِ به رحمت جو دروازو کليل رهندو آهي، جيڪڏهن اهو (موت کان پهريان مسلم جي ڪيني کان) توبهه ڪري ته ان جي توبهه مقبول آهي، هي ٻڌي ڪري حضورِ انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جنتُ البقيع جي طرف تشريف کڻي ويا ۽ سجدي ۾ وڃي انهن الفاظن ۾ دعا گهري: اَعُوْذُ بِعَفْوِكَ مِنْ عِقَابِكَ وَاَعُوْذُ بِرِضَاكَ مِنْ سَخَطِكَ وَاَعُوْذُ بِكَ مِنْكَ جَلَّ ثَنَاؤُكَ لَا

اَبْلُغُ الثَّنَاءَ عَلَيْكَ اَنْتَ كَمَا اَثْنَيْتَ عَلٰى نَفْسِكَ (يعني اي الله عَزَّوَجَلَّ) مان تنهنجي عذاب کان تنهنجي معافي، تنهنجي ناراضي کان تنهنجي رضا ۽ تون کان تنهنجي پناهه طلب ڪريان ٿو، تنهنجي تعريف بلند آهي، مان تنهنجي تعريف جو حق ادا نه ٿو ڪري سگهان، تنهنجو حقيقي شان اهو ئي آهي جيڪو تو پاڻ بيان فرمايو.

حضرت سيدنا جبرائيل عَـلَيْـهِ الـسَّـلاَم رات جي چوٿين پهر نازل ٿيا ۽ عرض ڪيائون: اي محمد صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! پنهنجو سِر مبارڪ آسمان جي طرف مٿي کڻو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجو سر مبارڪ آسمان جي طرف مٿي کنيو ته ڏٺائون ته رحمت جا دروازه کليل آهن. پهرين دروازي تي هڪ فرشتو هيئن پڪاري رهيو هو ته مبارڪ هجي ان کي جيڪو هن رات عبادت ڪري، ٻئي دروازي تي هڪ فرشتو هيئن پڪاري رهيو ته مبارڪ هجي ان کي جيڪو هن رات سجدو ڪري، ٽيئن دروازي تي هڪ فرشتو ندا ڪري رهيو هو مبارڪ هجي ان کي جيڪو هن رات رڪوع ڪري چوٿين دروازي تي هڪ فرشتو پڪاري رهيو هو مبارڪ هجي ان کي جيڪو هن رات ۾ پنهنجي رب عَزَّوَجَلَّ کان دعا گهري، پنجين دروازي تي به هڪ فرشتو هيئن پڪاري رهيو هو ته مبارڪ هجي ان کي جيڪو هن رات پنهنجي رب تعالٰي سان مناجات ڪرڻ ۾ مشغول هجي ڇهين دروازي تي به هڪ فرشتو پڪاري رهيو هو ته هن رات ۾ مسلمانن کي مبارڪ هجي ستين دروازي تي به هڪ فرشتو پڪاري رهيو هو ته خدا تعالٰي کي هڪ مڃڻ وارن کي مبارڪ هجي، اٺين دروازي تي به هڪ فرشتو

پڪاري رهيو هو ته آهي ڪو مغفرت طلب ڪرڻ وارو ته ان جي مغفرت ڪئي وڃي؟ ۽ ڏهين دروازي تي به هڪ فرشتو پڪاري رهيو آهي ته آهي ڪو دعا ڪرڻ وارو ته ان جي دعا قبول ڪئي وڃي؟ پوءِ رحمتِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: اي جبرائيل! رحمت جا هي دروازا ڪيستائين کليل رهندا آهن؟ عرض ڪيائون: رات جي ابتداء کان طلوعِ فجر تائين، پياري آقا محمد مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ان رات ٻڪرين جي وارن کان به وڌيڪ مغفرت ڪئي ويندي آهي، انهيء رات ماڻهن جي سڄي سال جا اعمال (آسمانن جي طرف) بلند ڪيا ويندا آهن ۽ انهيء رات رزق تقسيم ڪيو ويندو آهي.

مہہِ شعبان کے صدقے میں کر فضل و کرم مولیٰ

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

سبحٰن الله عَزَّوَجَلَّ! مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان ته الله عَزَّوَجَلَّ شعبانُ المعظم جي مهيني کي ڪيتري قدر شرافت ۽ بزرگي عطا فرمائي آهي بالخصوص 15 شعبانُ المعظم جي نوراني رات (يعني شبِ براءَت) ته ايتري قدر عظيم الشان رات آهي جو بارگاهِ خداونديءَ عَزَّوَجَلَّ مان مغفرت جا پروانا تقسيم ڪيا ويندا آهن، جڏهن هيءَ مبارڪ رات تشريف کڻي ايندي آهي ته رحمت جا دريا ايتري قدر جوش ۾ ايندا آهن ته ربّ ڪائنات عَزَّوَجَلَّ پنهنجي ٻانهن جي لاءِ رحمت جا 300 دروازا کولي ڇڏيندو آهي ۽ الله عَزَّوَجَلَّ جا معصوم فرشتا ان رات رڪوع ۽ سجدا ۽ نفلي عبادت ڪرڻ دعائن ۾ مشعول رهڻ ۽ گريه وزاري ڪرڻ

وارن کي خوشخبريون ٻڌايون ٿا هن عظيم رات جي عظمت جي سبب کان نبي رحمت شفيع امت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم 15 شعبان المعظم جي رات ڪڏهن جنت البقيع جي قبرستان تشريف کڻي ويندا ۽ اُتي بارگاھِ خداوندي عَزَّوَجَلَّ ۾ سجدي جي حالت ۾ گريه وزاري فرمائيندا ۽ ڪڏهن گهر مبارڪ ۾ نفلن ۽ دعا وغيرا ۾ مشغول رهندا هئا. جيئن ته

اُمّ الۡمُؤمِنِين حضرتِ سيدتنا عائشه صديقه، طيبه طاهِره رَضِیَ اللهُ تَعَالٰی عَنْهَا فرمائن ٿيون: كَانَ رَسُوْلُ اللهِ صَلَّی اللّٰہُ عَلَیْہِ وَسَلَّمَ يَدْعُوْ وَهُوَ سَاجِدٌ لَيْلَةَ النِّصْفِ مِنْ شَعْبَانَ يعني نصف شعبان جي رات منهنجي سر تاج، صاحبِ معراج صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سجدي جي حالت ۾ دعائون گهرندا هئا.

مٺا مٺا اسلامي ڀائرو! ياد رکو ته مالڪِ ڪونين، رحمتِ دارين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو ان مقدّس رات ۾ خصوصيت سان گڏ نفل ادا ڪرڻ ۽ الله عَزَّوَجَلَّ جي عذاب ۽ ان جي ناراضگيءَ کان پناھ طلب ڪرڻ تعليمِ اُمّت جي لاءِ هو ته جيئن پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اُمّت به پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي پيروي ڪندي ان رات ۾ وڌ کان وڌ عبادتون ڪن ۽ الله عَزَّوَجَلَّ جي بارگاھ مان ملڻ وارن انعامن ۽ رحمتن جي فيضان کان بالڪل محروم نه رهن. پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو توبھ ۽ استغفار فرمائڻ ۽ رب تعالٰي جي ناراضگيءَ ۽ ان جي غضب کان پناھ گُهرڻ هر گز پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان معاذالله عَزَّوَجَلَّ گناهن جي صادر ٿيڻ سبب نه هيو ڇو ته صرف نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پر تمام انبياءِ ڪرام عَـلَيْهِمُ الرِّضْوَانْ الله جي فضل ۽ ڪرم سان گناهن کان معصوم (يعني گناهن کان پاڪ) آهن جو انهن کان گناه ٿي ئي نٿا سگهن جيئن ته

شارِحِ بُخاري حضرت علامه مفتي شريف الحق امجدي عَلَیهِ رَحْمَةُ اﷲِ القَوِی ارشاد فرمائين ٿا: سڀ انبياءِ ڪرام عَـلَيْهِمُ الرِّضْوَانْ بالخصوص اسان جي حضورِاقدس صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تمام گناهن کان معصوم آهن . (فتاويٰ شارِح بُخاري، 363/1) اهڙي طرح مفسّرِ شهير حڪيم الامّت مفتي احمد يار خان عَـلَيْهِ رَحْـمَةُ الْـمَنَّانْ فرمائن ٿا: حضراتِ انبياءِ گناهن کان معصوم آهن جو گناھ ڪري ئي نٿا سگهن.

مٺا مٺا اسلامي ڀائرو! يقيناً اسان سڀ وڏا خوش قسمت آهيون جو الله تعالٰي اسان کي پنهنجي محبوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اُمّت ۾ پيدا فرمائي ڪري اسلام جي دولت سان مالامال فرمايو ۽ پنهنجي محبوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي صدقي اسان کي شعبان المعظم جهڙو برڪتن سان مالا مال مُبارڪ مهينو عطا فرمايو، تنهنڪري اسان کي گهرجي ته فرضن ۽ واجبن جي پابنديءَ سان گڏ و گڏ ان مهيني ۾ عام ڏينهن جي مقابلي ۾ ذڪر ۽ درود ، تلاوتِ قرآنِ ڪريم، هفتي وار سنّتن ڀريواجتماع ۽ هفتي وار مدني مذاڪري ۾ شرڪت، اميرِ اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ ۽ دعوتِ اسلامي جي اشاعتي اداري مڪتبةالمدينه جي شائع ٿيل ڪتابن ۽ رسالن جو مطالعو، صدقو ۽ خيرات، تقسيمِ رسائل، مرحومن جي لاءِ ايصالِ ثواب ۽ دعاءِ مغفرت ڪرڻ جي لاءِ سنت جي نيّت سان قبرستان جي حاضري ۽ ٻي نفلي عبادت جو به گهڻي کان گهڻو اهتمام ڪيون، هونئن به هي مبارڪ مهينو اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو پسنديده ۽ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي گهڻي کان گهڻا درودِ پاڪ پڙهڻ جو مهينو آهي. جيئن ته

غنية الطالبين ۾ آهي ته شعبان المعظم ۾ سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن تي درودِ پاڪ جي ڪثرت ڪئي ويندي آهي ۽ هي نبيءِ مختار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي درود موڪلڻ جو مهينو آهي. (غنية الطالبين، القسم الثالث: مجالس في مواعظ القرآن..... الخ، 342) تنهنڪري ان مهيني ۾ ڪثرت سان درودِ پاڪ پڙهڻ گهرجي. اچو درودِ پاڪ جي عادت ٺاهڻ جي لاءِ هڪ حڪايت ٻڌون ٿا. جيئن ته

حجّةالاسلام حضرت سيدنا امام محمد غزالي عَلَیۡهِ رَحْمَةُ اﷲِ الۡوَالِی ارشاد فرمائن ٿا: هڪ ماڻهو حُضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي درود شريف نه پڙهندو هو، هڪ رات خواب ۾ زيارت سان مشرّف ٿيو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن جي طرف توجهه نه فرمائي، هُن عرض ڪيو: ”اي الله عَزَّوَجَلَّ جا رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! ڇا توهان مون کان ناراض آهيو؟ ”فرمايائون“ نه. هن شخص پڇيو: پوءِ توهان منهنجي طرف توجهه ڇو نه ٿا فرمايو؟ ”فرمايائون“ ”هن لاءِ جو آئون توکي نٿو سڃاڻان“. هن شخص عرض ڪيو: ”حضور! توهان مونکي ڪيئن نٿا سڃاڻون، آئون ته توهان جي امّت جو هڪ فرد آهيان. ”۽ عُلماء فرمائيندا آهن ته توهان پنهنجي امتين کي ان کان به وڌيڪ سڃاڻيندا آهيو جيئن ڪا ماءُ پنهنجي پُٽ کي سڃاڻيندي آهي. پاڻ فرمايائون: ”عُلماء سچ چيو، پر تون مونکي درود شريف جي ذريعي ياد نه ٿو ڪرين ۽ آئون پنهنجي اُمّت جي ماڻهن کي درود پاڪ پڙهڻ جي سبب سان سڃاڻيندو آهيان، جيتري قدر اُهي مون تي درود پڙهندا آهن ايتري قدر آئون انهن کي سڃاڻيندو آهيان“.

جڏهن اُهو شخص سڄاڳ ٿيو ته هن پنهنجي مٿان لازم ڪيو ته حضور سرورِ ڪائنات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي روزانو هڪ سؤ ڀيرا درود پاڪ پڙهندس، هاڻي هن شخص روزانو سؤ ڀيرا درود پاڪ پڙهڻ پنهنجو معمول بڻائي ڇڏيو. ڪجهه عرصي بعد وري حضور اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ديدار سان مُشرّف ٿيو، پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن فرمايو: مون توکي هاڻي سڃاتو آهي ۽ آئون تنهنجي شَفاعت به فرمائيندس.

(مڪاشفة القلوب، ص 79 ملخصاً)

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته درود پاڪ پڙهڻ واري کان نبّي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم نه صرف خوش ٿيندا آهن بلڪه شربتِ ديدار سان به نوازيندا آهن. تنهنڪري اسان به جيڪڏهن نبيءِ اڪرم، نورِ مجسّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي رِضا چاهيون ٿا ۽ حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زيارت جا خواهشمند آهيون ته اسان کي ضبح ۽ شام جو وظيفو بنائڻ گهرجي، سچي لگن سان گڏ ان ۾ مگن رهندا سين ته اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ هڪ نه هڪ ڏينهن ضرور اسان تي ڪرم ٿيندو ۽ اسان کي به زيارت نصيب ٿي ويندي.

اعلٰي حضرت، امام اهلسنّت مولانا شاهه امام احمد رضا خان عَـلَيْهِ رَحْمَةُ الـرَّحْمٰن ”الوظيفة الڪريمہ“ ۾ لکن ٿا (درودِ پاڪ) خالص تعظيمِ شانِ اَقدس جي لاءِ پڙهي، هن نيّت کي به (دل ۾) جڳهه نه ڏي ته مونکي زيارت عطا ٿئي، اڳتي انهن جو ڪرم تمام وسيع ۽ بي انتها آهي. منهن مدينهء طيّبه جي طرف هجي ۽ دل حُضورِ اَقدس صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي طرف،

دَسۡت بَسۡته (هٿ ٻڌي ڪري) پڙهي ۽ هي تصوّر ڪري ته روضهء انور تي حاضر آهيان ۽ يقين ڄاڻي ته حضور انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مونکي ڏسي رهيا آهن، منهنجي آواز ٻڌي رهيا آهن، ۽ منهنجي دل جي خيالن کان به آگاھ آهن. (الوضيفته الڪريمه، ص 28)

مٺا مٺا اسلامي ڀائرو! جيڪڏهن اسان به اعلٰي حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي ٻڌايل طريقن تي عمل ڪندي اخلاص ۽ استقامت سان درود پاڪ پڙهڻ جي عادت بڻايون ته اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ ديدارِ مصطفٰي کان مُستفيض ٿيڻ سان گڏوگڏ دين ۽ دنيا جا بيشمار فائدا حاصل ٿيندا. اچو ترغيب جي لاءِ درود پاڪ جا وڌيڪ فضائل ٻڌون ٿا جيئن ته

حضرتِ سَيّدُنا شيخ عبدالحق مُحدّث دهلوي عَـلَيْهِ رَحْـمَةُ الـلّٰـهِ الۡـقَـوِی ”جذبُ القُلوب“ ۾ ارشاد فرمائن ٿا: جڏهن مؤمن ٻانهو هڪ ڀيري درود شريف پڙهندو آهي ته الله (عَزَّوَجَلَّ) هن تي ڏهه (10) ڀيرا رحمت موڪليندو آهي، (ڏهه (10) درجا بلند ڪندو آهي، ڏهه (10) نيڪيون عطا فرمائيندو آهي، ڏهه (10) غلام آزاد ڪرڻ جو ثواب ۽ ويھ (20) غزوات (جنگين) ۾ شامل ٿيڻ جو ثواب عطا فرمائيندو آهي درودِ پاڪ دعا جي قبوليت جو سبب آهي، هن جي پڙهڻ سان شفاعتِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم واجب ٿي ويندي آهي. بابِ جنَّت وٽ مصطفٰي جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو قُرب نصيب ٿيندو، درود پاڪ تمام پريشانين کي دور ڪرڻ جي لاءِ ۽ تمام حاجتن کي پورو ڪرڻ جي لاءِ ڪافي آهي، دُرود پاڪ گناهن جو ڪفارو آهي، صدقي جي قائم مقام بلڪ صدقي کان به افضل آهي. وڌيڪ فرمائن

ٿا: درود شريف سان مصيبتون ٽرنديون آهن، بيمارين کان شفاء حاصل ٿيندي آهي، خوف دور ٿيندو آهي، ظُلم کان نجات حاصل ٿيندي آهي، دشمن تي فتح حاصل ٿيندي آهي، الله (عَزَّوَجَلَّ) جي رِضا حاصل ٿيندي آهي ۽ دل ۾ اُن جي محبّت پيدا ٿيندي آهي، فرشتا هن جو ذڪر ڪندا آهن، عملن جي تڪميل ٿيندي آهي ، دل ۽ جان ، اسباب ۽ مال جي پاڪائي حاصل ٿيندي آهي، پڙهڻ وارو خوشحال ٿي ويندو آهي، برڪتون حاصل ٿينديون آهن، اولاد در اولاد چئن (4) نسلن تائين برڪت رهندي آهي.

(جذب القلوب ص 229)

اهڙي طرح هڪ ٻي حديثِ پاڪ ۾ آهي: رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم شَعبان کان وڌيڪ ڪنهن مهيني ۾ روزا نه رکندا هئا، بلڪ پوري شعبان جا ئي روزا رکندا هئا ۽ فرمائيندا هئا: پنهنجي اِسۡتِطاعَت مطابق عمل ڪريو جو الله عَزَّوَجَلَّ ان وقت تائين پنهنجو فضل ناهي روڪيندو جيستائين توهان تنگ نه ٿيو .

(صحيح بخاري ج 1 ص 648 حديث 1970 دارالڪتب العلمية بيروت)

شارِحِ بُخاري حضرت علامه مفتي شريف الحق امجدي عَلَیهِ رَحْمَةُ اﷲِ القَوِی هن حديث پاڪ شرح ۽ وضاهت بيان فرمائن ٿا ان جو خلاصو آهي ته رسولِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم شَعبان جي اڪثر ڏينهن جا روزا رکندا هئا، انهيء کي غلبي ۽ زيادت جي لحاظ کان سڀ لفظ سان تَعبِير ڪيو آهي (يعني سڄي مهيني روزا رکڻ مراد ورتي وئي آهي) جيئن چوندا آهن: ”فلان سڄي رات عبادت ڪئي.“ جڏهن ته هن رات

جو ماني به کاڌي هوندي ۽ ضروري حاجتن کان به فراغت حاصل ڪئي هوندي، وڌيڪ فرمائن ٿا: هن حديث مان خبر پئي ته شَعبان ۾ جنهن کي طاقت هجي اهو وڌ کان وڌ روزا رکي. (نزهة القاري، ج. 3، ص. 380)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! شعبان المعظم اسان جي پياري آقا، مڪي مدني مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو پسنديده مهينو هو. جيئن ته اُمُّ المؤمنين حضرت عائشه صديقه رَضِیَ اللهُ تَعَالٰی عَنْهَا کان مروي آهي ته سرڪارِ نامدار، سُرور قلب و سينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو پسنديده مهينو شعبان المعظم هو ته ان ۾ روزا رکندا هئا پوءِ ان کي رمضان المبارڪ سان ملائي ڇڏيندا هئا. (ابو دائود، ج. 2، ص. 476، حديث 2431) پڻ نبي اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي طرف منسوب ڪندي هي به ارشاد فرمايو: شَعْبَانُ شَھْرِیْ وَرَمَضَانُ شَھْرُ اللہ يعني شعبان منهنجو مهينو آهي ۽ رمضان الله عَزَّوَجَلَّ جو مهينو آهي.

(جامع الصغير، حرف الشين، ص. 301، حديث 4889)

علامه عبد الرؤف مُناوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هن حديثِ پاڪ جي شرح ۾ فرمائن ٿا: رحمتِ عالم، نور مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن شعبان المعظم کي انهيءَ لاءِ پنهنجو مهينو فرمايو ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن مهيني ۾ (ڪثرت سان نفل) روزا رکندا هئا ڇاڪاڻ ته هي روزا پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي واجب نه هئا ۽ رمضان المبارڪ

کي انهيءِ لاءِ الله تعالٰي جو مهينو فرمايو ته الله تعاليٰ هن مهيني ۾ روزا فرض فرمايا آهن. (فيض القدير، ج. 4، ص. 213، حديث 4889)

مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان ته اسان جي بخشي بخشائي آقا، اسان گنهگارن کي بخشائڻ واري مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي عبادت جو عالم هي هو ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن مقدس مهيني جي اڪثر ڏينهن جا روزا رکندا هئا ۽ هڪ اسان آهيون جو اسان جي زندگي ۾ نه ڄاڻي ڪيترا ڀيرا شعبان المعظم جو مبارڪ مهينو تشريف کڻي آيو ۽ بخشش ۽ مغفرت جا پروانا تقسيم ڪندي رخصت ٿي ويو پر اسان هن مبارڪ مهيني ۾ پنهنجي گناهن کان توبهه ڪرڻ، آئنده گناهن کان بچڻ جو پڪو ارادو ڪرڻ، فرض نمازن جو اهتمام ڪرڻ، صدقو ۽ خيرات ڪرڻ، تلاوتِ قرآن ڪريم، ذڪر و درود، روزن ۽ ٻين نفلي عبادت جي ڪثرت ڪرڻ ۽ پنهنجي رب عَزَّوَجَلَّ کي راضي ڪرڻ ۾ ناڪام رهياسين. ڇاڪاڻ ته نبي ڪريم، رؤوف رحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن هن مهيني ۾ اسان کي ڪثرتِ عبادت جي ترغيب ڏيڻ جي لاءِ نه صرف عملي طور تي خود نفلي عبادت جو اهتمام فرمايو بلڪه هن مهيني جي اهميت بيان ڪندي ارشاد فرمايو ته رجب ۽ رمضان جي درميان ۾ هي مهينو آهي، ماڻهو ان کان غافل آهن. هي اهو مهينو آهي جنهن ۾ ماڻهن جا اعمال رب العٰلمين عَزَّوَجَلَّ جي (بارگاهه جي) طرف کنيا ويندا آهن ۽ مون کي اهو محبوب آهي ته منهنجو عمل هن حال ۾ اٿاريو وڃي جو مان روزيدار هجان. (نسائي، ص. 387، حديث 2354) ۽ بالخصوص 15 شعبان المعظم جي رات ۾ نفلي عبادت ڪرڻ ۽ ڏينهن ۾ روزو رکڻ جو باقاعده حڪم ڏيندي ارشاد

فرمايو: جڏهن شعبان المعظم جي رات اچي ته ان ۾ قيام (يعني عبادت) ڪريو ان ڏينهن ۾ روزو رکو جو الله عَزَّوَجَلَّ غروبِ آفتاب کان آسمانِ دنيا تي خاص تجلي فرمائيندو ۽ فرمائيندو آهي ته آهي ڪو مون کان مغفرت طلب ڪرڻ وارو ته ان کي بخشي ڇڏيان! آهي ڪو روزي طلب ڪرڻ وارو ته ان کي روزي ڏيان! آهي ڪو مصيبت زده ته ان کي عافيت عطا ڪريان! آهي ڪو اهڙو! آهي ڪو اهڙو! ۽ هي ان وقت تائين فرمائيندو آهي جيسيتائين فجر طلوع ٿي وڃي.

(ابن ماجه ڪتاب الصلاة، ج. 2، ص. 160، حديث 1388)

مٺا مٺا اسلامي ڀائرو! بيان ڪيل احاديث مبارڪه مان شعبان المعظم ۾ نفلي عبادت جي ترغيب سان گڏو گڏ هي به معلوم ٿيو ته هن مهيني ۾ سڄي سال جا اعمال الله عَزَّوَجَلَّ جي بارگاهه ۾ پيش ڪيا وڃن ٿا. ايئن سمجهو ته اعمال نامن جي تبديلي جي اعتبار سان 14 شعبان المعظم سال جو آخري ڏينهن ۽ 15 شعبان المعظم سال جو پهريون ڏينهن هوندو آهي، لهٰذا اسان کي گهرجي ته بالعموم شعبان المعظم جي سڄي مهيني ۽ بالخصوص 15 شعبان جو عبادت ۽ استغفار جي خوب ڪثرت ڪريون ۽ الله عَزَّوَجَلَّ کان پنهنجي دنيا ۽ آخرت جي بهتري جي لاءِ دعائون گهرون.

پاره 25 سورة الدخان جي آيت نمبر 3 ۽ 4 ۾ خداءِ حنّان و منّان عَزَّوَجَلَّ ارشاد فرمائي ٿو:

اِنَّاۤ اَنۡزَلۡنٰہُ فِیۡ لَیۡلَۃٍ مُّبٰرَکَۃٍ اِنَّا کُنَّا مُنۡذِرِیۡنَ ﴿۳﴾ فِیۡہَا یُفۡرَقُ کُلُّ اَمۡرٍ حَکِیۡمٍ ۙ﴿۴﴾

ترجمو ڪنز لايمان: بيشڪ اسان ان کي برڪت واري رات ۾ لاٿو بيشڪ اسان ڊيڄاريندڙ آهيون. ان ۾ ورهايو ويندو آهي هر حڪمت وارو ڪم .

مفسر شهير، حڪيم الامت مفتي احمد يار خان عَـلَيْهِ رَحْـمَةُ الْـمَنَّانْ بيان ڪيل آيتِ مبارڪه جي تحت فرمائن ٿا: هن رات مان مراد يا شبِ قدر آهي (يعني رمضان المبارڪ جي) 27 وين رات يا شبِ معراج يا شب براءت (يعني 15 وين) شعبان. هن رات ۾ پورو قرآن لوحِ محفوظ کان دنياوي آسمان جي طرف لاٿو ويو پوءِ اتان کان 23 سال جي عرصي ۾ ٿورو ٿورو حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي نازل ٿيو. هن آيت مان معلوم ٿيو ته جنهن رات ۾ قرآن نازل ٿيو اها مبارڪ آهي ته جنهن رات ۾ صاحبِ قرآن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم دنيا ۾ تشريف کڻي آيا اها به مبارڪ آهي. هن رات ۾ سڄي سال جو رزق، موت، زندگي، عزت و ذلت، غرض تمام انتظامي امور لوحِ محفوظ کان فرشتن جي صحيفن ۾ نقل ڪري هر صحيفو ان محڪمي جي فرشتن کي ڏئي ڇڏيندا آهن. جيئن ملڪ الموت عَـلَيْـهِ الـسَّـلاَم کي سڀني مرڻ وارن جي فهرست وغيره. (نور العرفان، ص. 790)

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته شبِ براءَت بيحد اهم رات آهي لهٰذا ڪنهن صورت به ان کي غفلت ۾ نه گذاريو وڃي، ته ان رات الله عَزَّوَجَلَّ بيشمار ماڻهن جي مغفرت فرمائيندو آهي. جيئن ته نبي ڪريم رؤوف رّحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو ته مون وٽ جبرائيل عَـلَيْـهِ الـسَّـلاَم آيا ۽ چيو ته هي شعبان جي (15 وين) رات آهي، ان ۾ الله تعالٰي بني ڪلب جي ٻڪرين جي وارن جي تعداد جي برابر ماڻهن کي جهنم کان آزاد فرمائيندو آهي. (شعب الايمان، ج. 3، ص. 384، حديث 3837)

پر افسوس! ڪجهه بدنصيب ماڻهو اهڙا به آهن جنهن تي شبِ براءت يعني ڇوٽڪاري پائڻ جي رات به نه بخشڻ جي وعيد آهي. جيئن ته حضرت سيدنا امام بيهقي شافعي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ”فضائل الاوقات“ ۾ نقل ڪن ٿا: رسولِ اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو فرمانِ عبرت نشان آهي: ڇهه (6) ماڻهن جي هن رات به بخشش ناهي ٿيندي: (1) شراب جو عادي (2) ماءُ پيءُ جو نافرمان (3) بدڪاري جو عادي (4) قطع تعلق ڪرڻ وارو (5) تصوير بنائڻ وارو ۽ (6) چغل خور. (فضائل الاوقات، ج. 1، ص. 130، حديث 27) اهڙي طرح ڪجهه ٻين احاديثِ مبارڪن ۾ ڪاهن، جادوگر، تڪبر سان گڏ پاجامو يا تهبند ڀيڄين کان هيٺ لٽڪائڻ وارا ۽ ڪنهن مسلمان سان بلا اجازت شرعي بغض و ڪينو رکڻ وارن جي لاءِ به هن رات مغفرت کان محرومي جي وعيد بيان ڪئي وئي آهي، جيئن ته تمام مسلمانن کي گهرجي ته بيان ڪيل گناهن مان جيڪڏهن ڪنهن گناهه ۾ مبتلا هجي ته بالخصوص ان گناهه کان ۽ بالعموم سڀني گناهن کان شبِ براءَت جي اچڻ کان پهريان بلڪه اڄ ۽ هاڻي سچي توبهه ڪري ڇڏيو ۽ ڪيڏو نه سٺو

هجي جو توبهه تي استقامت پائڻ، پنهنجي دنيا ۽ آخرت جي بهتري جي لاءِ نيڪيون ڪرڻ گناهن کان بچڻ ۽ پنهنجي دل ۾ خوب خوب نفلي عبادت ڪرڻ جو جذبو پيدا ڪرڻ جي لاءِ دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي وڃون ۽ ذيلي حلقي جي 12 مدني ڪمن ۾به وڌي ڪري حصو وٺڻ وارا بڻجي وڃون .

اچو! هاڻي احاديثِ مبارڪه جي روشني ۾ نفل روزن جا ڪجهه فضائل به ٻڌون ٿا. جيئن ته

حضرت سيدنا انس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان مروي آهي: قيامت جي ڏينهن روزه دار قبرن مان نڪرندا ته اهي روزي جي خوشبو سان سڃاڻا ويندا، انهن جي لاءِ دسترخوان ۽ پاڻي جا ڪوزا رکيا ويندا، جنهن تي مشڪ جي مهر هوندي، انهن کي چيو ويندو ته کائو (ڇو ته روزا رکڻ جي سبب دنيا ۾) توهان بکيا هئو، پيو (ڇو ته روزا رکڻ جي سبب دنيا ۾) توهان اڃارا هئو، آرام ڪريو (ڇو ته روزا رکڻ جي سبب دنيا ۾) توهان ٿڪيل هئو، اهي کائيندا ۽ پيئندا حالانڪه ماڻهو اڃا حساب جي مشقت ۽ اُڃ ۾ مبتلا هوندا.

(ڪنز العمال، ج. 4، ص. 213، حديث 23639)

حضرت سيدتنا عُماره رَضِیَ اللهُ تَعَالٰی عَنْهَا فرمائن ٿيون: حضورِ پاڪ، صاحبِ لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مون وٽ تشريف کڻي آيا ته مون پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمتِ سراپا برڪت ۾ کاڌو پيش ڪيو ته ارشاد فرمايائون: توهان به کائو. مون عرض ڪيو: مان روزي ۾ آهيان. ته رحمتِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: جيستائين روزه دار جي سامهون کاڌو کايو ويندو آهي، فرشتا ان (روزه دار) جي لاءِ دعاء مغفرت ڪندا رهندا آهن. (ترمذي، ج. 2، ص. 205، حديث 785)

حضرت سيدنا بلال رَضِىَ اللهُ تَعَالٰی عَـنْهُ نبي اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمت ۾ حاضر ٿيا، ان وقت حضورِ پر

نور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ناشتو ڪري رهيا هئا، فرمايائون: اي بلال! ناشتو ڪريو. عرض ڪيو: يا رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! مان روزه دار آهيان. ته رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: اسان پنهنجي روزي کائي رهيا آهيون ۽ بلال جو رزق جنت ۾ وڌي رهيو آهي. اي بلال! ڇا تو کي خبر آهي ته جيستائين روزه دار جي سامهون ڪجهه کايو وڃي تيستائين ان جون هڏيون تسبيح ڪنديون آهن ۽ فرشتا ان جي لاءَ استغفار ڪندا آهن. (شعب الايمان، ج. 3، ص. 297، حديث 3586)

صَلُّوۡا عَلَى الۡحَبِيۡب صَلَّى اللّٰهُ تَعَالٰى عَلٰى مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اڄ جي بيان ۾ اسان شعبان المعظم جي مهيني ۾

v نفل عبادت ڪرڻ جي اهميت جي باري ۾ ٻڌو ته ماهِ مقدَّسه ۾ عبادت گزارن کي الله عَزَّوَجَلَّ جي معصوم فرشتن جي دعائن سان حصو نصيب ٿيندو آهي.

v اسان جا مٺا مٺا آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم به وقتا فوقتا هن مهيني جي اهميت کي مختلف طريقن سان اجاگر فرمايو، پاڻ به ان ۾ نفلن ۽ روزن جي ڪثرت فرمائي ۽ پنهنجي امت کي به ڪثرت سان نفلي عبادت جي ترغيب ڏيارڻ ۽ خوابِ غفلت کان بيدار ڪرڻ جي لاءِ ارشاد فرمايو ته رجب ۽ رمضان جي درميان هي اهو مهينو آهي جو ماڻهو ان کان غافل آهن حالانڪه ان ۾ ماڻهن جا اعمال الله عَزَّوَجَلَّ جي بارگاهه ۾ پيش ڪيا ويندا آهن.

v 15 شعبانُ المعظم جي مبارڪ ۽ مقدس رات ۾ عبادت ڪرڻ وارن کي خوشخبريون ڏنيون ويون آهن ته مبارڪ هجي ان کي جيڪو هن رات عبادت ڪري، مبارڪ هجي ان کي جيڪو هن رات ۾ سجدو ڪري، مبارڪ هجي ان کي جيڪو هن رات ۾ رڪوع ڪري، مبارڪ هجي ان کي جيڪو هن رات پنهنجي رب عَزَّوَجَلَّ کان دعا گهري، مبارڪ هجي ان کي جيڪو هن رات ۾ پنهنجي رب تعالٰي سان مناجات ڪرڻ ۾ مشغول هجي، ڪڏهن ايئن صدائون لڳايون وينديون آهن: آهي ڪو توبهه ڪرڻ وارو ته ان جي توبهه قبول ڪئي وڃي؟ آهي ڪو مغفرت طلب ڪرڻ وارو جو ان جي مغفرت ڪئي وڃي؟ آهي ڪو دعا ڪرڻ وارو جو ان جي دعا قبول ڪئي وڃي؟

v 15 شعبان جي رات ڇو ته ان ماهِ مقدس جي سڀ کان اهم رات آهي ۽ جنهن کي شبِ براءَت يعني ڇوٽڪاري جي رات به چيو ويندو آهي ڇو ته هن رات ۾ الله عَزَّوَجَلَّ قبيله بني ڪلب جي ٻڪرين جي وارن جي تعداد جي برابر ماڻهن کي جهنم کان ڇوٽڪارو عطا فرمائيندو آهي، پڻ انهيءَ رات سڄي سال جي لاءِ ماڻهن جي موت ۽ حيات، عزت ۽ ذلت، رزق ۽ ان جي ٻين اهم ترين معلامات جا فيصلا ڪيا ويندا آهن.

اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم 15 شعبانُ المعظم جي رات خصوصي طور تي عبادت جي ترغيب ارشاد فرمائيندي فرمايو ته هن رات کي عبادت ۾ گذاريو ۽ ڏينهن جو روزو رکو، اهو ئي سبب آهي جو صحابه ڪرام عَـلَيْهِمُ الرِّضْوَانْ ۽ بزرگانِ دين رَحِمَهُمُ الـلّٰـهُ تَـعَالٰی هن

v مهيني ۾ پاڻ به نفل عبادت جي ڪثرت فرمائيندا ۽ مسلمانن کي به ان جي ترغيب ارشاد فرمائيندا، لهٰذا اسان کي به گهرجي ته شور ۽ غل ۽ طرح طرح جي گناهن جي ذريعي هن مبارڪ رات جو تقدس پامال ڪرڻ، آتش بازي ڪري پنهنجي ۽ ٻين مسلمانن جي جان ۽ مال کي نقصان پهچائڻ ۽ الله عَزَّوَجَلَّ ۽ رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي ناراض ڪرڻ جي بجاء انهن کي راضي ڪرڻ جي لاءِ تمام گناهن کان سچي توبهه ڪري ذڪر و درود، تلاوتِ قرآن ۽ ٻين نيڪ اعمال جي ذريعي خوب خوب نفل عبادت ڪيون.

v شعبانُ المعظم ڇو ته ڪثرتِ درود جو مهينو به قرار ڏنو ويو آهي، لهٰذا فضول گوئي کان بچندي زياده کان زياده درودِ پاڪ پڙهڻ جي به ڪوشش ڪئي وڃي.

v هن ماهِ مبارڪ جي حقيقي برڪتون حاصل ڪرڻ، گناهن کان بچڻ ۽ نيڪيون ڪرڻ جي لاءِ دعوتِ اسلامي جي مدني ماحُول سان وابسته ٿي وڃو، دعوتِ اسلامي جي تحت هفتيوار ۽ ٻين اجتماعاتِ ذڪر و نعت ۾ شرڪت کي به پنهنجو معمول بڻائي ڇڏيو ۽ مدني چينل به ڏسندا رهو. الله عَزَّوَجَلَّ اسان سڀني جي مغفرت فرمائي ڪري اسان کي شعبانُ المعظم جي برڪتن سان مالامال فرمائي.

اٰمِيۡن بِجَاهِ النَّبِيِّ الۡاَمِيۡن صَلَّى اللّٰهُ تَعَالٰى عَلَيۡهِ وَاٰلِه وَسَلَّم

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! بيان جي آخر ۾ سنّت جي فضيلت ۽ ڪجهه سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو. تاجدار رسالت

شهنشاهه نبوت مصطفٰي جان رحمت شمع بزم هدايت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ جنّت نشان آهي: ”جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي. ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو“. (مشڪاة المصابيح، ج 1، ص 55، حديث 75 ادارا الکتب العلميه بيروت)

سینہ تری سنّت کا مدینہ بنے آقا

جنّت میں پڑوسی مجھے تم اپنا بنانا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

اچو! شيخِ طريقت ، امير اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي ”163 مدني گل“ مان عمامي شريف ٻڌڻ جون سنتون ۽ آداب ٻڌون ٿا:

پهريان ٻه فرامين مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ملاحظه فرمايو: * عمامي سان گڏ ٻه رڪعتون نماز (پڙهڻ) بغير عمامي جي ستر 70 رڪعتن کان افضل آهي ـ (فردوس الاخبار، ج 2 ص 265 حديث 3233) * ٽوپي تي عمامو اسان ۽ مشرڪن جي وچ ۾ فرق آهي، هر ورَ جيڪو مسلمان پنهنجي مٿي تي ڏيندو ان کي قيامت جي ڏينهن هڪ نور عطا ڪيو ويندو. (الجامع الصغير، ص353 حديث 5725) * عمامو قبلي ڏانهن منهن ڪري بيهي ٻڌو. (ڪشف الالتباس في استحباب اللباس للشيخ عبدالحق الدهلوي ص 38) * عمامي ۾ سنت هي آهي ته اڍائي گز کان گھٽ نه هجي، نه ڇهه گز کان وڌيڪ، ۽ ان جي بندش گنبذ نما هجي. (فتاويٰ رضويه ج22 ص 186) * رومال

جيڪڏهن ايترو وڏو هجي جيڪو مٿي کي چڱي طرح ڍڪي ته اهو عمامو ئي آهي ۽ * ننڍو رومال جنهن سان صرف هڪ ٻه وَر اچي سگھن ٻَڌڻ مڪروهه آهي. (فتاويٰ رضويه ج 7 ص 299) * عمامي کي جڏهن نئي سري سان ٻڌڻو هجي ته جيئن ٻڌيل آهي ائين کوليو ۽ هڪ مهل زمين تي نه اُڇلايو. (فتاويٰ عالمگيري ج 5 ص 330) * جيڪڏهن ضرورتاً لاٿو، ۽ ٻيهر ٻَڌڻ جي نيّت هجي ته هڪ هڪ پيچ کولڻ تي هڪ هڪ گناهه مٽايو ويندو . (ملخص از فتاويٰ رضويه ج 6 ص 214) عمامي جا ڇهه طبي فائدا ملاحظه فرمايو: مٿي اگھاڙي رهندڙ جي وارن تي سردي گرمي ۽ اُس وغيره سڌو سنئون اثر ڪندي آهي ان سان نه صرف وار بلڪ دماغ ۽ چهرو به متاثر ٿيندو آهي ۽ صحت کي نقصان پهچي سگھي ٿو، تنهن ڪري سنت جي پيروي جي نيت سان عمامو شريف ٻڌڻ ۾ ٻنهي جهانن ۾ عافيّت آهي .

هزارين سنّتون سکڻ جي لاءِ مڪتبة المدينه جا شايع ٿيل ٻه ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب ” سنّتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهو۔ سنّتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنّتن ڀريو سفر آهي۔
