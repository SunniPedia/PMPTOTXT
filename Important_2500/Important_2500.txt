بيان ڪرڻ جون نيّتون

مان به نيت ڪريان ٿو * الله جي رضا حاصل ڪرڻ ۽ ثواب حاصل ڪرڻ جي لاءِ بيان ڪندس * ڏسي بيان ڪندس * سيپاري 14 سورة النحل

آيت نمبر 125: اُدۡعُ اِلٰی سَبِیۡلِ رَبِّکَ بِالۡحِکۡمَۃِ وَ الۡمَوۡعِظَۃِ الۡحَسَنَۃِ ” ترجمو ڪنزالايمان: پنهنجي ربّ جي واٽ جي طرف سڏ پڪي سمجهه ۽ چڱي نصيحت سا ن“ ۽ بخاري شريف (حديث؛ 4361) جي هن فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيهِ وَاٰلِهٖ وَسَلَّم: بَلِّغُوۡا عَنِّيۡ وَلَوۡ آيَة يعني ” منهنجي طرفان پهچايو جيتوڻيڪ هڪ ئي آيت هجي“ ۾ ڏنل احڪام جي پيروي ڪندس * نيڪي جو حڪم ڏيندس ۽ بُرائي کان منع ڪندس * شعر پڙهڻ، عربي، انگريزي ۽ ڏکيا لفظ ڳالهائڻ دؤران دل جي اخلاص تي توجُّھ رکندس يعني پنهنجي علميت جو ڌاڪو ويهارڻ مقصود ٿيو ته ڳالهائڻ کان بچندس * مدني قافلي، مدني انعامات ۽ علاقائي دوره براءِ نيڪي جي دعوت وغيره جي رغبت ڏياريندس * ٽهڪ ڏيڻ ۽ ڏيارڻ کان بچندس * نظر جي حفاظت بڻائڻ جي خاطر حتي الامڪان نگاهون جُهڪائي رکندس۔

مٺا مٺا اسلامي ڀائرو! حضرتِ سَيّدُنا اِسمٰعيل عَـلَيْـهِ الـسَّـلَام ، الله عَزَّوَجَلَّ جا مقرّب ۽ چونڊيل نبي آهن، الله عَزَّوَجَلَّ سندن کي خصوصي انعامن سان نوازيو آهي، اهو ئي سبب آهي ته سندن تي الله عَزَّوَجَلَّ جي طرف کان جيتريون به آزمائشون آيون، پاڻ ان ۾ توفيقِ الٰهي سان ثابت قدم رهيا، پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو هڪ مشهور واقعو ۽ پوءِ ان مان حاصل ٿيندڙ مدني گُل به ٻڌڻ جي سعادت حاصل ڪنداسين.

ٽن راتين کان هڪ ئي خواب:

حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام اٺين ذوالحج جي رات هڪ خواب ڏٺو

جنهن ۾ ڪو چوڻ وارو چئي رهيو آهي: ”بيشڪ الله عَزَّوَجَلَّ توکي پنهنجو پُٽ ذبح ڪرڻ جو حڪم ڏئي ٿو.“ پاڻ صبح کان شام تائين انهيءَ باري ۾ غور فرمائيندا رهيا ته هي خواب الله عَزَّوَجَلَّ جي طرفان آهي يا شيطان جي طرفان؟ اٺ ذوالحج جو نالو یَوۡمُ التَّرۡوِیَہ (يعني سوچ ويچار جو ڏينهن) انهيءَ ڪري رکيو ويو. نائين (ذوالحج) جي رات وري ساڳيو خواب ڏٺائون ۽ صبح يقين ڪري ورتائون ته هي حڪم الله عَزَّوَجَلَّ جي طرفان آهي. ان ڪري 9 ذوالحج کي یَوۡمِ عَرفَه (يعني سڃاڻپ وارو ڏينهن) چيو ويندو آهي. ڏهين (ذوالحج) جي رات وري اهوئي خواب ڏسڻ کان بعد پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام صبح جو هن خواب تي عمل ڪرڻ يعني پُٽ جي قرباني جو پڪو ارادو ڪري ورتو، انهيءَ ڪري 10 ذوالحج کي یَوۡمُ النَّحۡر يعني ” ذَبۡح جو ڏينهن“ چيو ويندو آهي .

(تفسير ڪبير ج 9، ص 346)

مٺا مٺا اسلامي ڀائرو! انبياءِ ڪرام عَـلَـيْهِمُ السَّلَام جو مقام ۽ مرتبو تمام مخلوق کان بلند ۽ اعليٰ آهي، تنهنڪري انهن تي اچڻ وارا امتحان به ايتري قدر شديد هوندا آهن، پر قربان وڃو، ان مقدس هستين جي صبر و استقلال تي جو اهي ان راھ ۾ اچڻ واري هر مصيبت ۽ مشڪل کي خوشي خوشي برداشت ڪري نه صرف بارگاھِ الٰهي ۾ سُرخرو ٿي بلند درجات حاصل ڪندا آهن بلڪه مستقبل ۾ اچڻ وارن امتحانن جي لاءِ پنهنجو پاڻ کي ۽ گھروارن کي هر وقت تيار رکندا آهن، انهن جون اهي ئي قربانيون رهندڙ

دنيا تائين ماڻهن جي لاءِ مشعلِ راھ بڻجي وينديون آهن. جڏهن ته انبياءِ ڪرام عَـلَـيْهِمُ السَّلَام جا خواب وَحي هوندا آهن.

(مستدرڪ،تفسير سورة الصافات، ۳/۲۱۴، حديث:۳۶۶۵)

تنهنڪري حضرت ابراهيم عَـلَيْـهِ الـسَّلَام سمجھي ويا ته منهنجي ربّ عَزَّوَجَلَّ مون کي پنهنجي پُٽ کي ذبح ڪرڻ جو حڪم ارشاد فرمايو آهي. فورًا پنهنجي لَختِ جگر کي فرمانِ خداوندي تي قربان ڪرڻ جي لاءِ تيار ٿي ويا ۽ پاڻ سڄو معاملو پنهنجي ننڍي ڄمار واري پُٽ حضرتِ اِسمٰعيل عَـلَيْـهِ الـسَّـلَام کي به ٻڌائي ڇڏيو ته الله عَزَّوَجَلَّ جو فرمان آهي ته آئون توکي ذبح ڪريان، هاڻي تون ٻڌاءِ تنهنجي ڇا راءِ آهي. تفسيرِ خازِن ۾ آهي ته حضرت ابراهيم عَـلَيْـهِ الـسَّلَام حضرتِ اِسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کان ان لاءِ مشورو نه گھُريو ته جيڪڏهن انهن جي مرضي نه هجي ته انهن جي راءِ تي عمل ڪن بلڪه ان مان حضرتِ اِسمٰعيل عَـلَيْـهِ الـسَّـلَام جو امتحان مقصود هو ته الله عَزَّوَجَلَّ جي طرف کان ملڻ واري آزمائش تي انهن جا ڇا جذبات آهن ۽ الله عَزَّوَجَلَّ جي حڪم جي اطاعت ۽ ان جي طرف کان ملڻ واري مَشَقَّت تي صَبۡر ۽ ثابت قدمي جو علم ٿي وڃي ۽ اهي حڪمِ خداوندي جي بجا آوري تي ملڻ واري ثواب کي حاصل ڪرڻ ۾ به ڪامياب ٿي سگھن. (تفسير خازن۴/۲۲،ملخصاً)

حضرت سَيِّدنا ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن حضرتِ سَيّدُنا اِسمٰعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي جڏهن هي خواب بيان فرمايو ته ان پيڪرِ تسليم و رِضا جيڪو جواب ڏنو ان کي قرآنِ پاڪ ۾ سيپاري 23 سُورۃُ الصّٰفّٰت جي آيت نمبر 102 ۾ هنن لفظن سان بيان ڪيو ويو آهي:

یٰۤاَبَتِ افۡعَلۡ مَا تُؤۡمَـرُ ۫سَتَجِدُنِیۡۤ اِنۡ شَآ ءَ اللہُ مِنَ ا لصّٰبِرِ یۡنَ ﴿ ۱۰۲ ﴾

(سيپارو 23، سورھ الصّٰٓفّٰت آيت 102)

مونکي رسين سان مضبوط ٻڌي ڇڏجو

تفسيرِ خازن ۾ آهي ته حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالسَّلَام پنهنجي والد محترم کي عرض ڪيو: بابا جان! ذبح ڪرڻ کان پهريان مون کي رسين سان مضبوط ٻڌي ڇڏجو ته جيئن مان حرڪت نه ڪري سگهان ڇو ته مون کي ڊپ آهي ته منهنجي ثواب ۾ گهٽتائي نه اچي وڃي ۽ منهنجي رت جي ڇنڊن کان پنهنجن ڪپڙن کي بچائجو ته جيئن انهن کي ڏسي منهنجي امڙ غمگين نه ٿئي. ڇري چڱي طرح تيز ڪري وٺو ته جيئن منهنجو ڳلو فوراً چڱي طرح ڪٽجي وڃي ڇو ته موت ڏاڍو سخت هوندو آهي، توهان مون کي ذبح ڪرڻ وقت منهنجي پيشاني زمين جي طرف ڪجو ته جيئن توهان جي نظر منهنجي چهري تي نه پوي ۽ جڏهن توهان منهنجي امڙ ڏي وڃو ته منهنجا سلام پهچائجو ۽ جيڪڏهن توهان مناسب سمجهو ته منهنجي قميص انهن کي ڏئي ڇڏجو، هن سان انهن کي تسلي ٿيندي ۽ صبر اچي ويندو. حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ارشاد فرمايو: اي منهنجا پُٽ! توهان الله تعاليٰ جي حڪم تي عمل ڪرڻ ۾ منهنجا ڪيڏا نه عمده مددگار ثابت ٿي رهيا آهيو! پوءِ جيئن حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام فرمايو هو انهن کي اهڙي ئي طرح ٻڌي پنهنجي ڇُري تيز ڪئي، حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي پيشاني ڀر ليٽايو، انهن جي چهري تان نظر هٽائي ۽ انهن جي ڳلي تي ڇُري وهائي ڇڏي پر ڇُري پنهنجو ڪم نه ڪيو يعني ڳلو نه ڪٽيو.

(تفسير خازن ج 4، ص 22 ملخّصاً)

حضرت سَيّدُنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جڏهن حضرت سَيّدُنا اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي ذبح ڪرڻ جي لاءِ زمين تي ليٽايو ته حضرت جبرائيل عَـلَيْـهِ الـسَّـلَام الله تعالٰي جي حڪم سان فديي طور جنّت مان هڪ دنبو کڻي آيا ۽ پري کان وڏي آواز ۾ فرمايائون: اَللّٰهُ اَكۡبَرُ اَللّٰهُ اَكۡبَرُ جڏهن حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالسَّلَام هي آواز ٻڌو ته پنهنجو مٿو آسمان جي طرف کنيائون ۽ سمجهي ويا ته الله عَزَّوَجَلَّ جي طرفان ايندڙ آزمائش جو وقت گذري چڪو آهي ۽ پُٽ جي جڳهه تي فديي ۾ دنبو موڪليو ويو آهي، لهٰذا خوش ٿي فرمايائون: لَآ اِلٰهَ اِلَّا اللّٰهُ وَاللّٰهُ اَكۡبَرُ جڏهن حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام هي ٻڌو ته فرمايائون: اَللّٰهُ اَكۡبَرُ وَلِلّٰهِ الۡحَمۡدُ ، هن کان بعد انهن ٽنهي پاڪ هستين جي انهن مبارڪ لفظن جي ادائگي جي اها سنّت قِيامت تائين جاري ۽ ساري ٿي وئي. (بنايه شرح هدايه ج3، ص130)

مٺا مٺا اسلامي ڀائرو! بيان ڪيل واقعي ۾ اسان جي لاءِ به مدني گُل موجود آهن. جيئن ته توهان ملاحظه فرمايو ته جڏهن حضرت ابراهيم عَـلَيْـهِ الـسَّلَام جن حضرت اسماعيل عَـلَيْـهِ الـسَّلَام کي پنهنجو خواب ٻڌايو ته پاڻ بجاءِ غمگين ۽ خوفزده ٿيڻ جي فرحت ۽ مسرت سان ڄڻ جھومڻ لڳا ته منهنجي خوش قسمتي جو الله عَزَّوَجَلَّ مون کي قرباني جي لاءِ طلب فرمايو آهي ۽ بَخوشي الله عَزَّوَجَلَّ جي راھ ۾ قربان ٿيڻ جي لاءِ راضي ٿي ويا، ان جي برعڪس جيڪڏهن ڪنهن تي ٿوري آزمائش اچي وڃي يا ڪا تڪليف پهچي ته شِڪوه شڪايت بلڪه مَعَاذَ الله عَزَّوَجَلَّ بي صبري جو مظاهرو ڪندي گھڻو ڪري ڪفريات بَڪي ڇڏيندا آهن ۽ ربّ تَعَاليٰ جي ناراضي جو شڪار ٿي ويندا آهن، اهڙي موقعي تي خوب صبر جو مظاهرو ڪرڻ گھرجي ته هي

آزمائش به الله عَزَّوَجَلَّ جي طرف کان رحمت آهي، جيئن ته حضرت سَيّدُنا انس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته حضورِ پاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: ”بيشڪ گھڻو اَجۡر سخت آزمائش تي ئي آهي ۽ الله عَزَّوَجَلَّ جڏهن ڪنهن قوم سان محبت ڪندو آهي ته انهن کي آزمائش ۾ مبتلا ڪري ڇڏيندو آهي، ته جيڪو ان جي قضا تي راضي رهي ان جي لاءِ رضا آهي ۽ جيڪو ناراض ٿئي ان جي لاءِ ناراضي آهي.“ (ابن ماجہ، ڪتاب الفتن، باب الصبر علي البلاء، رقم ۴۰۳۱، ج۴ ،ص ۳۷۴) هن واقعي مان تربيتِ اولاد جو به درس ملي ٿو ته اسان جا ٻار جيڪڏهن اسان جي دل جو چين ۽ اکين جو نُور هجن پر ان کان پهريان اسين الله عَزَّوَجَلَّ جا ٻانها، نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا اُمّتي ۽ معاشري جا اهم فرد به آهيون . جيڪڏهن اسان جي تربيت انهن کي الله عَزَّوَجَلَّ جي عبادت، سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اطاعت ۽ سُنتن سان محبت ۽ گناهن کان نفرت نه ٿي ڏياري سگھي ته انهن کي پنهنجو فرمانبردار بڻائڻ جو خواب ڏسڻ به ڇڏي ڏيڻ گھرجي، ڇو جو هي اسلام ئي آهي جيڪو هڪ مسلمان کي پنهنجي والدين جو مطيع ۽ فرمانبردار بڻجڻ جي تعليم ڏئي ٿو. تنهن جي ڪري اسان کي اولاد جي ظاهري زيب و زينت، چڱي لباس ۽ ٻين ضرورتن سان گڏوگڏ انهن جي اخلاقي ۽ روحاني تربيت جي لاءِ به هر وقت تيار رهڻ گهرجي.

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

شهر ِمڪةُ المڪرّمه جي آبادڪاري

حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جڏهن حضرتِ سَيّدُنا اِسمٰعيل جي والده حضرتِ ہاجرہ رَضِیَ اللهُ تَعَالٰی عَنْهَا حضرتِ سَيّدُنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام کي مڪي شهر

وٺي آيا ۽ انهن کي اُتي ڇڏي ڏنو، انهي دوران هڪ ڊگهو عرصو گذري ويو ۽ مڪّهء مڪرمه ۾ ئي قبيلو جُرهم آباد ٿيو ۽ اهي به اُتي رهڻ لڳا. (انهي عرصي ۾ حضرتِ اِسمٰعيل عَـلَيْـهِ الـسَّلَام به جوان ٿي چڪا هئا) حضرت سَيّدُنا اِسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن انهن جي هڪ عورت سان شادي ڪري ورتي ۽ انهن جي والده حضرتِ سيِّدتنا ہاجرہ رَضِیَ اللهُ تَعَالٰی عَنْهَا جو ان عرصي ۾ وصال ٿي ويو. چڱو عرصو گذرڻ کان پوءِ حضرتِ ابراهيم عَـلَيْـهِ الـسَّلَام پنهنجي پُٽ حضرت سَيّدُنا اِسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي گھر تشريف وٺي آيا ۽ انهن جي گھرواريءَ کان پڇيو: توهان جا صاحب (مُڙس) ڪٿي آهن؟ انهن عرض ڪيو: اُهي شڪار جي لاءِ تشريف وٺي ويا آهن، الله عَزَّوَجَلَّ توهان تي رحم فرمائي!توهان تشريف رکو اهي اِنْ شَآءَالله عَزَّوَجَلَّ هاڻي بس اچي رهيا هوندا. حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن پڇيو: توهان وٽ کائڻ جي لاءِ ڪجهه آهي؟ انهن عرض ڪيو: جي ها! ۽ ان وقت ئي کير ۽ گوشت حاضرِ خدمت ڪيو. حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن انهن کان گذربسر جي باري ۾ پڇيو، ته حضرتِ سيّدنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام جي گھرواريءَ عرض ڪيو: اسان خير سان آهيون ۽ اَ لْحَمْدُلِلّٰہ عَزَّوَجَلَّ خوش حال به آهيون. ته حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن انهن ٻنهي جي لاءِ دعاءِ برڪت فرمائي پوءِ حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام انهن کي چيو ته جڏهن تنهنجا گھروارا اچن ته انهن کي منهنجا سلام ڏجو، ۽ چئجو ته پنهنجي دروازي جي چائنٺ کي باقي رکن. جڏهن حضرتِ سيّدنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام تشريف وٺي آيا ته انهن پنهنجي والدِ محترم جي خوشبو محسوس ڪئي ته پنهنجي گھرواريءَ کان پڇيو: ڇا اڄ هتي ڪو آيو هو؟ انهن جواب ڏنو: ها هڪ خوبصورت چهري وارا ۽ سٺي خوشبو وارا بزرگ تشريف وٺي آيا هئا، ان

کان بعد انهن تمام معاملو ٻڌايو. ۽ هي به چيو ته مون انهن جو مٿو مبارڪ ڌوتو هو ۽ انهن جي قدمن جا نشان آهن. حضرتِ سيّدنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام جن پورو واقعو ٻڌي ڪري ارشاد فرمايو: اهي منهنجا والد حضرت ابراهيم هئا ۽ منهنجي دروازي جي چائنٺ مان مراد تون آهين ۽ انهن مون کي حڪم ڏنو آهي ته آئون توکي روڪي رکان. (روح البيان،۱/۲۲۵ ملتقطاً)

مٺا مٺا اسلامي ڀائرو! هن واقعي مان اهو درس ملي ٿو ته جڏهن ڪو مهمان اچي ته اسان کي پنهنجي حيثيت جي مطابق ان جي مهماني ڪرڻ گهرجي ۽ پنهنجي غربت ۽ تنگدستي جو اظهار ڪرڻ جي بجاءِ هر حال ۾ الله عَزَّوَجَلَّ جو شڪر ادا ڪرڻ گھرجي، پنهنجي ڀينرن، ڌيئرن ۽ پنهنجي ٻارن جي ماءُ جي تربيت ڪندي انهن کي گھرواري جي حقن جي ادائگي ۽ ان جي شڪر گذار گھروارِي بنجڻ ۽ ناشڪري ۽ احسان فراموشي کان بچڻ جو ذهن ڏيڻ گھرجي. يقينًا نيڪ سيرت ۽ مثالي گھرواريءَ جي هڪ خوبي اها به آهي ته اها هميشه پنهنجي مڙس جي نعمتن جي شڪر گذار رهندي آهي ۽ ڪڏهن به انهن احسانن جو انڪار ڪري ناشڪري ناهي ڪندي اها چڱي طرح ڄاڻيندي آهي ته مڙس مون نازڪ جي لاءِ هڪ مضبوط سهارو ۽ الله عَزَّوَجَلَّ جي نعمت آهي، اهو ئي منهنجي ضرورتن کي پورو ڪندو آهي ۽ ان جي سبب مون کي اولاد جي نعمت ملي آهي. پر بدقسمتي سان اسان جي معاشري ۾ ڪيتريون ئي اسلامي ڀينرن ناشڪري جي وبال ۾ گرفتار آهن، ٿورو ڪنهن جو چڱو گھر، قيمتي ڪپڙا ۽ قيمتي زيورات ڏسن ته ناشڪري تي لهي اينديون آهن ۽ مَعَاذَالله عَزَّوَجَلَّ الله تَعَالٰي تي اعتراض ڪندي ڪجھ هن طرح جي گفتگو ڪنديون آهن ته ”خدا اسان

کي نه ڄاڻي ڪهڙي جُرم جي سزا ۾ مفلس ۽ غريب بڻائي ڇڏيو. آئون ته آهيان ئي بدقسمت جو نه پيڪَن ۾ سُک نصيب ٿيو نه ساهِرَن وٽ ڪجھ ڏٺو، فُلان کي ڏسو عيش ۽ آرام ۾ زندگي بسر ڪري رهي آهي ۽ آئون هتي فاقن سان مري رهي آهيان، وغيره وغيره.“ اهڙي طرح ڪن جي ته هي عادت هوندي آهي جو انهن کي سٺو ۽ عمده کاڌو کارائيندو پياريندو رهجي گڏوگڏ انهن کي پنهنجي حيثيت ۽ طاقت جي مطابق ڪپڙا، زيورات ۽ ٻيو سازوسامان ڏيندو رهجي پر ڪنهن مجبوري جي سبب ڪا خواهش پوري نه ڪري سگهجي ته اهي مڙس جا پوري زندگيءَ جا احسان وساري ان جي ناشڪري ڪندي چونديون آهن ته ”هاءِ مونکي هن گھر ۾ ڪڏهن سُک نصيب نه ٿيو، تون منهنجي ڪڏهن ڪا خواهش پوري ئي نه ڪئي. منهنجي ته قسمت ئي خراب آهي جو توسان شادي ڪئي وئي وغيره وغيره“ (جنّتي زيور،ص ۱۲۵،ملخصاً) ياد رهي! ته ناشڪري جا اهي لفظ نه صرف عورت جي دنياوي زندگي کي اُجاڙي ڇڏيندا آهن بلڪه ان جي آخرت به برباد ڪري سگھن ٿا. جيئن ته

ٻنهي جهانن جي تاجور، سلطانِ بحروبر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عبرت نشان آهي: مون جهنم ۾ اڪثريت عورتن جي ڏٺي . صحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان جن عرض ڪيو: يَا رَسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان جو ڇا سبب آهي؟ فرمايو: ڇو جو اهي ناشڪري ڪنديون آهن. پڇيو ويو ته ڇا اهي الله عَزَّوَجَلَّ جي ناشڪري ڪنديون آهن؟ ارشاد فرمايو: اهي مُڙس ۽ ان جي احسان جي ناشڪري ڪنديون آهن.۽ جڏهن توهان ڪنهن عورت سان پوري عُمر چڱو سلوڪ ڪيو پرتوهان ۾ ڪا ناپسنديده ڳالھ ڏسندي ته چوندي: مون ته ڪڏهن تو مان ڀلائي ڏٺي ئي ناهي.

(بخاري،ڪتاب النڪاح ، باب ڪفران العشير…الخ، ۳/۴۶۳، حديث:۵۱۹۷)

۽ جيڪي اسلامي ڀينرون عقلمندي جو ثبوت ڏيندي پنهنجي مُڙس جي فرمانبرداري ۽ شڪر گذار گھرواري بڻجڻ جو ڪردار نڀائينديون آهن، انهن جي لاءِ جنت جي بشارت آهي، جيئن ته فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: جيڪا عورت پنج نمازون پڙهي، رمضان جا روزا رکي، پنهنجي عزت ۽ ناموس جي حفاظت ڪري ۽ پنهنجي مُڙس جي اطاعت ڪري ته ان کي چيو ويندو ته جنت جي جهڙي به دروازي مان چاهين جنت ۾ داخل ٿي وڃ.

(مسند امام احمد، مسند عبد الرحمن بن عوف، ۱/ ۴۰۶، حديث:۱۶۶۱)

ياد رکو! مڙس جي ناشڪري کان ان لاءِ به بچڻ گھرجي جو مسلسل ان طرح جون ڳالهيون مڙس جي دل ۾ نفرت ۽ عداوت جو طوفان پيدا ڪنديون آهن ۽ جيڪڏهن خدانخواسته ان جي سبب تعلّقات ڪمزور ٿيڻ لڳا ته پوري عُمر پڇتائڻ کان علاوه ڪجھ هٿ نه ايندو. بهرحال ڪامياب زال اها ئي آهي جيڪا ڪڏهن به ناشڪري جا لفظ زبان تي نه آڻي ۽ هميشه مڙس جي شڪر گذار بڻجي ان جي دل خوش ڪندي رهي. زال مڙس جي درميان محبت ۽ الفت جو هي خوبصورت رشتو چڱي نموني سان انهيءَ وقت ئي هلي سگھي ٿو جڏهن اهي ٻئي هڪ ٻئي جا حق چڱي طرح ادا ڪن.

پاڻ اَبُو العرب (اهلِ عرب جا والد) آهن. آبِ زم زم سندن جو معجزهء ارهاصي (اعلانِ نبوت کان پهريان صادر ٿيڻ وارو معجزو) آهي جيڪو قيامت تائين باقي آهي قيامت تائين ٻه (2) معجزا باقي رهڻ وارا آهن، هڪ آبِ زم زم جو چشمو جيڪو اسمٰعيل عَـلَيْـهِ الـسَّلَام جي کُڙي مان نڪتو ۽ ٻيو دينِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم يعني قرآنِ ڪريم، حديثِ پاڪ ۽ ان جا قانون ۽ عبادتون. (تفسير نعيمي، ۱۶/۲۸۵)

حضرت سَيّدُنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام جي يادگار

مٺا مٺا اسلامي ڀائرو! آبِ زم زم شريف قيامت تائين باقي رهڻ وارو معجزو آهي، ان کي حضرت سَيّدُنا اِسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام سان نسبت جي سبب دنيا جي سڀني پاڻين تي ڪيترن ئي اعتبارن سان فضليت ۽ فوقيت حاصل آهي. ڪيترائي سال گذرڻ جي باوجود به هي پاڻي مخلوقِ خدا کي فيضياب ڪري رهيو آهي ۽ ظاهري ۽ باطني مرضن کان شفا جي دولت به تقسيم ڪري رهيو آهي. منهنجي آقا اعلٰي حضرت امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ارشاد فرمايو: زم زم شريف جو هڪ معجزو هي به آهي هر وقت مزو تبديل ٿيندو رهندو آهي. ڪڏهن ٿورو کارو، ڪڏهن تمام مٺو ۽ رات جو ٻه (2) بجي کان پوءِ پيتو وڃي ته تازو ڏوهيل ڳئون جو خالص کير معلوم ٿيندو آهي. (وڌيڪ فرمائن ٿا:) زم زم شريف ڪنهن وٽ ڪافي مقدار ۾ هجي ان کي نه ڪنهن غذا جي ضرورت، نه دوا جي.

(ملفوظاتِ اعلٰي حضرت، ص۴۳۵)

حديث شريف ۾ آهي: زم زم کاڌي جي جڳھ تي کاڌو ۽ دَوا جي جڳھ تي دَوا آهي. (مصنف ابن ابي شيبہ،ڪتاب الحج،باب في فضل زمزم، ۴/۳۵۸،حديث:۲) اچو ته! هاڻي آبِ زم زم جي فضائل تي ٽي (3) فرمانِ مصطفٰي ٻڌون ٿا. جيئن ته

1. آبِ زم زم دنيا ۽ آخرت جي جنهن مقصد جي لاءِ به پيتو وڃي ڪافي آهي.

(سنن ابن ماجة، ابواب المناسڪ، باب الشرف من زمزم، ج ۳، ص۴۹۰، حديث:۳۰۶۲، بدون ”من امرالدنيا والآخرة“)

2. آبِ زم زم پيٽ ڀري پيئڻ نفاق کان ڇوٽڪارو ڏيندو آهي.

(معجم الڪبير،حديث:۱۱۱۶۷ ،۱۱/۸۰)

آبِ زمزم شعائرُالله آهي

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته الله تبارڪ وتعاليٰ ان مقدس پاڻي کي ڪيتري قدر شان ۽ شوڪت ۽ برڪتن سان نوازيو آهي ته جتي هي مسلمانن جي دلين ۾ قيامت تائين الله عَزَّوَجَلَّ جي نبي حضرت سَيّدُنا اِسمٰعيل ذبيح الله عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي ياد ۽ ان جي محبت تازي ڪندو رهندو، اُتي هي بابرڪت پاڻي بيمارن، پريشان حالن، ڏکايلن ۽ غم ۾ ورتلن جي جسماني ۽ روحاني مرضن ۾ شفا جو سبب به ٿيندو. واقعي جنهن شيءِ کي الله وارن جي مبارڪ جسم سان نسبت حاصل ٿي وڃي ان جو شرف ۽ عظمت وڌي ويندي آهي بلڪه اهي شعائرُ الله (الله عَزَّوَجَلَّ جون نشانيون) بڻجي وينديون آهن. جيئن ته مُفسّر شهير، حڪيمُ الاُمّت، حضرتِ مفتي احمد يار خان عَلَيْهِ رَحْمَةُ الحَنّان فرمائن ٿا: صفا ۽ مروه اهي جبل آهن جن تي حضرت هاجره (رَضِیَ اللهُ تَعَالٰی عَنْهَا) پاڻي جي تلاش ۾ ست (7) ڀيرا چڙهيون ۽ لٿيون. ان الله واريءَ جا قدم پوڻ جي برڪت سان هي ٻئي جبل شَعائرُالله بڻجي ويا ۽ قيامت تائين حاجين تي ان پاڪ بي بي جي نَقَل ڪندي ان تي ست (7) ڀيرا چڙهڻ ۽ لهڻ لازم ٿي ويو. بزرگن جا قدم لڳڻ سان اها شيءِ شَعائرُالله بڻجي ويندي

آهي. (مزيد فرمائن ٿا:) طورِ سينا جبل ۽ مڪهء معظمه ان لاءِ عظمت وارا بڻجي ويا جو طور کي ڪَلِيۡمُ الله (يعني موسٰي عَـلَيْـهِ الـسَّلَام) سان ۽ مڪهء معظمه کي حبيبُ الله (يعني اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) سان نسبت ٿي وئي. خلاصو هي آهي ته الله جي پيارن جون شيون شَعائرُالله آهن جيئن قرآن شريف ، خانهء ڪعبه ، صفا مروه جبل، مڪهء معظمه ، بيت المقدس ، طورسينا ، اولياءِ ڪرام ۽ انبياءِ ڪرام جا مقبرا (يعني مزارات) آبِ زمزم وغيره.

(علم القرآن،ص۴۸-۵۰ملتقطاً)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

قرآن ڪريم ۾ حضرت سَيّدُنا اِسمٰعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو اسمِ گرامي ڪيترين ئي جڳهن تي آيو آهي ۽ هر جڳھ تي سندن جي شان ۽ عظمت جو ذڪر موجود آهي. جيئن ته هڪ جڳهه تي الله عَزَّوَجَلَّ سندن عَـلَيْـهِ الـسَّلَام جون خوبيون هنن لفظن سان بيان فرمايون آهن: جيئن ته سيپاري 16 سُوۡرَۃُ مَرْيَم جي آيت نمبر 54 ۽ 55 ۾ ارشادِ باري تعالٰي آهي:

وَ اذۡکُرۡ فِی الۡکِتٰبِ اِسۡمٰعِیۡلَ ۫ اِنَّہٗ کَانَ صَادِقَ الۡوَعۡدِ وَ کَانَ رَسُوۡلًا نَّبِیًّا ﴿ۚ۵۴﴾ وَ کَانَ یَاۡمُرُ اَہۡلَہٗ بِالصَّلٰوۃِ وَ الزَّکٰوۃِ ۪ وَ کَانَ عِنۡدَ رَبِّہٖ مَرۡضِیًّا ﴿۵۵ ﴾ (پ:16، سورهء مريم آيت 54،55)

واعدي جا سچا!

مٺا مٺا اسلامي ڀائرو! هتي اها ڳالھ ذهن نشين فرمائي وٺو ته سڀ انبياءِ ڪرام عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام واعدي جا سچا آهن، پر حضرت سَيّدُنا اِسمٰعيل عَـلَيْـهِ الـسَّلَام ان صفت ۾ خاص شهرت رکندا آهن. پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن ننڍپڻ ۾ پنهنجي والدِ محترم حضرت سَيّدُنا ابراهيم خليل الله عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام سان ڪيل واعدو ذبح جي وقت صبر ۽ رضا سان گڏ بخوبي پورو فرمايو.

هڪ ڀيري پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام هڪ شخص سان هڪ جڳھ تي ملڻ جو واعدو ڪيو ۽ پاڻ ان مقام تي پهچي ويا پر جنهن شخص کي اچڻو هو ان کان وسري ويو ايستائين جو شام ٿي وئي، پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام انهيءَ جڳھ تي پوري رات گذاري ڇڏي. صبح جڏهن اهو شخص آيو ته توهان کي موجود ڏٺائين ته حيرت ۾ پئجي ويو ۽ عرض ڪيو:توهان هتان کان ويا ناهيو؟ ته پاڻ عَـلَيْـهِ الـسَّلَام ارشاد فرمايو: نه، تنهنجي اچڻ کان پهريان آئون ڪيئن هليو وڃان ها.

حضور پُرنور، سيّدُ العٰلمين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمائن ٿا: ”واعدي خلافي هي ناهي ته ماڻهو واعدو ڪري ۽ ان جي نيت ان کي پورو ڪرڻ جي به هجي بلڪه واعدي خلافي هي آهي ته ماڻهو واعدو ڪري ۽ ان جي نيت ان کي پورو ڪرڻ جي نه هجي.“ (الجامع لاخلاق الراوي للخطيب البغدادي، ص۳۱۵، رقم۱۱۶۸) هڪ ٻي حديثِ پاڪ ۾ آهي ته جڏهن ڪو شخص پنهنجي ڀاءُ سان واعدو ڪري ۽ ان جي نيت ان کي پورو ڪرڻ جي هجي پوءِ پورو نه ڪري سگھي، واعدي تي نه اچي سگھي ته ان تي گناھ ناهي.

(سُنَنِ ابوداؤد ج۴ ص ۳۸۸ حديث ۴۹۹۵)

ڳالھ اثر انداز نه ٿيندي ۽ گھربل نتيجا حاصل ٿيڻ دشوار ٿيندا. تنهن جي ڪري پنهنجي اولاد کي نماز وغيره جي ترغيب ڏيارڻ کان پهرين پاڻ نماز پڙهڻ جي مضبوط عادت ٺاهڻي پوندي. قرآنِ ۽ حديث ۾ نماز پڙهڻ وارن جي لاءِ ڪيتريون ئي بشارتون ۽ نماز جون بي شمار فضيلتون بيان ڪيون ويون آهن، اچو ته علمِ دين حاصل ڪرڻ ۽ نماز جي پابندي جو ذهن بڻائڻ جي لاءِ فرمانِ خداوندي ٻڌون ٿا: سيپاري 6 سُورَۃُ الْمَائِدہ جي آيت نمبر 12 ۾ ارشاد ٿئي ٿو:

وَ لَقَدۡ اَخَذَ اللہُ مِیۡثَاقَ بَنِیۡۤ اِسۡرَآءِیۡلَ ۚ وَ بَعَثۡنَا مِنۡہُمُ اثۡنَیۡ عَشَرَ نَقِیۡبًا ؕ وَ قَالَ اللہُ اِنِّیۡ مَعَکُمۡ ؕ لَئِنۡ اَقَمۡتُمُ الصَّلٰوۃَ وَ اٰتَیۡتُمُ الزَّکٰوۃَ وَ اٰمَنۡتُمۡ بِرُسُلِیۡ وَ عَزَّرۡتُمُوۡہُمۡ وَ اَقۡرَضۡتُمُ اللہَ قَرۡضًا حَسَنًا لَّاُکَفِّرَنَّ عَنۡکُمۡ سَیِّاٰتِکُمۡ وَ لَاُدۡخِلَنَّکُمۡ جَنّٰتٍ تَجۡرِیۡ مِنۡ تَحۡتِہَا الۡاَنۡہٰرُ ۚ

(پ:6، سورهء المائده آيت 12)

ترجمو ڪنز الايمان: ۽ بي شڪ الله بني اسرائيل کان عهد ورتو ۽ اسان انهن مان ٻارنهن سردار قائم ڪيا ۽ الله فرمايو بي شڪ مان توهان سان گڏ آهيان ضرور جيڪڏهن توهان نماز قائم رکو ۽ زڪوات ڏيو ۽ منهنجي رسولن تي ايمان آڻيو ۽ انهن جي تعظيم ڪريو ۽ الله کي قرض حسنه ڏيو بي شڪ مان توهان جا گناھ لاهي ڇڏيندس ۽ ضرور توهان کي باغن ۾ وٺي ويندس جن جي هيٺان نهرون جاري آهن.

سُبحٰن الله عَزَّوَجَلَّ نمازين جي لاءِ الله عَزَّوَجَلَّ جي بارگاھ ۾ ڪهڙا ڪهڙا انعام آهن جو انهن کي جنت ۽ مغفرت جون بشارتون ڏنيون پيون وڃن ۽ اجرِ عظيم جون خوشخبريون به ٻڌايون پيون وڃن. احاديثِ مبارڪه ۾ به نماز جي وڏي اهميت ۽ رغبت ڏياري وئي آهي، اچو ته ترغيب جي لاءِ (2) فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌون ٿا.

2. جيڪڏهن توهان مان ڪنهن جي اڱڻ ۾ واھ هجي، هر روز پنج ڀيرا ان ۾ غسل ڪري ته ڇا ان تي ڪجھ مير رهندو؟ ماڻهن عرض ڪيو: جي نه. پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: نماز گناهن کي اهڙي نموني ڌوئي ڇڏيندي آهي جيئن پاڻي مير کي ڌوئي ڇڏيندو آهي.

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَن ْ زِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([1])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سيدُنا ابنِ عباس رَضِي اللّٰہُ تَعَالٰي عَنہُمَا کان رِوايت آهي ته سرڪارِ مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([2])

﴿2﴾ هر رات عبادت ۾ گذارڻ جو آسان نسخو

غرائب القرآن صفحه 187 تي هڪ روايت نقل ڪئي وئي آهي ته جيڪو شخص رات ۾ ٽي ڀيرا هي دعا پڙهندو ته ان ڄڻ شبِ قدر کي حاصل ڪري ورتو لهٰذا هر رات هي دعا پڙهڻ گهرجي. دعا هي آهي:

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

”يعني خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي.“

(فيضانِ سنّت ج اوّل، ص 1163، 1164)
