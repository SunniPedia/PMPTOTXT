بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندَس ٭ ٽيڪ لڳائي ويهڻ

بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندَس ٭ ضرورت پوڻ تي ٻين اسلامي ڀينرن جي لاءِ سِرڪي جاءِ ڪشاده ڪندَس ٭ ڌِڪو وغيره لڳو ته صبر ڪندَس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندَس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ واريءَ جي دِل جوئي جي لاءِ آهسته آواز سان جواب ڏيندَس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندَس ۔ دوران بيان موبائل جي غيرضروري استعمال کان بچندَس،نه بيان ريڪارڊ ڪندَس نه ئي ٻئي ڪنهن قسم جو آواز ڇو ته ان جي اجازت ناهي،جيڪو ڪجهه ٻُدَندَس،ان کي ٻڌي ۽ سمجهي ڪري ان تي عمل ڪرڻ ۽ ان کي بعد ۾ ٻين تائين پهچائي نيڪي جي دعوت عام ڪرڻ جي سعادت حاصل ڪندَس.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! اڄ اسان جي بَيان جو موضوع آهي ”امام مالڪ جو عشقِ رسول“ ، جنهن ۾ اسان عالِمِ مدينه حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو مختصر تعارف، سندن ديني خدمتون ۽ سندن جي باري ۾ فرمانِ رسول ٻڌڻ جي سعادت حاصل ڪنديون سين، شهرِ مدينه، خاڪ مدينه ۽ حديثِ پاڪ جي ادب ۽ تعظيم جي باري ۾ سندن سيرت جا ڪجھ ايمان افروز واقعا به ٻڌنديون سين، امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪيترا وڏا عبادت گذار هئا، تلاوتِ قرآن سان پاڻ کي ڪيتري مَحَبَّت هئي ۽ سندن عبادت جو انداز ڪيتري قدر پيارو هو ان باري ۾ عالم سڳورن جا ارشاد به اسان ٻڌنديون سين، الله ڪري ته دِلجمعي سان شروع کان آخر سٺين سٺين نيتن سان بيان ٻڌڻ جي سعادت حاصل ڪريون۔

امام مالڪ ۽ مسجدِ نبوي جو ادب

خليفي ابو جعفر منصور حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه سان مسجدِ نبوي شريف ۾ مناظرو ڪيو، مناظري دوران ابو جعفر جو آواز ڪجھ بلند

ٿيو ته امام مالڪ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه ان کي (نيڪي جي دعوت ڏيندي) فرمايائون: اي اميرُالمؤمنين! هن مسجد ۾ پنهنجو آواز بلند نه ڪيو ڇوته الله پاڪ هڪ جماعت کي ادب سيکاريندي ارشاد فرمايو:

یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا لَا تَرْفَعُوْۤا اَصْوَاتَكُمْ فَوْقَ صَوْتِ النَّبِیِّ (سيپاري:۲۶، الحجرات:۲)

تَرجَمَو ڪنز الايمان: اي ايمان وارؤ! پنهنجو آواز مٿي نه ڪريو هن نبي (غيب جون خبرون ٻڌائيندڙ) جي آواز کان ۔

ٻي جماعت جي ساراھ ڪندي ارشاد فرمايو ويو:

اِنَّ الَّذِیْنَ یَغُضُّوْنَ اَصْوَاتَهُمْ عِنْدَ رَسُوْلِ اللّٰهِ

ترجموڪنزالايمان ؛ بيشڪ جيڪي پنهنجو آواز جهڪو رکندا آهن الله جي رسول وٽ،

اِنَّ الَّذِیْنَ یُنَادُوْنَكَ مِنْ وَّرَآءِ الْحُجُرٰتِ

ترجموڪنزالايمان ؛ بيشڪ اهي جيڪي تو کي حُجري جي ٻاهران پُڪارن ٿا

پوءِ حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن فرمايائون: بيشڪ نور جي پيڪر، تمام نبين جي سرور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي عزت ۽ حرمت اڃا به اهڙي طرح آهي جهڙي طرح پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ظاهري حيات ۾ هئي، اهو ٻڌي ابو جعفر چُپ ٿي ويا، پوءِ پڇيائين: اي ابو عبدالله! مان قبلي ڏانهن منهن ڪري دعا گھران يا، رسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي طرف متوجھ ٿي ڪري؟ فرمايائون: توهان ڇو حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان منهن ڦيريو ٿا جڏهن ته حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم توهان جا ۽ توهان جي والد حضرت آدم عَـلَيْـهِ الـسَّـلاَم جا بروز قيامت الله پاڪ جي بارگاھ ۾ وسيلو آهن بلڪه سندن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ئي ڏانهن متوجھ ٿي پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان شفاعت گھرو پوءِ الله پاڪ پاڻ ڪريم صَلَّى اللّٰهُ تعاليٰ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي شفاعت قبول فرمائيندو . (شفاء،القسم الثاني،الباب الثالث، فصل واعلم انّ حرمۃ النبي۔۔۔ الخ، الجزء الثاني ، ص۴۱)

اعليٰ حضرت، امامِ اهلسنت مولانا شاھ امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه بارگاھ رسالت ۾ عرض ڪن ٿا:

تجھ سے چھپاؤں منہ تو کروں کس کے سامنے کیا اور بھی کسی سے توقع نظر کی ہے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! ٻڌو اوهان! ڪروڙين مالِڪين جي عظيم پيشوا حضرت سيدُنا امام مالِڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪيترا زبردست عاشقِ رسول هئا، پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي ذات جي لاءِ ته سڀ ڪجھ برداشت ڪري وٺندا هئا، پر جيڪڏهن ڪنهن شخص کي سندن جي ۽ اسان جي آقا و موليٰ،جنابِ احمد مجتبيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مسجد شريف ۾ آواز بلند ڪري بي ادبي ڪندي ڏسندا هئا ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي غيرتِ ايماني کي جوش اچي ويندو هو، پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ان جي ان ناگوار حرڪت تي چپ ڪري نه ويهندا هئا بلڪ انهيءَ وقت ان کي نيڪي جي دعوت ڏيندي بارگاھ رسالت جا آداب (Manners) ياد ڏياريندا هئا ته هي اهو مقدس مقام آهي جنهن جا آداب اسان جي پياري ربِّ ڪريم پنهنجي پاڪيزه ڪلام قرآنِ ڪريم ۾ بيان ڪيا آهن۔ هن واقعي ۾ جتي حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي عشقِ رسول جي خبر پئي اتي هي به معلوم ٿيو ته حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه نيڪي جي دعوت جو ڪو موقعو هٿن مان وڃڻ نه ڏيندا هئا، پر افسوس! هاڻي نيڪي جي دعوت جو جذبو گهٽ ٿيندو پيو وڃي۔ جيڪڏهن اسان نيڪ عمل ڪري به وٺون ته ٻين اسلامي ڀينرن کي نيڪي جي دعوت ڏيڻ ۽ برائي کان روڪڻ ۾ سستي جو شڪار ٿي وينديون آهيون، جڏهن ته حديث سڳورين ۾ نيڪي جي دعوت ڏيڻ ۽ بُرائيءَ سان روڪڻ جون ترغيبون موجود آهن، جيئن

حُضُورِ پاڪ، صاحِبِ لَولاڪ،سَياحِ اَفلاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عبرت نشان آهي: توهان مان ڪو جڏهن ڪنهن بُرائي کي ڏسي ته ان کي گھرجي ته

بُرائي کي پنهنجي هٿ سان بدلي ڇڏي ۽ جيڪو پنهنجي هٿ سان بدلڻ جي قوت نه رکي، ان کي گھرجي ته پنهنجي زَبان سان بدلي ڇڏي ۽ جيڪو پنهنجي زَبان سان بدلڻ جي به اِستِطاعت نه رکي ان کي گھرجي ته پنهنجي دل ۾ بُرو سمجھي ۽ اها ڪمزور ترين ايمان جي نشاني آهي۔ (مسلم، ڪتاب الايمان، باب بيان ڪون النھي عن المنڪر۔۔۔الخ، ص۴۸، حديث: ۱۷۷)

ڇا اسان دل ۾ بُرو سمجھون ٿيون؟

حضرت سَيدُنا مُصعَب بن عبدالله رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جڏهن حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي سامهون نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو ذڪر ڪيو ويندو ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي چهري جو رنگ بدلجي (Change) ويندو هو ۽ چيلھ مبارڪ جھڪي پوندي هئي، هڪ ڏينهن حاضرين امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه

کان ان جي ان ڪيفيت جي باري ۾ پڇيو ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمايائون: جيڪو ڪجھ مون ڏٺو آهي، توهان ڏسو ها ته مون تي اعتراض نه ڪيو ها. مون قارين جي سردار حضرت سَيدُنا محمد بن منڪدِر رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان جڏهن به ڪا حديث پڇا ڪئي ته هو عظمتِ حديث ۽ يادِ رسول ۾ روئي پوندا هئا ايستائين جو مون کي ان جي حال تي رحم اچڻ لڳندو هو۔ (الشفاء،الباب الثالث، ۲/۴۲ملخصاً) اي ڪاش! اسان کي به عشقِ رسول ۽ يادِ نبي ۾ روئڻ نصيب ٿي وڃي۔ اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ

ترے عشق میں کاش! روتا رہوں میں رہے تیری اُلفت سے معمور سینہ

امامِ مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ۽ تعظيمِ خاڪ مدينه

حضرت سَيدُنا امام شافعي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: مون حضرت سيدنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي دروازي تي خراسان ۽ مصر جا گهوڙا ٻڌل ڏٺا، ايترا اعليٰ قسم جا گهوڙا مون ڪڏهن به نه ڏٺا هئا، ان ڪري مون عرض ڪيو؛ اهي گهوڙا ڪيڏا نه بهترين آهن؟ پاڻ فرمايائون؛ مان اهي سڀ تو کي تحفي ۾ ڏيان ٿو، مون عرض ڪيو؛ پنهنجي لاءِ ته هڪ گھوڙو رکي ڇڏيو فرمايائون؛ مون کي الله پاڪ کان حياءُ ٿيو ٿئي ته هن مبارڪ زمين کي پنهنجي گهوڙي جي پيرن هيٺان لتاڙيان جنهن ۾ الله پاڪ جا رسول، رسولِ مقبول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تشريف فرما هجن ۔

(احياء علوم الدين،ڪتاب العلم،باب ثاني في العلم المحمود والمذموم واقسامھما واحڪامھما،ج۱/ ۴۸)

قضاءِ حاجت جي لاءِ حرم کان ٻاهر ويندا هئا

حضرت سيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي مديني پاڪ جي مٽيءَ جي تعظيم خاطر مدينه منوره ۾ ڪڏهن به قضاءِ حاجت نه ڪئي ان ڪري هميشه مديني پاڪ جي لاءِ حرم کان ٻاهر هليا ويندا هئا، پر مرض جي حالت ۾ مجبور هئا۔ (بستان المحدثين ،ص۱۹)

مدینہ اس لیے عطارؔ جان و دل سے ہے پیارا کہ رہتے ہیں مِرے آقا مرے سرور مدینے میں

8 مدني ڪمن مان هڪ مدني ڪم ”گهردرس“

پياريون پياريون اسلامي ڀينرون! ٻڌو اوهان! حضرت سَيدُنا امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪيترا زبردست عاشقِ رسول هئا، جيڪي نه صرف پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ذات بلڪ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان نسبت رکڻ جي سبب حديثِ رسول، خاڪ مدينه ۽ شهرِ مصطفٰي سان ڪيتري قدر عقيدت ۽ محبت ڪندا هئا، اي ڪاش! امام مالڪ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جي صدقي اسان کي به حديثِ رسول جي ادب جي مدني سوچ نصيب ٿي وڃي ۽ اسان به شهرِ مُصطَفٰي جي حسين تصورات ۾ گم رهڻ جي عادت بڻايون ۔ اَلْحَمْدُ لِلّٰه! عاشقانِ رسول جي مدني تحريڪ دعوتِ اِسلامي جي مَدَني ماحول سان وابسته رهڻ عشقِ رسول حاصل ڪرڻ جو بهترين ذريعو آهي، تنهنڪري اوهان به هن مدني ماحول سان وابسته رهو ۽ ذيلي حلقي جي 8 مدني ڪمن ۾ ڪجھ نه ڪجھ وقت ضرور ڏيو ۔ ياد رهي! ذيلي حلقي جي 8 مدني ڪمن مان هڪ مدني ڪم گهر درس به آهي توهان سڀني اسلامي ڀينرن کي گزارش آهي ته گهر ۾ مدني ماحول بڻائڻ جي لاءِ روزانو گهٽ ۾ گهٽ هڪ ڀيرو فيضانِ سنت جو درس ڏيڻ يا ٻڌڻ جي ترڪيب ضرور بڻايو (جنهن ۾ نا محرم نه هجن) امير اهلسنت دامت برڪاتهم العاليه جي تخريج شده رسالن مان به درس ڏئي سگهجي ٿو(دورانيو 7 منٽ آهي) ماڻهن تائين علم دين جون ڳالهيون پهچائڻ اجر ۽ ثواب جو ڪم آهي، جيئن ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: جيڪو شخص منهنجي امت تائين ڪا اسلامي ڳالھ پهچائي ته جيئن ان سان سنت قائم ڪئي وڃي يا ان سان بدمذهبي دور ڪئي وڃي اهو جنتي آهي، لهٰذا توهان به درس ڏيڻ جي نيت ڪري ڇڏيو، اچو! ترغيب جي لاءِ هڪ مدني بهار ملاحظو فرمايو

دل جي ڪيفيت بدلجي وئي

باب المدينه (ڪراچي) جي هڪ اسلامي ڀيڻ کي ڊائجسٽ پڙهڻ جو تمام گهڻو

شوق هو ان جي زندگي ۾ عمل جي بهار ڪجھ هن طرح آئي، ان جو وڏو ڀاءُ جيڪو دعوتِ اسلامي جي مدني ماحول سان وابسته هو اهو مختلف رسالا وٺي گهر آڻيندو هو جنهن کي پڙهي ڪري ان اسلامي ڀيڻ کي نهايت سٺو لڳندو هو ، آهستي آهستي دعوت اسلامي جي اجتماعن ۾ شرڪت جي برڪت سان مدني ماحول سان وابسته ٿي وئي ۽ ان بي پردگي کان توبه ڪئي ۽ مدني برقعو سجائڻ، نمازن جي پابندي ڪرڻ جو عزم ڪري ورتو ۽ نيڪيون ڪرڻ ۾ مصروف ٿي وئي پر دعوت اسلامي جو مدني ڪم ڪرڻ کان شرمائيندي هئي، سندس ڀاءُ باربار ان کي گهر ۾ فيضان سنت جو درس ڏيڻ جي ترغيب ڏيندو هو پر هوءَ درس ڏيڻ کان گهٻرائيندي هئي، اهڙي طرح سلسلو هلندو رهيو، هڪ ڏينهن سندس ڀاءُ اجتماع کان واپسي تي ” بھیانک اُونٹ “ نالي هڪ رسالو آندو، رسالي جو نالو نهايت وڻندڙ هو، تنهنڪري ان اسلامي ڀيڻ رسالي کي پڙهڻ شروع ڪيو ۽ جڏهن هنن لفظن کي پڙهيائين ” اسان جي آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي ماڻهن پٿر هنيا پر پوءِ به پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن دين جي تبليغ فرمائي ۽ ماڻهو اسان تي گل نڇاور ڪن ٿا پوءِ به اسان جيءُ چرايون، اسان پکن، قالينن تي ويهي به دين جي تبليغ نه ڪيون “ هي پڙهي ڪري سندس جي ڪيفيت بدلجي وئي، جڏهن پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي عظيم قربانين جي متعلق خبر پئي ته خود بخود ان جي اکين مان لڙڪ جاري ٿي ويا، ان همت ڪري درس فيضانِ سنت ڏيڻ جو ارادو ڪري ورتو ۽ جلدي ئي پنهنجي گهر ۾ درس ڏيڻ شروع ڪيائين جنهن ۾ پاڙي جون اسلامي ڀينرون به شرڪت ڪرڻ لڳيون، فيضانِ سنت جي درس جي برڪت سان اسلامي ڀينرون عمل جي طرف رغبت ڪرڻ لڳيون، الله پاڪ جي ڪرم سان ٻين اسلامي ڀينرن به درس ڏيڻ شروع ڪري ورتو جلد ئي انهن جي علائقي ۾ ٽن جڳهن تي درس فيضان سنت شروع ٿي ويو.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! ياد رکو! جهڙي طرح عالِمِ مدينه عشقِ رسول جي لازوال دولت سان مالامال هئا، جهڙي طرح عالِمِ مدينه هڪ سچا عاشقِ رسول هئا، جهڙي طرح عالِمِ مدينه جو سينو شهرِ مُصطَفٰي جي محبت سان سرشار هو، جهڙي طرح عالِمِ مدينه نسبتِ رسول جي اهميت ۽ فضيلت سان پوري طرح واقف ها، جهڙي طرح عالِمِ مدينه حديثِ رسول جو دلنشين انداز ۾ ادب و احترام ڪندا هئا، جهڙي طرح عالِمِ مدينه حديثِ رسول جي خدمت جي ڪري عوام ۽ خواص ۾ مشهور ۽ معروف هئا، اهڙي طرح عالِمِ مدينه جي مبارڪ سيرت جو هڪ دلڪش پهلو هي به آهي ته پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه عبادت و رياضت ۽ تلاوتِ قرآن جا به تمام گھڻا شيدائي هئا۔ اچو! عالم سڳورن جي زباني سندن عبادت، رياضت ۽ تلاوتِ قرآن سان محبت جون ڪجھ ايمان افروز جھلڪيون ٻڌو ۽ عبادت و رياضت ۾ مصروف ٿي وڃو،

امام مالڪ جي عبادت ۽ رياضت بابت بزرگن جا قول

٭ حضرت سيدُنا علامه قاضي عياض مالڪي رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه لکن ٿا ته حضرت سَيدُنَا زبير بن حبيب رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جن فرمايو: جڏهن به ڪنهن (اسلامي) مهيني جي آمد ٿيندي هئي ته امام مالڪ ان مهيني جي پهرين رات جو شب بيداري ڪندا هئا، ٭ حضرت سيدنا زبير بن حبيب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: منهنجي خيال ۾ اوهان اها عبادت مهيني جو استقبال ڪرڻ جي غرض سان ڪندا هئا، ٭ صاحبزادي فاطمه بن مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْهَا فرمائن ٿيون: امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هر رات پنهنجو وظيفو پورو ڪندا هئا ۽ جڏهن جمعي جي رات ايندي هئي ته سڄي رات الله پاڪ جي عبادت ۾ مشروف ٿي ويندا هئا، ٭ حضرت سَيدُنَا مغيره رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جو بيان آهي: هڪ ڀيري رات جي وقت ماڻهن جي سمهڻ کان پوءِ مان امام مالڪ بن انس وٽان لنگھيس، مون ڏٺو ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه بيهي نماز ۾ مشغول آهن، جڏهن پاڻ سوره فاتحه پڙھي چڪا ته سوره ” اَلْهٰکمُ التَّکاثُرُ “ شروع ڪري ڇڏي، امام مالڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جڏهن هن آيت سڳوري تي پهتا:

ثُمَّ لَتُسْــٴَـلُنَّ یَوْمَىٕذٍ عَنِ النَّعِیْمِ۠(۸) (پ،۳۰،التڪاثر:۸)

ترجمو ڪنزالايمان: پوءِ بيشڪ ضرور ان ڏينهن توهان کان نعمتن جي پڇا ٿيندي ـ

ته گھڻي دير تائين روئندا رهيا، مان پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جي تلاوت ٻڌڻ ۾ مشغول ٿي ويس ۽ اتي بيٺو رهيس، پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه انهيءَ آيت کي بار بار پڙهندا رهيا ۽ روئندا رهيا ايستائين جو صبح چمڪڻ لڳي، پوءِ پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه رڪوع ڪيائون۔ مان پنهنجي گھر ڏانهن روانو ٿيس، مان وضو ڪري مسجد ۾ حاضر ٿيس ته ڇا ٿو ڏسان ته مسجد ۾ (علمِ دين جي) مجلس قائم آهي، ماڻهو سندن جي آسپاس حلقو (Circle) ٺاهي ويٺل آهن ۽ پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جي مبارڪ چهري ۾ خوبصورت نور چمڪي رهيو آهي۔ ٭ محمد بن خالد رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه فرمائن ٿا: جڏهن به ان امام مالڪ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه جو مبارڪ چهرو ڏسان ٿو ته سندن چهري ۾ مون کي آخرت (جو خوف رکڻ وارن) جون نشانيون نظر اچن ٿيون، جڏهن اوهان گفتگو ڪندا آهن ته مان ڄاڻي وٺندو آهيان ته حق سندن جي واتان نڪرندو آهي، ٭ حضرت ابو مصعب رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه فرمائن ٿا: امام مالڪ رات جي هڪ حصي ۾ ڊگھا رڪوع ۽ سجدا ادا ڪندا هئا، جڏهن پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه نماز ۾ بيهندا هئا ته ايئن لڳندو هو جيئن ڪا سڪل ڪاٺي هجي، جڏهن پاڻ رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه کي دُرن جي سزا ڏني وئي هئي ان وقت سندن رَحْمَةُ اللّٰهِ تعاليٰ عَلَيۡه کي چيو ويو: مختصر نماز پڙھي وٺو، فرمايائون: ٻانهي کي گھرجي ته هو الله پاڪ جي لاءِ جيڪو به عمل ڪري، چڱي طرح ڪري۔ (سيپاروه 29 سُورَ ۃ المُلڪ جي آيت نمبر 8 ۾) الله پاڪ فرمائي ٿو:

لِیَبْلُوَكُمْ اَیُّكُمْ اَحْسَنُ عَمَلًاؕ (پ:۲۹، المُلڪ:۲)

ص۸۷،رقم:۱۱۸) ٭ جيڪڏهن ڪا رڪاوٽ نه هجي ته رستي جي ڪناري ڪناري درمياني چال هلو، نه ايترو تيز جو ماڻهن جون نگاهون اوهان ڏانهن ٿين ۽ نه ايترو آهستي جو اوهان بيمار لڳو ٭ رستي تي هلندي پريشان نظري(يعني بلاضرورت هيڏانهن هوڏانهن ڏسڻ) سنت ناهي، هيٺ نگاهه ڪري باوقار طريقي سان هلو ٭ هلڻ يا ڏاڪڻ تان لهڻ مهل هي احتياط ڪيو ته جوتن جو آواز پيدا نه ٿئي ٭ ڪجھ ماڻهن جي عادت هوندي آهي ته رستي تي هلندي جيڪا به شيء سامهون آئي ان کي لتون هڻندي گذرندا آهن، هي بلڪل غير مُهَذَّب طريقو آهي ائين ڪرڻ سان ڪنهن جو پير زخمي ٿيڻ جو انديشو آهي ۽ اخبار يا لکائي وارن ڊٻن، پيڪٽن، ۽ مِنرل واٽر جي خالي بوتلن وغيره کي لت هڻڻ به بي ادبي آهي.

ايندڙ هفتيواراجتماع جي بيان جون جھلڪيون

پياريون پياريون اسلامي ڀينرون! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! آئنده هفتي جي بَيان جو موضوع آهي. ”امدادِ مصطفيٰ جا واقعا“ هن بيان ۾ اصحاب ڪرام، بزرگانِ دين، ۽ ٻين ماڻهن ايستائن جو جانورن جي رسول اڪرم صَلَّى اللّٰهُ تعاليٰ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي بارگاه ۾ فرياد ڪرڻ ۽ انهن جي فرياد ڪرڻ تي نبي رحمت، شفيع امت، تاجدارِ رسالت صَلَّى اللّٰهُ تعاليٰ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جي انهن جي امداد فرمائڻ جا ڪجھ ايمان افروز واقعا بيان ٿيندا، تنهنڪري اچڻ واري اربع تي به اجتماع ۾ حاضري جي نيت ڪيو ۽ هٿ مٿي ڪري چئو؛ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ!
