مڪان ۽ ڪِرائيدارن ۾ هاٿا پائي ٿي رهي آهي، ٭ ڪٿي ڪنڊيڪٽر ۽ سُوارين ۾ لڙائي جھيڙو ٿي رهيو آهي ته ڪٿي ريڙهو لڳائڻ وارن ۾ تڪرار ٿي رهيو آهي، ٭ ڪٿي ڊاڪٽر (Doctor) ۽ مريض هڪ ٻئي سان بي صبري جو مظاهرو ڪري رهيا آهن ته ڪٿي ٺيڪيدار ۽ ملازم پاڻ ۾ ڳنڍجي پيا آهن، ٭ ڪٿي پاڙيسري هڪ ٻئي جي رت جا پياسا آهن ته ڪٿي رشتيدارن ۾ ناراضي آهي، ٭ ڪٿي امام ۽ مقتدين ۾ دُوريون وڌنديون پيون وڃن ته ڪٿي مسجد ڪميٽي ۽ نمازي حضرات پاڻ ۾ ناراض آهن، ٭ ڪٿي سالن جي گھاٽن دوستن (Friends) ۾ اَڻبڻت هلي رهي آهي ته ڪٿي سڄو گھر ئي ميدانِ جنگ بڻيل آهي۔ ٭ ڪٿي خوني رشتا ۽ انهن جو احترام داؤ تي لڳل آهي ته ڪٿي سڳن ڀائرن ۾ دوريون وڌي چڪيون آهن۔ مطلب ته! ٭ اهي جيڪي ڪل تائين هڪ ٻئي جا محافظ هئا، ٭ اهي جيڪي ڪل تائين هڪ ٻئي تي جان قربان ڪرڻ جي دعوا ڪندا هئا، ٭ اهي جيڪي ڪل تائين هڪ ٻئي جي عزّت جا مُحافِظ هئا، ٭ اهي جو ڪلهه تائين جن جي دوستي جو مثال ڏنو ويندو هو، ٭ اهي جو ڪل تائين جن جي اِتِّفاق ۽ ٻڌي جي مشهوري هئي۔ ٭ اهي جيڪي ڪل تائين هڪ ٻئي جي خلاف هڪ لفظ تائين ٻڌڻ به پسند نه ڪندا هئا، ٭ اهي جيڪي ڪل تائين هڪ ٻئي جي بغير ماني تائين نه کائيندا هئا، ٭ اهي جيڪي ڪل تائين بُري وَقت ۾ هڪ ٻئي جا مددگار هئا، ٭ اهي جيڪي ڪل تائين هڪ ٻئي کي نيڪي جي ڪمن جون ترغيبون ڏيندا هئا، ٭ اهي جيڪي ڪل تائين سنتن ڀري اجتماعن ۾ گڏ ايندا ويندا هئا، ٭ اهي جيڪي ڪل تائين دعوتِ اسلامي جي مدني ڪمن ۾ هڪ ٻئي جي حوصله افزائي ڪندا هئا، لڙائي جھيڙي جهڙي مَنحوس شيطاني ڪم جي نَحوست جي سبب اُنهن جي وچ ۾ نفرتن جون اهڙيون مضبوط ديوارون قائم ٿي وينديون آهن جو پوءِ اهي هڪ ٻئي کي ڏسڻ به پسند ناهن ڪندا۔ ائين سمجھو ته جهڙي نموني باھ (Fire) گھرن، فيڪٽرين، ڪمپنين، گودامن، جهنگلن، ڳوٺن ۽ مختلف شين کي ڪلاڪن بلڪه مِنٽن ۾ ساڙي تباھ ۽ برباد ڪندي آهي، اهڙي طرح خوش

۽ آباد ملڪن، شهرن، نسلن، قومن، گھرن، خاندانن، ادارن ۽ تنظيمن جو اَمن تباھ برباد ڪرڻ ۽ دلين ۾ نفرتن جو ٻج پوکڻ ۾ اڪثر لڙائي جھڳڙن جون تباھ ڪاريون ئي شامل هونديون آهن۔ يقيناً جيڪڏهن اسان قُرآني اَحڪام کي نظر انداز نه ڪريون ها، جيڪڏهن اسان رسولِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي فَرمانن تي عمل پيرا ٿيون ها، جيڪڏهن اسان پنهنجي بُزرگ سڳورن رَحْمَةُ اللّٰهِ عَلَيۡهم اَجْمَعِين جي اِرشادن مان نصيحت جا مَدَني گل چُونڊيون ها، جيڪڏهن اسان عُلَماءِ حق جي دامنِ ڪرم سان وابسته رهون ها، جيڪڏهن اسان لڙائي جھيڙن جي تباھ ڪارين کي اکين آڏو رکون ها ته اڄ اسان جو معاشرو به اَمن ۽ سڪون وارو هجي ها۔

پيارا پيارا اسلامي ڀائرو! بزرگانِ دين جي معافي ۽ معاشري جي اصلاح جو جذبو مرحبا! اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! بُزرگانِ دين جي صحبت سان تربيتِ يافته شيخِ طريقت، اميراهلسنت باني دعوت اسلامي حضرت علامه و مولانا محمد الياس عطار قادري رضوي به معاشري جي اصلاح ڪرڻ جي معاملي ۾تمام گهڻا اِيڪِٽِوۡ (Active) آهن ۔ پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ اصلاحِ معاشره جي لاءِ ڪيئي اهم ڪمن ۾ مصروف آهن، پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ ماڻهن کي نرمي ۽ محبت سان نيڪي جي دعوت ڏئي گناهن کان روڪڻ جي لاءِ هردم ڪوشش ڪري رهيا آهن ۔ اَميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جتي انفرادي ڪوشش جي ذريعي معاشري جي اصلاح ڪري رهيا آهن، اتي پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ پنهنجي بيانن ۽ مدني مذاڪرن جي ذريعي به معاشري جي اصلاح ۾ اهم ڪردار ادا ڪري رهيا آهن، اميراهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جا بيانَ ۽ مدني مذاڪرا تاثير جو تير بڻجي اثرانداز ٿي رهيا آهن ۔ سندن جي سنّتن ڀري اِصلاحي بيانن ۽ مدني مذاڪرن کي ٻڌڻ وارن جي شوق جو عَالَم ڏسڻ جي قابل هوندو آهي ۔ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي جا سنتن ڀريا اجتماع هجن يا هر هفتي ٿيندڙ مدني

مذاڪرو هجي، ڪثير عاشقانِ رسول سندن بيان ۽ رنگ برنگي مدني گلن سان مستفيض ٿيندا آهن، مدني چينل، انٽرنيٽ ذريعي ڏسڻ ۽ ٻڌڻ وارن جو تعداد ان کان علاوه آهي ۔ صرف اهو ئي نه، بلڪ سندن بيان ۽ مدني مذاڪرا مدني چينل جي ذريعي گھرن، دڪانن (Shops) ، مسجدن، جامعات وغيره ۾ به نهايت شوق سان ٻڌا ويندا آهن ۔ انهن بيانن کي مڪتبة المدينه شايع ڪندو آهي ۔ سندس جو اندازِ بيان تمام سادو ۽ آسان ۽ سمجھائڻ جو طريقو اهڙو همدردي وارو هوندو آهي جو ٻڌڻ واري جي دل ۾ لهي ويندو آهي ۔ لکين مسلمان سندن جي بيانن جي برڪت سان توبه ڪري سڌي واٽ تي اچي چڪا آهن ۔ اچو! پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي بيان جو هڪ واقعو ٻڌون ٿا:

نمازي ڊ اڪو

باب المدينه (ڪراچي) جي هڪ ذمه دار اسلامي ڀاءُ جو بيان آهي منهنجو هڪ دوست تمام گھڻو ماڊرن، جوا ۽ شراب جو عادي ۽ گناهن تي وڏو دلير هو۔ ڪنهن طرح مڙندو نه هو، هڪ ڀيري باب المدينه کان ڪولمبو روانگي مهل، مون ان جي سامان ۾ اميراهلسنت جي سنتن ڀري بيان جي ڪيسٽ ''نمازي ڊاڪو'' وجھي ڇڏي۔ ڪولمبو پهچي ان بيان ٻڌو۔ امير اهلسنت جا لفظ تاثير جو تير بڻجي ان جي جگر ۾ سمائجي ويا۔ ان بيان جي برڪت سان ان ۾ حيرت انگيز طور تي مدني انقلاب اچي ويو۔ ايستائين جو ان ڏاڙھي ۽ سبز سبز عمامي شريف جو مدني تاج مستقل طور تي سجائي ورتو۔ دعوت اسلامي جي مدني ڪمن ۾ مشغوليت اختيار ڪري ورتي ۽ امير اهلسنت سان بيعت ٿي ڪري عطاري ٿيڻ جي سعادت به ماڻي ورتي۔ (تعارف امير اهلسنت،۲۶)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

اميرِ اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جو فيضان

پيارا پيارا اسلامي ڀائرو! اميرِ اهلسنَّت، حضرتِ علامه مولانا ابو بلال محمدالياس عطار قادري رَضَوي ضِيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ کي جڏهن نيڪي جي

دعوت عام ڪرڻ ۽ معاشري جي اصلاح جي مقدس جذبي تحت ماڻهن کي بُرائي کان روڪڻ جي تڙپ دل ۾ پيدا ٿي ته پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ اُمّتِ مسلمه کي نماز روزي جو پابند بڻائڻ، سُنَّتون سکڻ سيکارڻ، نيڪين جو جَذبو ڏيارڻ، گُناهن کان بچڻ جو ذِهن بڻائڻ جي لاءِ اِنفرادي ۽ اجتماعي ڪوشش ۽ بيانَ ڪرڻ شُروع ڪيائون۔ پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ نه صِرف خود اِستقامت سان نيڪي جي دعوت عام ڪندا رهيا بلڪ وقت به وقت اسلامي ڀائرن کي به پنهنجي ۽ سڄي دُنيا جي ماڻهن جي اِصلاح جي ڪوشش ڪرڻ جو ذِهن ڏيندا رهيا۔ هن مدني مقصد (Aim) کي دُنيا جي ڪنڊ ڪنڊ ۾ عام ڪرڻ جي لاءِ دعوت اسلامي جو بنياد رکيائون ۽ ان ۾ زندگي جي تقريباً هر شعبي سان وابسته ماڻهن جي اصلاح جي لاءِ ڪيئي شعبن کي ترتيب ڏنائون ته جيئن ان شعبي سان وابسته اسلامي ڀائر ماڻهن کي نمازن جي پابندي، تقوي ۽ پرهيزگاري، معاشري ۾ رهڻ سهڻ جو انداز سيکاري معاشري جي اصلاح ۾ اهم ڪردار ادا ڪن، پاڻ دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ پنهنجن سُنَّتن ڀري بيانن ۽ ڪتابن ۽ رسالن جي ذَريعي نيڪي جي دعوت جون خُوب خُوب ڌُومون مچايون۔ سندن جي آسان، پُر تاثير تحرير ۽ سُنَّتن ڀري بيانن جي بَرڪت سان لکين مُسلمان خاص ڪري نوجوانن جي زِندگين ۾ مدني اِنقلاب پيدا ٿي ويو ۽ اهي جُرم جي دُنيا کي ڇڏي ڪري صلوٰ ۃ وسُنَّت جي واٽ تي هلڻ شروع ٿي ويا۔ جيڪڏهن اسان به نيڪي جي دعوت ڏيڻ ۽ ٻين کي بُرائي کان روڪڻ جو ذِهن رکو ٿا ۽ پنهنجي معاشري کي امن ۽ سڪون جو گهوارو بڻائڻ چاهيو ٿا ته دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي وڃو، ٻين کي به هن ماحول سان وابسته ڪريو، پاڻ به مدني انعامات جا عامل بڻجو، ٻين کي به بڻايو، پاڻ به هر مهيني ٽن (3) ڏينهن جي مدني قافلي ۾ سفر ڪريو ٻين کي به سفر ڪرايو۔

معاشري جي اصلاح سان سرشار امير اهلسنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ پنهنجن مُريدن ۽ مُحِبن ۽ تمام عاشقانِ رسول کي سنتن جي مطابق زندگي گذارڻ جي لاءِ مدني انعامات عطا فرمايا آهن ان جي مدني انعام نمبر 61 ۾ آهي: ڇا هن

مهيني اوهان جي انفرادي ڪوشش سان گھٽ مان گھٽ هڪ اسلامي ڀاءُ مدني قافلي ۾ سفر ڪيو؟ ۽ گھٽ مان گھٽ هڪ مدني انعامات جو رسالو جمع ڪرايو؟

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

(3)معاشري جا مختلف گناهه

پيارا پيارا اسلامي ڀائرو! بيان جي شروع ۾ جهڙي طرح اسان ٻڌو ته معاشرو فرد سان بڻجندو آهي، جيڪڏهن هرهڪ فرد پنهنجي اصلاح جي ڪوشش ڪرڻ ۾ لڳي وڃي ته معاشرو خود بخود بهتر ٿيندو ويندو۔ معاشري ۾ هن وقت جيڪي گناه ٿي رهيا آهن انهن مان هڪ غيبت به آهي جيڪا معاشري جي تباهي جو سبب بڻجي رهي آهي۔

غيبت جي تعريف: انسان جي ڪنهن اهڙي عيب جو ذڪر ڪرڻ، جيڪو ان ۾ موجود هجي غيبت چورائيندو آهي. غيبت جي نحوست منجهان هي آهي ته اها بُري خاتمي جو سبب آهي، ڪثرت سان غيبت ڪرڻ واري جي دُعا قَبول ناهي ٿيندي، غيبت سان نَماز روزي جي نُورانيت ختم ٿي ويندي آهي، غيبت جي نحوست جو اَندازو هن فرمانِ مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مان لڳائي سگھجي ٿو ته ارشاد فرمايائين: اَلْغِيبَۃُ اَشَدُّ مِنَ الزِّنَا يعني غيبت بدڪاري کان وڏو گناھ آهي۔ (الترغيب والترهيب ، ۳، / ۳۳۱ ، حديث:۲۴) اهڙي طرح چغلي جو به معاشري جي تباهي ۾ وڏو ڪردار آهي۔ ان جي ڪري به گھرن جي بربادي، پاڻ ۾ رنجشون ۽ بغض ۽ ڪينو پيدا ٿيندو آهي ۽ اهڙي شخص کي الله پاڪ به پسند ناهي ڪندو.

چغلي جي تعريف: ماڻهن ۾ فساد ڪرائڻ جي لاءِ انهن جون ڳالهيون هڪٻئي تائين پهچائڻ چغلي چورائيندو آهي. (شرح مسلم للنووی، ۲/ ۱۱۲) حديث ِپاڪ ۾ آهي ته الله پاڪ جا نيڪ ٻانها اهي آهن جن کي ڏسجي ته الله پاڪ ياد اچي وڃي ۽ الله پاڪ جا بُرا ٻانها اهي آهن جيڪي چُغل خوري ڪن ٿا، دوستن ۾ جدائي وجھن ٿا ۽ نيڪ ماڻهن جا عيب (Defect) تلاش ڪن ٿا۔ (مسند احمد، ۶/ ۲۹۱، حديث ۱۸۰۲۰)

چغلي وانگر گارگند به اسان جي معاشري ۾ وڌندڙ برن ڪمن مان هڪ آهي، ان سان به فتنو ۽ فساد پيدا ٿيندو آهي، مثال طور پاڻ ۾ نفرتون پيدا ٿينديون آهن، خون خرابو، جھيڙا ۽ کوڙ ٻيون تباھ ڪاريون ظاهر ٿينديون آهن۔

نبي ڪريم ،رؤف رحيم صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ جن فرمايو: ''مسلمان کي گار ڏيڻ پاڻ کي هلاڪت ۾ وجھڻ جي مترادف آهي۔'' (الترغيب والترھيب، ڪتاب الادب ،، ۳/۳۱۱، الحديث:۴۲۶۳) اهڙي طرح حسد به نهايت بري خصلت ۽ وڏي گناھ سان گڏوگڏ معاشري کي خراب ڪرڻ وارو ڪم آهي۔

حسد جي تعريف: ڪنهن وٽ ڪا نعمت ڏسي تمنا ڪرڻ ته ڪاش! هن کان اها نعمت کسجي مون کي ملي وڃي حسد چورائيندو آهي. (بُرے خاتمے کے اسباب،ص۱۳ ملخصاً) حاسد جي سڄي زِندگي ساڙ ۽ گهٻراهٽ جي باھ ۾ سڙندي رهندي آهي ۽ ان کي چين ۽ سڪون نصيب ناهي ٿيندو، حسد نيڪين کي اهڙي طرح کائي ويندو آهي جيئن باھ ڪاٺيءَ کي۔ ايئن ئي تڪبُّر (Arrogance) کي ڏٺو وڃي ته ان جي سبب الله پاڪ ۽ رسولِ اڪرم، نبي مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ناراضي، مخلوق جي بيزاري، ميدانِ محشر ۾ ذلّت ۽ رُسوائي، رَبّ جي رحمت ۽ جنتي اِنعامن کان محرومي ۽ جهنَّم جو حقدار بڻجڻ جهڙن وڏن وڏن نُقصانن جو انديشو ٿي سگھي ٿو۔

تڪبر جي تعريف: پاڻ کي بهتر ۽ ٻين کي بيڪار سمجهڻ جو نالو تڪبر آهي. (تڪبر، ص16) نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: ''جنهن جي دل ۾ رائي جي داڻي جيترو به تَڪبُّر هوندو اهو جنّت ۾ داخل نه ٿيندو۔'' (مسلم،ڪتاب الايمان،باب تحريم الڪبروبيانه، الحديث:۲۶۶،ص۶۱)

ڪنهن جي نافرماني ڪريو ٿ ا

پيارا پيارا اسلامي ڀائرو! ٻڌو اوهان اهي گُناھ معاشري ۾ ڪهڙين ڪهڙين بُرائين کي جَنم ڏين ٿا۔ لهٰذا گُناھ چاهي ننڍو هجي يا وڏو ان کان بچڻ ۾ ئي عافيت آهي جيئن ته حضرت سَيدُنا بلال بن سعد رَضِىَ اللهُ عَنْه فرمائن ٿا: گُناھ جي ننڍي هئڻ کي نه ڏسو بلڪ اهو ڏسو ته توهان ڪنهن جي نافرماني ڪري رهيا

آهيو. (الزواجرعن اقتراف الڪبائر، مقدمة في تعريف الڪبيرة، خاتمة في التحذير۔۔الخ، ۱/۲۷) تنهنڪري پهرين ڳالھ ته هيءَ آهي جو ايترو علم هجي جو ٻانهو ظاهري ۽ باطني گناهن کي ڄاڻي سگهي. ظاهري ۽ باطني گناهن جي ضروري معلومات هئڻ به فرضن ۽ لازم علوم مان آهي. ٻين لازم علومن وانگر هن جي به معلومات لازم آهي. وڌيڪ جيڪڏهن گُناھ جو اِرادو ڪرڻ وقت اسان جي اها مدني سوچ بڻجي وڃي ته مان جنهن رَبِّ ڪريم جي نافرماني ڪري رهيو آهيان هو ته مون کي هر وَقت هر حال ۾ ڏسي رهيو آهي ته اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! هن طرح ڪافي حد تائين گُناهن کان ڇوٽڪارو نصيب ٿي ويندو۔ گُناهن کان نَفرت ڪرڻ ۽ ڇُوٽڪارو ماڻڻ جو هڪ بهترين ذَرِيعو ڪنهن چڱي ماحول سان وابسته ٿيڻ به آهي۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! اڄ جي هن پُرفِتَن دَور ۾ دعوتِ اسلامي جو مدني ماحول، الله پاڪ جي عظيم نعمت آهي۔ اوهان به هن مهڪندڙ مُشڪبار مدني ماحول سان هر دَم وابسته رهو، اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! دُنيا ۽ آخرت جون ڀلايون حاصل ٿينديون ۔

(2) ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
