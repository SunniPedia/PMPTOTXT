مٺا مٺا اسلامي ڀائرو! دنيا جي مَحَبَّت شيطان جو اهڙو ڄار آهي جيڪو ان ۾ ڦاسجي انسان نيڪ ڪمن کان پري ٿي ويندو آهي، مثال طور پهريان مستحبّات کان پري ٿيندو آهي، پوءِ سُنَّتن کان غافل ٿيندو آهي، ان کان بعد فرض ۽ واجب ڇڏڻ جي عادت بڻجندي آهي ۽ پوءِ آهستي آهستي حرام ڪمن جو عادي بڻجي ويندو آهي۔ نيٺ نمازون ڇڏڻ جو معمول بڻجي ويندو آهي، ڪوڙ ڳالهائڻ، غيبت ڪرڻ، ٻين جي دل ڏُکائڻ، ڳائڻ وڄائڻ ٻڌڻ ۽ طرح طرح جي حرام ۽ ناجائز ڪمن ۾ زندگي گذرڻ لڳندي آهي۔ ان جي علاوه دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو پنهنجن کي وساري ويهندو آهي، دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو مخلص دوستن کان محروم ٿي ويندو آهي، دنيا ۾ ضرورت کان وڌيڪ مشغول رهڻ وارو غريبن کي حقير ۽ ڪمتر سمجھڻ لڳندو آهي، دُنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو ڪنجوس (Miser) ٿي ويندو آهي، دُنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو تَڪبُّر جي آفت ۾ مبتلا ٿي ويندو آهي،دنيا ۾ ضرورت کان زِياده مشغول رهڻ واري تي نصيحت اثر ناهي ڪندي، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو حلال ۽ حرام ۾ تميز ناهي ڪري سگھندو، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو حُقُوْقُ اللہ سان گڏوگڏ حُقُوقُ العِبَاد يعني ٻانهن جي حقن کان به غافل ٿي ويندو آهي، دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو چورن، ڌاڙيلن ۽ لٽيرن جو شڪار ٿي ويندو آهي۔ مطلب ته دنيا ۾ ضرورت کان زِياده مشغول رهڻ وارو طرح طرح جي آفتن ۾ مبُتلا رهندو آهي۔دنيا ۾ ضرورت کان زِياده مشغول رهڻ وار جيڪڏهن دنيا جي حقيقت کي سمجھي وڃن ته ڪڏهن به ان سان دل نه لڳائن ها۔ قرآنِ ڪريم جي مختلف آيت سڳورين ۾ دُنيا جي حقيقت کي بيان ڪيو ويو آهي جيئن

سيپارو 27 سُورَۃُ الْحَدِيد جي آيت نمبر 20 ۾ ارشاد آهي:

اِعْلَمُوْۤا اَنَّمَا الْحَیٰوةُ الدُّنْیَا لَعِبٌ وَّ لَهْوٌ وَّ زِیْنَةٌ وَّ تَفَاخُرٌۢ بَیْنَكُمْ وَ تَكَاثُرٌ فِی الْاَمْوَالِ وَ الْاَوْلَادِؕ- (پ۲۷،الحديد: ۲۰)

تَرجَمو ڪنز الايمان: يقين ڄاڻو ته دنيا جي زندگي ته فقط راند روند ئي آهي ۽ سينگار ۽ توهان جو هڪٻئي ۾ فخر ڪرڻ ۽ مال ۽ اولاد ۾ هڪٻئي تي زيادتي جي خواهش.

بيان ڪيل آيت سڳوري تحت تفسير صِراطُ الجِنان ۾ لکيل آهي: هن آيت ۾ دُنيا جي حقيقت بيان ڪئي پئي وڃي ته جيئن مسلمان ان طرف راغب

نه ٿين ڇوته دُنيا تمام گھٽ نفعي واري ۽ جلد ختم ٿيڻ واري آهي۔ هن آيت ۾ اللہ پاڪ دُنيا جي باري ۾ پنج (5) شيون بيان ڪيون آهن (1،2)… دنيا جي زندگي ته صرف راند روند آهي جيڪو ٻارن جو ڪم آهي ۽ صرف ان جي حصول ۾ محنت ۽ مشقت ڪندو رهڻ وقت ضائع ڪرڻ کان سوا ٻيو ڪجھ نه آهي۔(3)… دنيا جي زندگي زِينت ۽ آرائش جو نالو آهي جيڪو عورتن جو شيوو آهي۔ (4،5)…دنيا جي زندگي پاڻ ۾ فَخر ۽ غرور ڪرڻ ۽ مال ۽ اولاد ۾ هڪ ٻئي تي زيادتي چاهڻ جو نالو آهي ۽ جڏهن دنيا جو هي حال آهي ۽ ان ۾ اهڙيون بُرائيون موجود آهن ته ان ۾ دل لڳائڻ ۽ ان جي حصول جي ڪوشش ڪندي رهڻ بدران انهن ڪمن ۾ مشغول هئڻ گھرجي، جن سان اُخروي زندگي سَنورجي سگھي ٿي۔ (صراط الجنان، 9/740)

شيطان جي ڌيءَ

حضرت سيدنا علي خَـوَّاص رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا؛ دنيا اِبليس لَعِين (يعني لعنتي شيطان) جي ڌيءَ آهي ۽ دنيا کان محبت ڪرڻ وارو هر شخص ان جي

ڌيءَ جو مڙس آهي، ابليس پنهنجي ڌيءَ جي ڪري ان دنيادار شخص وٽ ايندو ويندو آهي، تنهنڪري منهنجا ڀاءُ! جيڪڏهن اوهان شيطان کان محفوظ رهڻ چاهيو ٿا ته ان جي ڌيءَ (يعني دنيا) سان رشتو قائم نه ڪريو. (اَلْحَدِيقَۃُ النَّدِيۃ، ۱ / ۱۹ ملخصا)

نيري اکين واري بدصورت پوڙهي

حضرت سيدنا فُضَيل بن عِياض رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا، حضرت سيدنا عبد اللہ بن عباس رَضِىَ اللهُ تَعَالٰی عَنْهُما جن فرمايو؛ قيامت جي ڏينهن نيري اکين واري نهايت بدصورت پوڙهي جنهن جا ڏند اڳيان نڪتل هوندا ماڻهن جي سامهون ظاهر ٿيندي ۽ انهن کان پڇا ڪئي ويندي؛ هن کي سڃاڻو ٿا؟ ماڻهو چوندا؛ اسان ان جي سڃاڻپ کان اللہ عَزَّوَجَلَّ جي پناهه چاهيون ٿا، چيو ويندو؛ هي اها ئي دنيا آهي جنهن تي اوهان فخر ڪندا هئا، هن دنيا جي ڪري قطعِ رِحمي ڪندا يعني رشتيداريون ٽوڙيندا هئا، انهيءَ جي ڪري هڪٻئي سان حسد ۽ دشمني ڪندا هئا، پوءِ ان (پوڙهي جھڙي دنيا) کي جهنم ۾ وڌو ويندو ته رڙيون ڪندي؛ اي منهنجا پروردگار عَزَّوَجَلَّ! منهنجي پيروي ڪرڻ وارا ۽ منهنجي جماعت ڪٿي آهي؟ اللہ عَزَّوَجَلَّ فرمائيندو؛ انهن کي به ان سان گڏ ڪيو . (ذمُّ الدّنيا مع موسوعۃالامام ابن ابي الدنيا، ۵ / ۷۲ ،رقم۱۲۳)

مٺا مٺا اسلامي ڀائرو! اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ بُزرگانِ دين جو ڪردار اسان جي لاءِ مشعلِ راھ آهي، اهي حضرات دنيا (World) جي حقيقت کي چڱي طرح سڃاڻندا هئا۔ دُنيا انهن ڏانهن ايندي آهي پر اهي دُنيا جي مُقابلي ۾ هميشه آخرت کي ئي ترجيح ڏيندا آهن۔ اچو! ترغيب طور هڪ ايمان افروز حڪايت ٻڌون ٿا، جيئن ته

دنيا کان بي رغبت خليفو

اميرُالمومنين حضرت سَيدُنا عمربن عبدُالعزيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جڏهن خليفو بڻيو ته زُهد ۽ قَناعت (يعني دنيا کان بي رغبتي کي) اِختيار ڪري ورتو، عَيش ۽ عِشرت کي ڇڏي ڏنو، لذيذ کاڌا ڇڏي ڏنا ۽ تمام سادي غِذا استعمال ڪرڻ لڳا جيئن ته

نُعَيم بن سلامت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا ته ۾ اميرالمومنين عُمر بن عبدُ العَزِيز وٽ ويس ته ڏٺو ته پاڻ زيتون جي تيل سان ماني کائي رهيا هئا۔ (سيرتِ ابن جوزي، ص ۱۸۰) يونس بن شيب رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جو بيان آهي ته مون اميرالمومنين حضرت سَيدُنا عمر بن عبدالعزيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي خلافت کان پوءِ هن حال ۾ ڏٺو ته جيڪڏهن مان چاهيان ها ته بغير هٿ لڳائڻ جي سندن پاسراٽيون ڳڻي سگھان ها۔

(سيرتِ ابن جوزي، ص۱۸۱)

اوهان جا غلام فرمائن ٿا ته هڪ ڏينهن مان پنهنجي آقا اميرُالمومنين وٽ آيس ته پاڻ مسور جي دال کائي رهيا هئا، مون چيو: کُلَّ يومٍ عَدَسٌ! يعني روز روز دال! اوهان جي گھرواريءَ فرمايائون: هذَاطَعَامُ مَولَاک اَمِيرِ الْمُؤْمِنِين! يعني توهان جي آقا اميرالمؤمنين رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي اها ئي غذا آهي۔ (سيرت ابن جوزي،ص۱۸۱ ملخصا)

واه ڇا ڳالھ ڪجي ”مسور“ جي

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! قربان وڃجي! اميرالمومنين حضرت سَيدُنا عُمر بن عبدُ العَزِيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي دنيا کان بي رغبتي تي! جيڪي ايتري وڏي سَلطَنت جا خليفو هئڻ باوجود به سادي غذا استعمال ڪندا هئا، تنهنڪري اسان کي به گھرجي ته اسان به اللہ وارن جي نقشِ قدم تي هلندي سادگي اپنايون ۽ دال سبزي به خوشي خوشي کائيندا ڪيون۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! مسور جي دال جي ته ڇا شان آهي جو حديثِ پاڪ ۾ ان کي برڪت واري شيء قرار ڏنو ويو آهي جيئن رسولِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: توهان مسور ضرور کائيندا ڪيو ڇوته اها برڪت واري شيء آهي۔ جيڪا دل کي نرم ڪندي ۽ لڙڪن کي وڌائيندي آهي۔ هن ۾ 70 انبياءِ ڪرام عَلَيهِ السَّلَام جون برڪتون شامل آهن، جن ۾ حضرت عيسيٰ عَلَيهِ السَّلَام به شامل آهن۔ (فردوس الاخبار، ۲ / ۶۷، حديث: ۳۸۷۶)

امام ثعلبي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا ته حضرت سَيدُنا عُمر بن عبدُالعَزيز رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هڪ ڏينهن زيتون، هڪ ڏينهن گوشت ۽ هڪ ڏينهن مسور جي دال سان ماني کائيندا هئا، هڪ بُزرگ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: مسور ۽ زيتون نيڪن جي غذا آهي، مسور بدن (Body) کي سنهو ڪري ٿي ۽ سنهو بدن عبادت ۾ مددگار ٿيندو آهي، مسور سان اهڙي شهوت ناهي ڀڙڪندي جهڙي گوشت کائڻ سان ڀڙڪندي آهي۔ (تفسير قرطبي ،۱ / ۳۴۶، ملتقطا)

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن .
