دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جو نهايت ئي پيارو ڪتاب ”نيڪياں ضائع هونے سے بچائيے“ جي صفحي نمبر 20 ۽ 21 تي آهي: حضرت سَيدُنا عبدُالله بن عباس رَضِىَ اللّٰهُ تَعَالٰى عَنْهما کان روايت آهي ته اميرالمؤمنين حضرت سَيدُنا ابوبڪر صديق رَضِىَ اللّٰهُ تَعَالٰى عَنْه جي خلافت واري زماني ۾ ڏڪار ٿيو ته پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه ماڻهن کي فرمايائون: توهان کي ڪا تڪليف نه پهچندي ايستائين جو الله پاڪ توهان کي هن ڏڪار کان نجات ڏيندو۔ پوءِ جڏهن اڳيون ڏينهن ٿيو ته سندن وٽ خوشخبري ڏيڻ وارو آيو ته حضرت سَيدُنا عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه وٽ ڪڻڪ ۽ خوراڪ جي سامان جا هڪ هزار اٺ اچي رهيا آهن۔ (پوءِ جڏهن حضرت سَيدُنا عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه وٽ ڪڻڪ ۽ ٻين کائڻ جي شين جا هڪ هزار اٺ آيا) ته اڳئين ڏينهن واپاري سندن گھر پهتا ۽ دروازي تي دستڪ ڏنائون۔ پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه ٻاهر تشريف وٺي آيا ۽ انهن کان پڇيائون: توهان ڇا چاهيو ٿا؟ انهن چيو: اسان کي معلوم ٿيو آهي ته اوهان وٽ ڪڻڪ ۽ ٻئي اناج جا هڪ هزار اٺ آيا آهن، اوهان اهي اسان کي وڪڻي ڇڏيو ته جيئن مدينه منوره جي ضرورت مندن تي رِزق جي ڪشادگي ٿي وڃي۔ حضرت سَيدُنا عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه ارشاد فرمايائون: اندر اچو۔ اهي ماڻهو

اندر آيا ته پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه انهن کان پڇيائون: توهان ماڻهو ملڪ شام جي ريٽ مطابق ڇا نفعو ڏيندو؟ انهن چيو: ڏھ جا ٻارهن يعني ٻيڻو نفعو ڏينداسين، پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه ارشاد فرمايائون: مون کي هن کان وڌيڪ نفعو ملي رهيو آهي۔ واپارين چيو: ڏھ جا چوڏهن وٺو۔ پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه فرمايائون: مون کي وڌيڪ ملي پيو۔ انهن چيو: ڏھ جا پندرهن وٺو۔ پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه فرمايائون: مون کي هن کان وڌيڪ نفعو ملي رهيو آهي۔ انهن چيو: مديني جا واپاري (Dealers) ته اسان آهيون، اوهان کي ڪير وڌيڪ نفعو ڏئي رهيو آهي؟ حضرت سَيدُنا عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه ارشاد فرمايائون: مون کي هڪ روپيي تي ڏھ روپيا منافعو ملي رهيو آهي، توهان هن کان وڌيڪ ڏيندو؟ انهن چيو: اسان ايترو نفعو نٿا ڏئي سگھون۔ پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه فرمايائون: اي واپارين جي جماعت! هن ڳالھ تي شاهد ٿي وڃو ته مون اهي سموريون شيون مديني جي ضرورت مندن جي لاءِ صدقو ڪيون آهن۔ حضرت سَيدُنا عبدُ الله بن عباس رَضِىَ اللّٰهُ تَعَالٰى عَنْهما فرمائن ٿا: مان جڏهن رات جو سُتس ته خواب ۾ رسولِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪئي، پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن کي نور جي چادر پاتل هئي، مبارڪ هٿن ۾ نور جي لٺ ۽ پير مبارڪ ۾ جيڪي نعلين هئا ان جا تسمه(ڪَها) پڻ نوراني هئا. مون عرض ڪيو: بِاَبِي اَنْتَ وَاُمِّي يَارَسُوْلَ الله! لَقَدْطَالَ شَوْقِي اِلَيْكَ يعني منهنجا ماءُ پيءُ اوهان تي قربان يارسولَ الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ! اوهان ڏانهن منهنجو شوق وڌندو پيو وڃي۔ نبي اڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ارشاد فرمايائون: مان جلدي ۾ آهيان، عثمان (رَضِىَ اللّٰهُ تَعَالٰى عَنْه) هڪ هزار اٺن جو ٻوجھ ڪڻڪ وغيره صدقو ڪيو آهي۔ الله ڪريم عثمان (رَضِىَ اللّٰهُ تَعَالٰى عَنْه) جو اهو عمل قبول فرمائي جنتي حور سان ان جو نڪاح ڪيو آهي ۔

پاڻ رَضِیَ اللهُ تَعَالٰی عَنْه جو مبارڪ نالو، ”عُثمان“ ۽ ڪنيت’’ابوعَمرو‘‘ آهي۔ جڏهن ته سندن لقب ”اَميرُ المُؤمنين، ذُوالنُّورَين، ڪاملُ الحَياءِوَالاِيمان، (حيا ۽ ايمان ۾ ڪامل) جامِعُ القرآن (قرآن گڏ ڪرڻ وارا)، سَيدُالاَسخِياء(سخين جا سردار)، عُثمانِ باحَيا وغَيره مشهور (Famous) آهن۔ (ڪرامات عثمان غني ص۳، ۵،۱۱) پر پاڻ رَضِیَ اللهُ تَعَالٰی عَنْه جي سمورن لقبن مان ”ذُوالنُّورَين“ (يعني ٻن نور وارو) گھڻو مَشهور آهي۔ هن لَقب جو سبَب هي آهي ته حضور پر نُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّمَ جون ٻه شَهزاديون حضرت سَيدَتُنا رُقَيه ۽ حضرت سَيدَتُنا اُمِّ ڪلثُوم رَضِىَ اللّٰهُ تَعَالٰى عَنْهما هڪ کان پوءِ ٻي سندن نڪاح ۾ آيون۔ انهي ڪري سندن کي’’ذُوالنُّورَين‘‘ (يعني ٻن نورن وارا) چيو ويندو آهي۔ (تھذيب الاسماء، باب العين و الثاء المثلة،۱/۲۹۷) انهيء طرف اِشارو ڪندي سيدي اَعليٰ حضرت رَضِیَ اللهُ تَعَالٰی عَنْه فرمائن ٿا۔

نُور کی سرکار سے پایا دوشالہ نُور کا ہو مُبارَک تم کو ذُوالنُّوْرَیْن جوڑا نُور کا

پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه خُلفاءِ راشِدين ۾ ٽيان خليفه آهن۔ (جنتي زيور،ص۱۸۲ملخصاً) پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه اَمِيرُ المُؤمِنِين ابوبڪرصديق رَضِىَ اللّٰهُ تَعَالٰى عَنْه جي ڪوششن سان اسلام آندائون ۽ اسلام قبول ڪرڻ وارن ۾ سندن جو شمار چوٿين نمبر تي ٿئي ٿو، جيئن ته پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْه خود اِرشاد فَرمائن ٿا: ا ِنِّيْ لَرَابِعُ اَرْبَعَةٍ فِي الْاِسْلَامِ يعني مان اِسلام قَبول ڪرڻ وارن چئن شخصن مان چوٿون آهيان۔ (معجم ڪبير،۱/۸۵، حديث:۱۲۴)(اسد الغابة،عثمان بن عفان،۳ / ۶۰۶ ملخصاً) حضرت سَيدُنا عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه کي جمعي جي ڏينهن 35 هجري ذُوالحِجَّة جي مهيني ۾شهيد ڪيو ويو۔حضرت سَيدُنا جُبَيربن مُطعِم رَضِىَ اللّٰهُ تَعَالٰى عَنْه سندن جنازي نماز پڙھائي ۽ سندن رَضِىَ اللّٰهُ تَعَالٰى عَنْه کي جنَّتُ البَقِيع ۾ دفن ڪيو ويو۔ (اسدالغابة،عثمان بن عفان،۳ / ۶۱۴-۶۱۶ ملخصاً)

دنيا ڇڏي سگھان ٿو پر اِيمان نه:

اَميرُ المُؤمِنِين حضرت سَيدُناعثمانِ غَني رَضِىَ اللّٰهُ تَعَالٰى عَنْه جڏهن اِسلام قبول ڪيائون ته نه رڳو پنهنجن گھر وارن بلڪ سڄي خاندان جي شديد مُخالفت کي منهن ڏيڻو پيو، پاڻ رَضِیَ اللهُ تَعَالٰی عَنْه کي مارڪُٽ ڪئي وئي ايستائين جو سندن رَضِیَ اللهُ تَعَالٰی عَنْه جو چاچو حَڪم بن اَبي العاص ته ايتري قَدر ناراض ٿيو جو سندن کي پڪڙي هڪ رَسّي سان ٻڌي چوڻ لڳو: تو پنهنجي پيءُ ڏاڏي جو دِين ڇڏي ٻيو مَذهب اِختيار ڪيو آهي، جيستائين تون نئين مَذهب کي نه ڇڏيندين اسان تون کي نه ڇڏينداسين، اهڙي طرح ٻڌي رکنداسين۔ اهو ٻڌي حضرت سَيدُنا عثمانِ غَني رَضِیَ اللهُ تَعَالٰی عَنْه جن فرمايو: رب ذُوالجَلال جو قسم! مان اسلام کي ڪڏهن نه ڇڏيندس، نه ڪڏهن هن دولت کي ڇڏي سگھان ٿو، دل مان دينِ اسلام نڪري وڃي اهو هرگز نٿو ٿي سگھي۔ حَڪم بن اَبي العاص جڏهن سندن رَضِیَ اللهُ تَعَالٰی عَنْه جو اهو جذبو (Passion) ڏٺو ته مَجبور ٿي سندن رَضِیَ اللهُ تَعَالٰی عَنْه کي قيد مان آزاد ڪري ڇڏيو۔ (تاريخ مدينة دمشق،عثمان بن عفان ،۳۹ /۲۶)

ڪن ٿا، طرح طرح سان ستائن ٿا ته جيئن ڪهڙي نموني مَعَاذَ الله عَزَّوَجَلَّ هي دينِ اِسلام کان ڦري مُرتدّ ٿي وڃن۔ پر ياد رکو! هر حالت ۾ اِيمان جي حِفاظت اِنتهائي ضَروري آهي، چاهي ڪهڙي ئي آفت اچي اِيمان جي دولت هٿ مان نه وڃڻ کپي بلڪ ايمان تي استقامت جي لاءِ الله پاڪ جي بارگاھ ۾ دعا ڪندو رهڻ گھرجي۔

ايمان تي خاتمي جي لاءِ ”شجرهء، قادريه، رضويه عطاريه“ ۾ هڪ نهايت پيارو وظيفو به آهي، جيڪو به صبح ۽ شام 3،3 ڀيرا هن کي پڙهندو، اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ پڙهڻ واري جو خاتمو ايمان تي ٿيندو. اهو پيارو وظيفو شجري شريف جي صفحي نمبر 15 تي موجود آهي.

سيرتِ عثمانِ غني رَضِىَ اللّٰهُ تَعَالٰى عَنْه ۾ انهن اسلامي ڀينرن جي لاءِ سکڻ جو مدني گل آهي جن کي گھر وارن يا ٻين رشتيدارن جي طرفان سنتن جي خدمت ڪرڻ سبب طرح طرح سان ستايو ويندو آهي ته اهي همت هاري مدني ماحول جي برڪتن کان پنهنجو پاڻ کي محروم ڪنديون آهن، انهن کي گھرجي ته هو ان طرح جي رنڊڪن جي ڪري هرگز دل ننڍي نه ڪن بلڪ نبين سڳورن عَلَيهمُ السَّلام ۽ بزرگانِ دين رَحْمَةُ اللهِ تَعَال ٰی عَلَيْهم اجمعين خاص ڪري شهداءِ ڪربلا رَضِىَ اللّٰهُ تَعَالٰى عَنْهم اجمعين تي ايندڙ مصيبتن ۽ تڪليفن ۽ ان تي انهن جي ثابت قدمي کي پنهنجي پيشِ نظر رکن، سُنّتن جي خدمت جي سلسلي کي جاري رکن ۽ دعوتِ اسلامي جي مدني ماحول سان مضبوطي سان جُڙيل رهن ڇوته دعوتِ اسلامي جي مدني ماحول سان وابسته رهڻ به ايمان تي استقامت جو بهترين ذريعو آهي۔ الله ڪريم اسان کي ايمان تي استقامت، دعوتِ اسلامي ۽ امير اهلسنت دَامَتْ بَرَکاتُهمُ الْعَالِيه جي سچي پڪي غلامي نصيب فرمائي۔

(جمع الجوامع للسيوطي ج 5 ص 249 حديث 14867) ٭ مسواڪ جي ٿولھ چيچ جي برابر هجي ٭ ڏندڻ هڪ گرانٺ کان ڊگهو نه هجي نه ته ان تي شيطان ويهندو آهي ٭ ان جا تاندورا نرم هجن جو سخت تاندورا ڏندن ۽ مهارن جي وچ ۾ وٿي (Gap) جو سبب بڻجندا آهن ٭ مسواڪ تازو هجي ته سٺو نه ته ڪجھ دير پاڻي جي گلاس ۾ پسائي نرم ڪري ڇڏيو ٭ ان جا تاندورا روزانو ڪٽيندا رهو جو تاندورا ان وقت تائين ڪم جا آهن جيستائين انهن ۾ ڪؤڙاڻ باقي آهي ٭ ڏندن جي ويڪر ۾ ڏندڻ ڪريو ٭ جڏهن به ڏندڻ ڪرڻو هجي ته گھٽ ۾ گھٽ ٽي ڀيرا ڪريو ٭ هر ڀيري ڌوئي ڇڏيو ٭ ڏندڻ ساڄي هٿ ۾ اهڙي طرح کڻو جو چيچ ان جي هيٺان ۽ وچ واريون ٽي آڱريون مٿان ۽ آڱوٺو ڇيڙي تي هجي ٭ پهريان ساڄي طرف کان مٿين ڏندن تي ۽ پوءِ کاٻي طرف کان هيٺين ڏندن تي ڏندڻ ڪيو ٭ مسواڪ مٺ ۾ بند ڪري ڪرڻ سان بواسير ٿيڻ جو انديشو آهي مسواڪ جي فضيلتن ۽ فائدن جي حوالي سان وڌيڪ معلومات حاصل ڪرڻ جي لاءِ شيخِ طريقت،اميرِ اهلسنّت دَامَتْ بَرَکاتُهمُ الْعَالِيه جي رسالي ” مسواک شريف کے فضائل “ جو مطالعو ڪريو۔
