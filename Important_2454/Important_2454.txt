مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته تلاوتِ قرآنِ ڪريم جون ڪهڙيون برڪتون آهن جو جڏهن امام ناصرُ الدين بستي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي هوش آيو ۽ انهن پاڻ کي دفنايل ڏٺو ته ان پريشاني جي عالَم ۾ توهان سورة يٰسين جي تلاوت شروع ڪري ڇڏي، جنهن جي برڪت هي ظاهر ٿي ته الله عَزَّوَجَلَّ غيب کان سندن جي نجات جا اسباب پيدا فرمايا ۽ پاڻ زنده سلامت قبر کان ٻاهر تشريف کڻي آيا ۽ سندن جي جان بچي وئي. يقينً ا قرآنِ مجيد الله عَزَّوَجَلَّ جي تمام ئي پياري نعمت ۽ رحمتن ۽ برڪتن وارو ڪتاب آهي. جيڪو الله عَزَّوَجَلَّ پنهنجي ٻانهن جي رهنمائي ۽ انهن جي فلاح و ڪامراني جي لاءِ سَرورِ ڪائنات، شاھِ مَوجُودات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي قلبِ منير تي نازل فرمايو. ان جي عظمت جي لاءِ ايترو ئي ڪافي آهي ته هي الله جو ڪلام آهي (يعني الله جو پيارو پيارو ڪلام آهي)، هي مبارڪ ڪتاب هر لحاظ کان ڪامل آهي، ان کي نازل ڪرڻ وارو رَبُّ الْعَالَمِیْن آهي ته آڻڻ وارو رُوحُ الاَمِين (حضرتِ سيدُنا جبريل عَـلَيْـهِ الـسَّـلَام) آهي ۽ جنهن امت جي لاءِ آيو اها سڀني امتن کان بهترين آهي، جنهن زبان ۾ نازل ٿيو اها عربي مبين آهي ته جنهن مهيني ۾ آيو اهو سڀ مهينن ۾ معزز ترين آهي، جنهن رات نازل ٿيو اها افضل ترين آهي جن جڳھن تي نازل ٿيو اهي اعلٰي ترين آهن، قرآنِ ڪريم وحيِ الٰهي آهي، هي قربِ خداوندي جو ذريعو آهي، رهندڙ دنيا جي لاءِ نسخهءِ ڪيميا آهي،

سڀني آسماني ڪتابن جو لبّ لباب آهي، تمام علوم جو سرچشمو آهي، هي هدايت جو مجموعو، رحمتن جو خزانو ۽ برڪتن جو چشمو آهي، هي اهڙو دستور آهي جنهن تي عمل پيرا ٿي ڪري سڀئي مسئلا حل ڪري سگھجن ٿا، اهڙو نور آهي، جنهن سان گمراهي جا تمام انڌيرا دور ڪري سگھجن ٿيا، اهڙو رستو آهي، جيڪو سڌو الله عَزَّوَجَلَّ جي رضا ۽ جنت تائين وٺي ويندو آهي، اصلاح ۽ تربيت جو اهڙو نظام آهي، جنهن سان انسان جو تزڪيو ڪري (يعني انسان کي پاڪ ڪري) ان کي مثالي بڻائي ڇڏيندو آهي، اهڙو وڻ آهي جنهن جي ڇانوَ ۾ ويهڻ وارو قلبي سڪون محسوس ڪندو آهي، اهڙو باوفا ساٿي آهي، جيڪو قبر ۾ به ساٿ نڀائيندو آهي ۽ حشر ۾ به وفا جو حق ادا ڪندو. ان ۾ بيمار دلين جي لاءِ شِفا آهي، جنهن ان کي مضبوطي سان پڪڙيو، اهو هدايت يافته ٿي ويو، جنهن ان تي عمل ڪيو اهو فلاحِ دارَين (يعني دنيا ۽ آخرت ۾ ڪاميابي) حاصل ڪري ويو.

سڀ کان سٺو ڪتاب

تفسيِر خازِن ۾ قرآنِ پاڪ جي اَحْسَنُ الْحَديث (يعني سڀ کان سٺي ڪتاب) ہئڻ جون ٻه صُورتون بيان ڪيون ويون آهن: (1) لَفظ جي اِعتبار کان ۽ (2) معني جي اِعتبار کان۔

(1) لفظ جي اِعتبار سان هن طرح جو قرآنِ ڪريم فَصاحَت وبَلاغَت جي سڀ کان اونچي دَرَجي تي فائز آهي، نه هي اَشعار جي جِنس مان آهي ۽ نه هي عوامي خطبن ۽ رَسائل جي طريقي تي آهي بلڪه هي ڪلام جو هڪ اهڙو قِسم آهي، جيڪو پنهنجي نموني ۾ سڀ کان جُدا آهي۔

(2) معنٰي جي اِعتبار سان ايئن ته قرآنِ مجيد ۾ ڪٿي به تَعارُض (يعني ٽڪراءُ) ۽ اِختلاف ناهي ۽ ان ۾ ماضي جون خبرون ، پهريان جا واقعات، غيب جون ڪثير خبرون، واعدي ۽ وعيد ۽ جنت ۽ دوزخ جو بيان آهي۔

(تفسير الخازن پ۲۳ الزمر تحت الاية:۲۳ ،۴/ ۵۳)

مُفَسِّرِشَہيرحڪيمُ الْاُمَّت حضرتِ مُفتي احمد يارخان عَلَيْهِ رَحْمَةُ اللهِ الْحَنَّان ان حديثِ پاڪ جي تحت لکن ٿا: حديث جي معنيٰ مُطلقًا ڳالھ ۽ ڪلام آهي، تنهنڪري ان معنيٰ جي اعتبار کان قرآن به حديث آهي ۽ ماڻهن جو ڪلام به، پر اِصطلاح ۾ صرف حُضُور جي فرمان ۽ ڪم کي حديث چيو ويندو آهي۔ هتي لُغوي معنيٰ ۾ آهي، اللہ جو ڪلام سڀ ڪلامن تي ايئن ئي بُزرگ (معزز) آهي جيئن خُود پروَردگار پنهنجي مخلوق تي۔ (مراٰۃ المناجيح ،۱/۱۴۶)

مٺا مٺا اسلامي ڀائرو! هاڻي توهان قرآنِ ڪريم جا فَضائل ۽ ڪمالات سَماعت فرمايا، واقعي قرآن ِ ڪريم جي عظمت ۽ رِفعت جو اَندازو نه ٿو لڳائي سگهجي، ان جو پڙھڻ عبادت، ان جو ٻڌڻ عبادت، ان کي ڇُهڻ عبادت، ايستائين جو ان کي ڏسڻ به عبادت آهي۔ هي مُقدَّس ڪلام رحمتن ۽ برڪتن سان ڀرپُور آهي۔ اچو ته!پہرين ان مُقدَّس ڪتاب سان مَحَبَّت ڪرڻ وارن جا فضائل ٻڌون ٿا۔ جيئن ته

الله ۽ ان جي رسول سان محبت کي ڄاڻڻ جو طريقو

حضرتِ سيدُنا عبداللہ بن مسعود رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: جنهن کي هي ڄاڻڻ پسند ہجي ته اهو اللہ عَزَّوَجَلَّ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان ڪيتري مَحَبَّت ڪندو آهي ته اهو ڏسي ته جيڪڏهن اهو قرآن سان مَحَبَّت ڪندو آهي (يعني ان جي تلاوت ۽ ان تي عمل ڪندو آهي۔ شرح الشفا للملاعلي قاري) ته اهو اللہ عَزَّوَجَلَّ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان به مَحَبَّت ڪندو آهي۔ (المعجم الڪبير للطبراني ۹/۱۳۲ الحديث:۸۶۵۷)

حضرتِ سيّدُنا سہل بن عبداللہ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: اللہ عَزَّوَجَلَّ سان مَحَبَّت جي علامت قرآن سان مَحَبَّت ڪرڻ آهي، قرآن سان مَحَبَّت جي علامت نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت ڪرڻ آهي، نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت جي علامت سُنَّت (يعني توهان جي حديثن ۽ اَحوال وغيره) سان مَحَبَّت ڪرڻ آهي، سُنَّت سان مَحَبَّت جي علامت آخرت سان مَحَبَّت ڪرڻ آهي، آخرت سان مَحَبَّت جي علامت دُنيا سان بُغض رکڻ آهي ۽ دُنيا سان بُغض جي علامت هي آهي ته ان مان بَقدرِ ضرورت ورتو وڃي ۽ ايتري مِقدار ۾ ورتو وڃي جيڪو آخرت تائين پهچائي ڇڏي۔

(الشفاء بتعريف حقوق المصطفيٰ ،جزء ثاني ص۲۸)

’’ جَامِعُ الْعُلُوم وَالْحِکَم ‘‘ ۾ آهي: مَحَبَّت ڪرڻ وارن جي ويجھو ڪا به شيءِ محبوب جي ڪلام کان وڌيڪ مٺي ۽ لذيذ ناهي هوندي، محبوب جو ڪلام سندن دلين جي لاءِ ڪيف وسُرورجو باعث ہوندو آهي ۽ اهو ان جي مَقصود ومَطلُوب جي اِنتہا ہوندو آهي۔ (جامع العلوم والحڪم ٤٥٢)

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته قرآنِ ڪريم سان مَحَبَّت ڪرڻ ڪيتري اَہَميت جو حامل آهي جو قرآن سان مَحَبَّت کي اللہ عَزَّوَجَلَّ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان مَحَبَّت جي علامت قَرار ڏنو ويو آهي ۽ قرآنِ پاڪ سان مَحَبَّت جي علامت ان جي تلاوت سان گڏوگڏ ان تي عمل ڪرڻ به آهي۔ اَلْحَمْدُلِلّٰہ عَزَّوَجَلَّ شيخِ طريقت، اميرِ اَہلسُنَّت، باني دعوتِ اسلامي حضرت ِ علامہ مولانا ابُوبلال محمدالياس عطّار قادري رَضَوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه تلاوتِ قرآن کي پنهنجو معمول بنائڻ ۽ ان جي مَحَبَّت دل ۾وڌائڻ جي لاءِ 72 مدني انعامات ۾ مدني انعام نمبر21 ۾ اِرشاد فرمائن ٿا ’’ڇا توهان اڄ ڪَنزُ الايمان مان ڪم اَز ڪم ٽي آيتون (ترجمي ۽ تفسيرسان گڏ) تلاوت

ڪرڻ يا ٻڌڻ جي سعادت حاصل ڪئي؟‘‘ تنهنڪري اسان کي به پنهنجي هي عادت بنائڻ گھرجي ته روزانه ڪنزُالايمان شريف ساڻ تفسير خزائن العرفان يا صِراطُ الجِنان مان ترجمي ۽ تفسير سميت ڀرپور غور و فڪر سان قرآنِ ڪريم جي تلاوت ڪندا ڪريون، اهڙي طرح مدني اِنعام تي عمل ڪرڻ سان گڏوگڏ تلاوت جون بي شمار ڀلايون ۽ برڪتون به نصيب ٿينديون ۽ عمل جو جذبو به پيدا ٿيندو۔ اِنْ شَآءَالله عَزَّوَجَلَّ

کنز الایماں اے خدا میں کاش روزانہ پڑھوں

پڑھ کے تفسیر اس کی پھر اس پر عمل کرتا رہوں

مٺا مٺا اسلامي ڀائرو! يقيناً اهي ماڻهو وڏا خُوش نصيب آهن جيڪي قرآنِ ڪريم سان مَحَبَّت ڪندا آهن ۽ ان جي تِلاوت ڪرڻ سان گڏوگڏ ان تي عمل به ڪندا آهن، پرافسوس صَد ڪروڙ اَفسوس اسان جي مُعاشري ۾ هڪ وڏو تعداد اهڙن ماڻهن جو به آهي جيڪو قرآنِ پاڪ کان تمام پَري آهي ۽ مهينا گُذري وڃڻ جي باوجود به ان جي تلاوت جي توفيق نصيب ناهي ٿيندي، اها ئي وجهه آهي جو اڄ اسان طرح طرح جي پريشانين کي منهن ڏئي رهيا آهيون، ناچاقين، نااِتفاقين ۽ بي روزگارين بُري طرح اسان کي پنهنجي لپيٽ وٺي ڇڏيو آهي، اسان مان تقريباً سڀني جي گھرن ۾ قرآنِ ڪريم مَوجود ہوندو ۽ اها سعادت جي ڳالھ به آهي، پر اسين ڄڻ رکي وساري ڇڏيو آهي ۽ ڪڏهن کولي ڪري ڏسندا به نه آهيون، جڏهن ته گھرن ۾ تلاوتِ قرآن ڪريم ڪرڻ جي وڏي فضيلت آهي۔ جيئن ته حضرتِ سيدُنا ابُوہُرَيرہ رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: ’’جنهن گھر ۾ قرآن پڙهيو ويندو آهي اهو پنهنجي رهڻ وارن تي ڪُشادہ ٿي ويندو آهي، ان جون ڀلايون گھڻيون هونديون آهن، ان ۾ فِرشتا حاضر ٿيندا ۽ شياطين ان مان نڪري ويندا آهن

۽ جنهن گھر ۾ قرآن ناهي پڙھيو ويندو، اهو پنهنجي رہڻ وارن تي تنگ (سوڙهو) ٿي ويندو آهي، ان جي ڀلائي گھٽ ٿي ويندي آهي، ان مان فِرشتا نڪري ويندا ۽ شياطين اچي ويندا آهن۔‘‘ (احياء العلوم (مترجم) ، ج۱ ، ص ۸۲۶)

اميرِ اہلسنت پنهنجي خواهش جو اظہار فرمائن ٿا:

الٰہی خوب دیدے شوق قرآں کی تلاوت کا

شرف دے گنبدِ خضرا کے سائے میں شہادت کا

نبين جي سُلطان، رَحمتِ عالميان، سردارِ دو جہان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ بَرَڪت نشان آهي: توهان قرآن سان تعلُّق باقي رکو، ان کي مُستقِل پڙھندا رہو، سو قسم آهي اُن جو، جنهن جي قبضي ۾ منهنجي جان آهي، يقيناً قرآن زيادہ آزاد ٿيڻ تي آماده آهي، انهن اُٺن کان، جيڪي پنهنجي رسين سان ٻڌل هجن۔ (صحيح البخاري ج۳ ص ۴۱۲ حديث۵۰۳۳) تنهنڪري منهنجا پيارا اسلامي ڀائرو! قرآنِ پاڪ جا حقيقي عاشق بڻجي وڃي، ان سان سچي دل لڳائي وٺو، ان جي روزانه تلاوت کي پنهنجو معمول بڻائي وٺو، اِنْ شَآءَالله عَزَّوَجَلَّ پوءِ ان جون رحمتون ۽ برڪتون اسان کي به نصيب ٿينديون، گھر مان پريشانيون دُور ٿينديون، رِزق ۾ برڪت ٿيندي ۽ اِنْ شَآءَالله عَزَّوَجَلَّ سڀ مُعاملا آسان ۽ حل ٿيندا ويندا۔

اميرِ اہلسنّت لکن ٿا:

تلاوت کی توفیق دیدے الٰہی گناہوں کی ہو دُور دل سے سیاہی

اچو ته تلاوتِ قرآنِ ڪريم جي عظمتن ۽ فضيلتن جي باري ۾ ڪجھ ٻڌون ٿا ته جيئن اسان جي دِلين ۾ به قرآنِ پاڪ جي اہميت ۽ روزانه پابندي سان ان جي تلاوت جي طرف رغبت پيدا ٿئي ۔ الله عَزَّوَجَلَّ سيپاري 22 سُوْرَۃ الفاطِر آيت 29 ۾ اِرشاد فرمائي ٿو:

اِنَّ الَّذِیۡنَ یَتۡلُوۡنَ کِتٰبَ اللہِ وَ اَقَامُوا الصَّلٰوۃَ وَ اَنۡفَقُوۡا مِمَّا رَزَقۡنٰہُمۡ سِرًّا وَّ عَلَانِیَۃً یَّرۡجُوۡنَ تِجَارَۃً لَّنۡ تَبُوۡرَ ﴿ۙ۲۹﴾

ترجمو ڪنزالايمان: بيشڪ جيڪي الله جو ڪتاب پڙهن ٿا ۽ نماز قائم رکن ٿا ۽ اسان جي ڏني مان ڪجهه اسان جي راهه ۾ خرچ ڪندا آهن لڪل ۽ ظاهر اهي اهڙي واپار جا اميدوار آهن جنهن ۾ ٽوٽو (نقصان) هرگز نه آهي.

تفسيرِ بَغَوي ۾ ان آيتِ مُبارَڪه جي تحت ذڪر ڪيل آهي ته ’’تِجَارَ ۃ ‘‘ مان مُراد اهو ثواب آهي جنهن جو الله عَزَّوَجَلَّ واعدو فرمايو آهي، ان ۾ هرگز نقصان ناهي، يعني هي اجر نه فاسد ٿيندو نه ہلاڪ ٿيندو۔ (تفسير بغوي ج۳ ص ۴۹۲) ڄڻ ته الله عَزَّوَجَلَّ قارئينِ قرآنِ ڪريم (يعني قرآنِ پاڪ جي تلاوت ڪرڻ وارن کي) اَجرِ عظيم جي بشارت عطا فرمائي رهيو آهي۔ هڪ ٻي مقام تي تلاوت ڪرڻ وارن جي تعريف ۾ ايئن ارشاد ٿئي ٿو:

اَلَّذِیۡنَ اٰتَیۡنٰہُمُ الۡکِتٰبَ یَتۡلُوۡنَہٗ حَقَّ تِلَاوَتِہٖ ؕ اُولٰٓئِکَ یُؤۡمِنُوۡنَ بِہٖ ؕوَ مَنۡ یَّکۡفُرۡ بِہٖ فَاُولٰٓئِکَ ہُمُ الۡخٰسِرُوۡنَ ﴿ ۱۲۱ ﴾٪ (پ1، البقرة، آيت 121)

ترجمو ڪنزالايمان: جن کي اسان ڪتاب ڏنو آهي،اُهي جيئن پڙهڻ جو حق آهي (تيئن) ان جي تلاوت ڪن ٿا، اهي ان تي ايمان رکن ٿا، ۽ جيڪي ان جا منڪر ٿينداته اهي ئي نقصان (کڻڻ) وارا آهن.

حضرت سيدُنا قتادہ رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان مروي آهي ته ان آيت ۾ اُولٰٓئِکَ یُؤۡمِنُوۡنَ بِہٖؕ يعني اهي ئي ان تي ايمان رکندا آهن مان مُراد نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا اهي اَصحاب آهن، جن اللہ عَزَّوَجَلَّ جي آيتن تي ايمان آندو ۽ انهن جي تصديق ڪئي۔ (تفسير در منثور ج۱ص ۲۷۳) معلوم ٿيو ته قرآنِ پاڪ جي تلاوت ڪرڻ، ايمان وارن جو حصو ۽ انهن جي ئي خصوصيت آهي۔ ان آيت جي تحت

مڪتبة المدينہ کان ڇپجڻ واري عظيم ۽ عام فہم تفسير "صراطُ الجِنان" جلد1، صفحي 200 تي آهي۔

قرآن مجيد جا حق:

ان مان اهو به معلوم ٿيو ته ڪتاب اللہ جا ڪيترائي حق به آهن۔ قرآن جو حق اهو آهي ته ان جي تعظيم ڪئي وڃي، ان سان محبت ڪئي وڃي، ان جي تلاوت ڪئي وڃي، ان کي سمجھيو وڃي، ان تي ايمان رکيو وڃي، ان تي عمل ڪيو وڃي ۽ ان کي ٻين تائين پہچايو وڃي. اهڙي طرح اَحاديثِ مُبارَڪه ۾ به تلاوتِ قرآنِ ڪريم جا بي شُمار فضائل مَوجُود آهن۔ انهيءَ متعلق چار(4) فرامينِ مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌو۔

مٺا مٺا اسلامي ڀائرو!صُبح ۽ شام قرآنِ پاڪ جي تلاوت ڪرڻ واري کي جتي بي شمار ا َ جر و ثواب ملندو آهي، ان جي برڪت سان ظاہري و باطني بيمارين کان شفا به نصيب ٿيندي آهي۔ هڪ شخص نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي بارگاھ ۾ نِڙي ۾ سور جي شڪايت ڪئي ته پاڻ اِرشاد فرمايو: قرآن پڙھڻ اختيار ڪر۔ (شعب الايمان،٢/٥١٩، حديث:٢٥٨٠) هڪ شخص حاضر ٿي ڪري عرض ڪيو: منهنجي ڇاتيءَ ۾ تڪليف آهي۔ ارشاد فرمايو: قرآن پڙھو، اللہ عَزَّوَجَلَّ فرمائي ٿو: وَشِفَآءٌ لِّمَا فِی الصُّدُوْرِ (ترجمو ڪنزالايمان: ۽ دلين جي صحت) (الدرالمنثور،٤/٣٦٦) (پارہ١١، يونس:٥٧) بلڪه قرآنِ ڪريم ته مختلف بيمارين جي بہترين دَوا به آهي۔ جيئن ته فرمانِ مُصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: خَیْرُالدَّوَاءِ اَلْقُرْاٰنُ يعني بہترين دَوا قرآنِ ڪريم آهي۔ (سنن ابن ماجہ،٤/۱۱۶۔١١۷،حدیث:٣٥٠١)

حضرت علامہ عبدالر ؤ ف مناوي عَلَیۡهِ رَحْمَةُ اللـهِ الۡھَادِی ان حديثِ پاڪ جي تحت لکن ٿا: يعني بہترين تعويذ اهو آهي، جيڪو ڪنهن قرآني آيت جي ذَريعي سان ڪيو وڃي، اِرشادِ باري تعاليٰ آهي:

وَ نُنَزِّلُ مِنَ الۡقُرۡاٰنِ مَا ہُوَ شِفَآءٌ وَّ رَحۡمَۃٌ لِّلۡمُؤۡمِنِیۡنَ ۙ

(پ15، بني اسرائيل، آيت 82)

ترجمو ڪنزالايمان: ۽ اسان قرآن ۾ نازل ڪريون ٿا اها شيء جيڪا ايمان وارن جي لاءِ شفاء ۽ رحمت آهي.

ته قرآنِ مجيد دل، بدن ۽ رُوح سڀ جي لاءِ دَوا آهي۔ جڏهن ته ڪجھ ڪلامن جون به خاصيتون ۽ فائدا ہوندا آهن، ته پوءِ رَبُّ العالمين جَلَّ جَلَالُہٗ جي ڪلام جي باري ۾ توهان جو ڇا خيال آهي، جنهن جي فضيلت ٻين ڪلامن تي اهڙي آهي، جيئن اللہ عَزَّوَجَلَّ جي سندس مخلوق تي۔ قرآنِ پاڪ ۾ ڪجھ اهڙيون آيتون آهن جيڪي خاص بيمارين ۽ مصيبتن جي خاتمي جي لاءِ آهن، انهن آيتن جي مَعرِ فت خاص ماڻهن کي ہوندي آهي۔ (فيض القدير،٣/٦٢٨) پُورو قرآنِ ڪريم ئي بيمارين جي لاءِ شِفا ۽ مصيبتن ۽ تڪليفن ۾ مشڪل ڪُشا آهي ۽ قرآنِ پاڪ جي ہر سُورت جي پنهنجي ئي فضيلت ۽ شان آهي جيڪا جان ۽ مال جي حِفاظَت ڪرڻ، رَنج ۽ غم ختم ڪري دل ۾ فَرحت ۽ مَسرت داخل ڪرڻ ۽ مَرض کان نجات ڏيارڻ جي لاءِ ڪافي آهي ۔ جيئن ته مڪتبةالمدينہ جي مطبوعہ 679 صفحات تي مُشتمل ڪتاب ’’ جنَّتي زيور ‘‘مان قرآنِ پاڪ جي ڪجھ سُورتن جا مُختَصر فَضائل ٻڌون ٿا۔ سُوۡرَۃُ الْفاتِحَہ 100مرتبہ پڙهي ڪري جيڪا دُعا گھُري وڃي، قَبول ٿيندي آهي، سُوْرَۃُ الْبَقَرہ جي تلاوت سان شيطان گھر مان ڀڄي ويندو آهي، آیَۃُ الْکُرسی پڙهڻ سان مُحتاجي دُور ٿيندي آهي، سُوْرَۃُ الْکَہْف کي ہميشہ پڙھڻ وارو دَجّال جي فِتنن کان محفوظ رهندو، والدين جي قَبر تي ہر جُمعي سُورۂ یٰسٓ جي تِلاوت ڪرڻ سان ان جي حُروف جي تعداد جي برابر ان جا گُناہه بخشيا ويندا آهن۔ سُوْرَۃُالدُّخَان پڙهڻ سان مُشڪل دُور ٿيندي آهي، جيڪو قريبُ المرگ ہجي ان تي سُوْرَۃُ الْجاثِیَہ جو دَم ڪرڻ سان ان جو خاتِمو خير سان ٿيندو سُوْرَۃُالحُجُرات جو پڙهڻ ۽ دَم ڪري پيئڻ گھر ۾ خير و برڪت جي لاءِ مُفيد آهي،

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اللھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ مان گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اللھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سيدُنا انس رَضِي اللہُ تَعَالٰي عَنہُ کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([1])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

﴿4﴾ ڇھ لک دُرُودشريفن جو ثواب

اللھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ

اللھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحاب َ ه ڪرام رَضِي اللہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته ائين پڙهندو آهي. ([1])

﴿ 6 ﴾ دُرُود شَفَاعت

اللہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَن ْ زِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللہُ تَعَالٰی عَلٰی مُحَمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سيدُنا ابنِ عباس رَضِي اللہُ تَعَالٰي عَنہُمَا کان رِوايت آهي ته سرڪارِ مدينه صَلَّي اللہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿2﴾ هر رات عبادت ۾ گذارڻ جو آسان نسخو

غرائب القرآن صفحه 187 تي هڪ روايت نقل ڪئي وئي آهي ته جيڪو شخص رات ۾ ٽي ڀيرا هي دعا پڙهندو ته ان ڄڻ شبِ قدر کي حاصل ڪري ورتو لهٰذا هر رات هي دعا پڙهڻ گهرجي. دعا هي آهي:

لَآ اِلٰہَ اِلَّا اللہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

”يعني خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي.“

(فيضانِ سنّت ج اوّل، ص 1163، 1164)
