دُرُود شريف جي فضيلت

اُمُّ المؤمنين حضرت سَيدَتُنا عائشه صِدِّيقَه رَضِىَ اللّٰهُ تَعَالٰى عَنْها کان روايت آهي ته نبي رَحمت، شفيعِ اُمّت، محبوبِ ربُّ العزّت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ شفاعت نشان آهي: ” مَنْ صَلّٰي عَلَي يوْمَ الْجُمُعَۃِ کانَتْ شَفَاعَۃٌ لَّه عِنْدِي يوْمَ الْقِيامَۃ (يعني) جيڪو شخص جُمُعي جي ڏينهن مون تي دُرُود شريف پڙھندو ته بروزِ قيامت ان جي شفاعت منهنجي ذِمّه ڪرم تي هوندي۔“ (ڪنز العمال،ڪتاب الاذڪار،الباب السادس في الصلاۃ عليه،الجزء الاول،۱/۲۵۵ ،حديث: ۲۲۳۶)

پياريون پياريون اسلامي ڀينرون! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نِي ّ تون ڪري وٺون ٿ يون . فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِهٖ “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندَس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندَس ٭ ضرورت پوڻ تي ٻين اسلامي ڀينرن جي لاءِ سِرڪي جاءِ ڪشاده ڪندَس ٭ ڌِڪو وغيره لڳو ته صبر ڪندَس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندَس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ واريءَ جي دِل جوئي جي لاءِ آهسته آواز سان جواب ڏيندَس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندَس ۔ دوران بيان موبائل جي غيرضروري استعمال کان بچندَس،نه بيان ريڪارڊ ڪندَس نه ئي ٻئي ڪنهن قسم جو آواز ڇو ته ان جي اجازت ناهي،جيڪو ڪجهه ٻُدَندَس،ان کي ٻڌي ۽ سمجهي ڪري ان تي عمل ڪرڻ ۽ ان کي بعد ۾ ٻين تائين پهچائي نيڪي جي دعوت عام ڪرڻ جي سعادت حاصل ڪندَس.

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! اڄوڪو دؤر قِيامت جي نشانين ۽ ان کان پهريان ظاهر ٿيڻ وارن بيشمار فتنن سان ڀريل آهي۔ هر ايندڙ ڏينهن نون نون فتنن سان گڏ ظاهر ٿي رهيو آهي۔ ڪيتري اَفسوس جي ڳالھ آهي جو جنهن اُمّت جي حقيقي خَير خواھ حُضور نبي ڪريم، محبوبِ ربِّ عظيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ پنهنجي ظاهري حياتي ۾ فتنن جي دَور جي هولناڪين جي باري ۾ فڪر مند رهندا هئا، آزمائشن، پريشانين ۽ فتنن جي هن هولناڪ زماني ۾ منهنجي اُمّت ڪٿي حق واري واٽ تان ڀٽڪي نه وڃي، اڄ انهيءَ اُمّت جا ماڻهو انهن فتنن ۾ پئجي پنهنجن ئي هٿن سان دِين ۽ ديانت، حق ۽ اَمانت ۽ باهمي مَحَبَّت

و شَرافت جو سر عام قَتلِ عام ڪري رهيا آهن۔ تنهنڪري حالات هن ڳالھ جي تقاضا ڪري رهيا آهن ته هر طبقي سان تَعَلُّق رکندڙ مسلمانن تائين نبي غيب دان، مڪي مدني سُلطان صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا اهي اِرشاداتِ عالي شان پهچايا وڃن، جن ۾ پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ الله ڪريم جي عطا ڪيل عِلمِ غَيب جي ذريعي 14 سو سال پهريان ئي موجوده ۽ آئنده دور جي فِتنن جي پيشن گوئي ڪندي اسان کي انهن جي ڀڙڪندڙ باھ مان اٿڻ وارن شُعلن کان پنهنجو دامن بچائڻ جي ترغيب ڏياري آهي۔ اچو! هن باري ۾ ڪجھ اَحاديثِ مبارَڪه ۽ ان مان حاصِل ٿيڻ واري نصيحت جا مَدَني گل گڏ ڪريون ٿيون، جيئن

گھر وارن جي هٿان هلاڪت

غَيب ڄاڻڻ وارا آقا، مڪي مدني مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ عبرت نشان آهي: ماڻهن تي هڪ زمانو اهڙو ايندو جو ان شخص جي علاوه ڪنهن جو دِين محفوظ نه رهندو، جيڪي پنهنجي دِين کي کڻي (يعني ان جي حفاظت جي خاطر) هڪ جبل کان ٻئي جبل ۽ هڪ غار کان ٻئي غار ڏانهن ڀڄي۔ ان وقت معيشت (روزي وغيره) جو حصول الله پاڪ کي ناراض ڪرڻ بغير نه ٿيندو۔ جڏهن اها صُورتِ حال هوندي ته انسان پنهنجي ٻارن ٻچن هٿان هلاڪت ۾ پئجي ويندو، جيڪڏهن ٻار ٻچا نه هوندا ته والدين جي هٿان ان جي هلاڪت ٿيندي ۽ جيڪڏهن والدين به نه هجن ته ان جي هلاڪت رشتيدارن يا پاڙيسرين جي هٿان ٿيندي۔“ صحابَه ڪرام عَلَيهمُ الرِّضْوَان عرض ڪيو: يارَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! اهو ڪئين ٿيندو؟ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”اهي ان کي غُربت تي عار (شرم) ڏياريندا، ان وقت اهو پنهنجو پاڻ کي هلاڪت جي هنڌن ڏانهن وٺي ويندو.“ (الزهد الڪبير للبيهقي،الجزء الثاني،فصل في ترڪ الدنيا … الخ، ص۱۸۳، حديث: ۴۳۹)

پياريون پياريون اسلامي ڀينرون! بَيان ڪيل حديثِ پاڪ ۾ جهڙي طرح مَردن جي لاءِ عبرت ۽ نصيحت جا مَدَني گل آهن، اتي انهن عَورَتن جي لاءِ به هن ۾ عبرت آهي، جيڪي پنهنجي مڙسن کي انهن جي آمدني تي طرح طرح جا طعنا ڏيندي

ڪجھ هن طرح جا تيز تکا جملا چونديون آهن: ”اڙي ڏس! فُلاڻي وٽ ته ڪيئي ڪيئي بنگلا، فيڪٽريون ۽ جائيدادون (Properties) آهن پر اوهان مون کي ڪرائي جي ننڍڙي گھر ۾ ويهاريو آهي، منهنجو ته هتي ساھ ٿو گُھٽجي! مون کي ڪشادو گھر کپي، فُلاڻي کي ته ڏسو هو پنهنجي فيملي سان گڏ عاليشان گاڏين ۾ گُھمي ڦري ٿو پر اوهان مون کي بسن، ٽيڪسين ۽ رڪشن جا ڌڪا کارايو ٿا، فُلاڻي ته لوڊشيڊنگ کان بچڻ جي لاءِ جنريٽر ورتو آهي اوهان گھٽ مان گھٽ يو پي ايس (U.P.S) يا چارجنگ فين ئي وٺي ڏيو، فُلاڻي ته هن عِيد تي پنهنجي ٻارن جي اَمڙ کي ايترن هزارن جو لباس يا سون (Gold) جو سيٽ ٺاهرائي ڏنو آهي پر اوهان مون کي عِيد تي سون جي منڊي يا بريس ليٽ يا واليون ئي وٺي ڏيو ها، فُلاڻي ته پنهنجن ٻارن جي امڙ کي فلاڻي شاپنگ مال مان شاپنگ ڪرائي آهي تنهنڪري مون کي به ڪنهن سٺي شاپنگ مال مان شاپنگ ڪرڻي آهي۔ فُلاڻي کي ڏسو ته ڪيترو خوشحال ٿي ويو آهي، اوهان به ته ڪجھ ڪريو، فُلاڻي جي پگھار لکن ۾ آهي پر اوهان ايترو عرصو ڪم ڪرڻ کان بعد به اتي جا اتي بيٺا آهيو۔“ وغيره جڏهن ته ٻارن جون فرمائشون ان جي علاوه آهن۔ ته روز روز جڏهن اهو شخص فرما ئشن ۽ طعنن کي ٻڌندو آهي ته ذِهني اَذِيت ۽ مايوسي ان کي هر پاسي کان وڪوڙي وٺندي آهي، ان کي ڪجھ سمجھ ناهي ايندو ته آخر هو ڪري ته ڇا ڪري، جيئن ته ان کي ٻارن ٻچن جون فرمائشون به پُوريون ڪرڻون هونديون آهن ان وٽ وسائل به گھٽ هوندا آهن، ان ڪري هو انهن جون اجايون فرمائشون پوري ڪرڻ جي لاءِ حرام و حلال جي پروا ڪرڻ بغير ناجائز ذريعا اختيار ڪري پنهنجي قَبر ۽ آخرت کي داءُ تي لڳائي ڇڏيندو آهي، جيئن ته

حلال و حرام جي مُعامَلي ۾ بي پرواهي

حضرت سَيدُنا ابُوهرَيره رَضِىَ اللّٰهُ تَعَالٰى عَنْه کان روايت آهي ته نبي پاڪ، صاحبِ لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: يَاْتِى عَلَى النَّاسِ زَمَانٌ،لاَ يُبَالِى الْمَرْءُ مَا اَخَذَ مِنْهُ اَمِنَ الْحَلاَلِ اَمْ مِنَ الْحَرَام يعني ماڻهن تي هڪ زمانو اهڙو ايندو جو ماڻهو کي هن ڳالھ جي ڪا

پروا نه هوندي ته ان (مال) ڪٿان حاصل ڪيو؟ حرام مان يا حلال مان۔

(بخاری، کتاب البیوع، باب من لم یبالی من حیث … الخ، ۲/۷، حدیث: ۲۰۵۹)

حڪيم الاُمَّت مفتي احمد يار خان رَحْمَة اللّٰهِ تَعَالٰى عَلَيه هن حديث جي شرح ۾ فرمائن ٿا: ”يعني آخري زماني ۾ ماڻهو دِين کان بي پروا ٿي ويندا، پيٽ جي فڪر ۾ هر طرح ڦاسي ويندا، آمدني وڌائڻ، مال گڏ ڪرڻ جي فڪر ڪندا، هر حرام وحلال وٺڻ تي دلير ٿي ويندا جيئن اڄڪلھ عام آهي۔“ (مراٰۃالمناجیح،۴/۲۲۹)

پياريون پياريون اسلامي ڀينرون! ياد رکو! توڙي جو ماءُ پيءُ ٻارن ۽ قَرابتدارن جي حُقُوق جي ادائيگي يقيناً اسان جي ئي ذميداري آهي، پر جيڪڏهن اسان انهن جون خواهشون پُوريون ڪرڻ ۽ انهن جي طعنن کان بچڻ جي لاءِ حرام ۽ حلال جي پروا ڪرڻ بغير مال و دولت گڏ ڪنديون رهيون سين، مشڪل وقت ۾ انهن کي صبرو شڪر، توڪل وقناعت جو ذهن نه ڏنو ۽ حلال وحرام جي تميز نه سيکاري ته ٿي سگھي ٿو ته سڀاڻي قيامت جي ڏينهن اهي ئي اسان جي خلاف بارگاھ الٰهي ۾ مُقدمو ڪن ۽ اسان جي پڪڙ ٿي وڃي، جيئن ته

بارگاهه الٰهي ۾ دعويٰ

حضرتِ سَيدُنا فقيه ابوللَّيث سَمر قندي رَحْمَة اللّٰهِ تَعَالٰى عَلَيه نَقل ڪن ٿا: مروي آهي ته مرد سان تعلق رکندڙن ۾ پهرين ان جي زال ۽ اولاد آهي. اهي سڀ (يعني زال ۽ ٻار قيامت ۾) الله پاڪ جي سامهون بيهي ڪري عرض ڪندا: ”اي اسان جا ڪريم رب! اسان کي هن شخص کان اسان جو حق وٺي ڏي، ڇاڪاڻ ته هن ڪڏهن اسان کي ديني معاملن جي تعليم نه ڏني ۽ هي اسان کي حرام کارائيندو هو، جنهن جي اسان کي خبر نه هئي.“ پوءِ انهيءَ شخص کي حرام ڪمائڻ تي ايتري قدر ماريو ويندو جو هن جو گوشت ڇڻي ويندو. پوءِ ان کي ميزان (ساهمي) وٽ آندو ويندو. ملائڪ جبل جي برابر ان جون نيڪيون آڻيندا ته انهيءَ جي خاندان مان هڪ شخص اڳيان وڌي ڪري چوندو: ”منهنجون نيڪيون گهٽ آهن.“ ته هو ان جي نيڪين مان کڻي وٺندو. پوءِ ٻيو اچي ڪري چوندو: ”تو

مون کي وياج کارايو هو.“ ۽ ان جي نيڪين مان کڻي وٺندو. اهڙي طرح ان جا گهر وارا انهيءَ جون سڀ نيڪيون کڻي ويندا ۽ هو پنهنجن گهروارن ڏانهن حسرت ۽ مايوسيءَ سان ڏسي چوندو: ”هاڻي منهنجي ڳچي تي اُهي گناهه ۽ ظلم رهجي ويا، جيڪي مون توهان جي لاءِ ڪيا هئا.“ ملائڪ چوندا: ”هي اُهو بدنصيب شخص آهي، جنهن جون نيڪيون هن جا گهر وارا کڻي ويا ۽ هي انهن جي ڪري دوزخ ۾ هليو ويو.“ (الروض الفائق،الباب الثامن في عقوبة قاتل … الخ، ص۴۰۱)

پياريون پياريون اسلامي ڀينرون! غور ڪريو! ان شخص جي بد نصيبي جو ڪهڙو عالَم هوندو جنهن جون سموريون نيڪيون ان جا گھروارا کڻي ويندا ته هو پاڻ ڪنگال ٿي ويندو. تنهنڪري غنيمت سمجھندي غفلت جي ننڊ مان بيدار ٿيڻ گھرجي، الله پاڪ ۽ اُن جي پياري حبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي فرمانبرداري ڪندي پنهنجي اولاد جي دِيني تربيت ڪريو، دُنيوي تعليم جي فِڪر ڪرڻ سان گڏوگڏ پنهنجي اولاد کي قرآنِ ڪريم پڙھڻ به سيکاريو وڃي، وضو ۽ غُسل، پاڪي ۽ ناپاڪي وغيره، اسلامي آداب به سيکاريا وڃن، انهن جا اَخلاق به سنوارڻ جي ڪوشش ڪئي وڃي، الله پاڪ ۽ ان جي پياري مَحبوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي محبت سيکاري وڃي۔ آخرت جو فڪر ڏياريو وڃي، موت جي تياري ڪرائي وڃي، مختلف گناهن جي باري ۾ ٻڌايو وڃي ۽ انهن گناهن کان بچايو به وڃي۔

ڪٿي ائين نه ٿئي جو اهو وَقت (Time) اچي جنهن جي نشاندهي الله پاڪ جي مَحبوب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ فرمائي هئي ته جنهن ۾ سُنّت تي عمل ڪرڻ، دِين تي قائم رهڻ ۽ ان راھ ۾ ايندڙ تڪليفن تي صبر ڪرڻ گھڻو مشڪل هوندو۔ اچو! هن باري ۾ 2 فرمانِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ٻڌو.

(1) اِرشاد فرمايائون: اَلْمُتَمَسِّكُ بِسُنَّتِي عِنْدَ اِخْتِلَافِ اُمَّتِي كَالْقَابِضِ عَلَى الْجَمْر يعني امت جي

(فتني ۽) فساد جي وقت منهنجي سنّت تي عمل ڪرڻ وارو باهه جوٽانڊو جهلڻ واري وانگر هوندو۔ (نوادر الاصول، الاصل الثالث عشر،الجزء الاول، ص۶۸، حديث:۸۷)

(2) اِرشاد فرمايائون: يَاْتِي عَلَى النَّاسِ زَمَانٌ الصَّابِرُ فِيْهِمْ عَلٰى دِينِهٖ كَالقَابِضِ عَلَى الجَمْرِ يعني ماڻهن تي اهڙو زمانو ايندو جو انهن ۾ پنهنجي دين تي صبر ڪرڻ وارو ٽانڊو جهلڻ واري جيان هوندو۔ (ترمذي، ڪتاب الفتن،باب ما جاء في النھي عن سب الرياح،۴/ ۱۱۵،حديث: ۲۲۶۷)

پياريون پياريون اسلامي ڀينرون! مفتي صاحِب حديث پاڪ جي هي شرح پنهنجي دور مطابق ڪئي هئي ۽ هاڻي ته ان کان ڪيئي ڀيرا وڌيڪ بيباڪي ۽ بي راھ روي جو دور هلي رهيو آهي، اها ڳالھ ڪنهن کان لڪيل ناهي ته في زمانه جيڪي اسلامي ڀينرون بُرقعو پائن ٿيون، شَرعي پردو ڪن ٿيون ۽ شريعت و سُنّت مطابق زندگي گذارڻ چاهين ٿيون، سخت آزمائش آهي. اهي اسلامي ڀائر ديانت داري سان پنهنجي فرائض جي منصب کي ادا ڪندي وياج ۽ رِشوت جي نحوست کان بچندي رِزقِ حلال ڪمائڻ چاهين ٿا انهن جي لاءِ به سخت آزمائش آهي۔ اهي عاشقانِ رسول جيڪي شريعت جي دائري ۾ رهندي، ناجائز رسمن کان بچندي شادي جي تقريب

الله پاڪ اسان کي هر طرح جي فتني کان محفوظ فرمائي ۽ آخرت جي تياري ڪرڻ جي تَوفِيق عطا فرمائي۔ اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

پياريون پياريون اسلامي ڀينرون! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪريان ٿي، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّم َ جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو. سُرمو لڳائڻ جون سنتون ۽ آداب

اچو! شيخ طريقت،اميرِ اَهلسُنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جي رسالي 101 مدني گل صفحي نمبر27 تان سُرمو لڳائڻ جون سنتون ۽ آداب ٻڌون ٿيون: فرمانِ مُصطَفٰي صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ آهي ؛ ”سڀني سرمن مان بهتر سرمو ”اِثمِد“ آهي جو اهو نظر کي روشن ڪري ۽ پلڪون وڌائي ٿو“۔ (ابن ماجه،۴/ ۱۱۵ ،حديث:۳۴۹۷)

٭ پٿر جو سرمو استعمال ڪرڻ ۾ حرج ناهي ۽ ڪارو سرمو يا ڪجل زينت جي قصد (يعني زينت جي ارادي) سان مرد کي لڳائڻ مڪروهه آهي ۽ زينت جو ارادو نه هجي ته ڪراهت ناهي۔ (فتاويٰ هنديه،۵ /۳۵۹) ٭ سرمو سمهڻ مهل استعمال ڪرڻ سنّت آهي۔ (مرآۃالمناجيح ،۶/۱۸۰) ٭ سرمي استعمال ڪرڻ جي ٽن منقول طريقن جو خلاصو پيش خدمت آهي: (1) ڪڏهن ٻنهي اکين ۾ ٽي ٽي سرائيون (2) ڪڏهن ساڄي اک ۾ ٽي ۽ کاٻي ۾ ٻه، (3) ته ڪڏهن ٻنهي اکين ۾ ٻه ٻه ۽ وري آخر ۾ هڪ سرائي کي سرمي واري ڪري ان کي واري واري سان ٻنهي اکين ۾ لڳايو۔ (شعب الايمان ج 5 ص 218، 219) اهڙي طرح ڪرڻ سان اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ٽنهي طريقن تي عمل ٿيندو رهندو۔ ٭ تڪريم جا جيترا به ڪم هوندا هئا اسان جا پيارا آقا صَلَّى اللّٰه تَعَالٰى عَلَيه واٰلِهٖ وَسَلَّمَ ساڄي پاسي کان شروع ڪندا هئا، تنهنڪري پهريان سرمو ساڄي اک ۾ پوءِ کاٻي اک ۾ پايو۔
