(8) وظائف پڙهڻ جي دوران بدن ۽ ڪپڙن جي پاڪي ۽ صاف سٿرائي جو خاص طور تي خيال ۽ لحاظ رکي بلڪ ۽ ظاهري پاڪائي ۽ صفائي سان گڏوگڏ پنهنجا اخلاق ۽ ڪردار ۽ باطني صفائيءَ جو به اهتمام رکي. ٭ سُوۡرَۃُ الْفاتِحَہ 100 ڀيرا پڙهي ڪري جيڪا دُعا گهري وڃي، قبول ٿيندي آهي، v سُوْرَۃُ الْبَقَرہ جي تلاوت سان شيطان گهر مان ڀڄي ويندو آهي آیَۃُ الْکُرسی پڙهڻ سان محتاجي دور ٿي ويندي آهي. ٭ سُوْرَۃُ الْکَہْف کي هميشه پڙهڻ سان دجَّال جي فتنن کان حفاظت ٿيندي، ٭ سُورةُ الدُّخان پڙهڻ سان مشڪل دور ٿي ويندي آهي، ٭ سُوْرَۃُ الحُجُرات جو پڙهڻ ۽ دم ڪري پيئڻ گهر ۾ خير ۽ برڪت جي لاءِ مفيد آهي، سُوْرَهٔ قٓ پڙهڻ سان باغ ۾ ميون جي ڪثرت ٿيندي آهي، سُوْرَۃُ الرَّحْمٰن 111 مرتبا پڙهڻ سان تمام مقصد پورا ٿي ويندا آهن، سُوْرَۃُالْواقِعَہ روزانو پڙهڻ سان اُن کي ڪڏهن به فاقو نه ٿيندو، سُوْرَۃُ الْمُلْک کي هر رات پڙهڻ سان عذابِ قبر کان حفاظت ٿيندي، سُوْرَۃُالْمُزَّمِّل 111 مرتبا پڙهڻ سان هر مشڪل آسان ٿي ويندي آهي، سُوْرَۃُالنّٰزِعٰت پڙهڻ سان سڪرات جي تڪليف نه ٿيندي، سُوْرَۂ اَلَمۡ نَشۡرَح کي جنهن مال تي پڙهيو وڃي، اُن ۾خوب برڪت ٿيندي، سُوْرَۃُالتِّیْن 3 ڀيرا پڙهڻ سان اخلاق ۽ ڪردار بهترين ٿيندا آهن، سُوْرَۃُ الْعَلَق ۾ جوڙن جي سور جو علاج آهي، سُوْرَۃُالْقَدۡر جيڪو صبح ۽ شام پڙهي، الله پاڪ اُن جي عزت وڌائي ڇڏيندو، سُوْرَۃُ الْبَیِّنَہ ڪوڙھ ۽ يرقان جو علاج آهي، سُوْرَۃُ الزِّلْزال چوٿائي قرآن جي

برابر آهي، جنهن کي نظر لڳي هجي، اُن تي سُوْرَۃُالْعٰدِیٰت جو دم مفيد آهي، سُوْرَۃُ الْقارِعَہ پڙهڻ سان بلائن کان حفاظت رهندي آهي، سُوْرَۃُالتَّکَاثُر کي 300 ڀيرا پڙهڻ سان تمام جلد قرض اَدا ٿي ويندو آهي، سُوْرَۃُ الْعَصْر پڙهڻ سان غم دور ٿي ويندو آهي، سُوْرَۃُ الْھُمَزَہ ۽ سُوْرَۃُالْفِیْل دشمن جي شر کان حِفاظَت ۽ سُوْرَۂ قُریش جان جي حِفاظت جي لاءِ مجرب آهي، سُوْرَۃُالْماعُوْن وڏي مشڪل جي وقت پڙهڻ گهڻو مفيد آهي، سُوْرَۃُالْکَوثَر جي تلاوت سان بي اولاد، کي اولاد ملي ويندي آهي، سُوْرَۃُ الْکافِرون چوٿائي قرآن جي برابر آهي، سُوْرَۃُ الْاِخْلاص ٽهائي قرآن جي برابر آهي، هن جا ڪيترائي فضائل آهن، سُوْرَۃُ الْفَلَق ۽ سُوۡرَۃُ النَّاس سان جنن ۽ شيطانن ۽ حاسدن جي شر کان حِفاظت رهندي آهي. (جنتی زیور،ص۵۸۸)

سُـبْحٰـنَ ال ـ لّٰــه عَزَّوَجَلَّ! ٻڌو اوهان قرآني سورتن جا ڪهڙا ڪهڙا فضائل ۽ برڪتون آهن، لهٰذا روزانو ڪم از ڪم هڪ سيپارو يا جيترو ممڪن ٿي سگهي تلاوتِ قرآن جو معمول بڻائي ڇڏيو، جيڪڏهن قرآن پڙهڻ نٿو اچي ته عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي جي تحت اسلامي ڀينرن جي لڳڻ واري مدرسة المدينه بالغان ۾ پڙهو. اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ڪجھ ئي عرصي ۾ درست قرآن پڙهڻ سان گڏوگڏ ڪيترن ئي شرعي مسئلن جي به ڄاڻ نصيب ٿيندي. مدرسة المدينه بالغات ۾ پڙهڻ تي استقامت حاصل ڪرڻ جي لاءِ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي جي مدني ماحول سان وابسته رهو ۽ نيڪيءَ جي دعوت عام ڪرڻ جي لاءِ دعوتِ اسلامي جو ساٿ ڏيو.

مجلس رابطه براءِ عالمات

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوت اسلامي سڄي دنيا ۾ 200 کان وڌيڪ ملڪن ۾ ڪم و پيش 105 شعبه جات ۾ نيڪي جي دعوت جون ڌومون مچائڻ ۾ مصروفِ عمل آهي. انهيء شعبن مان هڪ شعبو ”مجلسِ رابطه براءِ عالمات) به آهي. هن شعبي جو بنيادي مقصد سنّي عالمات کي دعوت اسلامي جي ديني خدمتن کان آگاهه ڪرڻ، مدني ڪمن ۾ انهن جي مدد حاصل ڪرڻ، انهن جون دعائون وٺڻ، سنّي جامعات ۽ مدرسن ۾ مدني ڪمن جي ترڪيب بڻائڻ، هفته وارا سنتن ڀري اجتماع ۾ سنّي جامعات ۽ مدرسن جي طالبات جي شرڪت ڪرائڻ آهي. الله ڪريم دعوت اسلامي ۽ ”مجلس رابطه براءِ عالمات“ کي وڌيڪ ترقيون عطا فرمائي ۽ اسان کي دعوت اسلامي جي مدني ماحول ۾ استقامت نصيب فرمائي . اٰمِــيْن بِـجَـاهِ الـنَّـبِـیِّ الْاَمِــيْن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

پياريون پياريون اسلامي ڀينرون! بيان کي اختتام جي طرف آڻيندي ذڪر ۽ دورود جا ڪجھ مدني گل ٻڌڻ جي سعادت حاصل ڪنديون سين. پهريان فرامينِ مصطفيٰ ملاحظه ڪيو: (1) فرمايائين: پنهنجي ربِّ ڪريم جو ذڪر ڪرڻ وارن ۽ نه ڪرڻ وارن جي مثال زنده ۽ مرده جي طرح آهي. (بخاری، کتاب الدعوات، ۴/۲۲۰، حدیث:۶۴۰۷)

(1) فرمايائين: قيامت جي ڏينهن ماڻهن ۾ منهنجي تمام ويجهو اهو هوندو جنهن دُنيا ۾ مون تي زياده دُرودِ پاڪ پڙهيا هوندا. (تِرمِذی، ۲/۲۷، حدیث:۴۸۴)

پياريون پياريون اسلامي ڀينرون! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! ايندڙ هفتيوار اجتماع ۾ اسان اصحابِِ صُفَّہ عَـلَيْهِمُ الرِّضْوَانْ جي شان ۽ عظمت ۽ انهن جي قُربانين جي باري ۾ ٻڌنديون سين. هنن الله جي پيارن مان ڪجھ مشهور صحابي سڳورن عَـلَيْهِمُ الرِّضْوَانْ جي مبارڪ زندگين جي باري ۾ ڪجھ ايمان افروز واقعا به ٻڌنديون سين. اصحابِ صُفَّہ جي سيرت مان اسان کي ڪهڙو درس ملندو آهي، اهي ڪهڙا سبب هئا جو اصحابِ صُفَّہ پنهنجا گهرٻار، اهل ۽ عيال ۽ فڪرِ روزگار سان بي نياز ٿي ڪري مسجدِ نبوي ۾ ئي رهندا هئا، هي سڀ ايندڙ اجتماع جي بيان ۾ اسان ٻڌڻ جي سعادت حاصل ڪنديون سين، تنهنڪري اچڻ واري اربع تي به اجتماع ۾ حاضري جي نيت ڪيو ۽ هٿ مٿي ڪري چئو؛ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ!
