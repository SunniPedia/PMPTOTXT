محشر جي ڏينهن رُسوائي جو تصور هوندو ته اسان گھٽ گفتگو ڪرڻ وارا ۽ فُضُول گفتگو کان بچڻ وارا بڻجي وينداسين ڇوته جنهن کي ڪو دُنياوي فڪر لاحق ٿيندي آهي ته هو ان جي باري ۾ سوچيندو رهندو آهي، مثال طور ڪنهن جي ماءُ پيءُ يا رشتيدار جو اوچتو انتقال (Death) ٿي وڃي، يا هلندڙ ڪاروبار اوچتو ڪنهن حادثي جو شڪار ٿي وڃي يا بند ٿي وڃي، ڳالھ ڳالھ تي گهريلو جهيڙا شروع ٿي وڃن ته ٻانهي تي خاموشي طاري ٿي ويندي آهي، اهڙي طرح جيڪڏهن اسان کي جهنم جي عذابن جو تذڪرو ٻڌي فڪرِ آخرت نصيب ٿي وڃي، خوفِ خدا سان اسان جي دل ڏڪي وڃي ته اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ اسان جي چپن تي به خاموشي طاري ٿي ويندي۔ پر اسان ته ڪنهن گھڙي خاموش نٿا رهون، هر وقت بغير سوچي سمجهي ڳالهيندا رهندا آهيون۔ جيڪڏهن اسان پنهنجي پياري آقا مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مبارڪ سيرت جو مطالعو ڪريون ته خبر پوندي ته اسان جا پيارا آقا صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تمام گھٽ گفتگو ڪندا هئا، اڪثر خاموش رهندا هئا جيئن ته حضرت سيدُنا جابر بن سَمُرَه رَضِىَ اللّٰه تَعَالٰى عَنْه فرمائن ٿا ته رسولِ پاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ تمام گھڻو خاموش رهندا هئا۔ (مسندامام احمد،مسندالبصريين،حديث جابربن سمرۃ،۷/ ۴۰۴، حديث:۲۰۸۳۶ملتقطا)

حڪيمُ الامت حضرت مُفتي احمد يار خان نعيمي رَحْمَةُ اللهِ تَعَالٰي عَلَيْه هن حديثِ پاڪ جي تحت فرمائن ٿا: خاموشي مان مُراد دُنياوي ڪلام کان خاموشي (آهي) نه ته حضورِ اقدس (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي زبان شريف الله (پاڪ) جي ذڪر ۾ تر رهندي هئي، ماڻهن کان بلاضرورت گفتگو نه ڪندا هئا، هي ذڪر آهي جائز ڪلام جو، ناجائزڪلام ته سڄي عُمر زبان شريف تي آيو ئي نه۔ حُضُور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) سراپا حق آهن، پوءِ اوهان تائين باطل جي رسائي ڪيئن ٿئي۔ (مرآۃ المناجيح،۸/۸۱)

(3)غيبت ۽ لڙائي جهيڙي ۾ مبتلا ٿي ويندو آهي

مٺا مٺا اسلامي ڀائرو! فضول ڳالهيون ۽ فضول سوال عام طور تي غيبت ۽ لڙائي جهيڙي جو سبب به بڻجي ويندا آهن، جيڪڏهن ڪنهن صحيح مقصد جي لاءِ ڪا ڳالھ پڇي وڃي ته حرج نه آهي پر عام طور سوالن جو صحيح مقصد ناهي هوندو، بلڪ پڇڻ صرف پڇڻ هوندو آهي مثال طور ڪنهن جي ڪا شيء ڏٺي ته ان کان پڇڻ ته اها ڪيتري ۾ ورتي آهي؟ ڪٿان کان ورتي آهي؟ ان جي گارنٽي ڪيترن سالن جي آهي؟ ياد رکو! بنا سبب اهي ڳالهيون پڇڻ فُضول گفتگو ۾ شمار ٿينديون آهن ۽ آخرت ۾ ان جو حساب ڏيڻو پوندو۔ ڪڏهن ڪڏهن ان سوالن سان غيبت، چغلي ۽ جھيڙي جھٽي جو دروازو کلي ويندو آهي، مثال طور ڪنهن ڪرائي تي گھر ورتو ته سوال ٿيندو آهي، ڪيترن ڪمرن جو آهي؟ ڪرايو ڪيترو آهي؟ ۽ مڪان مالڪ ڪيئن آهي؟ مالڪ مڪان جي باري ۾ اهو سوال نهايت خطرناڪ آهي ان جو جواب عام طور بلا اجازتِ شرعي ڪجھ ان طرح گناهن ڀريو ملندو آهي ته اسان جو مڪان مالڪ نهايت سخت مزاج آهي۔ بي رحم آهي ۽ خَردماغ آهي، ڪرائي ۾ هڪ ڏينهن جي به دير برداشت نٿو ڪري۔ ياد رکو! اهو سڀ غيبت جي زُمري ۾ اچي ٿو ۽ ڪنهن مسلمان جي غيبت ڪرڻ پاڻ کي هلاڪت ۾ مبتلا ڪرڻ آهي جيئن غيبت جون تباھ ڪاريون صفحي 26 تي آهي: غيبت ايمان کي ڪپي ڇڏيندي آهي ٭ غيبت بُري خاتمي جو سبب آهي ٭ گھڻي غيبت ڪرڻ واري جي دعا قبول ناهي ٿيندي ٭ غيبت سان نَماز روزي جي نورانيت ختم ٿي ويندي آهي ٭ غيبت سان نيڪيون برباد ٿينديون آهن ٭ غيبت نيڪيون ساڙي ڇڏيندي آهي ٭ غيبت ڪرڻ وارو توبه ڪري به وٺي تڏهن به سڀني کان آخِر ۾ جنَّت ۾ داخِل ٿيندو، مطلب ته غيبت گناھ ڪبيره، قطعي حرام ۽ جهنَّم ۾ وٺي وڃڻ وارو ڪم آهي۔

(4)ڪفريات ڳالهائي ڇڏيندو آهي

مٺا مٺا اسلامي ڀائرو! اڄوڪي دور ۾ علم کان دُوري جي ڪري ڪي ماڻهو ايتري قدر بي لگام ٿيندا پيا وڃن جو جيڪو وات ۾ ايندو آهي هڪدم بڪي ڇڏيندا آهن، گھڙي کن به نٿا غورڪن ته مان ڇا چئي رهيو آهيان؟ فضول گوئي جي اها عادت انسان کي نه ڳالهائيندڙ شيء ڳالهرائيندي آهي ۽ وڏي مصيبت ۾ ڦاسائيندي آهي۽ بعض اوقات انسان پنهنجي زبان مان ڪفريه جملا ڪڍي پنهنجو سڀ کان قيمتي سرمايي ايمان مان هٿ کڻي ويهندو آهي پر ان کي خبر تائين ناهي پوندي ته هو مسلمان ناهي رهيو ۽ ان جو نڪاح به ٽٽي ويو، ڪي ماڻهو ته گھڻا ئي بي حس هوندا آهن، ڳالھ ڳالھ تي هروڀرو ائين تائيد طلب ڪندا آهن، ڪيئن ادا! ٺيڪ آهي نه! مان غَلَط ته نه پيو چوان!'' اوهان جو ڇا خيال آهي؟'' هاڻي ڳالھ ڀلي ناقابلِ قبول هجي پر نه چاهيندي به ها ۾ ها ملائي ڪوڙ ڳالهائڻ جو گناھ ڪرڻو پوندو آهي، ان طرح فضول ڳالهائڻ وارا ماڻهو ڪڏهن ته گمراهي جون ڳالهيون بلڪه مَعَاذَ الله عَزَّ وَجَلَّ! ڪفريات بڪي به عادت مطابق تائيد حاصل ڪندا آهن ۽ ان طرح جا جملا مثال طور: ''ڪيئن سائين ٺيڪ چئي رهيو آهيان نه؟'' چئي سامهون واري کان ها چورائي بعض اوقات ان جو به ايمان برباد ڪرائيندا آهن؟ ڇوته هوش ۽ حواس سان ڪفر جي تائيد به ڪفر آهي۔ ڪفريه ڪلمات جي باري ۾ مزيد اهم معلومات جي لاءِ شيخِ طريقت ،اميراهلسنت دَامَتْ بَرَکاتُهمُ الْعَالِيه جو بهترين ڪتاب ”ڪفريه ڪلمات جي باري ۾ سوال جواب “ جو مطالعو انتهائي مفيد آهي ۔

ڪٿي اهو فُضول ڪلام نه ٿئي

٭ … امير المؤمنين حضرت سَيدُنا ابو بڪر صديق رَضِى الله تَعَالٰي عَنْه گفتگو کان بچڻ جي لاءِ پنهنجي وات ۾ پٿري رکندا هئا ۽ پنهنجي زبان ڏانهن اشارو ڪري فرمائيندا هئا: هيء ئي اها شيءَ آهي جيڪا مون کي هلاڪت واري جڳهن تي کڻي وئي آهي۔

٭ … هڪ صحابي رَضِى الله تَعَالٰي عَنْه ارشاد فرمائن ٿا: هڪ شخص مون سان ڪا ڳالھ ڪري ٿو ته ان جو جواب ڏيڻ مون کي ايترو وڌيڪ مرغوب ۽ پسنديده هوندو آهي جيترو هڪ اُڃايل شخص کي ٿڌو پاڻي به ناهي هوندو پر مان ان خوف سان ان جو جواب ناهيان ڏيندو ته ڪٿي هي فضول ڪلام نه هجي۔ (احياءالعلوم ،۳/ ۳۴۸)

٭ … حضرت سيدُنا موسٰي بن علي رَحْمَةُ اللهِ تَعَالٰي عَلَيْه فرمائن ٿا: بني اسرائيل جي هڪ راهب (يعني دنيا کان الڳ ٿلڳ عبادت ۾ مشغول شخص) جو قول آهي ته عورت جي زينت شرم ۽ حيا ۽ عقل مند جي زينت خاموشي آهي۔ (ايڪ چپ سو سڪه،ص۱۶)

٭ … حضرت سيدُنا رُبَيع بن خَيثَم رَحْمَةُ اللهِ تَعَالٰى عَلَيْه ويھ سالن تائين دُنياوي گفتگو نه ڪئي۔ جڏهن صبح ٿيندي هئي ته مس، ڪاغذ ۽ قلم رکندا هئا ۽ جيڪا گفتگو به ڪندا هئا ان کي لکي وٺندا هئا پوءِ شام جي وقت پنهنجي نفس جو مُحاسبو ڪندا هئا۔ (احياء العلوم،۳/۳۳۸)

مٺا مٺا اسلامي ڀائرو! ٻڌو اوهان ته اسان جي بُزرگانِ دين وٽ زبان جي ڪيتري قدر اهميت هئي جو اهي نيڪ ماڻهو هر گهڙي ذڪر۽ دُرُود ۾ گذارڻ جي باوجود به فضول ڳالهين کان بچڻ جي لاءِ وات ۾ پٿر (Stone) رکندا هئا ۽ ڪا ڳالھ ڪندا هئا ته ان کي لکي وٺندا هئا ۽ شام جو ان جي باري ۾ غور ۽ فڪر ڪندا هئا۔ اڄوڪي دور ۾ ان جي جهلڪ اسان جي شيخ ِطريقت،امير اهلسنَّت دَامَتْ بَرَکاتُهمُ الْعَالِيه ۾ نظر ايندي آهي، پاڻ دَامَتْ بَرَکاتُهمُ الْعَالِيه نه صرف پاڻ انهن بزرگن جي تعليمات تي عمل ڪرڻ وارا آهن بلڪ سندن فيضِ تربيت سان لکين ماڻهو به راھ هدايت تي گامزن ٿي چڪا آهن۔ شيخِ طريقت، اميرِ اهلسنَّت دَامَتْ بَرَکاتُهمُ الْعَالِيه اسان کي زبان جي حفاظت جي باري ۾ ڪيئي مَدني اِنعامات عطا ڪيا آهن: مثال طور مدني انعام نمبر 29 ۾ فرمائن ٿا: ڇا اڄ توهان ڪنهن کان اهڙا فضول سوال ته نه ڪيا جن جي ذريعي مسلمان عمومًا ڪوڙ جي ڪبيره گناهه ۾ مبتلا ٿي ويندا آهن. اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن

بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([1])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([4])

﴿ 6 ﴾ دُرُود شَفَاعت

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([1])
