پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ جَنَّت جي نوجوانن جا سردار آهن، اِمَام حَسَن مُجتبٰي رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي وِلادتِ مُبارَڪه 15 رَمَضانُ المُبارَڪ 3 ہجري تي رات جو مدينهء طيّبه زَادَھَا الـلّٰـهُ شَرَفًاوَّتَعۡظِیۡمًا ۾ ٿي۔ حُضُور سَيّدِ عَالم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن ستين ڏينهن اوهان جو عقيقو ڪيو ۽ وار جُدا ڪيا ويا ۽ حُڪم ڏنو ته وارن جي وزن جي برابر چاندي صَدَقو ڪئي وڃي۔

(تاریخ الخلفاء، باب الحسن بن علی بن ابی طالب، ص۱۴۹و روضۃ الشہداء (مترجم)، باب ششم،ج۱،ص۳۹۶)

سَيِّدُنَا اِمَام حَسَنِ مُجتبٰي رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو نالو، اِمَامُ الانبياء، سَيّدُ الاَسۡخِياء صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن رکيو۔ مڪمل واقِعو ڪجھ هن طرح آهي ته حَضۡرَتِ سَيّدَتُنا اسماء بنتِ عُميس رَضِىَ اللهُ تَعَالٰی عَـنْهَا جن بارگاھ رِسالَت صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ۾ حَضۡرَتِ سَيِّدُنَا اَمَام حَسن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي وِلادت جي خوشخبري پہچائي۔ (ته) حُضُور پُرنُور، شافعِ يومُ النُّشور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم تشريف وٺي آيا ۽ فرمايو ته اسماء منهنجي فرزند کي آڻيو، حَضۡرَتِ ِاسماء رَضِىَ اللهُ تَعَالٰی عَـنْهَا (اِمَام حسن کي) هڪ ڪپڙي ۾ (ويڙهي) حُضور سَيّدِ عَالَم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي خدمت ۾ حاضر ڪيو۔ حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن ساڄي ڪن ۾ اَذان ۽ کاٻي ۾ تڪبير فرمائي ۽ حَضۡرَتِ سَيِّدُنَا مولٰي عَلِيُّ المُرتَضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم کان پڇا فرمايو: توهان ان فرزندِ اَرۡجُمند جو ڇا نالو رکيو آهي؟ عرض ڪيو: يارَسُولَ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم! منهنجي ڪهڙي مَجال جو بغير اِذن ۽ اِجازت نالو رکڻ تي اڳرائي ڪريان، پر هاڻي توهان دريافت فرمايو آهي ته منهنجو خيال آهي ’’ حَرب‘ ‘ نالو رکيو وڃي، باقي اوهان مُختار آهيو، ته حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن شهزادي جو نالو حَسن رکيو۔ (سوانحِ کربلا ص۹۲ملخصاً)

وہ حسن مُجتبیٰ سیدُ الاسخیاء

راکبِ دوشِ عزّت پہ لاکھوں سلام

(حدائقِ بخشش)

شعر جي وضاحت: اهي اِمَام حسن مجتبٰي رَضِىَ اللهُ تَعَالٰی عَـنْهُ جيڪي سخين جا سردار آهن، جيڪي پنهنجي ناني جان، محبوبِ رحمٰن صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي پياري ڪلهن تي سُوار ٿيندا هئا،اُن جي ذاتِ مبارڪ تي لکين سلام هجن۔

سندن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي ننڍي ڀاءَ سَيدُالشُّہَداء، راڪبِ دَوشِ مُصطفٰي، حَضۡرَتِ

سَيِّدُنَا اِمَام حُسَيۡن رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي وِلادت 5 شعبانُ الۡمُعَّظم سن 4 ہجري مدينهء مُنوَّرہ زَادَھَا الـلّٰـهُ شَرَفًاوَّتَعۡظِیۡمًا ۾ ٿي۔ سندن نالو، حُضُور پرنور، شافع ِ يومُ النُّشور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ’’ حُسَيۡن ‘‘ ۽ ’’ شبير ‘‘ رکيو جڏهن ته سندن ڪنيت ’’ اَبُوعَبدُاللہ‘ ‘ ۽ لَقب به ’’سِبۡطُ رَسُوۡلِ اللہِ (يعني رَسُوۡلُ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جا ڏوهٽا) ‘‘۽’’ رَيحَانَةُ الرَّسُول ‘‘ (يعني رسول صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جو گُل) آهي ۽ پنهنجي وڏي ڀاءُ جيان پاڻ به جنَّتي جوانن جا سردار آهن۔

(اسدالغابة، باب الحاء والحسین،۱۱۷۳۔ الحسین بن علی،ص۲۵،۲۶ملتقطاًوسیر اعلام النبلاء،۲۷۰۔الحسین الشہید...الخ، ج۴،ص۴۰۲۔۴۰۴)

مٺا مٺا اسلامي ڀائرو! هينئر اسان ٻڌو ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن پنهنجن پيارن ڏُوهٽن جا نالا پاڻ تجويز فرمايا۔ اچو ته! انهيءَ متعلق نالي رکڻ جا ڪجھ آداب به ٻُڌون۔ سُٺو نالو رکڻ اولاد جي حقن ۾ شامل آهي ۽ والِدين جي طرفان پنهنجي ٻار جي لاءِ سڀ کان پہريون ۽ بُنيادي تحفو به آهي، جنهن کي اهو پوري عُمر پنهنجي سيني سان لڳائي رکندو آهي، ايستائين جو ميدانِ حَشر برپا ٿيندو ته اهو انهيءَ نالي سان مالڪِ ڪائِنات عَزَّوَجَلَّ جي سامهون گھُرايو ويندو، جيئن ته حَضۡرَتِ سَيِّدُنَا ابودَرداء رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان مَروي آهي ته حُضُورِ پاڪ، صاحبِ لَولاڪ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: ” قِيامت جي ڏينهن توهان پنهنجي ۽ پنهنجي ابن ڏاڏن جي نالن سان سَڏِيا ويندؤ، تنهنڪري پنهنجا سُٺا نالا رکندا ڪريو۔ “

(سنن ابی داود ،کتاب الادب باب فی تغییر الاسماء ،الحدیث ۴۹۴۸،ج۴،ص۳۷۴)

مٺا مٺا اسلامي ڀائرو! هن حديثِ پاڪ مان اُهي ماڻهو عِبرت حاصِل ڪن جيڪي پنهنجي ٻارن جا نالا ڪنهن گلوڪار، فِلمي اداڪار يا مَعَاذَاللّٰه عَزَّوَجَلَّ ڪافرن جي نالن تي رکي ڇڏيندا آهن، ان کان وڌيڪ بَدتَرين ذِلّت ڇا هوندي جو مُسلمان جي اولاد کي ميدانِ مَحشَر ۾ ڪافرن جي نالن سان سڏيو وڃي۔ وَالعِيَاذُ بِاللّٰه ، اسان جي مُعاشَري ۾ ٻار جي نالي جو انتخاب ڪرڻ جي ذميداري عام طور تي ڪنهن قريبي رشتيدار مثلاً ڏاڏي، پُڦي، چاچي وغيرہ کي سونپي ويندي آهي ۽ اڪثر ڪري علمِ دِين کان دُوري جي سبب اهي ٻار جو اهڙو نالو رکي ڇڏيندا آهن، جنهن جو ڪو مَطلب ئي نه هوندو آهي يا وري سٺي مَعنٰي نه هوندي آهي، يا وري نالو شرعًا دُرست نه هوندو آهي، اهڙا نالا رکڻ کان بچڻ گھرجي، ڪڏهن ڪڏهن اهڙو نالو به ڳوليو ويندو آهي جيڪو گھر، خاندان، محلي ۾ پري پري تائين ڪنهن جو به نه هجي، جڏهن به ڪو ٻڌي ته فورًا هي چوي ته اهو نالو ته پهريون ڀيرو ٻڌو آهي، ڪيڏو نه زبردست نالو رکيو آهي! اهي لفظ ٻڌي نالي رکڻ وارو ڪپڙن ۾ ناهي ماپندو، پر اهڙن کي سوچڻ گھرجي ته متان ڪٿي اها خوشي حُبِّ جاھ (يعني تعريف جي خواهش) جي مرض جو نتيجو ته ناهي؟ ، تنهنڪري اَنبِياءِ ڪِرام عَلِيـھمُ الـصَّلٰوة ُ وَالـسَّلَام جي اسماءِ مبارڪه، صحابهِ ڪِرام، تابعين عُظام ۽ اولياءِ ڪِرام رَحِمَهُمُ الـلّٰـهُ السَّلَام جي نالن تي نالا رکڻ گھرجن جنهن جو هڪ فائدو هي ٿيندو ته ٻار جو پنهنجي اَسلاف ڪرام (يعني بُزرگن) سان روحاني تعلق قائم ٿي ويندو ۽ ٻيو انهن نيڪ هستين جو نالو رکڻ جي برڪت سان ٻار جي زندگي تي مدني اثرَ مرتب ٿيندا۔ نالن جي حَوالي سان وڌيڪ دِلچسپ ۽ حيرت انگيز مَعلُومات حاصِل ڪرڻ جي لاءِ مڪتبة المدينة جو شايع ٿيل تقريبًا 180صَفَحن تي مُشتمِل ڪتاب ” نام رکھنے کے احکام “ جو مُطالَعو

ڪريو جو ان ڪتاب ۾ ٻارن جا نالا رکڻ جي لاءِ سَوين سُٺن نالن جي فہرست موجود آهي، گڏوگڏ ان کان عِلاوه ٻارن جا نالا رکڻ جي باري ۾ ڪثير مَدَني گُل پنهنجي خُوشبُو مَہڪائي رهيا آهن۔

صَــلُّـوا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! نبي ڪريم، رئوف رَّحيم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم مُختلف مَوقعن تي انهن حضرات جو اهڙو شان ۽ عَظَمَت بيان فرمائي جنهن کي ٻُڌي ڪري اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ توهان جي دل ۾ حَسنَين ڪَريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جي مَحَبَّت وڌندي۔ اچو ته انهن جي شَان ۽ عظمت سان تَعلق رکندڙ ڪجھ فرامينِ مُصطفٰي صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ٻُڌون۔

* هُمَا رَيْحَانَتَايَ مِنَ الدُّنْيَا يعني حَسَن ۽ حُسَيۡن (رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) دُنيا ۾ منهنجا ٻه (2) گُل آهن۔ (صحیح البخاری،کتاب فضائل اصحاب النبی،باب مناقب الحسن والحسین، الحدیث:۳۷۵۳،ج۲،ص۵۴۷)

عاشقِ صحابه ۽ اہلبيت سيدي اعلٰي حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه بارگاھِ رسالت ۾ عرض ڪن ٿا:

(حدائقِ بخشش)

* اَلْحَسَنُ وَالْحُسَيۡن سَيِّدَا شَبَابِ اَهْلِ الْجَنَّةِ يعني حَسَن ۽ حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جَنَّتي نوجوانن جا سردار آهن۔ (ترمذی ،ج۵،ص۴۲۶،ح۳۷۹۳)

حَضۡرَتِ سَيِّدُنَا عَبدُاللہ بن عباس رَضِیَ اللهُ تَعَالٰی عَنْهُمَا فرمائن ٿا ته: جڏهن هي آيتِ مُبارَڪه (سيپارو 25، سُورَةُ الشُّورٰي آيت نمبر 23) نازِل ٿي: ” قُلۡ لَّاۤ اَسۡـَٔلُکُمۡ عَلَیۡہِ اَجۡرًا اِلَّا الۡمَوَدَّۃَ فِی الۡقُرۡبٰی ؕ تَرجَموڪنزالايمان: تون فرماءِ مان ان تي توهان کان ڪو به بدلو نه ٿو گھُران، مگر قَرابَت جي مَحَبّت ‘‘

ته صَحابهء ڪرام عَلِيهِمُ الرِّضْوَان جن عَرض ڪيو: يارَسُولَ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم! توهان جا اهي ڪهڙا قَرابَت دار (يعني رشتيدار) آهن، جنهن سان مَحَبَّت ڪرڻ اسان تي واجب آهي؟ حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: عَلِي المُرتَضٰي ، فاطِمَةُ الزَّہرا ۽ سندن ٻئي پُٽ (يعني حَضۡرَتِ سَيِّدُنَا اِمَام حسن ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) ۔

مٺا مٺا اسلامي ڀائرو! مَعلُوم ٿيو ته اَہلِبيت جي مَحَبَّت واجب ۽ ضَروري آهي، ہر مُسلمان جي ويجهو اَہلِ بيتِ ڪرام، پنهنجي جان ۽ مال، عِزَّت ۽ آبرو، ماءُ پيءُ ۽ اَولاد کان به وڌيڪ محبوب ہئڻ گھرجن۔ انهن مبارڪ ہستين جي مَحَبَّت ، سَيدِ عالَم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي مَحَبَّت آهي ۽ حُضُور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي مَحَبَّت، ڪامل اِيمان جي نِشاني آهي۔ جيئن ته

نبي رَحمت، شَفيعِ اُمَّت صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عاليشان آهي: ” لَا یُؤْمِنُ عَبْدٌحَتّٰی اَکُوْنَ اَحَبَّ اِلَیْہِ مِنْ نَفْسِہ ٖ “ يعني ڪو ٻانهو ڪامل مؤمن نٿو ٿي سگھي، جيستائين جو اهو مونکي پنهنجي جان کان وڌيڪ نه چاهي ” وَذَاتِیْ اَحَبَّ اِلَیْہِ مِنْ ذَاتِہِ “ ۽ منهنجي ذات، ان کي پنهنجي ذات کان وڌيڪ محبوب نه ہجي ” وَتَکُوْنَ عِتْرتِیْ اَحَبَّ اِلَیْہِ مِنْ عِتْرَتِہِ “ ۽ منهنجي اَولاد ان کي پنهنجي اولاد کان وڌيڪ پياري نه ہجي” وَاَهْلِیْ اَحَبَّ اِلَیْہِ مِنْ اَهلِہِ “ ۽ منهنجا اَہلِ بيت ان کي پنهنجي گھروارن کان وڌيڪ پيارا ۽ محبوب نه ہجن۔

(شعب الایمان، باب فی حب النبی، ۲/۱۸۹،حدیث: ۱۵۰۵بتصرف ما)

مٺا مٺا اسلامي ڀائرو! اَہلِ بيتِ اَطہار عَلِيهِمُ الرِّضْوَان جي شان ۾ اللہ عَزَّوَجَلَّ سيپاري 22، سُوْرَۃُ الْاَحْزَاب جي آيت نمبر 33 ۾ ارشاد فرمائي ٿو:

اِنَّمَا یُرِیۡدُ اللہُ لِیُذۡہِبَ عَنۡکُمُ الرِّجۡسَ اَہۡلَ الۡبَیۡتِ وَ یُطَہِّرَکُمۡ تَطۡہِیۡرًا ﴿ۚ۳۳﴾

تَرجَموڪنزالايمان: الله ته اهو چاهي ٿو ته نبيءَ جا گھروارؤ جو توهان کان هر ناپاڪي دور فرمائي ڇڏي ۽ توهان کي پاڪ ڪري خوب سُٺو ڪري ڇڏي.

اڪثر مُفَسِّرينِ ڪرام جي راءِ آهي ته هي آيتِ مُبارَڪه حَضۡرَتِ سَيِّدُنَا عَلِي المُرتَضٰي ، حَضۡرَتِ سَيدہ فاطِمه زَہرا ، حَضۡرَتِ سَيِّدُنَا اِمَام حَسَن ۽ حَضۡرَتِ سَيِّدُنَا اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جي حق ۾ نازِل ٿي۔ اِمَام احمد رَحْمَةُ اللهِ تَعَالٰی عَلِيه حَضۡرَتِ ابُوسَعيد خُدري رَضِیَ اللهُ تَعَالٰی عَنْهُ کان رِوايت ڪئي آهي ته هي آيت پَنجتَن پاڪ جي شان ۾ نازِل ٿي۔ پَنجتَن مان مُراد حُضُور نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ۽ حَضۡرَتِ عَلِي ، حَضۡرَتِ فاطمہ ، حَضۡرَتِ اِمَام حَسَن ۽ حَضۡرَتِ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا آهن۔ (سوانح کربلا، ص۷۹،۸۰،ملتقطا)

هڪ رِوايت ۾ هي به آهي ته حُضُور، جانِ عالَم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم انهن حضرات سان گڏوگڏ پنهنجي صاحِبزادين ۽ قَرابت دارن (يعني رشتيدارن) ۽ اَزواجِ مُطَہَّرات کي به شامِل فرمايو۔ (الصواعق المحرقۃ، الباب الحادی عشر، الفصل الاول، ص۱۴۴)

آيتِ مُبارَڪه جي تَفسِير ڪندي، اِمَام طَبري رَحْمَةُ اللهِ تَعَالٰی عَلِيه فرمائن ٿا: يعني اي آلِ محمد (صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم)! اللہ تعالٰي چاہي ٿو ته توهان کان بُرين ڳالھين ۽ فُحش شين کي پَري رکي ۽ توهان کي گُناہن جي ميراڻ کان پاڪ ۽ صاف ڪري ڇڏي۔ (طبری ،پ ۲۲،الاحزاب،تحت الآیۃ۳۳ ،ج۱۰ص۲۹۶)

صَدۡرالاَفاضِل حَضۡرَتِ علامه مولانا سيِّد محمد نعيمُ الدّين مراد آبادي رَحْمَةُ اللهِ تَعَالٰی عَلِيه فرمائن ٿا: هي آيت ِڪريمه اَہلِ بيتِ اَطهَار جي فَضيلتن جو چشمو آهي ۽ معلوم ٿئي ٿو ته سمُورا اَخلاقِ دَنِيَّه ۽ اَحوالِ مَذمُومَه (يعني بُري اَخلاقن ۽ اَحوالن) کان انهن جي تطهير فرمائي وئي (يعني اِنهن کي بُري اَخلاقن کان محفوظ رکيو ويو) ۔ ڪجھ حديثن ۾ مَروِي آهي ته اَہلِ بيت ، باھ تي حَرام آهن (يعني اَہلِ بيت جنَّتي آهن) ۽ هي ان تَطہِير جو فائدو ۽ ثَمر آهي ۽ جيڪا شيء انهن جي

اَحوالِ شَريفه جي لائق نه ہجي ان کان انهن جو پَرۡوَرۡدۡگار عَزَّوَجَلَّ انهن کي محفوظ رکندو ۽ بچائيندو آهي۔ (سوانح کربلا،ص۸۲)

اسان کي به اَہلِ بيتِ پاڪ سان مَحَبَّت قائم رکندي، انهن جي نقشِ قدم تي هلڻ جي ڪوشِش ڪرڻ گھرجي، اللہ عَزَّوَجَلَّ انهن جي صَدقي اسان کي به گناہن کان بچڻ جي توفيق عطافرمائي ۽ خُوب خُوب نيڪيون ڪري جنّت ۾ ان نيڪ ہستين جو قُرب عطا فرمائي۔

اٰمِين بِجَاهِ النّبيِّ الاَمِين صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم

پارَ ہائے صُحُف غُنْچہ ہائے قُدُس

اَہْلِ بیتِ نَبوّت پہ لاکھوں سلام

آبِ تَطہیر سے جس میں پودے جمے

اس رِیاضِ نَجابت پہ لاکھوں سلام

خونِ خَیْرُ الرُّسُل سے ہے جن کا خمیر

اُن کی بے لَوث طِینت پہ لاکھوں سلام

صَــلُّـوا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! نبي ڪريم، رءُوفٌ رَحِيۡم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم کي اَہلِ بيت مان سڀ کان ع َ زِيز ۽ محبوب حَسنَين ڪَريمَين هئا ، حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ڪڏهن ٻنهي شهزادن کي پنهنجي ڪُلهن تي سُوار ڪري وٺندا هُئا، ايستائين جو نماز ۾ سجدي جي حالَت ۾ ٻئي پُشتِ اَطہر (يعني پُٺي مبارڪ) تي سُوار ٿيندا هئا ته حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سَجدو طَويل (ڊگھو) ڪري ڇڏيندا هئا ۽ جڏهن سجدي مان مٿو مبارڪ کڻندا هئا ته اِنهن کي آرام سان زمين تي ويھاري ڇڏيندا هئا۔

حَضۡرَتِ سَيِّدُنَا ابُوهرَيۡرَه رَضِیَ اللهُ تَعَالٰی عَنْهُ کان مَروي آهي ته هڪ ڏينهن اسان سرڪارِ نامدار، مديني جي تاجدار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان گڏ نمازِ عِشاء ادا ڪري رهيا هئاسين، سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جڏهن سَجدي ۾ ويا ته اِمَام حَسَن ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا ، حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي پُشت مُبارَڪ (پُٺي مبارڪ) تي سُوار ٿي ويا۔ سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن سجدي مان مٿو مبارڪ کنيو ته انهن کي نَرمي سان پڪڙي ڪري زمين تي ويهاري ڇڏيو، پوءِ جڏهن سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ٻيهر سجدي ۾ ويا ته اِمَام حَسَن ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جن ٻيهر ايئن ئي ڪيو، ايستائين جو حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن نماز مڪمَّل فرمائي ورتي ۽ انهن ٻنهي کي پنهنجي رانن تي ويهاري ڇڏيو۔ (مسند احمد،مسند ابی ھریرة، ۳/۵۹۲،حدیث:۱۰۶۶۴ بدون الی امھما،البداية والنهاية ثم دخلت سنة احدوستین،۵/۷۱۶) انهيءَ طرح ننڍپڻ ۾ هڪ ڀيري خُطبي جي دوران ٻئي شہزادا مسجد ۾ تَشريف وٺي آيا، ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم خُطبو ڇڏي ڪري انهن جي ويجھو ويا ۽ انهن کي کڻي ڪري پنهنجي سامهون ويهاريو۔ (ترمذی،ج۵،ص۴۲۹،ح۳۷۹۹)

حَضۡرَتِ سَيِّدُنَا عُرۡوَه بن زُبَير رَضِیَ اللهُ تَعَالٰی عَنْهُ پنهنجي والِد کان رِوايت ڪن ٿا: هڪ ڀيري سرڪارِ دوعالم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم اِمَام حَسَن رَضِیَ اللهُ تَعَالٰی عَنْهُ کي چُميو، سِنگھيو ۽ ڇاتيءِ سان لڳائي ورتو، ان وقت حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي ويجھو هڪ انصاري صَحابي بيٺل هئا، انهن سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي اِمَام حَسَن رَضِیَ اللهُ تَعَالٰی عَنْهُ تي ايتري قدر شَفۡقَت ڏسي ڪري عرض ڪيو: يارَسُوۡلَ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم! منهنجو به هڪ پُٽ آهي،

جيڪو هاڻي بالغ ٿي چڪو آهي، پر مون ان کي ڪڏهن ناهي چُميو، سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪڏهن اللہ عَزَّوَجَلَّ تنهنجي دل مان رحمت ڪڍي ورتي آهي ته ان ۾ منهنجو ڪهڙو قُصُور (ڏوھ) آهي۔

(المستدرک، من فضائل الحسن بن علی ،ج۴،ص ۱۶۱،حدیث:۴۸۴۶)

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته اسان کي به پنهنجي ٻارن سان پيار ۽ مَحَبَّت سان پيش اچڻ، ہر معاملي ۾ انهن سان شفقت ڀريو سُلوڪ ڪرڻ ۽ انهن کي پاڻ سان گڏ کارائڻ گھرجي۔ ڳالھ ڳالھ تي مارڪُٽ ڪرڻ، جِھڻڪڻ، اکيون ڏيکارڻ اِنتہِائي نُقصان جو باعِث بڻجي سگھي ٿو، تنهنڪري ٻارن جي دِلجُوئي ۽ انهن جي بہتر تَربِيت ۽ پَروَرِش جي پُوري پُوري ڪوشِش ڪرڻ گھرجي۔ مڪتبة المدينه جو ڪتاب ” تربيتِ اولاد “ حاصل ڪريو، توهان کي معلوم ٿيندو ته اولاد جي تربيت ڪيئن ڪرڻي آهي؟ اهڙي طرح سرڪارِ اعَلِيٰ حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه جو رِسالو ” اولاد کے حقوق “ جيڪو مڪتبة المدينه تان آسان ڪري شائع ڪيو ويو آهي، ان جو مطالعو به مُفيد رهندو۔ اِنۡ شَآءَاللّٰه عَزَّوَجَلَّ . اچو ته! هاڻي ٻُڌون ته ٻارن کي خوش ڪرڻ جي ڇا فضيلت آهي، جيئن ته:

حَضۡرَتِ سَيدَتُنا عائِشه صِديقه رَضِیَ اللهُ تَعَالٰی عَنْهَا روايت ڪن ٿيون ته خَاتَمُ الۡمُرۡسَلِين، رَحۡمَةُ لِّلۡعٰلَمِين صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: ”بيشڪ جنَّت ۾ هڪ گھر آهي جنهن کي ” اَلۡفَرۡحۡ “ چيو ويندو آهي۔ ان ۾ اهي ئي ماڻهو داخل ٿيندا جيڪي ٻارن کي خُوش ڪندا آهن۔“ (جامع صغیر ،الحدیث۲۳۲۱، ص۱۴۰)

اعَلِيٰ حَضۡرت، اِمَام اَہلسُنّت مولانا شاھ اِمَام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلِيه هڪ سُوال جي جَواب ۾ پيءُ تي اَولاد جا حقَ بيان ڪندي فرمائن ٿا ته پيءُ

”خدا جي ان اَمانتن سان شَفقَت ۽ مَحَبَّت جو سُلوڪ رکي، انهن سان پِيار ڪري، بَدن سان چنبڙائي، ڪلهن تي ويهاري، انهن سان کِلڻ، کيڏڻ ۽ وندر جون ڳالھيون ڪري، انهن جي دِلجُوئي، دِلداري، لحاظ ۽ حفَاظَت ہر وَقت ايستائين جو نماز ۽ خُطبي ۾ به مَلحُوظ (ڌيان) رکي۔ نَئون ميوو، نَئون ثمر پهريان انهن (ئي) کي ڏئي جو اهي به تازا ميوا آهن، نئين کي نئون مُناسب آهي ۔ ڪڏهن ڪڏهن وَس آهر انهن کي شيرني (مٺائي) وغيرہ کارائڻ، پارائڻ، کيڏڻ جون سُٺيون شيون (جيڪي) شَرعًا جائز آهن، ڏيندو رهي۔ وندرائڻ جي لاءِ ڪوڙو واعدو نه ڪري، بلڪه ٻار سان به واعدو اهو ئي جائز آهي جنهن کي پُوري ڪرڻ جو قَصد (اِرادو) رکندو ہجي۔ ڪجھ ٻار ہجن ته جيڪا شيءِ ڏي سڀني کي برابر ۽ هڪجھڙي ڏي، هڪڙي کي ٻئي تي بي فضيلتِ ديني (يعني دِيني فضيلت جي بغير) ترجيح نه ڏي۔‘‘

(فتاویٰ ر ضویہ، ٢٤ / ۴۵۳)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اچو ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي حَسنَين ڪَريمَين رَضِیَ اللهُ تَعَالٰی عَنْهَا سان مَحَبَّت جو هڪ ٻيو پہلو سَماعَت فرمايون، جيئن ته

حَضۡرَتِ سَيِّدُنَا اِبنِ عَبَّاسَ رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کان رِوايت آهي ته ٻنهي جہانن جا تاجۡوَر، سلطانِ بَحروبَر صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم حَضۡرَتِ سَيِّدُنَا اِمَام حَسَن ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کي ڪلِماتِ تَعوُّذ سان گڏ دَم فرمائي رهيا هئا۔ حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: توهان جا جَدّ اَمۡجد يعني حَضۡرَتِ

ابرَاہِيم عليه الصلات والسلام به پنهنجي صاحِبزادن حَضۡرَتِ اسماعيل ۽ حَضۡرَتِ اسحاق عَلِيهِمُ السَّلَام کي اِنهي ڪلِمات سان گڏ دَم فرمائيندا هئا: اَعُوْذُ بِکَلِمَاتِ اللّٰہِ التَّامَّۃِ مِنْ کُلِّ شَیْطٰنٍ وَّھَامَّۃٍ وَّمِنْ کُلِّ عَیْنٍ لَّامَّۃٍ يعني آئون اللہ عَزَّوَجَلَّ جي ڪامل ڪلِمن جي ذريعي ہر شيطان ۽ زہريلي جانور ۽ ہر نَظرِ بَد کان پناھ گُھران ٿو. (بخاری،کتاب احادیث الانیباء،۲/۴۲۹، حدیث:۳۳۷۱)

مُفَسِّرِ شَہِير،حڪيم الاُمَّت مُفتي احمد يارخان رَحْمَةُ اللهِ تَعَالٰی عَلِيه ان حديثِ پاڪ جي شَرح ۾ فرمائن ٿا: ” ڪلِمَاتُ اللہ (اللہ جا ڪَلِمه) مان مُراد سڀئي اسماءِ اِلٰہي (اللہ جا نالا) آهن، اهي ہرنَقص (ڪمي) ۽ خرابي کان پاڪ آهن، ان لاءِ انهن کي ” تَامّات“ چيو ويندو آهي، جهڙي طرح اللہ (عَزَّوَجَلَّ) جي پناھ گھُرڻ ضروري آهي، اهڙي طرح ئي ان جي نالن جي پناھ به ضروري آهي۔“ مزيد فرمائن ٿا: جنّ ۽ نَظرِ بَد سان به انسان بيمار ٿي ويندو آهي، جنّ جو اثَر قرآنِ ڪريم مان ثابت آهي۔

(مراٰة المناجیح،ج۲،ص ۴۰۹ ملتقطاً)

مٺا مٺا اسلامي ڀائرو! بَيان ڪيل حديثِ پاڪ مان دَم وغيرہ جي جواز جو ثبوت ملي ٿو ته اسان جي پياري آقا، مڪي مدني مُصطفٰي صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم پنهنجي پياري نَواسن (ڏوهٽن) کي دَم فرمائيندا هئا۔ قرآنِ مجيد جي آياتِ مُبارَڪه جي ذريعي بيمارن تي پڙهي ڪري دم ڪرڻ جي حوالي سان ڪيئي روايتون مَوجُود آهن: جيئن ته، اُمُّ الۡمُؤمنين حَضۡرَتِ سَيدَتُنا عائِشه صِدّيقه رَضِیَ اللهُ تَعَالٰی عَنْهَا بيان فرمائن ٿيون: جڏهن رَسُولُ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي اَہل (يعني گھروارن) مان ڪو بيمار ٿيندو هو ته سرڪار صَلَّى

اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ان تي قُلْ اَعُوْذُ بِرَبِّ الْفَلَقِ ٭ ۽ قُلْ اَعُوْذُ بِرَبِّ النَّاس ٭ پڙهي ڪري دَم فرمائيندا هئا۔ (مسلم،کتاب السلام،باب رقية المریض بالمعوذات والنفث، ص ۱۲۰۵، حدیث:۲۱۹۲)

اَعۡليٰ حَضۡرَتِ، اِمَام اَہلسنَّت، مولانا شاھ اِمَام احمدرضا خان رَحْمَةُ اللهِ تَعَالٰی عَلِيه فرمائن ٿا: جائز تعويذ جيڪو قرآنِ ڪريم يا اَسماءِ اِلٰہي (يعني اللہ عَزَّوَجَلَّ جا نالا) يا ٻين اَذڪار ۽ دُعائن مان ہجي، ان ۾ اَصۡلاً (بلڪل ڪو) حَرَج ناهي بلڪه مُسۡتَحَب آهي۔ رَسُوۡلُ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن اهڙي ئي مَقام ۾ فرمايو ته ” مَنِ اسْتَطَاعَ مِنْکُم اَنْ یَّنْفَعَ اَخَاہُ فَلْیَنْفَعْہٗ يعني توهان مان جيڪو شخص پنهنجي مُسلمان ڀاءُ کي نَفعو پہچائي سگھي (ته ان کي نَفعو) پہچائي۔“ (مسلم،کتاب السلام،باب استحباب رقیة من العين۔۔۔الخ،ص۱۲۰۸،فتاویٰ افریقہ ،ص ۱۶۸) البته غير شَرعي تعويذات ۽ غير شرعي ڪَلِمات وارو دَم ناجائز آهن جيئن ته اَعَلِيٰ حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه فرمائن ٿا ”اهي مَقصدَ جنهن جي لاءِ تعويذ يا عمل ڪيو وڃي، جيڪڏهن خِلافِ شَرع هجن، ناجائز ٿي ويندا، جيئن عورتون تَسخِيرِ شوهر (مڙس کي مَغلُوب ڪرڻ) جي لاءِ تعويذ ڪرائينديون آهن، هي حڪمِ شرع جو عڪس آهي (خِلافِ شريعت آهي) ، اهڙي طرح تَفريق ۽ عَداوَت (يعني پاڻ ۾ جُدائي وجھڻ ۽ دُشمني پيدا ڪرڻ) جي عَمل ۽ تعويذ جيڪي مَحارِم (رشتيدارن) ۾ ڪيا وڃن، مثلاً ڀاءُ کي ڀاءُ کان جُدا ڪرڻ، هي قَطعِ رَحم (رشتو ٽوڙڻ) آهي ۽ قَطعِ رحم حرام (آهي) ، اهڙي طرح زال ۽ مڙس ۾ نِفاق وجھڻ (به حرام آهي) ۔ (فتاوی رضویہ ،ج۲۴،ص۱۹۶)

اَلحَمدُلِلّٰه عَزَّوَجَلَّ في زمانه شيخِ طريقت، اميرِ اہلسنّت، باني دعوتِ اِسلامي حَضۡرَتِ علامه مولانا ابُو بلال محمد الياس عطّار قادِري رَضَوي دَامَـتْ

بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه مُسلمانن جي خَيرخُواہي جي جَذبي جي تحت جتي ٻيا شعبا قائم فرمايا آهن، اُتي مَجلِس مڪتُوبات و تَعوِيذاتِ عطاريه جو شعبو به قائم فرمايو آهي، جنهن جي تحت نه صرف مڪتوبات جي ذريعي پريشان حالن جي غمخواري ڪئي ويندي آهي، بلڪه اَميرِ اہلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي عطا ڪيل تعويذات ۽ اوراد ۽ وظائف جي ذريعي مُختَلف پريشانين جو حل ۽ فِي سَبِيۡلِ اللّٰه بيمارن جو علاج ڪرڻ جي ڪوشش ڪئي ويندي آهي، ہر مهيني تقريبًا 1,25,000 (هڪ لَک پنجويھ ہزار) مريضن کي 4 لَک کان وڌڪ تعويذات ۽ اورادِ عطاريه ڏنا ويندا آهن، هي تعويذات توهان ’’ تعويذاتِ عطاريه ‘‘ جي بستي تان فِي سَبِيۡلِ اللّٰه با آساني حاصِل ڪري سگھو ٿا۔

اللہ کرم ایسا کرے تجھ پہ جہاں میں

اے دعوتِ اسلامی تیری دُھوم مچی ہو!

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! مرحبا حَسنَين ڪَريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا سان رحمتِ ڪونين صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم ڪيتري قدر محبت ۽ شفقت ڪندا هئا، مزيد عشقِ حَسنَين ڪَريمَين وڌائڻ واري ڳالھ سماعت ڪريو ۽ ايمان تازو ڪريو، جيئن ته عاشقِ صحابه ۽ اہلبيت اعَلِيٰ حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه لکن ٿا:

معدوم نہ تھا سایہء شاہِ ثقلین

تمثیل نے اُس سایہ کے دو حصے کیے

اُس نور کی جلوہ گاہ تھی ذاتِ حسنین

آدھے سےحسن بنے ہیں آدھے سے حسین

رُباعي جي وضاحت: ايئن ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جو مبارڪ پاڇو سج جي تيز روشني ۽ چنڊ جي روشني ۾ زمين تي نه پَوندو هو، پر جڏهن سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي فيضان جو پاڇو حَسنَين ڪَريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا تي پَيو ته سيني تائين اِمَام حسن مجتبٰي رَضِیَ اللهُ تَعَالٰی عَنْهُ، حُضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان مشابه ٿي ويا ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُ، سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان سيني کان پيرن تائين مشابه ٿي ويا۔ قصيدہء نور ۾ سَيِّدي اَعَلِيٰ حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه لکن ٿا:

ایک سینہ تک مشابہ اِک وہاں سے پاؤں تک

صاف شکلِ پاک ہے دونوں کے ملنے سے عِیاں

حُسنِ سبطین ان کے جاموں میں ہے نِیما نورکا

خطِ توام میں لکھا ہے یہ دو وَرقہ نورکا

ياد رکو! سَيّدي اعَلِيٰ حَضۡرَتِ رَحْمَةُ اللهِ تَعَالٰی عَلِيه جي شاعري قرآن ۽ حديث جي ترجماني ۽ بزرگن جي اقوال ۽ احوال جي مطابق آهي، اعَلِيٰ حَضۡرَتِ ، حَسنَين ڪَريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان مشابہت کي ايئن ئي ناهي لکي ڇڏيو بلڪه

ترمذي شريف ۾ آهي: سَيِّد الۡاَولِيَاء، مَوليٰ مُشڪل ڪُشاء، شيرِ خُدا حَضۡرَتِ سَيِّدُنَا عَلِي المرتضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم فرمائن ٿا: اِمَام حَسن رَضِیَ اللهُ تَعَالٰی عَنْهُ سِيني ۽ مٿي جي درميان (وچ تائين) محبوبِ رحمٰن صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان تمام گھڻي مشابہت رکندا هئا ۽ اِمَام حُسَيۡن رَضِیَ اللهُ تَعَالٰی عَنْهُ ان کان هيٺ واري حصي ۾ رَسُوۡلُ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان تمام گھڻي مشابهت رکندا هئا۔

مُفَسِّرِ شَہِير،حڪيم الاُمَّت مُفتي احمد يارخان رَحْمَةُ اللهِ تَعَالٰی عَلِيه ان حديثِ پاڪ جي شَرح ۾ فرمائن ٿا: خيال رهي ته حَضرَتِ فاطمه زہرا رَضِیَ اللهُ تَعَالٰی عَنْهَا از سر تا قدم (يعني مٿي کان پيرن تائين) بلڪل هَم شَڪلِ مصطفيٰ هيون۔ ۽ خاتونِ جنت رَضِیَ اللهُ تَعَالٰی عَنْهَا جي شهزادن يعني حسنين ڪريمين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا ۾ اها مشابہت تقسيم ڪئي وئي هئي، حَضرَتِ اِمَام حسين جي پِني قدم شريف تائين ۽ پيرن جي کُڙي بلڪل حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي مشابه هئي، حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان قدرتي مشابہت به اللہ تعاليٰ جي نعمت آهي جو پنهنجي ڪنهن عمل کي حضور جي مشابه ڪري ڇڏي ته ان جي بخشش ٿي ويندي آهي، ته جنهن کي خدا تعاليٰ پنهنجي محبوب جي مشابه ڪري، ان جي محبوبيت جو ڇا عالَم هوندو۔ . (مرآۃ جلد8ص480)

صَــلُّـوا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! جڏهن صَحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم کي پنهنجي اَہۡلِ بيت ۽ پيارن نَواسن (ڏوهٽن) سان بي انتِہا مَحَبَّت ڪندي ڏسندا هئا ته حضور صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان نسبت جي سبب اهي حضرات به انهن سان مَحَبَّت ۽ شَفۡقَت سان پيش ايندا هئا ۽ سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي وصالِ ظاہري کان پوءِ به سرڪار صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جي اہلِ بيتِ اطہار ۽ بالۡخُصوص حَسَنَينِ ڪرِيمَين جو بيحد خَيال رکن دا هئا ، جيئن ته

حَضرَتِ سَيِّدُنَا اَبُوبَڪرصِدّيق رَضِیَ اللهُ تَعَالٰی عَنْهُ جڏهن اَمِيرُالۡمُؤمِنِين، خَلِيفَةُ الۡمُسۡلِمِين مُنتَخَب ٿيا ته رسولِ ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم سان تَعَلُّق جي

سبب پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهُ اہلِ بيتِ اطہار جو تمام گھڻو خَيال رکندا هئا ۽ اَہۡلِ بيتِ اَطہار جي باري ۾ فرمائيندا هئا ته ’’نبي ڪريم، رءُوفٌ رَّحيم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جا اَقارِب (يعني رشتيدار) مونکي پنهنجي رشتيدارن کان وڌيڪ عَزِيز آهن۔‘‘ (بخاری، کتاب المغازی، باب حدیث بنی نضیر، الحدیث: ۴۰۳۶، ج۳، ص۲۹)

باغ جنّت کے ہیں بہرِ مدح خوانِ اہلبیت تم کو مُژدہ نار کا اے دُشمنانِ اہلبیت

(ذوقِ نعت)

حَضرَتِ سَيِّدُنَا اِمَام حُسَين رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا: آئون هڪ ڏينهن اَمِيرُالْمُومنين حَضرَتِ سَيِّدُنَا عمر فارُوقِ اعظم رَضِیَ اللهُ تَعَالٰی عَنْهُ جي گھر ويس، پر پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهُ ، حَضرَتِ اَمِيرِ مُعاوِيه رَضِیَ اللهُ تَعَالٰی عَنْهُ سان گڏ عَلِيحدگي ۾ گفتگو فرمائي رهيا هئا ۽ سندن رَضِیَ اللهُ تَعَالٰی عَنْهُ جا پُٽ حَضرَتِ عبدُ اللہ رَضِیَ اللهُ تَعَالٰی عَنْهُ دروازي تي بيهي اِنتظار ڪري رهيا هئا۔ ڪجھ دير انتظار کان پوءِ اهي واپس موٽڻ لڳا ته ان سان گڏ آئون به واپس موٽي آيس۔ بعد ۾ اَمِيرُالۡمُومنين حَضرَتِ سَيِّدُنَا عمر فارُوقِ اعظم رَضِیَ اللهُ تَعَالٰی عَنْهُ سان منهنجي مُلاقات ٿي ته مون عرض ڪيو: ” يا اَمِيرُالمُومنين! آئون توهان وٽ آيو هئس، پر توهان حَضرَتِ اَمِيرِ مُعاويه رَضِیَ اللهُ تَعَالٰی عَنْهُ سان گفتگو ۾ مَصۡروف هئا۔ توهان جو پُٽ عبدُ اللہ رَضِیَ اللهُ تَعَالٰی عَنْهُ به ٻاہر بيهي انتظار ڪري رهيو هو (مون سوچيو ته جڏهن پُٽ کي اندر اچڻ جي اجازت ناهي ته مونکي ڪيئن اجازت ٿي سگھي ٿي) تنهنڪري آئون به انهن سان ئي گڏجي واپس هليو ويس۔“ ته فارُوقِ اَعۡظَم رَضِیَ اللهُ تَعَالٰی عَنْهُ جن فرمايو: اي منهنجا پُٽ حُسَيۡن! منهنجي اَولاد کان وڌيڪ، توهان

ان ڳالھ جا حقدار آهيو ته توهان اَندر اچي وڃو ها ۽ اسان جي مٿي تي هي جيڪي وارَ آهن، اللہ عَزَّوَجَلَّ کان پوءِ ڪنهن ظاهر ڪيا آهن، توهان سادات ِڪرام ئي ته ظاهر ڪيا آهن۔“ (تاریخِ ابن عساکر،ج ۱۴،ص۱۷۵)

حَضرَتِ سَيِّدُنَا اَصۡبَغۡ بن نُباته رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا: هڪ ڀيري حَضرَتِ سَيِّدُنَا اِمَام حسن مُجتبيٰ رَضِیَ اللهُ تَعَالٰی عَنْهُ بيمار ٿي پيا ته حَضرَتِ سَيِّدُنَا عَلِي الۡمُرتَضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم ان جي عِيادَت جي لاءِ تشريف وٺي ويا، اسان به انهن سان عِيادَت جي لاءِ حاضِر ٿياسين۔ حَضرَتِ سَيِّدُنَا عَلِي المُرتَضٰي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم خَيرِيت پڇندي فرمايو: اي نواسهء رسول! هاڻي طَبِيعَت ڪيئن آهي؟ عَرض ڪيو: اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ بہتر آهيان، پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمايو: جيڪڏهن اللہ عَزَّوَجَلَّ چاہيو ته بہتر ئي رہندو، پوءِ حَضرَتِ سَيِّدُنَا اِمَام حسن رَضِیَ اللهُ تَعَالٰی عَنْهُ جن عرض ڪيو: مون کي سہارو ڏئي ڪري ويهاريو، حَضرَتِ سَيِّدُنَا عَلِي رَضِیَ اللهُ تَعَالٰی عَنْهُ جن انهن کي پنهنجي سيني سان ٽيڪ لڳائي ڪري ويهاري ڇڏيو، پوءِ حَضرَتِ سَيِّدُنَا اِمَام حسن رَضِیَ اللهُ تَعَالٰی عَنْهُ جن فرمايو: هڪ ڏينهن مون کي نانا جان، رحمتِ عالميان صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن فرمايو هو: اي منهنجا پُٽ! جنّت ۾ هڪ وڻ آهي جنهن کي شَجَرَةُ البَلويٰ چيو ويندو آهي، آزمائش ۾ مبتلا ماڻهن کي قيامت جي ڏينهن ان وڻ جي ويجھو جَمع ڪيو ويندو، جڏهن ته ان وقت نه ميزان رکيو ويو هوندو نه ئي اعمال نامه کوليا ويا هوندا، انهن کي پورو پورو اجر عطا ڪيو ويندو۔ پوءِ سرڪارِ دوعالم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن اها آيتِ مبُارَڪه تِلاوَت فرمائي،

اِنَّمَا یُوَفَّی الصّٰبِرُوۡنَ اَجۡرَہُمۡ بِغَیۡرِ حِسَابٍ ﴿۱۰﴾ (تَرجَموڪنزالايمان: صبر وارن کي سندن ثواب ڀرپور ڏنو ويندوبغير حساب جي ۔) (پ۲۳،الزمر،آيت ۱۰) (کتاب الدعاء للطبراني،ص۳۴۷)

مٺا مٺا اسلامي ڀائرو! ان واقِعي مان جتي حَضرَتِ سَيِّدُنَا عَلِي كَرَّمَ اللّٰهُ تَعَالٰي وَجۡهَهُ الۡكَرِيۡم جي پنهنجي شہزادي اِمَام حَسَن رَضِیَ اللهُ تَعَالٰی عَنْهُ سان مَحَبَّت جوعِلم ٿيو، اُتي اِمَام حَسَن رَضِیَ اللهُ تَعَالٰی عَنْهُ جي بيان ڪيل فرمانِ مُصطفيٰ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم مان هي به معلُوم ٿيو ته پريشانين، مُصيبتن ۽آزمائشن تي صَبۡر ڪرڻ وارن کي قِيامَت جي ڏينهن ان جي صَبۡر جو پُورو پُورو اَجۡر ڏنو ويندو۔ يادرکو! اللہ عَزَّوَجَلَّ جي ہر ڪم ۾ ہزارين حِڪمَتون پَوشِيدہ ہونديون آهن، جنهن جو اسان کي عِلم ناهي ہوندو۔ تنهنڪري ہر هڪ جي سَامهون پنهنجي پريشاني، غريبي ۽ مُفلِسي جو روئڻ، پنهنجا ڏُک تڪليفون ٻُڌائڻ ۽ تَنگدَستي جي سَبَب مَعَاذَاللّٰه عَزَّوَجَلَّ رَبّ تعاليٰ جي ذات تي فضول اِعتِراضَ ڪري پنهنجي زَبان مان ڪفرِيات بَڪڻ جي بَجاءِ، اِنهن آزمائشن ۽ تَڪليفن کي منهن ڏيندي صَبر ۽ تَحَمُّل کان ڪم وٺڻ گھرجي، ڇو جو هي مُصِيبتون ۽ بَلائون گُناہن جو ڪَفّارو ۽ دَرَجات ۾ بَلندي جو سبب ٿينديون آهن ۔ اللہ عَزَّوَجَلَّ جي مَحۡبُوب ، داناءِ غُيوب صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: جڏهن قِيامت جي ڏينهن اَہلِ بَلا (يعني بيمارين ۽ آفت ۾ مبتلا رهڻ وارن) کي ثَواب عَطا ڪيو ويندو، ته عافِيت وارا تمنّا ڪندا ته ڪاش! دُنيا ۾اسان جون کَلون قَينچين سان ڪٽيون وڃن ها ۔ (سُنَنُ التِّرْمِذِی ج٤ ص١٨٠حدیث ٢٤١٠ دارا لفکر بیروت) مُفَسِّرِ شَہِير،حڪيم الاُمَّت مُفتي احمد يارخان رَحْمَةُ اللهِ تَعَالٰی عَلِيه هن حديثِ پاڪ جي الفاظن ” ڪاش! دُنيا ۾ اسان جون کَلون قَينچين سان ڪٽيون وڃن ها “ جي تحت فرمائن ٿا: ”يعني تمنّا ۽ آرزُو ڪندا ته اسان تي دُنيا ۾ اهڙيون بيماريون

اچن ها، ته جيئن اسان کي به اهو ثواب اڄ ملي ها جيڪو ٻين بيمارين ۽ آفت ۾ مبتلا رهڻ وارن کي ملي رهيو آهي۔“ (مراٰۃ ،ج۲ص،۴۲۴)

اڪثر ڪري مُسلمان پنهنجون پريشانيون ۽ ڪافِرن جي عيش ۽ عشرت سان ڀرپُور زِندگي کي ڏسي ڪري به وَسوَسن جو شڪار ٿي ويندا آهن ۽ انهن جي ذِہن ۾ طَرح طَرح جا سُوالَ پيدا ٿيندا آهن، جڏهن ته ان ۾ به اللہ رَبُّ الۡعٰلَمِين جَلَّ جَلالُهٗ جي تمام وڏي حِڪمَت پوشِيدہ آهي۔ جيئن ته،

حَضرَتِ سَيِّدُنَا اِبۡنِ عَبَّاسَ رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جن فرمايو ته هڪ نبي عَلِيـهِ الـسَّـلَام پنهنجي پَروَردگار عَزَّوَجَلَّ جي درٻار ۾ عرض ڪيو: اي منهنجا رَبّ عَزَّوَجَلَّ! مؤمن ٻانهو تنهنجي اِطاعت ڪندو آهي ۽ تنهنجي مَعصِيّت (نافرماني) کان بچندو آهي (پر) تُون (ربّ عَزَّوَجَلَّ) ان جي لاءِ دُنيا تنگ فرمائي ان کي آزمائشن ۾ مبتلا ڪري ڇڏيندو آهين ۽ ڪافر تنهنجي اِطاعت ناهي ڪندو بلڪه توتي ۽ تنهنجي مَعۡصِيّت (نافرماني) تي جُراءت ڪندو آهي، پر تُون (ربّ عَزَّوَجَلَّ) ان کي مُصيبت کان پَري رکندو آهين ۽ ان جي لاءِ دُنيا ڪشاده ڪري ڇڏيندو آهين، (آخر ان ۾ ڇا حڪمت آهي؟) اللہ عَزَّوَجَلَّ ان (نبي جي) طرف وَحي فرمائي: ٻانها به منهنجا آهن ۽ مُصيبَت به منهنجي اختيار ۾ آهي ۽ سڀ منهنجي حمد سان گڏ منهنجي تسبيح ڪندا آهن، مُؤمِن جي ذِمي گُناھ ٿيندا آهن ته آئون ان کان دُنيا کي پَري ڪري ان کي آزمائش ۾ مبتلا ڪندو آهيان ته هي (آزمائش ۽ مُصيبت) ان جي گُناہن جو ڪَفّارو بڻجي ويندي آهي، ايستائين جو اهو مون سان مُلاقات ڪندو ته آئون ان کي نيڪيون بدلي ۾ ڏيندس ۽ ڪافر جون (دنياوي اعتبار سان) ڪجھ نيڪيون ہونديون

آهن، ته آئون ان جي لاءِ رِزق ڪشادو ڪندو آهيان ۽ مُصيبت کي اُن کان پَري رکندو آہيان ته ايئن ان جي نيڪين جو بدلو دُنيا ۾ ئي ڏئي ڇڏيندو آہيان، ايستائين جو جڏهن اهو مون سان مُلاقات ڪندو ته آءُ ان جي گُناہن جي ان کي سَزا ڏيندس۔ ([1])

بہر حال اسان کي مُسلمان هئڻ جي حيثيت سان اللہ عَزَّوَجَلَّ جي ہر ڪم کي حڪمت تي مُشۡتَمِل سمجھڻ گھرجي ۽ مُصِيبت تي صَبۡر جو مُظاہَرو ڪندي اَجۡر ۽ ثواب جو خُوب خُوب ذَخيرو ڪرڻ گھرجي۔ اللہ عَزَّوَجَلَّ اسان کي بي صَبري ۽ ناشُڪرِي کان بچائي ۽ صَبر ۽ شُڪر جو عادي بڻائي۔

حَضرَتِ سَيِّدُنَا ابُوهرَيرہ رَضِیَ اللهُ تَعَالٰی عَنْهُ کان مَروي آهي ته رَسُولُ اللہ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جن فرمايو آهي ته ڪنهن مسلمان جي لاءِ هي ڳالھ جائز ناهي ته اهو پنهنجي ڀاءُ سان ٽي (3) ڏينهن رات کان وڌيڪ قَطع تَعَلُّق ڪري۔ انهن ۾ جيڪو ڳالھ ٻولھ ڪرڻ ۾ اڳرائي ڪندو، اهو جنت جي طرف وڃڻ ۾ به سبقت ڪندو۔ حَضرَتِ ابُو ہُرَيرہ رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا ته مون تائين اها ڳالھ پہتي آهي ته حَضراتِ حَسَنَينِ ڪرِيمَين جي دَرمِيان ڪا معمولي اڻبڻت ٿي وئي آهي۔ آئون اِمَام حُسَين رَضِیَ اللهُ تَعَالٰی عَنْهُ جي خِدمَت ۾ حاضِر ٿيس ۽ عَرۡض

ڪيو: ماڻهو توهان جي اِقۡتِداء ڪندا آهن ۽ توهان حَضرات هڪ ٻئي کان ناراض آهيو ۽ توهان پاڻ ۾ قَطع تَعلُّقي ڪري ويٺا آهيو۔ توهان هاڻي اِمَام حَسَن رَضِیَ اللهُ تَعَالٰی عَنْهُ وٽ وڃو ۽ انهن کي راضي ڪريو ڇو جو توهان ان کان ننڍا آهيو، اِمَام حُسين رَضِیَ اللهُ تَعَالٰی عَنْهُ جن فرمايو: جيڪڏهن مون نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم کي هي فرمائيندي نه ٻُڌو هجي ها ته جڏهن ٻن ماڻهن جي درميان قَطع تَعَلُّقي ٿي وڃي، ته انهن ۾ جيڪو ڳالھ ٻولھ ڪرڻ ۾ اڳرائي ڪندو اهو جنَّت ۾ پهريان ويندو ، آئون مُلاقات ڪرڻ ۾ ضَرُور اڳرائي ڪريان ها، پر آئون ان ڳالھ کي پسند نه ٿو ڪريان ته آئون انهن کان پهريان جنَّت ۾ هليو وڃان۔

حَضرَتِ ابُوهرَيرہ رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا: ان کان بعد ۾ حَضرَتِ سَيِّدُنَا اِمَام حسن رَضِیَ اللهُ تَعَالٰی عَنْهُ جي بارگاھ ۾ حاضر ٿيس ۽ انهن کي سڄو واقعو ٻُڌايو: اِمَام حسن رَضِیَ اللهُ تَعَالٰی عَنْهُ جن فرمايو ته اِمَام حسين جيڪا ڳالھ چئي آهي اها درست آهي۔ پوءِ پاڻ رَضِیَ اللهُ تَعَالٰی عَنْهُ اِمَام حُسين رَضِیَ اللهُ تَعَالٰی عَنْهُ وٽ تَشريف وٺي ويا، انهن سان مُلاقات ڪئي ۽ اهڙي طرح ٻنهي ڀائرن جو پاڻ ۾ صُلح ٿي ويو۔ . مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته ڪنهن مسلمان جي لاءِ جائز ناهي ته هو پنهنجي مسلمان ڀاءُ کان ٽي ڏينهن رات کان وڌيڪ قَطع تَعلُّق ڪري (يعني رشتو ٽوڙي) ۔ پر افسوس! اڄڪلھ ٿوري ٿوري ڳالھ تي ماڻهو ناراض ٿي ويندا آهن ۽ هڪ ٻئي جي شڪل ڏسڻ به پسند نه ڪندا آهن، معمولي رنجش تي خاندان جُدا ٿي ويندا آهن، اڪثر ڪري خوني رِشتا به قتل ۽

غارت گِري تي لهي ايندا آهن۔ هي مدني ماحول کان دُوري ۽ علمِ دِين جي ڪمي جي سبب ٿيندو آهي، تنهنڪري اسان کي گھرجي ته مدني قافلن ۾ سفرڪري، ہفتيوار سُنّتن ڀري اجتماع ۽ مدني مذاڪري ۾ شرڪت ڪري علمِ دين حاصل ڪريون ته جيئن جہالت جي سبب جيڪي گناھ ٿيندا آهن، اُن کان بچي سگھون۔

فرمانِ مُصطَفيٰ صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم: سومر ۽ خميس تي اللہ تَعاليٰ جي بارگاھ ۾ ماڻهن جا اعمال پيش ٿيندا آهن، ته اللہ عَزَّوَجَلَّ پاڻ ۾ عَداوت (دشمني) رکڻ وارن ۽ قطعِ رِحمي ڪرڻ وارن کان عِلاوه سڀني جي مَغۡفِرَت فرمائي ڇڏيندو آهي۔ (اَلْمُعْجَمُ الْکبِیر لِلطّبَرانی ج،١ص١٦٧،حدیث:٤٠٩)

حَضرَتِ سَيِّدُنَا اَعۡمش رَحْمَةُ اللهِ تَعَالٰی عَلِيه کان مَنقُول آهي، حَضرَتِ سَيِّدُنَا عبدُالله ابنِ مَسۡعُود رَضِیَ اللهُ تَعَالٰی عَنْهُ هڪ ڀيري صُبح جي وَقت مَجلس ۾ تشريف فرما هئا، اُنهن فرمايو: آئون قاطِع رِحم (يعني رشتي ٽوڙڻ واري) کي اللہ عَزَّوَجَلَّ جو قسم ڏيان ٿو ته اهو هتان کان اُٿي وڃي ته جيئن اسان اللہ عَزَّوَجَلَّ کان مغفرت جي دُعا ڪريون، ڇو جو قاطِع رِحم(يعني رِشتي ٽوڙڻ واري) تي آسمان جا دروازا بند رہندا آهن۔ (يعني جيڪڏهن اهو هتي مَوۡجُود رهندو ته رحمت نازل نه ٿيندي ۽ اسان جي دُعا قَبول نه ٿيندي۔) (اَلْمُعْجَمُ الْکبِیر ج ٩ ص١٥٨ رقم٨٧٩٣)

صَــلُّـوا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! نيڪيون ڪرڻ، گُناہن کان بچڻ ۽ نيڪي ج ي دعوت کي عام ڪرڻ جي لاءِ ذيلي حلقي جي ٻارنهن 12 مدني ڪمن ۾ اڳتي وڌي حِصّو وٺو۔ ذيلي حلقي جي ٻارنہن مدني ڪمن مان هڪ مدني ڪم پنهنجي عملن جو مُحاسبو ڪندي مَدني انعامات تي عمل ڪرڻ به آهي۔ اسان جا بزرگانِ دين رَحِمَهُمُ الـلّٰـهُ السَّلَام به نه صرف پاڻ فڪرِ آخرت ۾ پنهنجي عملن جو مُحاسبو ڪندا هئا بلڪه ماڻهن کي به ان جو ذِہن ڏيندا هئا، جيئن ته اَمِيرُ المُؤمِنين حَضرَتِ سَيِّدُنَا عُمر فارُوق رَضِیَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا:''اي انسانو! پنهنجي عملن جو حساب ڪري وٺو، ان کان پهريان جو قيامت اچي وڃي ۽ توهان کان انهن جو حساب ورتو وڃي۔ (حلیۃ الاولیاء ،ج۱،ص۵۶) شيخِ طريقت ،اميرِ اہلسُنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه هن پُرفتن دَؤر ۾ فڪرِ آخرت جو ذِہن بنائڻ، نيڪيون ڪرڻ ۽ گناہن کان بچڻ جي طريقن تي مُشتمل مَدني اِنعامات بَصُورتِ سُوالات عطا فرمايا آهن۔ اِسلامي ڀائرن جي لاءِ 72، اِسلامي ڀينرن جي لاءِ 63، اسڪولن، ڪاليجن ۽ جامِعات جي طَلباء جي لاءِ 92، طالبات جي لاءِ 83، ۽ مدرسةُ المدينه جي مدني مُنّن جي لاءِ 40 مَدَني اِنعامات آهن، اهڙي طرح خُصُوصي يعني گونگن ٻوڙن ۽ نابينا اِسلامي ڀائرن ۽ قيدين

جي لاءِ به مَدَني اِنعامات مُرتَّب فرمايا آهن۔ مَدَني اِنۡعامات جو رسالو مَڪتبة الۡمَدِينه جي ڪنهن به شاخ تان ہديةً حاصل ڪري سگھجي ٿو، ان جو بَغَور مُطالَعَو ڪرڻ کان پوءِ توهان ان نتيجي تي پہچندا ته هي دَرۡاَصۡل خُود اِحۡتِسابي جو هڪ جامع نِظام آهي جنهن کي اپنائڻ کان پوءِ نيڪ بنجڻ جي راھ ۾ اچڻ واريون رُڪاوٽون اللہ عَزَّوَجَلَّ جي فَضۡل ۽ ڪرم سان آہستي آہستي پَري ٿي وينديون آهن ۽ ان جي بَرَڪت سان پابندِ سُنّت بنجڻ، گُناہن کان نَفرت ڪرڻ ۽ ايمان جي حِفاظت جي لاءِ ڪُڙهڻ جو ذِہۡن بنجندو۔ اچو ته مدني انعامات جي رسالي جي هڪ مدني بہار مُلاحظه ڪريون۔ جيئن ته،

نيو ڪراچي جي هڪ اِسلامي ڀاءُ جو ڪجھ هن طرح بيان آهي: علائقي جي مسجد جا اِمَام صاحب جيڪي دعوتِ اِسلامي سان وابسته آهن، انهن انفرادي ڪوشش ڪندي منهنجي وڏي ڀاءُ کي مَدَني انعامات جو هڪ رسالو تحفي ۾ ڏنو، اهو گھر کڻي آيا ۽ پڙهيو ته حيران ٿي ويا ته ان مختصر رسالي ۾ هڪ مسلمان کي اِسلامي زندگي گذارڻ جا ايترا زبردست فارمولا ڏنا ويا آهن۔ مَدَني انعامات جو رسالو ملڻ جي برڪت سان اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ ان کي نماز جو جذبو مليو ۽ نمازِ باجماعت جي ادائيگي جي لاءِ مسجد ۾ حاضر ٿي ويا ۽ هاڻي پنج وقت جا نمازي بڻجي چڪا آهن، ڏاڙھي مبارڪ به سجائي ورتي آهي ۽ مدني انعامات جو رسالو به ڀريندا آهن۔

مدنی اِنعامات کے عامِل پہ ہر دم ہر گھڑی یا اِلٰہی! خوب برسا رحمتوں کی تُو جھڑی

مٺا مٺا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلِيهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان محبت ڪئي ان مون سان محبت ڪئي ۽ جنهن مون سان محبت ڪئي اهو جنّت ۾ مون سان گڏ هوندو . اچوته شيخِ طريقت اَميرِ اهلسُنت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي رسالي ”163 مَدَني گُل “ مان مُنڊي پائڻ جا ڪجھ اهم مَدَني گُل ٻُڌون ٿا.

* مرد کي سون جي منڊي پائڻ حرام آهي ـ * (نابالغ) ڇوڪَر کي سَون جا زيور پارائڻ حرام آهي ۽ جنهن پارايا اهو گنهگار ٿيندو. * لوهه جي منڊي جهنّمين جو زيور آهي (ترمذي ج 3 ص 305 حديث 1792) * مرد جي لاءِ اها ئي مُنڊي جائز آهي جيڪا هڪ نگيني واري هجي ۽ جيڪڏهن ان ۾ (هڪ کان وڌيڪ) گھڻا نگينا هجن ته پوءِ کڻي چاندي جي ئي هجي مَرد جي لاءِ ناجائز آهي (ردالمحتار ج 9 ص 597) * بغير نگيني واري منڊي پائڻ ناجائز آهي جو هي منڊي ناهي، ڇَلّو آهي * حُروفِ مُقَطّعَات واري منڊي پائڻ جائز آهي پر حُروفِ مُقطّعات واري منڊي بغير وضو پائڻ ۽ هٿ لڳائڻ يا مصافحو

ڪندي مهل هٿ ملائڻ واري جو منڊي کي بي وضو هٿ لڳائڻ جائز ناهي * اهڙي طرح مردن جي لاءِ هڪ کان وڌيڪ (جائز) مُنڊي پائڻ يا (هڪ کان وڌيڪ) ڇلّا پائڻ به ناجائز آهي. * چاندي جي هڪ منڊي هڪ نگ وار ي جيڪا وزن ۾ ساڍا چار ماشه (يعني 4 گرام 374 ملي گرام) کان گھٽ هجي پائڻ جائز آهي، ها تڪبُّر يا زنانو سينگار (ليڊيز اسٽائل جو فيشن) يا ٻي ڪنهن غرضِ مَذمُوم (يعني قابل مَذمت مقصد) نيّت ۾ هجي ته رڳو منڊي (ئي) ڇا بلڪه انهيءَ نيّت سان (ته) سٺا ڪپـڙا پائڻ به جائز ناهن ـ (فتاويٰ رضويه ج 22 ص 141) * عيدن تي منڊي پائڻ مُستَحب آهي ـ (بهار شريعت ج 1 ص 779، 780) * لوهه جي منڊي تي چاندي جو خول چاڙهيو جو لوهه بلڪل ڏسڻ ۾ نه اچي ته ان منڊي جو پائڻ (مرد ۽ عورت ڪنهن کي به) منع نه آهي (فتاوي عالمگيري ج 5 ص 335) * منَّت (يعني باس) جو دم ڪيل ڌاتو (Metal) جو ڪڙو به مرد کي پائڻ ناجائز ۽ گناهه آهي ايئن ئي مدينه منوّره زادَها اللهُ شَرَفاً وتَعۡظِيۡماً يا اجمير شريف يا ڪنهن درگاھ جا چاندي جا ڇلا يا ڪنهن ٻئي ڌات جا ڇلا ۽ اسٽيل جي منڊي به جائز ناهي * جيڪڏهن ڪنهن اسلامي ڀاءُ ڌاتو جو ڪڙو يا ڇلّو، ناجائز منڊي يا ڌاتو جي زنجير (Chain- Bracelet) پاتي آهي ته هاڻي جو هاڻي لاهي توبهه ڪري ڇڏيو ۽ آئنده نه پائڻ جو عهد ڪيو.
