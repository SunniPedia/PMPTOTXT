ان عَلَوي هڪ هزار درهم جي ٿيلهي موڪلي ڏنائين۔ ڪجھ دير کان پوءِ حضرت وَاقِدِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي هڪ دوست جي چٺي حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڏانهن آئي، ته ”رمضان شريف جي مهيني ۾ خرچ جي لاءِ مون کي هڪ هزار درهم جي ضرورت آهي۔“ حضرت واقِدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اها ئي ٿيلهي ان ڏانهن موڪلي ڏني۔ ٻئي ڏينهن اهو ئي علوي دوست جنهن کان حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه قرض ورتو هيو ۽ اهو ٻيو دوست جنهن حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان قرض ورتو هيو ٻئي حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي گھر آيا۔ علوي چوڻ لڳو، رمضان المبارڪ جو مهينو اچي رهيو آهي ۽ مون وٽ ان هڪ هزار درهمن کانسواءِ ٻيو ڪجھ نه هيو، پر جڏهن اوهان جي چٺي آئي ته مون هي هزار درهم اوهان کي موڪلي ڏنا ۽ پنهنجي ضرورت جي لاءِ پنهنجي هن دوست کي چٺي لکي ته مون کي هڪ هزار درهم قرض طور موڪلي ڏي۔ ان اها ئي ٿيلهي جيڪا مون اوهان کي موڪلي هئي، مون ڏانهن موڪلي ڏني ته خبر پئي ته اوهان مون کان قرض گھريو، وري مون پنهجي هن دوست کان قرض گهريو ۽ انهن وري اوهان کان گھريو ۽ جيڪا ٿيلهي مون اوهان کي موڪلي هئي اها توهان هن کي موڪلي ڇڏي ۽ ان اها ٿيلهي مون کي موڪلي ڏني۔ پوءِ هنن ٽنهي بزرگن اتفاق راءِ سان ان رقم جا ٽي حصا ڪري پاڻ ۾ ورهائي ڇڏيا ۔ انهيءَ رات حضرتِ سيّدُنا واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي خواب ۾ جناب رسالت مآب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جي زيارت ٿي ۽ فرمايائون ، سڀاڻي توهان کي گھڻو ڪجھ ملي ويندو۔ اهڙيءَ طرح ٻئي ڏينهن امير يَحۡيى برمڪي سيّدُنا واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي گھرائي پُڇيو، ”مون رات خواب ۾ توهان کي پريشان ڏٺو آهي ڇا ڳالهه آهي؟ حضرت سيّدُنا واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه سڄو قصو ٻُڌايو ته يَحۡيى برمڪي چيو،” مان هي نٿو چئي سگهان ته اوهان ٽنهي مان ڪير وڌيڪ سخي آهي؟ بيشڪ اوهان ٽئي سخي ۽ واجب الاحترام آهيو۔ پوءِ ان ٽيهه هزار درهم حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي ۽ ويهه ويهه هزار انهن ٻنهي کي ڏنا ۽ حضرت واقدي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي قاضي (Judge) به مقرر ڪري ڇڏيو۔ (حُجَّة الله عَلَي الْعٰلَمِين ، ص ۵۷۷ مُلَخَّصاً)

تلاوتِ قرآن جا فضائل

مٺا مٺا اسلامي ڀائرو! اچو! تلاوت قرآن جا ڪجهه فضائل ٻڌو ۽ جهومو، جيئن ته تلاوت ڪرڻ وارن جي تعريف ۽ توصيف قرآن ڪريم ۾ بيان ٿي آهي. (پ 1، البقرة، آيت 121) قرآن پاڪ سان محبت ڪرڻ (يعني ان جي تلاوت ۽ ان تي عمل ڪرڻ) اللہ پاڪ ۽ ان جي رسول صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان محبت جو ذريعو آهي. (معجم الڪبير 9/132،حديث 9487). قرآنِ پاڪ جي هڪ حرف پڙهڻ تي 10 نيڪين جو ثواب ملندو آهي. جنهن گهر ۾ قرآن پڙهيو ويندو آهي اهو پنهنجي رهڻ وارن

تي ڪشادو ٿي ويندو آهي، ان جي ڀلائي گهڻي ٿيندي آهي، ان ۾ ملائڪ حاضر ٿيندا آهن، ان گهر مان شيطان نڪري ويندا آهن. (احياء العلوم،1/826) قرآن ڪريم قيامت

جي ڏينهن پنهنجي پڙهڻ وارن جي شفاعت ڪرڻ جي لاءِ ايندو. (مسلم، ڪتاب فضائل القرآن،ص403 حديث 804) تلاوت قرآن ڪرڻ وارن کي اللہ پاڪ شڪر گزارن جي ثواب کان افضل عطا فرمائيندو. (ڪنزالعمال ج 1 ص273) اهلِ قرآن (يعني ان جي تلاوت ڪرڻ وارا ۽ ان جي احڪامن تي عمل ڪرڻ وارا) اللہ وارا ۽ ان جا خاص ٻانها آهن. (ابن ماجه ج 1 ص 140) تلاوت قرآن هن امت جي افضل عبادت آهي. (شعب الايمان للبيهقي ج 2 ص 354) جيڪو نماز ۾ بيهي تلاوت ڪري ان جي لاءِ هر حرف جي بدلي 100 نيڪيون آهن. جيڪو ويهي تلاوت ڪري ان جي لاءِ هر حرف جي بدلي 50 نيڪيون آهن. جيڪو نماز کان پوءِ با وضو تلاوت ڪري ان جي لاءِ 25 نيڪيون آهن. (احياء العلوم ج 1 ، ص366) تلاوت قرآن دلين جي صفائي جو ذريعو آهي. (شعب الايمان ، التاسع عشر من شعب الايمان .. الخ فصل في ادمان تلاوتہ،2/352 حديث 2014) اللہ ڪريم اسان کي تلاوت قرآن جو ذوق ۽ شوق نصيب فرمائي. اٰمِیْن بِجَاہِ النَّبِیِ الْاَمِیْن صَلَّی اللّٰہُ تَعَالٰی عَلَیْہِ واٰلِہٖ وَسَلَّمَ

روزي جا ٽي (3) دَرَجا

مٺا مٺا اسلامي ڀائرو! روزي جا ٽي درجا آهن: (1) عوام جو روزو (2) خواص جو روزو (3) اَخَصُّ الخَواص جو روزو۔ اچو! هاڻي هنن مان هر هڪ جي وضاحت مُلاحظه ڪريو:

(1) عوام جو روزو: روزي جي لغوي معنيٰ آهي: ”رڪجڻ“ شريعت جي اصطلاح ۾ صبح صادق کان وٺي سج لهڻ تائين ڄاڻي واڻي کائڻ پيئڻ ۽ جماع کان ”رُڪيل رهڻ“ کي روزو چئبو آهي ۽ اهو ئي عوام يعني عام ماڻهن جو روزو

آهي۔ (2) خواص جو روزو: کائڻ پيئڻ ۽ جماع کان رڪيل رهڻ سان گڏوگڏ جسم جي تمام عضون کي بُرائين کان”روڪڻ“ خواص يعني خاص ماڻهن جو روزو

آهي ۔ (3) اَخَصُّ الخَواص جو روزو: پنهنجو پاڻ کي تمام ڪمن کان ”روڪي“ ڪري صرف ۽ صرف اللہ ڪريم جي طرف متوجهه ٿيڻ، هي اخصُّ الۡخواص يعني خاص الخاص ماڻهن جو روزو آهي ۔ (الجوهرة النيرة ،ڪتاب اصوم، ص 175)فيضان رمضان،ص۹۰ملخصاً)

مٺا مٺا اسلامي ڀائرو! روزي جي حالت ۾ اسان کي جنهن ڳالھ جي گھڻي وڌيڪ ضرورت آهي اها هي آهي ته اسان کائڻ پيئڻ وغيره کان ”رُڪيل رهڻ“ سان گڏوگڏ پنهنجي بدن جي سمورن عضوئن (Body Parts) کي به روزي جو پابند بڻايون ۽ پاڻ کي هر قسم جي برائين کان بچايون۔ ڇوته جڏهن اسان رَمضانُ المبارَڪ ۾ روزو رکي ڏينهن جي وقت کائڻ پيئڻ ڇڏي ڏيندا آهيون جڏهن ته هي کائڻ پيئڻ ان کان اڳ، ڏينهن جي وقت به بلڪل جائز هيو، سو پاڻ ئي سوچيو ته جيڪــي شيون رمضان شريف کان پهريان حلال هيون، اهي به جڏهن هن مبارڪ مهيني جي مقدس ڏهاڙن ۾ منع ڪيون ويون. ته جيڪي شيون رمضان شريف کان پهريان به حرام هيون، مثال طور: ڪوڙ، غيبت، چُغلي، بدگماني، گارگند، فلمون ڊراما، گانا باجا، بدنگاهي، ڏاڙهي ڪوڙائڻ، يا هڪ مٺ کان گهٽائڻ، والدين کي ستائڻ، بنا شرعي اجازت جي ماڻهن جي دل ڏکائڻ وغيره اهي رمضان المبارڪ ۾ ڇو نه اڃان به وڌيڪ حرام ٿينديون؟ هاڻي غور ڪيو، جيڪو شخص حلال ۽ پاڪ کائڻ، پيئڻ ته ڇڏي ڏئي، پر حرام ڪم ۽ جهنم ۾ وٺي وڃڻ وارا ڪم بدستور (يعني اڳ جيان) جاري رکي۔ اهو ڪهڙي قسم جو روزيدار آهي؟ ياد رکو! نبي پاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ عبرت نشان آهي،”جيڪو بُري ڳالهه چوڻ ۽ ان تي عمل ڪرڻ نه ڇڏي ته ان جي بُکايل ۽ اُڃارو رهڻ جي اللہ پاڪ کي ڪابه ضرورت ڪانهي. (بُخاري، ڪتاب الصوم ،باب من لم يدع قول الزور۔۔الخ،۱/ ۶۲۸،حديث:۱۹۰۳)

ڪنن جو گناھ هي آهي ته انهن سان انهن ڳالهين کي ٻڌڻ جن کان اللہ ڪريم منع ڪيو آهي، لهٰذا هرگز هرگز ڪنن سان گانا باجا ۽ موسيقي نه ٻڌي، ڪوڙا لطيفا نه ٻڌي، ڪنهن جي غِيبَت، چُغلي ۽ ٻين جا عيب نه ٻُڌي، جڏهن ٻه ماڻهو لڪي ڳالھ ڪن ته ڪن لڳائي نه ٻڌي۔ ڪنن جو روزو رکي! اهو هيئن جو صرف ۽ صرف جائز ڳالهيون ٻڌي، مثال طور ڪنن سان تلاوت، حمد و نعت ۽ بزرگن جون منقبتون، سُنّتن ڀريا بيان ۽ مدني مذاڪرا ۽ سٺيون سٺيون ڳالهيون ٻڌو، اَذان ۽ اِقامت ٻُڌو ۽ ٻڌي ڪري جواب به ڏيو وغيره۔

زبان جو گناھ

زبان جو گناھ هي آهي ته هن سان ڳالهيون ڪرڻ جيڪي اللہ ڪريم جي ناراضي جو سبب ٿين، ان ڪري گار گند، ڪوڙ، غِيبَت، چُغلي وغيره کان زَبان کي ناپاڪ ٿيڻ کان بچايو۔ اي ڪاش! زَبان صِرف ۽ صِرف نيڪ ۽ جائز ڳالهين جي لاءِ ئي حَرَڪت ۾ اچي، مثال طور زَبان سان تلاوتِ قرآن ڪريو، ذِڪر و دُرُود جو وِرد ڪريو۔ حَمد ۽ نعت شريف پڙھو، دَرس ڏيو، سُنّتن ڀريو بيان ڪريو، نيڪي جي دعوت ڏيو وغيره۔

هٿ جو گناھ

هٿن کي ربِّ ڪريم جي نافرماني ۾ استعمال ڪرڻ هٿن جو گناھ آهي، تنهنڪري اسان جي اها ئي ڪوشش هجي ته ڪنهن تي ظلماً هٿ نه کڻون، رِشوت ڏيڻ وٺڻ جي لاءِ نه اُٿن، نه ڪنهن جو مال چوري ڪيون، نه تاس کيڏون نه پتنگ اُڏايون، نه ڪنهن نامحرم عورت سان مصافحو ڪريون۔ (بلڪه لذّت اچي ته اَمرد سان به هٿ نه ملايون)۔ جڏهن به هٿ کڻون، صِرف نيڪ ڪمن جي لاءِ کڻون، مثال طور باوُضو قرآنِ ڪريم کي هٿ لڳايو، نيڪ ماڻهن سان مصافحو ڪريو وغيره۔

پيرن جو گناھ

پيرن جو گناھ هي آهي ته هنن سان اهڙين جڳهن تي وڃڻ جو جن کان اللہ ڪريم منع ڪيو آهي، مثال طور اسان کي گھرجي ته اسان هر گز هرگز سينما

گھر ڏانهن نه وڃون، بُرن دوستن جي مجلسن ڏانهن نه هلون، شطرنج ،لُڊو، تاس، ڪرڪٽ، فٽ بال، وِڊيوگيمز، ٽيبل فٽ بال وغيره کيڏڻ يا ڏسڻ ڏانهن نه وڃون، ڪاش! پيرکڻون ته صِرف ۽ صِرف نيڪ ڪمن جي لاءِ کڻون، مثال طور پير هلن ته مسجدن، مزاراتِ اَوليا، عالمن ۽ نيڪن جي زِيارت، نيڪي جي دعوت ڏيڻ جي لاءِ هلن، سُنّتن ڀري اِجتماع ڏانهن هلن، سُنّتن جي تربيت جي مَدَني قافلن ۾ سفر(Travel) جي لاءِ هلن۔ اللہ ڪريم اسان کي رَمَضان ۽ غيرِ رَمَضان ۾ پنهنجن عضوئن کي گناهن کان بچائڻ ۽ انهن جي ذريعي نيڪ ۽ جائز ڪم ڪرڻ جي تَوفِيق عطا فرمائي۔ اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اِعتِڪاف جون سُنتون ۽ آداب

مٺا مٺا اسلامي ڀائرو! اچو! اِعتڪاف جون ڪجھ سُنّتون ۽ آداب ٻڌڻ جي سعادت حاصِل ڪريون ٿا۔ اچو! پهريان 2 فرمانِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مُلاحَظَه

ڪريو: (1) فرمايائون: جنهن شخص ايمان سان ثواب حاصل ڪرڻ جي نيّت سان اعتڪاف ڪيو ان جا تمام پويان گناهه بخشيا ويندا۔ (جامع صغير، ص۵۱۶، حديث:۸۴۸۰) (2) فرمايائون: جنهن رَمَضانُ المبارَڪ ۾ 10 ڏينهن اِعتڪاف ڪري ورتو اهو ائين آهي جيئن ٻه حج ۽ ٻه عُمرا ڪيا۔ (شعب الايمان،الرابع والعشرين من شعب الايمان،۳/۴۲۵،حديث: ۳۹۶۶) ٭ رَمَضانُ المبارَڪ جي آخري عَشَري (ڏهي) جو اِعتڪاف سُنّتِ مُؤَڪدَه عَلَي الْڪفَايه آهي، (يعني) جيڪڏهن سڀ ترڪ ڪن ته سڀني کان مُطالَبو ٿيندو ۽ شهر ۾ هڪ ڪري ورتو ته سڀئي بَرِي الذِّمَّه ٿي ويندا۔ (فيضانِ رمضان ،ص ۲۳۵) ٭ نَفلي اِعتڪاف جي لاءِ نه روزو شرط آهي نه ڪا وَقت جي قيد جڏهن به مسجد ۾ داخل ٿيو ته اِعتڪاف جي نِيت ڪري وٺو۔ (فيضانِ رمضان،ص ۲۳۵ملخصاً) ٭ رَمَضانُ المبارَڪ ۾ اِعتڪاف ڪرڻ جو سڀ کان وڏو مَقصَد شبِ قَدر جي تلاش آهي۔ (فيضانِ رمضان،ص ۲۳۲) ٭ سڀ کان افضل مسجد حرم شريف ۾ اِعتڪاف آهي پوء مسجدِ نَبَوِي ۾ پوءِ مسجدِ اَقصيٰ (يعني بَيتُ المَقدِس) ۾ پوءِ اتي جتي وڏي جماعت ٿيندي هجي۔ (فيضانِ رمضان، ص۲۴۰) ٭ مُعتَڪف اِعتڪاف جي ڪري جن نيڪين کان محروم ٿي ويو جيئن زيارتِ قُبور، مسلمانن سان ملاقات، بيمار جي تيمارداري، نمازِ جنازه ۾ حاضِري ان کي هنن سڀني نيڪين جو ثواب انهيء طرح ملندو آهي جيئن هي ڪم ڪرڻ وارن کي ثواب ملندو آهي۔ (مرآة المناجيح،۳/۲۱۷) ٭ اسلامي ڀينرون مسجدِ بَيت ۾ اِعتڪاف ڪن، مسجدِ بَيت ان جڳھ کي چوندا آهن جيڪا عَورَت گھر ۾ پنهنجي نماز جي لاءِ مخصوص ڪندي آهي۔ (فيضانِ رمضان،ص ۲۸۹) ٭ اِعتڪاف جي دوران ٻن سببن جي ڪري مسجد مان ٻاهر نڪرڻ جي اجازت آهي۔ (1) حاجتِ شرعي مثال طور جُمعه نماز ادا ڪرڻ جي لاءِ وڃڻ۔ (2) حاجتِ طبعي يعني اها ضَرورت جنهن جي بِغير چارو ناهي، مثال طور پَيشاب، پاخان و وغيره۔ (فيضانِ رمضان،ص ۲۶۵)

دعوتِ اسلامي جي هفتيوار سنتن ڀري ا ِ جتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

) 1 (جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

) 2 (سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى الله تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا

گناھ معاف ڪيا ويندا. ([1])

) 3 (رَحمت جا ستّر دروازا

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([4])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([5])

) 1 (هڪ هزار ڏينهن جون نيڪيون

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

) 2 (ڄڻ شبِ قدر حاصل ڪري ورتي
