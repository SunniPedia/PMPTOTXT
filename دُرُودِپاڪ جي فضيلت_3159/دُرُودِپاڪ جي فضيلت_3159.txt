پنهنجو ايمان ظاهر نه ڪري سگھندو هو جڏهن ته اڄ اسلامي علامتون ظاهر آهن۔ پوءِ ڪجھ دير بعد پاڻ رَضِى اللّٰه تَعَالٰي عَنْه تندور واري جي دُڪان تي ويا ۽ ماني خريدڻ جي لاءِ ان کي دقيانوسي روپيو ڏنو، جنهن جو رواج صديون اڳ ۾ ختم ٿي ويو هو ۽ ان کي ڏسڻ وارو به ڪو باقي نه رهيو هو۔ بازار وارن خيال ڪيو ته ڪو پراڻو خزانو اِن کي هٿ لڳو آهي، پوءِ اهي ان کي پڪڙي حاڪم وٽ وٺي ويا، اهو نيڪ شخص هو، هن ان کان پڇيو ته خزانو ڪٿي آهي؟ انهن چيو ته خزانو ڪٿي ناهي۔ اهو روپيو اسان جو پنهنجو آهي۔ حاڪم چيو: اها ڳالھ ڪنهن طرح يقين جوڳي نه آهي، ڇوته ان ۾ جيڪو سال لکيل آهي اهو 300 سال اڳ جو آهي ۽ اوهان نوجوان آهيو، اسان پوڙها آهيون، اسان ڪڏهن اهو سڪو ناهي ڏٺو۔ پاڻ رَضِى اللّٰه تَعَالٰي عَنْه فرمايو: ٺيڪ آهي مان جيڪو توهان کان معلوم ڪيان اهو توهان ٺيڪ ٺيڪ ٻڌايو ته مسئلو حل ٿي ويندو۔ اهو ٻڌايو ته دقيانوس بادشاھ ڪهڙي حال ۾ آهي؟ حاڪم چيو: اڄ رُوئي زمين تي ان نالي جو ڪو بادشاھ ناهي، سوين سال اڳ هڪ بي ايمان بادشاھ ان نالي جو گذريو هو۔ پاڻ فرمايائون: ڪلھ ئي ته اسان ان جي خوف سبب جان بچائي ڀڄيا آهيون ۽ منهنجا ساٿي ويجھو جبل ۾ هڪ غار اندر پناھ ورتي آهي، هلو مان توهان کي انهن سان ملايان، حاڪم ۽ شهر جا سردار ۽ هڪ وڏو تعداد ان سان گڏ غار جي ڪناري پهچي ويا۔ اَصحابِ ڪهف يَمْلِيخَا جي انتظار ۾ ها، جڏهن انهن گھڻن ماڻهن جي اچڻ جو آواز ٻڌو ته سمجھيا ته يَمْلِيخَا پڪڙجي پيو آهي ۽ دقيانوسي فوج اسان جي تلاش ۾ اچي پئي۔ پوءِ اهي الله عَزَّوَجَلَّ جي حمد ۾ مشغول ٿي ويا۔ ايتري ۾ شهر جا ماڻهو پهچي ويا ۽ يَمْلِيخَا باقين کي سمورو قصّو ٻڌايو، انهن حضرات سمجھي ورتو ته اسان الله تعاليٰ جي حڪم سان ايتري وڏي وقت (يعني 300سال کان به وڌيڪ عرصي تائين) ستل رهياسين ۽

هاڻي ان ڪري اٿاريا ويا آهيون ته ماڻهن جي لاءِ موت جي بعد زندهه ٿيڻ جو دليل ۽ نشاني بڻجون۔

جڏهن شهر جو حاڪم غار جي ڪناري پهتو ان ٽامي جو صندوق ڏٺو، ان کي کولايو ويو ته تختي نڪتي، ان تختي ۾ اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم ۽ ان جي ڪتي جو نالو لکيل هو۔ (صِراطُ الجِنان،۵/۵۴۲،بتغیر قلیل)

اَصحابِ ڪهف رَضِي اللّٰه تَعَالٰي عَنْهُم جو تعداد ۽ نالا مبارڪ

خليفه مفتي اعظم هند، شيخ الحديث حضرت علامه عبد المصطفيٰ اعظمي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جي تعداد جي باري ۾ جڏهن ماڻهن ۾ اختلاف ٿيو ته اها آيت نازل ٿي؛

قُلۡ رَّبِّیۡۤ اَعۡـلَمُ بِعِدَّ تِہِمۡ مَّا یَعۡلَمُہُمۡ اِلَّا قَلِیۡلٌ

ترجموڪنزالايمان: تون فرماءِ منهنجو رب انهن جي تعداد کي خوب ڄاڻي ٿو انهن کي نه ٿا ڄاڻن مگر ٿورا۔ (پ 15 الڪهف؛ 22) (عجائب القرآن مع غرائب القرآن،ص۱۵۳)

حضرت عبد الله ابن عباس رَضِیَ الـلّٰـهُ تَعَالٰی عَـنْهُمَا فرمايو: ته مان انهن گھٽ ماڻهن مان آهيان جيڪي اصحاب ڪهف جو تعداد ڄاڻن ٿا. اصحابِ ڪهف رَضِیَ الـلّٰـهُ تَعَالٰی عَـنْهُم جو تعداد ست آهي انهن جا نالا مبارڪ هي آهن:(1) مَکۡسَلْمِينَا ، (2) يَمْلِيخَا ، (3) مَرْطُوْنَس ، (4) بَينُوْنَس ، (5) سَارِينُونَس ، (6) ذُوْنوانس ، (7) کشْفِيطَطْنُوْنَس رَضِى اللّٰه تَعَالٰي عَنْهم اَجْمَعِين ۽ اٺون انهن جو ڪتو جنهن جو نالو قِطْمِير آهي ۔ ([1])

مٺامٺا اسلامي ڀائرو! بزرگانِ دين رَحمَةُ اللّٰه تَعَالٰي عَلَيهمْ پنهنجي ڪتابن ۾ اَصحابِ

ڪهف جي نالن جا فائدا ۽ خاصيتون بيان ڪيون آهن، اچو! اسان به هنن نالن جون برڪتون ٻڌون ٿا،

منقول آهي ته جيڪڏهن اهي نالا لکي ڪري دروازي تي لڳايا وڃن ته گھر سڙڻ کان محفوظ رهندو آهي، مال دولت تي رکيو وڃي ته چوري ناهي ٿيندو، ڪشتي يا جهاز ان جي برڪت سان غرق ناهن ٿيندا، ڀڳل شخص ان جي برڪت سان واپس موٽي ايندو آهي، ڪٿي باھ لڳل هجي ۽ اهي نالا ڪپڙي ۾ ويڙهي وڌا وڃن ته الله جي حڪم سان باھ وسامي ويندي آهي، ٻار جي روئڻ، مُدي واري بخار، مٿي جي سور، اُمُّ الصِّبْيان (يعني خاص قسم جي دماغي جھٽڪن ۽ دورن جي بيماري) ،خشڪي ۽ تري جي سفر ۾، جان و مال جي حفاظت، عقل جي تيزي ۽ قيدين جي آزادي جي لاءِ اهي نالا فائدي مند آهن۔ ([1])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جي صندوق مان نڪرڻ واري تختي ۾ انهن جي نالن جي علاوه اهو به لکيل هو ته اها جماعت پنهنجي دِين جي حفاظت جي لاءِ دقيانوس جي ڊپ مان ان غار ۾ پناھ ورتي آهي، دقيانوس کي خبر ملي ته هڪ ديوار سان انهن کي غار ۾ بند ڪرڻ جو حڪم ڏنو، اسان اهو حال ان ڪري لکون ٿا ته جيئن جڏهن اها غار کُلي ته ماڻهو انهن جي احوال تي آگاھ ٿين۔ اها تختي پڙھي سڀني کي تعجب ٿيو ۽ ماڻهو الله عَزَّوَجَلَّ جي حمد ۽ ساراھ ڪئي ته ان اهڙي نشاني ظاهر فرمائي، جنهن سان موت کان پوءِ اُٿڻ جو يقين حاصل ٿيو۔ حاڪمِ شهر پنهنجي بادشاھ بَيد رُوس کي ان واقعي جو اطلاع ڏنو، پوءِ بادشاھ به باقي معزز ماڻهن ۽ سردارن کي وٺي حاضر ٿيو ۽ شڪرِ

الٰهي جو سجدو ادا ڪيو ته الله تعاليٰ ان جي دعا قبول ڪئي۔ اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم بادشاھ سان مُعانقو ڪيو يعني گلي مِليا ۽ فرمايائون: اسان توهان کي الله عَزَّوَجَلَّ جي حوالي ڪيون ٿا۔ وَالسَّلَامُ عَلَيکَ وَرَحْمَۃُ اللّٰهِ وَبَرَکاتُهُ ، الله تعاليٰ تنهنجي، تنهنجي مُلڪ جي حفاظت فرمائي ۽ جنّن ۽ انسانن جي شر کان بچائي۔ بادشاھ بيٺو ئي هو جو اُهي حضرات پنهنجي خواب گاهن ڏانهن واپس موٽي مصروفِ خواب ٿيا ۽ الله تعاليٰ انهن کي وفات ڏني، بادشاھ هڪ وڻ جي ڪاٺي جي صندوق ۾ انهن جي جسمن کي محفوظ ڪيو ۽ الله تعاليٰ رُعب سان انهن جي حفاظت فرمائي جو ڪنهن کي مجال ناهي جو اتي پهچي سگھي۔ بادشاھ غار جي مُنڍ تي مسجد بنائڻ جو حڪم ڏنو ۽ هڪ خوشي جو ڏينهن مُعَيّن (يعني مخصوص) ڪري ڇڏيو ته هر سال ماڻهو عيد جيان اتي ايندا ڪن۔ ([1])

مٺامٺا اسلامي ڀائرو! بيان ڪيل واقعي مان اسان کي 3 مدني گل حاصل ٿيا: پهريون اهو ته مرڻ کان بعد زنده ٿي اٿڻ حق آهي ۽ اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جو واقعو ان جي نشاني ۽ دليل آهي۔ ٻيو اهو ته ڪراماتِ اَولياء حق آهن، اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم نبي نه بلڪ بني اسرائيل جا ولي آهن، انهن جي ڪرامت اها بيان ٿي ته غار ۾ 300 سالن کان وڌيڪ عرصي تائين ستل رهيا، ايترو عرصو بي غذا سمهڻ ۽ فنا نه ٿيڻ يقيناً هڪ ڪرامت آهي۔ هتي اها ڳالھ ذهن نشين رهي ته ڪرامت ستل هئڻ به ظاهر ٿي سگھي ٿي، ان جي لاءِ ولي جو جاڳڻ ضروري ناهي جيئن ته

دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جي تفسيرِ قرآن، بنام”صِراطُ الجِنان“ ۾ آهي: ڪرامت ولي کان سمهڻ مهل به ظاهر ٿي سگھي

ٿي ۽ اهڙي طرح بعدِ موت به۔ انهن جي جسمن کي مٽيءَ جو نه کائڻ اهو به ڪرامتِ اولياء مان آهي۔ اهو ضروري ناهي ته ولي پنهنجي اختيار سان ڪرامت ظاهر ڪري ۽ ان کي علم به هجي بلڪ بعض اوقات ولي جي اختيار بغير ۽ ان کي علم نه هئڻ باوجود به ڪرامت ظاهر ٿيندي آهي جيئن اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جي واقعي مان ٿيو۔ ([1]) ٽيو مدني گل اهو مليو ته بزرگن جي مزارن جي ويجھو مسجدون بنائڻ ۽ هر سال اُن جو عُرس ملهائڻ هرگز ممنوع ناهي بلڪه نيڪ ماڻهن جو طريقو آهي جيئن ته صدْرُ الْافَاضِل حضرت علّامه مولانا سَيد مفتي محمد نعيمُ الدين مُراد آبادي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه انهيء واقعي جي روشني ۾ ارشاد فرمائن ٿا: بزرگن جي مزارن جي ويجھو مسجدون بنائڻ اَهلِ اِيمان جو قديم طريقو آهي۔ بزرگن جي قُرب ۾ برڪت حاصل ٿيندي آهي، ان ڪري الله وارن جي مزار تي ماڻهو برڪت حاصل ڪرڻ جي لاءِ ويندا آهن، قبرن جي زيارت سُنّت ۽ ثواب جو باعث آهي۔ ([2])

تاجدارِ رسالت، شهنشاھِ نبوت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هر سال جي شروعات ۾ ش ُ هدَاءِ اُحُد عَلَيهمُ الرِّضْوَان جي قبرن جي زيارت لاءِ ويندا هئا ۽ اُنهن جي لاءِ دُعا ڪندا هئا (فرمائيندا هئا) ته سلامتي هجي توهان تي (راھ حق ۾) توهان جي صبر ڪرڻ جي سبب ۽ ڪهڙو نه بهتر آهي آخرت جو گھر (جيڪو توهان کي ان جي بدلي ۾ عطا ٿيو) ۔ ([3])

سِراجُ الهند، حضرت سَيدُنا شاھ عبدالعزيز محدثِ دِهلَوي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: اولياءِ ڪرام عَلَيهم رَحْمَةُ اللّٰه ِالسَّلام جو عُرس ملهائڻ ان حديثِ پاڪ مان ثابت آهي۔ ([4])

مُفَسِّرِ شَهير،حڪيم الاُمَّت مفتي احمد يار خان نعيمي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه ارشاد

فرمائن ٿا: بزرگن جي آستانن جي برابر مسجد بنائڻ ۽ برڪت جي لاءِ اُتي نمازون پڙھڻ، قرآن شريف ۽ کوڙ حديثن مان ثابت آهي۔ سوره ڪهف ۾ آهي: لَنَتَّخِذَنَّ عَلَی ۡ ہ ِ م ۡ مَّسۡجِدًا ﴿۲۱﴾ ([1]) (ترجموڪنز الايمان: قَسم آهي اسان ته اُنهن تي مسجد ٺاهِينداسين .) حضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي روضي مبارڪ ۽ اڪثر صحابه ڪرام عَلَيهمُ الرِّضْوَان جي مزارن وٽ مسجدون آهن، اهي خود صحابه رَضِى اللّٰه تَعَالٰي عَنْهم اَجْمَعِين يا نيڪ ٻانهن عَلَيهمْ رَحمَةُ اللّٰهِ الْمُبِين بڻايون، هاڻي مزاراتِ اولياء وٽ مسلمان مسجدون بڻائيندا آهن، مقبولن جي قرب ۾ نماز وڌيڪ قبول ٿيندي آهي۔ مسجدِ نبوي ۾ هڪ نماز جو ثواب پنجاھ هزار(50000) نمازن جي برابر آهي، حُضُوْرِ انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي قُرب جي وجه سان، ربّ تعاليٰ گنهگار اسرائيلين کي فرمايو هو: وَّ ادۡخُلُوا الۡبَابَ سُجَّدًا وَّ قُوۡلُوۡا حِطَّۃٌ ([2]) (ترجمو ڪنز الايمان: ۽ دروازي کان سجدو ڪندي داخِل ٿيو ۽ چئو ته اسان جا گُناھ مُعاف ٿين ۔) نبين جي قبرن جي برڪت سان توبه قبول ٿيندي۔ (الله تعاليٰ) حضرت سَيدُنا زڪريا عَلَيه السَّلَام جو واقعو بيان فرمائي ٿو: ہُنَالِکَ دَعَا زَکَرِیَّا رَبَّہٗ ۚ ([3]) (ترجمو ڪنز الايمان: اُتي زڪريا پنهنجي رب کي سڏيو ۔) حضرت سَيدُنا زڪريا عَلَيه السَّلَام حضرت سَيدَتُنا بي بي مريم رَضِي اللّٰه تَعَالٰي عَنْها وٽ بيهي پٽ جي دعا گھري، معلوم ٿيو ته بزرگن جي قُرب ۾ توب ھ ۽ دُعا وڌيڪ قبول ٿيندي آهي۔ ([4])

صدرُ الافاضِل حضرت علامه مولاناسيد مفتي محمد نعيم الدّين مُراد آبادي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا ته تفسيرِ ثَعْلَبي ۾ آهي: جيڪو هنن ڪلمات وَ کَلۡبُہ ُ م ۡ بَاسِطٌ ذِرَاعَی ۡ ہ ِ بِالۡوَصِی ۡ دِ ؕ کي لکي ڪر پاڻ سان گڏ رکي ڪتّي جي شر کان اَمن ۾ رهندو۔ ([2]) جيڪڏهن ڪتّو ڀونڪندي اڳتي وڌي حملو ڪري ته اهي قرآني ڪلمات پڙھي وٺو۔ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ اهو ڪجھ نه بگاڙي سگھندو۔ ([3])

اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جي ڪتّي جو مقام ۽ مرتبو بيان ڪندي اعليٰ حضرت،امامِ اهلسنت مولانا شاھ امام احمد رضا خان رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: اَصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جو ڪتَّو بلعم باعُور جي شڪل (جو) بڻجي جنّت ۾

ويندو ۽ اهو ان ڪتّي جي شڪل (جو) ٿي دوزخ ۾ پوندو۔اُن (يعني اصحابِ ڪهف جي ڪتّي) محبوبانِ خدا جو ساٿ ڏنو، الله (عَزَّ وَجَلَّ) ان کي انسان بڻائي جنّت عطا فرمائي ۽ ان (يعني بلعم باعُور) محبوبانِ خدا سان عداوت (يعني دشمني) ڪئي (نتيجي ۾ اهو تباھ و برباد ٿي ويو) ۔ ([1])

مُفَسّرِشهير،حڪيمُ الْاُمَّت مفتي احمد يار خان نعيمي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: ڪجھ جانور جنّت ۾ ويندا: (1) حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ڏاچي قَصوا، (2) اَصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جو ڪتّو، (3) حضرت سَيدُنا صالح عَلٰي نَبِينَا وَ عَلَيه الصَّلٰوةُ وَالسَّلَام جي ڏاچي ۽ (4) حضرت سَيدُنا عيسيٰ رُوْحُ الله عَلٰي نَبِينَا وَ عَلَيه الصَّلٰوةُ وَالسَّلَام جو دراز گوش۔ ([2]) هڪ ٻي مقام تي ارشاد فرمائن ٿا: جيڪڏهن زمين مسجد جي لاءِ وقف ٿئي ته ان جو شان ۽ عظمت وڌي ويندي آهي،اصحابِ ڪهف رَضِى اللّٰه تَعَالٰي عَنْهم جي ڪتّي پنهنجي زندگي الله تعاليٰ جي پيارن جي لاءِ وقف ڪئي ته ان کي هميشه جي زندگي ملي وئي، زمين ۽ ڪتّو زندگي وقف ڪرڻ جي سبب سان شان وارا ٿي ويا، ته جيڪڏهن انسان پنهنجي زندگي الله تعاليٰ جي رِضا ۽ خوشنودي جي حُصُول جي خاطِر دِين جي لاءِ وقف ڪري ڇڏي ته اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ملائڪن کان به افضل ٿي ويندو۔ ([3])

حضرت سَيدُنا اِمام ابو عبدُالله محمد بن اَحمد قرطبي رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: جڏهن نيڪ ٻانهن ۽ اولياء ڪرام عَلَيهم رَحْمَةُ اللّٰه ِالسَّلام جي صحبت ۾ رهڻ جي برڪت سان هڪ ڪتّو ايترو بلند مقام ماڻي ويو جو الله تعاليٰ ان جو ذڪر قرآنِ پاڪ ۾ فرمايو ته ان مسلمان جي باري ۾ توهان جو ڇا خيال آهي جيڪو

اولياء ڪرام عَلَيهم رَحْمَةُ اللّٰه ِالسَّلام ۽ نيڪ ٻانهن سان محبت ڪرڻ وارو ۽ ان جي صحبت کان فيضياب ٿيڻ وارو آهي بلڪه هن آيت ۾ ان مسلمانن جي لاءِ تسلّي آهي جيڪي ڪنهن بلند مرتبي تي فائز نه آهن۔ ([1]) يعني ان جي لاءِ تسلّي آهي جو اهي پنهنجي ان محبت ۽ عقيدت جي ڪري الله تعاليٰ جي بارگاھ ۾ ڪامياب ٿيندا۔ ([2])

بي ادبي جي مذمّت بيان ڪندي نبي ڪريم،رؤفٌ رحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: اَدَبُ السُّو ْءِ كَعِرْقِ السُّو ْءِ يعني بي ادبي نهايت بُري عادت جيان آهي۔ ([4]) حضرت سَيدُنا عبدُ الله بن مُبارَڪ رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه فرمائن ٿا: اسان کي گھڻي علم

جي مقابلي ۾ ٿوڙي ادب جي وڌيڪ ضرورت آهي۔ ([1]) اعليٰ حضرت،امامِ اهلسنت مولانا شاھ امام احمدرضاخان رَحْمَةُ اللّٰه ِتَعَالٰي عَلَيه ارشاد فرمائن ٿا: لَادِينَ لِمَنْ لَّا اَدَبَ لَه يعني جيڪو با ادب ناهي ان جو ڪو دِين ناهي۔ ([2]) ڪنهن دانا جو قول آهي: مَاوَصَلَ مَنْ وَصَلَ اِلَّا بِالْحُرْمَۃِوَمَاسَقَطَ مَنْ سَقَطَ اِلَّا بِتَرْکِ الْحُرْمَۃِ يعني جنهن جيڪو ڪجھ حاصل ڪيو اَدب ۽ احترام ڪرڻ جي سبب ئي ماڻيو ۽ جنهن جيڪو ڪجھ وڃايو اهو اَدب ۽ احترام نه ڪرڻ جي سبب ئي وڃايو ۔ ([3])

مشهور چوڻي آهي ته ”با ادب با نصيب، بے ادب بد نصيب“ تنهنڪري هر مسلمان کي گھرجي ته اهو بي ادبي۽ بي ادبن کان پري رهي۽ دِل و جان سان الله عَزَّ وَجَلَّ جي ولين جو ادب ڪري ڇاڪاڻ ته ادب انسان کي دنيا ۽ آخرت ۾ ڪاميابيون ۽ عزت ۽ شهرت ڏياريندو آهي، هيءَ اها انمول شيء آهي جو جنهن جي تعليم خود ربِّ ڪائنات عَزَّ وَجَلَّ پنهنجي مدني حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي ارشاد فرمائي جيئن نبي اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: اَدَّبَنِيْ رَبـِّيْ فَاَحْسَنَ تَاْدِيْبِـيْ يعني مونکي منهنجي ربّ عَزَّ وَجَلَّ ادب سيکاريو ۽ نهايت بهتر ادب سيکاريو۔ (جامع صغیر،حرف الہمزہ،حدیث: ۳۱۰ ،ص ۲۵) بهرحال عزّت ۽ ذِلّت سڀ الله عَزَّ وَجَلَّ جي اختيار ۾ آهي، اهو چاهي ته پنهنجي پيارن سان نسبت ۽ ان جو ادب ڪرڻ جي برڪت سان هڪ ڪتّي کي به نوازي ڇڏي ۽ ان جي مٿي تي عزّت جو تاج سجائي ان کي جنّت جو پروانو عطا ڪري ڇڏي ۽ چاهي ته بلعم بن باعور جهڙي مقبولِ بارگاھ کي بي ادبي ۽ گستاخي جي سبب سان پنهنجي بارگاھ مان ڌتڪاري ڇڏي۔

بَلْعَم باعُور بني اِسرائيل ۾ تمام وڏو عالِم هو۔ مُسْتَجَابُ الدَّعَـوَات هو (يعني اُن جي دُعا قبول ٿيندي هئي) مال جي لالچ ۾ حضرت موسيٰ عَلَيه الصَّلٰوةُ وَالسَّلَام جي لاءِ بَد دُعا ڪرڻ چاهي، جيڪي لفظ حضرت سَيدُنا موسيٰ عَلَيه الصَّلٰوةُ وَالسَّلَام جي لاءِ چوڻ چاهي پيو، پنهنجي لاءِ نڪرندا ها الله (عَزَّوَجَلَّ) ان کي هلاڪ ڪري ڇڏيو۔ (ملفوظاتِ اعليٰ حضرت،ص۳۶۷بتغير قليل) اسان الله عَزَّ وَجَلَّ کان عافيت جو سوال ڪيون ٿا۔

اچو! شيخِ طريقت،اميرِ اهلسنّت حضرت علامه مولانا ابو بلال محمد الياس عطار قادري رضوي ضيائي دَامَتْ بَـرَکاتُهمُ الْعَالِيه جي رسالي 101 مدني گل مان گھر ۾ اچڻ وڃڻ جون ڪجھ سنتون ۽ آداب ٻڌون ٿا: * جڏهن گهران ٻاهر نڪرو ته هي دعا پڙهو؛ ” بِسْمِ اللهِ تَوَکَّلْتُ عَلَی اللهِ لَاحَوْلَ وَلَا قُوَّۃَ اِلَّا بِاللّٰہِ ‘‘ ترجمو؛ الله جي نالي سان، مون الله تي ڀروسو ڪيو۔ الله کان سواءِ نه (گناهن کان بچڻ جي) قوت آهي ۽ نه (نيڪيون ڪرڻ جي) طاقت آهي۔ ([2]) اِنْ شَآءَ الله عَزَّ وَجَلَّ انهيءَ دعا کي پڙهڻ جي برڪت سان سڌي واٽ تي رهندو مصيبتن کان حفاظت ٿيندي ۽ الله عَزَّوَجَلَّ جي مدد شاملِ حال رهندي ۔ * پنهنجي گهر ۾ ايندي ويندي مَحارِم ۽ مَحرمات (مثال طور ماءُ، پيءُ، ڀاءُ، ڀيڻ، ٻارن ٻچن وغيره) کي سلام ڪريو ۔ * بنا الله (عَزَّوَجَلَّ) جو نالو وٺڻ جي مثال طور بِسْمِ الله چوڻ بنا جيڪو پنهنجي گھر ۾ داخل ٿيندو آهي، شيطان به اُن سان گڏ داخل ٿي ويندو آهي ۔ * جيڪڏهن اهڙي گهر (کڻي پنهنجي خالي گهر) ۾ وڃڻ ٿئي جو اُن ۾ ڪو به نه هجي ته هيئن چئو؛ اَلسَّلَامُ عَلَیْنَا وَعَلٰی عِبَادِ اللهِ الصّٰلِحِیْن (يعني اسان تي ۽ الله عَزَّوَجَلَّ جي نيڪ ٻانهن تي سلام) ملائڪ انهي سلام جو

جواب ڏيندا۔ ([1]) يا هن ريت چوي؛ اَلسَّلامُ عَلَیْکَ اَیُّـھَاالنَّبِیُّ (يعني يا نبي توهان تي سلام) ڇوته حضور اقدس صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو روح مبارڪ مسلمانن جي گهرن ۾ تشريف فرما هوندو آهي ۔ ([2]) * جڏهن ڪنهن جي گهر ۾ داخل ٿيڻ چاهيو ته هيئن چئو؛ اَلسَّلامُ عَلَیْکُم ڇا مان اندر اچي سگھان ٿو؟ * جيڪڏهن داخل ٿيڻ جي اجازت نه ملي ته خوشيءَ سان موٽي وڃو، ٿي سگھي ٿو ڪنهن مجبوري جي ڪري گهر جي مالڪ اجازت نه ڏني هجي * جواب ۾ نالو ٻڌائڻ کان پوءِ دروازي کان هٽي بيهو ته جيئن دروازو کلندي ئي گهر ۾ نظر نه پوي * ڪنهن جي گهر وڃو ته اُتان جي انتظامات تي بي جا تنقيد نه ڪريو ان سان اُن جي دل آزاري ٿي سگھي ٿي * واپسيءَ تي گهروارن جي باري ۾ دعا به ڪريو ۽ شڪريو به ادا ڪريو ۽ سلام به ۽ ٿي سگھي ته ڪو سنتن ڀريو رسالو وغيره تحفي ۾ پيش ڪريو۔
