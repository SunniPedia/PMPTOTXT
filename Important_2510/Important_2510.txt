اَلۡـحَـمۡـدُ لِـلّٰـہِ رَبِّ الۡـعٰـلَـمِیۡنَ وَ الـصَّـلٰـوۃُ وَ الـسَّـلَامُ عَـلٰی سَـیِّـدِ الۡـمُـرۡ سَـلِـیۡنَ

اَمَّـا بَــعۡـدُ فَـاَعُـوۡذُ بِـا لـلّٰـہِ مِـنَ الـشَّـیۡـطٰنِ الـرَّجِیۡمِؕ بِـسۡمِ اللـہِ الـرَّحۡـمٰنِ الـرَّحِـیۡمِؕ

حضرت ابراهيم جون قربانيون

وَعَلٰى اٰلِكَ وَاَصۡحٰبِكَ يَا حَبِيۡبَ الله

اَلصَّلٰوةُ وَالسَّلَامُ عَلَيۡكَ يَا رَسُوۡلَ الله

وَعَلٰى اٰلِكَ وَاَصۡحٰبِكَ يَا نُوۡرَ الله

اَلصَّلٰوةُ وَالسَّلَامُ عَلَيۡكَ يَا نَبِيَّ الله

نَوَيۡتُ سُنَّتَ الۡاِعۡتِكَافِ

(مون سنّتِ اعتڪاف جي نيّت ڪئي)

جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي نفلي اعتڪاف جي نيّت ڪري وٺندا ڪريو، جيستائين مسجد ۾ رهندؤ نفلي اعتڪاف جو ثواب حاصل ٿيندو رهندو ۽ ضمنًا مسجد ۾ کائڻ پيئڻ به جائز ٿي ويندو۔

دُرودِ پاڪ جي فضيلت

فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهي: جنهن کي اهو پسند هجي ته الله عَزَّوَجَلَّ جي بارگاھ ۾ پيش ٿيڻ وقت الله عَزَّوَجَلَّ ان کان راضي هجي، ان کي گھرجي ته مون تي ڪثرت سان دُرود شريف پڙهي.

(فِردوسُ الاخباربماثور الخطاب،ج۲ص۲۸۴، حديث۶۰۸۳)

بیٹھتے، اُٹھتے، جاگتے، سوتے

ہو اِلٰہی مِرا شِعار دُرُود

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نيتون ڪري وٺون ٿا.

فرمانِ مصطفي ٰ صَلَّي اللهُ تَعَالٰي عَلَيهِ وَاٰلِهٖ وَسَلَّم: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِه “ يعني مسلمان جي نيت ان جي عمل کان بهتر آهي. [1]

ٻه مدني گُل: (1) بغير سٺي نيت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نيتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نيّتون

* نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس * ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس * ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس * ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس * صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي (يعني دلداري) جي لاءِ بلند آواز سان جواب ڏيندس * بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

پر افضل اسان جا پيارا آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ئي آهن. الله عَزَّوَجَلَّ سيپاري 3 سُوْرَۃُ الْبَقَرہ جي آيت نمبر 253 ۾ ارشاد فرمائي ٿو:

تِلۡکَ الرُّسُلُ فَضَّلۡنَا بَعۡضَہُمۡ عَلٰی بَعۡضٍ ۘ مِنۡہُمۡ مَّنۡ کَلَّمَ اللہُ وَ رَفَعَ بَعۡضَہُمۡ دَرَجٰتٍ ؕ

(سيپارو: 3 سُورَةُ البَقرہ آيت نمبر 253)

ترجمو ڪنزالايمان: هي رسول آهن اسان انهن مان هڪڙن کي ٻين تي فضيلت ڏني انهن مان ڪنهن سان الله ڪلام فرمايو ۽ ڪو اهو آهي جنهن کي سڀني تي درجا بلند ڪيائين.

صَدۡرالاَفاضِل حضرت علامه مولانا سيِّد محمد نعيمُ الدّين مراد آبادي عَلَیهِ رَحْمَةُ اﷲِ الهادي الله عَزَّوَجَلَّ جي ان فرمان ته ” ڪو اهو آهي جنهن کي سڀني تي درجا بلند ڪيائين “ جي متعلق فرمائن ٿا: ان مان مراد حضور پُرنور سَيِّد انبياء محمد مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهن توهان کي ڪيترن ئي درجن جي سبب سڀني انبياء عَـلَـيْهِمُ السَّلَام تي افضل ڪيو. ان تي پوري اُمّت جو اِجۡماع آهي ۽ اڪثر احاديث کان ثابت آهي. حُضُور عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جا اهي خَصائِص ۽ ڪمالات جن ۾ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام سڀني انبياء تي فوقيت رکندڙ ۽ افضل آهن ۽ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو ڪو شريڪ (يعني هڪجهڙي مرتبي جو) ناهي، بيشمار آهن جو قرآن ڪريم ۾ هي ارشاد ٿيو ته ” درجا بلند ڪيائين “ ان درجن جو ڪو شمار قرآن ڪريم ۾ ذڪر ناهي فرمايو ته هاڻي ڪير حد لڳائي سگھي ٿو.

(خزائن العرفان، پارہ:۳، بقرہ: تحت الآيہ:۲۵۳، ملتقطاً)

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته اسان جي پياري آقا ۽ مولٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سڀني انبياء عَـلَيْـهِمُ الـسَّلَام کان افضل آهن، ايستائين جو جنابِ ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي به جيڪي خصوصيتون ۽ ڪمالات ۽ بلند مرتبا حاصل ٿيا اهي به اسان جي آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو صَدۡقو آهي. ڇو جو پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي صدقي سان ئي دُنْیَا وَمَافِیْہَا (دنيا ۽ جيڪو ڪجھ

الـصَّلٰوة ُ وَالـسَّلَام سان گڏ ملي صبر ۽ رِضا جو اهڙو منظر پيش فرمايو، جنهن جو مثال نه ٿو ملي. پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام اٺين ذوالحج جي رات هڪ خواب ڏٺو جنهن ۾ ڪو چوڻ وارو چئي رهيو هو: ”بيشڪ الله عَزَّوَجَلَّ توکي پنهنجو پٽ ذَبۡح ڪرڻ جو حڪم ڏئي ٿو“. پاڻ صبح کان شام تائين انهيءَ باري ۾ غور فرمائيندا رهيا ته هي خواب الله عَزَّوَجَلَّ جي طرفان آهي يا شيطان جي طرفان؟ اٺ ذوالحج جو نالو یَوۡمُ التَّرۡوِیَہ (يعني سوچ ويچار جو ڏينهن) انهيءَ ڪري رکيو ويو. نائين رات وري ساڳيو خواب ڏٺائون ۽ صبح يقين ڪري ورتائون ته هي حڪم الله عَزَّوَجَلَّ جي طرفان آهي. ان ڪري 9 ذوالحج کي یَوۡمِ عَرفَه (يعني سڃاڻپ وارو ڏينهن) چيو ويندو آهي. ڏهين رات وري اهوئي خواب ڏسڻ کان بعد پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام صبح جو هن خواب تي عمل ڪرڻ يعني پٽ جي قرباني جو پڪو ارادو ڪري ورتو، انهيءَ ڪري 10 ذوالحج کي یَوۡمُ النَّحۡر (يعني ذبح جو ڏينهن) چيو ويندو آهي. (تفسير ڪبير ج 9، ص 346) الله عَزَّوَجَلَّ جي حڪم تي عمل ڪندي پُٽ جي قرباني جي لاءِ حضرت اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جڏهن پنهنجي پياري پُٽ حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي جنهن جي عمر ان وقت 7 سال (يا 13 سال يا هن کان ڪجھ وڌيڪ) هئي گڏ وٺي هليا. (پُٽ هجي ته اهڙو) پوءِ جيئن حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام فرمايو هو انهن کي اهڙي ئي طرح ٻَڌي ڇڏيو، پنهنجي ڇُري تيز ڪئي، حضرت اسماعيل عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي پيشاني ڀر ليٽايو، انهن جي چهري تان نظر هٽائي ۽ انهن جي ڳلي تي ڇُري وهائي ڇڏي پر ڇُري پنهنجو ڪم نه ڪيو يعني ڳلو نه ڪٽيو. انهيءَ وقت حضرت اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام تي وَحي نازِل ٿي . جيئن ته سيپاري 23 سُورَةُ الصّٰفّٰت آيت نمبر 104 کان 107 ۾ ارشاد ٿئي ٿو:

وَ نَادَیۡنٰہُ اَنۡ یّٰۤاِبۡرٰہِیۡمُ ﴿۱۰۴﴾ۙ قَدۡ صَدَّقۡتَ الرُّءۡیَا ۚ اِنَّا کَذٰلِکَ نَجۡزِی الۡمُحۡسِنِیۡنَ ﴿۱۰۵﴾ اِنَّ ہٰذَا لَہُوَ الۡبَلٰٓـؤُا الۡمُبِیۡنُ ﴿۱۰۶﴾ وَ فَدَیۡنٰہُ بِذِبۡحٍ عَظِیۡمٍ ﴿۱۰۷﴾

ترجمو ڪنزالايمان: ۽ پيءُ پٽ کي پيشاني ڀر ليٽايو. (ان وقت جو حال نه پڇ) ۽ اسان ان کي سڏڻ فرمايو اي ابراهيم بي شڪ تو خواب سچو ڪري ڏيکاريو اسان ايئن انعام ڏيندا آهيون چڱن کي. بيشڪ اهو روشن امتحان هو. ۽ اسان هڪ وڏو ذبيحو ان جي فديي ۾ ڏيئي ان کي بچائي ورتو.

(تفسير خازن ج ۴ ص ۲۲ملخّصاً)

حضرت سَيّدُنا ابراهيم عَـلَيْـهِ الـسَّـلَام جو تعارف

مٺا مٺا اسلامي ڀائرو! ان واقعي مان جتي حضرت سَيّدُنا اسماعيل عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي اعلٰي صَبۡر ۽ رِضا جو مقام ثابت ٿئي ٿو، اُتي حضرت سَيّدُنا ابراهيم خليل الله عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو پنهنجي ربّ عَزَّوَجَلَّ جو حد درجي جو مطيع ۽ فرمانبردار هئڻ به ظاهر ٿئي ٿو.

ڇا هر ڪو خواب ڏسي پنهنجو پُٽ ذبح ڪري سگهي ٿو؟

ياد رهي! ڪو به شخص خواب يا غيبي آواز ٻڌي ڪري پنهنجي يا ٻئي ڪنهن جي ٻار يا ڪنهن انسان کي ذبح نٿو ڪري سگهي، ڪندو ته سخت گنهگار ۽ جهنّم جي عذاب جو حقدار ٿيندو. حضرت اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام خواب ڏسي پنهنجي پُٽ کي قربان ڪرڻ جي لاءِ تيار ٿي ويا اهو حق آهي ڇو ته پاڻ نبي هئا ۽ نبي جو خواب وَحۡی ِ الٰھی هوندو آهي. انهن هستين جو امتحان هو، حضرت جبرائيل عَـلَيْـهِ الـسَّـلَام جَنَّتي دنبو کڻي آيا ۽ الله تعاليٰ جي حڪم سان حضرت اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام پنهنجي پياري پُٽ جي جڳهه تي

ان جنّتي دُنۡبي کي ذبح فرمائي ڇڏيو. (پُٽ هجي ته اهڙو)

پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو نالو ” ابراهيم “ آهي. ابراهيم سُرياني زبان جو لفظ آهي. ان جي معنٰي آهي اَبٌ رَحِیْمٌ (يعني مهربان پيءُ) ڇو جو پاڻ ٻارن تي وڏا مهربان هئا. گڏوگڏ مهمان نوازي ۽ رَحۡم و ڪرم ۾ پاڻ مشهور آهن. تنهن جي ڪري سندن کي ” ابراهيم “ چيو ويندو آهي. ڪجھ ماڻهن چيو آهي ته ابراهيم اصل ۾ ابرم هو. جنهن جي معنٰي آهي بزرگ . ڇو جو پاڻ ڪيترن ئي انبياءِ ڪرام (عَـلَيْـهِمُ الـصَّلٰوة ُ وَالـسَّلَام) جا والد آهن، تنهنڪري سندن جو نامِ نامي ابراهيم آهي. (تفسيرِ نعيمي ۱/۶۱۸، ملتقطاً) پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي ڪُنۡيت ابُو الضَّیْفان (وڏو مهمان نواز) آهي. ڇو جو سندن جو گھر رستي جي ڪناري تي هو جيڪو به اُتان کان گذرندو هو توهان انهن جي مهمان نوازي ڪندا هئا. (تفسيرِ خازن،تحت قولہ و من احسن ديناممن اسلم،۱/۴۳۴) پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي ولادت ” اهواز “ جي سرزمين جي ” سوس “ مقام تي ٿي. پوءِ سندن جا والد کين ” بابِل “ نمرود جي ملڪ وٺي آيا. الله عَزَّوَجَلَّ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي حڪمت ۽ دانائي سان سرفراز فرمايو ۽ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي زمين ۽ آسمان جي سڀ شين جو مشاهدو به ڪرايو. جيئن ته ارشادِ رَبّاني آهي ته: ﴿ وَ کَذٰلِکَ نُرِیۡۤ اِبۡرٰہِیۡمَ مَلَکُوۡتَ السَّمٰوٰتِ وَ الۡاَرۡضِ وَ لِیَکُوۡنَ مِنَ الۡمُوۡقِنِیۡنَ ﴾ ترجمو ڪنزالايمان: ۽ اهڙي طرح اسين ابراهيم کي ڏيکاريون ٿا سموري بادشاهي آسمانن ۽ زمين جي ۽ هن ڪري ته اهو عين اليقين وارن مان ٿي وڃي. (سيپارو 7، الانعام:75)

10خاص فضائل

پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي ڏھ (10) اهڙا فضائل حاصل آهن جيڪي پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي لاءِ خاص آهن. اهي خوبيون هي آهن: | رسولِ

پاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن کان پوءِ حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام سڀني کان افضل آهن. | حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام پاڻ کان پوءِ ايندڙ سڀني نبين جا والد آهن. (بہارِ شريعت، ۱/۵۲) | هر آسماني دين ۾ سندن جي ئي پيروي ۽ اطاعت آهي. | هر دين واراتوهان جي تعظيم ڪندا آهن. | سندن جي ئي ياد قرباني آهي. | حج جا ارڪان به سندن جي ئي ياد آهن. | پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ئي ڪعبي شريف جا پهريان تعمير ڪرڻ وارا يعني ان کي گهر جي صورت ۾ ٺاهڻ وارا آهن. | جنهن پٿر (مقام ابراهيم) تي بيهي پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ڪعبو شريف ٺاهيو اتي قيام ۽ سجدا ٿيڻ لڳا. | فوت ٿي ويل مسلمان ٻارن جي پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ۽ سندن بيبي صاحبه حضرت ساره رَضِیَ اللهُ تَعَالٰی عَنْهَا عالَمِ برزخ ۾ پرورش ڪن ٿيون. | قيامت ۾ سڀ کان پهريان سندن کي ئي عمده لباس عطا ڪيو ويندو ان کان فوراً بعد اسان جي پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن کي.

(تفسير نعيمي ج۱ص۶۲۱ مُلَخّصًا)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

سڀني کان افضل اسان جو نبي

مٺا مٺا اسلامي ڀائرو! بيان ڪيل آخري فضيلت ته قيامت واري ڏينهن سڀ کان پهريان پاڻ عَلَيْـهِ الصَّلٰوة ُ وَالسَّلَام کي بهترين لباس ملندو بعد ۾ اسان جي نبي پاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي، ان کي ٻڌي ڪري ڪنهن جي ذهن ۾ اهو خيال اچي سگھي ٿو ته حضرت سَيِّدُنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام حُضور سيِّد عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان افضل آهن. قيامت جي ڏينهن سڀ کان پهرين حضرت سَيِّدُنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي حُلّو (بهشتي لباس) عطا ٿيڻ پنهنجي جڳھ،

پر افضل اسان جا پيارا آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ئي آهن. الله عَزَّوَجَلَّ سيپاري 3 سُوْرَۃُ الْبَقَرہ جي آيت نمبر 253 ۾ ارشاد فرمائي ٿو:

تِلۡکَ الرُّسُلُ فَضَّلۡنَا بَعۡضَہُمۡ عَلٰی بَعۡضٍ ۘ مِنۡہُمۡ مَّنۡ کَلَّمَ اللہُ وَ رَفَعَ بَعۡضَہُمۡ دَرَجٰتٍ ؕ

(سيپارو: 3 سُورَةُ البَقرہ آيت نمبر 253)

ترجمو ڪنزالايمان: هي رسول آهن اسان انهن مان هڪڙن کي ٻين تي فضيلت ڏني انهن مان ڪنهن سان الله ڪلام فرمايو ۽ ڪو اهو آهي جنهن کي سڀني تي درجا بلند ڪيائين.

صَدۡرالاَفاضِل حضرت علامه مولانا سيِّد محمد نعيمُ الدّين مراد آبادي عَلَیهِ رَحْمَةُ اﷲِ الهادي الله عَزَّوَجَلَّ جي ان فرمان ته ” ڪو اهو آهي جنهن کي سڀني تي درجا بلند ڪيائين “ جي متعلق فرمائن ٿا: ان مان مراد حضور پُرنور سَيِّد انبياء محمد مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم آهن توهان کي ڪيترن ئي درجن جي سبب سڀني انبياء عَـلَـيْهِمُ السَّلَام تي افضل ڪيو. ان تي پوري اُمّت جو اِجۡماع آهي ۽ اڪثر احاديث کان ثابت آهي. حُضُور عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جا اهي خَصائِص ۽ ڪمالات جن ۾ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام سڀني انبياء تي فوقيت رکندڙ ۽ افضل آهن ۽ پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو ڪو شريڪ (يعني هڪجهڙي مرتبي جو) ناهي، بيشمار آهن جو قرآن ڪريم ۾ هي ارشاد ٿيو ته ” درجا بلند ڪيائين “ ان درجن جو ڪو شمار قرآن ڪريم ۾ ذڪر ناهي فرمايو ته هاڻي ڪير حد لڳائي سگھي ٿو.

(خزائن العرفان، پارہ:۳، بقرہ: تحت الآيہ:۲۵۳، ملتقطاً)

مٺا مٺا اسلامي ڀائرو! معلوم ٿيو ته اسان جي پياري آقا ۽ مولٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سڀني انبياء عَـلَيْـهِمُ الـسَّلَام کان افضل آهن، ايستائين جو جنابِ ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي به جيڪي خصوصيتون ۽ ڪمالات ۽ بلند مرتبا حاصل ٿيا اهي به اسان جي آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو صَدۡقو آهي. ڇو جو پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي صدقي سان ئي دُنْیَا وَمَافِیْہَا (دنيا ۽ جيڪو ڪجھ

ان ۾ آهي سڀ) کي پيدا ڪيو ويو. اعلٰي حضرت امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه انهي حقيقت جي طرف اشارو ڪندي فرمائن ٿا:

ہوتے کہاں خلیل و بِنا، کَعبہ و مِنیٰ

لَوْلاک والے صاحِبی سب تیرے گھر کی ہے

(حدائقِ بخشش، ص۲۰۳)

ٻئي جهان توهان جي لاءِ ٺهيا

يعني اعلٰي حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه هي فرمائي رهيا آهن: هر ڪنهن جو وجود حُضُور عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي نور جي فيض سان آهي. جيڪڏهن حُضُور عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام نه هجن ها ته نه ڪعبو تعمير ڪرڻ وارا حضرت خليلُ الله عَـلَيْـهِ الـسَّلَام هجن ها، نه ڪعبي شريف جو بنياد رکيو وڃي ها ۽ نه ئي منٰي جون رونقون هجن ها. (شرح حدائقِ بخشش ازمولانا حسن قادري، ص:۵۸۵) ته عالم جي پيدائش به حُضُور پُرنور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي طفيل ئي ٿي آهي ۽ ان ڪائنات جو رنگ ۽ خوشبو به مٺي مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي دم قدم سان آهي. زمينن ۽ آسمانن جي خالق عَزَّوَجَلَّ ڪائنات بڻائڻ جو ارادو فرمايو ته ان پنهنجي حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي نُور جي تخليق فرمائي، ان وقت نه جنّ هئا نه انسان، نه لَوح نه قَلَم، نه جنّت ۽ نه دوزخ ۽ نه حُور ۽ مَلَڪ هئا نه زمين ۽ آسمان ۽ نه ئي هي سج ۽ چنڊ وجود ۾ آيا هئا. ان وقت الله عَزَّوَجَلَّ پنهنجي حبيب جي نُور سان لوح ۽ قَلَم ۽ عَرش و ڪُرسي پيدا فرمايا، پوءِ اِن پاڪ نور سان آسمان ۽ زمين، جنّت ۽ دوزخ کي بڻايو، مقصد هي ته حُضُورِ پاڪ، صاحبِ لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ذاتِ پاڪ ئي ڪائنات جي تخليق جو مقصد آهي. جيئن ته حديثِ قُدسي جو مفهوم آهي ته

حضرت سَيِّدُنا عبدُالله بن عباس رَضِیَ الله تَعَالٰی عَنْہُمَا کان روايت آهي: الله عَزَّوَجَلَّ حضرتِ سَيِّدُنا عيسٰي رُوحُ الله عَلٰی نَبِیِّنَاوَعَلَیْہِ الصَّلٰو ۃ ُ وَالسَّلَام ڏانهن وحي موڪلي: ”اي عيسيٰ! محمد (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) تي ايمان آڻيو! ۽ توهان جي اُمّت مان جنهن کي انهن جو زمانو ملي، انهن کي به حڪم ڪجو ته انهن تي ايمان آڻين فَلَوْلَامُحَمَّدٌ مَاخَلَقْتُ اٰدَمَ وَلَاالْجَنَّۃَ وَلَاالنَّارَ ، يعني ڇو جو جيڪڏهن مُحَمّد عربي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ذاتِ گرامي نه هجي ها ته آئون نه آدم کي پيدا ڪريان ها ۽ نه ئي جنّت ۽ دوزخ بڻايان ها.“ جڏهن مون عرش کي پاڻي تي بڻايو ته اهو ان وقت جنبش (حرڪت) ڪري رهيو هو مون ان تي ’’ لَا ۤ اِلٰہَ اِلَّا اللہُ مُحَمَّدٌ رَّسُوْلُ اللہِ ‘‘ لکي ڇڏيو، بس اهو رُڪجي ويو. (الخصائص الڪبريٰ، باب خصوصيتہ بڪتابة اسمہ الشريف …الخ ، ۱/۱۴) بس! ان ڪائنات جون سڀ رونقون حُضُور عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي سبب ئي آهن.

حديثِ قُدسي آهي، الله عَزَّوَجَلَّ ارشاد فرمائي ٿو: لَقَدْخَلَقْتُ الْدُنْیَا وَاَھْلَھَا لِاُعَرِّفَھُمْ کَرَامَتَکَ وَمَنْزِ لَتَکَ عِنْدِیْ وَلَوْ لاَکَ یَا مُحَمَّدُ! مَا خَلَقْتُ الدُّنْیَا۔ يعني اي منهنجا حبيب (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) مون دنيا ۽ اهلِ دنيا کي ان لاءِ پيدا ڪيو ته جيڪا عزت ۽ منزلت تنهنجي مون وٽ آهي، آئون انهن کي ان جي سُڃاڻ ڪرائي ڇڏيان ۽ اي منهنجا حبيب (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم)! جيڪڏهن تون نه هجين ها ته آئون دنيا کي نه پيدا ڪريان ها. (تاريخِ دمشق،۳/۵۱۸،از ملفوظاتِ اعليٰ حضرت، ص۵۲۱) معلوم ٿيو ته دنيا جون سموريون شيون ايستائين جو سڀني انبياء عَـلَيْـهِمُ الـصَّلٰوة ُ وَالـسَّلَام کي به وجود جي دولت اسان جي آقا، مڪي مدني مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ئي مُهابي ملي آهي، توهان ئي اصل ڪائنات ۽ مَنۡبعِ موجودات آهيو. تنهنڪري ته اعلٰي حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا.

(1) زمین و زَماں تمہارے لیے، مَکین و مَکاں تمہارے

چُنِیْن و چُناں تمہارے لیے، بنے دو جہاں تمہارے لیے

(2) فِرِشتے خِدَم، رسولِ حِشَم، تمامِ اُمَم، غلامِ کَرَم

وُجُود و عَدَم، حُدُوث و قِدَم، جَہاں میں عِیاں تمہارے لیے

(حدائقِ بخشش، ص۳۴۸)

اشعار جي وضاحت

(1) يارسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! هي پوري زمين اوهان جي لاءِ بڻائي وئي، جيترا به زمانا آيا، توهان جي خاطر، هر شيءِ اوهان جي خاطر بلڪه ٻئي جهان ئي توهان جي لاءِ آهن.

(2) يارسولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! فِرشتا اوهان جا خدمت گذار آهن، انبياء ۽ رسول عَـلَيْـهِمُ الـصَّلٰوة ُ وَالـسَّلَام توهان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا خيرخواه آهن، سڀئي امتي توهان جي ڪرم جا سُوالي آهن، وجود هجي يا عدم (يعني ٿيڻ يا نه ٿيڻ) عالمِ حُدوث هجي يا قدم (يعني زندگي يا موت) نئون هجي يا پراڻو، رات هجي يا ڏينهن انهن سڀني جو ظاهر ٿيڻ توهان جي بابرڪت ذات جي طفيل آهن.

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

قوم کي نيڪي جي دعوت ڏيڻ

مٺا مٺا اسلامي ڀائرو! حضرت ابراهيم خليلُ الله عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي الله عَزَّوَجَلَّ اسان جي پياري نبي، مڪي مدني صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان بعد تمام انبياء ۾ سڀ کان وڏو رتبو عطا فرمايو آهي. الله عَزَّوَجَلَّ پنهنجي مُقرَّب ۽

محبوب ٻانهن کي آسانين سان گڏ ڪيترن ئي مشڪلاتن ۾ مبتَلا فرمائي ڪري انهن جي آزمائش به فرمائيندو آهي ۽ هي نُفوسِ قُدسِيه حرفِ شڪايت زبان تي آڻڻ جي بجاءِ هميشه خوش مزاجي سان انهن مصيبتن ۽ مشڪلاتن کي برداشت ڪندا آهن. الله عَزَّوَجَلَّ حضرت ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي به ڪيترن ئي شين سان آزمايو ۽ پاڻ عَـلَيْـهِ الـسَّلَام الله عَزَّوَجَلَّ جي فضل ۽ ڪرم سان هر امتحان ۾ ڪامياب ٿيا. پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي قوم شرڪ جي لعنت ۾ مبتلا هئي. جڏهن پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام اعلانِ نبوت فرمايو ته سڀ کان پهريان اهل ۽ عيال مان پنهنجي چاچي کان آغاز فرمايو ۽ ان کي شرڪ کان باز رهڻ ۽ الله عَزَّوَجَلَّ کي ئي معبودِ حقيقي مڃڻ جي دعوت ڏني. ان جو نتيجو هي نڪتو ته انهن عَـلَيْـهِ الـسَّلَام جو چاچو سندن جي ڳالھ مڃڻ جي بجاءِ انهن عَـلَيْـهِ الـسَّلَام جو دشمن ٿي ويو. قرآنِ پاڪ ۾ هي واقعو هن ريت بيان ڪيو ويو، سيپاري 16، سورهء مريم جي آيت نمبر 42 ،43 ۾ ارشاد ٿئي ٿو:

اِذۡ قَالَ لِاَبِیۡہِ یٰۤاَبَتِ لِمَ تَعۡبُدُ مَا لَا یَسۡمَعُ وَ لَا یُبۡصِرُ وَ لَا یُغۡنِیۡ عَنۡکَ شَیۡئًا ﴿۴۲﴾ یٰۤاَبَتِ اِنِّیۡ قَدۡ جَآءَنِیۡ مِنَ الۡعِلۡمِ مَا لَمۡ یَاۡتِکَ فَاتَّبِعۡنِیۡۤ اَہۡدِکَ صِرَاطًا سَوِیًّا ﴿۴۳﴾

(سيپاري 16، سورهء مريم آيت نمبر 42 ،43)

ترجمو ڪنزالايمان: جڏهن پنهنجي پيءَ کي چيائين اي منهنجا پيءُ اهڙي کي ڇو ٿو پوڄين جيڪو نه ٻڌي نه ڏسي ۽ نه تنهنجي ڪنهن ڪم اچي. اي منهنجا پيءُ بيشڪ مون وٽ اهو علم آيو جيڪو تو وٽ نه آيو، تون منهنجي پٺيان هليو اچ ته مان توکي سڌي واٽ ڏيکاريان.

مٺا مٺا اسلامي ڀائرو! آيت مبارڪه ۾ حضرت ابراهيم عَـلَيْـهِ الـسَّلَام جي چاچي کي پيءُ سڏيو ويو آهي، ان جو سبب بيان ڪندي حضرت صَدۡرُالاَفاضِل حضرت علّامه مولانا سيِّد محمد نعيمُ الدّين مُراد آبادي عَلَیهِ رَحْمَةُ اﷲِ الهادي فرمائن ٿا:

قامُوس ۾ آهي ته آزر، حضرت ابراهيم عَـلَيْـهِ الـسَّلَام جي چاچي جو نالو آهي. امام علّامه جلالُ الدّين سِيُوطي (رَحْمَةُ اللهِ تَعَالٰی عَلَيْه) مسالڪ الحُنَفاء ۾ به ايئن ئي لکيو آهي، چاچي کي پيءُ چوڻ سڀني ملڪن ۾ معمول آهي بالخصوص عرب ۾، قرآن ڪريم ۾ آهي ﴿ نَعۡبُدُ اِلٰہَکَ وَ اِلٰـہَ اٰبَآئِکَ اِبۡرٰہٖمَ وَ اِسۡمٰعِیۡلَ وَ اِسۡحٰقَ اِلٰـہًا وَّاحِدًا ﴾ ان ۾ حضرت اِسمٰعِيل کي حضرت يعقوب جي ابن ۾ ذڪر ڪيو ويو آهي ان جي باوجود جو پاڻ چاچا آهن. (خزائن العرفان،ص۲۶۱)

الغَرۡض پاڻ عَـلَيْـهِ الـسَّلَام پنهنجي چاچي کي هر نموني سمجھايو پر اُهي پنهنجي ابن ڏاڏن جي دين کي ڇڏڻ تي راضي نه ٿيا ۽ نهايت سخت لهجي ۾ جواب ڏنو، جيئن قرآنِ پاڪ جي سيپاري 16، سورهء مريم جي آيت نمبر 46 ۾ انهن لفظن سان بيان فرمايو آهي:

قَالَ اَرَاغِبٌ اَنۡتَ عَنۡ اٰلِہَتِیۡ یٰۤـاِبۡرٰہِیۡمُ ۚ لَئِنۡ لَّمۡ تَنۡتَہِ لَاَرۡجُمَنَّکَ وَ اہۡجُرۡنِیۡ مَلِیًّا ﴿۴۶﴾

(سيپارو 16، سورة مريم آيت نمبر 46)

ترجمو ڪنزالايمان: چيائين ڇا تون منهنجي خدائن کان منهن موڙين ٿو اي ابراهيم بيشڪ جيڪڏهن تون نه مُڙين ته مان توتي پٿراءُ ڪندس۽ مون کان جدا ٿي وڃ ڊگھي زماني تائين.

ٿڌي با ھ

پاڻ عَـلَيْـهِ الـسَّلَام جي چاچي جڏهن توهان جي ڳالھ مڃڻ کان انڪار ڪيو ۽ اهو به توهان جو دشمن ٿي ويو ته ان کان پوءِ پاڻ عَـلَيْـهِ الـسَّلَام پنهنجي قوم کي دعوت ڏني ته انهن به سندن عَـلَيْـهِ الـسَّلَام جي ڳالھ مڃڻ کان انڪار ڪري ڇڏيو. پر پاڻ عَـلَيْـهِ الـسَّلَام مسلسل پنهنجي قوم کي نيڪي جي دعوت ڏيڻ ۽ انهن کي ڪُفۡر ۽ شِرڪ جي اونداھي وادين کان ڪڍڻ جي لاءِ هر وقت تيار رهيا پر اهي باز نه آيا ۽ اهي به پاڻ عَـلَيْـهِ الـسَّلَام جا دشمن ٿي ويا ۽ چوڻ لڳا:

﴿ قَالُوا اقۡتُلُوۡہُ اَوۡ حَرِّقُوۡہُ ﴾ ترجمو ڪنزالايمان: چيائون هن کي قتل ڪري ڇڏيو يا ساڙي ڇڏيو (پارہ:۲۰، العنکبوت:۲۴) صَدۡرُالاَفاضِل حضرت علّامه مولانا سيِّد محمد نعيمُ الدّين مراد آبادي عَلَیهِ رَحْمَةُ اﷲِ الهادي فرمائن ٿا: نمرود ۽ ان جي قوم پاڻ عَـلَيْـهِ الـسَّلَام کي باھ ۾ ساڙڻ تي متفق ٿي وئي ۽ انهن پاڻ عَـلَيْـهِ الـسَّلَام کي هڪ مڪان ۾ قيد ڪري ڇڏيو ۽ قريه ڪوثيٰ ۾ هڪ عمارت ٺاهي ۽ هڪ مهيني تائين ڪوشش سان سڀني قِسمن جون ڪاٺيون جمع ڪيون ۽ هڪ وڏي باھ ٻاري، جنهن جي تپش سان هوا ۾ اڏامڻ وارا پکي به سڙي ويندا هئا ۽ هڪ مَنۡجَنِيق (مَنۡ.جَ.نِيۡق پٿر اڇلائڻ واري توپ) بيهاري ۽ پاڻ عَـلَيْـهِ الـسَّلَام کي ٻَڌي ان ۾ رکي باھ ۾ اڇلايو، ان وقت سندن عَـلَيْـهِ الـسَّلَام جي زبان مبارڪ تي هو حَسْبِیَ اﷲُ وَ نِعْمَ الْوَکِیْلُ (يعني الله عَزَّوَجَلَّ مونکي ڪافي آهي ۽ ڪيترو ئي بهتر ڪارساز) جبريلِ امين عَـلَيْـهِ الـسَّلَام پاڻ عَـلَيْـهِ الـسَّلَام کي عرض ڪيو ته ڇا ڪو ڪم آهي؟ پاڻ عَـلَيْـهِ الـسَّلَام جن فرمايو تون سان نه، جبريل عَـلَيْـهِ الـسَّلَام عرض ڪيو: ته پنهنجي رب کان سُوال ڪريو! فرمايو: سُوال ڪرڻ سان ان جو منهنجي حال کي ڄاڻڻ مون کي ڪفايت ڪري ٿو. (خزائن العرفان،سيپارو:۱۷، الانبيا، تحت الآية:۶۸) تڏهن الله عَزَّوَجَلَّ ان باھ کي حڪم فرمايو ﴿ یٰنَارُ نَارُ کُوۡنِیۡ بَرۡدًا وَّ سَلٰمًا عَلٰۤی اِبۡرٰہِیۡمَ ﴿ۙ۶۹﴾ ﴾ ترجمو ڪنزالايمان: اي باھ ٿي وڃ ٿڌي ۽ سلامتي واري ابراهيم تي. ته باھ سواءِ توهان جي بندش (رسين وغيره) جي ٻيو ڪجھ نه ساڙيو ۽ باھ جي گرمي ختم ٿي وئي ۽ روشني باقي رهي. (خزائن العرفان،سيپارو:۱۷، الانبيا، تحت الآية:۶۹)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

آزمائش تي صبر انبياء جو طريقو آهي

مٺا مٺا اسلامي ڀائرو! توهان ڏٺو ته! حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جن پنهنجي قوم کي دينِ حَق جي دعوت ڏيڻ ۾ ڪهڙي همت، استقامت، حوصلي ۽ صبر کان ڪم ورتو. پهرين سندن عَـلَيْـهِ الـسَّلَام جا گھروارا سندن جا دشمن ٿيا، پوءِ سندن عَـلَيْـهِ الـسَّلَام قوم به دشمن ٿي وئي، ان جي باوجود پاڻ عَـلَيْـهِ الـسَّلَام دين جي دعوت ڏيڻ تَرڪ نه فرمائي، بلڪه انهن جي اصلاح جي ڪوشش جاري رکي. ايستائين جو انهن پاڻ عَـلَيْـهِ الـسَّلَام کي زندھ باھ ۾ ساڙڻ جو فيصلو ڪري ورتو، تڏهن به پاڻ عَـلَيْـهِ الـسَّلَام انهن جي اڳيان جھڪڻ گوارا نه ڪيو ۽ صبر جي انتها ته ڏسو! ان وقت حضرتِ سَيِّدُنا جبريلِ امين عَـلَيْـهِ الـسَّلَام حاضر ٿي عرض به ڪيو ته ڪا حاجت هجي ته فرمايو، پوري ڪري ڇڏيان؟ پر قربان وڃو! حضرت سَيِّدُنا ابراهيم خليلُ الله عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام تي جو ان وقت به صبر ۽ رضا جا پيڪر رهيا. ان واقعي مان انهن اسلامي ڀائرن کي به درس حاصل ڪرڻ گھرجي جيڪي راھِ خدا ۾ نيڪي جي دعوت ڏيندي پيش اچڻ واري مصيبتن تي شڪوه ڪندي نظر ايندا آهن. ياد رکو! دين جي تبليغ ڪرڻ انبياءِ ڪرام عَـلَيْـهِمُ الـسَّلَام جي سُنّتِ ڪريمه آهي ۽ ان راھ ۾ ملڻ وارين تڪليفن تي صَبۡر ڪرڻ به انهن مقدّس هستين جو طريقو آهي. الله عَزَّوَجَلَّ اَنۡبياء ۽ مرسلين عَـلَيْـهِمُ الـصَّلٰوةِ وَالـتَّسۡلِیۡم ۽ اولياءِ ڪاملين رَحِمَهُمُ اللـهُ الْـمُبین جي آزمائش فرمائيندو آهي ۽ انهن مصيبتن ۽ مشڪلاتن تي صبر ڪرڻ تي اهو رحيم ۽ ڪريم ربّ عَزَّوَجَلَّ پنهنجي رحمت سان انهن کي بيشمار اَجۡر ۽ ثواب ۽ بلند دَرَجات سان نوازيندو آهي. اسان کي به گھرجي ته جڏهن به نيڪي جي دعوت ڏيڻ وقت ڪٿي مشڪلات ۽

مصيبتن کي منهن ڏيڻو پوي ته رضاءِ الٰهي جي خاطر ان کي برداشت ڪندي صبر کان ڪم وٺون. ا ِنْ شَآءَ الله عَزَّوَجَلَّ ان جي برڪت سان اجر ۽ ثواب جو خزانو اسان جي هٿ ايندو. انهيءَ جي مُتعلق ٻه فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌو ۽ عمل جو جذبو پيدا ڪريو.

(1) جنهن مصيبت تي صبر ڪيو ايستائين جو ان (مصيبت) کي چڱي صبر سان موٽائي ڇڏيو ته الله تبارڪ و تعالٰي ان جي لاءِ ٽي سؤ (300) درجات لکندو، هر هڪ درجي جي درميان زمين ۽ آسمان فاصلو هوندو.

(اَ لْجَامِعُ الصَّغير لِلسّيوْطِي ص۳۱۷ حديث ۵۱۳۷)

(2) الله عَزَّوَجَلَّ ٻانهي کي تڪليف ۾ مبتلا رکندو آهي ايستائين جو اها تڪليف ان جا سمورا گناھ مٽائي ڇڏيندي آهي.

(المستدرڪ،ڪتاب الجنائز،باب المريضي ڪتب لہ من۔۔۔۔۔۔ الخ، الحديث: ۱۳۲۶، ج۱،ص ۶۶۸)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

عشق جو امتحان

حضرت سَيِّدُنا اسماعيل عَلَيْهِ السَّلَام حضرت سَيِّدُنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي زوجهءِ محترمه حضرت بي بي هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا جي بَطَنِ مبارڪ مان پيدا ٿيا. انهن جي پيدائش کان پوءِ پاڻ (حضرت سَيِّدنا اِبراهيم) عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام تي وحي نازل ٿي ته اوهان حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا ۽حضرت اسماعيل عَلَيْهِ السَّلَام کي اهڙي هنڌ ڇڏي اچو جتي برپٽ ميدان ۽ خشڪ جبلن کان علاوه ٻيو ڪجھ نه هجي. حضرت ابراهيم عَلَيْهِ السَّلَام حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا ۽ اسماعيل کي گڏ کڻي سفر فرمايو ۽ ان هنڌ آيا جتي ڪعبهءِ مُعَظمه آهي. اتي ان وقت

نه ڪا آبادي هئي نه ڪو چشمو ۽ پري پري تائين ڪنهن پاڻي يا ماڻهو جو نالو نشان تائين نه هو. حضرت ابراهيم عَلَيْهِ السَّلَام اتي هڪ ٿانوَ ۾ ڪجھ کجورون ۽ هڪ مشڪيزي ۾ پاڻي رکي روانا ٿي ويا. حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا فرياد ڪئي: اي الله عَزَّوَجَلَّ جا نبي هن بَرپَٽ بيابان ۾جتي نه ڪو مددگار نه ڪو غمخوار، اوهان اسان کي بي يارومددگار ڇڏي ڪيڏانهن وڃي رهيا آهيو؟ ڪيئي ڀيرا حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا اوهان کي پڪاريو پر اوهان ڪو جواب نه ڏنو، آخر ۾ حضرت بيبي هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا سُوال ڪيو ته اوهان ايترو ته ٻڌايو ته اوهان پنهنجي مرضي سان ڇڏي پيا وڃو يا خُداوَندِ قُدّوس جي حڪم سان اوهان ائين ڪيو آهي، ته پاڻ عَلَيْهِ السَّلَام فرمايائون: اي هاجره! مون جيڪو ڪجھ ڪيو آهي اهو الله تعاليٰ جي حڪم سان ڪيو آهي. اهو ٻڌي حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا چيو ته پوءِ اوهان وڃو، مون کي ڪامل يقين آهي ۽ پورو پورو اطمينان آهي ته الله ڪريم عَزَّوَجَلَّ مون کي ۽ منهنجي پٽ کي ضايع نه فرمائيندو۔ ڪجھ ڏينهن کان پوءِ کجور ۽ پاڻي ختم ٿيڻ تي حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا تي بُک ۽ اُڃ جو غَلَبو ٿيو ۽ انهن جي سيني ۾ کير به سُڪي ويو ۽ ٻار بُک ۽ اڃ جي ڪري تڙپڻ لڳو، حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا پاڻي جي ڳولها ۽ جُستجو ۾ صفا مروه جي ٻنهي جبلن جا ست (7) چڪر لڳايا پر پري پري تائين پاڻي جو پتو نه مليو، حضرت اسماعيل عَلَيْهِ السَّلَام اڃ جي شدّت جي ڪري کڙيون هڻي ڪري روئي رهيا هئا، حضرت جبرئيل عَلَيْهِ السَّلَام اوهان جي کڙين وٽ زمين تي پنهنجو پير هڻي چشمو جاري ڪري ڇڏيو. هڪ روايت جي مطابق حضرت اسماعيل عَلَيْهِ السَّلَام جي کُڙي سان جاري ٿيو (مراٰة المناجيح ،۴/۶۷) ان پاڻي ۾ کير جهڙي خاصيت هئي جو اهو پاڻي ۽ غذا ٻنهي جو ڪم ڏيندو

هو، اهو زم زم جو پاڻي پي ڪري حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا ۽ حضرت اسماعيل عَلَيْهِ السَّلَام زنده رهيا، ايستائين جو حضرت اسماعيل عَلَيْهِ السَّلَام جوان ٿي ويا ۽ شڪار ڪرڻ لڳا ۽ شڪار جي گوشت ۽ زم زم جي پاڻي تي گذر بسر ٿيندو رهيو. پوءِ جُرهم قبيلي جا ڪجھ ماڻهو پنهنجون ٻڪريون چاريندي هن ميدان ۾ آيا ۽ پاڻي جو چشمو ڏسي حضرت هاجره رَضِیَ اللـهُ تَعَالٰی عَـنْهَا جي اجازت سان هتي آباد ٿي ويا ۽ ان قبيلي جي هڪ ڇوڪريءَ سان حضرت اسماعيل عَلَيْهِ السَّلَام جي شادي به ٿي وئي ۽ آهستي آهستي هتي هڪ آبادي ٿي وئي. (عجائب القرآن مع غرائب القرآن، ص۱۴۵، ملخصاً)

اطاعت گذار هجي ته اهڙو!

مٺا مٺا اسلامي ڀائرو! ان واقعي کان اسان کي اهو معلوم ٿيو ته حضرت سَيِّدُنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام پنهنجي رب تعالٰي جا تمام ئي اطاعت گذار ۽ فرمانبردار هئا جو (توهان رضاءِ الٰهي جي خاطر پنهنجو) اهو ٻار جيڪو وڏي دعائن کان بعد پوڙهائپ ۾ مليوهو، جيڪو سندن اکين جو نُور ۽ دل جو سُرور هو، فطري طور تي حضرت سَيِّدنا اِبراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ان کي پنهنجو پاڻ کان ڪڏهن جدا نٿي ڪري سگھيا، پر الله عَزَّوَجَلَّ جي رضا جي خاطر پنهنجي پياري فرزند ۽ گھرواريءَ کي واديءِ بَطۡحا جي ان ويران جڳھ تي ڇڏي آيا جتي مٿي ڍڪڻ جي لاءِ وڻ جو پتو ۽ اُڃ ختم ڪرڻ جي لاءِ پاڻي جو هڪ قطرو به نه هو، نه اُتي ڪو يار ۽ مددگار، نه ڪو دوست ۽ غمخوار. اسان مان ڪو هجي ها ته شايد ان جي تصور سان ئي ان جي ڇاتيءَ ۾ دل ڌڙڪڻ لڳي ها، بلڪه غم جي شدّت کان دل ڦاٽي پوي ها. پر حضرت اِبراهيم خليلُ الله عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام الله عَزَّوَجَلَّ جو هي حڪم ٻُڌي نه

فڪر مند ٿيا، نه هڪ لمحي جي لاءِ سوچ ويچار ۾ پيا، نه رنج ۽ غم کان بيحال ٿيا بلڪه فورًا ئي پنهنجي ربّ عَزَّوَجَلَّ جو حڪم بجا آڻڻ جي لاءِ زال ۽ ٻار کي وٺي ڪري ملڪِ شام کان مڪي جي سرزمين ڏانهن هلي ويا ۽ اُتي زال ۽ ٻار کي ڇڏي شام ملڪ واپس هلي آيا. اللهُ اَڪبَر! ان اطاعت جي جذبي ۽ فرمانبرداري جي جوش تي اسان جي جان قربان!

اي ڪاش! اسان به رضاءِ الٰهي جي خاطر اسلام جي تعليمات کي عام ڪرڻ، سُنّتون سکڻ ۽ ماڻهن ۾ نيڪي جي دعوت کي عام ڪرڻ جي لاءِ پاڻ به ۽ پنهنجي جان جي ٽڪرن، اکين جي تارن کي پاڻ کان جُدا ڪري ڪيئي سالن يا مهينن جي لاءِ نه بلڪه مهيني ۾ ٽن (3) ڏينهن جي لاءِ راھِ خدا ۾ عاشقانِ رسول سان گڏ مدني قافلن ۾ سفر تي روانو ڪرڻ وارا بڻجي وڃون. هفتيوار سُنتن ڀرئي اجتماع ۾ پنهنجي سمجھدار ۽ وڏن ٻارن سان گڏ پاڻ به شرڪت ڪرڻ اسان جو معمول بڻجي وڃي. ڏاڙهي شريف، عِمامو شريف ۽ ٻين سُنّتن تي عمل ڪرڻ ۽ پنهنجي ٻارن کي به سُنّتن تي عمل ڪرڻ جي ترغيب ڏيڻ جي سعادت نصيب ٿي وڃي.

سُنَّتوں کی کروں خُوب خِدْمت

ہر کسی کو دُوں نیکی کی دعوت

نیک میں بھی بنوں اِلْتِجا ہے

یاخُدا تجھ سے میری دُعا ہے

(وسائلِ بخشش، ص۱۳۹)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

راضي برضاءِ الٰهي

مٺا مٺا اسلامي ڀائرو! حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي پوري زندگي راھِ خدا عَزَّوَجَلَّ ۾ قربانيون ڏيندي گذري. جڏهن به حڪمِ

خداوندي آيو ته پاڻ ان جي تعميل ڪندي ڪڏهن گھروارن کي شرڪ کان روڪيو ته ڪڏهن قوم کي نيڪي جي دعوت ڏئي انهن جي مخالفت کي منهن ڏنو، ڪڏهن وقت جي بادشاھ کي توحيد ۽ رسالت جو پيغامِ حَق ٻُڌايو، ڪڏهن پنهنجي قوم جي هٿان باهه ۾ سڙڻو پيو، وڏي چاهتن ۽ مُرادن کان پوءِ پوڙهائپ ۾ حاصل ٿيڻ واري کير پِياڪ پُٽ کي ان جي والده سان گڏ ويران بَرپَٽ ۽ ان ميدان ۾ جتي ڪو ڇانو لاءِ وڻ نه هو، رب عَزَّوَجَلَّ جي حڪم سان بغير ڪنهن دُنيوي سهاري جي اڪيلو ڇڏي ڏنو، ته ڪڏهن انهيءَ پُٽ جي نرم ۽ نازڪ ڪنڌ تي پنهنجي هٿن سان ڇُري هلائڻ کان به پوئتي نه هٽيا. الغرض پاڻ عَـلَيْـهِ الـسَّـلَام جي پوري زندگي اطاعتِ الٰهي ۾ گذرندي نظر اچي ٿي. پاڻ عَـلَيْـهِ الـسَّـلَام جي حياتي جي داستان جو هڪ هڪ گوشو رضاءِ الٰهي تي راضي رهڻ جو درس ڏئي ٿو. پاڻ عَـلَيْـهِ الـسَّـلَام جو هر حال ۾ راضي برضاءِ الٰهي رهڻ جو جذبو صد ڪروڙ مرحبا! ڪاش! جو اسان به پاڻ عَـلَيْـهِ الـسَّـلَام وانگر پنهنجي مولٰي عَزَّوَجَلَّ جي رضا ۾ راضي رهڻ وارا بڻجي وڃون، ڪاش! جو اسان به اطاعتِ الٰهي سان مَعۡمور زندگي گذارڻ وارا بڻجي وڃون.

حضرت سَيّدُنا ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام راھِ خدا ۾ پنهنجي اولاد تائين قربان ڪرڻ لاءِ تيار ٿي ويا، اي ڪاش! اسان پنهنجو ڪجھ وقت ئي راھِ خدا ۾ گذارڻ وارا بڻجي وڃون. خوُب خُوب مدني قافلن ۾سفر ڪرڻ سان گڏوگڏ مدني انعامات تي عمل ڪرڻ وارا ۽ مدني ڪمن جون ڌومون مچائڻ وارا بڻجي وڃون.

حضرت سَيّدُنا ابراهيم عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي حياتِ طيّبه مان اسان کي هي به درس ملي ٿو ته اسان تي ڪيتري ئي وڏي مصيبت ڇو نه اچي وڃي

۽ ڪهڙي به وڏي آزمائش ۾ ڇو نه مبتلا ٿي وڃون، پر پنهنجي ربّ عَزَّوَجَلَّ جي رضا تي راضي رهندي صبر ۽ شڪر سان گڏ ان وقت کي گذارڻ گھرجي. بالفرض! جيڪڏهن اسان ڪنهن مصيبت جو شڪار ٿي وڃون، تڏهن به واويلا مچائڻ جي بجاءِ پاڻ کي ملڻ واري عظيم ثواب جي طرف نظر ڪرڻ گھرجي. ڪيتريون ئي نعمتون اهڙيون آهن جيڪي بغير گھُري عطا ٿيون آهن ۽ انهن جو سلسلو وڌندو ئي ٿو وڃي. يقينًا خداءِ اَحْکَمُ الْحَاکِمِیْن عَزَّوَجَلَّ جون بيشمار نعمتون پوري ڪائنات جي ذرّي ذرّي تي مينهن جي قطرن، وڻن جي پتن، سمنڊ جي قطرن ۽ واريءَ جي ذرّن کان به وڌيڪ هر لمحي، هر گهڙي بغير گھُري طوفاني بارش جيان تيز تر وسنديون رهنديون آهن. جنهن کي شمار ڪرڻ جو تصور به نٿو ڪري سگھجي ۽ ان جو اعلان الله عَزَّوَجَلَّ خود پنهنجي پياري ڪلام ۾ فرمائي رهيو آهي، جيئن ته ارشاد ٿئي ٿو:

وَ اِنۡ تَعُدُّوۡا نِعۡمَتَ اللہِ لَا تُحۡصُوۡہَا ؕ

(سيپارو: 13، ابراهيم: 34)

ترجمو ڪنزالايمان: ۽ جيڪڏهن الله جون نعمتون شمار ڪريو ته ڳڻي نه سگھندؤ.

مٺا مٺا اسلامي ڀائرو! يقينًا ان ۾ ڪو شڪ ناهي ته ٻانهو ڪنهن به وقت، ڪنهن به لمحي، ڪنهن به حالت ۾ پنهنجي خالق ۽ مالڪ عَزَّوَجَلَّ جي بيحد ۽ لاتعداد نعمتن کان لاتعلق نه ٿو ٿي سگھي. تنهنڪري جيڪڏهن ڪا آزمائش جي گھڙي مقدر ۾ اچي وڃي، يا ڪا نعمت ختم ٿي وڃي ، تڏهن به الله عَزَّوَجَلَّ جون ٻيون لاتعداد نعمتون ته موجود هونديون آهن. تنهنڪري عقلمند اهو ئي آهي جيڪو انهن گھڙين ۾ صبر جو دامن نه ڇڏي ۽ رضاءِ الٰهي جي خاطر انهن کي برداشت ڪري، اِنْ شَآءَالله عَزَّوَجَلَّ ان جي برڪت سان اجر ۽ ثواب جي عظيم خزانو اسان جي هٿ ايندو. جيئن ته ٻه فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌو:

* مسلمان کي جيڪا به تڪليف، اذيّت، انديشو، غم ۽ ڏُک پهچي ايستائين جو جيڪڏهن ان کي ڪنڊو به لڳي ته الله عَزَّوَجَلَّ ان تڪليفن جي سبب ان جا گناھ مٽائي ڇڏيندو آهي.

(صحيح البخاري، ڪتاب المرضي ،باب ڪفارة المرض، الحديث ۵۶۴۰،ج،۴،ص،۳)

* قيامت جي ڏينهن جڏهن مُنادي (سڏڻ وارو) نِدا (سڏ) ڪندو: ڪير آهي جنهن جو الله عَزَّوَجَلَّ تي قرض آهي؟ ته مخلوق چوندي: اهڙو ڪير آهي جنهن جو قرض الله عَزَّوَجَلَّ تي هجي؟ فِرشتا چوندا: اهو جنهن کي (دنيا ۾) اهڙي مصيبت ۾ مبتلا ڪيو ويو جنهن سان ان جي دل غمگين ٿي، اکين مان لُڙڪ وهيا پر ان ثواب جي اُميد تي الله عَزَّوَجَلَّ جي رِضا جي لاءِ صَبۡر ڪيو اڄ اهو اُٿي بيهي ۽ الله عَزَّوَجَلَّ کان پنهنجو اَجۡر حاصل ڪري.

(نيکيوں کی جزائیں اور گناہوں کی سزائیں، ص۶۳)

قرباني جي اهميت

مٺا مٺا اسلامي ڀائرو! اسان کي گھرجي ته اسان پنهنجي اندر صبر ۽ شڪر جهڙيون عادتون پيدا ڪرڻ جي لاءِ ڪوشش ڪريون ۽ ان ساراهيل خوبين کي پنهنجي ذات تي نافذ ڪرڻ جو هڪ بهترين ذريعو هي به آهي ته اسان پنهنجي بزرگن جي سيرت ۽ ڪردار جو مطالعو ڪريون. جڏهن انبياءِ ڪرام عَـلَـيْهِمُ السَّلَام ۽ اولياءِ عظام رَحِمَهُمُ اللـهُ السَّلَام جي واقعن کي پڙهنداسين ته اِنْ شَآءَالله عَزَّوَجَلَّ انهن جي ڪردارن جي مدني خوشبو سان اسان جي زندگي به مهڪڻ لڳندي. اڄ اسان حضرت سَيّدُنا ابراهيم عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي متعلق ٻڌو. پاڻ عَـلَيْـهِ الـسَّـلَام پوري زندگي الله عَزَّوَجَلَّ جي اطاعت ۽ فرمانبرداري ۾

گذاري، حڪمِ خداوندي ۾ پنهنجو تَن، مَن ۽ ڌَن قربان ڪرڻ جو عملي مظاهرو ڪيو. ايستائين جو چنڊ جهڙو پُٽ به راھِ خدا ۾ قربان ڪرڻ کان نه ڪيٻايو. الله عَزَّوَجَلَّ کي سندن عَـلَيْـهِ الـسَّـلَام جي هي ادا ايتري پسند آئي جو تمام اُمّتِ مسلمه کي حڪم فرمائي ڇڏيو ته توهان به منهنجي خليل (عَـلَيْـهِ الـسَّـلَام) جي ان ادا تي عمل ڪندي جانور ذَبۡح ڪندا ڪريو. جيئن ته هر بالغ، مقيم، مسلمان مرد ۽ عورت، مالڪِ نصاب تي قرباني واجب آهي. (عالمگيري ج٥ص٢٩٢، ملتقطاً) مالڪِ نصاب هئڻ مان هي مراد آهي ته ان شخص وٽ ساڍا ٻاونجاھ (52.5) تولا چاندي يا ان جيتري ماليت جي رقم يا ان جيتري ماليت جو تجارت جو مال يا ان جيتري ماليت جو (حاجاتِ اَصۡليّه کان علاوه) سامان هجي ۽ ان تي الله عَزَّوَجَلَّ يا ٻانهن جو ايترو قرض نه هجي جنهن کي ادا ڪرڻ کان بعد بيان ڪيل نصاب باقي نه رهي. (عالمگيري، ۱/۱۸۷، ملخصاً) فُقهاءِ ڪرام رَحِمَهُمُ اللـهُ السَّلَام فرمائن ٿا: حاجتِ اَصۡلِيّه (يعني ضرورياتِ زندگي) مان مراد اهي شيون آهن جن جي عام طور تي انسان کي ضرورت هوندي آهي ۽ ان جي بغير وقت گذرڻ ۾ شديد تنگي ۽ ڏکيائي محسوس ٿيندي آهي جيئن رهڻ جي لاءِ گھر، پائڻ جي لاءِ ڪپڙا، سُواري، علمِ دين جي متعلق ڪتاب ۽ پيشي جي متعلق اوزار وغيره. (عالمگيري، ۱/۱۷۲، ملخصاً)

ڪيترين ئي احاديثِ مبارڪه ۾ قرباني جا فضائل وارد ٿيا آهن. اچو ته انهن مان ٻه (2) فرمانِ مصطفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ٻڌون ٿا.

حضرتِ سَيّدُنا زيد بن اَرۡقم رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: صحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان عرض ڪيو، يَا رَسولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هي قربانيون ڇا آهن؟ پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن ارشاد فرمايو: توهان جي پيءُ اِبراهيم عَـلَيْـهِ الـسَّـلَام جي

1) سُنّت آهن. صحابهء ڪرام عَـلَيْهِمُ الرِّضْوَان جن عرض ڪيو: يَا رَسولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هن ۾ اسان جي لاءِ ڇا ثواب آهي؟ فرمايو: هر وار جي بدلي هڪ نيڪي آهي. عرض ڪيو: ۽ اُن ۾؟ فرمايو: ان جي هر وار جي بدلي به هڪ نيڪي آهي.

(ابن ماجه ،ڪتاب الاضا حي، باب ثواب الاضحية، رقم ۳۱۲۷ ،ج۳ ،ص ۵۳۱)

2) هڪ ٻي حديثِ پاڪ ۾ آهي: اي انسانو! خوش دلي سان قرباني ڪندا ڪريو ۽ اُن جي رت تي رضاءِ الٰهي عَزَّوَجَلَّ ۽ اَجۡر جي اُميد رکو چاهي اها زمين تي ڪري چڪي هجي، ڇو جو اها الله عَزَّوَجَلَّ جي حفاظت ۾ ڪرندي آهي. (العجم الاوسط ، الحديث: ۸۳۱۹، ج۶، ص ۱۴۸)

مٺا مٺا اسلامي ڀائرو! ياد رکو! جنهن شخص تي شَرۡعي اُصولن جي بنياد تي قرباني واجب آهي ان تي قرباني جي متعلق مسئلا سکڻ به لازم آهن. في زمانو علمِ دين کان دُوري جي سبب مسلمان ان اهم فريضي کي به جهڙو ان جو حق آهي، ادا نه ٿا ڪن، اسان جي معاشري جو هڪ تعداد اهڙو آهي جنهن جي ديني معلومات جو حال هي آهي ته هڪ ئي گهر ۾ رهڻ وارن مختلف مالڪِ نصاب فرد پوري گهر جي طرفان هڪ ئي ٻَڪرو قربان ڪري ڇڏيندا آهن ۽ قرباني جي لاءِ جانور ڪهڙو هئڻ گھرجي؟ يا ڪهڙي عيب جي سبب قرباني نه ٿيندي؟ انهن کي معلوم ئي نه هوندو آهي ۽ اهي قرباني ڪرڻ کان بعد ان خوش فهمي ۾ مبتلا رهندا آهن ته اسان به سُنّتِ ابراهيمي تي عمل ڪري ورتو ۽ اسان جو واجب ادا ٿي ويو. جڏهن ته انهن جي ذمّي واجب جي ادائيگي باقي رهندي آهي. ان لاءِ قرباني جي احڪامات جو علم هئڻ ضروري آهي. قرباني ۾ پيش اچڻ وارن مسئلن جي متعلق شيخِ طريقت،اميرِ اَهلسُنَّت،بانيِ دعوتِ اسلامي حضرت علامه مولانا ابُو بلال محمدالياس عطّارقادري رَضَوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جو 48 صفحن تي

مشتمل رسالي ” ابلق گھوڙي سُوار “ جو مطالعو بيحد مُفيد آهي. هر اسلامي ڀاءُ کي گھرجي ته قرباني ڪرڻ کان پهريان گهٽ ۾ گهٽ هڪ ڀيرو ضرور هن رسالي جو مطالعو فرمائي وٺي. اِنْ شَآءَالله عَزَّوَجَلَّ علمِ دين جو عظيم خزانو هٿ ايندو ۽ قرباني جي ڪيترن ئي مسئلن کان به آگاهي نصيب ٿيندي. قرباني جي متعلق مزيد معلومات جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب ” بهارِ شريعت جلد 3 حصو 15 “ مان قرباني جي مسائل جو مطالعو ڪريو يا دارُ الافتاء اهلسُنّت سان رجوع ڪريو.

مٺا مٺا اسلامي ڀائرو! اڄ اسان حضرت سَيّدُنا ابراهيم عَـلَيْـهِ الـسَّلَام جي قربانين جي باري ۾ ٻڌو ۽ ضمنًا ان جي متعلق مدني گُل به حاصل ڪيا. پاڻ عَـلَيْـهِ الـسَّلَام جن پنهنجي پوري زندگي اطاعتِ خداوندي ۽ رضاءِ الٰهي ۾ گذاري ڇڏي . وقت بوقت مصيبتن ۽ مشڪلاتن کي منهن ڏيڻو پيو پر پاڻ صبر ۽ شڪر ۽ رضاءِ الٰهي تي راضي رهڻ کي ترجيح ڏني ۽ ڪڏهن به پنهنجي زبان تي شڪايت جو جملو نه آندو. اسان کي به گھرجي ته پنهنجي زندگي کي اطاعتِ الٰهي ۾ گذاريون، هر معاملي ۾ نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سُنّتن جي پيروي ڪندا رهون. آزمائشن تي صبر ۽ نعمتون ملڻ تي شڪرِ الٰهي بجا آڻيون ته اِنْ شَآءَالله عَزَّوَجَلَّ ٻنهي جهانن ۾ سعادتون اسان جو مقدر ٿينديون.

مجلسِ مزاراتِ اولياءَ

مٺا مٺا اسلامي ڀائرو! اَلۡحَمۡدُلِلّٰهِ عَزَّوَجَلَّ دعوتِ اسلامي سڄي دنيا ۾ نيڪي جي دعوت عام ڪرڻ، سنّتن جي خوشبو لٽائڻ ۽ علمِ دين جي روشني ڦهلائڻ ۾ مصروف آهي، دنيا جي تقريب ً ا 192 مُلڪن ۾ هن جو مدني پيغام پهچي چڪو آهي، سڄي دنيا ۾ مدني ڪم کي مُنظَّم ڪرڻ جي لاءِ تقريب ً ا 95 کان وڌيڪ شعبا قائم آهن، اِنهن مان هڪ شعبو ” مجلسِ مزاراتِ اولياءَ “ به آهي. هِن مجلس جا ذميدار اسلامي ڀائر ٻين مدني ڪمن سان گڏوگڏ بُزُرگانِ دين جي مبارڪ مزارن تي حاضر ٿي مختلف ديني خدمتون سر انجام ڏيندا آهن، مثل ً ا: جيتريقدر ٿي سگهي صاحبِ مزار جي عُرس جي موقعي تي اجتماعِ ذڪر و

نعت جو انعقاد، مزارن سان مليل مسجدن ۾ عاشقانِ رسول جا مدني قافلا سفر ڪرائڻ ۽ بالخصوص عُرس شريف جي موقعي تي مزار جي چؤديواري ۾ سنّتن ڀريا مدني حلقه لڳائڻ جن ۾ وُضُو، غسل، تيمُّم، نماز ۽ ايصالِ ثواب جو طريقو، مزارن تي حاضري جا آداب ۽ سرڪارِمدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جون سُنّتون سيکاريون وينديون آهن ۽ گڏوگڏ دعوتِ اسلامي جي هفتيوار سنّتن ڀري اجتماعن ۾ شرڪت، مدني قافلن ۾ سفر ۽ مدني انعامات تي عمل ڪرڻ جي ترغيب ڏياري ويندي آهي. عُرس جي ڏينهن ۾ صاحبِ مزار جي خدمت ۾ گھڻي کان گھڻا ايصالِ ثواب جا تحفا پيش ڪرڻ سان گڏ صاحبِ مزار بُزُرگ جي سجَّاده نشين، خليفن ۽ مزارن جي متوَلّي صاحبان سان وقت بوقت ملاقات ڪري اُنهن کي دعوتِ اسلامي جي خدمتن، جامعة المدينه ۽ مدارسُ المدينه ۽ مُلڪ ۽ بيرونِ مُلڪن ۾ ٿيڻ وارن مدني ڪمن کان واقف ڪرائڻ وغيره. الله عَزَّوَجَلَّ دعوتِ اسلامي کي ترقي ۽ عُروج عطا فرمائي. اٰمِــيْن بِـجَـاهِ الـنَّـبِـیِّ الْاَمِــيْن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

12 مدني ڪمن ۾ حصو وٺو!

مٺا مٺا اسلامي ڀائرو! توهان به سُنّتن جي خدمت جي لاءِ دعوتِ اسلامي جي مدني ڪمن ۾ اڳتي وڌي حصو وٺو. ذيلي حلقي جا ٻارنهن مدني ڪم مسلمانن کي سُنّت جي راھ تي هلائڻ جي لاءِ وڏا مددگار آهن. هن 12 مدني ڪمن مان هڪ مدني ڪم ” مدرسةُ المدينه بالِغان “ ۾ پڙهڻ

پڙهائڻ به آهي. قرآن پاڪ الله عَزَّوَجَلَّ جو مبارڪ ڪلام آهي، ان جو پڙهڻ، پڙهائڻ ۽ ٻڌڻ ٻڌائڻ سڀ ثواب جو ڪم آهي. پر ثواب ان وقت ملندو جڏهن دُرُست تَلَفّظ سان پڙهيو وڃي نه ته گھڻو ڪري ثواب جي بجاءِ ٻانهو عذاب جو مستحق بڻجي ويندو آهي. منهنجا آقا اعليٰ حضرت، امام اهلسنّت، مُجَدّد دين ۽ ملت، پروانهء شمع رسالت، مولانا شاهه امام احمد رضا خان عَلَيْهِ رَحْمَةُ الرَّحْمٰن فرمائن ٿا: ”بيشڪ ايتري تجويد جنهن سان تَصحيحِ (تَصۡ. حِي. حِ) حُروف ٿئي (يعني تجويد جي قاعدن جي مطابق حُروف کي دُرُست مخارج سان ادا ڪري سگھي) ۽ غلط خواني (يعني غلط پڙهڻ) کان بچي، فرضِ عين آهي.“ (فتاويٰ رضويہ مُخَرَّجَہ ج۶ ص ۳۴۳) قرآنِ ڪريم پڙهڻ ۽ پڙهائڻ ڪيتري قدر فضيلت وارو آهي. جيئن ته نبيِّ مُڪرّم، نُورِ مُجسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ مُعظَّم آهي ” خَیْرُکُمْ مَنْ تَعَلَّمَ الْـقُرْاٰنَ وَعَلَّمَهٗ “ يعني توهان مان بهترين شخص اهو آهي جيڪو قرآنِ پاڪ سکي ۽ سيکاري. (بخاري، ڪتاب فضائل القرآن، باب خيرکم من تعلم القرآن...الخ، ۳/۴۱۰،حديث:۵۰۲۷) حضرت سيِّدُنا ابو عبدالرحمٰن سُلَمي رَضِىَ اللهُ تَعَالٰی عَـنْهُ مسجد ۾ قرآن پاڪ پڙهائيندا هئا، ۽ فرمائيندا هئا؛ انهي حديثِ پاڪ ئي مون کي هتي ويهاريو آهي.

(فَيضُ القَدير ج 3 ص 618 تحت الحديث 3983)

مٺا مٺا اسلامي ڀائرو! جيڪڏهن اسان به دُرُست مخارج ۽ قاعدن سان گڏ قرآن ڪريم پڙهڻ جا خواهشمند آهيون ۽ پڙهڻ به گھرجي ته مدرسة المدينه (بالغان) ۾ ضرور شرڪت ڪيو. اَلۡحَمۡدُلِلهِ عَزَّوَجَلَّ مختلف جڳهن ۽ مسجدن وغيره ۾ عام طور تي نمازِ عشاء کان بعد هزارين مدرسة المدينه (بالغان) جي ترڪيب هوندي آهي، جنهن ۾ وڏي عمر جا اسلامي ڀائر صحيح مخارج سان حروف جي ادائيگي سان گڏ قرآن ڪريم سکندا ۽ دعائون ياد ڪندا آهن، نماز وغيره دُرست ڪندا ۽ سُنّتن جي مفت تعليم حاصل ڪندا آهن. ان کان علاوه دنيا جي مختلف ملڪن ۾ گھرن جي اندر تقريبًا روزانه

هزارين مدرسا مدرسة المدينه (براءِ بالغات) جي نالي سان لڳايا ويندا آهن، جن ۾ اسلامي ڀينرون قرآنِ پاڪ، نمازن ۽ سُنّتن جي مفت تعليم حاصل ڪنديون ۽ دعائون ياد ڪنديون آهن. اَلۡحَمۡدُلِلهِ عَزَّوَجَلَّ مدرسة المدينه (بالغان) جي بَرَڪت سان ڪيترن ئي اسلامي ڀائرن جي اصلاح به ٿيندي آهي. اچو ته انهيءَ حوالي سان هڪ ايمان افروز مدني بهار ٻڌون ٿا.

منهنجي زندگي ۾ بهار اچي وئي:

زم زم نگر، حيدرآباد (باب الاسلام، سنڌ) جي علائقي آفندي ٽاؤن جي هڪ مُقيم اسلامي ڀاءُ جي بيان جو خلاصو آهي: آئون هڪ فيشن پرست نوجوان هئس، دنيا جي موج مستي ۾ گُم، پنهنجي آخرت جي انجام کان غافل حياتي جا ڏينهن گذاري رهيو هئس ته منهنجي سُتل قسمت جاڳي پئي، مون کي مدرسة المدينه (بالغان) جون روحاني فضائون ته ڇا مُيسَر آيون منهنجي ته خوش بختي جي سفر جو آغاز ٿي ويو. مدرسة المدينه (بالغان) جي برڪتن منهنجي دل کي خوفِ خدا ۽ عشقِ مصطفيٰ جي ڏيئي سان روشن ڪري ڇڏيو. ان ۾ مون کي قرآن جي تعليم سان گڏوگڏ سُنّتن تي عمل جو جذبو به مليو ۽ هفتيوار سُنّتن ڀرئي اجتماع ۾ شرڪت جي سعادت به نصيب ٿي. اَلۡحَمۡدُلِلهِ عَزَّوَجَلَّ مدرسة المدينه (بالغان) ۾ پڙهڻ جي برڪت سان منهنجي زندگي ۾ بهار اچي وئي، فيشن پرستي ۽ موج مستي کان نجات حاصل ٿي ۽ آئون دعوتِ اسلامي جي مهڪندڙ مهڪندڙ مدني ماحول سان وابسته ٿي ويس. اَلۡحَمۡدُلِلهِ عَزَّوَجَلَّ هي تحرير لکڻ وقت مدني قافله ذميدار جي حيثيت سان مدني ڪمن جون ڌومون مچائي رهيو آهيان.

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺامٺا اسلامي ڀائرو! بيان کي ختم ڪندي سنّت جي فضيلت ۽ ڪجھ سنّتون ۽ آداب بيان ڪرڻ جي سعادت حاصل ڪيان ٿو، تاجدار رسالت، شهنشاههِ نُبُوَّت، مصطفيٰ جانِ رحمت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمان جنّت نشان آهي: جنهن منهنجي سنّت سان مَحَبّت ڪئي ان مون سان مَحَبّت ڪئي ۽ جنهن مون سان مَحَبّت ڪئي اهو جنّت ۾ مون سان گڏ هوندو۔

(مِشڪاة المَصابيح ، ج۱ ص۵۵ حديث ۱۷۵ دارالڪتب العلمية بيروت)

سینہ تری سنّت کا مدینہ بنے آقا جنت میں پڑوسی مجھے تم اپنا بنانا

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺامٺا اسلامي ڀائرو! اڪثر ڪري اسان کي سفر جي ضرورت پيش ايندي آهي، تنهنڪري اسان ڪوشش ڪري سفر جون به ڪجھ نه ڪجھ سُنّتون ۽ آداب سکي وٺون ته جيئن ان تي عمل ڪري اسان پنهنجي سفر کي ثواب حاصل ڪرڻ جو ذريعو بڻائي سگھون. * ممڪن هجي ته خميس جي ڏينهن سفر شروع ڪرڻ سُنّت آهي. (اشعة اللمعات،ج۵،ص۱۶۱) * جيڪڏهن سهولت هجي ته رات جو سفر ڪيو وڃي جو رات جو سفر جلدي طيءِ ٿيندو آهي * جيڪڏهن ڪجھ اسلامي ڀائر ملي ڪري مدني قافلي جي صورت ۾ سفر ڪن ته ڪنهن هڪ کي امير بڻائي وٺن. * روانو ٿيڻ وقت عزيزن، دوستن کان قصور معاف ڪرايو ۽ جن کان معافي طلب ڪئي وڃي ان تي لازم آهي ته دل سان معاف ڪري ڇڏين. (بہارِ شريعت،حصہ ۶،ص۱۹) * سفر جو لباس پائي جيڪڏهن مڪروھ وقت نه هجي ته گھر ۾ (4) رڪعتون نفل ” اَلْحَمْدُ وَ قُلْ “ سان پڙهي ٻاهر نڪرو، اهي رڪعتون واپسي تائين اهل ۽

مال جي نگھباني ڪنديون. * اسان جڏهن به سفر تي روانا ٿيون ته اسان کي گھرجي ته اسان پنهنجي اهل ۽ مال کي الله عَزَّوَجَلَّ جي حوالي ڪري وڃون. بلڪه ٿي سگھي ته گھروارن کي هيٺ ڏنل ڪلمات چئي سفر تي روانا ٿيون. اَسْتَوْدِعُکَ اللہَ الَّذِیْ لَایُضِیْعُ وَدَائِعَہٗ ۔ ترجمو: آئون توهان کي الله عَزَّوَجَلَّ جي حوالي ڪريان ٿو جيڪو سونپيل امانتن کي ضايع نه ڪندو آهي. (سنن ابن ماجہ، ڪتاب الجہاد، باب تشبيع الغزوة ووداعھم، الحديث، ۲۸۲۵،ج۳،ص۳۷۲) * واپار جي لاءِ سفر ڪرڻ وارن اسلامي ڀائرن کي گھرجي ته هي پنج سورتون پڙهي وٺن (۱) قُلْ یٰاَیُّھَاالْکٰفِرُوْنَ آخر تائين (۲) اِذَا جَآءَ نَصْرُ اللہِ وَالْفَتْحُ آخر تائين (۳) قُلْ ھُوَ اللہُ اَحَدٌ آخر تائين (۴) قُلْ اَعُوْذُ بِرَ بِّ الْفَلَقِ آخر تائين (۵) قُلْ اَعُوْذُ بِرَ بِّ النَّاسِ آخر تائين * جڏهن ڏاڪڻين تي چڙهو يا اونچي جڳھ جي طرف هلو، يا اسان جي بس وغيره ڪنهن اهڙي روڊ کان گذري جيڪو اونچائي جي طرف وڃي رهيو هجي ته ” اَللہُ اَکْبَرُ “ چوڻ سُنّت آهي ۽ جڏهن ڏاڪڻين کان لهڻ يا لاهي (يعني هيٺاهين) جي طرف هلڻ وقت ” سُبْحَانَ اللہِ “ چوڻ سُنّت آهي. * مسافر کي گھرجي ته اهو دعا کان غفلت نه ڪري هي جيستائين سفر ۾ آهي ان جي دُعا قبول ٿيندي آهي بلڪه جيستائين گھر نه پهچي ان وقت تائين دعا مقبول آهي. * منزل تي لهو ته وقت بوقت هي دعا پڙهو، اِنْ شَآءَالله عَزَّوَجَلَّ هر نقصان کان بچندا. دعا هي آهي: اَعُوْذُ بِکَلِمَاتِ اللہِ التَّآمَّاتِ مِنْ شَرِّمَا خَلَقَ ترجمو: الله عَزَّوَجَلَّ جي ڪلماتِ تامه جي پناھ گھران ٿو ان جي شر کان جنهن ان کي پيدا ڪيو. (ڪنز العمال ،ڪتاب السفر ، الفصل الثاني في آداب السفر، الحديث ۱۷۵۰۸،ج۶،ص۳۰۱) * سفر کان واپسي تي گھروارن جي لاءِ ڪو تحفو وٺي اچو جو هي سُنّت آهي. سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

فرمايو: جڏهن سفر کان ڪو واپس اچي ته گھروارن جي لاءِ ڪجھ نه ڪجھ تحفو آڻي، چاهي پنهنجي جھولي ۾ پٿر ئي آڻي. (ڪنز العمال ،ڪتاب السفر ،الفصل الثاني في آداب السفر ،الحديث ۱۷۵۰۲،ج۶ ،ص۳۰۱)

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي زيارت ڪندو ۽ قبر ۾

داخل ٿيندي وقت به، اهو ڏسندو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن ، ([1])

﴿ 2 ﴾ سمورا گُناھ مُعاف

اللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سيدُنا انس رَضِىَ اللهُ تَعَالٰی عَـنْهُ کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪو شَخص اهو دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا. ([2])

﴿ 3 ﴾ رَحمت جا ستّر دروازا

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

اللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي پنهنجي ۽ صِدّيقِ اڪبر رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي وچ ۾ ويهاريو، ان تي صَحابَه ڪرام رَضِیَ اللهُ تَعَالٰی عَنْهم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: اهو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته ايئن پڙهندو آهي. ([2])

﴿ 6 ﴾ دُرُود شَفَاعت

اللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([3])

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سيدُنا ابنِ عباس رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: انهي درود شريف کي پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿2﴾ هر رات عبادت ۾ گذارڻ جو آسان نسخو

غرائب القرآن صفحه 187 تي هڪ روايت نقل ڪئي وئي آهي ته جيڪو شخص رات ۾ ٽي ڀيرا هي دعا پڙهندو ته ان ڄڻ شبِ قدر کي حاصل ڪري ورتو لهٰذا هر رات هي دعا پڙهڻ گهرجي. دعا هي آهي:

لَاۤ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

“يعني خداءِ حليم ۽ ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي.” (فيضانِ سنّت ج اوّل، ص 1163، 1164)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد
