اميرُالمُؤمنين حضرت سيدنا ابوبڪر صديق رَضِىَ اللهُ تَعَالٰی عَنْهُ کي پنهنجي غريب ماسات ۽ نادار بدري صحابي حضرت سيدنا مِسطَح رَضِىَ اللهُ تَعَالٰی عَنْهُ جنهن جو خرچ پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ کڻندا هئا، ان کان سخت رنج پهتو ۽ اهو هي ته ان اوهان رَضِىَ اللهُ تَعَالٰی عَنْهُ جي پياري ڌيءَ يعني اُمُّ المُومنين حضرت سيدتنا عائشه صديقه رَضِىَ اللهُ تَعَالٰی عَنْها تي تهمت (يعني بهتان) لڳائڻ وارن سان مُوافقت ڪئي يعني انهن جو ساٿ ڏنو ، انهيءَ ڪري پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ (سيدنا مِسطح بن اُثاثه کي) خرچ نه ڏيڻ جو قسم کنيو، ان تي سيپاري 18 سورة النُّور جي آيت نمبر 22 نازل ٿي؛

جڏهن هي آيت سيد عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پڙهي ته حضرت سيِّدُنا ابُوبَڪر صِديق رَضِىَ اللهُ تَعَالٰی عَنْهُ چيو؛ بيشڪ منهنجي آرزو آهي ته الله عَزَّوَجَلَّ منهنجي مغفرت ڪري ۽ مان مِسطح (رَضِىَ اللهُ تَعَالٰی عَنْهُ) سان جيڪو سلوڪ ڪندو هيس ان کي به ڪڏهن موقوف (يعني بند) نه ڪندس، پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ ان (مالي تعاون) کي جاري رکيو، (خزائن العرفان ص ۵۶۳ از نیکی کی دعوت ۱۶۰)

مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان! اَمِیْرُ الْمُؤمِنِیْن حضرتِ سَيّدُنا صِدّيقِ اڪبر رَضِیَ اللہُ تَعَالٰی عَنْہُ پنهنجي ماسات (يعني پنهنجي ڪزن) حضرت سَيّدُنا مِسطَح بن اُثَاثَه رَضِیَ اللہُ تَعَالٰی عَنْہُ کان قطع تَعلّقي (يعني لاڳاپا ختم) ڪرڻ جو قسم کنيو هو پر جڏهن اها آيت سڳوري نازل ٿي ته پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ رضاءِ الٰهي جي سبب حضرت سَيِّدُنا مِسطَح بن اُثَاثَه رَضِیَ اللہُ تَعَالٰی عَنْہُ کي معاف ڪري ڇڏيو۔ غور ڪيو! جيڪڏهن اهڙو معاملو اسان سان ٿئي ته اسان اهڙي ماڻهو سان ڳالھ ٻولھ، ميل ميلاپ ايستائين جو سلام دعا به ڇڏي ڏيندا آهيون، بلڪ اسان ته ننڍين ننڍين ڳالهين تي مِٽ مائٽن سان لاڳاپا ختم ڪري ڇڏيندا آهيون، حسن سلوڪ يعني سٺي ورتاءُ کان پاسو ڪندا آهيون ۽ ڳالھ ٻولھ ڇڏي ڏيندا آهيون اسان سڀني کي گھرجي ته غور ڪيون ته خاندان ۾ ڪنهن ڪنهن سان اَڻبڻت آهي جڏهن پتو پئجي وڃي ته جيڪڏهن شرعي عذر نه هجي ته هڪدم ناراض رشتيدارن سان ٺاھ ۽ صلح جي ترڪيب شروع ڪيون، جيڪڏهن جُھڪِڻو پوي ته بيشڪ الله جي رضا جي لاءِ جهڪي پون، اِنْ شَآءَاللہ عَزَّ وَجَلَّ

بلندي نصيب ٿيندي ۔ فرمانِ مصطَفٰي صَلَّی اللّٰہُ تَعَالٰی عَلَیْہِ واٰلِہٖ وَسَلَّمَ آهي: مَنْ تَوَاضَعَ لِلّٰہِ رَفَعَہُ اللّٰہُ يعني جيڪو اللہ عَزَّ وَجَلَّ جي لاءِ عاجِزي ڪندو آهي اللہ عَزَّ وَجَلَّ ان کي بلندي عطا فرمائيندو آهي ۔ (شُعَبُ الْاِیمان ج۶ ص۲۷۶ حدیث۸۱۴۰) پنهنجي گھرن ۽ معاشري کي امن وارو بڻائڻ جي لاءِ پنهنجي عزيزن ۽ رشتيدارن سان سهڻو سلوڪ ۽ صلهءِ رحمي جي عادت بڻايو ۽ جيترو ممڪن ٿئي قطع تعلقي کان بچڻ جي ڪوشش ڪيون، ڇوته ٻين گناهن جو وبال ته رڳو گناھ ڪرڻ واري ٿيندو آهي پر مائٽي ٽوڙڻ جي نحوست سبب سڄي قوم رحمتِ الٰهي کان محروم ٿي ويندي آهي.

اللہ عَزَّ وَجَلَّ اسان کي رشتيدارن، يتيمن، بي سهارن سان صلهءِ رحمي ڪرڻ جو حڪم ڏنو اهي جيئن سپاري 21 سُوْرَۃُ الرُّوْم جي آيت نمبر38 ۾ ارشاد آهي:

مُفسّرِ شهير، حڪيمُ الاُمَّت مُفتي احمد يار خان رَحْمَۃُ اللہِ تَعَالٰی عَلَیْہ هن آيتِ سڳوري جي تحت فرمائن ٿا؛ اها آيت سڳوري سڀني عزيزن جي حقن کي ادا ڪرڻ جو حڪم ڏئي رهي آهي، ان مان معلوم ٿيو ته هر عزيز جو حق آهي، ان ۾ سڀئي عزيز شامل آهن، هن آيت مان اهو به معلوم ٿيو ته عزيزن رشتيدارن سان سهڻو ورتاء ۽ صدقو خيرات، نالي ڪڍائڻ، رسم و رواج

’’ بہشت کی کُنجیاں ‘‘ نهايت ايمان افروز ڪتاب آهي، هن ڪتاب ۾ رنگ برنگي مدني گل انمول قيمتي خوشبو مهڪائي رهيا آهن، اچو ٻڌون ٿا؛ جنت ڇا آهي، جنت ڪٿي آهي؟ جنتون ڪيتريون آهن؟ جنت وارن جي عمر ڪيتري هوندي؟ جنت ۾ وٺي وڃڻ وار عمل ڪهڙا آهن؟ تلاوت قرآن جون برڪتون، شهادت واري درجي جو سولو طريقو، مسلمانن جي پرده پوشي جا فضائل، ماڻهن جون خطائون معاف ڪرڻ جي فضيلت ۽ ڪافي ديني معلومات سان ڀرپور اهو ڪتاب مڪتبة المدينه تان هديو ڏئي حاصل ڪري سگھجي ٿو، هن شاندار ڪتاب ۾ شَيخُ الحديث، حضرتِ علّامه مَولانا عبدُ المصطفٰي اَعظمي رَحْمَۃُ اللّٰہ ِتَعَالٰی عَلَیْہ هنن حالتن ڏانهن اشارو ڪندي فرمائن ٿا؛ هن زماني ۾ ننڍين ننڍين ڳالهين تي ماڻهو اهو چئي ڇڏيندا آهن ته مان اڄ کان تنهنجو ڀاءُ نه آهيان ۽ تو منهنجي ڀيڻ نه آهين، اهڙي طرح ڀاءُ ڀاءُ کي چوندو آهي ته مان اڄ کان تنهنجو ڀاءُ نه آهيان ۽ تون منهنجو ڀاءُ نه آهين، اهو قطع رحم يعني رشتن کي ٽوڙڻ آهي جيڪو حرام ۽ جھنّم ۾ وٺي وڃڻ وارو ڪم آهي، ان ڪري هر مسلمان کي هميشه ان جو خيال رکڻ گھرجي ته ڪنهن مائٽ کان تعلق نه ٽوڙي بلڪ هميشه ان ڪوشش ۾ لڳل رهي ته رشتيدارن سان تعلق قائم رهي ۽ ڪڏهن به رشتو نه ٽٽي، ڪي ماڻهو اهو چوندا آهن ته جيڪو مائٽ اسان سان تعلق رکندو اسان به ان سان تعلق رکنداسين ۽ جيڪو اسان سان جدا ٿيندو اسان به ان سان جدائي ڪنداسين، اهو چوڻ ۽ اهو طريقو به اسلام جي خلاف آهي (وڌيڪ فرمائن ٿا؛) رشتيدارن سان تعلق ٽوڙڻ جي هڪ ئي صورت جائز آهي اها هي ته شريعت جي معاملي ۾ تعلق ٽوڙيو وڃي مثال طور ڪو رشتيدار جيتوڻيڪ قريبي رشتيدار ئي ڇو نه هجي، جيڪڏهن اهو مرتد (يعني اسلام کان ڦري وڃي) يا گمراھ ۽ بد دين ٿي وڃي ته پوءِ ان کان تعلق ٽوڙڻ واجب آهي، يا ڪو رشتيدار ڪنهن ڪبيره گناھ ۾ گرفتار آهي ۽

مٺا مٺا اسلامي ڀائرو! رشتيدارن جي حقن جي ادائگي ۽ انهن سان صلهءِ رحمي ڪرڻ يقيناً سعادت جو سبب آهي پر اسان کي هن جو ڪامل فائدو ان صورت ۾ نصيب ٿيندو جڏهن اسان دل سان هن نيڪي کي ادا ڪيون، افسوس هاڻي رشتيداري به صرف نالي جي رهجي وئي آهي ۽ اڄڪلھ صلهءِ رحمي ڪرڻ ڄڻ ته هڪ مجبوري بڻجي وئي آهي ڪي ماڻهو ظاهري طور ته وڏا ملنسار لڳندا آهن پر انهن جو سينو مسلمانن جي ڪيني سان ڀريل هوندو آهي، ڪي نادان نفس ۽ شيطان جي ورغلائڻ تي ذاتي سببن جي ڪري يا هروڀرو رشتيدارن سان قطع تعلق ڪندا آهن ۽ دوستن تي دل کولي خرچ ڪندا آهن، پر هاءِ افسوس! پنهنجي والدين، ڀيڻ ڀائرن چاچن ڀائٽن ۽ ڀاڻيجن وغيره جي حقن کان بلڪل ئي غافل هوندا آهن، شادي وغيره جي تقريب هجي يا مبارڪ ڏينهن جي آمد، بزرگن جي ايصال ثواب جي نذر ونياز جو اهتمام هجي يا اجتماع ذڪر ونعت وغيره ڪن ماڻهن جو اهو معمول هوندو آهي ته ان ۾ صرف انهن مائٽن کي دعوت ڏيڻ جي تڪليف ڪندا آهن يا انهن جي گھرن ۾ کاڌو وغيره موڪليندا آهن جيڪي انهن کي پنهنجي تقريبن ۾ سڏيندا آهن يا جن عزيزن سان انهن جا مفاد هوندا آهن، ان جي برعڪس جيڪي مائٽ ان جي ڪم نه ايندا آهن يا ويچارا غربت ۽

مسڪيني جي ڪري ان کي پاڻ وٽ نٿا دعوت ڏين ته اهڙن کي پنهنجي دعوت ۾ سڏائڻ ته رهيو پر انهن سان سلام دعا جي حد تائين به تعلق رکڻ انهن کي نه وڻندو آهي، ائين مستحق زڪوٰة رشتيدارن کي مسلسل نظر انداز ڪيو ويندو اهي، ايستائين جو ڪن خاندان ۾ ذاتي دشمنيون انهن کي فوتگي جي موقعي تي ڪفن دفن، جنازو نماز پڙهڻ ۽ تعزيت جي معاملي ۾ رڪاوٽ بڻجي وينديون آهن، مطلب ته رشتيدارن ۾ پهريان جيان محبت، خلوص ۽ خيرخواهي جو جذبو بلڪل ئي ختم ٿيندو پيو نظر اچي ڇو ته اسان پاڻ انهن جي وچ ۾ درجه بنديون ڪيون آهن اهڙن نازڪ حالتن کي ڏسي اهو اندازو لڳائڻ مشڪل ناهي ته حقيقي صلهءِ رحمي کي هاڻي هڪ بوجھ سمجھيو ويندو آهي۔

قريبي عزيزن سان قطع تعلق ڪرڻ ۽ مشڪل ۾ انهن جي ڪم نه اچڻ جي مذمت بابت ڪجھ عبرت آموز مبارڪ حديثون ٻڌو، خوف خدا کان ڊڄو ۽ پنهنجي عزيزن سان سهڻي ورتاءُ ڪرڻ جي نيّت ڪريو.

1. اي امّت محمد (صَلَّى اللهُ تَعَالٰى عَلَيهِ وَاٰلِهٖ وَسَلَّم) قسم آهي ان ذات جو! جنهن مون کي حق سان گڏ موڪليو، الله تعاليٰ ان شخص جي صدقي کي قبول ناهي ڪندو جنهن جا عزيز ان جي ڀلائي جا محتاج هجن ۽ هو غيرن کي ڏئي، قسم آهي ان ذات جو! جنهن جي دست قدرت ۾ منهنجي جان آهي الله تعاليٰ اهڙي شخص ڏانهن قيامت جي ڏينهن رحمت جي نظر نه ڪندو۔ (مجمع الزوائد، کتاب الزکاۃ، باب الصدقۃ... الخ، ۳/۲۹۷،حدیث: ۴۶۵۲)

جيڪو شخص پنهنجي ڪنهن قريبي عزيز وٽ اچي ان جي حاجت کان علاوه واري اها شيء گھري جيڪا الله عَزَّ وَجَلَّ ان کي عطا ڪئي آهي پر

1. اهو ان تي بخل ڪري، ته الله عَزَّ وَجَلَّ جھنّم مان هڪ ازدهه نانگ ڪڍندو جنهن جو نالو شجاع هوندو اهو زبان کي هلائي رهيو هوندو ۽ ان شخص جي ڳچي جو هار بڻجي ويندو. (معجم الاوسط، ۴/۱۶۷،حدیث:۵۵۹۳)

2. جنهن گناھ جي سزا دنيا ۾ به جلدي ڏني وڃي ۽ ان جي لاءِ آخرت ۾ به عذاب جو ذخيرو رهندو، اهو بغاوت ۽ قطع رحم کي وڌيڪ ناهي.

(ترمذی ،کتاب صفۃ القیامۃ،باب:۱۲۲،حدیث:۲۵۱۹، ۴/۲۲۹)

حضرت سيِّدُنا فَقِيه اَبُوالليث سَمَرقَندِي عَلَیهِ رَحْمَةُ اﷲِ القوي ”تنبيه الغَافِلين“ ۾ نقل ڪن ٿا، حضرت سيِّدنا يحيٰ بِن سُلَيم عَلَیهِ رَحْمَةُ اﷲِ الباري فرمائن ٿا؛ مڪي پاڪ زَادَھَا اللهُ شَرَفًا وَّ تَعۡظِیۡمًا ۾ هڪ نيڪ شخص خراسان جو رهڻ وارو هو، ماڻهو ان وٽ امانتون رکرائيندا هئا۔ هڪ شخص ان وٽ ڏهه هزار اشرفيون امانت طور رکرائي پنهنجي ڪنهن ضرورت سان سفر تي هليو ويو۔ جڏهن اهو واپس موٽيو ته اهو خراساني فوت ٿي چڪو هو۔ هُن انهن جي وارثن کان پنهنجي امانت جو حال پڇيو، پر انهن لاعلمي ظاهر ڪئي۔ امانت رکرائڻ واري مڪي پاڪ جي عالمن کان پڇيو ته مون کي ڇا ڪرڻ گھرجي؟ انهن چيو؛ اسان کي اميد آهي ته اهو خراساني جنّتي هوندو توهان ائين ڪيو ته اڌ رات يا رات جي ٽئين حصي گذرڻ کان پوءِ زم زم جي کوھ وٽ وڃي ان جو نالو وٺي ان کان پڇجو، هن ٽي راتيون ائين ئي ڪيو، پر اتان ڪو جواب نه مليو۔ ان ٻيهر وڃي عالمن کي ٻڌايو۔ انهن إِنَّا لِلَّهِ وَإِنَّا إِلَيْهِ رَاجِعُونَ پڙهي ڪري چيو ”اسان کي ڊپ آهي ته اهو شايد جنّتين مان ناهي،“ توهان يمن وڃو اتي برهوت نالي وادي ۾ هڪ کوھ آهي اتي وڃي اهڙي طرح آواز ڏي۔ ان ائين ڪيو ته پهرين آواز ۾ جواب مليو، مون ان کي گھر ۾ فلاڻي جڳھ تي دفن ڪئي آهي ۽ مون گھر وارن وٽ به امانت ناهي رکي، منهنجي پٽ وٽ وڃو

۽ انهيءَ جڳھ کي کوٽيو توهان کي ملي ويندي پوءِ ان ائين ڪيو ته مال ملي ويو، مون ان کان پڇا ڪيو؛ تون ته نيڪ هئين تون هت پهچي وئين؟ ته ان چيو؛ منهنجا ڪجھ رشتيدار خراسان ۾ هئا جن کان مون قطع تعلق ڪيو (يعني مائٽي ٽوڙي) انهي حالت ۾ مون کي موت اچي ويو انهي ڪري مون کي اها سزا ڏني وئي ۽ انهي جڳھ تي پهچايو ويو۔ (تنبیہ الغافلین ص ۷۲ مُلَخّصاً)

* مسڪرائي ۽ خوش اخلاقيءَ سان ڳالهه ٻولهه ڪيو * مسلمانن جي دلجوئي جي نيّت سان ننڍن سان شفقت وارو ۽ وڏن سان ادب وارو لهجو اختيار ڪيو۔ * وڏي آواز سان ڳالهائڻ کان حد درجي احتياط ڪيو * کڻي هڪ ڏينهن جو ٻار هجي سٺين سٺين نيتن سان ان سان به اوهان توهان سان گفتگو ڪرڻ جي عادت بڻايو، اوهان جا اخلاق به ا ن شآءَالله عَزَّوَجَلَّ عمدا ٿيندا ۽ ٻار به آدب سکندو * ڳالهه ٻولهه ڪرڻ وقت پردي جي جاءِ تي هٿ لڳائڻ، آڱرين سان بدن تان مير لاهڻ، ٻين جي سامهون بار بار نڪ کي هٿ لڳائڻ، يا نڪ يا ڪن ۾ آڱر وجهڻ، ٿُڪ اڇلائڻ سٺي ڳالهه ناهي * جيستائين ٻيو ڳالهائي رهيو هجي ته اطمينان سان ٻُڌو، ڳالهه ڪٽڻ کان بچو ڪنهن سان ڳالهه ٻولهه ڪجي ته ان جو ڪو صحيح مقصد به هئڻ گهرجي ۽ هميشه مخاطب جي ظرف ۽ ان جي نفسيات جي مطابق ڳالهه ڪئي وڃي * بدزباني ۽ بي حيائي جي ڳالهين کان هر وقت پرهيز ڪريو، گارگند کان بچندا رهو ۽ ياد رکو ته ڪنهن مسلمان کي بنا اجازتِ شرعي گار ڏيڻ حرامِ قطعي آهي۔ (فتاوي رضويه، ج 21 ص 127) ۽ بي حيائي جي ڳالهه ڪرڻ واري تي جنّت حرام آهي۔ تاجدار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمائن ٿا؛ ”اُن شخص تي جنت حرام آهي جيڪو فُحش گوئي (بي حيائي جي ڳالهه) کان ڪم وٺي ٿو۔“

(ڪتاب الصمت مع موسوعه الامام ابن ابي الدنيا ج 7 ص 204، رقم 325)

طرح طرح جون هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جا شايع ٿيل ٻه ڪتاب ” بهار شريعت “ حصو 16 (304 صفحا) ۽ 120 صفحن جو ڪتاب ” سنتیں اور آداب “ هديو ڏئي حاصل ڪيو ۽ پڙهو۔ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي۔
