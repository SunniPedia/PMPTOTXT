اَلۡحَمۡدُ لِلّٰہِ رَبِّ الۡعٰلَمِیۡنَ وَالصَّلٰوة وَالسَّلَامُ عَلٰی سَیِّدِ الۡمُرۡسَلِیۡن ط

اَمَّا بَعۡدُ! فَاَعُوۡذُ بِاللّٰہِ مِنَ الشَّیۡطٰنِ الرَّجِیۡم ط بِسۡمِ اللّٰہِ الرَّحۡمٰنِ الرَّحِیۡم ط

اَلصَّلٰوة وَ السَّلَامُ عَلَیۡكَ یَا رَسُولَ اللہ

وَعَلٰی اٰلِكَ وَ اَصۡحٰبِكَ یَا حَبِیۡبَ اللہ

اَلصَّلٰوة وَ السَّلَامُ عَلَیۡكَ یَا نَبِیَّ اللہ

وَعَلٰی اٰلِكَ وَ اَصۡحٰبِكَ یَا نُوۡرَ اللہ

نَـوَيتُ سُنَّتَ الاِعتِکاف (مون سنّتِ اعتڪاف جي نيّت ڪئي)

جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي نفلي اعتڪاف جي نيّت ڪري وٺو، جيستائين مسجد ۾ رهندؤ نفلي اعتڪاف جو ثواب پڻ ملندو رهندو ۽ ضمنًا مسجد ۾ کائڻ پيئڻ به جائز ٿي ويندو۔

رَسُولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عاليشان آهي: ’’جيڪو مون تي هڪ ڀيرو دُرُودِ پاڪ پڙھندو، الله عَزَّ وَجَلَّ ان تي ڏھ (10) رَحمتون نازِل فرمائيندو ۽ جيڪو مون تي ڏھ (10) ڀيرا دُرُودِ پاڪ موڪليندو، الله عَزَّ وَجَلَّ اُن تي سو (100) رحمتون نازل فرمائيندو ۽ جيڪو مون تي سو (100) ڀيرا دُرُودِ پاڪ موڪليندو، الله عَزَّ وَجَلَّ ان جي ٻنهي اکين جي وچ ۾ لکي ڇڏيندو ته اهو ٻانهو نِفاق ۽ جھنم جي باھ کان بَري آهي ۽ قِيامت جي ڏينهن ان کي شهيدن سان گڏ رکندو۔‘‘

(مجمع الزوائد،ڪتاب الادعيۃ،في الصلوۃ علي النبي۔۔۔الخ،الحديث: ۱۷۲۹۸ ،ج ۱۰ ،ص ۲۵۳)

ذِکر و دُرُود ہر گھڑی وِردِ زَباں رہے

میری فُضُول گوئی کی عادت نکال دو

(وسائلِ بخشش مُرمّم،ص305)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! اسان ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون نيتون ڪيون ٿا. فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم: نِيَّةُ المُؤمِنِ خَيرٌ مِّن عَمَلِهٖ. يعني مسلمان جي نيت ان جي عمل کان بهتر آهي. [1]

ٻه مدني گُل: 1 بنا سٺي نيت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

2 جيتريون سٺيون نيتون وڌيڪ، اوترو ثواب به وڌيڪ.

* نگاهون جھڪائي انتهائي توجھ سان بيان ٻُڌندس * ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو گوڏا ڀڃي ويهندس * ضرورت طور سوڙهو ٿي سِرڪي ٻين جي لاءِ جاءِ ڪُشادي ڪندس * ڌِڪو وغيره لڳو ته صبر ڪندس، گھورڻ، جھڻڪڻ ۽ اَٽڪڻ کان بچندس * صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله، تُوبُوا اِلَـي الله! وغيره ٻُڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي (يعني دلداري) جي لاءِ بلند آواز سان جواب ڏيندس * بيان کان پوءِ پاڻ اڳتي وڌي سلام، مُصافحو ۽ اِنفرادي ڪوشش ڪندس۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

حَضْرت سَيدُنا حُذيفه بن يمان رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: ’’ هڪ ڀيرو مون پنهنجي امڙ سائڻ رَضِىَ اللهُ تَعَالٰی عَـنْها کي عَرض ڪيو: اوهان مون کي اجازت عطا ڪيو ته مان نبيء رحمت، شفيعِ اُمّت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اقتدا ۾ مَغرِب جي نماز ادا ڪيان ۽ عَرض ڪيان ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مون ۽ توهان لاءِ مغفرت جي دعا

فرمائن، حضرتِ سَيدُنا حذيفه رَضِىَ اللهُ تَعَالٰی عَنْهُ فرمائن ٿا: مان نبي ڪريم، محبوبِ ربِّ عظيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي بابرڪت بارگاھ ۾ حاضر ٿيس ۽ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي اِقتدا ۾ مَغرِب نماز پڙھي، جڏهن پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم عشاء جي ادائيگي کان به فارغ ٿي ويا ۽ وڃڻ لڳا ته مان به پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي پويان پويان هلڻ لڳس۔

نبي غيب دان، رسولِ ذيشان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم منهنجو آواز ٻڌو ته فرمايائون: ڪير؟ ڇا حذيفه؟ عرض ڪيم: جي ها! اِرشاد فرمايائون: توهان جي ڪهڙي حاجت آهي؟ غَفَرَ اللهُ لَکَ وَ لِاُمِّکَ يعني الله عَزَّوَجَلَّ توهان جي ۽ توهان جي امڙ جي مغفرت فرمائي! پوءِ فرمايائون: اهو هڪ ملائڪ آهي، جيڪو ان رات کان پهريان ڪڏهن به زمين تي ناهي لٿو، ان پنهنجي رب عَزَّوَجَلَّ کان اجازت گھري ته مون کي سلام ڪري ۽ مون کي بِشارت ڏي ته: ’’ بِاَنَّ فَاطِمَۃَ سَیِّدَۃُ نِسَآءِ اَہْلِ الْجَنَّۃِ يعني فاطمه (رَضِىَ اللهُ تَعَالٰی عَـنْها) جنّتي عورتن جون سردار آهن، وَاَنَّ الْحَسَنَ وَالْحُسَیْنَ سَیِّدَا شَبَابِ اَہْلِ الْجَنَّۃِ ۽ حَسَن ۽ حُسَين (رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) جنّتي نوجوانن جا سردار آهن۔

(سنن الترمذي، ابواب الماقب عن رسول الله، باب مناقب حسن بن علي بن ابي طالب، حديث:۳۷۸۸)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! ڏٺو اوهان ته تاجدارِ رِسالت، شهنشاھ نبوت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن نبوّت جي نور سان حَضرت حذيفه رَضِىَ اللهُ تَعَالٰی عَنْهُ کي سڃاڻي به ورتو ۽ انهن جي دلي حاجت به مَعلُوم ڪري ورتي ته هي ڇو اچي رهيا آهن؟ ۽ انهن جي چَوڻ بِنا انهن جي لاءِ ۽ انهن جي والده جي لاءِ مغفرت جي دعا به فرمائي۔ يقيناً اسان جا آقا، مڪي مَدَني مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي دل جي حالت پوشيده نه آهي، پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي ته پٿرن جي حالت به ظاهر آهي،

جيئن مبارڪ حديثن جي سڀ کان معتبر ڪتاب ”بخاري شريف“ ۾ آهي:

سرڪارِ مدينه، قَرارِ قلب و سينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اِرشاد فرمايو: اُحُدٌ جَبَلٌ یُحِبُّنَا وَنُحِبُّہٗ يعني اُحُد جبل اسان سان مَحَبَّت ڪري ٿو ۽ اسان به ان سان مَحَبَّت ڪيون ٿا۔ (بخاري، ڪتاب الزڪاۃ، باب خرص التمر، ج۱،ص ۵۰۰،حديث:۱۴۸۲)

مٺا مٺا اسلامي ڀائرو! غور ڪيو! جڏهن آقا عَلَيه السَّلَام کي هڪ پٿر جو حال مَعلُوم آهي ۽ اهي ڄاڻن ٿا ته ڪنهن پٿر ۾ اسان سان ڪيتري مَحَبَّت آهي ته اهي مؤمن ٻانهي جي دلي احوال کان به يقيناً واقف آهن، تڏهن ته پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم حَضْرتِ حذيفه رَضِىَ اللهُ تَعَالٰی عَنْهُ جي چوڻ بغير ئي ان جي دلي خواهش تي آگاھ ٿيا ۽ ان جي لاءِ ۽ ان جي امڙ جي لاءِ مغفرت جي دُعا فرمائي، هن رِوايت کي ٻڌي ڪري جيڪڏهن اهو وسوسو اچي ته دلي احوال ۽ ڪيفيتن جو علم ته غيب آهي ۽ علمِ غيب صرف الله عَزَّوَجَلَّ وٽ آهي ته ياد رکو ته ان ۾ ڪوئي شڪ ناهي ته الله عَزَّوَجَلَّ عَالِمُ الغَيبِ وَالشَّهادَة آهي، ان جو علمِ غيب ذاتي يعني ان جو پنهنجو آهي ۽ هميشه هميشه کان آهي جڏهن ته نبين سڳورن عَلَيهمُ الصَّلٰو ۃ ُ وَالسَّلامُ ۽ ولين سڳورن رَحِمَهمُ الله السّلام جو علمِ غيب ذاتي ناهي بلڪ عطائي يعني الله تَعَالٰي جو ڏنل آهي ۽ انهن وٽ اهو علم هميشه هميشه کان ناهي بلڪ جڏهن کان الله عَزَّ وَجَلَّ انهن کي ڏنو آهي، تڏهن کان آهي ۽ جيترو ڏنو آهي اوترو آهي ان جي ڏيڻ کان بغير هڪ ذرّي جو به علم انهن کي حاصل ناهي، تنهنڪري الله عَزَّوَجَلَّ ئي انبيا ۽ اولياء کي علم عطا فرمائيندو آهي ۽ اهي ان علم کي اڳيان به ڦهلائيندا آهن، جيئن علمِ غيبِ مُصطَفي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي باري ۾قرآنِ پاڪ ۾ اِرشاد آهي:

وَ مَا ہُوَ عَلَی الۡغَیۡبِ بِضَنِیۡنٍ ﴿ۚ۲۴﴾

(پاره:۳۰، سورۃ التڪوير:۲۴)

ترجمو ڪنزالايمان: ۽ هي نبي غيب ٻڌائڻ ۾ بخيل نه آهي

ان آيت سڳوري جي حوالي سان تفسيرِ خازِن ۾ آهي: (هن آيت مان) مُراد هي آهي ته سرڪار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم وٽ علمِ غيب ايندو آهي ته ماڻهن کي ان جي بيان ڪرڻ ۾ بخل نٿا ڪن بلڪ توهان کي ٻڌائيندا آهن۔ (خازن،۴/۳) هن آيت ۽ تفسير مان معلوم ٿيو ته الله عَزَّوَجَلَّ جا محبوب، داناءِ غيوب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ماڻهن کي غيب ٻڌائيندا آهن ۽ ظاهر آهي ٻڌائيندو اهو جيڪو پاڻ به ڄاڻندو هجي۔

سيدي اعليٰ حَضْرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ’’حدائقِ بخشش شريف‘‘ ۾ ڇا خوب فرمائن ٿا:

اور کوئی غیب کیا تم سے نِہاں ہو بھلا

جب نہ خُدا ہی چھپا تم پہ کروڑوں دُرُود

شرحِ ڪلامِ رضا:

يا رَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم اوهان جي شان جي ڇا ڳالھ ڪجي! شبِ معراج عين بيداري جي حالت ۾ اوهان پنهنجي مُبارَڪ سر جي اکين سان پنهنجي پاڪ پَروَردگار عَزَّوَجَلَّ جو ديدار ڪيو، ته ائين الله عَزَّوَجَلَّ جيڪو غيبُ الغيب آهي، اهو به پنهنجي فضل ۽ ڪرم سان پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي ظاهر ۽ آشڪار ٿي ويو ته هاڻي ڪو ٻيو غيب پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان ڪهڙي طرح لڪي سگھي ٿو، پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي ڪروڙين ڀيرا دُرُود ۽ سلام نازل ٿئي۔

بچه بچه نور ڪا!

مٺا مٺا اسلامي ڀائرو! حَضْرت سَيدُنا حُذَيفه بن يمان رَضِىَ اللهُ تَعَالٰی عَنْهُ کان مروي اِيمان افروز حديثِ پاڪ مان آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي نياڻي شهزادِي جي وڏي شان سان گڏوگڏ سرڪار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا ڏوهٽا حَسَنَينِ ڪريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا

جو شان ۽ شوڪت به ظاهر ٿيو۔ ٻئي شهزادا پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا مشابهت رکندا هئا۔

مٺا مٺا اسلامي ڀائرو! يقيناً حَضْرت سَيدُنا امام حَسَن ۽ حَضْرت سَيدُنا امام حُسَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا ڪرم ۽ شفقت جي باغ جا مهڪندڙ گل آهن، اهي ٻئي سرڪارِ ڪائنات، فخرِ مَوْجُودات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا ڏوهٽا آهن، آقا ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي هنن سان بيحد مَحَبَّت هئي، اچو! انهن جي مَحَبَّت کي وڌائڻ ۽ پنهنجي لاءِ نيڪيون لکائڻ جي نِيت سان انهن جو ذڪرِ خير ٻڌون ٿا:

نالو، ڪُنيت ۽ لقب:

انهن ٻنهي شهزادن مان وڏا حضرتِ سَيدُنا امام حَسَن رَضِىَ اللهُ تَعَالٰی عَنْهُ آهن۔ سندن جي ڪنْيت ”ابو محمد“ آهي۔ لقب”تَقِي ۽ سَيد“ جڏهن ته عرف ’’ سِبْطُ رَسُولِ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم “ ۽ ”سبطِ اڪبر“ آهي۔ اوهان کي ” رَيۡحَانَۃُالرَّسُوْل “ (يعني رسولِ خدا جا گل) ۽ ” آخِرُالۡخُلَفَاءِ بِالنَّصِّ “ (يعني حديثِ پاڪ جي مطابق آخري خليفه) به چون ٿا۔ پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ جي مُبارَڪ ولادت 15 رَمَضانُ المُبارَڪ 3 هجري جي رات ۾ مديني شريف ۾ ٿي، حُضُور سَيّدِ عالم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ستين ڏينهن اوهان جو عقيقو ڪيو ۽ وار جُدا ڪيا ويا ۽ حڪم ڏنو ويو ته وارن جي وزن برابر چاندي صدقو ڪئي وڃي. (تاريخُ الْخُلَفَا، بابُ الحسن بن علي بن ابي طالب، ص ۱۴۹ وروضۃ الشهداء (مترجم)، باب ششم،۶/ ۱۳۹) اوهان جو نالو اِمامُ الانبياء، سَيدُ الاَسْخِيا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ئي ’’حسن‘‘رکيو۔ (سوانحِ ڪربلا،ص۹۲)

سندن رَضِىَ اللهُ تَعَالٰی عَنْهُ جا ننڍا ڀاءُ امامِ عالي مقام، حَضرتِ سَيدُنا امامِ حُسَين رَضِىَ اللهُ تَعَالٰی عَنْهُ جي ولادت 5 شعبانُ المعظم سن 4 ھ تي مديني شريف ۾ ٿي، انهن جو نالو به حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ’’حُسَين‘‘ ۽ ’’شبّير‘‘ رکيو جڏهن ته اوهان جي ڪنْيت ’’ابو عبدالله‘‘ ۽ اوهان جو لقب به ’’ سِبۡطُ رَسُوۡلِ الله ‘‘(يعني رسولِ خدا جا ڏوهٽا) ۽ ’’ رَيحَانَۃُ الرَّسُوْل ‘‘(يعني رسولِ خدا جو گل) آهي ۽ پاڻ رَضِىَ اللهُ تَعَالٰی عَنْهُ به جنتي جوانن جا سردار آهن۔ (اسد الغابۃ، باب الحاء والحسين، ۱۱۷۳ ۔الحسين بن علي،ص ۲۵ ، ۲۶ ملتقطاًوسير اعلام النبلاء، ۲۷۰ ۔الحسين...الخ، ۴/۴۰۲ ۔ ۴۰۴)

اُمُّ الْمُؤمنين حضرت سَيّدَتُنا عائشه صديقه رَضِي اللّٰه تَعَالٰي عَنْها فرمائن ٿيون ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن حضرت سَيّدُنا امام حَسن ۽ امام حُسين رَضِي اللّٰه تَعَالٰي عَنْهمَا جي پيدائش جي ستين ڏينهن، انهن جي طرفان ٻه ٻه ٻڪريون عَقيقي ۾ ذَبْح ڪيون ۔ (مصنف عبد الرزاق ،باب العقيقه ، ۴/۲۵۴,حديث: ۷۹۹۳)

مٺا مٺا اسلامي ڀائرو! عقيقو ڪرڻ سنّت عمل آهي. (فتاويٰ رضويہ ج 20 ص 586) صدرُ الشَّريعه، بدرُ الطَّريقه ، حضرت علامه مولانا مفتي محمد امجد علي اعظمي عَلَيْهِ رَحْمَةُ اللّٰهِ الۡـقَوِی فرمائن ٿا؛ عقيقي جي لاءِ ستون ڏينهن بهتر آهي ۽ جيڪڏهن ستين ڏينهن نه ڪري سگهو ته جڏهن چاهيو ڪري سگهو ٿا، سنّت ادا ٿي ويندي۔ ڪِن اهو چيو آهي ته ستين يا چوڏهين يا ايڪيهين ڏينهن يعني ستن ڏينهن جو لحاظ رکيو وڃي، اِهو بهتر آهي۔ ۽ ياد نه رهـي ته هيئن ڪريو جو ٻار جنهن ڏينهن پيدا ٿيو ان ڏينهن کي ياد رکو ان کان هڪ ڏينهن پهريان وارو ڏينهن جڏهن اچي ته اهو ستون ڏينهن ٿيندو، مثلاً جمعي تي پيدا ٿيو ته خميس ستون ڏينهن آهي ۽ ڇنڇر تي پيدا ٿيو ته ستون ڏينهن جمعو ٿيندو، پهرين صورت ۾ خميس جي ڏينهن ۽ ٻي صورت ۾ جمعي جي ڏينهن عقيقو ڪندو ته اهو ستين ڏينهن ۾ شمار ٿيندو۔ (بهارِشريعت ج ۳ ص ۳۵۶)

عقيقي جي باري ۾ تفصيلي معلومات حاصل ڪرڻ چاهيو ٿا ته اميرِ اهلسنّت دَامَتْ بَـرَکاتُهمُ الْعَالِيه جو رِسالو "عقيقي جي باري ۾ سوال جواب" جو مطالعو فرمايو۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! هتي اها ڳالھ غور لائق آهي ته عام طور تي جڏهن به ٻار پيدا ٿيندو آهي عقيقو ان جا والدين ئي ڪندا آهن، پر اسان ڏسون ٿا ته جڏهن حضرت سَيدُنا امامِ حسن رَضِىَ اللهُ تَعَالٰی عَنْهُ جي ولادت ٿي تڏهن به ۽ جڏهن حضرتِ سَيدُنا امام حُسين رَضِىَ اللهُ تَعَالٰی عَنْهُ پيدا ٿيا تڏهن به عقيقو انهن جي والِدَين نه ڪيو بلڪ انهن جا نانا جان، رحمتِ عالميان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ئي پنهنجي پيارن ڏوهٽن جو عقيقو فرمايو، ان مان اندازو لڳائي سگھجي ٿو ته حُضُور عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام پنهنجن ڏوهٽن سان ڪيتري مَحَبَّت ڪندا هئا؟ آقا عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام نه صرف

حَسَنَينِ ڪريمَين سان پيار ۽ مَحَبَّت ڪندا هئا بلڪ پاڻ عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام ڪيئي موقعن تي انهن جي شان ۽ عظمت کي به بيان فرمايو آهي، اچو ’’حَسَنَين‘‘ جي پنج (5) اکرن جي نسبت سان انهن جي عظمت ۽ شان جي باري ۾ پنج (5) حديث سڳوريون ٻڌون ٿا:

1) اميرُالمؤمنين حضرت سَيدُنا عُمر فارُوقِ اعظم رَضِىَ اللهُ تَعَالٰی عَنْهُ کان روايت آهي ته مون حضرت سَيدُنا امام حَسن ۽ حضرتِ سَيدُنا امام حُسين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کي سرڪار (عَلَيه الصَّلٰوۃُ وَالسَّلَام) جي ڪلهن تي سُوار ڏٺو ته چيم: اوهان ٻنهي جي سُواري ڪيتري شاندار آهي؟ ته نبي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اِرشاد فرمايو: ۽ سوار به ته ڪيترا لاجواب آهن۔

(ترمذي، باب مناقبِ حسن رضي الله عنه ج۵، ص۴۳۲، حديث ۳۸۰۹)

2) هڪ ٻي حديثِ پاڪ ۾ اِرْشاد فرمايائون: مَنْ اَحَبَّھُمَا فَقَدْ اَحَبَّنِي وَمَنْ اَبْغَضَھُمَا فَقَدْ اَبْغَضَنِي يعني جنهن اِنهن سان مَحَّبت ڪئي، اُن مون سان مَحَّبت ڪئي ۽ جنهن انهن سان عَداوت (يعني دُشمني) ڪئي ان مون سان عَداوت (يعني دُشمني) ڪئي۔ (ابن ماجه،ڪتاب السنۃ،باب في فضائل اصحاب رسول اللّٰه،۱/۹۶، حديث:۱۴۳)

3) حَسَن ۽ حُسَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا جَنَّتي جوانن جا سردار آهن۔ (سنن الترمذي،ڪتاب المناقب، باب مناقب ابي محمد الحسن بن علي...الخ، الحديث: ۳۷۹۳ ،ج ۵ ،ص ۴۲۶)

4) حُضُورِ اقدس صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم انهن ٻنهن ننڍڙن کي سِنگھندي مبارڪ سيني تي ليٽائيندا هئا۔ (سنن الترمذي،ڪتاب المناقب،باب مناقب ابي محمد الحسن بن علي...الخ، الحديث: ۳۷۹۷ ،ج ۵ ،ص ۴۲۸)

5) هُمَا رَيْحَانَتَايَ مِنْ الدُّنْيَا يعني حَسن ۽ حُسَين (رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) دُنيا ۾ منهنجا ٻه گل آهن۔ (ترمذي ، باب مناقبِ حسن رضي الله عنه ج۵،ص ۴۲۷،حديث ۳۷۹۵)

مٺا مٺا اسلامي ڀائرو! غور ڪيو! ته پياري آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي حَسَنَينِ

ڪريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا سان ڪمال جي مَحَبَّت ۽ اُلفت هئي، اهي ماڻهو جيڪي پنهنجن ٻارن سان ان ڪري پيار ۽ مَحَبَّت سان پيش نٿا اچن ته ڪٿي انهن جو رُعب ۽ دٻدٻو ختم نه ٿي وڃي، انهن جي لاءِ آقا عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام جو هي عمل نصيحت جوڳو درس آهي ته حُضُورِ اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سيدِ ڪائنات هئڻ جي باوجود به پنهنجي ڏوهٽن کي ڪلهن تي سوار ڪندا هئا، آقا عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام حَسَنَينِ ڪريمَين جي محبت کي پنهنجي مَحَبَّت قرار ڏنو ۽ حَسَنَينِ ڪريمَين سان دشمني کي پاڻ سان دشمني قرار ڏنو، پنهنجي اولاد کي چُمڻ، سيني سان لڳائڻ، ڪلهن تي سوار ڪرڻ ۽ اُنهن کي سيني سان لڳائڻ اهو ته ٿئي ٿو پر آقا عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام حَسَنَينِ ڪريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا کي س ِ نگھندا به هئا ۽ فرمائيندا هئا حسن ۽ حسين (رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) دنيا ۾ جنت جا گل آهن۔

(مراٰۃ المناجيح،باب مناقب اهل بيت النبي، الفصل الاول، ۸/۴۶۲)

مٺا مٺا اسلامي ڀائرو! حَسَنَينِ ڪريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا رَسُولُ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم

جا جگر بند، حَضْرت بي بي فاطمه زهرا رَضِي اللّٰه تَعَالٰي عَنْها جا شهزادا ۽ اَمِيرُ المؤمنين حَضرتِ سَيدُنا علي المرتضيٰ کرَّمَ الله تَعَالٰي وَجْهه الْکرِيم جا دُلارا (نهايت ئي پيارا ۽ دادلا) آهن، انهن جا فضائل ۽ مناقب بي شمار آهن، ڇو نه هجن جو انهن جو گھراڻو ئي سمورن فضيلتن ۽ ڪمالن ۽ برڪتن ۽ چڱائين جو منبع ۽ مرڪز آهي۔ اهي ٻئي شهزادا رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ڪمالات جا مظهر ۽ جامع هئا۔ رسول الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجي اهلِ بيت مان حَسَنَينِ ڪريمَين رَضِیَ اللهُ تَعَالٰی عَنْهُمَا سان بي پناه مَحَبَّت ڪندا هئا،

حَضْرت اُسامه بن زيد رَضِي اللّٰه تَعَالٰي عَنْه فرمائن ٿا ته مان هڪ رات ڪنهن ڪم سبب نبي اڪرم،نورِ مجسّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي خدمت ۾ حاضر ٿيس ته نبي پاڪ،صاحبِ لولاڪ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان طرح تشريف فرما ٿيا جو پاڻ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ڪا شيء هنج ۾ کنئي هئي، مون کي خبر نه هئي ته اها ڇا آهي، جڏهن مان پنهنجي ضرورت کان فارغ ٿيس مون پڇيو هي ڇا آهي؟ جيڪا اوهان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هنج ۾ کنئي آهي؟ حُضُور عَلَيه الصَّلٰو ۃ ُ وَالسَّلَام ان کي کوليو ته حَسن ۽ حُسين (رَضِیَ اللهُ تَعَالٰی عَنْهُمَا) سندن (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) جي هنج ۾ هئا، پوءِ سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن اِرشاد فرمايو: اهي ٻئي منهنجا پٽ، منهنجي ڌيء جا پٽ آهن۔ الٰهي! مان انهن ٻنهي سان مَحَبَّت ڪيان ٿو تون به هنن سان مَحَبَّت فرما ۽ جيڪو هنن سان مَحَبَّت ڪري ان سان به مَحَبَّت فرماءِ!

مَعْلُوم ٿيو ته پنهنجي اولاد ۽ ٻارن کي خوش ڪرڻ جنّت ۾ داخلي جو سبب به آهي۔ ٻار جي نفسياتي ضرورتن مان به هي آهي ته انهن سان مَحَبَّت ۽ شفقت ڀريو ورتاؤ ڪيو وڃي، اهي ٻار جيڪي ماءُ پيءُ جي مَحَبَّت کان محروم رهندا آهن، ڪڏهن ڪڏهن اُهي نفسياتي بيمارين ۾ مبتلا نظر ايندا آهن، بهترين کاڌن، ڪپڙن ۽ رانديڪن جي بدران ٻار ان ڳالھ جو وڌيڪ محتاج هوندو آهي ته ان جا ماءُ پيءُ ان تي توجهه ڏين ۽ ان سان مَحَبَّت ڪن۔

ڪي ماڻهو پنهنجي ٻارن جي اڳيان مال و اسباب جا ته ڍير جمع ڪندا آهن، پر انهن کي پنهنجي حقيقي توجهه ۽ پيار محبت کان محروم رکندا آهن، انهن وٽ ايترو به وقت ناهي هوندو جو اهي پنهنجي اولاد سان مَحَبَّت ڀريون ڪجھ ڳالهيون ئي ڪري وٺن۔ اهڙي طرح بعض والِدَين اهڙا به هوندا آهن، جيڪي پنهنجي ٻارن تي اجائي سختي ڪندا آهن، حالانڪ جهڙي طرح هڪ ٻار جي چڱي صحت جي لاءِ متوازن غذا ضروري آهي، اهڙي طرح ٻارن جي بهترين تربيت جي لاءِ والِدَين جو مناسب پيار به نهايت ضروري آهي۔ اچو! ٻارن سان محبت جي اِظهار جا ڪجھ طريقا ٻڌون ٿا:

1. انهن جي جائز خواهشن کي پنهنجي وُسعت جي مطابق پورو ڪرڻ جي ڪوشش ڪئي وڃي۔

2. انهن جي ڳالهين کي توجهه سان ٻڌڻ جي عادت بڻائي وڃي، ڪڏهن ڪڏهن ٻار ڪجھ معلوماتي سُوال ڪندا آهن، ان موقعي تي به انهن جي پوري ڳالھ ٻڌڻ گھرجي، ان سان انهن جي حوصله افزائي به ٿيندي ۽ وڌيڪ سکڻ جو ذهن بڻجندو، جڏهن ته فضول گفتگو ڪرڻ کان ننڍپڻ کان ئي انهن کي بچڻ جو ذهن ڏنو وڃي۔

3. ڪو به چڱو ڪم سر انجام ڏيڻ (مثال طور نماز جي پابندي ڪرڻ، تلاوتِ قرآن ڪرڻ، مدني انعامات تي عمل ڪرڻ وغيره) تي انهن کي تحفا ڏنا وڃن يا انهن جي حوصله افزائي ڪئي وڃي۔
