اهڙي طرح حضرت سَيدُنا سفيان ثوري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: مون طواف ڪري رهيو هيس ان دوران هڪ شخص کي هر قدم تي حُضور نبي ڪريم، صاحبِ خُلقِ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم تي دُرُودِ پاڪ پڙھندي ڏٺو ته ان کان پڇا ڪيم: اي ڀاءُ! سُبْحٰنَ الله ، لَآاِلٰه اِلَّا الله جي بدران دُرُودِ پاڪ جو وِرد ڪندا پيا وڃو، هن ۾ ڪهڙو راز آهي؟ ته هو پڇڻ لڳو: الله پاڪ اوهان کي معاف فرمائي، اوهان ڪير آهيو؟ مون ٻڌايو: مان سفيان ثوري آهيان۔پوءِ ان چيو: جيڪڏهن اوهان زماني وارن ۾ اجنبي نه هجو ها ته مان اوهان کي پنهنجي حال ۽ ان راز جي خَبَر نه ڏيان ها، پوءِ چيائين ته مان پنهنجي والدِ گرامي سان گڏ بيتُ الله جي حجِ جي ارادي سان نڪتس۔ سفر دوران منهنجو والد صاحب بيمار ٿي پيو، اُن جو چهرو ڪارو (Black) ٿي ويو، اکيون نيريون ٿي ويون ۽ پيٽ ڦوڪجي ويو هو، مون روئيندي هي پڙھيو: اِنّالِلّٰه وَاِنَّا اِلَيه رَاجِعُوْنَ (پ۲، البقرة:۱۵۶) (ترجَمو ڪنز الايمان: اسان الله جي مِلڪ آهيون، ۽ اسان کي ان جي طرف موٽڻو آهي) (ٿوري دير کان پوءِ) انهيءَ ويراني ۾ منهنجي والد جو انتقال ٿي ويو، پوءِ مون ان جي چهري تي چادر وجهي ڇڏي۔ اوچتو مون تي ننڊ جو غَلَبواچي ويو ۽ مان سمهي پيس، مون هڪ شخص کي ڏٺو ان کان وڌيڪ حسين وجميل ۽ صاف سٿري لباس وارو مون ڪڏهن نه ڏٺو هو۔ هو منهنجي والدِ محترم جي ويجهو تشريف کڻي آيا ۽ ان جي چهري تان چادر هٽائي پنهنجو مبارَڪ هٿ چهري تي ڦيرڻ لڳا۔ ته ان جو چهرو کير کان وڌيڪ سفيد ٿي ويو۔ پوءِ ان شخص پيٽ تي هٿ ڦيريو ته اهو پهريان وانگر ٿي ويو، پوءِ هو موٽڻ لڳا ته مان بيهي

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس ٭ ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس ٭ ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي جي لاءِ بلند آواز سان جواب ڏيندس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ربيع الاول جو مبارڪ مهينو پنهنجون برڪتون لُٽائي رهيو آهي، هر پاسي مٺي مٺي آقا، ٻنهي جھانن جي داتا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ذِڪر سان عاشقانِ رسول پنهنجي دِل ۽ دِماغ کي مهڪائي رهيا آهن، ڪٿي سندن ولادت جو ذِڪر آهي ته ڪٿي سندن ڪمالات (Virtues) جي ڳالھ آهي۔ ڪٿي سندن سيرت بابت بيان آهي ته ڪٿي سندن مبارڪ صورت جو ذِڪر خير آهي۔ ڪٿي عِبادتِ مُصطَفٰي جو بيان آهي ته ڪٿي سلطنتِ مُصطَفٰي جا چرچا آهن۔ ڪٿي شفاعتِ مُصطَفٰي جا تذڪرا آهن ته ڪٿي سندن عنايتن جا چرچا آهن۔ ڪٿي سندن عظمتن جو بيان آهي ته ڪٿي سندن رِفعتن (بُلندين) جا تذڪرا آهن۔ ڪٿي طاقتِ مُصطَفٰي جو بيان آهي ته ڪٿي شُجاعتِ مُصطَفٰي جا گيت ڳايا پيا وڃن۔ ڪٿي نگاھ مُصطَفٰي جون ڳالهيون آهن ته ڪٿي عطاءِ مُصطَفٰي جو بيان آهي۔ ڪٿي خاندانِ مُصطَفٰي جا چرچا آهن ته ڪٿي برڪاتِ نبوت جو بيان آهي۔ ڪٿي معجزاتِ مُصطَفٰي جون ڳالهيون آهن ته ڪٿي اَخلاقِ مُصطَفٰي جو بيان آهي۔ ڪٿي اِختياراتِ مُصطَفٰي جون ڳالهيون آهن ته ڪٿي نوازشاتِ مُصطَفٰي جو بيان آهي۔ ڪٿي هجرتِ مُصطَفٰي جون ڳالهيون آهن ته

ڪٿي اِستقبالِ مُصطَفٰي جو بيان آهي۔ ڄڻ ته ذرو ذرو ميلادِ مُصطَفٰي جي برڪتن مان پنهنجو حصو ماڻي رهيو آهي۔

اچو! انهيء مناسبت سان اڄ اسان به ”اِمدادِ مُصطَفٰي جي واقعن“ تي مُشتَمِل بيان ٻڌڻ جي سعادت حاصل ڪريون ٿا۔ صحابَه ڪرام ۽ ٻيا بزرگ سڳورا رِضْوَانُ الله عَلَيهمْ اَجْمَعِين مشڪل حالتن ۾ ڪهڙي طرح رسولِ خدا، شاھ ارض و سما صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کان فرياد ڪن ٿا ۽ اِمدادِ مُصطَفٰي سان خالي جھوليون ڪيئن ڀرين ٿا، اهو به ٻڌنداسين۔ اسان جا آقا، مڪي مدني مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ڪيئن پنهنجي اُمّتين جي مدد ڪن ٿا اهو به ٻڌنداسين۔ اهڙي طرح پيارا آقا، مڪي مدني مُصطَفٰي عَلَيه الصَّلٰوة وَالسَّلَام جانورن جي به فرياد ٻڌندا هئا ۽ مدد ڪندا هئا، ان بابت به ڪجھ واقعا ٻڌنداسين۔ ان جي علاوه ٻيا به ڪيئي طرح جا مدني گل ٻڌڻ جي سعادت حاصل ڪنداسين۔ الله ڪري ته دِلجمعي سان اسان شروع کان آخر تائين سٺين سٺين نيتن سان بيان ٻڌڻ جي سعادت حاصل ڪريون۔

اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللّٰهُ عَلَيۡهِ واٰلِهٖ وَسَلَّمَ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

سَيفُ الله جو فرياد

دعوتِ اسلامي جي اشاعتي اداري مڪتبة المدينه جي ڪتاب ”فيضانِ صديقِ اڪبر“ جي صفحي 384 تي آهي: اميرالمؤمنين حضرت سَيدُنا ابوبڪر صديق رَضِىَ اللهُ تَعَالٰی عَـنْهُ حضرت سَيدُنا عِڪرِمه رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي مُسَيلِمَه ڪذَّاب جي سر قلم ڪرڻ جي لاءِ روانو فرمايو ۽ پوءِ حضرت سَيدُنا شُرَحۡبِيل رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي ان جي مدد جي لاءِ موڪليو، پر انهن ٻنهي جي اڳيان ان بدبخت هٿيار ڦٽا نه ڪيا، ڇوته حُضورِ اڪرم، نُورِ مجسَّم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي دنيا مان تشريف کڻي وڃڻ کان بعد مُسَيلِمَه ڪذَّاب جو ڪاروبار چمڪي پيو هو ۽ تقريبًا هڪ لک (100000) کان وڌيڪ ماڻهو ان جي چوڌاري گڏ ٿي ويا هئا، حضرت سَيدُنا عِڪرِمه ۽ حضرت سَيدُنا شُرَحبِيل رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا کان به ان جي خُوب لڙائي ٿي،

جنهن ۾ ان جا ڪيئي ماڻهو مارجي ويا، ايتري ۾ انهن ٻنهن صحابه رَضِىَ اللهُ تَعَالٰی عَـنْهُمَا جي مدد جي لاءِ حضرت سَيدُنا خالد بن وَليد رَضِىَ اللهُ تَعَالٰی عَـنْهُ به پهچي ويا۔ سندس رَضِىَ اللهُ تَعَالٰی عَـنْهُ جي لشڪر جو تعداد چويھ هزار(24000) هئي ۽ مُسَيلِمَه ڪذَّاب وٽ ان وقت چاليھ هزار (40000) فوج هئي، ٻئي لشڪر بهادري سان وڙهيا ۽ جنگ جو نقشو ڪيئي ڀيرا تبديل ٿيو، ڪڏهن حالتون مسلمانن جي حق ۾ ٿي وينديون هيون ۽ ڪڏهن مُرتَدن (يعني اِسلام کان ڦري وڃڻ وارن) جو پلڙو ڀاري رهندو هو ۔ ثُمَّ بَرَزَ خَالِدٌ وَدَعَا اِلَي الْبَرَّازِ وَنَاديٰ بِشِعَارِھِمْ وَکانَ شِعَارُھُمْ يا مُحَمَّدَاه، فَلَمْ يبْرُزْ اِلَيه اَحَدٌ اِلَّا قَتَلَه يعني جڏهن حضرت سَيدُنا خالد بن وَلِيد رَضِىَ اللهُ تَعَالٰی عَـنْهُ کي يقين ٿي ويو ته بَنُو حَنِيفَه قبيلي وارا ان وقت تائين نه هٽندا جيستائين مُسَيلِمَه ڪذَّاب کي قتل نه ڪيو وڃي ته پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ بذاتِ خُود ميدان ۾ لٿا ۽ مقابلي جي لاءِ مُسَيلِمَه ڪذَّاب جي شهسوارن کي طلب ڪيائون ۽ مسلمانن جي شِعَار يعني عادت جي مطابق’’يامُحَمَّدَاه‘‘ جو نعرو (Chant) لڳايو ۽ ان وقت جنگ ۾ مسلمانن جو شِعَار اهو هو جو اهي مشڪل وقت ۾ وڏي آواز سان اهو نعرو لڳائيندا هئا ’’يامُحَمَّدَاه‘‘ يعني يا رسُولَ الله (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم)! اسان جي مدد فرمايو۔ اهڙي طرح حضرت سَيدُنا خالد بن وَلِيد رَضِىَ اللهُ تَعَالٰی عَـنْهُ به نعرو هنيو ۽ پوءِ دشمنن جي طرفان جيڪو به مقابلي ۾ آيو، پاڻ رَضِىَ اللهُ تَعَالٰی عَـنْهُ ان جو ڪنڌ لاهي ڇڏيو۔ نيٺ مُسَيلِمَه ڪذَّاب جي ساٿين کي شڪست ٿي ۽ اهي سڀئي ڀڄڻ لڳا۔ مسلمانن جي هڪ جماعت انهن جو پيڇو ڪيو، گھڻن کي جهنم جي حوالي ڪيائون ۽ گھڻن کي گرفتار ڪري قيدي بڻائي ورتائون ۽ کوڙ غنيمت جو مال مسلمانن کي هٿ آيو۔ (فيضانِ صديقِ اڪبر،ص۳۸۴بتغير قليل)

اعليٰ حضرت،اِمام ِاَهلسنّت مولانا شاھ امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه پنهنجي نعتيه ديوان ”حدائقِ بخشش“ ۾ هن طرح عرض ڪن ٿا:

فریاد اُمَّتی جو کرے حالِ زار میں ممکن نہیں کہ خیرِ بشَر کو خَبَر نہ ہو

شعر جي مختصر وضاحت: يعني ڀلا اهو ڪيئن ٿي سگھي ٿو ته پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو مصيبت ۾ ڦاٿل ڪو اُمَّتي خُلوصِ نِيت سان پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي سڏي ۽ سموري مخلوقات ۾ سڀ کان بهترين شخصيت يعني حضورِ انور، شافعِ محشر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان تي ايندڙ مصيبت کان بي خَبَر رهن۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! بيان ڪيل واقعي ۾ جنهن جنگ جو ذِڪرڪيو اها جنگِ يمَامَه آهي، جنهن ۾ حضرت سَيدُنا خالد بن وَلِيد رَضِىَ اللهُ تَعَالٰی عَـنْهُ مشڪل گھڙي ۾ شهنشاھ ڪون ومڪان، مَحبوبِ رحمٰن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي وِصالِ ظاهري کان بعد مدينَه مُنَوَّرَه کان تمام پري ٿي ڪري به ”يا مُحَمَّدَاه “ جا نعرا لڳائي رهيا آهن۔ غَور ڪريو ته اهي خالد بن وَلِيد رَضِىَ اللهُ عَنْه جن کي مڪي مَدَني سُلطان، رَحمت ِ عالَمِيان صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سَيفٌ مِن سُيُوفِ الله يعني الله پاڪ جي تَلوارن مان هڪ تَلوار جي خِطَاب سان نوازيو، اهي خالدبن وليد رَضِىَ اللهُ تَعَالٰی عَـنْهُ جيڪي پاڻ معلمِ انسانيت،شفيعِ اُمّت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا تربيت يافته هئا، اهي خالدبن وليد رَضِىَ اللهُ تَعَالٰی عَـنْهُ جيڪي اهڙي عظيم لشڪرِ اسلام جا سپه سالار (Chief) هئا جنهن ۾ رَسُولِ خُدا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا تَربِيت يافتَه ۽ شريعت جا پابند وڏا وڏا صحابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ موجود هئا، جيڪڏهن نبي ڪريم، رؤف و رحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي وصالِ ظاهري کان بعد پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي مدد جي لاءِ سڏڻ جائز نه هجي ها ته يقيناً ٻيا صحابي سڳورا عَـلَيْهِمُ الرِّضْوَانْ چپ نه رهن ها بلڪ انهيء وقت پاڻ رَضِىَ اللهُ عَنْه کي ان عمل کان روڪين ها، پر انهن نه روڪيو، تنهنڪري ان جو نه روڪڻ ئي هن ڳالھ جو دليل بڻجي ويو ته حضرت سَيدُنا خالد بن وَلِيد رَضِىَ اللهُ تَعَالٰی عَـنْهُ جو عمل شرعاً جائز هو۔ لهٰذا جڏهن به ڪا مصيبت (Trouble) اچي ته پوري تَوَجُّه ۽ ڪامل يقين سان رَسولِ ڪائنات، فَخرِمَوجُودات صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي مدد جي لاءِ پُڪاريو ۽ دل ۾ اهو يقين رکو ته رسولِ خدا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم نه صِرف منهنجو فَرياد ٻڌي رهيا آهن بلڪ الله

پاڪ جي عطا سان منهنجي مدد به فرمائيندا۔

یا رَسُوْلَ اللہ کے نعرے سے ہم کو پیار ہے ہم نے یہ نعرہ لگایا اپنا بیڑا پار ہے

خُلد میں ہوگا ہمارا داخِلَہ اِس شان سے یا رَسُوْلَ اللہ کا نعرہ لگاتے جائیں گے

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! صحابَه ڪرام عَـلَيْهِمُ الرِّضْوَانْ جو آقا ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي سڏڻ مرحبا! هن ۾ اسان جي به تربيت جو سامان آهي۔ اسان کي گھرجي ته اسان به نبي اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي هن طرح پڪاريندا ڪريون، مثال طور: يارَسُوْلَ الله ، يا حَبِيبَ الله ، يا نَبِيَّ الله ، يانُورَالله ۔ جيئن سرڪارِ اعليٰ حضرت رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: الله پاڪ فرمايو:

لَا تَجْعَلُوْا دُعَآءَ الرَّسُوْلِ بَیْنَكُمْ كَدُعَآءِ بَعْضِكُمْ بَعْضًاؕ- (پ ۱۸، النور:۶۳)

ترجَمَو ڪنزُالايمان: رسول جي سڏڻ کي پاڻ ۾ ايئن نه بنايو جيئن توهان مان هڪڙو ٻئي کي سڏي ٿو .

جو اي زيد، اي عمرو۔ بلڪه هن طرح عرض ڪريو: يارَسُولَ الله ، يانَبِي الله ، ياسَيدَالْمُرْسَلِين ، ياخَاتَمَ النَّبِيين ، ياشَفِيعَ الْمُذْنِبِين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم (فتاويٰ رضويه، ۳۰/۱۵۶)

مٺا مٺا اسلامي ڀائرو! خوفِ خدا ۽ عشقِ مُصطَفٰي جو پيڪر بڻجڻ، گناهن کان نفرت ڪرڻ، نيڪي جي دعوت ۽ اُمّتِ مُسلمه جي خيرخواهي وغمخواري جو جذبو ماڻڻ جي لاءِ عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي وڃو ۽ ذيلي حلقي جي 12مدني ڪمن ۾ وڌ کان وڌ حصو وٺو، ذيلي حلقي جي 12مدني ڪمن مان هڪ مدني ڪم ”مدني مذاڪرو“ آهي، هن مدني ڪم جابيشُماردِيني ۽ دُنيوي فائدا آهن۔ جيئن ٭ مدني مذاڪري جي برڪت سان گناهن کان بچڻ جو ذِهن ملندو آهي۔ مدني مذاڪري جي برڪت سان علمِ دِين حاصل ٿيندو آهي۔ ٭ مدني مذاڪري جي برڪت سان گناهن کان بچڻ جو ذِهن ملندو آهي۔ ٭ مدني مذاڪري جي برڪت سان دِيني معلومات (Islamic Information) سان گڏوگڏ اَخلاقي ۽ تنظيمي تربيت به نصيب ٿيندي آهي۔ ٭ مدني مذاڪري جي برڪت سان اُمّتِ مُسلمه جي خيرخواهي جو جذبو نصيب ٿيندو آهي۔ ٭ مدني مذاڪري جي برڪت سان

عاشقانِ رسول جي صحبت نصيب ٿيندي آهي۔

هن مدني ڪم جي تفصيلي معلومات ڄاڻڻ جي لاءِ رِسالو " هفتيوار مدني مذاڪره " جو مطالعو ڪريو۔ تمام ذِمه داران هن رِسالي کي حاصل ڪن، هن شعبي جي مدني ڪم کي مزيد مضبوط ۽ منظّم ڪرڻ جي لاءِ هن رِسالي ۾ ڏنل طريقيڪار جو عملي طور تي نفاذ ڪريو ۽ ڪرايو۔

اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! هفتيوار مدني مذاڪري جي علاوه مختلف موقعن تي به مدني مذاڪرن جو سلسلو ٿيندو رهندو آهي، جيئن ته ربيع الاوّل جي مدني مذاڪرن جي ڌوم ڌام هئي، اهڙي طرح ماه فاخِر، ماه عبدالقادِر، ماه ربيع الآخر ۾ به ڪم وبيش 11 ڏينهن مسلسل مدني مذاڪرن جو سلسله ٿيندو۔ اچو! اسان هاڻي کان نيت ڪريون ٿا ته ربيع الآخر جي مدني مذاڪرن ۾ پاڻ به شرڪت جي سعادت حاصل ڪنداسين ۽ ٻين کي به ان جي دعوت ڏينداسين۔ اِنْ شَــآءَالـلّٰـه عَزَّوَجَلَّ

اچو! تَرغيب لاءِ هڪ مَدَني بَهار ٻڌو ۽ مدني مذاڪري ۾ شرڪت ڪرڻ جي نِيت ڪريو۔ جيئن

گناهن کان توبه ڪري ورتي

واه ڪينٽ (پنجاب) جو هڪ اسلامي ڀاءُ ٻين نوجوانن جيان ڪيترين ئي بُرائين ۾ ڦاٿل هو۔ فلمون ڊراما ڏسڻ، راند روند ۾ وَقت برباد ڪرڻ ان جو محبوب مشغلو هو۔ هڪ ڏينهن ان جي گھر ۾ مدني چينل هلڻ جي برڪت سان مدني مذاڪرو ڏسڻ جي سعادت نصيب ٿي ، اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! مدني مذاڪرو ڏسڻ جي برڪت سان ان جي اڳوڻن گناهن کان توبه نصيب ٿي ۽ هو فرض ۽ واجبات جو عامِل بڻجڻ جي ڪوشش ۾ لڳي ويو، چهري تي هڪ مُٺ ڏاڙھي مُبارڪه سجائي ورتي۔ الله پاڪ جو وڌيڪ فضل ۽ ڪرم اهو ٿيو ته ان جي والدين خوشي سان ان کي دعوتِ اسلامي جي مدني ڪمن جي لاءِ ”وقفِ مدينه“ ڪري ڇڏيو۔

تمہیں لطف آ جائے گا زندگی کا قریب آکے دیکھو ذرا مدنی ماحول

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اِمدادِ مُصطَفٰي جو باب گھڻو وسيع ۽ ڪشادو آهي، نبين جي سُلطان، مَحبوبِ رحمٰن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي مشڪل ڪشائي جو اهو فَيضان صِرف انسانن تائين ئي مَحدود ناهي، ان ڪري جو پاڻ سڳورا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم صِرف انسانن ئي جا نبي نه آهن بلڪه تمام مخلوقات ڏانهن نبي بڻائي موڪليا ويا آهن، تنهنڪري جھڙي طرح پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پنهنجي ڏکويل اُمَّتين جا فَرياد ٻڌن ٿا ۽ الله پاڪ جي عطا سان انهن جي مُشڪل ڪشائي فرمائن ٿا، اهڙي طرح پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم مصيبتن ۾ڦاٿل جانورن، پکين ايستائين جو بي جان شين جو به فرياد ٻڌن ٿا، انهن جون ٻوليون سمجھن ٿا ۽ اُنهن جي دَست گِيري به ڪن ٿا،

مشهور مُفسِّرِ قُرآن،حڪيم الاُمَّت مفتي احمد يار خان نعيمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: حُضورِ انور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم پاڻ ته عَرَبي ڳالهائين ٿا پر سموريون زبانون به سمجھن ٿا، ايستائين جو جانورن جون ٻوليون به سمجھن ٿا، ان ڪري اُٺن، جھرڪين (Sparrows) حُضورِ انور، شاھ بحروبر صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي بارگاھ ۾ فرياد ڪيو ۽ انهن جي فرياد رسي ڪياؤن۔ (مرآة المناجيح،۸/۱۱۹بتغير قليل) هڪ ٻي هنڌ فرمائن ٿا:(حُضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي) پٿر سلام ڪندا هئا، ڪاٺي جو ٿڙ استنِ حَنَّانَه حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) جي فِرَاق (يعني پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي جُدائي) ۾ روئي پيو، پاڻ (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) کي دل جو ڏُک دَرد چيائين ۽ حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) سڀ ڪجھ سمجھي ورتو۔

اڄ حضور اڪرم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي درٻار ۾ هر شخص پنهنجي ٻولي ۾ حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) سان فرياد ڪري ٿو، ڪو به وچ ۾ ترجمو ڪرڻ وارو ناهي هوندو سڀني جي ٻڌن ٿا سمجھن ٿا، سڀني جي فرياد رَسِي ڪن ٿا، اهو آهي حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) جو سڀ زبانون ڄاڻڻ جو ثُبُوت۔ (مرآة المناجيح،۸/۳۸) مزيد

فرمائن ٿا: حضرت (سَيدُنا) سُلَيمان (عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام) صِرف جھرڪين، ڪيلين جي ٻولي سمجھندا هئا، حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) شَجَر و حَجر، خُشڪ و تَر تمام مخلوق جي ٻولي ڄاڻن ٿا، حُضور (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) حاجت رَوا، مُشڪل ڪشا آهن۔ هي اهو مسئلو آهي جنهن کي جانور به مڃن ٿا۔ (مرآة المناجيح، ۸/۲۳۹)

هر ڻ يءَ جو فرياد

حضرت سَيدُنا زَيد بن اَرْقَم رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا ته هڪ ڀيري مان رَحمتِ عالَم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان گڏ مديني جي گھٽين مان گذري رهيو هئس، اسان هڪ اَعرابي جي خيمي وٽان لنگھياسين، جنهن وٽ هڪ هرڻي به ٻڌل هئي۔ هرڻي female deer)) عرض ڪيو: يارَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! هي خيمي وارو اَعرابي مون کي شِڪار ڪري کڻي آيو آهي، جڏهن ته منهنجا ٻه (2) ٻچا جنگل ۾ موجود آهن، منهنجي ٿڻن ۾ کير گهاٽو ٿي رهيو آهي هي نه وري مون کي ذَبح ڪري ٿو ته جيئن مان هن تڪليف مان نجات ماڻيان ۽ نه ئي مون کي ڇڏي ٿو ته جهنگ ۾ پنهنجي ٻارن کي کير پياري اچان۔ هرڻي جو فرياد ٻڌي سَروَرِ معصوم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: جيڪڏهن مان تون کي ڇڏي ڏيان ته ڇا تون موٽي واپس ايندينءَ؟ عرض ڪيائين: جي ها! جيڪڏهن مان ائين نه ڪيان ته الله پاڪ مون کي (ناجائز) ٽيڪس وُصول ڪرڻ واري جهڙو عذاب ڏي۔ پوءِ حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي ڇڏي ڏنو، اها وڏي تيزي ۽ بيقراري سان جهنگ ڏانهن هلي وئي، اڃا ٿوري ئي دير گذري هئي ته اها خوشي خوشي واپس اچي وئي۔ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي خيمي ساڻ ٻڌي ڇڏيو۔ ايتري ۾ اهو اَعرابي به پاڻي جي مشڪ کڻي بارگاھِ رسالت ۾ حاضر ٿي ويو۔ نبي ڪريم، رؤف و رحيم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان کي فرمايائون: ڇا اها هرڻي اسان کي وڪڻندو؟ ان عَرض ڪيو: يا رَسُولَ الله صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم! هي هديي طور پيشِ خدمت آهي۔ ته پوءِ پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم هرڻي کي آزاد

اُ ٺ جو فَرياد

حضرت سَيدُنا يعليٰ بن مُرَّه رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: هڪ ڏينهن اسان حُضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سان گڏجي سفر تي وڃي رهيا هئاسين، اوچتو اسان جو گذر هڪ اهڙي اُٺ وٽان ٿيو، جنهن تي پاڻي جا مشڪيز لَڏيا پئي ويا، جڏهن اُٺ حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم کي ڏٺو ته اهو آواز ڪرڻ لڳو ۽ پنهنجي ڪنڌ کي جُھڪائي ورتو، حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان وٽ تشريف کڻي ويا ۽ فرمايائون: هن اُٺ جو مالڪ ڪٿي آهي؟ (ايتري ۾) اهو اچي ويو، حُضُور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايائون: توهان هن کي مون کي وڪڻي ڇڏيو، ان عرض ڪيو بلڪ هي مان اوهان کي تحفي طور پيش ڪيان ٿو، پاڻ سڳورن صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: بلڪ ان کي وڪڻي ڇڏيو، ان عرض ڪيو بلڪ تحفي طور پيش آهي، البت اهو اُٺ هڪ اهڙي گھراڻي جو آهي جو جنهن جي روزگار جو ذريعو ان کان ٻيو ڪو ناهي، حُضور صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: بهرحال جڏهن توهان ان جو مُعَامَلو ذِڪر ڪري ئي ورتو آهي ته ٻڌو! هن ڪم گھڻو وٺڻ ۽ گاهه گھٽ ڏيڻ جي شڪايت ڪئي آهي، تنهنڪري هن سان سٺو سُلوڪ ڪندا ڪيو۔ (دلائل النبوة،ذڪر المعجزات الثلاث التي شهدن …الخ،۶/۲۳)

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([3])

﴿4﴾ ڇ ھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([1])

﴿5﴾ قُربِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَي ۡ هِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪڙو ماڻهو آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته هي اهڙي مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([2])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([3])

﴿ 1 ﴾ هڪ هزار ڏ ينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([1])

﴿ 2 ﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
