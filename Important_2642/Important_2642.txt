اَلۡـحَمۡدُ لِلہِ رَبِّ الۡـعٰلَمِیۡنَ وَ الـصَّلٰوۃُ وَ السَّلَامُ عَـلٰی سَـیِّـدِ الۡـمُرۡ سَلِیۡنَ اَمَّا بَـعۡـدُ فَـاَعُوۡذُ بِا للہِ مِنَ الـشَّیۡطٰنِ الرَّ جِیۡمِؕ بِسۡمِ اللہِ الرَّحۡمٰنِ الرَّحِیۡمِؕ

فيضان جشن ولادت

* اَيَّامُ الله (الله جي ڏينهن) کي ياد ڪرڻ جو حڪم قرآن شريف ۾ آهي (پ: 13، ابراهيم: 5 ماخوذا) ۽ اَيَّامُ الله (الله جي ڏينهن) ۾ سڀ کان مقدس ”نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ميلاد جو ڏينهن“ آهي ڇو ته سڀئي ڏينهن هن مبارڪ ڏينهن جي سبب بابرڪت. ٿيا * قرآن ڪريم اسان کي رحمت الٰهي تي خوشي ملهائڻ جو حڪم ڏنو آهي. (پ: 11، يونس: 58) ۽ قرآن ڪريم ئي نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن کي سڀني جهانن جي لاءِ رحمت الهٰي ٻڌايو آهي. (پ 17، الانبياءِ 107) معلوم ٿيو ته جشن ولادت جي خوشي ملهائڻ در اصل قرآن جي حڪم تي عمل ڪرڻ آهي. (اذاقة الانام لمانعي عمل الموندو والقيام، ص 96) * اسان جا پيارا آقا مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سومر شريف تي روزو رکي پنهنجي ولادت جو ڏينهن ملهائيندا هئا، جيئن ته حضرت سيِّدنا ابو قتاده رَضِىَ اللهُ تَعَالٰی عَـنْهُ فرمائن ٿا: بارگاههِ رسالت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ۾ سومر جي روزي جي باري ۾ پڇيو ويو ته ارشاد فرمايائون: انهيءَ ڏينهن منهنجي ولادت ٿي ۽ انهيءَ ڏينهن مون تي وحي نازل ٿي. (صحيح مسلم شريف ص 591 حديث 198(1162) * جڏهن ابو لهب مري ويو ته ان جي ڪجهه گهر جي ڀاتين ان کي خواب ۾ بري حال ۾ ڏٺو. پڇا ڪيو: ڇا مليو؟ چيائين: توهان کان جدا ٿي ڪري مون کي ڪو خير نصيب ناهي ٿيو. پوءِ پنهنجي آڱوٺي جي هيٺان موجود سوراخ ڏانهن اشارو ڪندي چوڻ لڳو: سواءِ هن جي جو مونکي هن مان پاڻي پياريو ويندو آهي ڇو ته مون ثويبه ٻانهيءَ کي آزاد ڪيو هو . (مصنف عبدارزاق جلد 9 ص 9 الحد يث 16661، عمدة القاري ج 14 ص 44 تحت الحديث 5101) * هن روايت جي تحت سيدنا شيخ عبدالحق محدث دهلوي عَـلَيْهِ رَحْـمَةُ الـلّٰـهِ الۡـقَـوِی فرمائن ٿا: هن واقعي ۾ ميلاد شريف وارن لاءِ وڏي دليل آهي جيڪي تاجدار رسالت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ولادت واري رات ۾ خوشيون ملهائيندا ۽ مال خرچ ڪندا آهنـ يعني ابولهب جيڪو ڪافر هو جڏهن اهو تاجدار نبوت صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي ولادت جي خبر ٻڌي خوش ٿيڻ ۽ پنهنجي ٻانهي (ثويبه) کي کير پيارڻ جي سبب آزاد ڪرڻ جو بدلو ڏنو ويو ته ان مسلمان جو ڇا حال هوندو جيڪو محبت ۽ خوشين سان ڀريل آهي ۽ مال خرچ ڪري رهيو آهي، پر هي ضروري آهي ته ميلاد شريف جي محفل گانن باجن ۽ موسيقيءَ جي آوازن کان پاڪ هجي . (مدارج النبوة ج 2 ص 19) * جشنِ ولادت ڇا آهي؟ جشنِ ولادت اهو مبارڪ اجتماع آهي جنهن ۾ معتبر ڪتابن مان نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جا فضائل ۽ مُعجزات، سيرت (خصلتون) حالات، زندگي، رضاعت ۽ بعثت جا واقعات بيان ڪيا وڃن . (سبل الهدي الرشاد، الباب الثالث عشر في اقوال العلماء..... الخ، 363،1 ماخوذا) * صَدرُالشريعه مُفتي محمد اَمۡجَد علي اعظمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: انهن شين (يعني نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جا فضائل ۽ معجزات وغيره) جو ذڪر حديثن ۾ به آهي ۽ قرآن مجيد ۾ به آهي. جيڪڏهن مسلمان پنهنجي محفل ۾ بيان ڪري بلڪه خاص انهن ڳالهين کي بيان ڪرڻ جي لاءِ محفل منعقد ڪري ته هن جي ناجائز ٿيڻ جي ڪو سبب ناهي. هن محفل جي لاءِ ماڻهن کي گهرائڻ ۽ شريڪ ڪرڻ ڀلائيءَ جي طرف سڏڻ آهي، جهڙي طرح وعظ ۽ جلسن جا اعلان ڪيا ويندا آهن. اشتهارات ڇپائي ڪري تقسيم ڪيا ويندا آهن. اخبارن ۾ هن جي متعلق مضمون شائع ڪيا ويندا آهن ۽ ان جي سبب کان اهي وعظ ۽ جلسا ناجائز ناهين ٿيندا. اهڙي طرح ذڪر پاڪ جي لاءِ سڏڻ ان محفل کي ناجائز ۽ بدعت نٿو چئي سگهجي. (بهار شريعت، 644،4) * جشن ولادت ملهائڻ جو حڪم: ”اهلسنّت جي مذهب ۾ مجلسِ ميلادِ پاڪ افضل ترين مُستحبات ۽ اعلٰي ترين مُسۡتَحۡسَنات (نيڪ ڪمن) مان آهي (الحق المبين، ص 100) جشنِ ولادت ملهائڻ جون فضيلتون ۽ برڪتون: جَشنِ ولادت ملهائڻ وارن کي الله عَزَّوَجَلَّ جي طرفان بيشمار ديني ۽ دنياوي رحمتون ملنديون آهن جيئن ته حضرت سيّدنا امام عبدالرحمٰن ابن جوزي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائين ٿا: جشنِ ولادت تي خوشيءِ ۽ مسرت ڪرڻ وارن جي لاءِ هي خوشي جنّهم کان رڪاوٽ بڻجندي، جيڪو جشنِ ولادت جي خوشي ۾ هڪ درهم خرچ ڪري ته نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم ان جي شفاعت فرمائيندا ، الله عَزَّوَجَلَّ ان جي هڪ درهم جي بدلي ڏهه درهم عطا فرمائيندو. اي امتِ محبوب! توهان جي لاءِ خوشخبري هجي توهان دنيا ۽ آخرت ۾ ڪثير ڀلائيءَ جا حقدار آهيو. حضرت سيّدنا احمد مجتبيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جو جشنِ ولادت ملهائڻ وارن کي برڪت، عزت، ڀلائي ۽ فخر ملندو، موتين جو عمامو ۽ سبز جنتي

لباس پائي جنّت ۾ داخل ٿيندو، بيشمار محلات ان کي عطا ڪيا ويندا ۽ هر محل ۾ حور هوندي. نبي خيرالانام صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن تي گهڻو دُرُود شريف پڙهو. جشنِ ولادت ملهائي ان کي تمام گهڻو عام ڪيو وڃي . (مجموع لطيف النبي في صيغ المولد النبوي القدسي، مولد العروس، ص 281) * حافظُ الحديث امام ابو الخير محمد بن عبدالرحمٰن سخاوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جشنِ ولادت ملهائڻ وارن تي هن عمل جي برڪتن سان فضلِ عظيم ظاهر ٿيندو آهي. (سبل الهدي والرشاد، الباب الثالث عشر في اقوال العلماء..... الخ، 363،1) حضرت سيّدنا امام احمد بن محمد قسطلاني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: ”ولادتِ باسعادت جي ڏينهن ۾ محفل ميلاد ڪرڻ جي فاعدن سان هي امر مُجرَّب (يعني تجربي شده) آهي ته هن سال امن امان رهندو آهي. الله عَزَّوَجَلَّ ان شخص تي رحمت نازل فرمائي جنهن ولادت جي مهيننجي راتين کي عيد بڻائي ڇڏيو.“ (مواهب الدنية، المقصد الاول، ذڪر رضاعة، 148،1). * جَشنِ ميلاد ملهائڻ وارن جي لاءِ بزرگن جي ڪلام ۾ جنّت جي بشارت به آهي جيئن ته شيخ محقق شاهه عبدالحق مُحَدّث دهلوي عَـلَيْهِ رَحْـمَةُ الـلّٰـهِ الۡـقَـوِی فرمائن ٿا: سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي ولادت جي رات خوشي ملهائڻ وارن جي جزا هي آهي ته الله تعاليٰ انهن کي پنهنجي فضل ۽ ڪرم سان جَنَّاتُ النَّعِيم ۾ داخل فرمائيندو، مسلمان هميشه کان محفل ميلاد منعقد ڪندا آيا آهن ۽ ولادت جي خوشيءَ ۾ دعوتون ڏيندا، کاڌا پڪرائيندا ۽ خوب صدقا ۽ خيراتون ڏيندا آيا آهن. خوب خوشيءَ جو اظهار ڪندا آهن ۽ دل کولي خرچ ڪندا آهن ۽ پڻ سرڪارمدينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي ولادت با سعادت جي ذڪر جو اهتمام ڪندا آهن ۽ پنهنجن گهرن کي سينگاريندا آهن ۽ انهن تمام سٺن ڪمن جي برڪت سان انهن ماڻهن تي الله تعاليٰ جي رحمتين جو نزول ٿيندو آهن . (ماثبت باالسنة ص 74 ملتقطاً) علماء ۽ محدثين جا فرامين: * حضرت سيّدنا امام عبدالرّحمٰن ابن جوزي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: هن فعل (يعني جشنِ ولادت ملهائڻ) ۾ شيطان جي تذليل ۽ اهل ايمان جي تقويت جي سواءِ ڪجهه ناهي . (سبل الهدي الرشاد، الباب الثالث عشر في اقوال العلماء..... الخ، 363،1) * امام جلال الدين عبدالرحمٰن سيوطي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جشنِ ولادت ملهائڻ وارو ثواب حاصل ڪندو آهي جو ان ۾ نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي تعظيم ۽ ولادتِ با سعادت تي خوشي ۽ شادماني جو اظهار آهي. (الحاوي للفتاوي، حسن المقصد في عمل المولد، 222،1) * شارحِ بخاري حضرت سيّدنا احمد بن حجر عسقلاني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: اسان جي حق ۾ مستحب آهي ته جشنِ ولادت تي اظهارِ شڪر جي لاءِ اجتماع ڪريون. کاڌو کارايون ۽ اهڙي طرح جون ٻيون نيڪيون ۽ اظهار خوشي ۽ سرور آڻيون. (الحاوي للفتاوي، حسن المقصد في عمل المولد، 230،1) * حضرت سيّدنا ابو موسٰي زرهوني رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي خواب ۾نبي ڪريم صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن جي زيارت ٿي ته پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ميلاد جي خوشي ۾ اهتمام ڪرڻ جي متعلق عرض ڪيو ته سرڪار صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم فرمايو: مَنْ فَرِحَ بِنَا فَرِحنَا بِہ (يعني جيڪو اسان کان خوش ٿيندو آهي اسين ان کان خوش ٿيندا آهيون. (سبل الهدي والرشاد، الباب الثالث عشر في اقوال العلماء.... الخ، 363،1) * امير اهلسنّت باني دعوت اسلامي حضرت علامه مولانا ابو بلال محمد الياس عطار قادري دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جن جو رسالو ”صبح بهاران“ مان ڪجهه منتخب مدني گل ملاحظه ڪريو: (1) اسان جا پيارا آقا مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم سومر شريف تي روزو رکي پنهنجي ولادت جو ڏينهن ملهائيندا رهيا. توهان به ياد مصطفٰي ۾ 12 ربيع الاول شريف جو روزو رکو. (2) يارهين جي شام نه ته ٻارهينءَ جي رات غسل ڪيو، ٿي سگهي ته هن عيدن جي عيد جي تعظيم جي نيت سان پنهنجي استعمال جي هرشيءِ ممڪن هجي ته نئي وٺو. (3) جشن ولادت جي خوشي ۾ ڪو به اهڙو ڪم نه ڪيو جنهن سان حقوق العباد ضايع ٿئي. (4) سڀئي اسلامي ڀائر ۽ اسلامي ڀينرون جشنِ ولادت جي خوشِ ۾ سنّتن جي مطابق زندگي گذارڻ جي نيّت ڪن. (5) جشن ولادت جي خوشي ۾ مسجدن، گهرن، دڪانن ۽ پنهنجي پاڙي ۾به ساوا ساوا جهنڊا لڳايو، خوب روشني ڪيو، پر بجلي فراهم ڪندڙ اداري سان رابطو ڪري جائز ذريعي سانروشني ڪرڻ جي ترڪيب ڪريو. (6) جلوس ۾ حتي الامڪان با وضو رهو ۽ باجماعت نماز جي پابنديءَ جو خيال رکو. (وڌيڪ معلومات جي لاءِ امير اهلسنّت حضرت علامه مولانا ابو بلال محمد الياس عطاري قادري دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـهْ جو رسالو صبح بهاران جو مطالعو ڪريو)

نيڪين تي استقامت پائڻ ۽ سنّتن ڀري زندگي گذارڻ جي لاءِ دعوت اسلامي جي مدني ماحول سان وابسته ٿي وڃو.
