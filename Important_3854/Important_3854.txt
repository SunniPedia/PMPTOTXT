آٓفتن، پريشانين ۽ مصيبتن ۾ گرفتار ٿي ويندو آهي۔ ٭ لالچ جي ڪري انسان ڪيئي نعمتن کان محروم ٿي ويندو آهي۔ ٭ لالچ پوڙهائپ تائين به جان نٿي ڇڏي۔ ٭ لالچ جي گندگي سان بُزرگانِ دِين رَحْمَةُ اللہ تَعَالٰی عَلَيْهم اَجْمَعِين جو دامن ڪڏهن داغ دار نه ٿيو۔ ٭ لالچ جي ڪري انسان ڪيئي بُرائين ۾ مُبتَلا ٿي ويندو آهي۔ ٭ لالچ خون خرابو ڪرائيندي آهي۔ ٭ لالچ جي ڪري انسان ٻين کي ڌوڪو ڏيندو آهي۔ ٭ لالچ اللہ پاڪ جي ياد کان غافل ڪندي آهي۔ ٭ لالچ چڱي صحبت کان محروم ڪندي آهي۔ ٭ لالچ حلال ۽ حرام جي تميز وساري ڇڏيندي آهي۔ اللہ ڪريم اسان کي لالچ جهڙي بري آفت کان محفوظ ڪري نيڪين جي حِرص جي لازوال نعمت سان سرفراز فرمائي۔

اٰمِين بِجَاه النَّبِي الْاَمِين صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

تَوَڪل ۽ قَناعت جا مَدَني گل

مٺا مٺا اسلامي ڀائرو! بَيان کي اِختِتام ڏانهن آڻيندي تَوَڪل ۽ قَناعت جا ڪجھ مَدَني گل ٻڌڻ جي سعادت حاصِل ڪريون ٿا، پهريان ٻه فرمانِ مُصطَفٰي صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مُلاحَظَه ڪريو: (1) قَناعت ڪڏهن خَتم نه ٿيندڙ خزانو آهي۔ (الزھد الڪبير،ص۸۸،حديث:۱۰۴) (2) بيشڪ ڪامياب ٿي ويو اهو شخص جنهن اِسلام آندو اُن کي بَقَدرِ ڪفايت (ضرورت مطابق) رزق ڏنو ويو ۽ اللہ عَزَّوَجَلَّ اُن کي جيڪو ڪجھ ڏنو اُن تي قناعت به عطا فرمائي۔ (مسلم،ڪتاب الزڪاة،باب في الڪفاف والقناعة، ص۴۰۶، حديث:۱۰۵۴) ٭ انسان کي جيڪو ڪجھ اللہ پاڪ جي طرفان مِلي ان تي راضي ٿي ڪري زندگي گذاريندي حِرص ۽ لالچ کي ڇڏي ڏيڻ کي قَناعت چئبو آهي۔ (جنتي زيور،ص۱۳۶ملخصاً) ٭ روزمَرّه اِستِعمال ٿيڻ وارين شين جي نه هئڻ تي به راضي رهڻ قَناعت آهي۔ (التعريفات للجرجاني،باب القاف،تحت اللفظ:القناعة، ص۱۲۶) ٭ شيخ ابو علي دَقَّاق رَحْمَةُ اللہ تَعَالٰی عَلَيْه فرمائن ٿا: تَوَڪل جا ٽي (3) دَرَجا آهن: (1) اللہ پاڪ جي ذات تي ڀروسو ڪرڻ، (2) ان جي حڪم اڳيان سَرِ تسليمِ خَم ڪرڻ ۽ (3)

پنهنجو هر مُعامَلو ان جي حوالي ڪرڻ۔ (رساله قشيرية،ص۲۰۳) ٭ دُنياوي شين ۾ قَناعت ۽ صَبر چڱو آهي پر آخِرَت جي شين ۾ حِرص ۽ بي صَبري اعليٰ آهي، دِين جي ڪنهن دَرَجي تي پهچي قَناعت نه ڪريو اڳتي وڌڻ جي ڪوشِش ڪريو۔ (مرآة المناجيح،۷/۱۱۲) ٭ لالچ نهايت ئي بُري خَصلَت ۽ نِهايت خراب عادت آهي، اللہ پاڪ جي طرفان ٻانهي کي جيڪو رِزق ۽ نعمت ۽ مال ۽ دَولت يا عزّت ۽ رتبو مِليو آهي، انهيء تي راضِي ٿي قَناعت ڪرڻ گھرجي۔ (جنتي زيور،۱۱۰ ملخصاً) ٭ جنهن جون لالچ ڀريون نگاهون ماڻهن جي مال کي ڏسنديون رهن اهو هميشه غمگين رهندو۔ (رساله قشيرية،۱۹۸) ٭ بَلعَم بن باعُوراء جيڪو تمام وڏو عالِم ۽ مُستَجَابُ الدَّعَوات هو، حِرص ۽ لالچ ان کي دنيا ۽ آخرت ۾ تباھ ۽ برباد ڪري ڇڏيو۔ (ملفوظات اعليٰ حضرت،ص۳۶۷ ماخوذاً) ٭ اللہ پاڪ فرمائي ٿو: اهو شخص منهنجي ويجھو سڀ کان وڌيڪ مالدار آهي جيڪو منهنجي ڏنل شين تي سڀ کان وڌيڪ قَناعت ڪرڻ وارو آهي۔ (ابنِ عساڪر ،موسيٰ بن عمران بن يصھر بن قاھث،۶۱/ ۱۳۹ملخصاً)

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

طرح طرح جون هزارين سنتون سکڻ جي لاءِ مڪتبة المدينه جي شايع ٿيل ڪتاب 312 صفحن تي مشتمل ڪتاب ” بهار شريعت “ حصو 16 ۽ 120 صفحن جو ڪتاب “ سنتیں ۽ آداب ” هديو ڏئي حاصل ڪيو ۽ پڙهوـ سنتن جي تربيت جو هڪ بهترين ذريعو دعوت اسلامي جي مدني قافلن ۾ عاشقان رسول سان گڏ سنتن ڀريو سفر آهي .

مجھ کو جذبہ دے سفر کرتا رہوں پَروَردگار سُنّتوں کی تَرْبِیَت کے قافِلے میں بار بار

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ6 درود شريف ۽ 2 دعائون

﴿1﴾ جمعه جي رات جو دُرُود

اَللّٰھُمَّ صَلِّ وَسَلِّمْ وَبَارِكْ عَلٰی سَیِّدِنَا مُحَمَّدٍ النَّبِیِّ الْاُمِّیِّ الْحَبِیْبِ الْعَالِی الْقَدْرِ الْعَظِیْمِ الْجَاہِ وَ عَلٰی اٰلِہٖ وَ صَحْبِہٖ وَ سَلِّمْ

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ ۾ گھٽ هڪ ڀيرو پڙهندو موت جي وقت سرڪار مدينه صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيڻ وقت به ايستائين هو ڏسندو ته سرڪار مدينه صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن . ([1])

﴿2﴾ سمورا گُناھ مُعاف

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

حضرتِ سَيِّدُنا انس رَضِى اللہ تَعَالٰى عَنْه کان روايت آهي ته تاجدارِ مدينہ صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: جيڪو شَخص هي دُرُود شريف پڙهي جيڪڏهن بيٺل هو تو ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا ويندا . ([2])

﴿3﴾ رَحمت جا ستّر دروازا

صَلَّی اللّٰہُ عَلٰی مُحَمَّدٍ

جيڪو هي دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا

ويندا آهن . ([1])

﴿ 4 ﴾ ڇھ لک دُرُودشريفن جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِنَا مُحَمَّدٍ عَدَدَ مَا فِیْ عِلْمِ اللّٰہِ صَلَاۃً دَ اۤ ئِمَۃًۢ بِدَوَامِ مُلْكِ اللّٰہِ

حضرت اَحمد صاوِي عَلَيہِ رَحمَۃُ اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي . ([2])

﴿ 5 ﴾ قُربِ مُصطَفٰي صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي . ([3])

﴿ 6 ﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي . ([4])

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى اللہ تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن . ([1])

﴿2﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، اللہ عَزَّ وَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللہ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو . (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
