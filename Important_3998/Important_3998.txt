امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جا والد سڳورا تمام وڏا ممتاز بُزرگ ۽ زَبَردَست عالِمِ دين هئا، پاڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه امام بُخاري جي اُستاد حضرت سَيدُنا عَبدُ اللہ بِن مبارَڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي صُحبت ۾ رهندا هئا، پاڻ صاحبِ روايت مُحَدِّث هئا، پاڻ حضرت سَيدُنا عَبدُ اللہ بن مبارَڪ، حضرت سَيدُنا امام مالِڪ رَحْمَةُ اللهِ تَعَالٰی عَلَيْهما ، انهن جي شاگردن ۽ اُن زماني جي مُحَدِّثن کان روايت ڪندا هئا، نهايت ئي مُستَجَابُ الدَّعوَات بزرگ هئا (يعني سندن دعائون تمام گھڻيون مقبول ٿينديون هيون) ، ائين جو بارگاھ خداوندي ۾ عَرض ڪندا هئا ته منهنجون سڀ دعائون دنيا ئي ۾ نه قَبول ڪر ڪجھ آخرت جي لاءِ رهڻ ڏي، حلال کائڻ جا اهڙا پابند هئا جو حرام ته رهيو پري، پر مشڪوڪ شين کان به بچندا هئا، ايستائين جو وِصالِ ظاهري وَقت فرمايائون: مون وٽ جيترو به مال آهي اُن ۾ هڪ دِرهم به شڪ ۽ شُبهي وارو نه آهي۔ (نزهۃ القاري۱/۱۰۷،ملخصاً)(ارشاد الساري، ترجمۃ الامام بخاري، ۱/۵۵)

حضرت محمد بن اَبِي حاتِم ۽ حاشِد بن اِسماعيل رَحْمَةُ اللهِ تَعَالٰی عَ لَيْهما بَيان ڪن ٿا

ته حضرت سَيدُنا امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ننڍي عُمر ۾ اسان سان گڏ حديث جي لاءِ بَصره جي مشائخن جي خدمت ۾ حاضِر ٿيندا هئا، امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي علاوه اسان تمام ساٿي حديثن کي محفوظ ڪرڻ جي لاءِ لکندا هئاسين، سورهن ڏينهن (Sixteen Days) گذري وڃڻ کان پوءِ هڪ ڏينهن اسان امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کي چيو ته توهان حديثون محفوظ نه ڪري ايترن ڏينهن جي محنت ضائع ڪري ڇڏي۔ اهو ٻڌي امام بخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه اسان کي چيو: چڱو توهان پنهنجا لکيل صَفحا کڻي اچو، پوءِ اسان پنهنجا پنهنجا صَفحا کڻي آياسين، امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه حديثون ٻڌائڻ شروع ڪيون، ايستائين جو اُنهن پندرهن هزار (15000) کان وڌيڪ حديثون بَيان ڪري ورتائون، جن کي ٻڌي اسان کي اهو گُمان ٿيندو هو ته ڄڻ اسان کي اهي روايتون امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ئي لکرائيون آهن۔ (ارشاد الساري،ترجمۃ الامام بخاري ،۱/۵۹)

خَتمِ بُخاري جا فائدا

(ڪجهه)عارف سڳورن رَحْمَةُ اللهِ تَعَالٰی عَلَيْهم اَجمَعِين کان منقول آهي: جيڪڏهن ڪنهن مشڪل ۾ صحيح بخاري کي پڙھيو وڃي ته اها مشڪل حل ٿي ويندي آهي ۽ جنهن ڪشتي (Boat) ۾ صحيح بُخاري هجي اها غَرق ناهي ٿيندي۔ حافظ اِبنِ ڪثير رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: ڏڪار ۾ ”صحيح بُخاري“ پڙھڻ سان برسات ٿيندي آهي۔ (تذڪرۃ المحدثين، ص۱۹۸) مشهور مُحَدِّث شيخ عبدُ العزيز مُحَدِّثِ دِهلَوِي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: شِدَّت جي وقت، دشمن جي خوف، مَرَض جي سختي، ڏڪار ۽ ٻين مصيبتن ۾ هن ڪتاب جو پڙھڻ ترياق (علاج) جو ڪم ڏيندو آهي، اڪثر هن جو تَجرِبو ٿي چڪو آهي۔ (بستان المحدثين،ص۲۷۴) حڪيمُ الاُمّت حضرت مفتي احمد يار خان نعيمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: قرآن شريف کان پوءِ صحيح تر ين ڪتاب بُخاري شريف مڃيو ويو آهي، مصيبتن ۾ خَتمِ بُخاري ڪيو ويندو آهي، جنهن جي برڪت ۽ اللہ پاڪ جي فَضل سان مصيبتون ختم ٿي وينديون آهن۔ (مرآۃ المناجيح،۱/۱۱ ملخصاً)

علامه تاجُ الدِّين سُبڪي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي وِصالِ ظاهري جي (200 سال) کان پوءِ سَمَرقَند ۾ خشڪ سالي جي ڪري ڏڪار ٿيو، ماڻهن ڪيئي ڀيرا نماز اِستِسقاء پڙھي ۽ دعائون گھريون پر برسات نه ٿي، پوءِ هڪ نيڪ شخص جيڪو زُهد وتَقويٰ ۽ پرهيزگاري جي ڪري مشهور هو، شهر جي قاضي وٽ ويو ۽ ان کي مشورو ڏنائين ته توهان شهر جي ماڻهن کي وٺي امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي قَبر تي وڃو ۽ اتي وڃي اللہ پاڪ کان برسات جي دعا گھرو، شايد اللہ پاڪ توهان جي دعا قَبول ڪري۔ قاضي اهو مشورو قَبول ڪيو ۽ ماڻهن کي گڏ ڪري امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي قَبر تي آٓيو، امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي وسيلي سان دعائون گھريون ۽ گريه وزاري ڪئي، امام بُخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه کان قَبوليتِ دعا جي لاءِ سفارش ڪئي، نهايت ئي خُشوع وخُضوع سان ان دعا گهري، انهيءَ وَقت آسمان تي بادل ڇائنجي ويا ۽ ست ڏينهن (Seven Day) تائين مُسَلسَل برسات ٿيندي رهي ۽ ايتري برسات ٿي جو ماڻهن جي لاءِ مقام ”خَرتَنڪ“ کان ”سَمَرقَند“ تائين پهچڻ به مشڪل ٿي ويو۔ (طبقات الشافعيۃ الڪبري،الطبقۃ الثانيۃ،۲/۲۳۴) (خَرتَنڪ کان سَمَرقَند تائين فقط 3 مِيل جو فاصلو آهي۔)

حديثِ پاڪ جي باري ۾ مَدَني گل

مٺا مٺا اسلامي ڀائرو! بَيان کي اِختِتام ڏانهن آڻيندي حديثِ پاڪ جي باري ۾ ڪجھ مَدَني گل ٻڌڻ جي سعادت حاصِل ڪريون ٿا۔ پهريان 2 فرمانِ مُصطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ مُلاحَظَه ڪريو: (1) فرمايائون: جيڪو شخص ديني معامَلات جي باري ۾ چاليھ (40) حديثون ياد ڪري منهنجي اُمّت تائين پهچائيندو اللہ پاڪ (قِيامت جي ڏينهن) ان کي ان شان سان اٿاريندو جو اهو فقيه هوندو ۽ مان قِيامت جي ڏينهن ان جي شفاعت ڪندس ۽ ان جي لاءِ شاهدي ڏيندس۔) مشڪاۃ المصابيح، ڪتاب العلم، الفصل الثالث،۱/۶۸، حديث:۲۵۸ ((2) فرمايائون: اللہ پاڪ ان کي تَروتازه رکي جيڪو منهنجي حديث کي ٻڌي، ياد رکي ۽ ٻين تائين پهچائي۔ (تِرمِذي، ۴/۲۹۸، حديث:۲۶۶۵) ٭ اسلام ۾ ڪلامُ اللہ (يعني قرآن) کان بعد ڪلامِ رسولِ اللہ (يعني حديث) جو درجو آهي۔ (مرآۃ المناجيح،۱/۲) ٭ هر انسان تي حُضور عَلَيه السَّلَام جي اِطاعت فَرض آهي ۽ اها اِطاعت بغير حديث ۽ سُنّت ڄاڻڻ جي ناممڪن آهي۔ (مرآۃ المناجيح ،۱/۹) ٭ حديثن جي اِنڪار کان پوءِ قرآن تي اِيمان جي دعويٰ بلڪل باطِل آهي۔ (نزهۃ القاري ،۱/۶ ۳) ٭ جيستائين اهو معلوم نه ٿئي ته اها واقعي حديث مبارَڪه آهي، ان وَقت تائين بَيان نه ڪريون ۔ (فيضان فاروق اعظم ،۲/۴۵۱) ٭ آقا عَلَيه السَّلَام جو فرمان آهي: جيستائين توهان کي يقيني عِلم نه هجي منهنجي طرفان حديث بيان ڪرڻ کان بچو، جنهن ڄاڻي واڻي مون ڏانهن ڪوڙ منسوب ڪيو ان کي گھرجي ته هو پنهنجو ٺڪاڻو جهنم ۾ ٺاهي۔ (ترمذي، ڪتاب تفسير القرآن عن رسول الله ، باب ماجاء في الذي ۔۔۔الخ،۴/۴۳۹،حديث:۲۹۶۰)

٭ ايس ايم ايس (SMS) جي ذريعي بغير حوالي جي ڪا به حديث ان وقت تائين اڳتي نه موڪليو جيستائين ڪنهن سُنّي صحيحُ العقيده مفتي صاحب يا ڪنهن عالِمِ دين کان تصديق نه ڪرائي وٺو۔ (فيضان فاروق اعظم،۲/۴۴۰)

دعوتِ اسلامي جي هفتيوار سنتن ڀري

اِجتِماع ۾ پڙهيا ويندڙ (6)درود شريف

﴿ 1 ﴾ جمعه جي رات جو دُرُود

اَللّٰہُمَّ صَلِّ وَسَلِّم وَبَارِک عَلٰي سَيدِنَامُحَمَّدِ نِ النَّبِي الاُمِّي

الحَبِيبِ العَالِي القَدرِالعَظِيمِ الجَاہِ وَعَلٰي اٰلِہٖ وَصَحبِہٖ وَسَلِّم

بزرگن فرمايو آهي ته جيڪو شخص هر جمعرات (جمعه ۽ خميس جي وچين رات) هن دُرُود شريف کي پابندي سان گھٽ مان گھٽ هڪ ڀيرا پڙهندو موت جي وقت سرڪار مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جي زيارت ڪندو ۽ قبر ۾ داخل ٿيندي وقت اهو به ڏسندو ته سرڪار مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ ان کي پنهنجي هٿن سان قبر ۾ داخل ڪري رهيا آهن ، ([1])

﴿2﴾ سمورا گُناھ مُعاف

اَللّٰہُمَّ صَلِّ عَلٰي سَيدِ نَا وَمَولَانَا مُحَمَّدٍ وَّعَلٰي اٰلِہٖ وَسَلِّم

حضرتِ سيدُنا انس رَضِي اللّٰہُ تَعَالٰي عَنہُ کان روايت آهي ته تاجدارِمدينہ صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو:جيڪو شَخص اهو دُرُود شريف پڙهي جيڪڏهن بيٺل هو ته ويهڻ کان پهريان ۽ ويٺل هو ته بيهڻ کان پهريان، ان جا گناھ معاف ڪيا

ويندا. ([1])

﴿3﴾رَحمت جا ستّر دروازا

صَلَّي اللّٰہُ عَلٰي مُحَمَّد

جيڪو اهو دُرُود شريف پڙهندو آهي ان تي رحمت جا ستر دروازا کوليا ويندا آهن. ([2])

﴿4﴾ڇھ لک دُرُودشريف جو ثواب

اَللّٰھُمَّ صَلِّ عَلٰي سَيدِنَامُحَمَّدٍعَدَدَمَافِي عِلمِ اللّٰہ ِصَلَاة دَآئِمَة ۢبِدَوَامِ مُلکِ اللّٰہ

حضرت اَحمدصاوِي عَلَيہِ رَحمَة اللّٰہ ِالہَادِي ڪن بُزرگن کان نقل ڪن ٿا هن دُرُود شريف کي هڪ ڀيرو پڙهڻ سان ڇھ لک دُرُودشريف پڙهڻ جو ثواب حاصل ٿيندو آهي. ([3])

﴿5﴾قُربِ مُصطَفٰي صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ

اَللّٰہُمَّ صَلِّ عَلٰي مُحَمَّدٍکَمَا تُحِبُّ وَتَرضٰي لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَ سَلَّمَ ان کي پنهنجي ۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهي سبب صَحاب َ ه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو: اهو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته ائين پڙهندو آهي. (القول البدیع،الباب الاول،ص125)

﴿6﴾دُرُودِ شَفاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنۡزِلۡہُ الۡمَقۡعَدَ الۡمُقَرَّبَ عِنۡدَکَ یَوۡمَ الۡقِیَامَة

شافِعِ اُمَم صَلَّی اللّٰہُ تَعَالٰی عَلَیۡہِ واٰلِہٖ وَسَلَّمَ جو فرمانِ مُعَظَّم آهي جيڪو شخص هن درود شريف کي پڙهي ته ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي۔ ([1])

﴿1﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَي اللّٰہ ُ عَنَّا مُحَمَّدًا مَّا ھُوَ اَھلُہ ٗ

حضرتِ سيدُنا ابنِ عباس رَضِي اللّٰہُ تَعَالٰي عَنہُمَا کان رِوايت آهي ته سرڪارِمدينه صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ جن فرمايو؛ انهي درود شريف کي پڙهڻ واري جي لاءَ ستر ملائڪ هڪ هزار ڏينهن تي نيڪيون لکندا آهن ([2])

﴿1﴾ ڄڻ شب قدر حاصل ڪري ورتي ۔

فرمان مصطفيٰ صَلَّي اللّٰہُ تَعَالٰي عَلَيہِ واٰلِہٖ وَسَلَّمَ آهي جنهن هن دعا کي 3 ڀيرا پڙهيو ته ان ڄڻ شب قدر حاصل ڪري ورتي۔ ([3])

لَآ اِلٰہَ اِلَّااللہُ الۡحَلِیۡمُ الۡـکَرِیۡمُ ،سُبحٰنَ اللہ ِ رَبِّ السَّمٰوٰتِ السَّبۡعِ وَرَبِّ الۡعَرۡشِ الۡعَظِیۡم

(خداءِ حليم ۽ ڪريم کان سواءِ ڪو به عبادت جي لائق ناهي الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

صَلُّوۡا عَلَي الۡحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد
