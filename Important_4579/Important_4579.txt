اَمِيرِ اهلسنّت،باني دعوتِ اسلامي حضرت ِ علّامه مَولانا ابو بلال محمد الياس عطّار قادِري رَضَوِي ضِيائي پنهنجي رسالي " لڙڪن جي برسات " ۾ لکن ٿا: حضرت سيدنا امامِ اعظم رَحْمَةُ اللهِ تَعَالٰی عَلَيْه 70 هجري ۾ عراق جي مشهور شهر ڪوفي ۾ پيدا ٿيا ۽ 80 سالن جي عمر ۾ 2 شعبان المعظم 150 هجري ۾ وفات ٿي۔ (لڙڪن جي برسات، ص۲) اڄ به بغداد شريف جي قبرستان خيزران ۾ سندن مزارِ فائضُ الاَنوار موجود آهي۔ (تاريخِ بغداد ،۱۳/۳۲۵ مفهوماً) حضرت سيدنا امامِ اعظم رَحْمَةُ اللهِ تَعَالٰی عَلَيْه ڪجھ صَحابي سڳورن عَلَيۡهِمُ الرِّضْوَان سان مُلاقات جو شَرَف ماڻيو هو ۽ بعض صَحابَه ڪرام عَلَيۡهِمُ الرِّضْوَان کان براهِ راست پياري پياري آقا، مٺي مٺي مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جا اِرشاد به ٻُڌا آهن۔ (الخيرات الحسان، ص:۳۳ ملتقطاً)

۽ عمده ٻڌائي ٻين کي ڌوڪو ڏنو ويندو آهي۔ ٭ اسان جي مُعاشَري ۾ ٻين جي مجبورين مان فائدو کڻي انهن کي ڌوڪو ڏنو ويندو آهي۔ ٭ اسان جي مُعاشَري ۾ سٺي شيء ڏيکاري پوءِ هلڪي شيء ڏئي ٻين کي ڌوڪو ڏنو ويندو آهي۔ ٭ اسان جي مُعاشَري ۾ ماپ تور ۾ ڪمي ڪري ٻين کي ڌوڪو ڏنو ويندو آهي۔ اسان جي اَخلاقي حالت ته ايتري قدر تباھ ٿي چڪي آهي جو جيڪڏهن دڪان تي ويهڻ وارو اسان جو ٻار ڌوڪي وغيره سان ڪا شيء وڪڻڻ ۾ ڪامياب ٿي وڃي ته ان کي هڪ شاندار ڪارنامو سمجھيو ويندو آهي۔ ٻار کي شاباش ڏني ويندي آهي، ان جي پٺي ٿپڪائي ويندي آهي ۽ هن طرح جا جُملا چوندا آهن ته پٽ هاڻي تون به سکي ويو آهين، تو کي ڪاروبار ڪرڻ اچي ويو آهي، تون سمجھدار ٿي ويو آهين وغيره ۔ جڏهن ته اهڙي موقعي تي ته ٻار کي اهو سيکارڻ گھرجي ته پٽ ڪوڙ ڳالهائي ڪاروبار ناهي ڪبو، پٽ ٻين کي ڌوڪو ڏئي ڪاروبار ناهي هلائبو، پٽ ڏيتي ليتي ۾ ڪوڙا قسم ناهن کڻبا، وغيره وغيره۔ پر پيسو ڪمائڻ جي لالچ صحيح ۽ غلط جي فرق کان بي پرواھ (Independent) ڪندي آهي، پيسو ڪمائڻ جي لالچ حلال ۽ حرام جو فرق مٽائيندي آهي، پيسو ڪمائڻ جي لالچ الله پاڪ جي ياد کان غافل ڪندي آهي، پيسو ڪمائڻ جي لالچ نمازِ باجماعت کان غافل ڪندي آهي، پيسو ڪمائڻ جي لالچ تلاوتِ قرآن کان غافل ڪندي آهي۔ پيسو ڪمائڻ جي لالچ ڪوڙ ڳالهائڻ تي مجبور ڪندي آهي، پيسو ڪمائڻ جي لالچ ڌوڪو ڏيڻ تي اُڀاريندي آهي، جڏهن ته ڌوڪو ڏيڻ واري کي هن حديثِ پاڪ تي به غور ڪرڻ گھرجي ته صادق ۽ امين آقا، محسن ۽ ڪريم آقا صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جن فرمايو: لا يُؤْمِنُ اَحَدُكُمْ حَتّٰي يُحِبَّ لِاَخِيْهِ مَا يُحِبُّ لِنَفْسِهٖ يعني توهان مان ڪو به ان وقت تائين ڪامل مومن نٿو ٿي سگھي، جيستائين پنهنجي ڀاءُ جي لاءِ اها شيء پسند نه ڪري جيڪا پنهنجي لاءِ پسند ڪندو آهي۔ (بخاری ،کتاب الایمان ، باب من الایمان ان یحب لاخیه مایحب لنفسه،۱/۱۶، حدیث:۱۳)

اي عاشقانِ رسول اسلامي ڀينرون! اسان کي به ڪثرت س ان عبادتِ الٰهي کي پنهنجو معمول بنائڻ گھرجي۔ هن ۾ شڪ ناهي ته غفلت ڀري ماحول ۾ رهي ڪري انسان به غفلت جو شڪار ٿي ويندو آهي، جڏهن ته جيڪڏهن اهڙو ماحول ملي وڃي جنهن ۾ خوب خوب عبادت ڪرڻ جو ذهن ڏنو وڃي ته عبادت ڪرڻ آسان ٿي ويندو آهي۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ دعوتِ اسلامي جو مدني ماحول نه صرف گناهن جي نفرت دل ۾ بيدار ڪندو آهي بلڪ گھڻي عبادت تي به اُڀاريندو آهي، ذيلي حلقي جا 8 مدني ڪم به عبادتن ۾ شامل آهن۔ لهٰذا اسان کي به دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي ذيلي حلقي جي 8 مدني ڪمن ۾ وڌ کان وڌ حصو وٺندو رهڻ گھرجي۔

مطالعو ڪرڻ جا مدني گل

اي عاشقانِ رسول اسلامي ڀينرون! اچو! مطالعو ڪرڻ جي باري ۾ ڪجھ

مدني گل ٻڌڻ جي سعادت حاصل ڪريون ٿيون۔ پهريان 2 فرمان مصطفيٰ صَلَّي اللهُ تَعَالٰي عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ملاحظه ڪريو: (1) فرمايائون: بي شڪ علم سکڻ سان اچي ٿو۔ (ڪنز العمال،۱۰/۱۰۴، حديث: ۲۹۲۵۲) (2) فرمايائون: دنيا ملعون آهي ۽ الله پاڪ جو ذڪر، ان جا ولي يعني دوست ۽ ديني علم پڙھڻ ۽ پڙھائڻ واري کانسواءِ هر شيءِ به ملعون آهي۔ (تِرمِذي، ڪتاب الزهد،۴/۱۴۴،حديث:۲۳۲۹) ٭ مطالعو ايمان جي مضبوطي ۽ پختگي جو سبب آهي۔ (مطالعہ کیا،کیوں اور کیسے؟ص۱۶) ٭ مطالعي سان علم وڌي ٿو۔ (ايضاً ص۱۷) ٭ مطالعو معرفت حاصل ڪرڻ جو ذريعو آهي۔ (ايضاً،ص۱۸) ٭ اهڙن ڪتابن، رسالن ۽ اخبارن کان هميشه پري رهو جيڪي ايمان جي لاءِ زهر قاتل، حياء سوز ۽ اخلاق کان ڪريل هجن۔ (ايضاً،ص۲۹) ٭ پنهنجن اڪابرن جا حالات ڄاڻڻ ۽ انهن جي نيڪ ڪمن کي اپنائڻ جي لاءِ انهن جي احوال ۽ ڪوائف تي مشتمل ڪتابن جو مطالعو به ضروري آهي۔ (ايضاً،ص۳۳) ٭ امام غزالي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جڏهن توهان ڪو علم حاصل ڪرڻ لڳو يا مطالعو ڪرڻ چاهيو ته بهتر آهي ته توهان جو علم ۽ مطالعو تزڪيهءِ نفس ۽ دل جي اصلاح جو باعث ٿئي۔ (ايضاً،ص۳۲) ٭ ٭ ڪوشش ڪريو ته ڪتاب هر وقت گڏ رهي جو جتي موقعو ملي ڪجھ نه ڪجھ مطالعو ڪيو وڃي ۽ ڪتاب جي صحبت به ميسر رهي۔ (ايضاً،ص۳۶) ٭ مطالعو ڪرڻ کان بعد تمام مطالعي کي شروع کان آخر تائين سرسري نظر سان ڏٺو وڃي ۽ ان جو هڪ خلاصو پنهنجي ذهن ۾ محفوظ ڪيو وڃي۔ (ايضاً،ص 112) ٭ پنهنجو اِحتِساب ڪرڻ به فائديمند آهي ته مون کي هن مطالعي مان ڇا حاصل ٿيو ۽ ڪهڙو مواد ذهن نشين ٿيو ۽ ڪهڙو نه۔ (ايضاً،ص 112) ٭ ڪنهن ڳالھ کي ياد ڪرڻ جي لاءِ اکيون بند ڪري يادداشت تي زور ڏيڻ مفيد آهي۔ (ايضاً، ص 117) ٭ جيڪو مطالعو ڪريو ان کي چڱين نيتن سان ٻين اسلامي ڀينرن کي ٻڌايو اهڙي نموني به معلومات جو ذخيرو گھڻي عرصي تائين اسان جي دماغ ۾ محفوظ رهندو۔ (ايضاً، ص112) ٭ جيڪو ڪجھ پڙھون ان جي دهرائي (Repeat) ڪنديون رهو۔ (ايضاً،ص 112) ٭ شيخِ طريقت اميرِ اهلسنّت دَامَتْ بَرَکاتُهمُ الْعَالِيه مدَني مذاڪرن ۾ علمِ دين جا رنگ برنگي مدني گل ارشاد فرمائن ٿا هنن مدني مذاڪرن جي برڪت سان پنهنجي علم تي عمل ڪرڻ، ان کي ٻين تائين پهچائڻ ۽ مزيد مطالعو ڪرڻ جو جذبو پيدا ٿيندو آهي۔ (ايضاً،ص 115)
