اَلْحَمْدُ لِلّٰہِ رَبِّ الْعٰلَمِیْنَ وَ الصَّلٰوۃُ وَالسَّلَامُ عَلٰی سَیِّدِ الْمُرْسَلِیْنَ ط

اَمَّا بَعْدُ فَاَعُوْذُ بِاللّٰہِ مِنَ الشَّیْطٰنِ الرَّجِیْمِ ط بِسْمِ اللہِ الرَّحْمٰنِ الرَّ حِیْم ط

تبرڪات جون برڪتون

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا رَسُولَ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا حَبِیْبَ اللہ

اَلصَّلٰوۃُ وَ السَّلَامُ عَلَیْكَ یَا نَبِیَّ اللہ وَعَلٰی اٰلِكَ وَ اَصْحٰبِكَ یَا نُوْرَ اللہ

نَوَيۡتُ سُنَّتَ الۡاِعۡتِكَافِ (مون سنّتِ اعتڪاف جي نيّت ڪئي)

مٺا مٺا اسلامي ڀائرو! جڏهن به مسجد ۾ داخل ٿيو، ياد اچڻ تي اعتڪاف جي نيت ڪري وٺو، ڇوته جيستائين مسجد ۾ رهندؤ اعتڪاف جو ثواب ملندو رهندو۔ ياد رکو! مسجد ۾ کائڻ، پيئڻ، سمهڻ يا سحري، افطاري ڪرڻ، ايستائين جو آبِ زم زم يا دَم ڪيل پاڻي پيئڻ جي به شرعاً اجازت ناهي ، البت جيڪڏهن اعتڪاف جي نيت هوندي ته اهي سڀ شيون ضمناً جائز ٿي وينديون۔ اعتڪاف جي نيت به صرف کائڻ، پيئڻ يا سمهڻ جي لاءِ نه هئڻ گھرجي بلڪ ان جو مقصد الله تعاليٰ جي رِضا هجي۔ ”فتاويٰ شامي“ ۾ آهي: جيڪڏهن ڪو مسجد ۾ کائڻ پيئڻ يا سمهڻ چاهي ته اعتڪاف جي نيت ڪري ۽ ڪجھ دير ذِڪرُاللہ ڪري پوءِ جيڪو چاهي (يعني هاڻي چاهي ته کائي پي يا سمهي سگھي ٿو)

دُرُود شريف جي فضيلت

حضورِ اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو فرمانِ عاليشان آهي: الله پاڪ جي خاطر پاڻ ۾ محبت رکڻ وارا جڏهن پاڻ ۾ ملن ۽ مصافحو ڪن (يعني هٿ ملائن) ۽ نبي ڪريم (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) تي دُرودِ پاڪ موڪلين ته انهن جي جُدا ٿيڻ کان پهرين ٻنهي جا گناھ معاف ڪيا ويندا آهن. (مُسْنَدُ اَبِي يعْلٰي،۳/۹۵،حديث:۲۹۵۱)

کعبہ کےبدر الدُّجیٰ تم پہ کروروں دُرُود طیبہ کے شمسُ الضُّحٰی تم پہ کروروں دُرُود

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! ثواب حاصل ڪرڻ خاطر بيان ٻڌڻ کان پهريان سٺيون سٺيون

نِي ّ تون ڪري وٺون ٿا. فرمانِ مصطفي ٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: ” نِيَّةُ الۡمُؤۡمِنِ خَيۡرٌ مِّنۡ عَمَلِه  “ يعني مسلمان جي نِيّت ان جي عمل کان بهتر آهي .

(المعجم الڪبير،سهل بن الساعدي...الخ ج 6، ص 185 ، الحديث: 5942)

ٻه مدني گُل: (1) بغير سٺي نيّت جي ڪنهن به نيڪ عمل جو ثواب ناهي ملندو.

(2) جيتريون سٺيون نِيّتون وڌيڪ، اوترو ثواب به وڌيڪ.

بيان ٻڌڻ جون نِيَّتون

٭ نگاهون جھڪائي پوري توجھ سان بيان ٻڌندس ٭ ٽيڪ لڳائي ويهڻ بدران علمِ دين جي تعظيم خاطر جيترو ٿي سگهيو ٻئي گوڏا وڇائي ويهندس ٭ ضرورت پوڻ تي ٻين جي لاءِ سِرڪي جاءِ ڪشاده ڪندس ٭ ڌِڪو وغيره لڳو ته صبر ڪندس، گھوري ڏسڻ ۽ اَٽڪڻ کان بچندس ٭ صَـلُّـوا عَلَي الحَبِيـب! اُذکُرُالله, تُوبُوا اِلَـي الله! وغيره ٻڌي ثواب ڪمائڻ ۽ صدا لڳائڻ وارن جي دِل جوئي جي لاءِ بلند آواز سان جواب ڏيندس ٭ بيان کان پوءِ پاڻ اڳتي وڌي سلام، مصافحو ۽ انفرادي ڪوشش ڪندس۔

صَلُّوْا عَلَي الْحَبِيب! صَلَّي اللّٰه تَعَالٰي عَلٰي مُحَمَّد

مٺا مٺا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! اڄ اسان ”تبرڪات جون برڪتون“ ٻڌڻ جي سعادت حاصل ڪنداسين۔ تبرڪات ۽ ان جي ثبوت ۾ ٻه (2) قرآني واقعا، صحابه ڪرام عَلَيهمُ الرِّضْوَان نبي اڪرم رسولِ محتشم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مقدس تبرڪات جي ڪيتري تعظيم ڪندا هئا؟ تبرڪات جي تعظيم مان ڪهڙا ڪهڙا فائدا ٿيندا آهن، تبرڪات جي بي ادبي سان ڪهڙا نقصان ٿيندا آهن اهو به ٻڌنداسين۔ الله پاڪ ڪري ته دِلجمعي سان اسان اوّل کان آخر سٺين سٺين نيتن سان بيان ٻُڌڻ جي سعادت حاصل ڪريون۔

يوسف جي قميص مان شفاء

سوره يوسف ۾ الله پاڪ جي نبي حضرت سيِّدنا يعقوب ۽ حضرت سيدنا يوسف عَلَيهمَا السَّلَام جو ڊگھو واقعو بيان ٿيو آهي، جڏهن حضرت يوسف عَلَيه السَّلَام

کي ان جي ويڳن(ماٽيلن) ڀائرن ڌوڪي سان کوھ ۾ اڇلائي ڇڏيو ۽ ڪي واپاري ان کي کوھ مان ڪڍي مصر کڻي ويا ۽ اتي وڪڻي ڇڏيائون، حضرت يعقوب عَلَيه السَّلَام پنهنجي پٽ حضرت يوسف عَلَيه السَّلَام جي جدائي ۾ گھڻو غمگين ٿيا ۽ ان غم ۾ لڙڪ وهائي وهائي سندن اکين جي روشني متاثر ٿي وئي، ڪيئي سالن کان بعد جڏهن حضرت يوسف عَلَيه السَّلَام کي پنهنجي ڀائرن جي ذريعي والد محترم جي بصارت جي ڪيفيت جو معلوم ٿيو ته انهن پنهنجي قميص تبرڪ طور پنهنجي والد محترم جي لاءِ موڪليائون، ۽ جيڪو ڪجھ فرمايائون قرآنِ پاڪ ۾اهو هن طرح بيان ڪيو ويو آهي، جيئن ته سيپارو 13 ، سورة يوسف جي آيت نمبر 93 ۾ ارشاد آهي: اِذْهَبُوْا بِقَمِیْصِیْ هٰذَا فَاَلْقُوْهُ عَلٰى وَجْهِ اَبِیْ یَاْتِ بَصِیْرًاۚ-وَ اْتُوْنِیْ بِاَهْلِكُمْ اَجْمَعِیْنَ۠(۹۳) (پ 13، يوسف 93)

ترجموڪنزالايمان: منهنجي هي قميص کڻي وڃو ان کي منهنجي پيءُ جي منهن تي وجھو ته ان جون اکيون نور واريون ٿي وينديون .

جڏهن حضرت يوسف عَلَيه السَّلَام جي ڀائرن اهو ڪرتو حضرت يعقوب عَلَيه السَّلَام جي چهري تي وڌائون ته ڇا ٿيو، ان کي ڪجھ آيتن کان بعد هن طرح بيان ڪيو ويو آهي:

فَلَمَّاۤ اَنْ جَآءَ الْبَشِیْرُ اَلْقٰىهُ عَلٰى وَجْهِهٖ فَارْتَدَّ بَصِیْرًاۚ- (پ 13، يوسف 96)

ترجموڪنزالايمان: پوءِ جڏهن خوشخبري ٻڌائيندڙ آيو ان اها قميص يعقوب جي منهن تي رکي ان وقت سندس اکيون ٿي ويون۔

تفسيرِ صراط الجنان ۾ آهي: جمهور مفسّر فرمائن ٿا ته خوشخبري ٻڌائڻ وارو حضرت يوسف عَلَيه السَّلَام جو ڀاءُ يهودا هيو. يهودا چيو ته حضرت يعقوب عَلَيه السَّلَام وٽ رت سان ڀريل قميص به مان ئي کڻي ويو هئس، مون ئي چيو هو ته حضرت يوسف عَلَيه السَّلَام کي بگهڙ کائي ويو آهي، مون ئي انهن کي غمگين ڪيو هو، ان ڪري اڄ ڪرتو به مان ئي کڻي ويندس ۽ حضرت يوسف عَلَيه السَّلَام جي زندگي جي خبر به مان ئي ٻڌائيندس. ته پوءِ يهودا ڪرتو کڻي 80 فرسنگ

(يعني 240 ميل) ڊوڙندي آيا. يهودا جڏهن حضرت يوسف عَلَيه السَّلَام جي قميص حضرت يعقوب عَلَيه السَّلَام جي چهري تي وڌي ته انهيء وقت انهن جون اکيون درست ٿي ويون ۽ ڪمزوري کان پوءِ طاقت ۽ ڏک کان پوءِ خوشي موٽي آئي، پوءِ حضرت يعقوب عَلَيه السَّلَام فرمايو مون توهان کي نه چيو هو ته مان الله پاڪ جي طرف کان اها ڳالهه ڄاڻيان ٿو جيڪا توهان نه ٿا ڄاڻو جو حضرت يوسف عَلَيه السَّلَام زندهه آهن ۽ الله پاڪ اسان کي پاڻ ۾ ملائي ڇڏيندو. (تفسیرکبیر، یوسف، تحت الآیۃ: ۹۶، ۶/۵۰۸، جمل مع جلالین، یوسف، تحت الآیۃ: ۹۶، ۴/۸۰، ملتقطاً)(صراط الجنان،۵/۵۴ملخصاً)

مٺا مٺا اسلامي ڀائرو! غو ر ڪريو! حضرت سيدنا يوسف عَلَيه السَّلَام خود نبي آهن، پاڻ هڪ ٻي نبي، پنهنجي والد حضرت سيدنا يعقوب عَلَيه السَّلَام جي اکين جي مرض جي لاءِ بطور تبرڪ پنهنجو ڪرتو موڪلي رهيا آهن، ۽ جڏهن ان ڪرتي کي ان جي چهري تي وڌو ويو ته الله پاڪ انهن کي اکين جي بيماري مان شفا عطا فرمائي۔ هن مان معلوم ٿيو ته بزرگن سان نسبت رکڻ وارين شين کي تبرڪ سمجھڻ ۽ انهن مان برڪتون ماڻڻ انبياءِ ڪرام عَلَيهمُ السَّلَام جو طريقو رهيو آهي، ۽ هن واقعي کي قرآنِ پاڪ ۾ ذڪر ڪرڻ هن ڳالھ جو اعلان آهي ته تبرڪات مان فائدو ٿيندو آهي۔

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله تَعَالٰي عَلٰي مُحَمَّد

ڏڪار مان ڇوٽڪارو ملي ويو!

حضرتِ سيدنا شيخ عبدالحق محدثِ دهلوي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: هڪ ڀيري سخت ڏُڪار پيو، ماڻهن جي گهڻين دعائن جي باوجود بارش نه ٿي۔ حضرت سيّدنا نظام الدين رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جن پنهنجي امڙ سائڻ رَحْمَةُ اللهِ تَعَالٰی عَلَيْهَا جي ڪپڙي جو هڪ ڌاڳو هٿ ۾ کڻي عرض ڪيو؛ يا الله پاڪ! هي اُن عورت جي دامن جو ڌاڳو آهي جنهن تي ڪڏهن ڪنهن نامَحرم جي نظر نه پئي، منهنجا ڪريم رب! انهيءَ جي ئي صدقي رحمت جو مينهن وساءِ! اڃا دعا ختم به نه ٿي هئي جو رحمت جا ڪَڪَر ڇائنجي ويا ۽ ڇما ڇم برسات شروع ٿي وئي۔

تبرڪات جا فائدا

مٺا مٺا اسلامي ڀائرو! ٻڌو اوهان ته بزرگن جي جسم سان نسبت رکڻ واري لباس جي هڪ ڌ اڳي جو به ڪيترو شان آهي جو ان کي هٿ ۾ رکي دعا ڪئي وئي ته قبول ٿي وئي۔ الله پاڪ جيڪو سمورين برڪتن جو مالڪ آهي ان ڪريم رب پنهنجي انهن نيڪ ٻانهن کي دنيا جهان جي اهڙين برڪتن سان نوازيو هوندو آهي جو جيڪي شيون انهن سان منسوب ٿي وڃن اهي به بابرڪت ٿي وينديون آهن۔ انبياءِ ڪرام عَلَيهمُ السَّلَام صحابه ڪرام عَـلَيْهِمُ الرِّضْوَانْ ۽ اولياءِ عظام رَحْمَةُ اللهِ تَعَالٰی عَلَيْهمْ اَجْمَعِين سان منسوب شيون وڏيون بابرڪت ۽ فيض واريون هونديون آهن، يعني انهن مقدس تبرڪات جو ادب ۽ احترام ڪرڻ وارن خوش نصيب عاشقانِ رسول کي انهن تبرڪات مان خوب فيض ملندو آهي ۔

تَبَرُّڪ ڪنهن کي چئبو آهي

مٺا مٺا اسلامي ڀائرو! تبرڪ مان مراد (انبياءِ ڪرام عَلَيهمُ السَّلَام ، صحابَه ڪرام عَلَيهمُ الرِّضْوَان) بزرگانِ دين جون اهي شيون جيڪي برڪت طور تي رکيون وڃن۔ (تبرڪات ڪا ثبوت،ص۲ بتغير) پياري پياري آقا، مديني واري مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي جسم سان مَسّ (Touch) ٿيندڙ ۽ نسبت رکڻ واري هر هر شيء به مُتَبَرَّڪ آهي، اهڙي طرح صحابَه ڪرام ۽ بزرگانِ دين رِضْوَانُ الله تَعَالٰي عَلَيهمْ اَجْمَعِين جي مبارڪ جسمن سان مس ٿيندڙ (۽ نسبت رکڻ واري) هر شيء به مُتَبَرَّڪ آهي۔ (تبرڪات ڪا ثبوت،ص۳،۴ ملتقطاً) ان ڪري اسان کي هر ان شيء جو ادب ۽ احترام ڪرڻ گھرجي جنهن کي بزرگن سان نسبت ٿي وڃي۔ انهن جا وار مبارڪ، قميص، جبو، دستار، پيالو مطلب ته! انهن سان نسبت رکڻ وارو ڪو ڪک هجي يا لباس جو ڌاڳو هجي ان جو ادب ۽ احترام ڪرڻ سان به اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ برڪتون نصيب ٿينديون۔

اهڙي طرح تبرڪات جي معاملي ۾ شڪ نه ڪرڻ گهرجي، البت اهو ياد رهي ته ولي شريعت جو پابند هوندو آهي. جيڪڏهن وڏن وڏن جبن وارو، تمام گهڻيون منڊيون پائڻ وارو، بي نمازي شخص اسان کي ڪنهن مزار جي ٻاهر ملي وڃي ته اهو ولي ڪيئن ٿو ٿي سگهي ؟ اهو به ياد رهي ته ولي ” عالم “ هوندو آهي. توهان داتا صاحب، خواجه صاحب، غوثِ پاڪ (رَحمۃُ اللہِ تعالیٰ عَلَیْھِم اَجمعین) بلڪ ڪنهن به مشهور بزرگ جا واقعا پڙهندؤ ته توهان کي سڀني جي سيرت ۾ اهو ئي ملندو ته انهن جي زندگي جو شرعاتي حصو علمِ دين حاصل ڪندي گذريو. ياد رکو! حسنِ ظن ۾ فائدا ئي فائدا آهن، جڏهن ته بدگماني ۾ نقصان ئي نقصان آهن۔ افسوس! اڄ جي دور ۾ اسان جو هڪ تعداد آهي جيڪو حسنِ ظن بدران بدگماني ڏانهن گھڻو مائل هوندو آهي۔ ڳالھ ڳالھ تي بدگماني جا نظارا عام آهن۔ ڪنهن کي فون ڪريو ۽ اهو ريسيو Receive نه ڪري ته بدگُماني ۔۔۔۔۔۔ پٽ جو ڌيان ماءُ کان گھٽ ٿي وڃي ته هڪدم ننهن بابت بدگُماني۔۔۔۔۔۔ ڪنهن سٺي نوڪري کان فارغ ٿيا ته دفتر جي ڪنهن ماڻهو بابت بدگُماني ۔۔۔۔ ڪاروبار ۾ نُقصان ٿيو ته قريبي دڪاندار کان بدگُماني ۔۔۔۔۔۔

پنهنجي ڪمزور ڪارڪردگيءَ يا ڪنهن تنظيمي مصلحت جي پيشِ نظر تنظيمي ذميداري ختم ڪئي وئي يا تبديل ٿي وئي ته ذِميدارن کان بدگُماني۔۔۔۔۔۔اِجتِماعِ ذِڪر ونعت جي اِنتِظَامن ۾ ڪمزوري ٿي ته انتظام ڪرڻ وارن کان هڪدم بدگُماني ۔۔۔۔۔۔ اِجتِماع ذِڪر ونعت ۾ ڪو شخص عشقِ رسول ۾ جُھومي رهيو آهي يا پنهنجي گناهن کي ياد ڪري روئي رهيو آهي ته بد گُماني۔۔۔۔۔۔ ڪنهن بُزرگ يا پِير پنهنجي مُرِيدن يا مُتَعِّلِقن جي ترغِيب يا تحديثِ نعمت (يعني نعمت جي چرچي) جي لاءِ ڪو پنهنجو واقِعو بيان ڪيو ته هڪدم ان کان بدگُماني ۔۔۔۔ جنهن قَرض ورتو ۽ اهو رَابِطي ۾ نه پيو اچي يا جنهن سان مال بُڪ ڪرايو اهو نه پيو ملي ته هڪدم بدگُماني ۔۔۔۔۔۔ ڪنهن وَقت ڏنو ۽ اچڻ ۾ دير ٿي وئي ته بد گُماني ۔۔۔۔۔۔ ڪنهن وٽ ٿورڙي ئي عرصي ۾ گاڏي، سٺو گھر ۽ ٻيون سهولتون اچي ويون ته بدگُماني۔ مطلب ته! اسان جو معاشرو هن وقت بدگماني جي خوفناڪ آفت جي وڪڙ ۾ آهي۔

ياد رکو! بدگماني ٻين ڪيئي گناهن ۾ مبتلا ڪندي آهي۔۔۔ بدگماني ٻين جي عيبن جي ٽوھ ۾ لڳائيندي آهي۔۔۔ بدگماني حسد تي اڀاريندي آهي۔۔۔ بدگماني غيبت ڪرائيندي آهي۔۔۔ بدگماني دل ۾ نفرتن جو ٻج پوکيندي آهي۔۔۔ بدگماني پاڻ ۾ محبتون ختم ڪري جدايون ڪرائيندي آهي ۔۔۔ بدگماني حسنِ سلوڪ کان محروم ڪرائيندي آهي۔۔۔ بدگماني ٻانهي کي بدخُلق ٺاهيندي آهي۔۔۔ بدگماني الزام تراشي تي اڀاريندي آهي۔۔۔ بدگماني دنيا ۽ آخرت ۾ رسوا ڪرائيندي آهي۔ تنهنڪري عقل مند اهو ئي آهي جيڪو بدگماني جي بدران حسنِ ظن جي عادت اپنائي، ڇوته ٭ حسنِ ظن هڪ عمده عبادت آهي، ٭ حسنِ ظن گناهن کان بچائڻ وارو آهي، ٭ حسنِ ظن رکڻ ايمان جي تقاضن مان آهي، ٭ حسنِ ظن ايمان جو حصو آهي، ٭ حسنِ ظن نيڪ ماڻهن جي عادتن مان آهي، ٭ حسنِ ظن ٻانهي کي ثواب جو مستحق ڪندو آهي، ٭ حسنِ ظن ٻين جي عزتن جو محافظ آهي، ٭ حسنِ ظن کان سڪون ۽ قرار ملندو آهي، ٭ حسنِ ظن ٻانهي کي شيطان جي چنگل کان بچائيندو آهي، ٭ حسنِ ظن ايمان کي

قوت ڏيندو آهي، ٭ حسنِ ظن دل ۽ روح کي پاڪيزه ڪندو آهي، ٭ حسنِ ظن ٻانهي کي نيڪ بڻائيندو آهي، ٭ حسنِ ظن سان الله پاڪ ۽ ان جي مدني حبيب، حبيبِ لبيب صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي رضا ملندي آهي۔

هڪ ڀيري نبي اڪرم، نورِ مجسم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن ڪعبي پاڪ کي مخاطب ڪري ارشاد فرمايائون؛ تون پاڻ ۽ تنهنجي فضا ڪيڏي پياري آهي، تون ڪيڏو نه عظمت وارو آهين ۽ تنهنجي حُرمت ڪيڏي عظيم آهي، ان پاڪ ذات جو قسم جنهن جي قبضهءِ قدرت ۾ محمد (صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم) جي جان آهي! الله پاڪ جي ويجھو هڪ مؤمن جي جان ۽ مال ۽ ان کان سٺو گمان رکڻ جي حرمت تنهنجي حرمت کان به وڌيڪ آهي.

(سنن ابن ماجه، ابواب الفتن، باب حرمۃ دم المؤمن وماله ،۴/۳۱۹، حديث:۳۹۳۲)

خُدایا عطا کر دے رحمت کا پانی

رہے قَلب اُجلا دُھلے بدگُمانی

صَلُّو ْا عَلَي الْحَبِيب! صَلَّي الله تَعَالٰي عَلٰي مُحَمَّد

نبي پاڪ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جا تبرڪات:

مٺا مٺا اسلامي ڀائرو! صحابه ڪرام عَلَيهمُ الرِّضْوَان هن ڳالھ مان بخوبي واقف هئا جو نبي مڪرم، رحمت عالم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سراپا رحمت ۽ برڪت آ هن، جنهن شيء کي به آقا ڪريم، رؤف رحيم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان نسبت جو شرف ملندو آهي صحابه ڪرام عَلَيهمُ الرِّضْوَان ان کي برڪت سمجھندا هئا۔ ان ڪري صحابه ڪرام عَلَيهمُ الرِّضْوَان ڪيئي طريقن سان پياري آقا عَلَيه الصَّلٰوةُ وَالسَّلَام کان برڪتون ماڻڻ جي ڪوشش ڪندا هئا، اهي پاڻ ڪريم عَلَيه الصَّلٰوةُ وَالسَّلَام جي جسمِ اطهر کي تبرڪاً هٿ لائيندا هئا، پاڻ عَلَيه الصَّلٰوةُ وَالسَّلَام جي وضو جي بچيل پاڻي کي تبرڪ سمجھي ان مان برڪتون حاصل ڪندا هئا، جنهن پاڻي سان پاڻ ڪريم عَلَيه الصَّلٰوةُ وَالسَّلَام پنهنجا مبارڪ هٿ ڌوئيندا هئا صحابه ڪرام عَلَيهمُ الرِّضْوَان ان کي پنهنجن

چهرن ۽ بدن جي حصن تي تبرڪ طور مليندا هئا، پاڻ ڪريم عَلَيه الصَّلٰوةُ وَالسَّلَام جي بچيل کاڌي مان تبرڪ حاصل ڪندا هئا، پاڻ ڪريم عَلَيه الصَّلٰوةُ وَالسَّلَام جي پگھر مبارڪ، لعابِ دهن، موءِ مبارڪ، منڊي مبارڪ، بستر مبارڪ، لباس مبارڪ، کٽ مبارڪ ۽ پاڻ ڪريم عَلَيه الصَّلٰوةُ وَالسَّلَام جي استعمال ڪيل چٽائي مبارڪ مان به برڪتون حاصل ڪندا هئا، مطلب ته! هر اها شيء جنهن کي تاجدارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ سان ٿوري يا گھڻي نسبت حاصل ٿي ويندي هئي صحابه ڪرام عَلَيهمُ الرِّضْوَان ان کي پنهنجي لاءِ تبرڪ بڻائيندا هئا، حديث سڳورين ۾ هن طرح جا کوڙ واقعا ملن ٿا۔ اچو! انهن مان 3 واقعا ٻڌون ٿا:

(1) منقول آهي ته مشهور صحابي رسول، حضرت سَيدُنا اميرمعاويه رَضِىَ اللّٰهُ تَعَالٰى عَنْه وٽ نبي اڪرم، شاھ بني آدم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي مبارڪ قميص ۽ مقدس ننهن جا ڪجھ تراشا (لاٿل ننهن مبارڪ) هئا۔ ج ڏهن حضرت سَيدُنا اميرمعاويه رَضِىَ اللّٰهُ تَعَالٰى عَنْه جي وصال جو وقت آيو ته انهن وصيت فرمائي ته مون کي ان قميص ۾ ڪفن ڏنو وڃي جيڪا پياري آقا، مڪي مدني مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ انهن کي عطا فرمائي هئي ۽ اها مبارڪ قميص منهنجي جسم سان بلڪل ملائي رکي وڃي۔ جڏهن ته آقا عَلَيه الصَّلٰوةُ وَالسَّلَام جي مبارڪ ننهن جي باري ۾ وصيت فرمائي ته سنها سنها ڪري انهن جي اکين ۽ منهن تي رکيا وڃن۔ حضرت سَيدُنا اميرمعاويه رَضِىَ اللّٰهُ تَعَالٰى عَنْه وڌيڪ فرمايو: اهو ڪم ضرور ڪجو ۽ مون کي سڀ کان وڌيڪ رحم ۽ ڪرم ڪرڻ واري ربِّ ڪريم جي حوالي ڪري ڇڏجو۔

(اسدالغابۃ في معرفۃ الصحابۃ،باب الميم ولعين،۵ /۲۲۳)

(2) مسلم شريف ۾ آهي ته اميرالمُؤمنين حضرت سَيدُنا ابُوبڪرصديق رَضِىَ اللّٰهُ تَعَالٰى عَنْه جي صاحبزادي حضرت سَيدَتُنا اسماء رَضِىَ اللّٰهُ تَعَالٰى عَنْها وٽ حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو هڪ جُبّو (Jubbah) هو (هڪ ڀيري) حضرت سَيدَتُنا اسماء رَضِىَ اللّٰهُ تَعَالٰى عَنْها اهو جبو ڪڍي فرمايائون: هي جبو رَسُولُ الله صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جو آهي، پاڻ ڪريم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ ان کي پائيندا هئا هاڻي اسان ان کي

بيمارن جي لاءِ ڌوئيندا آهيون ۽ ان مان شفاحاصل ڪندا آهيون۔

(مسلم، ڪتاب اللباس و الزينۃ،باب تحريم لبس الحرير…الخ، حديث: ۲۰۶۹، ص۸۸۳)

مشهور مفسرِ قرآن،حضرت مفتي احمد يار خان نعيمي رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جڏهن ماڻهو ان جي زيارت ڪرڻ ايندا هئا ته پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْها اهو فرمائي زيارت ڪرائينديون هيون ته هي حضوراڪرم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ حيات شريف ۾ پائيندا هئا جنهن مان معلوم ٿئي ٿو ته حضور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي لباس جي زيارت ڪرائڻ سُنّتِ صحابه آهي جيئن ته اڄوڪي دور ۾ موءِ مبارڪ جي زيارت ڪرائي ويندي آهي۔ معلوم ٿيو! بزرگن جي تبرڪات جي زيارت ڪرائڻ انهن جو لباس ڌوئي بيمارن کي پيئارڻ سُنتِ صحابه آهي، ان ۾ شفا آهي۔ آبِ زمزم حضرت سَيدُنا اسماعيل عَـلَيْـهِ الـسَّـلاَم جي کڙي سبب نڪتو تمام بيمارين جي لاءِ شفا آهي۔ (مرآۃ المناجيح،۶/۹۸ ملتقطاً) ڪجهه حضرات امُ المُؤمنين عائشه صديقه رَضِىَ اللّٰهُ تَعَالٰى عَنْها جي خدمت ۾ حضور (صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ) جي تبرڪات جي زيارت ڪرڻ ايندا هئا، پاڻ رَضِىَ اللّٰهُ تَعَالٰى عَنْها انهن کي انهن جي زيارت ڪرائينديون هيون۔ (مرآۃ المناجيح،۶/۹۱)

مجلس”اِزديادِ حُب“

تنهنڪري زندگي کي غنيمت سمجھندي دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي وڃو۔ اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ ان جي برڪت سان نيڪ بڻجڻ، گناهن کان بچڻ ۽ پابندِ سنت بڻجڻ جو ذهن بڻجندو۔ اَلْـحَمْـدُ لـِلّٰـه عَزَّوَجَلَّ! عاشقانِ رسول جي مدني تحريڪ دعوتِ اسلامي سڄي دُنيا ۾ اِسلام جون بهارون عام ڪرڻ جي لاءِ ڪم و پيش 104 شُعبن ۾ دِينِ اسلام جو پَيغام عام ڪري رهي آهي، انهن شعبن مان هڪ ’’ مجلسِ ازدياد حب“ (يعني محبتون وڌائڻ وارو شعبو) به آهي۔ ٭ ”مَدَني انعام نمبر55“ ”ڇا اوهان هن هفتي گهٽ ۾ گهٽ هڪ اسلامي ڀاءُ کي (جيڪي پهريان مدني ماحول ۾ هئا پر هاڻي نه ٿا اچن) تلاش ڪري انهن کي مدني ماحول سان وابسته ڪرڻ جي ڪوشش ڪئي؟ (پر جنهن تي تنظيمي پابندي هجي ان کي نه ڇيڙيو)“ مدني انعامات نمبر 55 کي باقاعده تنظيمي طور تي عَمَلي جامو پارائيندي اهي پراڻا اسلامي ڀائر جيڪي اڳ ۾ ايندا هئا پر هاڻي نٿا اچن انهن کي مَدَني ماحول ۾ فَعال ڪري ”مسجد ڀريو تحريڪ“ ۾ شامل ڪرڻ، انهن کان پيشگي وَقت وٺي اِنفِرادي طور تي دُڪان، گھر يا دفتر وڃي ملاقات ڪرڻ، انهن کي سُنّتن ڀري اِجتماعن ۽ مَدَني مذاڪرن ۾ اچڻ جو ذهن ڏيڻ، اِجتماعي اِعتڪاف ۽ مَدَني ڪورسز (اِصلاحِ اَعمال ڪورس،فيضانِ نماز ڪورس وغيره) ڪرڻ جي ترغيب ڏيارڻ، مَدَني قافِلن ۾ سفرڪرائڻ، انهن جي گھرن ۾ مَدَني حلقي جي ترڪيب ڪرڻ، انهن کي مدرسةُ المدينه براءِ بالغان ۾ شرڪت ڪرائڻ، انهن جي خوشي، غمي،بيماري ۽

فوتگي وغيره جي معامَلن ۾ شريڪ ٿيڻ ۽ مشڪلات ۾ مڪتوبات ۽ تعويذاتِ عطّاريه جي ترڪيب ڪرڻ وغيره هن مجلس جي مقصدن ۾ شامل آهي۔ الله ڪريم”مجلس اِزدِيادِ حُبّ“ کي مزيد ترقيون عطا فرمائي. اسان کي دعوتِ اسلامي جي مدني ماحول سان وابسته ٿي نيڪي جي ڪمن ۾ وڌ کان وڌ حصو وٺڻ جي توفيق عطا فرمائي۔ آمِين بِجَاه النَّبِي الاَمِين صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ

مٺا مٺا اسلامي ڀائرو! اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ! آئنده هفتيوار اِجتماع جي بيان جو موضوع آهي ” اُمّتِ مصطفٰي جون خاصيتون“ جنهن ۾ اسان سڀ کان پهريان تورات شريف ۾ اُمّتِ مُصطَفٰي جي بيان ڪيل خاصيتن (Characteristics) ۽ فضائلن ۽ ڪمالات جي باري ۾ ٻڌنداسين، جڏهن حضرت سَيدُنا موسيٰ عَلَيه السَّلَام کي اُمّتِ محبوب جي خاصيتن جو عِلم ٿيو ته انهن ڪنهن شيء جي خواهش جو اِظهار ڪيو هو ان خواهش جي باري ۾ به اسان ٻڌنداسين، هن کان پوءِ اُمّتِ محبوب جي اَفضليت جو سبب ۽ اُمّتِ مصطفٰي جون ڇھ (6) عظيمُ الشَّان فضيلتون به بَيان ٿينديون، اُمّتِ محبوب جي ماڻهن جا حافِظا ڪيتري قدر مضبوط هوندا آهن، ان بابت بزرگن جا اَقوال ۽ حافظُ الحديث حضرت سَيدُنا امام بخاري رَحْمَةُ اللهِ تَعَالٰی عَلَيْه جي بي مثال يادداشت جو هڪ واقعو به اسان ٻڌنداسين۔ آخر ۾ غمخوارِ اُمّت، پيڪرِ عِلم وحڪمت صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جي پنهنجي اُمّت سان مَحَبَّت ۽ شفقت جون ڪجھ جھلڪيون به بَيان ٿينديون۔ تنهنڪري! آئنده جمعرات به اجتماع ۾ حاضري جي نيت ڪريو، نه صرف پاڻ اچڻ جي بلڪ انفرادي ڪوشش ڪري ٻين کي به گڏ آڻڻ جي نيت ڪريو ۽ هٿ کڻي زور سان چئو: اِنْ شَــآءَال ـ لّٰـه عَزَّوَجَلَّ .

اَللّٰھُمَّ صَلِّ عَلٰی سَیِّدِ نَا وَ مَوْلَانَا مُحَمَّدٍ وَّ عَلٰی اٰلِہٖ وَسَلِّمْ

اَللّٰھُمَّ صَلِّ عَلٰی مُحَمَّدٍ کَمَا تُحِبُّ وَ تَرضٰی لَہٗ

هڪ ڏينهن هڪ شَخص آيو ته حُضُورِ اَنور صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ کيس پنهنجي

۽ صِدّيقِ اڪبر رَضِي اللّٰہُ تَعَالٰي عَنہُ جي وچ ۾ ويهاريو، انهيءَ تي صَحابَه ڪرام رَضِي اللّٰہُ تَعَالٰي عَنہُم کي تَعَجُّب ٿيو ته اهو مرتبي وارو شخص ڪير آهي جڏهن اهو هليو ويو ته سرڪار مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هو جڏهن مون تي دُرُودِ پاڪ پڙهندو آهي ته هيئن پڙهندو آهي. ([1])

﴿6﴾ دُرُود شَفَاعت

اَللّٰہُمَّ صَلِّ عَلٰی مُحَمَّدٍ وَّاَنْزِلْہُ الْمَقْعَدَ الْمُقَرَّبَ عِنْدَکَ یَوْمَ الْقِیَامَۃِ

شافع امم صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن جو فرمان معظم آهي: جيڪو شخص هي درود پاڪ پڙهي ان جي لاءِ منهنجي شفاعت واجب ٿي ويندي آهي. ([2])

﴿ 1 ﴾ هڪ هزار ڏينهن جون نيڪيون

جَزَی اللّٰہُ عَنَّا مُحَمَّدًا مَا ھُوَ اَھْلُہٗ

حضرتِ سَيِّدُنا ابنِ عباس رَضِى الله تَعَالٰى عَنْهما کان رِوايت آهي ته سرڪارِ مدينه صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ جن فرمايو: هي درود شريف پڙهڻ واري جي لاءَ ستر فرشتا هڪ هزار ڏينهن تائين نيڪيون لکندا آهن. ([3])

﴿2﴾ ڄڻ شبِ قدر حاصل ڪري ورتي

لَآ اِلٰہَ اِلَّا اللّٰہُ الْحَلِیْمُ الْکَرِیْمُ، سُبْحٰنَ اللّٰہِ رَبِّ السَّمٰوٰتِ السَّبْعِ وَرَبِّ الْعَرْشِ الْعَظِیْم

(خداءِ حليم و ڪريم کان سواءِ ڪوئي عبادت جي لائق ناهي، الله عَزَّوَجَلَّ پاڪ آهي جيڪو ستن آسمانن ۽ عرشِ عظيم جو پروردگار آهي)

فرمانِ مصطفيٰ صَلَّى اللهُ تَعَالٰى عَلَيۡهِ وَاٰلِهٖ وَسَلَّمَ: جنهن هن دُعا کي ٽي ڀيرا پڙهيو ڄڻ ان شبِ قدر کي حاصل ڪري ورتو. (ابنِ عساڪر ج 19، ص 155، حديث 4455)

صَلُّوْا عَلَی الْحَبِیْب! صَلَّی اللّٰہُ تَعَالٰی عَلٰی مُحَمَّد
