اعليٰ حضرت امام احمد رضا خان رَحْمَةُ اللهِ تَعَالٰی عَلَيْه فرمائن ٿا: جنهن پنهنجي نَفۡس تي اعتماد ڪيو، اُن وڏي ڪَذّاب (يعني تمام وڏي ڪوڙي) تي اعتماد ڪيو (ملفوظاتِ اعلیٰ حضرت،ص۴۵۴) خاتمُ المرسلين، رحمةٌ لِّلعٰلمين صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو فرمانِ عبرت نشان آھي: جڏهن به ڪو شخص ڪنهن اجنبي عورت سا گڏ تنهائي اختيار ڪندو آهي، انهن سان گڏ ٽيون شيطان ضرور ھُوندو آھي۔“

ٿيڻ جي ترغيب ڏيندو آھي۔“ (فیض القدیر،حرف الھمزۃ،۳ /۱۰۲ ،تحتَ الحدیث ۲۷۹۵ملخصاً)

مٺا مٺا اسلامي ڀائرو! واقعي شيطان جي ڄار ۽ شَہۡوت جي وبال کان بچڻ جي لاءِ اجنبي مَرۡد ۽ عَورت جو هڪ ٻي کان پردو ڪرڻ ۽ پنهنجي نِگاہن جي حِفاظت ڪرڻ نہايت ضروري آهي، ڇاڪاڻ جو شَہۡوت پهريان مَرۡد ۽ عَورت جي دل ۾ هڪ ٻي جي قربت جو شوق ئي پيدا ڪندي آهي، قُرب حاصل ٿيڻ کان پوءِ ڳالھ ٻولھ جا سلسلا جاري ٿي ويندا آهن ۽ پوءِ اها ئي ڳالھ ٻولھ اڳتي هلي ڪري هڪ ٻي سان مذاق مسخري ۽ بي تڪلُّفيءَ ۾ تبديل ٿي ويندي آهي، جيڪڏهن پهريان عشقِ مجازيءَ جو جنون مٿي تي سُوار نه به ہجي يا ٻنهي مان ڪنهن هڪ جي دل ۾ ئي عشق پيدا ٿيو هجي، ڪنهن جهجڪ جي سبب اُن جو اِظہار نه ڪيو ويو هجي، ته ان بي تڪلُّفيءَ کان پوءِ ته عُمُومًا عشق ٿي ئي ويندو آهي ۽ ان جو اِظہار به با آساني ڪيو ويندو آهي ۽ پوءِ هڪ طرفه عشق، ٻه طرفه ٿي ڪري ڪهڙين ڪهڙين آفتن ۽ گُناہن ۾ مبُتلا ڪندو آهي، ان جو ہر عقلمند اندازو ڪري سگھي ٿو۔ اچو ته عشق ِمجازي جي آفتن ۽ حضرت سَيدُنا يوسُف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي طرف عشقِ مجازيءَ جي نسبت ڪرڻ جي غَلَط فہمي دُور ڪرڻ جي باري ۾ شيخِ طريقت، اميرِ اہلسنّت، بانيِ دعوتِ اسلامي حضرت علّامه مولانا ابُوبلال محمد الياس عطّار قادري رضوي ضيائي دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه جي تحرير ڪيل 616 صفحن تي مشتمل ڪتاب ”نيڪي جي دعوت“ جي صفحي 40 کان 43 مان ڪجھ ٻُڌون ٿا: جيئن ته،

شيخِ طريقت، اميرِ اہلسنّت دَامَـتْ بَـرَكَـاتُـهُـمُ الْـعَـالِـيَـه عشق مجازيءَ جون تباھ ڪاريون بيان ڪندي فرمائن ٿا ته عشق مجازيءَ جو اهڙو عجيب و غريب معاملو آهي عام طور جيڪو هڪ ڀيري ان جي وڪڙ ۾ اچي وڃي، ان جو ان مان بچي

نڪرڻ ڏکيو هوندو آهي. اڄڪلهه عشق مجازي جي خوب هوا هلي رهي آهي، هن جو سڀ کان وڏو سبب اڪثر مسلمانن ۾ اسلامي معلومات جي گھٽتائي ۽ ديني ماحول کان دوري آهي، انهيءَ ڪري هر پاسي گناهن جو سيلاب اچي چڪو آهي! V ، V.C.R ، ۽ انٽرنيٽ وغيره ۾ عِشۡقِيه فلمون ۽ فِسۡقِيه ڊرامن کي ڏسي يا عشق بازين تي مبالغه آميز اخباري خبرون، ۽ ناولن، بازاري ماهنامن، ڊائجسٽن ۾ فرضي افسانن کي پڙهي يا ڪاليجن ۽ يونيورسٽين جي مَخلوط (جتي ڇوڪرو ڇوڪري گڏ هجن اهڙن) ڪلاسن ۾ ويهي، يا نامَحرم رشتيدارن سان ملي پاڻ ۾ بي تڪلفيءَ واري راهه تي هلڻ سان اڪثر نوجوانن کي ڪنهن نه ڪنهن سان عِشق ٿي ويندو آهي، پهرين هڪ طرفان ٿيندو آهي پوءِ جڏهن فريق اول، فريق ثانيءَ کي ٻڌائيندو آهي ته بعض اوقات ٻه طرفا ٿي ويندو آهي ۽ پوءِ عموماً گناهه ۽ نافرمانين جو طوفان اچي ويندو آهي، فون تي دل کولي بي شرمانه ڳالهيون بلڪ بي حجاب ملاقاتن جا سلسلا ٿيندا آهن، خطن ۽ تحفن جا تبادلا ٿيندا آهن، شادي جا خُفيه قول ۽ قرار ٿي ويندا آهن، جيڪڏهن گھر وارا ديوار بڻجن ته ڪڏهن ٻئي فرار ٿي ويندا آهن، ان کان پوءِ اخبار ۾ اِشتِهار ڇپبا آهن، خاندان جي عزت، آبرو سرِ عام نيلام ٿيندي آهي، ڪڏهن ”ڪورٽ ميرج“ جي ترڪيب ٺهندي آهي ته ڪڏهن مَعَاذالله ائين ئي بنا نڪاح جي۔۔۔۔۔ ۽ اهڙن بي رحم انسانن جي ناجائز ٻارن جا لاش ڪچري ڪنڊي (يعني گندگي جي ڍير) ۾ ملندا آهن، ۽ ائين به ٿيندو آهي جو ڀڄي نه سگھن ته خودڪُشيءَ جي راهه ورتي ويندي آهي جنهن جون خبرون اخبار ۾ تقريباً روزانو شايع ٿينديون رهنديون آهن .

مٺا مٺا اسلامي ڀائرو! اڄڪلهه اسلامي معلومات جي ڪمي جو دؤر آهي، جهالت عام آهي، ڪجهه نادان عاشقَ، پنهنجي گندي عشق بازين تي پردو وجھڻ لاءِ ايستائين به چوندي ٻڌا ويا آهن ته حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام به زليخا سان عشق ڪيو هو! (مَعَاذالله) ائين هرگز ناهي، يقيناً ائين بڪواس ڪرڻ وارا نادان عاشق سخت خطا تي آهن، پنهنجي نفس جي شرارت جي معاملي ۾ شيطان جي ڳالهين ۾ اچي بنا سوچي سمجھي ڪنهن به نبي عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي باري ۾ زبان کولڻ ايمان جي لاءِ انتهائي خطرناڪ هوندي آهي. ياد رکو! نبي عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جي اَدنيٰ گستاخي به ڪفر آهي. حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام الله عَزَّوَجَلَّ جا نبي آهن ۽ هر نبي معصوم. نبي کان ڪڏهن به مَذموم حرڪت ناهي ٿيندي، جيئن دعوت اسلامي جي اشاعتي اداري مڪتبة المدينه جي شايع ٿيل ترجمي واري قرآن ” ڪنزُالايمان مع خزائنُ العِرفان “ صفحي 445 تي سيپاري 12 سورة يوسف جي آيت نمبر 24 ۾ الله تبارڪ و تعالي ارشاد فرمائي ٿو:

وَ لَقَدۡ ہَمَّتۡ بِہٖ ۚ وَ ہَمَّ بِہَا لَوۡ لَاۤ اَنۡ رَّاٰ بُرۡہَانَ رَبِّہٖ

ترجمو ڪنزالايمان: ۽ بيشڪ عورت ان جو ارادو ڪيو ۽ اهو به عورت جو ارادو ڪري ها جيڪڏهن پنهنجي رب جو دليل نه ڏسي ها.

صَدۡرالاَفاضِل حضرت علامه مولانا سيِّد محمد نعيمُ الدّين مراد آبادي عَـلَيْهِ رَحْ ـ مَةُ اللهِ الۡهادی فرمائن ٿا: الله تعاليٰ اَنبياء عَـلَيْـھمُ الـصَّلٰوة ُ وَالـسَّلَام جي پاڪ نفسن کي بُري اخلاق ۽ خراب اَفعَال کان پاڪ پيدا ڪيو آهي ۽ اخلاق شريفه طاهِره مقدَّسه تي انهن جي پيدائش فرمائي آهي، ان ڪري اهي هر بُري فعل کان پري رهن ٿا، هڪ

روايت ۾ آهي ته جنهن وقت زليخا سندن جي پويان لڳي، تنهن وقت پاڻ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام پنهنجي والد ماجد حضرت يعقوب عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي ڏٺائون ته آڱر مبارڪ ڏندن جي هيٺان دٻائي پري هٽڻ جو اشارو فرمائي رهيا هئا. (خزائن العرفان)

حقيقت هي آهي ته عشق صرف زليخا جي طرفان هو حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو دامن ان کان يقيناً پاڪ هو سيپاري 12 سوره يوسف آيت نمبر 30 ۾ مصر جي شريف ماڻهن جي عورتن جو قول هن طرح نقل ڪيو ويو آهي:

وَ قَالَ نِسۡوَۃٌ فِی الۡمَدِیۡنَۃِ امۡرَاَتُ الۡعَزِیۡزِ تُرَاوِدُ فَتٰىہَا عَنۡ نَّفۡسِہٖ ۚ قَدۡ شَغَفَہَا حُبًّا ؕ اِنَّا لَنَرٰىہَا فِیۡ ضَلٰلٍ مُّبِیۡنٍ ﴿۳۰﴾

حُجَّةُ الاِسلام حضرت سيِّدُنا امام ابو حامد محمّد بن محمّد بن محمد غزالي عَـلَيْهِ رَحْمَةُ اللهِ الۡوَالِی فرمائن ٿا: “زليخا کي رغبت هئي پر حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام طاقت ۽ قدرت رکڻ باوجود ان (يعني زليخا جي طرف رغبت) کان روڪيا رهيا، الله عَزَّوَجَلَّ قرآن ڪريم ۾ سندن جي محفوظ رهڻ جي عمل جي تعريف ڪئي آهي. ان سان اَظْہَر مِنَ الشَّمْسِ وَ اَبْیَن مِنَ الْاَ مْس يعني سج کان وڌيڪ روشن ۽ گذريل ڪالهه کان وڌيڪ يقين ٿي ويو ته اڄڪلهه جيڪي نادان عاشق پنهنجي گناهن ڀري بدبودار عشق کي صحيح ثابت ڪرڻ جي لاءِ مَعَاذَاللّٰه حضرت سيِّدُنا

يُوسُف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام ۽ زليخا جي واقعي کي آڙ بڻائن ٿا، اهو قرآن پاڪ جي حڪم جي سراسر خلاف آهي ۽ ڪئي صورتن ۾ سڌو ڪفر تائين وٺي وڃڻ وارو آهي، سورتِ يوسف ۾ صرف زليخا جي طرفان عشق جو تذڪرو آهي پر ڪٿي به ڪو اشارو نٿو ملي ته مَعَاذَاللّٰه حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام به هن عشق ۾ شريڪ هئا، تنهنڪري جيڪي حضرت سيدنا يوسف عَـلٰى نَبِـيِّـنَا وَ عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام کي به عشق ۾ شريڪ ڪندا آهن، اهي ان کان توبه ۽ تجديدِ ايمان ڪن يعني توبه ڪري نئين سري کان مسلمان ٿين. الله عَزَّوَجَلَّ جي نبي عَـلَيْـهِ الـصَّلٰوة ُ وَالـسَّلَام جو شان تمام بلند هوندو آهي ۽ اهي گناهن کان معصوم هوندا آهن.

مٺا مٺا اسلامي ڀائرو! ان حقيقت کان انڪار نٿو ڪري سگھجي ته موجودہ دَور ۾ ڪيتريون ئي اَخۡلاقي مُعاشَرَتي ۽ شَرۡعي بُرائين سان گڏوگڏ عشقِ مجازيءَ کي فروغ ڏيڻ ۾ به موبائل فون ۽ اِنٽر نيٽ اِنۡتِہائي اَہَم ۽ بُنيادي ڪردار ادا ڪيو آهي، اڄڪلھ عام طور تي بدمعاش ڇوڪرا نَفۡساني خواہشات جي تڪميل جي لاءِ ”ٽائيم پاس“ ڪرڻ جي خاطر ڪنهن نه ڪنهن طرح صِنۡفِ نازُڪ جو فون نمبر حاصل ڪري يا فيس بُڪ وغيرہ جي ذريعي هڪ ٻي سان رابطا قائم ڪري وٺندا آهن ۽ اڪثر ڪري پہريون قدم صِنۡفِ نازُڪ جي طرف کان کڄندو آهي، ائين ڪجھ ئي عرصي ۾ اجۡنَبِيت ختم ٿي ويندي آهي ۽ بي حيائي ۽ عشقِ مجازي جو نه ختم ٿيڻ وارو سِلۡسِلو شروع ٿي ويندو آهي، ان ۾ ڪو شڪ ناهي ته ہر زماني ۾ انساني ضرورَتن جي لحاظ کان سائنس دانن جي هڪ ٻي کان وڌيڪ حيرت انگيز اِيجادون منظرِ عام تي اچي رهيون آهن ۽ پنهنجي پنهنجي دَور جي ماڻهن کان خراجِ

تَحۡسِين وُصُول ڪنديون رهيون آهن، پر هي به هڪ حقيقت آهي ته هاڻي تائين جي تمام اِيجادن ۾ موبائل فون سڀ کان وڌيڪ حيرت انگيز اِيجاد آهي، جنهن ڄڻ ته پوري دُنيا کي ننڍو ڪري انسان جي تِري تي رکي ڇڏيو آهي۔ موبائل فون نه صرف پوري دنيا ۾ مقبول آهي، بلڪه زندگي جي تقريبًا تمام شعبن ۾ ان جو ڪنهن نه ڪنهن حد تائين عَمَل دَخَل ضرور آهي، اهو ئي سبب آهي ته اڄ موبائل فون هڪ طرح سان اسان جي ضرورياتِ زندگي جو حِصّو بڻجي چڪي آهي۔ ان جي ضرورت ۽ اَہميت جو اندازو ان ڳالھ سان لڳائي سگھجي ٿو ته ماڻهن جي هڪ تمام وڏي تعداد جا روز مرّھ جي مُعامَلن ۾ موبائل فون جو استعمال ايتري قدر وڌي چڪو آهي جو اهي ماڻهو ان جي بغير زندگي گُذارڻ جو تصوّر به نٿو ڪري سگھي، هلندي ڦرندي اُٿندي ويهندي سمهندي جاڳندي، ہر وقت موبائل فون سان ئي چِنبڙيل رهندا آهن ۽ اُن کي پاڻ سان چنبڙائي رکندا آهن، ايستائين جو ڪجھ بَد ذوق ماڻهو ته باٿ رُوم ۾ به ان جو استعمال جاري ئي رکندا آهن۔

بہرحال ان ۾ ڪو شڪ ناهي ته موبائل فون جي اِيجاد جتي انسان کي ڪيترا سارا فائدا پہچايا آهن، اُتي ناقابلِ تلافي نُقصانات ۽ ناقابلِ حل مشڪلاتن ۾ به مبتلا ڪري ڇڏيو آهي۔ ان لاءِ جيڪڏهن فائدي جي اعتبار کان هي هڪ طرح جي نعمت آهي ته نُقصانن جي لحاظ کان تمام وڏي مصيبت به آهي، پر حقيقت هيءَ آهي ته ان ۾ موبائل فون جو ڪو قُصُور ناهي، ڇو جو هي ته صرف هڪ اوزار آهي، پنهنجي ذات جي اعتبار کان نه چڱو آهي نه بُرو، ان جي چڱي بُري ٿيڻ جي بُنياد ۽ ان جي فائدن ۽ نُقصانن

جو دار و مدار ان جي استعمال تي آهي، جيڪڏهن جائز ڪمن ۾ ان جو صحيح استعمال ڪيو وڃي، ته ان ۾ فائدو ئي فائدو آهي، بلڪه جيڪڏهن اها ئي موبائل فون ضرورت پوڻ تي نيڪ ڪمن جي لاءِ استعمال ڪئي وڃي ته ڪارِ ثواب به ٿي سگھي ٿو، نيڪ ڪمن ۾ ان جو استعمال ڪو قابلِ تَعَجُّب ڳالھ ناهي، اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ دعوتِ اسلامي سان وابسته اسلامي ڀائر، مُبَلّغِين ۽ ذِمّه داران ان جي ذريعي به نيڪي جي دعوت ۽ سُنّتن جي خدمت ۾ مصروفِ عمل آهن، هاڻي ڪجھ ئي عرصو پهريان مدني چينل تي دعوتِ اسلامي جي مدارِسُ المدينه جي مدني عَطِيات جي لاءِ ٽيلي ٿون (مدني عطيات جمع ڪرڻ جي مہم جو سلسلو) ٿيو، ته ڪيترن ئي اسلامي ڀائرن موبائل فون ۽ اِنٽر نيٽ جي ذريعي به دعوتِ اسلامي جي مدني ڪمن ۾ ترقّي جي لاءِ پنهنجي پنهنجي طور تي ڪوششون ڪيون ۽ اَلۡحَمۡدُلِلّٰه عَزَّوَجَلَّ ڀرپُور انداز ۾ مدني عَطِيات جمع ڪرڻ جو سلسلو رہيو، بہرحال جيڪڏهن ان جو صحيح استعمال ڪيو وڃي ته هيءَ چڱي آهي ۽ ان جو فائدو ئي فائدو آهي ۽ جيڪڏهن غَلَط استعمال ڪيو وڃي ته هيءَ بُري آهي ۽ ان جو نُقصان ئي نُقصان آهي۔

اڄ ڪروڙن جي تعداد ۾ ماڻهون موبائل فون استعمال ڪري رهيا آهن، ان مان ڪجھ ته ضرورت جي سبب استعمال ڪندا آهن ۽ ڪيترائي فيشن، راند روند، ويڊيو گيمز، عادت يا گُناھ جي طور تي به استعمال ڪندا آهن، جن ۾ بد قسمتي سان مسلمان به شامل آهن، اهڙن ماڻهن کي موبائل فون جي استعمال ۾ جائز ۽ ناجائز جي ڪا پرواھ ناهي، يا ائين سمجهو ته جائز ۽ ناجائز جي طرف اُنهن جي ڪا توجّھ ناهي، جڏهن ته هڪ مسلمان جو ہر ڪم

شريعت جي عين مُطابِق ہئڻ گھرجي، بالفرض ناجائز ۽ گُناھ جي ڪمن ۾ ان جو استعمال نه به ڪيو وڃي، صرف فُضُول ڪمن ۾ ئي استعمال ڪيو وڃي، تڏهن به ان کان بچڻ ئي گھرجي، ڇو جو فُضُول ڪم وقت ضائع ڪرڻ سان گڏوگڏ انسان جي اسلام جي لَذَّت ، حَلاوَت ۽ حُسن تي به اَثَر انداز ٿيندا آهن۔ سرڪارِ مدينه، سُرورِ قلب ۽ سينه صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جو ارشادِ باقرينه آهي: مِنْ حُسْنِ اِسْلَامِ الْمَرْءِ تَرْكُهُ مَا لَا يَعْنِيْهِ يعني فُضُول ۽ بي مقصد ڪمن کي ڇڏي ڏيئڻ، اسلام جي حُسن مان آهي۔ (ترمذی،کتاب الزھد،باب(ت:۱۱)،۴/۱۴۲،حدیث:۲۳۲۴)

بہرحال ان ۾ ڪو شڪ ناهي ته اڄڪلھ موبائل فون جي فائدن کان وڌيڪ، ان جا نُقصان ۽ تباھ ڪاريون آهن۔ موبائل فون جي تباھ ڪارين ۽ فتني انگيزين جا مناظر ڏسي ڪري اکيون رَت جا ڳوڙها ٿيون وهائين۔ موبائل جي سبب اطمينان، سڪون، چين ۽ قرار ختم ٿي ويو…شرم ۽ حيا، عزّت ۽ وقّار هليو ويو… شريف ۽ عزّت دار عَورت جي حيا آهستي آهستي ختم ٿي رهي آهي…ابنِ آدم جي عزّت ۽ بنتِ حوّا جي عصمت ۽ پاڪدامني کي خطرن گھيري رکيو آهي…شرافت ۽ اَخۡلاق، تَہۡذِيب ۽ تمدُّن جو جنازو نڪري ويو…اِسراف ۽ فُضُول خرچيءَ جو بازار گرم ٿي ويو … ان جي غَلَط استعمال جي ڪثرت جي سبب وقت جي قدر ۽ قيمت جو احساس فنا ٿي ويو ۽ وقت ضائع ڪرڻ جو هڪ نئون دَور شروع ٿي ويو…رات جو اڪثر حصو موبائل جي ذريعي فُضُول گفتگو ڪرڻ جي سبب غير ضروري شَبِ بيدارين ۾ اضافو ٿي ويو… عشقِ مجازي جي بيماري زور پڪڙڻ لڳي…ناجائز اُلفت ۽ محبت کان نوجوانن جي زندگيون تباھ ٿيڻ لڳيون…

آهي، خود ڪُشي ۽ قتل جو باعث بنجڻ واري ان دَور جي سوشل ويب سائٽس ۽ موبائل مُہَذَّب دُنيا کي پريشانين ۾ وجھي ڇڏيو آهي ۔ ([1])

موبائل فون روڊ ايڪسيڊنٽ جو سبب به بڻجي رهيو آهي، سخت قانون جي باوجود 55 فيصد ماڻهو گاڏي هلائڻ وقت موبائل فون جي ذريعي گفتگو ڪرڻ يا گانا باجا ٻُڌڻ ۾ مصروف رہندا آهن، اهڙا به واقعات سامهون ايندا آهن ته پويان کان اچڻ واري گاڏيءَ جو ہارن نه ٻُڌڻ جي سبب خطرناڪ حادِثا ٿي چڪا آهن، موبائل هڪ طرح سان توهان جي لاءِ رحمت آهي ته زحمت به آهي، ان لاءِ ان جو استعمال ڪريون پر احتياط سان۔ ([2])

حال ۾ ئي فِن لينڊ جي سائنس دانن هڪ تحقيق ۾ ٻڌايو ته موبائل فون سان دِماغ ۾ تبديلي اچي ويندي آهي، ماہرين جي رائي آهي ته ان کان بچڻ جو هڪ ئي طريقو آهي، موبائل فون تي ڪم سان ڪم ۽ صرف مقصد جي ڳالھ ڪئي وڃي۔ فون دِماغي ڪينسر جهڙي سنگين بيمارين جو سبب آهي، موبائل فون دل جي ويجهي کيسي ۾ نه رکڻ گھرجي، ان سان دل تي منفي اَثَر پوندا آهي، ضرورت کان زيادہ دير تائين گفتگو ڪرڻ سان ڪَن تي ته ان جو بُرو اَثَر پوندو ئي آهي، گڏوگڏ هي دِماغ به ان جي اَثَر کان ناهي بچي سگھندو، خاص طور تي ٻارن جي نشوونما گڏوگڏ دِماغ جي لاءِ فون جو اضافي استعمال نُقصان دہ هوندو آهي۔ جي ہا! موبائل جو هڪ خطرناڪ ان طرح به آهي ته ان جي ذريعي، حامِله عَورت ۽ اُن جي پيٽ ۾ پَرۡوَرِش پائڻ واري ٻار جي صِحَّت ۽ اَخۡلاق تي وڏو بُرو اَثَر پئي ٿو، موبائل بجلي سان

هلندو آهي، موبائل کي پهريان بِجۡلي سان چارج ڪيو ويندو آهي، موبائل جي اندر ڪرَنۡٽ ۽ بِجۡلي جون شُعاعون ڀريل رہنديون آهن ۽ بجلي جون هي شُعاعون عام ماڻهو جي لاءِ به نُقصان دھ آهن، اهو ئي سبب آهي ته ڊاڪٽر حضرات موبائل کي سيني جي ويجھي کيسي ۾ رکڻ کان منع ڪندا آهن، ڇو جو ان سان دل جو مَرَض پيدا ٿيڻ جو انديشو رہندو آهي، اڄ موبائل مان نڪرڻ واريون شُعاعون حامِله عَورت ۽ اُن جي پيٽ ۾ پَرۡوَرِش پائڻ واري ٻار جي صِحَّت کي مَفۡلُوج ۽ ناڪارہ بڻائي رہيون آهن، تنهنڪري حامِله عَورتون دَورانِ حَمۡل موبائل جو گھڻو استعمال نه ڪن ۽ ضرورتًا موبائل استعمال ڪريون به ته ان کان خود کي دُور رکو، سِيراندي يا وهاڻي جي هيٺان موبائل ہرگز نه رکو، جديد ميڊيڪل سائنس جي تحقيق جي مُطابِق آخري ٽي چار ہفتن ۾ ماءُ جي پيٽ ۾ پرورش پائڻ واري ٻار جي ڪنن ۾ اُن جي ماءُ ۽ اِردگرد جون آوازون پہچڻ لڳنديون آهن۔ (ماہنامہ اشرفیہ اگست ۲۰۱۳بتغیر)

اڄ جون مائرون اَيامِ حَمۡل ۾ تڪليف جوعُذر پيش ڪري فرائض ۽ واجبات، نماز، روزي ۽ ٻين اوراد و وظائف ڇڏي ڏينديون آهن ۽ فُضُول ڳالهين ۽ بي ڪار ڪمن ۾ مشغول رہنديون آهن، فلمون ڏسنديون آهن، گانا ٻڌنديون آهن ۽ موبائل جي پويان پيل رهنديون آهن، جڏهن ته انهن کي اندازو ہئڻ گھرجي ته اُن جي فِلم بيني، فُحۡش گوئي ۽ گانن ٻُڌڻ جو بُرو اَثَر اُن جي پيٽ ۾ پرورش پائڻ تي به پئي رهيو آهي۔ اچو ته ماءُ جي پيٽ ۾ پلجڻ واري ٻار تي اُن جي ماءُ جي چڱين عادتن جو چڱو اَثَر پوڻ جو هڪ عظيمُ الشّان واقعو سماعت ڪريون۔ جيئن ته

چڱي عادت جو اولاد تي چڱو اَثَر

پيرانِ پير، روشن ضمير،قطبِ ربّاني، محبوبِ سبحاني حضرت سَيدُنا شيخ عبدُ القادر جيلاني رَضِىَ اللهُ تَعَالٰی عَـنْهُ پنج سالن جي عُمر ۾ جڏهن پهرين ڀيري بِسْمِ الله جي رَسۡم جي لاءِ ڪنهن بُزرگ وٽ ويٺا ته اَعُوْذُ بِاللہِ مِنَ الشَّیْطٰنِ الرَّجِیْم ۽ بِسْمِ اللّٰہِ الرَّحْمٰنِ الرَّحِیْم پڙھ ڪري سُورَۃ ُالْفاتِحہ ۽ الٓمّٓ کان 18 سيپاري تائين پڙھي ڪري ٻُڌائي ڇڏيو۔ اُن بُزُرگ پڇيو: پُٽ اڳتي پڙھ۔ فرمايائون: بس مون کي اِيترو ئي ياد آهي، ڇو جو منهنجي والدہ کي به ايترو ئي ياد هو، جڏهن آئون پنهنجي ماءُ جي پيٽ ۾ هئس، اُن وقت اهي پڙھنديون هيون، مون ٻُڌي ڪري ياد ڪري ورتو۔ (الحقائق فی الحدائق،۱/۱۴۰بتغیرقلیل)

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

مٺا مٺا اسلامي ڀائرو! غور ڪريو ته هي ماءُ جي چڱن ڪمن جو پيٽ ۾ پرورش پائڻ واري ٻار تي چڱو اَثَر ناهي ته ٻيو ڇا آهي؟ ياد رکو! ته ماءُ دَورانِ حمل قرآنِ ڪريم جي تلاوت ڪندي ته ضروري ناهي ته ٻار حافظِ قرآن بڻجي ڪري ئي پيدا ٿيندو، پر ايترو ضرور آهي ته تلاوتِ قرآنِ ڪريم ۽ ماءُ جي چڱن ڪمن جو اَثَر ٻار تي ضرور پوندو، حامِله عَورَتن کي ان سلسلي ۾ خاص توجّھ ڏيڻ ۽ حتّي الامڪان موبائل کان بچڻ جي ضرورت آهي۔ هونئن به عَورَتون گھر جي مالِڪه ۽ اُمورِ خانه جي نگہبان ۽ ذِمّه دار ہونديون آهن، تنهنڪري ان اعتبار سان به عَورَتن جي حق ۾ موبائل جو زيادہ استعمال زہرِ قاتل ۽ وڌيڪ خطرناڪ آهي، جيڪڏهن عَورَتون اڪثر ڪري موبائل ۾ مشغول رهنديون ته پوءِ گھريلو ذِمّي دارين کي ڪير سنڀاليندو؟ ان حقيقت کان انڪار نٿو ڪري سگھجي ته ڪم ڪاج جي دَوران ڪم سان گڏوگڏ مُسلسل موبائل فون جي طرف به توجّھ رکڻ سان نه صرف ذہن تي غير

ضروري دٻاؤ پوندو آهي، بلڪه ان غفلت جي سبب اڪثر ڪري گھر ۾ موجود ننڍن ٻارن کي ناقابلِ تلافي نُقصان به پہچندو آهي۔

(روزنامہ انقلاب، ۲۶ستمبر ۲۰۱۳جمعرات بتغیر)

اللہ عَزَّوَجَلَّ اسان سڀني کي پنهنجي حِفۡظ ۽ اَمان ۾ رکي ۽ پنهنجي زندگي کي قرآن ۽ سنّت جي اَحڪامن ۽ پياري آقا، مڪي مدني مُصۡطَفٰي صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي سنّتن جي مُطابِق گُذارڻ جي توفيق عطا فرمائي۔

صَــلُّـوْا عَـلَـی الْـحَبِيْـب! صَـلَّـی اللهُ تَعالٰـی عَلـٰی مُـحَـمَّد

سوال: جيڪڏهن ڪنهن سان عشق ٿي وڃي، بدنگاهي وغيره گناهن جو سلسلو به هجي ۽ شاديءَ جي ترڪيب به ممڪن نه هجي ته ان کان ڪيئن ڇوٽڪارو حاصل ڪري؟

جواب: واقعي اهو معاملو وڏو صبر آزما آهي، ان دؤران جيڪي به گناهه ٿيا هجن، انهن کان سچي دل سان توبهه ڪري ان عشق سراپا فِسق کان ڇوٽڪاري جي لاءِ الله عَزَّوَجَلَّ جي بارگاهه ۾ ٻاڏائي ڪري دعا گهري، ان کي ڏسڻ کان بچي، بلڪ جيڪڏهن ان جي تصوير، تحفو يا ڪا ٻي نشاني پاڻ وٽ هجي ته ان کي به نه ڏسي ۽ فورًا اهي شيون پاڻ کان الڳ ڪري ڇڏي، ان جي فون نه ٻڌي۔ ان جو عشقيه خط نه پڙهي ايتري تائين جو ان جي

تصّور کان به هر ممڪن صورت ۾ پيڇو ڇڏائي۔ پنهنجو پاڻ کي ديني ڪمن ۾ هڪدم مشغول ڪري ڇڏي۔ الله عَزَّوَجَلَّ ۽ ان جي پياري حبيب صَلَّى اللهُ تَعَالٰى عَلَيْهِ وَاٰلِهٖ وَسَلَّم جي محبت پنهنجي دل ۾ وڌائي ۽ بارگاههِ رسالت ۾ فرياد پيش ڪري:

محبَّت غیر کی دل سے نکالو یا رسولَ اللہ

مجھے اپنا ہی دیوانہ بنا لو یا رسولَ اللہ

عشق بازيءَ کان جان ڇڏائڻ جو روحاني علاج

سوال: عشق بازي کان جان ڇڏائڻ جي لاءِ ڪو وظيفو به ٻڌائي ڇڏيو؟

جواب: گذريل سوال جي جواب جي شروع ۾ جيڪي مدني گل پيش ڪيا، ان سان گڏوگڏ بيشڪ قرآني آيتن تي مشتمل هي ”عمل“ به ڪيو: بِسْمِ الله ِالرَّحْمٰنِ الرَّحِيْمِ. لَا اِلٰهَ إِلَّا أَنْتَ سُبْحٰنَكَ اِنِّي كُنْتُ مِنَ الظّٰلِمِيْنَ. اَللهُ نُورُ السَّمٰوٰتِ وَالْاَرْضِ. لاَ تَأْخُذُهُ سِنَةٌ وَلاَ نَوْمٌ ۔ باوضو ٽي ڀيرا پڙهي (اول آخر هڪ ڀيرو درود شريف) پاڻي تي دم ڪري پي ڇڏي، اهو عمل چاليهه ڏينهن تائين ڪري۔
